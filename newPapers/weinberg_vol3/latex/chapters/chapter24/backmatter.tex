\chapterbackmatter{Problems}

\begin{enumerate}
  \item Show that the most general symmetry algebra allowed under the
  assumptions of the Coleman--Mandula theorem in the case where all
  particles are massless consists of internal symmetry generators plus
  either the Poincar\'e algebra or the conformal algebra
  \eqref{eq:24.B.34}, \eqref{eq:24.B.35}.

  \item Show that the Gervais--Sakita action \eqref{eq:24.2.5} is invariant
  under the worldsheet supersymmetry transformation \eqref{eq:24.2.7}.

  \item Calculate the change in the Wess--Zumino Lagrangian density
  \eqref{eq:24.2.9} under the spacetime supersymmetry transformation
  \eqref{eq:24.2.8}.
\end{enumerate}
