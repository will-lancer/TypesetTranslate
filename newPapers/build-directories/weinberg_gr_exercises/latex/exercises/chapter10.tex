\begin{exercises}

\begin{exercise}[Counting radiative degrees of freedom]{Tong, Cambridge Part III Sheet 4}
For a vacuum plane wave
\(\bar h_{\mu\nu}=\Re(A_{\mu\nu}e^{\ii k\cdot x})\), impose harmonic gauge and
use residual gauge freedom to reach transverse-traceless gauge.  Show that
only two independent polarizations remain.
\end{exercise}

\begin{exercise}[A ring of test masses]{Tong, Cambridge Part III Sheet 4}
For a wave traveling in the \(+z\)-direction, write the \(+\) and \(\times\)
polarization tensors.  Use geodesic deviation to determine the motion of a
ring of freely falling particles in the \(xy\)-plane.
\end{exercise}

\begin{exercise}[Energy flux of a plane wave]{McGreevy, Problem Set 9}
For a TT wave, show that the averaged energy flux is
\[
\langle t^{0z}_{\rm GW}\rangle
=\frac1{32\pi G}
\left\langle\dot h^{\rm TT}_{ij}\dot h_{\rm TT}^{ij}\right\rangle.
\]
Evaluate it for \(h_+=A\cos[\omega(t-z)]\), \(h_\times=0\).
\end{exercise}

\begin{exercise}[Retarded quadrupole field]{Cambridge Part III, 2022 exam}
Starting from the retarded solution of the linearized field equation, use
\(\partial_\mu T^{\mu\nu}=0\) to show in the slow-source, far-zone limit that
\[
\bar h_{ij}(t,\bm x)=\frac{2G}{r}\ddot I_{ij}(t-r).
\]
State the additional TT projection needed for an observable wave.
\end{exercise}

\begin{exercise}[Circular binary luminosity]{Cambridge Part III, 2012 exam}
Two masses \(m_1,m_2\) move on a circular Newtonian orbit of separation \(a\).
Compute the trace-free quadrupole and show that the average radiated power is
\[
P=\frac{32}{5}\frac{G^4\mu^2M^3}{a^5},
\quad M=m_1+m_2,\quad \mu=\frac{m_1m_2}{M}.
\]
\end{exercise}

\begin{exercise}[Chirp mass]{MIT 8.962, Problem Set 13}
Use energy balance for the circular binary to derive \(\dot f\), where \(f\)
is the observed gravitational-wave frequency.  Show that the combination of
masses measured at leading order is
\(\mathcal M=\mu^{3/5}M^{2/5}\).
\end{exercise}

\begin{exercise}[Interferometer antenna pattern]{Tong, Cambridge Part III Sheet 4}
An equal-arm Michelson detector has orthogonal unit arm vectors
\(\hat{\bm u},\hat{\bm v}\).  Derive
\[
\frac{\delta L}{L}
=\frac12D^{ij}h^{\rm TT}_{ij},\qquad
D^{ij}=u^iu^j-v^iv^j.
\]
Find the response to a \(+\)-polarized wave arriving normal to the detector
plane with axes aligned to the arms.
\end{exercise}

\begin{exercise}[Velocity and displacement memory]{Cambridge Part III, 2024 exam}
A weak plane-wave pulse has profile \(A(u)\), \(u=t-z\), and produces
\(\ddot x=\epsilon y_0A'(u)\),
\(\ddot y=\epsilon x_0A'(u)\) to first order for initially resting masses.
Integrate through the pulse and identify the conditions for velocity memory
and pure displacement memory.
\end{exercise}

\begin{exercise}[Rotating rod]{Cambridge Part III, 2022 exam}
A thin rod of mass \(m\) and length \(2a\) rotates nonrelativistically in the
\(xy\)-plane with angular frequency \(\omega\).  Determine the wave frequency,
the polarization on the \(z\)-axis, and the scaling of its luminosity with
\(m,a,\omega\).
\end{exercise}

\begin{exercise}[Why there is no dipole radiation]{McGreevy, Problem Set 9}
Use conservation of total mass-energy and momentum to show that the monopole
and mass-dipole moments of an isolated slow source cannot radiate.  Explain
why the leading gravitational radiation is quadrupolar, in contrast with
electromagnetism.
\end{exercise}

\begin{exercise}[Quadrupole field of a compact source]{Cambridge Part III, 2007 exam, Question 3}
For a compact conserved source in linearized gravity, prove
\[
\int T^{ij}\dd^3x
=\frac12\frac{\dd^2}{\dd t^2}
\int T^{00}x^ix^j\dd^3x.
\]
Use the retarded solution to obtain the far-zone transverse-traceless field
in terms of the trace-free quadrupole.  Estimate its magnitude for a
self-gravitating source of mass \(M\), size \(R\), and compactness
\(\epsilon=GM/R\ll1\).
\end{exercise}

\begin{exercise}[Plane waves and residual gauge freedom]{Cambridge Part III, 2025 exam, Question 3}
Linearize Einstein's equation about
\(g_{\mu\nu}=\eta_{\mu\nu}+h_{\mu\nu}\) in wave gauge and derive
\[
\Box\bar h_{\mu\nu}=-16\pi G T_{\mu\nu},
\qquad
\partial^\mu\bar h_{\mu\nu}=0.
\]
Show that a residual transformation generated by \(\xi^\mu\) preserves the
gauge when \(\Box\xi^\mu=0\).  For a vacuum plane-wave packet
\(\bar h_{\mu\nu}=H_{\mu\nu}(k\cdot x)\), prove \(k^2=0\) and reduce the
perturbation to its two transverse-traceless polarizations.
\end{exercise}

\end{exercises}

\begin{solutions}

\begin{solution}
The symmetric tensor \(A_{\mu\nu}\) has ten components.  Harmonic gauge
\(k^\mu A_{\mu\nu}=0\) removes four combinations; residual transformations
with \(\Box\xi^\mu=0\) remove four more.  In a frame with
\(k^\mu=\omega(1,0,0,1)\), TT gauge leaves
\[
A_{xx}=-A_{yy}=A_+,\qquad A_{xy}=A_{yx}=A_\times,
\]
and no other components.
\end{solution}

\begin{solution}
\[
e^+_{ij}=\hat x_i\hat x_j-\hat y_i\hat y_j,\qquad
e^\times_{ij}=\hat x_i\hat y_j+\hat y_i\hat x_j.
\]
Geodesic deviation gives
\(\ddot s^i=\tfrac12\ddot h^{\rm TT}_{ij}s^j\).
The \(+\) mode alternately stretches the \(x\) and \(y\) axes; the
\(\times\) mode produces the same pattern rotated by \(45^\circ\).
\end{solution}

\begin{solution}
The Isaacson average of the quadratic Einstein tensor yields the stated
expression.  Here
\(\dot h_{ij}\dot h^{ij}=2A^2\omega^2\sin^2[\omega(t-z)]\), so
\[
\langle t^{0z}_{\rm GW}\rangle
=\frac{A^2\omega^2}{32\pi G}.
\]
\end{solution}

\begin{solution}
In the far zone,
\(\bar h_{ij}=4G r^{-1}\int T_{ij}(t-r,\bm x')\dd^3x'\).
Twice integrating the conservation law by parts gives
\(\int T_{ij}\dd^3x=\tfrac12\ddot I_{ij}\), hence the result.  The measurable
field is
\[
h^{\rm TT}_{ij}=\frac{2G}{r}
\Lambda_{ij,kl}(\hat{\bm n})\ddot I_{kl}(t-r),
\]
where \(\Lambda\) removes longitudinal and trace pieces.
\end{solution}

\begin{solution}
In center-of-mass coordinates the relative vector is
\(\bm r=a(\cos\Omega t,\sin\Omega t,0)\), and
\(Q_{ij}=\mu(r_ir_j-\delta_{ij}a^2/3)\).  Insert its third derivatives into
\[
P=\frac{G}{5}\left\langle\dddot Q_{ij}\dddot Q_{ij}\right\rangle
\]
and use \(\Omega^2=GM/a^3\).  This gives the stated
\(32G^4\mu^2M^3/(5a^5)\).
\end{solution}

\begin{solution}
The Newtonian orbital energy is
\(E=-G\mu M/(2a)\), while
\(\Omega^2=GM/a^3\) and \(f=\Omega/\pi\).  With \(\dot E=-P\),
\[
\dot f=\frac{96}{5}\pi^{8/3}
(G\mathcal M)^{5/3}f^{11/3},
\qquad
\mathcal M^{5/3}=\mu M^{2/3}.
\]
Thus \(\mathcal M=\mu^{3/5}M^{2/5}\).
\end{solution}

\begin{solution}
The proper arm change is
\(\delta L_{\hat n}/L=\tfrac12n^in^jh_{ij}^{\rm TT}\).  Subtracting the two
arms gives the detector tensor formula.  For a normal, aligned \(+\) wave,
\(u^iu^jh_{ij}=h_+\) and \(v^iv^jh_{ij}=-h_+\), so
\(\delta L/L=h_+\).
\end{solution}

\begin{solution}
Integrating once gives
\(\Delta\dot x=\epsilon y_0\int A'(u)\dd u\) and similarly for \(y\).
Thus a nonzero endpoint difference of \(A\) produces velocity memory.
If \(A\) returns to its initial value, the final velocity vanishes, but a
nonzero first moment of the profile leaves
\(\Delta x\propto-\epsilon y_0\int A(u)\dd u\), a displacement memory.
\end{solution}

\begin{solution}
The quadrupole depends on \(\cos2\omega t\) and \(\sin2\omega t\), so the wave
frequency is \(2\omega\).  On the rotation axis the two polarizations are in
quadrature, producing circular polarization.  Dimensional evaluation of the
quadrupole formula gives
\[
P\propto Gm^2a^4\omega^6,
\]
with the numerical coefficient fixed by the rod's moment
\(\int_{-a}^a\lambda \ell^2\dd\ell=ma^2/3\).
\end{solution}

\begin{solution}
The monopole \(M=\int T^{00}\dd^3x\) is constant.  The dipole
\(D^i=\int T^{00}x^i\dd^3x\) satisfies
\(\dot D^i=P^i\), and total momentum is also constant, so
\(\ddot D^i=0\).  Neither can have the time derivatives needed for outgoing
radiation.  The next moment is the quadrupole.  Electromagnetism permits
electric dipole radiation because positive and negative charge can move
relative to one another while total charge remains fixed; gravitational
``charge'' is mass-energy and its dipole is the center of mass.
\end{solution}

\begin{solution}
Twice differentiate the moment, use
\(\partial_\mu T^{\mu\nu}=0\), and integrate by parts twice; compact support
removes all surface terms and gives the stated identity.  The retarded
solution then yields
\[
h^{\rm TT}_{ij}(t,\bm x)
=\frac{2G}{r}\Lambda_{ij,kl}(\hat{\bm n})
\ddot I_{kl}(t-r),
\]
where \(\Lambda\) is the TT projector.  Virial motion has
\(v^2\sim GM/R=\epsilon\), \(I\sim MR^2\), and
\(\ddot I\sim Mv^2\sim M\epsilon\).  Hence
\[
|h^{\rm TT}|\sim\frac{GM}{r}\epsilon.
\]
\end{solution}

\begin{solution}
With
\(\bar h_{\mu\nu}=h_{\mu\nu}-\tfrac12\eta_{\mu\nu}h\), the linearized
Einstein tensor in wave gauge is
\(-\tfrac12\Box\bar h_{\mu\nu}\), giving the stated equation.  Under
\(h_{\mu\nu}\mapsto h_{\mu\nu}+2\partial_{(\mu}\xi_{\nu)}\),
\[
\partial^\mu\bar h_{\mu\nu}
\mapsto\partial^\mu\bar h_{\mu\nu}+\Box\xi_\nu,
\]
so \(\Box\xi^\mu=0\) is the residual freedom.

The vacuum wave equation gives \(k^2H_{\mu\nu}''=0\), hence \(k^2=0\) for a
nontrivial packet, while the gauge condition makes the amplitude transverse.
Choose \(k^\mu=\omega(1,0,0,1)\).  Residual gauge transformations remove
time and longitudinal components and the remaining trace, leaving
\[
h^{\rm TT}_{ij}
=\begin{pmatrix}
h_+&h_\times&0\\
h_\times&-h_+&0\\
0&0&0
\end{pmatrix}.
\]
\end{solution}

\end{solutions}
