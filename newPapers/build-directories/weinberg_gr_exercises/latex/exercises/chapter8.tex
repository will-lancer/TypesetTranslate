\begin{exercises}

\begin{exercise}[Deriving the Schwarzschild exterior]{Cambridge Part III, 2010 exam}
Insert
\[
\dd s^2=-A(r)\dd t^2+B(r)\dd r^2
+r^2(\dd\theta^2+\sin^2\theta\,\dd\phi^2)
\]
into the vacuum Einstein equation.  Solve for \(A,B\) using asymptotic
flatness and identify the integration constant with the mass.
\end{exercise}

\begin{exercise}[Circular timelike orbits]{Tong, Cambridge Part III Sheet 2}
For equatorial Schwarzschild geodesics, derive the radial effective potential.
Find the specific energy and angular momentum of a circular orbit and
determine the radius of the innermost stable circular orbit.
\end{exercise}

\begin{exercise}[Deflection of light]{Cambridge Part III, 2013 exam}
For a null Schwarzschild geodesic with impact parameter \(b\gg GM\), derive
the orbit equation and calculate the leading deflection angle.  Identify the
part that would be missed by perturbing only \(g_{00}\).
\end{exercise}

\begin{exercise}[Advance of perihelion]{MIT 8.962, Problem Set 5}
For a nearly Keplerian bound orbit with semi-major axis \(a\) and eccentricity
\(e\), use the Schwarzschild orbit equation to find the relativistic advance
of perihelion per revolution.
\end{exercise}

\begin{exercise}[Radar echo delay]{Cambridge Part III, 2016 exam}
A light ray passes a spherical mass with closest approach \(b\) and endpoints
at radii \(r_1,r_2\gg b\).  Compute the leading excess travel time over the
flat-spacetime result.
\end{exercise}

\begin{exercise}[Static-clock redshift]{MIT 8.962, Problem Set 4}
Two clocks remain at Schwarzschild radii \(r_1\) and \(r_2\).  Find their rate
ratio.  Expand for \(r_i\gg GM\) and estimate the sign of the effect for the
higher clock.
\end{exercise}

\begin{exercise}[Crossing the horizon]{McGreevy, Problem Set 7}
Introduce ingoing Eddington--Finkelstein time
\(v=t+r_\ast\), with
\(\dd r_\ast/\dd r=(1-2GM/r)^{-1}\).  Rewrite the Schwarzschild metric and
show directly that it is regular at \(r=2GM\).
\end{exercise}

\begin{exercise}[Photon sphere and capture]{Cambridge Part III, 2023 exam}
Find the circular null orbit in Schwarzschild spacetime and determine its
stability.  Derive the critical impact parameter separating capture from
scattering.
\end{exercise}

\begin{exercise}[Reissner--Nordstr\"om horizons]{Tong, Cambridge Part III Sheet 2}
For
\[
f(r)=1-\frac{2GM}{r}+\frac{GQ^2}{4\pi r^2},
\qquad
\dd s^2=-f\dd t^2+f^{-1}\dd r^2+r^2\dd\Omega_2^2,
\]
locate the horizons and distinguish the subextremal, extremal, and
overcharged cases.
\end{exercise}

\begin{exercise}[Parameterized post-Newtonian tests]{Cambridge Part III, 2019 exam}
Take the weak static isotropic metric
\[
\dd s^2=-(1-2U+2\beta U^2)\dd t^2
+(1+2\gamma U)\dd\bm x^2,\qquad U=GM/r.
\]
State how light deflection, Shapiro delay, and perihelion advance depend on
\(\beta,\gamma\), and recover their general-relativistic values.
\end{exercise}

\begin{exercise}[Linearized Schwarzschild light orbit]{Cambridge Part III, 2005 exam, Question 3}
Obtain the first-order Schwarzschild metric from the linearized Einstein
equations for a point mass.  For an equatorial null geodesic define
\(u=2GM/r\) and show that
\[
\left(\frac{\dd u}{\dd\phi}\right)^2
=C-u^2(1-u).
\]
Determine whether a circular null orbit exists and discuss its stability.
Why must the last numerical result be interpreted cautiously within a
strictly linearized calculation?
\end{exercise}

\begin{solutions}

\begin{solution}
The combination \(G^t{}_t-G^r{}_r=0\) gives \(AB=\) constant, set to one by
rescaling \(t\).  The remaining equation integrates to
\(r(1-B^{-1})=2GM\).  Therefore
\[
A(r)=1-\frac{2GM}{r},\qquad
B(r)=\left(1-\frac{2GM}{r}\right)^{-1}.
\]
The Newtonian limit \(g_{00}\simeq-(1+2\Phi)\), \(\Phi=-GM/r\), fixes the
constant.
\end{solution}

\begin{solution}
With \(E=f\dot t\), \(L=r^2\dot\phi\), and \(f=1-2GM/r\),
\[
\dot r^2+f\left(1+\frac{L^2}{r^2}\right)=E^2.
\]
The circularity conditions give
\[
L^2=\frac{GMr^2}{r-3GM},\qquad
E=\frac{1-2GM/r}{\sqrt{1-3GM/r}}.
\]
The second derivative of the potential changes sign at \(r=6GM\), the ISCO.
\end{solution}

\begin{solution}
For \(u=1/r\),
\[
\frac{\dd^2u}{\dd\phi^2}+u=3GMu^2.
\]
Insert the zeroth-order ray \(u_0=\sin\phi/b\) and solve for the first
correction.  Comparing the two asymptotes yields
\[
\Delta\phi=\frac{4GM}{b}.
\]
Half comes from the Newtonian potential in \(g_{00}\); the other half comes
from spatial curvature.
\end{solution}

\begin{solution}
For a timelike orbit the equation is
\[
u''+u=\frac{GM}{L^2}+3GMu^2.
\]
Using \(u_0=[1+e\cos\phi]/[a(1-e^2)]\), the resonant part of the perturbation
shifts the radial frequency.  The perihelion advance is
\[
\Delta\phi=\frac{6\pi GM}{a(1-e^2)}.
\]
\end{solution}

\begin{solution}
Expanding the null radial integral to first order in \(GM\) gives the one-way
delay
\[
\Delta t
\simeq2GM\ln\left(\frac{4r_1r_2}{b^2}\right)
\]
when both endpoints are far from closest approach.  A radar round trip
doubles this expression.
\end{solution}

\begin{solution}
A static clock measures
\(\dd\tau=\sqrt{1-2GM/r}\,\dd t\), hence
\[
\frac{\dd\tau_2}{\dd\tau_1}
=\sqrt{\frac{1-2GM/r_2}{1-2GM/r_1}}
\simeq1+GM\left(\frac1{r_1}-\frac1{r_2}\right).
\]
For \(r_2>r_1\), the higher clock runs faster.
\end{solution}

\begin{solution}
Since \(v=t+r_\ast\), the metric becomes
\[
\dd s^2=-f(r)\dd v^2+2\dd v\,\dd r+r^2\dd\Omega_2^2.
\]
Every coefficient and the determinant are finite and nondegenerate at
\(f=0\).  The divergence of Schwarzschild \(t\) there is therefore a
coordinate artifact.
\end{solution}

\begin{solution}
For null geodesics
\(\dot r^2+L^2f/r^2=E^2\).  Extremizing \(f/r^2\) gives
\(r_{\rm ph}=3GM\); the extremum is a maximum, so the orbit is unstable.
At that radius
\[
b_{\rm crit}^2=\frac{L^2}{E^2}
=\frac{r_{\rm ph}^2}{f(r_{\rm ph})}=27G^2M^2,
\]
so \(b_{\rm crit}=3\sqrt3\,GM\).
\end{solution}

\begin{solution}
The roots are
\[
r_\pm=GM\pm\sqrt{G^2M^2-\frac{GQ^2}{4\pi}}.
\]
Two distinct horizons occur when the discriminant is positive, one degenerate
horizon when it vanishes, and no horizon when it is negative.  In the last
case the curvature singularity is naked within the classical solution.
\end{solution}

\begin{solution}
Relative to the Einstein predictions,
\[
\Delta\theta=\frac{2(1+\gamma)GM}{b},\qquad
\Delta t=(1+\gamma)GM\ln\!\left(\frac{4r_1r_2}{b^2}\right),
\]
and
\[
\Delta\varpi
=\frac{6\pi GM}{a(1-e^2)}
\left(\frac{2+2\gamma-\beta}{3}\right).
\]
General relativity has \(\beta=\gamma=1\), making the multiplier in the last
line unity.
\end{solution}

\begin{solution}
In harmonic or isotropic gauge, solving the linearized field equations gives
\(g_{00}=-(1-2GM/r)\) and the corresponding spatial perturbation.  Conserved
energy and angular momentum reduce the null equation to the stated form.
For a circular orbit both the right side and its derivative must vanish.
The nontrivial stationary point satisfies
\[
-2u+3u^2=0,\qquad u=\frac23,
\]
so \(r=3GM\).  Since \(u^2(1-u)\) has a maximum there, the circular orbit is
unstable.  At \(r=3GM\), however, \(GM/r=1/3\) is not parametrically small;
the exact Schwarzschild geometry confirms the radius, but linearized theory
alone is not quantitatively controlled that deep in the field.
\end{solution}
