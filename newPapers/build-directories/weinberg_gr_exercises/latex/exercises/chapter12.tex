\begin{exercises}

\begin{exercise}[Stress tensor by metric variation]{Cambridge Part III, 2015 exam}
For a matter action \(I_{\rm m}\), define
\[
T_{\mu\nu}=-\frac{2}{\sqrt{-g}}
\frac{\delta I_{\rm m}}{\delta g^{\mu\nu}}.
\]
Apply the definition to a minimally coupled real scalar and to Maxwell theory.
\end{exercise}

\begin{exercise}[Diffeomorphism Noether identity]{Cambridge Part III, 2022 exam}
Let a generally covariant matter action depend on
\(g_{\mu\nu}\) and fields \(\psi\).  Vary it under an infinitesimal
diffeomorphism generated by \(\xi^\mu\) and show that the matter equations
imply \(\nabla_\mu T^{\mu\nu}=0\).
\end{exercise}

\begin{exercise}[Variation of the Einstein--Hilbert action]{Tong, Cambridge Part III Sheet 3}
Derive
\[
\delta(\sqrt{-g}R)
=\sqrt{-g}\,G_{\mu\nu}\delta g^{\mu\nu}
+\sqrt{-g}\,\nabla_\mu V^\mu.
\]
Give \(V^\mu\) in terms of \(\delta\Gamma^\rho{}_{\mu\nu}\).
\end{exercise}

\begin{exercise}[The boundary term]{MIT 8.962, Problem Set 9}
Explain why the Einstein--Hilbert action alone does not have a well-posed
Dirichlet variational principle on a manifold with boundary.  State the
Gibbons--Hawking--York term and the boundary data it fixes.
\end{exercise}

\begin{exercise}[Cosmological constant as an action term]{Cambridge Part III, 2011 exam}
Vary
\[
I_\Lambda=-\frac{\Lambda}{8\pi G}\int\dd^4x\sqrt{-g}
\]
and show that it contributes \(\Lambda g_{\mu\nu}\) to the field equation.
Compute its effective stress tensor and equation-of-state parameter.
\end{exercise}

\begin{exercise}[Palatini variation]{Cambridge Part III, 2013 exam}
Treat \(g_{\mu\nu}\) and a torsion-free connection
\(\Gamma^\rho{}_{\mu\nu}\) as independent in
\[
I=\frac1{16\pi G}\int\dd^4x\sqrt{-g}\,
g^{\mu\nu}R_{\mu\nu}(\Gamma).
\]
Show that the connection equation makes \(\Gamma\) Levi-Civita.
\end{exercise}

\begin{exercise}[Local Lorentz invariance]{McGreevy, Problem Set 6}
Write \(g_{\mu\nu}=e^a{}_\mu e^b{}_\nu\eta_{ab}\).  Show that
\(e^a{}_\mu\mapsto\Lambda^a{}_b(x)e^b{}_\mu\) leaves the metric invariant and
derive the inhomogeneous transformation of the spin connection.
\end{exercise}

\begin{exercise}[Tetrad Einstein equation]{Tong, Cambridge Part III Sheet 3}
Vary the gravitational action with respect to \(e^a{}_\mu\) rather than
\(g_{\mu\nu}\).  Show how the resulting equation is related to
\(G_{\mu\nu}=8\pi GT_{\mu\nu}\), and explain why its antisymmetric part
vanishes for locally Lorentz-invariant matter.
\end{exercise}

\begin{exercise}[Higher-curvature scalar mode]{Cambridge Part III, 2024 exam}
For \(f(R)=R+\alpha R^2\), take the trace of the metric field equation and
linearize around Minkowski space.  Find the mass of the additional scalar
curvature mode.
\end{exercise}

\begin{exercise}[Boundary terms and equivalent Lagrangians]{Cambridge Part III, 2014 exam}
Show that adding \(\nabla_\mu W^\mu\) to a Lagrangian density does not change
the bulk Euler--Lagrange equations for compactly supported variations.
Explain why such a term can nevertheless change conserved charges or the
variational problem at a boundary.
\end{exercise}

\begin{exercise}[Velocity-potential action for a perfect fluid]{Cambridge Part III, 2002 exam, Question 3}
Use the modern isentropic fluid action
\[
I=\int \sqrt{-g}\,P(X)\,\dd^4x,
\qquad
X=\sqrt{-g^{\mu\nu}\partial_\mu\chi\,\partial_\nu\chi}.
\]
Define \(n=P'(X)\),
\(u_\mu=-\partial_\mu\chi/X\), and
\(\rho=nX-P\).  Vary \(\chi\) and \(g^{\mu\nu}\) to derive particle-number
conservation and
\[
T_{\mu\nu}=(\rho+P)u_\mu u_\nu+Pg_{\mu\nu}.
\]
Then obtain the relativistic Euler equation from
\(\nabla_\mu T^{\mu\nu}=0\).
\end{exercise}

\end{exercises}

\begin{solutions}

\begin{solution}
For the scalar,
\[
T_{\mu\nu}
=\nabla_\mu\phi\nabla_\nu\phi
-g_{\mu\nu}\left[\frac12(\nabla\phi)^2+V(\phi)\right].
\]
For Maxwell theory,
\[
T_{\mu\nu}
=F_{\mu\rho}F_\nu{}^\rho
-\frac14g_{\mu\nu}F_{\rho\sigma}F^{\rho\sigma}.
\]
Both follow from
\(\delta\sqrt{-g}=-\tfrac12\sqrt{-g}g_{\mu\nu}\delta g^{\mu\nu}\).
\end{solution}

\begin{solution}
On the matter equations,
\[
0=\delta_\xi I_{\rm m}
=-\frac12\int\sqrt{-g}\,T_{\mu\nu}\Lie_\xi g^{\mu\nu}\dd^4x.
\]
Since \(\Lie_\xi g_{\mu\nu}=2\nabla_{(\mu}\xi_{\nu)}\), integration by parts
gives
\(\int\sqrt{-g}(\nabla_\mu T^{\mu\nu})\xi_\nu=0\).  Arbitrary compactly
supported \(\xi_\nu\) implies conservation.
\end{solution}

\begin{solution}
Use
\(\delta R_{\mu\nu}=\nabla_\rho\delta\Gamma^\rho{}_{\mu\nu}
-\nabla_\nu\delta\Gamma^\rho{}_{\mu\rho}\).  Then
\[
V^\rho=g^{\mu\nu}\delta\Gamma^\rho{}_{\mu\nu}
-g^{\rho\mu}\delta\Gamma^\nu{}_{\mu\nu}.
\]
Combining the Ricci variation with the determinant variation gives the stated
bulk term \(G_{\mu\nu}\delta g^{\mu\nu}\).
\end{solution}

\begin{solution}
The divergence in \(\delta I_{\rm EH}\) contains normal derivatives of
\(\delta g_{\mu\nu}\), which do not vanish when only the induced boundary
metric is fixed.  Adding
\[
I_{\rm GHY}=\frac{\epsilon}{8\pi G}
\int_{\partial M}\sqrt{|h|}\,K\,\dd^3x
\]
cancels those derivatives.  The resulting Dirichlet problem fixes the
induced metric \(h_{ij}\) on the boundary.
\end{solution}

\begin{solution}
The variation is
\[
\delta I_\Lambda
=\frac{\Lambda}{16\pi G}
\int\sqrt{-g}\,g_{\mu\nu}\delta g^{\mu\nu}\dd^4x.
\]
It therefore adds \(\Lambda g_{\mu\nu}\) to the geometric equation.  Moved to
the matter side,
\(T_{\mu\nu}^{(\Lambda)}=-\Lambda g_{\mu\nu}/(8\pi G)\), so
\(\rho_\Lambda=\Lambda/(8\pi G)\), \(p_\Lambda=-\rho_\Lambda\), and \(w=-1\).
\end{solution}

\begin{solution}
Connection variation and integration by parts give
\[
\nabla_\lambda(\sqrt{-g}\,g^{\mu\nu})
-\delta^{(\mu}_\lambda
\nabla_\rho(\sqrt{-g}\,g^{\nu)\rho})=0.
\]
Contracting indices removes the second term and yields
\(\nabla_\lambda(\sqrt{-g}g^{\mu\nu})=0\).  For a torsion-free connection in
dimension greater than two this is equivalent to
\(\nabla_\lambda g_{\mu\nu}=0\), uniquely selecting the Levi-Civita
connection.
\end{solution}

\begin{solution}
Metric invariance follows from
\(\Lambda^a{}_c\Lambda^b{}_d\eta_{ab}=\eta_{cd}\).  Requiring
\(D_\mu v^a\) to transform covariantly gives
\[
\omega'_\mu
=\Lambda\omega_\mu\Lambda^{-1}
-(\partial_\mu\Lambda)\Lambda^{-1}.
\]
The derivative term is the gauge-connection transformation for the local
Lorentz group.
\end{solution}

\begin{solution}
Since
\(\delta g_{\mu\nu}=2e_{a(\mu}\delta e^a{}_{\nu)}\), the tetrad variation
produces
\[
G^\mu{}_\nu e_a{}^\nu=8\pi G\,T^\mu{}_\nu e_a{}^\nu.
\]
Multiplying by the inverse tetrad recovers the metric equation.  An
antisymmetric tetrad variation is an infinitesimal local Lorentz rotation, so
local Lorentz invariance makes that part an identity and requires the
on-shell stress tensor to be symmetric.
\end{solution}

\begin{solution}
The trace equation is
\[
6\alpha\Box R-R=8\pi GT.
\]
In vacuum and to linear order,
\((\Box-m_s^2)R^{(1)}=0\) with
\[
m_s^2=\frac1{6\alpha}.
\]
A positive \(\alpha\) therefore gives a non-tachyonic scalar mode.
\end{solution}

\begin{solution}
The action changes by
\(\int_M\sqrt{-g}\nabla_\mu W^\mu
=\int_{\partial M}\sqrt{|h|}n_\mu W^\mu\).
For compactly supported variations this surface integral does not
contribute, so the bulk equations agree.  If the boundary is physical or at
infinity, however, its variation and value affect which data are fixed and
can shift the surface terms defining Hamiltonians and conserved charges.
\end{solution}

\begin{solution}
Since
\(\delta X=u^\mu\nabla_\mu\delta\chi\), variation of \(\chi\) followed by
integration by parts gives
\[
\nabla_\mu(nu^\mu)=0.
\]
For a metric variation,
\(\delta X=-\tfrac12X u_\mu u_\nu\delta g^{\mu\nu}\), while
\(\delta\sqrt{-g}=-\tfrac12\sqrt{-g}g_{\mu\nu}\delta g^{\mu\nu}\).
Using \(nX=\rho+P\) in
\(T_{\mu\nu}=-2(\sqrt{-g})^{-1}\delta I/\delta g^{\mu\nu}\) yields
\[
T_{\mu\nu}=(\rho+P)u_\mu u_\nu+Pg_{\mu\nu}.
\]
Projecting \(\nabla_\mu T^{\mu\nu}=0\) orthogonally to \(u^\nu\) gives
\[
(\rho+P)u^\mu\nabla_\mu u_\alpha
+(g_{\alpha}{}^\mu+u_\alpha u^\mu)\nabla_\mu P=0,
\]
the relativistic Euler equation.  The normalization \(u_\mu u^\mu=-1\)
follows directly from the definition of \(X\).
\end{solution}
