\begin{exercises}

\begin{exercise}[Geodesics from the particle action]{Cambridge Part III, 2024 exam}
Vary
\[
I[x]=-m\int\dd\lambda\,
\sqrt{-g_{\mu\nu}\dot x^\mu\dot x^\nu}
\]
with fixed endpoints.  Derive the geodesic equation in proper-time
parametrization and explain what changes for a general parameter.
\end{exercise}

\begin{exercise}[Charged motion in curved spacetime]{McGreevy, Problem Set 5}
Add \(q\int A_\mu\dd x^\mu\) to the particle action.  Show that
\[
m\Dworldline{u^\mu}=qF^\mu{}_\nu u^\nu
\]
and prove that \(u_\mu u^\mu\) remains constant.
\end{exercise}

\begin{exercise}[Charge conservation]{Tong, Cambridge Part III Sheet 3}
Starting from \(\nabla_\mu F^{\mu\nu}=J^\nu\), derive
\(\nabla_\nu J^\nu=0\).  Show that the charge
\[
Q_\Sigma=-\int_\Sigma n_\mu J^\mu\,\dd\Sigma
\]
is independent of the spacelike hypersurface \(\Sigma\), provided no current
flows through the timelike boundary.
\end{exercise}

\begin{exercise}[Dust follows geodesics]{Cambridge Part III, 2015 exam}
For pressureless matter,
\(T^{\mu\nu}=\rho u^\mu u^\nu\).  Decompose
\(\nabla_\mu T^{\mu\nu}=0\) parallel and orthogonal to \(u^\nu\), obtaining
both the continuity equation and the geodesic equation.
\end{exercise}

\begin{exercise}[Relativistic Euler equation]{McGreevy, Problem Set 6}
For a perfect fluid, project \(\nabla_\mu T^{\mu\nu}=0\) along and orthogonal
to \(u^\mu\).  Derive the energy equation and the relativistic Euler equation.
\end{exercise}

\begin{exercise}[Hydrostatic balance in a static field]{Cambridge Part III, 2016 exam}
In a static metric
\(\dd s^2=-e^{2\Phi(\bm x)}\dd t^2+h_{ij}\dd x^i\dd x^j\), a perfect fluid is
at rest.  Show that
\[
D_i p=-(\rho+p)D_i\Phi.
\]
Recover the Newtonian result when \(p\ll\rho\) and \(|\Phi|\ll1\).
\end{exercise}

\begin{exercise}[Conformal invariance of null rays]{MIT 8.962, Problem Set 5}
Let \(\widetilde g_{\mu\nu}=\Omega^2g_{\mu\nu}\).  Show that every null
geodesic of \(g_{\mu\nu}\) is an unparameterized null geodesic of
\(\widetilde g_{\mu\nu}\).  Find the relation between affine parameters.
\end{exercise}

\begin{exercise}[Geometric-optics amplitude]{Cambridge Part III, 2018 exam}
Use the ansatz
\(A_\mu=\Re[(a_\mu+\order(\epsilon))e^{\ii S/\epsilon}]\) in vacuum Maxwell
theory.  Show that \(k_\mu=\partial_\mu S\) is null and geodesic and that the
leading intensity obeys a flux-conservation law.
\end{exercise}

\begin{exercise}[Radiation fluid]{Tong Cosmology, Cambridge Part III Sheet 3}
For an isotropic gas of massless particles, derive \(p=\rho/3\) from kinetic
theory.  In a homogeneous expanding volume with scale factor \(a\), use local
energy conservation to show \(\rho\propto a^{-4}\).
\end{exercise}

\begin{exercise}[Killing energy of matter]{Cambridge Part III, 2024 exam}
Let \(K^\mu\) be a Killing field and \(T^{\mu\nu}\) a symmetric conserved
stress tensor.  Prove that \(J^\mu=T^\mu{}_\nu K^\nu\) is conserved.  State
the conditions under which its hypersurface integral is a conserved energy.
\end{exercise}

\begin{solutions}

\begin{solution}
The Euler--Lagrange equation becomes, after choosing
\(-g_{\mu\nu}\dot x^\mu\dot x^\nu=1\),
\[
\frac{\dd^2x^\mu}{\dd\tau^2}
+\Gamma^\mu{}_{\rho\sigma}
\frac{\dd x^\rho}{\dd\tau}\frac{\dd x^\sigma}{\dd\tau}=0.
\]
For a nonaffine parameter \(\lambda\), the right side is
\(f(\lambda)\dot x^\mu\); reparametrizing affinely removes it.
\end{solution}

\begin{solution}
Varying the interaction gives
\(q(\partial_\mu A_\nu-\partial_\nu A_\mu)u^\nu\delta x^\mu\).
Combining it with the free variation yields the stated Lorentz-force law.
Metric compatibility and antisymmetry of \(F_{\mu\nu}\) give
\[
\frac{\dd}{\dd\tau}(u_\mu u^\mu)
=2u_\mu\Dworldline{u^\mu}
=\frac{2q}{m}u^\mu F_{\mu\nu}u^\nu=0.
\]
\end{solution}

\begin{solution}
Taking a divergence and using antisymmetry,
\[
\nabla_\nu J^\nu
=\nabla_\nu\nabla_\mu F^{\mu\nu}
=\frac12[\nabla_\nu,\nabla_\mu]F^{\mu\nu}=0,
\]
because the curvature contractions cancel.  Integrating over the spacetime
slab between two slices and applying Stokes' theorem gives
\(Q_{\Sigma_2}-Q_{\Sigma_1}\) equal to minus the side-boundary flux, hence
zero under the stated condition.
\end{solution}

\begin{solution}
Expansion gives
\[
0=u^\nu\nabla_\mu(\rho u^\mu)+\rho u^\mu\nabla_\mu u^\nu.
\]
Contracting with \(u_\nu\) and using \(u_\nu u^\nu=-1\) yields
\(\nabla_\mu(\rho u^\mu)=0\).  Substitution then gives
\(u^\mu\nabla_\mu u^\nu=0\).
\end{solution}

\begin{solution}
With
\(P^{\mu\nu}=g^{\mu\nu}+u^\mu u^\nu\), the parallel projection is
\[
u^\mu\nabla_\mu\rho+(\rho+p)\nabla_\mu u^\mu=0.
\]
The orthogonal projection is
\[
(\rho+p)u^\mu\nabla_\mu u^\alpha
+P^{\alpha\mu}\nabla_\mu p=0.
\]
The pressure gradient accelerates only orthogonally to the four-velocity.
\end{solution}

\begin{solution}
The static four-velocity is \(u^\mu=e^{-\Phi}\delta^\mu_0\).  Its acceleration
is \(a_i=D_i\Phi\).  The spatial Euler equation therefore gives
\(D_ip=-(\rho+p)D_i\Phi\).  Neglecting \(p/\rho\) and weak-field corrections
leaves \(\bm\nabla p=-\rho\bm\nabla\Phi\).
\end{solution}

\begin{solution}
The transformed connection differs by
\[
C^\mu{}_{\rho\sigma}
=\delta^\mu_\rho\partial_\sigma\ln\Omega
+\delta^\mu_\sigma\partial_\rho\ln\Omega
-g_{\rho\sigma}\partial^\mu\ln\Omega.
\]
For a null tangent \(k^\mu\), the last term vanishes and the remaining terms
are proportional to \(k^\mu\), so only the parametrization changes.  If
\(\lambda\) is affine for \(g\), then an affine parameter for
\(\widetilde g\) satisfies
\(\dd\widetilde\lambda/\dd\lambda=\Omega^2\).
\end{solution}

\begin{solution}
The order \(\epsilon^{-2}\) Maxwell equation gives \(k_\mu k^\mu=0\) after a
transverse gauge choice \(k^\mu a_\mu=0\).  Since \(k_\mu\) is a gradient,
\[
k^\nu\nabla_\nu k_\mu
=\tfrac12\nabla_\mu(k^\nu k_\nu)=0.
\]
At the next order,
\(k^\mu\nabla_\mu |a|^2+|a|^2\nabla_\mu k^\mu=0\), or
\(\nabla_\mu(|a|^2k^\mu)=0\).
\end{solution}

\begin{solution}
For massless particles \(E=|\bm p|\) and isotropy gives
\[
T^{ij}=\int\frac{\dd^3p}{p^0}f\,p^ip^j
=\frac{\delta^{ij}}3\int\dd^3p\,f\,|\bm p|
=\frac{\rho}{3}\delta^{ij}.
\]
The homogeneous energy equation is
\(\dot\rho+3H(\rho+p)=0\), so
\(\dot\rho+4H\rho=0\) and \(\rho a^4=\) constant.
\end{solution}

\begin{solution}
\[
\nabla_\mu J^\mu
=(\nabla_\mu T^\mu{}_\nu)K^\nu
+T^{\mu\nu}\nabla_\mu K_\nu=0,
\]
because \(T^{\mu\nu}\) is symmetric while
\(\nabla_\mu K_\nu\) is antisymmetric.  The hypersurface charge is conserved
when the slices share a boundary, no flux escapes through that boundary, and
\(K^\mu\) is timelike in the region if the charge is to have the usual energy
interpretation.
\end{solution}
