\begin{exercises}

\begin{exercise}[Curvature from a commutator]{Tong, Cambridge Part III Sheet 2}
Starting from the coordinate expression for \(\nabla_\mu V^\rho\), compute
\([\nabla_\mu,\nabla_\nu]V^\rho\) and identify
\(R^\rho{}_{\sigma\mu\nu}\).  Repeat for a covector.
\end{exercise}

\begin{exercise}[Algebraic symmetries and component count]{Tong, Cambridge Part III Sheet 2}
For the Levi-Civita connection, derive the algebraic symmetries of
\(R_{\mu\nu\rho\sigma}\).  Show that the number of independent components in
\(n\) dimensions is \(n^2(n^2-1)/12\).
\end{exercise}

\begin{exercise}[Contracted Bianchi identity]{Tong, Cambridge Part III Sheet 2}
Contract the differential Bianchi identity and prove
\(\nabla_\mu G^{\mu\nu}=0\).  Explain why this identity strongly constrains
any local second-order field equation for the metric.
\end{exercise}

\begin{exercise}[Curvature of a two-sphere]{MIT 8.962, Problem Set 6}
For
\(\dd\ell^2=a^2(\dd\theta^2+\sin^2\theta\,\dd\phi^2)\), calculate the
connection, one independent Riemann component, the Ricci tensor, and the
scalar curvature.
\end{exercise}

\begin{exercise}[Geodesic deviation on a sphere]{MIT 8.962, Problem Set 7}
Consider neighboring meridians on a two-sphere, both parametrized by arc
length from the equator toward the north pole.  Compute their separation and
verify the geodesic-deviation equation directly.
\end{exercise}

\begin{exercise}[Infinitesimal holonomy]{MIT 8.962, Problem Set 7}
Parallel transport \(V^\mu\) around the infinitesimal parallelogram spanned by
\(\epsilon A^\mu\) and \(\epsilon B^\mu\).  Show that the leading change is
proportional to
\(R^\mu{}_{\nu\rho\sigma}V^\nu A^\rho B^\sigma\), including the sign for the
curvature convention of this edition.
\end{exercise}

\begin{exercise}[Weyl decomposition]{Tong, Cambridge Part III Sheet 2}
In \(n\geq3\) dimensions decompose the Riemann tensor into the Weyl tensor,
the traceless Ricci tensor, and the scalar curvature.  Show that the Weyl
tensor vanishes identically for \(n=3\).
\end{exercise}

\begin{exercise}[Conformally flat spacetime]{Cambridge Part III, 2017 exam}
For \(g_{\mu\nu}=e^{2\omega(x)}\eta_{\mu\nu}\), compute the Ricci tensor to
first order in \(\omega\).  Verify explicitly that the linearized Weyl tensor
vanishes.
\end{exercise}

\begin{exercise}[Schwarzschild tidal invariant]{Cambridge Part III, 2024 exam}
In an orthonormal freely falling frame outside a spherical mass, the electric
tidal tensor has eigenvalues
\((-2,1,1)GM/r^3\) up to the index-order sign convention.  Use this to find
\(R_{\mu\nu\rho\sigma}R^{\mu\nu\rho\sigma}\) and diagnose the surfaces
\(r=2GM\) and \(r=0\).
\end{exercise}

\begin{exercise}[Raychaudhuri focusing]{Cambridge Part III, 2021 exam}
For a hypersurface-orthogonal timelike geodesic congruence, derive the
Raychaudhuri equation.  Show that if \(R_{\mu\nu}u^\mu u^\nu\geq0\) and the
initial expansion is negative, a caustic forms within proper time at most
\(3/|\theta_0|\).
\end{exercise}

\begin{exercise}[Curvature identities from the connection]{Cambridge Part III, 2006 exam, Question 1}
Starting with a torsion-free metric connection, derive the algebraic
symmetries of \(R_{\mu\nu\rho\sigma}\), the first and second Bianchi
identities, and the contracted identity
\(\nabla_\mu G^{\mu\nu}=0\).  Also prove the Ricci commutator for vectors and
covectors and show that
\[
\nabla^\mu\nabla^\nu R_{\mu\nu\rho\sigma}=0.
\]
\end{exercise}

\begin{exercise}[A wave equation for curvature]{Cambridge Part III, 2025 exam, Question 2}
Derive the Riemann tensor from the commutator of covariant derivatives and
establish both Bianchi identities.  By differentiating the differential
Bianchi identity and commuting derivatives, show that in vacuum the Riemann
tensor obeys a nonlinear wave equation of the schematic form
\[
\Box R_{\mu\nu\rho\sigma}
=R_{\mu\nu\alpha\beta}R_{\rho\sigma}{}^{\alpha\beta}
+2R_{\alpha[\mu|\rho|\beta}
R^\alpha{}_{\nu]\sigma}{}^\beta
-2R_{\alpha[\mu|\sigma|\beta}
R^\alpha{}_{\nu]\rho}{}^\beta .
\]
What equation follows at first order for the curvature of
\(g_{\mu\nu}=\eta_{\mu\nu}+\epsilon h_{\mu\nu}\), and is it gauge
dependent?
\end{exercise}

\begin{solutions}

\begin{solution}
Direct expansion gives
\[
[\nabla_\mu,\nabla_\nu]V^\rho
=R^\rho{}_{\sigma\mu\nu}V^\sigma,
\quad
R^\rho{}_{\sigma\mu\nu}
=\partial_\mu\Gamma^\rho{}_{\nu\sigma}
-\partial_\nu\Gamma^\rho{}_{\mu\sigma}
+\Gamma^\rho{}_{\mu\lambda}\Gamma^\lambda{}_{\nu\sigma}
-\Gamma^\rho{}_{\nu\lambda}\Gamma^\lambda{}_{\mu\sigma}.
\]
For a covector the result is
\([\nabla_\mu,\nabla_\nu]\omega_\rho
=-R^\sigma{}_{\rho\mu\nu}\omega_\sigma\).
\end{solution}

\begin{solution}
Metric compatibility and zero torsion give
\[
R_{\mu\nu\rho\sigma}=-R_{\nu\mu\rho\sigma}
=-R_{\mu\nu\sigma\rho}
=R_{\rho\sigma\mu\nu},\qquad
R_{\mu[\nu\rho\sigma]}=0.
\]
View \(R\) as a symmetric bilinear form on the
\(N=n(n-1)/2\)-dimensional space of two-forms: it initially has
\(N(N+1)/2\) components.  The cyclic identity removes
\(\binom n4\), leaving \(n^2(n^2-1)/12\).
\end{solution}

\begin{solution}
From \(\nabla_{[\lambda}R_{\mu\nu]\rho\sigma}=0\), contract one index with
\(\rho\), then contract again with \(\mu\).  The result is
\[
\nabla_\mu R^{\mu\nu}-\frac12\nabla^\nu R=0,
\]
or \(\nabla_\mu G^{\mu\nu}=0\).  Thus a geometric field equation compatible
with conserved matter must have an identically divergence-free left side;
at second derivative order in four dimensions this essentially selects
\(G_{\mu\nu}+\Lambda g_{\mu\nu}\).
\end{solution}

\begin{solution}
The nonzero independent connection coefficients are
\(\Gamma^\theta{}_{\phi\phi}=-\sin\theta\cos\theta\) and
\(\Gamma^\phi{}_{\theta\phi}=\cot\theta\).  One finds
\[
R_{\theta\phi\theta\phi}=a^2\sin^2\theta,\qquad
R_{ij}=a^{-2}g_{ij},\qquad R=2a^{-2}.
\]
\end{solution}

\begin{solution}
If the meridians differ by a small angle \(\delta\phi\), their transverse
separation at arc length \(s=a\theta\) from the equator is
\(\ell(s)=a\delta\phi\cos(s/a)\).  Hence
\(\ell''=-\ell/a^2\).  The sectional curvature is \(1/a^2\), and the
geodesic-deviation equation gives precisely this focusing acceleration.
\end{solution}

\begin{solution}
Transport around the four sides and retain terms through \(\epsilon^2\).
The derivative terms and quadratic connection terms combine into curvature:
\[
\Delta V^\mu
=-\epsilon^2R^\mu{}_{\nu\rho\sigma}
V^\nu A^\rho B^\sigma+\order(\epsilon^3)
\]
for the path \(+A,+B,-A,-B\) and the convention used in this edition.
Reversing the loop reverses the sign.
\end{solution}

\begin{solution}
The decomposition is
\[
R_{\mu\nu\rho\sigma}
=C_{\mu\nu\rho\sigma}
+\frac{2}{n-2}
\left(g_{\mu[\rho}R_{\sigma]\nu}
-g_{\nu[\rho}R_{\sigma]\mu}\right)
-\frac{2R}{(n-1)(n-2)}
g_{\mu[\rho}g_{\sigma]\nu}.
\]
In three dimensions the Riemann and Ricci tensors both have six independent
components, so the trace-free remainder has none and \(C=0\).
\end{solution}

\begin{solution}
To linear order \(h_{\mu\nu}=2\omega\eta_{\mu\nu}\).  In four dimensions,
\[
R^{(1)}_{\mu\nu}
=-2\partial_\mu\partial_\nu\omega
-\eta_{\mu\nu}\Box\omega,\qquad
R^{(1)}=-6\Box\omega.
\]
Substitution into the four-dimensional Weyl decomposition cancels every
second-derivative term, so \(C^{(1)}_{\mu\nu\rho\sigma}=0\), as required by
conformal flatness.
\end{solution}

\begin{solution}
In vacuum the magnetic tidal tensor vanishes for Schwarzschild and the
Kretschmann scalar is eight times the squared norm of the electric tidal
tensor:
\[
K=8\left[(-2)^2+1^2+1^2\right]\frac{G^2M^2}{r^6}
=\frac{48G^2M^2}{r^6}.
\]
It is finite at \(r=2GM\), so that surface is a coordinate horizon, but
diverges at \(r=0\), a genuine curvature singularity.
\end{solution}

\begin{solution}
Decomposing \(\nabla_\nu u_\mu\) into expansion, shear, vorticity, and
acceleration and using the Ricci identity gives, for geodesic zero-vorticity
flow,
\[
\dot\theta=-\frac13\theta^2-\sigma_{\mu\nu}\sigma^{\mu\nu}
-R_{\mu\nu}u^\mu u^\nu.
\]
Under the stated condition \(\dot\theta\leq-\theta^2/3\).  Integration gives
\(1/\theta(\tau)\geq1/\theta_0+\tau/3\); a negative \(\theta_0\) therefore
diverges no later than \(3/|\theta_0|\).
\end{solution}

\begin{solution}
Expanding \([\nabla_\rho,\nabla_\sigma]V^\mu\) gives the coordinate
expression for curvature and its antisymmetry in \(\rho,\sigma\).  Metric
compatibility applied to the commutator acting on an inner product gives
antisymmetry in the first pair; the cyclic identity follows from zero
torsion, and pair exchange follows from these three relations.  The Jacobi
identity for three covariant derivatives gives
\(\nabla_{[\lambda}R_{\mu\nu]\rho\sigma}=0\).  Contracting twice yields
\(\nabla_\mu R^{\mu\nu}=\tfrac12\nabla^\nu R\), hence
\(\nabla_\mu G^{\mu\nu}=0\).

For a vector,
\([\nabla_\rho,\nabla_\sigma]V^\mu
=R^\mu{}_{\nu\rho\sigma}V^\nu\); for a covector the curvature term has the
opposite sign on its index.  Contract the differential Bianchi identity,
differentiate once more, and use the symmetry of the Ricci tensor together
with the commutator formula.  The remaining terms cancel pairwise, leaving
\(\nabla^\mu\nabla^\nu R_{\mu\nu\rho\sigma}=0\).
\end{solution}

\begin{solution}
The differential Bianchi identity gives
\[
\nabla^\lambda R_{\lambda\mu\rho\sigma}
=\nabla_\rho R_{\mu\sigma}-\nabla_\sigma R_{\mu\rho}.
\]
Apply another derivative, use \(R_{\mu\nu}=0\), and commute the derivative
past each index of the Riemann tensor.  Every commutator contributes one
curvature tensor, producing the displayed quadratic wave equation (up to
equivalent rearrangements using the Riemann symmetries).

On a flat background \(R^{(0)}_{\mu\nu\rho\sigma}=0\), so all quadratic
terms begin at order \(\epsilon^2\).  The order-\(\epsilon\) equation is
\[
\Box R^{(1)}_{\mu\nu\rho\sigma}=0.
\]
Because the background curvature vanishes, the linearized Riemann tensor is
unchanged by
\(h_{\mu\nu}\mapsto h_{\mu\nu}+2\partial_{(\mu}\xi_{\nu)}\); the equation is
therefore gauge invariant.
\end{solution}
