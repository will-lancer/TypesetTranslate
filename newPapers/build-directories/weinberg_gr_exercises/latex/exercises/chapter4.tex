\begin{exercises}

\begin{exercise}[Determinant as a density]{Tong, Cambridge Part III Sheet 1}
Under \(x^\mu\mapsto x'^\mu(x)\), derive the transformation of
\(g=\det g_{\mu\nu}\).  Show that \(\sqrt{-g}\,\dd^nx\) is invariant under an
orientation-preserving coordinate transformation.
\end{exercise}

\begin{exercise}[Lie derivative of a one-form]{Tong, Cambridge Part III Sheet 1}
Starting from \((\Lie_Xf)=X^\mu\partial_\mu f\), the Lie bracket, and the
Leibniz rule, derive \((\Lie_X\omega)_\mu\).  Then derive
\((\Lie_Xg)_{\mu\nu}\) and state the Killing equation.
\end{exercise}

\begin{exercise}[Difference of two connections]{Tong, Cambridge Part III Sheet 2}
Let \(\nabla\) and \(\widetilde\nabla\) be affine connections.  Prove that
\[
C^\rho{}_{\mu\nu}
=(\widetilde\Gamma^\rho{}_{\mu\nu}-\Gamma^\rho{}_{\mu\nu})
\]
is a \((1,2)\) tensor, even though neither set of connection coefficients is a
tensor.
\end{exercise}

\begin{exercise}[Exterior derivative identities]{Tong, Cambridge Part III Sheet 1}
For a \(p\)-form \(\alpha\) and a \(q\)-form \(\beta\), prove
\[
\dd(\alpha\wedge\beta)
=\dd\alpha\wedge\beta+(-1)^p\alpha\wedge\dd\beta,
\qquad \dd^2\alpha=0.
\]
Use these results to show that \(F=\dd A\) automatically satisfies the
homogeneous Maxwell equation.
\end{exercise}

\begin{exercise}[Covariant divergence]{McGreevy, Problem Set 4}
Show that
\[
\nabla_\mu V^\mu=\frac1{\sqrt{-g}}
\partial_\mu(\sqrt{-g}\,V^\mu).
\]
Use it to compute the divergence of a radial vector
\(\bm V=V^r(r)\partial_r\) in flat three-space written in spherical
coordinates.
\end{exercise}

\begin{exercise}[Laplacian in orthogonal coordinates]{MIT 8.962, Problem Set 3}
For a diagonal Euclidean metric
\(\dd\ell^2=\sum_i h_i^2(\dd q^i)^2\), derive the scalar Laplacian.  Specialize
to spherical polar coordinates and evaluate \(\nabla^2(r^nY_{\ell m})\).
\end{exercise}

\begin{exercise}[Parallel transport on a sphere]{Tong, Cambridge Part III Sheet 2}
A vector is parallel transported around a geodesic triangle on the unit
two-sphere with interior angles \(A,B,C\).  Show that its net rotation is the
spherical excess \(A+B+C-\pi\).  Verify this for the triangle bounded by the
equator and two meridians separated by angle \(\alpha\).
\end{exercise}

\begin{exercise}[Worldline covariant derivative]{McGreevy, Problem Set 5}
For \(u^\mu=\dd x^\mu/\dd\tau\) and a vector \(A^\mu(\tau)\), write
\(\Dworldline{A^\mu}\) in components.  Prove that
\[
\frac{\dd}{\dd\tau}(A_\mu B^\mu)
=\Dworldline{A_\mu}B^\mu+A_\mu\Dworldline{B^\mu}.
\]
Deduce that parallel transport preserves inner products.
\end{exercise}

\begin{exercise}[Hodge dual and Maxwell theory]{Cambridge Part III, 2024 exam}
In four-dimensional Lorentzian spacetime define
\((\star F)_{\mu\nu}=\tfrac12\epsilon_{\mu\nu\rho\sigma}F^{\rho\sigma}\).
Show that \(\star\star F=-F\), and rewrite the vacuum Maxwell equations as
\(\dd F=0=\dd\star F\).
\end{exercise}

\begin{exercise}[Non-coordinate frames]{Tong, Cambridge Part III Sheet 1}
Let \([e_a,e_b]=C^c{}_{ab}e_c\).  Derive the torsion-free relation between
\(\Gamma^c{}_{ab}\) and \(\Gamma^c{}_{ba}\).  For an orthonormal frame, use
metric compatibility to express the connection coefficients in terms of the
structure coefficients \(C^c{}_{ab}\).
\end{exercise}

\begin{exercise}[Projectively equivalent connections]{Cambridge Part III, 2001 exam, Question 1}
Let \(\nabla\) and \(\widetilde\nabla\) be torsion-free connections and set
\[
S^\rho{}_{\mu\nu}
=\widetilde\Gamma^\rho{}_{\mu\nu}-\Gamma^\rho{}_{\mu\nu}.
\]
First show that \(S^\rho{}_{\mu\nu}\) is a tensor.  If the two connections
have the same unparameterized geodesics, prove that
\[
S^\rho{}_{\mu\nu}
=\delta^\rho_\mu V_\nu+\delta^\rho_\nu V_\mu,
\qquad
V_\mu=\frac{1}{n+1}S^\rho{}_{\rho\mu}.
\]
\end{exercise}

\begin{exercise}[Forms, conformal symmetry, and Maxwell theory]{Cambridge Part III, 2025 exam, Question 4}
For a \(p\)-form \(\omega\), define \(\dd\omega\) in coordinates and prove
\(\dd^2=0\), naturality under pullback, and
\(\Lie_X\dd=\dd\Lie_X\).  In four dimensions, determine the constant
\(\alpha\) in the conformal Killing equation
\[
\nabla_\mu X_\nu+\nabla_\nu X_\mu
=\alpha g_{\mu\nu}\nabla_\rho X^\rho.
\]
Finally show that if \(\dd F=\dd\star F=0\), then
\(\widetilde F=\Lie_XF\) also solves the source-free Maxwell equations for
every conformal Killing field \(X\).
\end{exercise}

\begin{solution}
If \(J^\mu{}_\nu=\partial x'^\mu/\partial x^\nu\), then
\(g'=J^{-T}gJ^{-1}\), so
\[
\det g'=(\det J)^{-2}\det g.
\]
Meanwhile \(\dd^nx'=(\det J)\dd^nx\).  For \(\det J>0\),
\(\sqrt{-g'}\,\dd^nx'=\sqrt{-g}\,\dd^nx\).
\end{solution}

\begin{solution}
Applying \(\Lie_X\) to the scalar \(\omega(Y)\) gives
\[
(\Lie_X\omega)_\mu
=X^\nu\partial_\nu\omega_\mu+\omega_\nu\partial_\mu X^\nu.
\]
Leibniz then yields
\[
(\Lie_Xg)_{\mu\nu}
=X^\rho\partial_\rho g_{\mu\nu}
+g_{\rho\nu}\partial_\mu X^\rho
+g_{\mu\rho}\partial_\nu X^\rho
=\nabla_\mu X_\nu+\nabla_\nu X_\mu.
\]
A Killing field satisfies \(\nabla_{(\mu}X_{\nu)}=0\).
\end{solution}

\begin{solution}
For any one-form \(\omega_\rho\),
\[
(\widetilde\nabla_\mu-\nabla_\mu)\omega_\nu
=-C^\rho{}_{\mu\nu}\omega_\rho.
\]
The left side is the difference of two covariant derivatives and is therefore
tensorial.  Since \(\omega_\rho\) is arbitrary, \(C^\rho{}_{\mu\nu}\) must
transform as a tensor.  Equivalently, the inhomogeneous second-derivative
terms in the two connection transformation laws cancel.
\end{solution}

\begin{solution}
Antisymmetrizing the component derivative gives
\[
(\dd\alpha)_{\mu_0\cdots\mu_p}
=(p+1)\partial_{[\mu_0}\alpha_{\mu_1\cdots\mu_p]}.
\]
The graded product rule follows by distributing the derivative and moving it
past \(p\) antisymmetric differentials.  A second derivative vanishes because
partial derivatives commute while the indices are antisymmetrized.  Thus
\(\dd F=\dd^2A=0\).
\end{solution}

\begin{solution}
Using
\(\Gamma^\mu{}_{\mu\nu}=\partial_\nu\ln\sqrt{-g}\),
\[
\nabla_\mu V^\mu
=\partial_\mu V^\mu+\Gamma^\mu{}_{\mu\nu}V^\nu
=\frac1{\sqrt{-g}}\partial_\mu(\sqrt{-g}V^\mu).
\]
In spherical coordinates \(\sqrt g=r^2\sin\theta\), hence
\[
\bm\nabla\cdot\bm V=\frac1{r^2}\frac{\dd}{\dd r}(r^2V^r).
\]
\end{solution}

\begin{solution}
Here \(\sqrt g=\prod_i h_i\) and \(g^{ii}=h_i^{-2}\), so
\[
\nabla^2f=\frac1{\prod_jh_j}
\sum_i\partial_i\left(\frac{\prod_jh_j}{h_i^2}\partial_i f\right).
\]
In spherical coordinates this is the familiar radial term plus
\(r^{-2}\nabla^2_{S^2}\).  Since
\(\nabla^2_{S^2}Y_{\ell m}=-\ell(\ell+1)Y_{\ell m}\),
\[
\nabla^2(r^nY_{\ell m})
=\bigl[n(n+1)-\ell(\ell+1)\bigr]r^{n-2}Y_{\ell m}.
\]
\end{solution}

\begin{solution}
Gauss-Bonnet for a geodesic polygon gives
\(\int_\Delta K\,\dd A=A+B+C-\pi\).  Parallel transport rotates a vector by
the integrated Gaussian curvature, so the angle is the spherical excess.
For two meridians and the equator the angles are
\(\pi/2,\pi/2,\alpha\), hence the rotation is \(\alpha\).
\end{solution}

\begin{solution}
The derivative is
\[
\Dworldline{A^\mu}
=\frac{\dd A^\mu}{\dd\tau}
+\Gamma^\mu{}_{\rho\sigma}u^\rho A^\sigma.
\]
Apply the analogous formula to a covector.  The two connection terms cancel
after relabeling indices, leaving the ordinary derivative of \(A_\mu B^\mu\).
If both covariant derivatives vanish, their inner product is constant.
\end{solution}

\begin{solution}
Contracting two Levi-Civita tensors in signature \((-+++)\) gives
\[
\frac14\epsilon_{\mu\nu\rho\sigma}
\epsilon^{\rho\sigma\alpha\beta}F_{\alpha\beta}
=-F_{\mu\nu}.
\]
The Bianchi identity is precisely \(\dd F=0\).  The equation
\(\nabla_\mu F^{\mu\nu}=0\) is equivalent to the closure of the dual,
\(\dd\star F=0\).
\end{solution}

\begin{solution}
The torsion components are
\(T^c{}_{ab}=\Gamma^c{}_{ab}-\Gamma^c{}_{ba}-C^c{}_{ab}\), so zero torsion
gives
\(\Gamma^c{}_{ab}-\Gamma^c{}_{ba}=C^c{}_{ab}\).
With \(C_{abc}=\eta_{ad}C^d{}_{bc}\) and
\(\Gamma_{abc}=-\Gamma_{cba}\), cyclic permutations give
\[
\Gamma_{abc}
=\frac12\bigl(C_{abc}+C_{cab}-C_{bca}\bigr),
\]
with index placement adjusted to the stated definition of \(C^c{}_{ab}\).
\end{solution}

\begin{solution}
The inhomogeneous second-derivative terms in the transformation laws of the
two connections cancel, so their difference is a \((1,2)\) tensor.  Equality
of unparameterized geodesics means \(S(v,v)\) is parallel to \(v\) for every
tangent vector \(v\).  Polarizing this statement shows that the only
symmetric bilinear map with this property is
\[
S^\rho{}_{\mu\nu}
=\delta^\rho_\mu V_\nu+\delta^\rho_\nu V_\mu.
\]
Tracing \(\rho\) with \(\mu\) gives
\(S^\rho{}_{\rho\nu}=(n+1)V_\nu\), which fixes \(V\).  The extra term in the
geodesic equation is \(2(V_\mu\dot x^\mu)\dot x^\rho\), so it changes only
the affine parameter, not the curve.
\end{solution}

\begin{solution}
In a coordinate chart,
\[
(\dd\omega)_{\mu_0\cdots\mu_p}
=(p+1)\partial_{[\mu_0}\omega_{\mu_1\cdots\mu_p]}.
\]
Commuting partial derivatives proves \(\dd^2=0\); the chain rule proves
\(\dd(\phi^\ast\omega)=\phi^\ast(\dd\omega)\).  Differentiating the pullback
definition of \(\Lie_X\) gives
\(\Lie_X\dd=\dd\Lie_X\).  Tracing the conformal Killing equation in four
dimensions yields \(2\nabla\!\cdot X=4\alpha\nabla\!\cdot X\), hence
\(\alpha=\tfrac12\).

The Hodge star on two-forms is conformally invariant in four dimensions.
Consequently \([\Lie_X,\star]F=0\) for a conformal Killing field.  Therefore
\[
\dd(\Lie_XF)=\Lie_X(\dd F)=0,\qquad
\dd\star(\Lie_XF)=\dd\Lie_X(\star F)
=\Lie_X(\dd\star F)=0.
\]
\end{solution}
