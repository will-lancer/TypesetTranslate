\begin{exercises}

\begin{exercise}[Tolman--Oppenheimer--Volkoff equation]{MIT 8.962, Problem Set 12}
For a static spherical perfect fluid write
\[
\dd s^2=-e^{2\Phi(r)}\dd t^2+
\left(1-\frac{2Gm(r)}r\right)^{-1}\dd r^2+r^2\dd\Omega_2^2.
\]
Derive the equations for \(m'(r)\), \(\Phi'(r)\), and \(p'(r)\).
\end{exercise}

\begin{exercise}[Newtonian stellar limit]{Cambridge Part III, Structure and Evolution of Stars}
Take the weak-field, low-pressure limit of the TOV system.  Recover the
Newtonian equations of hydrostatic equilibrium and mass conservation, and
derive the virial scaling \(p_c\sim GM^2/R^4\).
\end{exercise}

\begin{exercise}[Lane--Emden reduction]{Cambridge Part III, Structure and Evolution of Stars}
For \(p=K\rho^{1+1/n}\), reduce Newtonian hydrostatic equilibrium to the
Lane--Emden equation.  Solve the \(n=1\) case and determine its first zero.
\end{exercise}

\begin{exercise}[White-dwarf mass scale]{Cambridge Part III, Structure and Evolution of Stars}
Use dimensional analysis for an ultra-relativistic degenerate electron gas,
\(p=K\rho^{4/3}\), to show that the equilibrium mass is independent of the
central density.  Determine its scaling with \(G,\hbar\), and the mass per
electron \(\mu_em_p\).
\end{exercise}

\begin{exercise}[Uniform-density interior]{McGreevy, Problem Set 8}
For \(\rho=\rho_0\) inside a star of radius \(R\), integrate the TOV equations
and obtain the pressure profile.  Match the interior metric to Schwarzschild
at the surface.
\end{exercise}

\begin{exercise}[Compactness bound]{Cambridge Part III, 2016 exam}
Use the central pressure of the uniform-density solution to show that it
diverges as \(R\to9GM/4\).  Explain the relation to the Buchdahl bound and
compare this radius with the Schwarzschild radius.
\end{exercise}

\begin{exercise}[Radial stability]{Cambridge Part III, Structure and Evolution of Stars}
For a Newtonian star with adiabatic index
\(\Gamma_1=(\partial\ln p/\partial\ln\rho)_s\), use a homologous radial
variation \(r\mapsto(1+\epsilon)r\) to show why stability requires
\(\Gamma_1>4/3\).
\end{exercise}

\begin{exercise}[Causality and the equation of state]{Rutgers Physics 617, General Relativity}
For a barotropic relativistic fluid, show that the local sound speed is
\(c_s^2=\dd p/\dd\rho\).  Explain the stability and causality restrictions on
\(c_s^2\), and derive \(p\leq\rho-\rho_s\) above a matching density
\(\rho_s\) if \(p(\rho_s)=0\).
\end{exercise}

\begin{exercise}[Oppenheimer--Snyder collapse]{Cambridge Part III, 2017 exam}
Model a collapsing homogeneous dust ball by a closed FLRW interior.  Show that
the scale factor can be written parametrically as
\[
a(\eta)=\frac{a_{\max}}2(1+\cos\eta),\qquad
t(\eta)=\frac{a_{\max}}2(\eta+\sin\eta).
\]
Identify the proper time to the singularity from maximum expansion.
\end{exercise}

\begin{exercise}[Matching a stellar surface]{Cambridge Part III, 2021 exam}
A spherical perfect-fluid interior is joined to a vacuum exterior across
\(r=R\).  State the continuity conditions for the induced metric and
extrinsic curvature.  Show why a smooth match to Schwarzschild requires
\(p(R)=0\) and identifies \(M=m(R)\).
\end{exercise}

\begin{exercise}[Radiation in hydrostatic equilibrium]{Cambridge Part III, 2008 exam, Question 1}
Let a static perfect fluid have energy density \(\rho\), pressure \(p\), and
four-velocity parallel to a timelike Killing field \(K^\mu\).  Derive
\[
\nabla_\mu p
=-(\rho+p)\nabla_\mu\ln\sqrt{-K^2}.
\]
For black-body radiation, \(p=\rho/3\), integrate this equation.  Explain why
a regular, compact, self-gravitating body made only of radiation cannot have
a free surface on which \(\rho\) tends continuously to zero.
\end{exercise}

\begin{solution}
The \(tt\) equation defines
\[
m'(r)=4\pi r^2\rho.
\]
The \(rr\) equation gives
\[
\Phi'=\frac{G[m+4\pi r^3p]}{r(r-2Gm)}.
\]
Finally \(\nabla_\mu T^{\mu r}=0\) gives
\[
p'=-(\rho+p)\Phi'
=-\frac{G(\rho+p)(m+4\pi r^3p)}{r(r-2Gm)}.
\]
\end{solution}

\begin{solution}
For \(2Gm/r\ll1\) and \(p\ll\rho\),
\[
\frac{\dd m}{\dd r}=4\pi r^2\rho,\qquad
\frac{\dd p}{\dd r}=-\frac{Gm\rho}{r^2}.
\]
Taking \(m\sim M\), \(\rho\sim M/R^3\), and a pressure gradient
\(p_c/R\) gives \(p_c\sim GM^2/R^4\).
\end{solution}

\begin{solution}
Set \(\rho=\rho_c\theta^n\) and
\(r=\alpha\xi\), with
\(\alpha^2=(n+1)K\rho_c^{1/n-1}/(4\pi G)\).  Then
\[
\frac1{\xi^2}\frac{\dd}{\dd\xi}
\left(\xi^2\frac{\dd\theta}{\dd\xi}\right)=-\theta^n,
\quad \theta(0)=1,\quad\theta'(0)=0.
\]
For \(n=1\), \(\theta=\sin\xi/\xi\), whose first zero is \(\xi_1=\pi\).
\end{solution}

\begin{solution}
Ultra-relativistic degeneracy gives
\[
K\sim\frac{\hbar}{(\mu_em_p)^{4/3}}.
\]
Balancing central degeneracy pressure
\(K(M/R^3)^{4/3}\) against \(GM^2/R^4\) cancels \(R\) and yields
\[
M_{\rm Ch}\sim
\frac{(\hbar/G)^{3/2}}{(\mu_em_p)^2},
\]
up to the dimensionless Lane--Emden coefficient.
\end{solution}

Here \(m(r)=4\pi\rho_0r^3/3=M(r/R)^3\).  Integration gives
\[
p(r)=\rho_0
\frac{\sqrt{1-2GMr^2/R^3}-\sqrt{1-2GM/R}}
{3\sqrt{1-2GM/R}-\sqrt{1-2GMr^2/R^3}}.
\]
The lapse is
\[
e^{\Phi(r)}
=\frac12\left[
3\sqrt{1-\frac{2GM}{R}}
-\sqrt{1-\frac{2GMr^2}{R^3}}\right],
\]
which matches the exterior lapse at \(r=R\).
\end{solution}

\begin{solution}
At \(r=0\), the denominator of the pressure is
\(3\sqrt{1-2GM/R}-1\).  It vanishes when
\(\sqrt{1-2GM/R}=1/3\), or \(R=9GM/4\).  Buchdahl's theorem extends
\(R>9GM/4\) to broad classes of isotropic stars with nonincreasing density.
This is larger than the horizon radius \(2GM\).
\end{solution}

\begin{solution}
Under \(R\mapsto\lambda R\) at fixed mass, density scales as
\(\lambda^{-3}\), internal energy as
\(U\propto\lambda^{-3(\Gamma_1-1)}\), and gravitational energy as
\(W\propto-\lambda^{-1}\).  Equilibrium cancels the first variation.
The second variation is proportional to \(3\Gamma_1-4\); it is positive only
for \(\Gamma_1>4/3\).
\end{solution}

\begin{solution}
Linearizing the local conservation equations in the fluid rest frame gives
\(\omega^2=c_s^2k^2\) with \(c_s^2=\dd p/\dd\rho\).  Stability requires
\(c_s^2\geq0\), while causality in units \(c=1\) requires \(c_s^2\leq1\).
Therefore
\[
p(\rho)=\int_{\rho_s}^{\rho}\frac{\dd p}{\dd\rho'}\dd\rho'
\leq\rho-\rho_s.
\]
\end{solution}

\begin{solution}
The dust Friedmann equation for positive curvature can be integrated by the
cycloid substitution stated in the question.  Maximum expansion is at
\(\eta=0\), while \(a=0\) at \(\eta=\pi\).  Hence the comoving proper time
from maximum radius to the singularity is
\[
\Delta t=\frac{\pi a_{\max}}2.
\]
\end{solution}

\begin{solution}
In the absence of a surface layer, the first and second fundamental forms
must agree across the surface.  Continuity of the angular metric fixes the
same areal radius; continuity of the radial metric fixes
\(M=m(R)\).  The normal-normal Einstein equation, or equivalently continuity
of the appropriate extrinsic-curvature component, requires the interior
radial pressure to equal the exterior pressure, hence \(p(R)=0\).
\end{solution}

\begin{solution}
Write \(N=\sqrt{-K^2}\), so the normalized fluid velocity is
\(u^\mu=K^\mu/N\).  Stationarity and the Killing equation give the
four-acceleration
\(a_\mu=u^\nu\nabla_\nu u_\mu=\nabla_\mu\ln N\).  Projecting
\(\nabla_\nu T^{\mu\nu}=0\) orthogonally to \(u^\mu\) gives
\(\nabla_\mu p=-(\rho+p)\nabla_\mu\ln N\).
For \(p=\rho/3\),
\[
\frac13\dd\ln\rho=-\frac43\dd\ln N,
\qquad
\rho N^4=\text{constant}.
\]
At a regular surface \(N\) is finite and nonzero.  Thus \(\rho\to0\) there
would force the constant, and hence \(\rho\), to vanish everywhere.  A
nontrivial radiation configuration therefore needs a confining boundary or
a non-free surface.
\end{solution}
