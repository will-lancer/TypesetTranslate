\begin{exercise}[PPN light deflection]{Cambridge Part III, 2019 exam}
For
\[
\dd s^2=-(1+2\Phi)\dd t^2+(1-2\gamma\Phi)\dd\bm x^2,
\qquad \Phi=-M/r,
\]
derive the leading light-deflection angle and show how it measures the PPN
parameter \(\gamma\).
\end{exercise}

\begin{exercise}[PPN Shapiro delay]{Cambridge Part III, 2016 exam}
Using the same PPN metric, compute the excess travel time for a ray passing a
mass with impact parameter \(b\) between radii \(r_1,r_2\gg b\).  Identify the
combination of PPN parameters tested by radar ranging.
\end{exercise}

\begin{exercise}[Lense--Thirring precession]{MIT 8.962, Problem Set 8}
From the far field of a rotating body with angular momentum \(\bm J\), derive
the gravitomagnetic precession of a stationary gyroscope.  Evaluate the
direction on the rotation axis and in the equatorial plane.
\end{exercise}

\begin{exercise}[Geodetic spin precession]{Cambridge Part III, 2022 exam}
Parallel transport a spin around a weak-field circular orbit.  Show that the
spin precesses relative to distant stars by \(3\pi M/r\) per orbit and
separate the special-relativistic Thomas and spatial-curvature contributions.
\end{exercise}

\begin{exercise}[Relativistic periastron advance of a binary]{Cambridge Part III, 2017 exam}
At first post-Newtonian order, show that a binary of total mass \(M\),
semimajor axis \(a\), and eccentricity \(e\) has periastron advance
\[
\Delta\varpi=\frac{6\pi M}{a(1-e^2)}
\]
per orbit.  Explain why the leading result is independent of the mass ratio.
\end{exercise}

\begin{exercise}[The binary mass function]{Cambridge Part III, General Relativity examples}
For a single-line spectroscopic or pulsar binary, derive
\[
f(m_1,m_2)=\frac{(m_2\sin i)^3}{(m_1+m_2)^2}
=\frac{P K_1^3}{2\pi G}(1-e^2)^{3/2}.
\]
Show how it gives a lower bound on the unseen companion mass.
\end{exercise}

\begin{exercise}[Dipole radiation in scalar--tensor gravity]{Cambridge Part III, 2013 exam}
Two compact bodies carry scalar charges \(\alpha_1m_1\) and
\(\alpha_2m_2\).  Show that their scalar dipole moment is proportional to
\((\alpha_1-\alpha_2)\) in the center-of-mass frame.  Compare the velocity
scaling of dipole and tensor-quadrupole power.
\end{exercise}

\begin{exercise}[Self-energy and strong-equivalence violation]{Cambridge Part III, 2017 exam}
Compute the Newtonian gravitational binding-energy fraction of a
uniform-density sphere.  If passive gravitational mass contains
\((1+\eta_N)E_g\) while inertial mass contains \(E_g\), derive the differential
acceleration of two bodies with different compactness.
\end{exercise}

\begin{exercise}[Gravitomagnetic force on a moving body]{McGreevy, Problem Set 8}
Write a stationary weak metric in terms of scalar and vector potentials.
Derive the slow-motion geodesic equation through first order in velocity and
show the analogy with an electric plus magnetic Lorentz force.  State which
numerical factors depend on the potential convention.
\end{exercise}
