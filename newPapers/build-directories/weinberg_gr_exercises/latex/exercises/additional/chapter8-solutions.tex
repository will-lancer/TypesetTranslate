\begin{solution}
Circularity and \(V_{\rm eff}'=0\) give
\[
E^2=\frac{(1-2M/r)^2}{1-3M/r},\qquad
L^2=\frac{Mr^2}{r-3M}.
\]
The second derivative changes sign at \(r=6M\).  There
\[
E_{\rm ISCO}=\sqrt{\frac89},\qquad L_{\rm ISCO}=2\sqrt3M,
\]
so the ideal binding efficiency is \(1-\sqrt{8/9}\simeq5.7\%\).
\end{solution}

\begin{solution}
With
\(r_\ast=r+2M\ln|r/(2M)-1|\) and \(v=t+r_\ast\),
\[
\dd s^2=-(1-2M/r)\dd v^2+2\dd v\,\dd r+r^2\dd\Omega^2.
\]
All coefficients are finite at \(r=2M\).  Ingoing rays have \(v=\) constant;
outgoing rays obey \(\dd r/\dd v=\tfrac12(1-2M/r)\) and become tangent to the
horizon.
\end{solution}

\begin{solution}
For \(u=t-r_\ast\), \(v=t+r_\ast\), set
\(U=-e^{-u/(4M)}\), \(V=e^{v/(4M)}\).  Then
\[
\dd s^2=-\frac{32M^3}{r}e^{-r/(2M)}\dd U\,\dd V+r^2\dd\Omega^2,
\quad
-UV=(r/2M-1)e^{r/(2M)}.
\]
The horizons are \(U=0\) or \(V=0\), the singularity is \(UV=1\), and the
signs of \(U,V\) distinguish two exteriors, a black-hole interior, and a
white-hole interior.
\end{solution}

\begin{solution}
For rest at infinity \(E=1\), so
\[
\frac{\dd r}{\dd\tau}=-\sqrt{\frac{2M}{r}}.
\]
Therefore
\[
\Delta\tau=\int_0^{2M}\sqrt{\frac{r}{2M}}\,\dd r=\frac{4M}{3}.
\]
Inside the horizon \(r\) is timelike: every future timelike curve moves
toward smaller \(r\), so a constant-\(r\) worldline would be spacelike.
\end{solution}

\begin{solution}
On a static radial slice of extremal Reissner--Nordstr\"om,
\[
\dd\ell=\frac{\dd r}{1-M/r}=\frac{r\,\dd r}{r-M}.
\]
Thus the distance contains
\[
\ell\sim M\ln\frac{r_{\rm fixed}-M}{\epsilon},
\]
which diverges as \(\epsilon\to0\).  The horizon area remains
\(4\pi M^2\); extremality creates an infinitely long throat, not an infinite
two-sphere.
\end{solution}

\begin{solution}
The lens equation is
\[
\beta=\theta-\frac{D_{LS}}{D_S}\frac{4M}{D_L\theta}.
\]
For \(\beta=0\),
\[
\theta_E^2=\frac{4M D_{LS}}{D_LD_S},\qquad
M=\frac{\theta_E^2D_LD_S}{4D_{LS}}.
\]
Alignment maps the source into a ring because of axial symmetry.
\end{solution}

\begin{solution}
For a null ray,
\[
b^2=\frac{r_0^2}{1-2M/r_0}
\]
at a turning point.  Its minimum occurs at \(r_0=3M\), giving
\[
b_c=3\sqrt3M,\qquad \sigma_{\rm capture}=\pi b_c^2=27\pi M^2.
\]
\end{solution}

\begin{solution}
For \(u^\mu=(1-2M/r)^{-1/2}\partial_t^\mu\),
\[
a=\sqrt{a_\mu a^\mu}
=\frac{M}{r^2\sqrt{1-2M/r}}.
\]
It diverges locally at the horizon.  Redshifting by the lapse gives
\(\sqrt{1-2M/r}\,a=M/r^2\to1/(4M)\), the Schwarzschild surface gravity.
\end{solution}
