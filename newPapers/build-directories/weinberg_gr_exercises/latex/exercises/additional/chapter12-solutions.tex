\begin{solution}
Einstein--Hilbert variation contains
\[
\delta I_G=\frac1{16\pi G}\int_{\mathcal M}\sqrt{-g}\,
G_{\mu\nu}\delta g^{\mu\nu}
+\frac1{16\pi G}\int_{\partial\mathcal M}\sqrt{|h|}\,n_\mu
(\nabla_\nu\delta g^{\mu\nu}-g_{\alpha\beta}\nabla^\mu\delta g^{\alpha\beta}).
\]
The term with normal derivatives is canceled by
\[
I_{\rm GHY}=\frac{\epsilon}{8\pi G}
\int_{\partial\mathcal M}\sqrt{|h|}\,K.
\]
The remaining boundary variation is proportional to \(\delta h_{ij}\), which
vanishes under Dirichlet data.
\end{solution}

\begin{solution}
The curvature decomposition is
\[
R={}^{(3)}R+K_{ij}K^{ij}-K^2+\text{divergence}.
\]
The divergence is precisely handled by the GHY term.  Hence
\[
I_G=\frac1{16\pi G}\int\dd t\,\dd^3x\,
N\sqrt h\left({}^{(3)}R+K_{ij}K^{ij}-K^2\right),
\]
where
\[
K_{ij}=\frac1{2N}
(\dot h_{ij}-D_iN_j-D_jN_i).
\]
\end{solution}

\begin{solution}
Writing the action canonically,
\[
I_G=\int\dd t\,\dd^3x
(\pi^{ij}\dot h_{ij}-N\mathcal H-N_i\mathcal H^i)
\]
up to boundary terms.  Thus
\[
\mathcal H=0,\qquad\mathcal H^i=0
\]
follow from varying \(N,N_i\).  Neither lapse nor shift has a time derivative
in \(I_G\), so their conjugate momenta vanish and they enforce gauge
constraints rather than propagate.
\end{solution}

\begin{solution}
Scalar variation gives
\[
\Box\phi-\xi R\phi-V'(\phi)=0.
\]
Metric variation gives
\[
T_{\mu\nu}
=\nabla_\mu\phi\nabla_\nu\phi
-\frac12g_{\mu\nu}(\nabla\phi)^2-g_{\mu\nu}V
+\xi\left(G_{\mu\nu}\phi^2
+g_{\mu\nu}\Box\phi^2-\nabla_\mu\nabla_\nu\phi^2\right).
\]
For \(V=0\) in four dimensions, local Weyl invariance requires
\(\xi=1/6\).
\end{solution}

\begin{solution}
Choose
\(\widetilde g_{\mu\nu}=F(\phi)g_{\mu\nu}/M_{\rm Pl}^2\).  Dropping the
transformed total derivative gives
\[
I=\int\sqrt{-\widetilde g}\left[
\frac{M_{\rm Pl}^2}{2}\widetilde R
-\frac12\mathcal K(\phi)(\widetilde\nabla\phi)^2
-\widetilde U(\phi)\right]\dd^4x,
\]
with
\[
\mathcal K=M_{\rm Pl}^2\left(\frac{Z}{F}
+\frac32\frac{F'^2}{F^2}\right),\qquad
\widetilde U=\frac{M_{\rm Pl}^4U}{F^2}.
\]
A further field redefinition makes the kinetic term canonical.
\end{solution}

\begin{solution}
The tetrad postulate is
\[
\partial_\mu e^a{}_\nu+\omega_\mu{}^a{}_b e^b{}_\nu
-\Gamma^\rho{}_{\mu\nu}e^a{}_\rho=0,
\]
so
\[
\omega_\mu{}^a{}_b
=e_b{}^\nu(\Gamma^\rho{}_{\mu\nu}e^a{}_\rho-\partial_\mu e^a{}_\nu).
\]
Under \(e\to\Lambda(x)e\),
\[
\omega_\mu\to\Lambda\omega_\mu\Lambda^{-1}
-(\partial_\mu\Lambda)\Lambda^{-1},
\]
the inhomogeneous transformation of a gauge connection.
\end{solution}

\begin{solution}
The metric variation of
\(\sqrt{-g}(R_{\mu\nu\rho\sigma}^2-4R_{\mu\nu}^2+R^2)\) produces the
Lanczos tensor.  In four dimensions the dimensionally dependent curvature
identity makes this tensor identically zero; only a boundary term remains.
For \(d>4\) the identity no longer holds, so the same variation gives
nontrivial second-order field equations.
\end{solution}

\begin{solution}
For fields \(\Phi\),
\[
\delta_\xi L=E_\Phi\Lie_\xi\Phi+\dd\Theta(\Phi,\Lie_\xi\Phi)
=\dd(\xi\cdot L).
\]
Thus on shell
\[
J_\xi=\Theta(\Phi,\Lie_\xi\Phi)-\xi\cdot L,\qquad \dd J_\xi=0.
\]
Locally the Poincar\'e lemma gives \(J_\xi=\dd Q_\xi\).  In Einstein gravity
\(Q_\xi\) is proportional to \(\star\dd\xi\), the Komar charge form.
\end{solution}

\begin{solution}
The on-shell variation is
\[
\delta I_{\rm on-shell}
=\frac{\epsilon}{16\pi G}\int_{\partial\mathcal M}\sqrt{|h|}
(K^{ij}-Kh^{ij})\delta h_{ij}.
\]
Hence
\[
\pi^{ij}=\frac{\epsilon\sqrt{|h|}}{16\pi G}(K^{ij}-Kh^{ij}),
\qquad
T_{\rm BY}^{ij}=\frac{2}{\sqrt{|h|}}\frac{\delta I}{\delta h_{ij}}
=\frac{\epsilon}{8\pi G}(K^{ij}-Kh^{ij}),
\]
up to reference or counterterm subtractions.
\end{solution}
