\begin{exercise}[Fixing the Newtonian normalization]{Tong, Cambridge Part III Sheet 3}
Linearize the \(00\) Einstein equation for a static weak field
\(g_{00}=-(1+2\Phi)\) sourced by nonrelativistic matter.  Recover Poisson's
equation and thereby fix the coefficient \(8\pi G\) in Einstein's equation.
\end{exercise}

\begin{exercise}[Energy conditions for a perfect fluid]{Cambridge Part III, General Relativity examples}
For a perfect fluid, express the null, weak, dominant, and strong energy
conditions as inequalities involving \(\rho\) and \(p\).  Apply them to dust,
radiation, vacuum energy, and a fluid with \(p=w\rho\).
\end{exercise}

\begin{exercise}[Vacuum curvature with a cosmological constant]{Cambridge Part III, 2011 exam}
In \(d\) dimensions solve the vacuum Einstein equation with cosmological
constant for \(R_{\mu\nu}\) and \(R\).  Determine the curvature radius of a
maximally symmetric solution and distinguish the signs corresponding to de
Sitter and anti-de Sitter spacetime.
\end{exercise}

\begin{exercise}[The linear field of a point mass]{MIT 8.962, Problem Set 8}
Solve the harmonic-gauge linearized Einstein equation for a static point mass.
Recover the weak Schwarzschild metric in isotropic coordinates and identify
the equal contributions of \(h_{00}\) and spatial curvature to light bending.
\end{exercise}

\begin{exercise}[Why the Fierz--Pauli mass is tuned]{Cambridge Part III, General Relativity examples}
Add the most general Lorentz-invariant quadratic mass term
\(-m^2(h_{\mu\nu}h^{\mu\nu}-\alpha h^2)/2\) to the linear spin-two action.
Show that \(\alpha=1\) imposes constraints leaving five propagating modes in
four dimensions, whereas generic \(\alpha\) propagates an additional scalar.
\end{exercise}

\begin{exercise}[Cosmological constant from unimodular gravity]{Cambridge Part III, General Relativity examples}
Assume the trace-free field equation
\[
R_{\mu\nu}-\frac14Rg_{\mu\nu}
=8\pi G\left(T_{\mu\nu}-\frac14Tg_{\mu\nu}\right).
\]
Use the Bianchi identity and stress-energy conservation to recover the full
Einstein equation with a cosmological constant as an integration constant.
\end{exercise}

\begin{exercise}[Time-symmetric initial data]{MIT 8.962, Problem Set 10}
Write the Hamiltonian and momentum constraints on a spacelike hypersurface.
Specialize to vacuum, \(\Lambda=0\), and time-symmetric data
\(K_{ij}=0\).  Show that a conformally flat metric
\(h_{ij}=\psi^4\delta_{ij}\) reduces the constraint to Laplace's equation and
recover the single-black-hole solution.
\end{exercise}

\begin{exercise}[Gravitational slip and anisotropic stress]{Cambridge Part III Cosmology, examples}
For static scalar perturbations
\[
\dd s^2=-(1+2\Phi)\dd t^2+(1-2\Psi)\dd\bm x^2,
\]
show from the traceless spatial Einstein equation that
\(\Phi=\Psi\) when anisotropic stress vanishes.  Explain how a measured
gravitational slip tests the matter model or the field equations.
\end{exercise}

\begin{exercise}[The trace source in scalar--tensor gravity]{Tong, Cambridge Part III Sheet 3}
Consider a scalar coupled as \(\phi R\).  Take the trace of the metric field
equation and combine it with the scalar equation to show why nonrelativistic
matter sources the scalar but conformal radiation does not.
\end{exercise}

\begin{exercise}[The four-dimensional Lovelock result]{Cambridge Part III, General Relativity examples}
Assume a local, symmetric, divergence-free rank-two tensor built from the
metric and at most its second derivatives, with field equations second order
in four dimensions.  Explain why only \(G_{\mu\nu}\) and \(g_{\mu\nu}\) occur.
Show explicitly why the Gauss--Bonnet density adds no local field equation.
\end{exercise}
