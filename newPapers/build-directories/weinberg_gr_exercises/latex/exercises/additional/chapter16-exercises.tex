\begin{exercise}[Scalaron mass in \(R+\alpha R^2\) gravity]{Cambridge Part III, 2024 exam}
Linearize the trace equation of \(f(R)=R+\alpha R^2\) gravity about
Minkowski spacetime.  Show that the extra scalar curvature mode is massive
and determine its mass.
\end{exercise}

\begin{exercise}[Cosmic no-hair with a cosmological constant]{Cambridge Part III Cosmology, examples}
For a Bianchi I universe containing matter with negligible anisotropic stress
and \(\Lambda>0\), derive the shear evolution.  Show that expansion approaches
de Sitter and quantify the decay of the shear relative to \(H\).
\end{exercise}

\begin{exercise}[A bounce and the null energy condition]{Cambridge Part III Cosmology, 2019 exam}
Use the Friedmann equations to determine the conditions for a nonsingular
bounce, \(H=0\) and \(\dot H>0\).  Show why a spatially flat bounce in ordinary
Einstein gravity violates the null energy condition.
\end{exercise}

\begin{exercise}[Finite time to a Big Rip]{Cambridge Part III Cosmology, examples}
For a flat universe dominated by a constant equation of state \(w<-1\), solve
for the scale factor and show that it diverges after a finite proper time.
Find the remaining lifetime in terms of the present Hubble rate.
\end{exercise}

\begin{exercise}[Oscillating scalar as an effective fluid]{Cambridge Part III Field Theory in Cosmology, examples}
A homogeneous scalar oscillates rapidly in
\(V(\phi)=\lambda|\phi|^n\).  Average over one oscillation to derive its
effective equation of state and density scaling.  Identify the cases that
mimic matter and radiation.
\end{exercise}

\begin{exercise}[Solar-system bound on Brans--Dicke theory]{Cambridge Part III, 2013 exam}
Starting from the weak-field Brans--Dicke solution, derive
\(\gamma=(1+\omega)/(2+\omega)\).  Translate a small observational bound on
\(|\gamma-1|\) into a lower bound on \(\omega\).
\end{exercise}

\begin{exercise}[The future de Sitter horizon]{Tong Cosmology, Cambridge Part III Sheet 2}
For exponential expansion, find the event-horizon proper distance and the
maximum comoving distance from which a signal emitted now can ever reach the
observer.  Compare comoving and proper descriptions.
\end{exercise}

\begin{exercise}[Chameleon screening in a dense environment]{Cambridge Part III, Modified Gravity examples}
Let a scalar have effective potential
\(V_{\rm eff}(\phi)=V(\phi)+\rho A(\phi)\).  Show how the density shifts the
minimum and mass.  Explain qualitatively how a thin shell suppresses the
fifth force of a large spherical body.
\end{exercise}

\begin{exercise}[Tolman's entropy problem for cyclic cosmology]{Cambridge Part III Cosmology, examples}
Assume ordinary entropy increases from cycle to cycle in a closed oscillating
universe.  Show qualitatively how the maximum scale factor and cycle duration
change, and explain why exactly periodic evolution is impossible without an
entropy-removal mechanism.
\end{exercise}

\begin{exercise}[Interacting dark-sector scaling solution]{Cambridge Part III Cosmology, examples}
Let pressureless matter and dark energy exchange energy through
\[
\dot\rho_m+3H\rho_m=Q,\qquad
\dot\rho_x+3H(1+w)\rho_x=-Q.
\]
For \(Q=3H\xi\rho_x\), find constant-ratio solutions
\(r=\rho_m/\rho_x\) and determine their effective equation of state.
\end{exercise}
