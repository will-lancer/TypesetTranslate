\begin{exercises}

\begin{exercise}[Killing fields on the two-sphere]{Tong, Cambridge Part III Sheet 3}
For
\(\dd\ell^2=\dd\theta^2+\sin^2\theta\,\dd\phi^2\), find three independent
Killing fields and verify that their Lie brackets form
\(\mathfrak{so}(3)\).
\end{exercise}

\begin{exercise}[Conserved quantity on a geodesic]{McGreevy, Problem Set 7}
If \(K^\mu\) is Killing and \(u^\mu\) is tangent to an affinely parameterized
geodesic, prove that \(K_\mu u^\mu\) is constant.  Generalize to a symmetric
Killing tensor \(K_{\mu_1\cdots\mu_p}\).
\end{exercise}

\begin{exercise}[Dimension of the isometry group]{Cambridge Part III, 2013 exam}
Show that a Killing field is fixed by its value \(K_\mu\) and antisymmetric
first derivative \(\nabla_\mu K_\nu\) at one point.  Deduce the upper bound
\(n(n+1)/2\) on the number of independent Killing fields in \(n\) dimensions.
\end{exercise}

\begin{exercise}[Curvature of a maximally symmetric space]{Cambridge Part III, 2014 exam}
Use maximal symmetry to show
\[
R_{\mu\nu\rho\sigma}
=K(g_{\mu\rho}g_{\nu\sigma}-g_{\mu\sigma}g_{\nu\rho}).
\]
Find \(R_{\mu\nu}\), \(R\), and \(K\) in terms of a cosmological constant in
\(n\) dimensions.
\end{exercise}

\begin{exercise}[de Sitter as a hyperboloid]{Tong, Cambridge Part III Sheet 3}
Embed \(n\)-dimensional de Sitter space as
\(-X_0^2+\sum_{i=1}^nX_i^2=\ell^2\) in \((1,n)\)-dimensional Minkowski space.
Introduce global coordinates and derive the induced metric.
\end{exercise}

\begin{exercise}[Anti-de Sitter global time]{Cambridge Part III, 2018 exam}
Embed anti-de Sitter space in a flat space with two timelike directions.
Derive global coordinates, identify the closed timelike curves of the
hyperboloid, and explain the universal-cover construction.
\end{exercise}

\begin{exercise}[Homogeneous and isotropic spatial metric]{Tong Cosmology, Cambridge Part III Sheet 1}
Starting only from spatial homogeneity and isotropy, show that a
three-dimensional spatial metric has constant curvature and may be written
\[
\dd\ell^2=a^2(t)\left[
\frac{\dd r^2}{1-kr^2}+r^2\dd\Omega_2^2\right].
\]
\end{exercise}

\begin{exercise}[Invariant rank-two tensors]{Cambridge Part III, 2012 exam}
Show that any symmetric rank-two tensor invariant under every isometry of a
maximally symmetric space is proportional to the metric.  Extend the argument
to invariant vectors and invariant scalars.
\end{exercise}

\begin{exercise}[Killing horizon surface gravity]{Cambridge Part III, 2009 exam}
Let a Killing field \(K^\mu\) become null on a Killing horizon.  Show that on
the horizon
\[
\nabla_\mu(K^\nu K_\nu)=-2\kappa K_\mu.
\]
Evaluate \(\kappa\) for a static metric
\(\dd s^2=-f(r)\dd t^2+f(r)^{-1}\dd r^2+\cdots\) at a simple root \(r_h\).
\end{exercise}

\begin{exercise}[Quotients and local symmetry]{Cambridge Part III, 2019 exam}
Flat space with one spatial identification \(x\sim x+L\) is locally maximally
symmetric but has fewer global isometries.  Determine which infinitesimal
Poincar\'e transformations descend to the quotient and explain the
distinction between local and global symmetry.
\end{exercise}

\begin{exercise}[Killing fields form a Lie algebra]{Cambridge Part III, 2003 exam, Question 1}
Starting from the pullback definition of the Lie derivative, derive its
component form on scalars, vectors, covectors, and general tensors.  Prove
\[
[\Lie_X,\Lie_Y]=\Lie_{[X,Y]}.
\]
Deduce that a constant linear combination and the commutator of two Killing
fields are again Killing.  If a stationary axisymmetric spacetime has
exactly two independent Killing fields \(T\) and \(\Phi\), show that
\([T,\Phi]=0\).
\end{exercise}

\begin{exercise}[Radial geodesics in anti-de Sitter space]{Cambridge Part III, 2025 exam, Question 1}
Consider unit-radius anti-de Sitter space in static coordinates,
\[
\dd s^2=-(1+r^2)\dd t^2+\frac{\dd r^2}{1+r^2}
+r^2\dd\Omega_2^2.
\]
Use the geodesic action \(I\) to find the radial first integrals.  Show that
every radial timelike geodesic leaving \(r=0\) returns after proper time
\(\pi\).  Show also that an outward radial null geodesic emitted from
\(r=t=0\) reaches \(r\to\infty\) only at infinite affine parameter but at
finite coordinate time \(t=\pi/2\).
\end{exercise}

\end{exercises}

\begin{solutions}

\begin{solution}
One convenient basis is
\[
J_3=\partial_\phi,\quad
J_1=-\sin\phi\,\partial_\theta-\cot\theta\cos\phi\,\partial_\phi,\quad
J_2=\cos\phi\,\partial_\theta-\cot\theta\sin\phi\,\partial_\phi.
\]
Direct substitution gives \(\Lie_{J_i}g=0\), and their brackets satisfy
\([J_i,J_j]=\epsilon_{ijk}J_k\), up to an overall sign from basis convention.
\end{solution}

\begin{solution}
\[
\DbyDtau(K_\mu u^\mu)
=u^\nu u^\mu\nabla_\nu K_\mu
=u^\nu u^\mu\nabla_{(\nu}K_{\mu)}=0.
\]
For a Killing tensor obeying
\(\nabla_{(\rho}K_{\mu_1\cdots\mu_p)}=0\), the same argument shows that
\[
K_{\mu_1\cdots\mu_p}u^{\mu_1}\cdots u^{\mu_p}
=\text{constant}.
\]
\end{solution}

\begin{solution}
Differentiating the Killing equation and commuting derivatives gives
\[
\nabla_\mu\nabla_\nu K_\rho
=R_{\rho\nu\mu}{}^\sigma K_\sigma
\]
with the convention of this edition.  Thus all higher derivatives are
determined by \(K_\mu\) and \(\nabla_{[\mu}K_{\nu]}\) at one point.  They have
\(n\) and \(n(n-1)/2\) components, totaling \(n(n+1)/2\).
\end{solution}

\begin{solution}
Isotropy leaves no invariant rank-four tensor with the Riemann symmetries
other than the stated metric combination.  Contraction gives
\[
R_{\mu\nu}=(n-1)Kg_{\mu\nu},\qquad
R=n(n-1)K.
\]
Vacuum Einstein equations imply
\[
K=\frac{2\Lambda}{(n-1)(n-2)}.
\]
\end{solution}

\begin{solution}
Set
\[
X_0=\ell\sinh(t/\ell),\qquad
X_i=\ell\cosh(t/\ell)\,n_i,\qquad \sum_i n_i^2=1.
\]
Pulling back the ambient metric gives
\[
\dd s^2=-\dd t^2+\ell^2\cosh^2(t/\ell)\dd\Omega_{n-1}^2.
\]
\end{solution}

\begin{solution}
In ambient signature \((-,-,+,\ldots,+)\), use
\[
X_{-1}=\ell\cosh\rho\cos t,\quad
X_0=\ell\cosh\rho\sin t,\quad
X_i=\ell\sinh\rho\,n_i.
\]
Then
\[
\dd s^2=\ell^2[-\cosh^2\rho\,\dd t^2+\dd\rho^2
+\sinh^2\rho\,\dd\Omega^2].
\]
The embedding identifies \(t\sim t+2\pi\), creating closed timelike curves.
Taking \(t\in\mathbb R\) unwraps them and defines the universal cover.
\end{solution}

\begin{solution}
Isotropy about every point forces the spatial Riemann tensor to have the
constant-curvature form.  Geodesic polar coordinates then give
\(\dd\ell^2=\dd\chi^2+S_k^2(\chi)\dd\Omega_2^2\), where
\(S_k=\sin\chi,\chi,\sinh\chi\).  Defining
\(r=S_k(\chi)\) yields the stated form, while homogeneity allows only an
overall time-dependent scale factor.
\end{solution}

\begin{solution}
At a point, the isotropy subgroup contains every spatial rotation (or the
appropriate Lorentz subgroup).  No nonzero vector is fixed by all of these
transformations, so an invariant vector vanishes.  Schur's-lemma reasoning
leaves only the metric as an invariant symmetric bilinear form:
\(S_{\mu\nu}=Cg_{\mu\nu}\).  Transitivity makes an invariant scalar, and the
coefficient \(C\), constant.
\end{solution}

\begin{solution}
On the horizon \(K^2=0\), and its gradient is normal to the horizon.  Since
\(K^\mu\) is also its null normal, the two are proportional, defining
\(\kappa\) by the stated equation.  For the static metric,
\[
\kappa=\frac12|f'(r_h)|.
\]
For Schwarzschild this gives \(1/(4GM)\).
\end{solution}

\begin{solution}
A global transformation must preserve the lattice generated by
\(L\partial_x\).  Continuous translations in every direction descend, as do
rotations and boosts that leave the compact \(x\)-direction fixed; generic
boosts mixing \(t\) with \(x\), and generic rotations mixing \(x\) with
noncompact directions, do not.  Curvature and the local Killing algebra
remain flat-space ones, but only lattice-preserving generators exponentiate
to global isometries of the quotient.
\end{solution}

\begin{solution}
Differentiating the pullback along the flow of \(X\) gives
\[
(\Lie_XY)^\mu=X^\nu\partial_\nu Y^\mu-Y^\nu\partial_\nu X^\mu,
\quad
(\Lie_X\omega)_\mu
=X^\nu\partial_\nu\omega_\mu+\omega_\nu\partial_\mu X^\nu,
\]
with one corresponding term for each tensor index.  The Jacobi identity for
vector-field commutators gives
\([\Lie_X,\Lie_Y]=\Lie_{[X,Y]}\).  Since
\(\Lie_Xg=\Lie_Yg=0\), both constant linear combinations and
\([X,Y]\) annihilate the metric.

Thus \([T,\Phi]\) is a Killing field and must be a constant linear
combination \(aT+b\Phi\).  In the asymptotically flat region,
\(T\to\partial_t\) and \(\Phi\to\partial_\phi\), whose commutator vanishes.
Therefore \(a=b=0\) and \([T,\Phi]=0\) everywhere.
\end{solution}

\begin{solution}
For radial motion the cyclic coordinate gives
\[
e=(1+r^2)\dot t,
\qquad
k=g_{\mu\nu}\dot x^\mu\dot x^\nu
=\frac{\dot r^2-e^2}{1+r^2},
\]
where \(k=-1\) for proper-time timelike motion and \(k=0\) for null motion.
For a timelike geodesic,
\[
\dot r^2=e^2-1-r^2,\qquad \ddot r=-r.
\]
The radial coordinate therefore returns to zero after
\(\Delta\tau=\pi\), independent of the amplitude.

For an outward null ray, \(\dot r=e\), so \(r=e\lambda\) after choosing the
origin of affine parameter.  Meanwhile
\[
\frac{\dd t}{\dd r}=\frac1{1+r^2},
\qquad
t(r)=\arctan r.
\]
Hence \(r\to\infty\) as \(\lambda\to\infty\), while
\(t\to\pi/2\), displaying the timelike conformal boundary of anti-de Sitter
space.
\end{solution}
