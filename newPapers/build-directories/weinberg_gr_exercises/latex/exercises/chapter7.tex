\begin{exercises}

\begin{exercise}[Why the Einstein tensor]{Cambridge Part III, 2015 exam}
Explain why \(R_{\mu\nu}=\kappa T_{\mu\nu}\) is generally incompatible with
stress-tensor conservation.  Determine the trace adjustment that produces a
divergence-free rank-two tensor containing at most second derivatives of the
metric.
\end{exercise}

\begin{exercise}[Vacuum energy and the cosmological constant]{Cambridge Part III, 2011 exam}
Take \(T_{\mu\nu}^{\rm vac}=-\rho_{\rm vac}g_{\mu\nu}\).  Show how it can be
absorbed into a cosmological constant, find the constant-curvature vacuum
solution in four dimensions, and relate its curvature radius to
\(\Lambda_{\rm eff}\).
\end{exercise}

\begin{exercise}[Trace reversal in \(d\) dimensions]{Cambridge Part III, 2012 exam}
From
\[
G_{\mu\nu}+\Lambda g_{\mu\nu}=8\pi G T_{\mu\nu}
\]
derive an expression for \(R_{\mu\nu}\) in \(d\) dimensions.  Specialize to a
traceless source and to vacuum.
\end{exercise}

\begin{exercise}[Scalar-field source]{Cambridge Part III, 2015 exam}
For
\[
I_\phi=-\int\dd^4x\sqrt{-g}
\left[\frac12\nabla_\mu\phi\nabla^\mu\phi+V(\phi)\right],
\]
derive \(T_{\mu\nu}\), the scalar equation, and
\(\nabla_\mu T^{\mu\nu}=0\) on shell.
\end{exercise}

\begin{exercise}[Brans-Dicke field equations]{Tong, Cambridge Part III Sheet 3}
Vary
\[
I=\frac1{16\pi G}\int\dd^4x\sqrt{-g}
\left(\phi R-\frac{\omega}{\phi}\nabla_\mu\phi\nabla^\mu\phi\right)
+I_{\rm m}.
\]
Obtain the metric and scalar equations, discarding boundary terms.
\end{exercise}

\begin{exercise}[Harmonic gauge]{Cambridge Part III, 2014 exam}
Linearize about Minkowski space, define
\(\bar h_{\mu\nu}=h_{\mu\nu}-\tfrac12\eta_{\mu\nu}h\), and show that the gauge
condition \(\partial^\mu\bar h_{\mu\nu}=0\) reduces the field equations to
wave equations.  Identify the residual gauge freedom.
\end{exercise}

\begin{exercise}[Hamiltonian constraint]{Cambridge Part III, 2018 exam}
For a spacelike hypersurface with induced metric \(h_{ij}\), unit normal
\(n^\mu\), and extrinsic curvature \(K_{ij}\), use the Gauss equation to show
\[
{}^{(3)}R+K^2-K_{ij}K^{ij}=16\pi G\,T_{\mu\nu}n^\mu n^\nu.
\]
Why is this a constraint rather than an evolution equation?
\end{exercise}

\begin{exercise}[ADM energy in the weak field]{Cambridge Part III, 2018 exam}
For an asymptotically flat spatial metric
\(h_{ij}=\delta_{ij}+q_{ij}\), evaluate
\[
E_{\rm ADM}=\frac1{16\pi G}\lim_{r\to\infty}
\int_{S_r}(\partial_jq_{ij}-\partial_iq_{jj})n^i\,\dd A
\]
for the asymptotic Schwarzschild field in isotropic coordinates.
\end{exercise}

\begin{exercise}[\(f(R)\) gravity]{Cambridge Part III, 2024 exam}
Vary
\[
I=\frac1{16\pi G}\int\dd^4x\sqrt{-g}\,f(R)+I_{\rm m}
\]
with respect to the inverse metric.  Derive the field equation and its trace.
Show that \(f(R)=R-2\Lambda\) returns Einstein gravity.
\end{exercise}

\begin{exercise}[Conserved total complex]{Cambridge Part III, 2010 exam}
In harmonic coordinates, write the nonlinear Einstein equation schematically
as a flat-spacetime wave equation for \(\bar h_{\mu\nu}\) sourced by
\(\tau_{\mu\nu}=T_{\mu\nu}+t_{\mu\nu}^{\rm grav}\).  Explain why
\(\partial_\mu\tau^{\mu\nu}=0\) does not make
\(t_{\mu\nu}^{\rm grav}\) a generally covariant local stress tensor.
\end{exercise}

\end{exercises}

\begin{solutions}

\begin{solution}
The Bianchi identity gives
\(\nabla^\mu R_{\mu\nu}=\tfrac12\nabla_\nu R\), which is not generally zero.
For \(R_{\mu\nu}+ag_{\mu\nu}R\), the divergence vanishes only when
\(a=-1/2\).  Thus the required tensor is
\(G_{\mu\nu}=R_{\mu\nu}-\tfrac12Rg_{\mu\nu}\), up to
\(\Lambda g_{\mu\nu}\).
\end{solution}

\begin{solution}
Moving the vacuum term to the geometric side gives
\(\Lambda_{\rm eff}=\Lambda+8\pi G\rho_{\rm vac}\).  In vacuum,
\(R_{\mu\nu}=\Lambda_{\rm eff}g_{\mu\nu}\) and
\[
R_{\mu\nu\rho\sigma}
=\frac{\Lambda_{\rm eff}}3
(g_{\mu\rho}g_{\nu\sigma}-g_{\mu\sigma}g_{\nu\rho}).
\]
The curvature radius obeys \(\ell^2=3/|\Lambda_{\rm eff}|\), with de Sitter
for positive and anti-de Sitter for negative \(\Lambda_{\rm eff}\).
\end{solution}

\begin{solution}
Taking the trace gives
\((1-d/2)R+d\Lambda=8\pi GT\).  Therefore
\[
R_{\mu\nu}
=8\pi G\left(T_{\mu\nu}-\frac{T}{d-2}g_{\mu\nu}\right)
+\frac{2\Lambda}{d-2}g_{\mu\nu}.
\]
For \(T=0\) the trace adjustment disappears; in vacuum only the final term
remains.
\end{solution}

\begin{solution}
Metric variation gives
\[
T_{\mu\nu}
=\nabla_\mu\phi\nabla_\nu\phi
-g_{\mu\nu}\left[\frac12(\nabla\phi)^2+V\right].
\]
Field variation gives
\(\Box\phi-V'(\phi)=0\).  Direct differentiation yields
\(\nabla_\mu T^{\mu\nu}
=(\Box\phi-V')\nabla^\nu\phi\), which vanishes on shell.
\end{solution}

\begin{solution}
The metric equation is
\[
\phi G_{\mu\nu}
=8\pi G T_{\mu\nu}
+\frac{\omega}{\phi}
\left(\nabla_\mu\phi\nabla_\nu\phi
-\frac12g_{\mu\nu}(\nabla\phi)^2\right)
+\nabla_\mu\nabla_\nu\phi-g_{\mu\nu}\Box\phi.
\]
Combining the direct scalar variation with the trace of this equation gives
\[
(2\omega+3)\Box\phi=8\pi G T
\]
when \(\omega\) is constant and matter does not couple directly to \(\phi\).
\end{solution}

\begin{solution}
Under
\(h_{\mu\nu}\mapsto h_{\mu\nu}-\partial_\mu\xi_\nu-\partial_\nu\xi_\mu\),
\(\partial^\mu\bar h_{\mu\nu}\mapsto
\partial^\mu\bar h_{\mu\nu}-\Box\xi_\nu\).
In harmonic gauge,
\[
\Box\bar h_{\mu\nu}=-16\pi G T_{\mu\nu}.
\]
Transformations with \(\Box\xi^\mu=0\) preserve the gauge and constitute the
residual freedom.
\end{solution}

\begin{solution}
Twice contracting the Gauss equation gives
\[
{}^{(3)}R+K^2-K_{ij}K^{ij}=2G_{\mu\nu}n^\mu n^\nu.
\]
Einstein's equation supplies the stated right side.  It contains no second
derivative normal to the hypersurface, so initial data must satisfy it before
evolution begins.
\end{solution}

\begin{solution}
In isotropic coordinates
\(h_{ij}=(1+GM/2r)^4\delta_{ij}
=\delta_{ij}+2GM\,\delta_{ij}/r+\order(r^{-2})\).
Thus
\[
\partial_jq_{ij}-\partial_iq_{jj}
=4GM\,\frac{x_i}{r^3}.
\]
The surface integral is \(16\pi GM\), and the prefactor gives
\(E_{\rm ADM}=M\).
\end{solution}

\begin{solution}
After integrating the variation of \(R\) by parts,
\[
f'(R)R_{\mu\nu}-\frac12f(R)g_{\mu\nu}
+(g_{\mu\nu}\Box-\nabla_\mu\nabla_\nu)f'(R)
=8\pi G T_{\mu\nu}.
\]
Its trace is
\[
f'R-2f+3\Box f'=8\pi GT.
\]
For \(f=R-2\Lambda\), \(f'=1\) and the derivative terms vanish.
\end{solution}

\begin{solution}
Moving all terms nonlinear in \(h_{\mu\nu}\) to the right gives
\(\Box\bar h_{\mu\nu}=-16\pi G\tau_{\mu\nu}\).
The harmonic condition implies ordinary conservation of the effective source.
However, a coordinate transformation can make the connection vanish at a
point and hence change or remove any local expression quadratic in first
derivatives.  The gravitational term is therefore a coordinate-dependent
pseudotensor; only suitable integrated asymptotic charges are invariant.
\end{solution}

\end{solutions}
