\begin{exercises}

\begin{exercise}[Rindler coordinates]{McGreevy, Problem Set 3}
Introduce coordinates \(t=\xi\sinh(a\eta)\),
\(x=\xi\cosh(a\eta)\) in the right Rindler wedge.  Find the metric, the proper
acceleration of a worldline \(\xi=\xi_0\), and the ratio of clock rates at two
fixed values of \(\xi\).
\end{exercise}

\begin{exercise}[Light in an accelerating laboratory]{Cambridge Part III, 2014 exam}
In a laboratory accelerating upward with acceleration \(g\), a horizontal
light ray crosses a distance \(L\).  Use the equivalence principle to find its
downward displacement and deflection angle to leading order in \(gL\).
\end{exercise}

\begin{exercise}[Newtonian limit of the geodesic equation]{Cambridge Part III, 2014 exam}
Let
\[
\dd s^2=-(1+2\Phi)\dd t^2+(1-2\Phi)\delta_{ij}\dd x^i\dd x^j,
\qquad |\Phi|\ll1.
\]
For slowly moving particles, derive \(\dd^2x^i/\dd t^2=-\partial_i\Phi\).
Which metric component is fixed by this argument?
\end{exercise}

\begin{exercise}[Gravitational frequency shift]{MIT 8.962, Problem Set 4}
Two stationary observers in a static spacetime exchange light.  Show that
\(\omega\sqrt{-g_{00}}\) is constant along the ray.  Apply the result near
Earth, where \(\Phi(z)=gz\), and find the fractional shift over a height \(h\).
\end{exercise}

\begin{exercise}[Rotating coordinates]{McGreevy, Problem Set 4}
Transform Minkowski space to cylindrical coordinates rotating with angular
velocity \(\Omega\).  Identify \(g_{00}\), the surface on which the rotating
time coordinate becomes null, and the leading centrifugal and Coriolis terms
in the nonrelativistic equations of motion.
\end{exercise}

\begin{exercise}[Universality of free fall]{Cambridge Part III, 2013 exam}
Suppose inertial and gravitational masses obey
\(m_g/m_i=1+\eta_A\) for material \(A\).  Derive the Eötvös parameter for two
test bodies \(A,B\) in the same external field, to first order in the small
\(\eta\)'s.  Explain why a universal constant ratio is unobservable.
\end{exercise}

\begin{exercise}[Local inertial coordinates]{Tong, Cambridge Part III Sheet 2}
At a point \(p\), show explicitly that a quadratic coordinate change can set
\(\Gamma^\rho{}_{\mu\nu}(p)=0\) for a torsion-free connection.  Count the
adjustable coefficients and explain why the first derivatives of the metric
can then be made to vanish for the Levi-Civita connection.
\end{exercise}

\begin{exercise}[What cannot be transformed away]{Cambridge Part III, 2024 exam}
Two neighboring freely falling particles have separation \(s^\mu\) and common
four-velocity \(u^\mu\).  Derive the observable relative acceleration in a
local inertial frame and explain why curvature, unlike the connection, cannot
be eliminated throughout a finite laboratory.
\end{exercise}

\begin{exercise}[Weight of internal energy]{McGreevy, Problem Set 2}
A perfectly reflecting box contains radiation of total rest-frame energy
\(E_\gamma\).  Compare the acceleration of the empty box and the filled box in
a weak, uniform gravitational field.  Show that consistency of the
equivalence principle assigns gravitational mass \(E_\gamma\) to the
radiation.
\end{exercise}

\begin{exercise}[Fermi-normal tidal metric]{Cambridge Part III, 2018 exam}
Along a freely falling, nonrotating worldline, the metric begins as
\[
g_{00}=-1-R_{0i0j}x^ix^j+\order(|\bm x|^3).
\]
Use the slow-particle geodesic equation to recover the tidal acceleration.
For a Newtonian point mass, give the three principal tidal eigenvalues.
\end{exercise}

\end{exercises}

\begin{solutions}

\begin{solution}
The transformed line element is
\[
\dd s^2=-a^2\xi^2\dd\eta^2+\dd\xi^2+\dd y^2+\dd z^2.
\]
A fixed-\(\xi_0\) observer has proper time
\(\dd\tau=a\xi_0\dd\eta\) and proper acceleration \(1/\xi_0\).
Thus two Rindler clocks satisfy
\(\dd\tau_2/\dd\tau_1=\xi_2/\xi_1\).
\end{solution}

\begin{solution}
The crossing time is \(L\) in units \(c=1\).  During this time the accelerating
floor rises by \(\tfrac12gL^2\), so the ray appears to fall by that amount.
Its final vertical velocity relative to the laboratory is \(-gL\), giving
\(\Delta\theta=-gL\).  Restoring \(c\) replaces \(gL\) by \(gL/c^2\).
\end{solution}

\begin{solution}
To the required order \(u^0\simeq1\) and
\[
\Gamma^i{}_{00}
=-\frac12\delta^{ij}\partial_jg_{00}
=\partial_i\Phi.
\]
The spatial geodesic equation becomes
\(\dd^2x^i/\dd t^2=-\Gamma^i{}_{00}=-\partial_i\Phi\).
Only \(g_{00}=-(1+2\Phi)\) is fixed by Newtonian particle motion; the spatial
metric requires additional relativistic information.
\end{solution}

\begin{solution}
For the timelike Killing field \(K=\partial_t\), the photon energy
\(-K_\mu k^\mu\) is conserved.  A static observer has
\(u^\mu=K^\mu/\sqrt{-K^2}\), hence
\(\omega=-u_\mu k^\mu=E_K/\sqrt{-g_{00}}\).  Therefore
\[
\frac{\Delta\omega}{\omega}=-\Delta\Phi=-gh
\]
to first order; an upward-traveling photon is redshifted.
\end{solution}

\begin{solution}
With \(\phi_{\rm inertial}=\phi+\Omega t\),
\[
\dd s^2=-(1-\Omega^2r^2)\dd t^2
+2\Omega r^2\dd t\,\dd\phi+\dd r^2+r^2\dd\phi^2+\dd z^2.
\]
The rotating time vector is null at \(r=|\Omega|^{-1}\).  Expanding the
geodesic equation at low speed gives
\(\ddot{\bm r}=-2\bm\Omega\times\dot{\bm r}
-\bm\Omega\times(\bm\Omega\times\bm r)\).
\end{solution}

\begin{solution}
The measured accelerations are
\(a_A=(1+\eta_A)g\) and \(a_B=(1+\eta_B)g\).  Hence
\[
\eta_{\rm E}=\frac{2(a_A-a_B)}{a_A+a_B}
\simeq\eta_A-\eta_B.
\]
If every material has the same ratio, it can be absorbed into the definition
of the gravitational field or of \(G\), so only composition dependence is
observable.
\end{solution}

\begin{solution}
Write
\[
x'^\rho=x^\rho+\frac12C^\rho{}_{\mu\nu}
(x-p)^\mu(x-p)^\nu .
\]
At \(p\),
\(\Gamma'^\rho{}_{\mu\nu}=\Gamma^\rho{}_{\mu\nu}-C^\rho{}_{\mu\nu}\);
choosing \(C=\Gamma(p)\) sets the connection to zero.  Both contain
\(n^2(n+1)/2\) independent coefficients for a torsion-free connection.  Since
\(\nabla_\rho g_{\mu\nu}=0\), vanishing \(\Gamma\) implies
\(\partial_\rho g_{\mu\nu}(p)=0\).
\end{solution}

\begin{solution}
Varying the geodesic equation through a one-parameter family gives
\[
\Dworldline{\Dworldline{s^\mu}}
=-R^\mu{}_{\nu\rho\sigma}u^\nu s^\rho u^\sigma
\]
for the curvature convention used here.  In local inertial coordinates this
is the directly measured relative coordinate acceleration at \(p\).  A
coordinate change can cancel first derivatives of the metric at one point,
but second derivatives combine into the tensor \(R^\mu{}_{\nu\rho\sigma}\)
and cannot be canceled when it is nonzero.
\end{solution}

\begin{solution}
The filled box has total inertial energy \(M_{\rm box}+E_\gamma\).  Universal
free fall requires the force to be
\((M_{\rm box}+E_\gamma)g\), so the acceleration remains \(g\).  If radiation
did not gravitate, the acceleration would instead be
\(M_{\rm box}g/(M_{\rm box}+E_\gamma)\), revealing the box's contents and
violating universality.
\end{solution}

\begin{solution}
For slow motion,
\[
\ddot x^i=-\Gamma^i{}_{00}
=-\frac12\partial_i(R_{0j0k}x^jx^k)
=-R^i{}_{0j0}x^j,
\]
the geodesic-deviation equation.  Outside a point mass the radial eigenvalue
is \(+2GM/r^3\) (stretching) and each transverse eigenvalue is
\(-GM/r^3\) (compression); their sum vanishes in vacuum.
\end{solution}
