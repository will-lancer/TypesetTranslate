\begin{exercises}

\begin{exercise}[Cosmological redshift]{Tong Cosmology, Cambridge Part III Sheet 1}
For radial null propagation in a Robertson--Walker spacetime, compare two
successive wavecrests and show that
\[
1+z=\frac{a(t_0)}{a(t_{\rm e})}.
\]
Derive the corresponding scaling of photon energy.
\end{exercise}

\begin{exercise}[Comoving, angular, and luminosity distances]{Tong Cosmology, Cambridge Part III Sheet 1}
For curvature \(k\), define
\[
\chi(z)=\int_0^z\frac{\dd z'}{H(z')},\qquad
S_k(\chi)=\{\sin\chi,\chi,\sinh\chi\}.
\]
Derive the transverse comoving distance, angular-diameter distance, and
luminosity distance, and prove Etherington reciprocity.
\end{exercise}

\begin{exercise}[Low-redshift Hubble law]{Cambridge Part III Cosmology, 2018 exam}
Expand the scale factor about the present and derive the luminosity-distance
series
\[
d_L(z)=\frac{z}{H_0}
\left[1+\frac12(1-q_0)z+\order(z^2)\right].
\]
State where spatial curvature first enters.
\end{exercise}

\begin{exercise}[Cosmological volume element]{Tong Cosmology, Cambridge Part III Sheet 2}
Derive the differential comoving volume per redshift and solid angle,
\[
\frac{\dd V_c}{\dd z\,\dd\Omega}
=\frac{D_M^2(z)}{H(z)}.
\]
Explain how the result changes if one asks for physical volume at emission.
\end{exercise}

\begin{exercise}[Tolman surface-brightness test]{Cambridge Part III Cosmology, 2013 exam}
For a source of fixed bolometric luminosity per proper emitting area, show
that observed surface brightness scales as \((1+z)^{-4}\).  Identify the four
powers of redshift.
\end{exercise}

\begin{exercise}[Euclidean number counts]{Tong Cosmology, Cambridge Part III Sheet 1}
In a static homogeneous Euclidean universe of identical sources, derive
\(N(>F)\propto F^{-3/2}\), where \(F\) is flux.  List three cosmological or
astrophysical effects that spoil this relation at large redshift.
\end{exercise}

\begin{exercise}[Curvature from distance triangles]{Cambridge Part III Cosmology, 2022 exam}
Show that the transverse distance satisfies
\[
\left[\frac{\dd D_M}{\dd\chi}\right]^2=1-kD_M^2.
\]
Explain how independent measurements of \(H(z)\) and \(D_M(z)\) give a
consistency test of Robertson--Walker geometry.
\end{exercise}

\begin{exercise}[Particle and event horizons]{MIT 8.962, Problem Set 13}
For \(a(t)\propto t^p\), determine when the particle-horizon distance
\(a(t)\int_0^t\dd t'/a(t')\) and event-horizon distance
\(a(t)\int_t^\infty\dd t'/a(t')\) are finite.
\end{exercise}

\begin{exercise}[Redshift drift]{Cambridge Part III Cosmology, 2024 exam}
A comoving source is observed at two nearby observer times.  Derive the
Sandage--Loeb drift
\[
\dot z=(1+z)H_0-H(z).
\]
Discuss its sign in a matter-dominated and a de Sitter universe.
\end{exercise}

\begin{exercise}[Steady-state counts]{Cambridge Part III Cosmology, 2009 exam}
In an exponentially expanding steady-state model \(a(t)=e^{H_0t}\), compute
the comoving distance and lookback time as functions of redshift.  Explain why
a constant physical density requires continuous matter creation.
\end{exercise}

\begin{solutions}

\begin{solution}
For a radial null ray,
\(\int_{t_{\rm e}}^{t_0}\dd t/a(t)\) is fixed by the comoving endpoints.
Varying the emission and reception times for adjacent crests gives
\(\delta t_0/a_0=\delta t_{\rm e}/a_{\rm e}\).  Therefore
\[
\frac{\nu_{\rm e}}{\nu_0}
=\frac{\delta t_0}{\delta t_{\rm e}}
=\frac{a_0}{a_{\rm e}}=1+z.
\]
Photon energy scales as \(E\propto\nu\propto a^{-1}\).
\end{solution}

\begin{solution}
With the curvature radius absorbed into \(S_k\),
\[
D_M=S_k(\chi),\qquad
d_A=\frac{D_M}{1+z},\qquad
d_L=(1+z)D_M.
\]
The first relation follows from the transverse metric.  Redshift reduces each
photon's energy and arrival rate by \(1+z\), giving
\(d_L=(1+z)^2d_A\), Etherington reciprocity.
\end{solution}

\begin{solution}
Write
\[
a(t_0-\Delta t)=a_0
\left[1-H_0\Delta t-\frac12q_0H_0^2\Delta t^2+\cdots\right]
\]
and expand both \(z\) and the null-distance integral, then eliminate
\(\Delta t\).  Multiplication by \(1+z\) gives the stated \(d_L\).
Spatial curvature first appears in the coefficient of \(z^3\).
\end{solution}

\begin{solution}
A shell has comoving volume
\(\dd V_c=D_M^2\dd\chi\,\dd\Omega\).  Since
\(\dd\chi=\dd z/H(z)\), the stated result follows.  Each physical length at
emission is smaller by \(a_{\rm e}/a_0=(1+z)^{-1}\), so
\(\dd V_{\rm phys,e}=\dd V_c/(1+z)^3\).
\end{solution}

\begin{solution}
Surface brightness is flux per observed solid angle:
\[
B_{\rm obs}=\frac{L/A_{\rm e}}{4\pi}
\frac{d_A^2}{d_L^2}
=\frac{B_{\rm e}}{(1+z)^4}.
\]
One power comes from photon-energy redshift, one from time dilation of the
arrival rate, and two from the relation between apparent angular area and
luminosity distance.
\end{solution}

\begin{solution}
Flux \(F=L/(4\pi r^2)\) implies
\(r_{\max}\propto F^{-1/2}\).  Homogeneity gives
\(N\propto r_{\max}^3\), hence \(N(>F)\propto F^{-3/2}\).
Expansion and curvature, redshift and \(K\)-corrections, source evolution,
absorption, and a luminosity distribution can all change the slope.
\end{solution}

\begin{solution}
Since \(D_M=S_k(\chi)\) and
\((S_k')^2=1-kS_k^2\), the identity follows.  Observations determine
\(\dd D_M/\dd z\) and \(H(z)\), hence
\[
k=\frac{1-[H(z)D_M'(z)]^2}{D_M^2(z)}.
\]
Robertson--Walker geometry requires this inferred curvature to be independent
of redshift.
\end{solution}

\begin{solution}
Near the big bang the particle-horizon integral behaves as
\(\int_0^t t'^{-p}\dd t'\), finite for \(p<1\) and divergent for
\(p\geq1\).  At late times the event-horizon integral
\(\int_t^\infty t'^{-p}\dd t'\) is finite for \(p>1\) and divergent for
\(p\leq1\).
\end{solution}

\begin{solution}
The fixed comoving separation obeys
\(\int_{t_e}^{t_0}\dd t/a(t)=\) constant, so
\(\dd t_e/\dd t_0=a_e/a_0=(1+z)^{-1}\).  Differentiating
\(1+z=a_0/a_e\) gives the stated drift.  For Einstein--de Sitter,
\(H=H_0(1+z)^{3/2}\), so the drift is negative for \(z>0\).  In de Sitter
\(H=H_0\), so \(\dot z=zH_0>0\).
\end{solution}

\begin{solution}
With \(a_0=1\), emission occurs at
\(t_e=t_0-H_0^{-1}\ln(1+z)\), so the lookback time is
\(H_0^{-1}\ln(1+z)\).  The comoving distance is
\[
\chi=\int_{t_e}^{t_0}e^{-H_0t}\dd t
=\frac{z}{H_0}
\]
after choosing \(a_0=1\).  Conserved dust would dilute as \(a^{-3}\);
maintaining constant physical density requires a source term
\(\dot\rho+3H_0\rho=3H_0\rho\).
\end{solution}
