% Exercise-edition support. This file emits no visible material by itself.

\newcounter{exercise}
\newcounter{solution}
\newcommand{\exercisechapter}{0}

\newcommand{\chapterexercises}[1]{%
  \renewcommand{\exercisechapter}{#1}%
  \setcounter{exercise}{0}%
  \InputIfFileExists{exercises/chapter#1.tex}{}{}%
}

\newenvironment{exercises}{%
  \clearpage
  \subsection*{Exercises}%
  \addcontentsline{toc}{subsection}{Exercises}%
  \setcounter{exercise}{0}%
}{%
  \par
}

\newenvironment{exercise}[2][]{%
  \refstepcounter{exercise}%
  \par\medskip
  \noindent\textbf{Exercise
    \exercisechapter.\arabic{exercise}}%
  \ifstrempty{#1}{}{\textbf{\ (#1)}}%
  \ifstrempty{#2}{}{%
    \par\nobreak\noindent
    {\footnotesize\textnormal{(#2)}}%
  }%
  \par\nobreak\smallskip\noindent
}{%
  \par\medskip
}

\newenvironment{solutions}{%
  \clearpage
  \subsection*{Solutions}%
  \addcontentsline{toc}{subsection}{Solutions}%
  \setcounter{solution}{0}%
}{%
  \par
}

\newenvironment{solution}[1][]{%
  \refstepcounter{solution}%
  \par\medskip
  \noindent\textbf{Solution
    \exercisechapter.\arabic{solution}}%
  \ifstrempty{#1}{}{\textbf{\ (#1)}}%
  \quad
}{%
  \par\medskip
}

% Audited exercise-edition interfaces.  The explicit stable identifier and
% display number make prompt--solution matching machine-checkable instead of
% depending on positional counters.  The legacy environments above remain
% available only while the provisional collection is being replaced.
\newenvironment{sourceexercise}[4]{%
  \setcounter{exercise}{#2}%
  \addtocounter{exercise}{-1}%
  \refstepcounter{exercise}%
  \def\sourceexerciseid{#1}%
  \label{exercise:#1}%
  \par\medskip
  \noindent\textbf{Exercise \exercisechapter.#2}%
  \ifstrempty{#3}{}{\textbf{\ (#3)}}%
  \ifstrempty{#4}{}{%
    \par\nobreak\noindent
    {\footnotesize\textnormal{(#4)}}%
  }%
  \par\nobreak\smallskip\noindent
}{%
  \par\medskip
}

\newenvironment{sourcesolution}[3]{%
  \setcounter{solution}{#2}%
  \addtocounter{solution}{-1}%
  \refstepcounter{solution}%
  \def\sourcesolutionid{#1}%
  \label{solution:#1}%
  \par\medskip
  \noindent\textbf{Solution \exercisechapter.#2}%
  \ifstrempty{#3}{}{\textbf{\ (#3)}}%
  \quad
}{%
  \par\medskip
}
