\begin{exercises}

\begin{exercise}[Post-Newtonian bookkeeping]{Cambridge Part III, 2011 exam}
For a bound weak-field system show that
\(v^2\sim U\sim p/\rho\sim\order(\epsilon)\), while
\(\partial_t\sim v\partial_i\).  Use this counting to determine the orders
needed for \(g_{00},g_{0\ii},g_{ij}\) in a first post-Newtonian equation of
motion.
\end{exercise}

\begin{exercise}[Point-mass metric in isotropic form]{MIT 8.962, Problem Set 8}
Transform the Schwarzschild metric to isotropic radius \(\varrho\) and expand
through \(U^2\), \(U=GM/\varrho\).  Read off the PPN parameters \(\beta\) and
\(\gamma\).
\end{exercise}

\begin{exercise}[Multipole expansion of a compact source]{Cambridge Part III, 2012 exam}
Expand the Newtonian potential of a localized density through quadrupole
order.  Show how the dipole term disappears in center-of-mass coordinates and
write the trace-free quadrupole contribution.
\end{exercise}

\begin{exercise}[Geodetic precession]{Cambridge Part III, 2022 exam}
A gyroscope follows a circular orbit of radius \(R\) in Schwarzschild
spacetime.  Parallel transport its spin for one revolution and show that the
precession relative to distant stars is
\[
\alpha=2\pi\left(1-\sqrt{1-\frac{3GM}{R}}\right).
\]
Give the weak-field limit.
\end{exercise}

\begin{exercise}[Frame dragging]{McGreevy, Problem Set 8}
In the far field of a slowly rotating body with angular momentum \(\bm J\),
take
\[
g_{0\ii}=-2G\frac{(\bm J\times\bm x)_i}{r^3}.
\]
Derive the leading Lense--Thirring precession of a stationary gyroscope.
\end{exercise}

\begin{exercise}[Perihelion shift in a PPN metric]{Cambridge Part III, 2019 exam}
Using the metric in Exercise 8.10, derive the secular perihelion advance and
show explicitly that it depends on the combination
\((2+2\gamma-\beta)/3\).
\end{exercise}

\begin{exercise}[Nordtvedt polarization]{Cambridge Part III, 2017 exam}
Suppose the ratio of gravitational to inertial mass of an extended body
depends on its gravitational binding energy,
\[
\frac{m_g}{m_i}=1+\eta_N\frac{E_g}{m_i}.
\]
Explain how the Sun's field polarizes the Earth--Moon orbit and identify the
PPN combination measured by \(\eta_N\).
\end{exercise}

\begin{exercise}[First post-Newtonian virial relation]{Cambridge Part III, 2016 exam}
For a stationary isolated system, use
\(\partial_\mu T^{\mu\nu}=0\) at leading order to prove
\(\int T^{ij}\dd^3x=\tfrac12\dd^2 I^{ij}/\dd t^2\), where
\(I^{ij}=\int T^{00}x^ix^j\dd^3x\).  State the time-averaged consequence for
a bound system.
\end{exercise}

\begin{exercise}[Conserved baryon density]{MIT 8.962, Problem Set 10}
For a perfect fluid define
\(\rho_\ast=\sqrt{-g}\,\rho_0u^0\), where \(\rho_0\) is rest-mass density.
Show that baryon conservation becomes
\[
\partial_t\rho_\ast+\partial_i(\rho_\ast v^i)=0.
\]
Why is \(\rho_\ast\) convenient in post-Newtonian theory?
\end{exercise}

\begin{exercise}[Brans--Dicke solar-system parameter]{Cambridge Part III, 2013 exam}
Linearize Brans--Dicke theory about a constant scalar background for a static
point mass.  Show that its PPN parameter is
\(\gamma=(1+\omega)/(2+\omega)\), and find the Einstein limit.
\end{exercise}

\begin{exercise}[Dragging inside a rotating shell]{Cambridge Part III, 2004 exam, Question 3}
A thin spherical shell of mass \(M\), radius \(R\), and small angular
velocity \(\bm\Omega\) is in the weak-field regime
\(\epsilon=GM/R\ll1\).  Solve the stationary linearized equations inside the
shell and show that the gravitomagnetic shift is linear in position.  Deduce
that local inertial axes rotate relative to infinity with
\[
\bm\omega_{\rm \dd rag}=\frac{4GM}{3R}\,\bm\Omega.
\]
Interpret the result as the interior Lense--Thirring effect.
\end{exercise}

\begin{solutions}

\begin{solution}
The virial estimate gives \(v^2\sim GM/R\sim U\); hydrostatic balance gives
\(p/\rho\sim U\).  Time variation is slower than spatial variation by \(v\).
To obtain acceleration through relative order \(\epsilon\), one needs
\[
g_{00}=-1+\order(\epsilon)+\order(\epsilon^2),\quad
g_{0\ii}=\order(\epsilon^{3/2}),\quad
g_{ij}=\delta_{ij}+\order(\epsilon).
\]
\end{solution}

\begin{solution}
The coordinates obey
\(r=\varrho(1+GM/2\varrho)^2\), giving
\[
\dd s^2
=-\left(\frac{1-U/2}{1+U/2}\right)^2\dd t^2
+\left(1+\frac U2\right)^4\dd\bm x^2.
\]
Thus
\(g_{00}=-1+2U-2U^2+\cdots\) and
\(g_{ij}=(1+2U+\cdots)\delta_{ij}\), so
\(\beta=\gamma=1\).
\end{solution}

\begin{solution}
For \(r\) outside the source,
\[
\Phi(\bm x)
=-G\left[\frac Mr+\frac{n_iD^i}{r^2}
+\frac{3n_in_j-\delta_{ij}}{2r^3}I^{ij}+\cdots\right].
\]
Here \(D^i=\int\rho x'^i\dd^3x'\), which vanishes at the center of mass.  With
\(Q^{ij}=\int\rho(3x'^ix'^j-r'^2\delta^{ij})\dd^3x'\), the quadrupole term is
\(-GQ^{ij}n_in_j/(2r^3)\).
\end{solution}

\begin{solution}
The circular geodesic has
\(\Omega^2=GM/R^3\).  Solving the parallel-transport equations over
\(\Delta t=2\pi/\Omega\) rotates the spin in the orbital plane by
\(2\pi\sqrt{1-3GM/R}\) relative to the radial direction.  Subtracting from a
full revolution gives the stated result.  For \(GM/R\ll1\),
\[
\alpha\simeq\frac{3\pi GM}{R}.
\]
\end{solution}

\begin{solution}
The gravitomagnetic field is the curl of the shift potential.  Parallel
transport of the spin gives
\[
\bm\Omega_{\rm LT}
=\frac{G}{r^3}\left[3\hat{\bm r}
(\bm J\cdot\hat{\bm r})-\bm J\right].
\]
It falls as \(r^{-3}\) and reverses with the source angular momentum.
\end{solution}

\begin{solution}
Expand the timelike geodesic action through \(U^2\) and \(Uv^2\), derive the
perturbed Binet equation, and retain the term resonant with the Keplerian
\(\cos\phi\).  The result is
\[
\Delta\varpi
=\frac{6\pi GM}{a(1-e^2)}
\left(\frac{2+2\gamma-\beta}{3}\right).
\]
The three pieces come respectively from nonlinear \(g_{00}\), spatial
curvature, and the velocity-dependent correction.
\end{solution}

\begin{solution}
Earth and Moon have different fractional gravitational binding energies, so
their accelerations toward the Sun differ by
\[
\frac{\Delta a}{a}
=\eta_N\left[
\left(\frac{E_g}{m}\right)_{\!E}
-\left(\frac{E_g}{m}\right)_{\!M}\right].
\]
This drives a synodic-period displacement of the lunar orbit toward the Sun.
In the standard PPN framework
\(\eta_N=4\beta-\gamma-3\); general relativity gives zero.
\end{solution}

\begin{solution}
Twice differentiate \(I^{ij}\), use
\(\partial_tT^{00}=-\partial_kT^{k0}\) and
\(\partial_tT^{0\ii}=-\partial_kT^{ki}\), and integrate by parts, dropping
surface terms.  This gives
\[
\ddot I^{ij}=2\int T^{ij}\dd^3x.
\]
The long-time average of a bounded \(I^{ij}\) has
\(\langle\ddot I^{ij}\rangle=0\), so
\(\langle\int T^{ij}\dd^3x\rangle=0\).
\end{solution}

\begin{solution}
Baryon conservation is
\(\nabla_\mu(\rho_0u^\mu)=0\), or
\(\partial_\mu(\sqrt{-g}\rho_0u^\mu)=0\).  Dividing spatial components by
\(u^0\) identifies \(v^i=u^i/u^0\) and yields the stated coordinate
continuity equation.  Its exactly conservative form permits use of the
familiar Newtonian fluid variables while all metric corrections are packaged
in \(\rho_\ast\).
\end{solution}

\begin{solution}
The scalar perturbation contributes an additional long-range \(1/r\) field.
Solving the linearized scalar and metric equations gives coefficients whose
ratio in \(g_{ij}\) and \(g_{00}\) is
\[
\gamma=\frac{1+\omega}{2+\omega}.
\]
As \(\omega\to\infty\), the scalar decouples and \(\gamma\to1\), the Einstein
value.
\end{solution}

\begin{solution}
The scalar part of the shell field is constant inside:
\(\Phi=-GM/R\).  The mass current
\(\rho\,\bm\Omega\times\bm r\) has only the \(\ell=1\) axial multipole, so
regularity at the center makes the interior vector potential linear in
\(\bm r\).  Matching it to the exterior dipole solution at \(r=R\) gives a
constant gravitomagnetic curl.  Substitution in the slow-motion geodesic
equation produces the Coriolis form
\[
\ddot{\bm r}=-2\bm\omega_{\rm \dd rag}\times\dot{\bm r}+\cdots,
\qquad
\bm\omega_{\rm \dd rag}=\frac{4GM}{3R}\bm\Omega.
\]
Thus a freely carried spatial triad inside the shell rotates with a fixed
fraction \(4GM/(3R)\) of the shell's angular velocity.  The effect vanishes
with \(G\) or \(M\) and reverses with the shell's rotation.
\end{solution}
