\begin{exercises}

\begin{exercise}[Olbers' paradox]{Tong Cosmology, Cambridge Part III Sheet 1}
In an infinite static Euclidean universe with uniform stellar luminosity
density, show that each radial shell contributes the same amount to the night
sky.  Explain separately how finite age, expansion, and absorption affect the
argument.
\end{exercise}

\begin{exercise}[Matter plus cosmological constant]{Tong Cosmology, Cambridge Part III Sheet 2}
For a flat universe with pressureless matter and positive \(\Lambda\), solve
the Friedmann equation to show
\[
a(t)\propto\sinh^{2/3}\left(\frac{3H_\Lambda t}{2}\right),
\qquad H_\Lambda=\sqrt{\Lambda/3}.
\]
Give the early- and late-time limits.
\end{exercise}

\begin{exercise}[Einstein static universe]{Cambridge Part III Cosmology, 2012 exam}
Find the radius and density of a closed static universe containing dust and a
cosmological constant.  Perturb the scale factor homogeneously and determine
the stability.
\end{exercise}

\begin{exercise}[A nonsingular bounce]{Cambridge Part III Cosmology, 2019 exam}
Use the Friedmann and acceleration equations to state the conditions at a
bounce \(H=0,\dot H>0\).  Show why a spatially flat bounce in Einstein gravity
requires violation of the null energy condition.
\end{exercise}

\begin{exercise}[Continuous creation in steady state]{Cambridge Part III Cosmology, 2009 exam}
Modify dust conservation to
\(\dot\rho+3H\rho=\mathcal C\).  Determine the creation rate
\(\mathcal C\) needed to keep \(\rho\) constant in exponential expansion and
interpret it as a particle-production rate.
\end{exercise}

\begin{exercise}[Brans--Dicke power-law cosmology]{McGreevy, Problem Set 8}
For a flat dust universe in Brans--Dicke theory, seek solutions
\(a\propto t^q\), \(\phi\propto t^s\).  Show that
\[
q=\frac{2(1+\omega)}{4+3\omega},
\qquad
s=\frac{2}{4+3\omega},
\]
and recover the Einstein dust limit.
\end{exercise}

\begin{exercise}[Bounds on a varying Newton constant]{Cambridge Part III, 2013 exam}
If the effective coupling is \(G_{\rm eff}\propto\phi^{-1}\), relate
\(\dot G/G\) to \(\dot\phi/\phi\).  For the power-law solution above,
determine its time dependence and explain qualitatively why nucleosynthesis
and solar-system clocks constrain it.
\end{exercise}

\begin{exercise}[Jordan and Einstein frames]{Tong, Cambridge Part III Sheet 3}
Starting from a scalar-tensor action
\(\int\sqrt{-g}[\phi R-\omega(\nabla\phi)^2/\phi]\), perform a conformal
transformation to make the Ricci term canonical.  Identify the resulting
scalar kinetic term and explain where the scalar coupling to matter moves.
\end{exercise}

\begin{exercise}[\(R+\alpha R^2\) as a scalar theory]{Cambridge Part III, 2024 exam}
Introduce an auxiliary field to rewrite
\(f(R)=R+\alpha R^2\) gravity as a scalar-tensor theory.  Find the scalar
potential before the conformal transformation and identify the stationary
point corresponding to Minkowski space.
\end{exercise}

\begin{exercise}[Quintessence fixed point]{Cambridge Part III Field Theory in Cosmology, 2022 exam}
For a canonical scalar with
\(V(\phi)=V_0e^{-\lambda\phi/M_{\rm Pl}}\) in a scalar-dominated flat
universe, find the constant equation-of-state power-law solution.  Determine
when it accelerates.
\end{exercise}

\begin{solutions}

\begin{solution}
A shell of radius \(r\) contains \(\propto r^2\dd r\) stars, each with flux
\(\propto r^{-2}\), so its contribution is proportional to \(\dd r\).
Integrating to infinity diverges.  A finite age cuts off the upper radius;
expansion redshifts energy and arrival rates and changes distances; ordinary
absorption alone merely heats the absorber, which reradiates in equilibrium
and does not resolve the paradox in an eternal static universe.
\end{solution}

\begin{solution}
With \(\rho_m=\rho_{m0}a^{-3}\),
\[
\dot a^2=H_\Lambda^2a^2+\frac{8\pi G\rho_{m0}}{3a}.
\]
Setting \(a^3\propto\sinh^2(3H_\Lambda t/2)\) solves the equation.  At early
times \(\sinh x\simeq x\), so \(a\propto t^{2/3}\); at late times
\(\sinh x\simeq e^x/2\), so \(a\propto e^{H_\Lambda t}\).
\end{solution}

\begin{solution}
For \(H=\ddot a=0\) with dust and \(k=+1\),
\[
\Lambda=4\pi G\rho,\qquad
\frac1{a_E^2}=4\pi G\rho=\Lambda.
\]
A homogeneous perturbation obeys
\(\delta\ddot a=\Lambda\,\delta a\), so one mode grows exponentially.  The
Einstein static universe is unstable to homogeneous expansion or contraction.
\end{solution}

\begin{solution}
At \(H=0\),
\[
\frac{k}{a_b^2}=\frac{8\pi G}{3}\rho_b,\qquad
\dot H_b=-4\pi G(\rho_b+p_b)+\frac{k}{a_b^2}>0.
\]
For \(k=0\), the first equation gives \(\rho_b=0\), while the second requires
\(\rho_b+p_b<0\).  This violates the null energy condition.
\end{solution}

\begin{solution}
For \(\dot\rho=0\) and constant \(H\),
\[
\mathcal C=3H\rho.
\]
If every created particle has mass \(m\), the number source is
\(\dot n+3Hn=\mathcal C/m=3Hn\): creation exactly replaces the particles
lost from a fixed physical density by volume expansion.
\end{solution}

\begin{solution}
Dust conservation gives \(\rho\propto a^{-3}\).  Substitution of
\(a\propto t^q\), \(\phi\propto t^s\) into the scalar and Friedmann equations
first gives \(3q+s=2\); solving the remaining algebraic relation yields the
stated \(q,s\).  As \(\omega\to\infty\), \(q\to2/3\) and \(s\to0\), so the
Einstein dust solution and constant \(G\) are recovered.
\end{solution}

\begin{solution}
Since \(G_{\rm eff}\propto\phi^{-1}\),
\[
\frac{\dot G}{G}=-\frac{\dot\phi}{\phi}=-\frac{s}{t}.
\]
The fractional change is largest at early times, shifting the expansion rate
and freeze-out during nucleosynthesis.  At late times it produces secular
drifts in orbital dynamics and clock comparisons, so both epochs bound
\(|s|\), and hence the scalar-tensor coupling.
\end{solution}

\begin{solution}
Choose
\(\widetilde g_{\mu\nu}=\phi\,g_{\mu\nu}/M_{\rm Pl}^2\).  Up to a boundary
term, the gravitational action becomes
\[
\int\sqrt{-\widetilde g}\left[
\frac{M_{\rm Pl}^2}{2}\widetilde R
-\frac{M_{\rm Pl}^2}{2}
\left(\omega+\frac32\right)(\widetilde\nabla\ln\phi)^2\right].
\]
A logarithmic field redefinition makes the scalar canonical.  Matter, which
was minimally coupled to \(g_{\mu\nu}\), now couples to the conformally
rescaled metric and hence directly to the scalar.
\end{solution}

\begin{solution}
Introduce \(\chi\) and write
\[
f(R)\longrightarrow f(\chi)+f'(\chi)(R-\chi)
=\Phi R-U(\Phi),
\]
where \(\Phi=1+2\alpha\chi\).  Solving
\(\chi=(\Phi-1)/(2\alpha)\) gives
\[
U(\Phi)=\frac{(\Phi-1)^2}{4\alpha}.
\]
Its stationary minimum is \(\Phi=1\), \(\chi=R=0\), corresponding to
Minkowski space.
\end{solution}

\begin{solution}
For the scalar-dominated fixed point,
\[
a(t)\propto t^{2/\lambda^2},\qquad
\phi(t)=\frac{2M_{\rm Pl}}{\lambda}\ln t+\text{constant}.
\]
The kinetic-to-potential ratio is constant and
\[
w_\phi=-1+\frac{\lambda^2}{3}.
\]
Acceleration requires \(w_\phi<-1/3\), hence \(\lambda^2<2\).
\end{solution}
