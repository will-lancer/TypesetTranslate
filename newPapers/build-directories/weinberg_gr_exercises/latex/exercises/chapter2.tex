\begin{exercises}

\begin{exercise}[Rapidity and successive boosts]{MIT 8.962, Problem Set 1}
For motion along the \(x\)-axis write \(v=\tanh\chi\).  Show that a boost through
rapidity \(\chi\) preserves \(-t^2+x^2\), that two collinear boosts add their
rapidities, and hence recover the relativistic velocity-addition law.
\end{exercise}

\begin{exercise}[Aberration and the moving sky]{McGreevy, Problem Set 1}
An isotropic distribution of photons is seen by an observer boosted with speed
\(\beta\) in the \(z\)-direction.  Derive the aberration formula relating
\(\theta\) and \(\theta'\), find \(\dd\Omega'/\dd\Omega\), and determine where
the photons concentrate as \(\beta\to1\).
\end{exercise}

\begin{exercise}[Invariant threshold energy]{Cambridge Part III, 2017 exam}
A proton of mass \(m\) strikes an identical proton initially at rest and
produces an additional particle of mass \(M\).  Using the invariant
\(s=-(p_1+p_2)^2\), find the minimum incident energy in the laboratory frame.
Compare it with the energy required in a symmetric collider.
\end{exercise}

\begin{exercise}[Electric and magnetic fields under a boost]{MIT 8.962, Problem Set 1}
Starting from the transformation of \(F_{\mu\nu}\), obtain the transformation
of \(\bm E\) and \(\bm B\) under a boost \(\bm v=v\hat{\bm x}\).  Apply the
result to a purely electric field \(E\hat{\bm z}\) and verify both field
invariants.
\end{exercise}

\begin{exercise}[Number current and proper density]{McGreevy, Problem Set 2}
A cold beam has proper number density \(n_0\) and four-velocity \(u^\mu\).
Show that \(J^\mu=n_0u^\mu\).  Determine the density and flux measured in a
frame in which the beam moves with velocity \(\bm v\), and prove that
\(\partial_\mu J^\mu=0\) gives the relativistic continuity equation.
\end{exercise}

\begin{exercise}[Perfect-fluid stress tensor]{Cambridge Part III, 2015 exam}
Use isotropy in the local rest frame to derive
\[
T^{\mu\nu}=(\rho+p)u^\mu u^\nu+p\eta^{\mu\nu}.
\]
For a fluid moving along \(x\), calculate \(T^{00}\), \(T^{0x}\), and
\(T^{xx}\), and interpret the pressure contribution to inertia.
\end{exercise}

\begin{exercise}[Constant proper acceleration]{McGreevy, Problem Set 3}
A particle moves in one spatial dimension with constant magnitude \(a\) of
four-acceleration.  Starting at rest at the origin, find \(t(\tau)\) and
\(x(\tau)\).  Show that the worldline is a hyperbola and identify its null
asymptotes.
\end{exercise}

\begin{exercise}[Thomas-Wigner rotation]{MIT 8.962, Problem Set 2}
Compose infinitesimal boosts \(\dd\bm v_1\) and \(\dd\bm v_2\) in different
directions.  Show to second order that their commutator is a rotation.  Deduce
the nonrelativistic Thomas-precession angular velocity for a particle with
velocity \(\bm v\) and acceleration \(\bm a\).
\end{exercise}

\begin{exercise}[Lorentz algebra]{Cambridge Part III, 2012 exam}
Define rotation generators \(J_i=\tfrac12\epsilon_{ijk}M^{jk}\) and boost
generators \(K_i=M^{0i}\).  Derive their commutators and show that
\(\bm J\cdot\bm K\) and \(\bm J^2-\bm K^2\) commute with every generator.
\end{exercise}

\begin{exercise}[Causal order of three events]{MIT 8.962, Problem Set 1}
Three events are observed in the order \(A,B,C\) in one inertial frame and
\(C,B,A\) in another.  Determine which pairwise separations must be spacelike.
Give a concrete set of \(1+1\)-dimensional coordinates for which a third
observer sees the order \(A,C,B\).
\end{exercise}

\begin{solution}
The boost is
\(t'=t\cosh\chi-x\sinh\chi\),
\(x'=x\cosh\chi-t\sinh\chi\), so
\(-t'^2+x'^2=-t^2+x^2\).  Matrix multiplication uses the hyperbolic addition
formulae and gives \(\chi_{12}=\chi_1+\chi_2\).  Therefore
\[
v_{12}=\tanh(\chi_1+\chi_2)
       =\frac{v_1+v_2}{1+v_1v_2}.
\]
\end{solution}

\begin{solution}
Boosting a null vector gives
\[
\cos\theta'=\frac{\cos\theta-\beta}{1-\beta\cos\theta},
\qquad
\dd\Omega'=\frac{\dd\Omega}{\gamma^2(1-\beta\cos\theta)^2}.
\]
Equivalently, an isotropic distribution in the unprimed frame has density per
primed solid angle proportional to
\(\gamma^{-2}(1+\beta\cos\theta')^{-2}\).  As \(\beta\to1\), almost the entire
sky is aberrated into a narrow cone about the direction of motion.
\end{solution}

\begin{solution}
In the laboratory \(s=2m^2+2mE\), where \(E\) is the incident proton's total
energy.  At threshold the final particles are mutually at rest in the
center-of-momentum frame, so \(s=(2m+M)^2\).  Thus
\[
E_{\min}=m+2M+\frac{M^2}{2m}.
\]
In a symmetric collider each beam needs only
\(E_{\rm beam}=m+\tfrac12M\); the laboratory target wastes energy in
center-of-mass motion.
\end{solution}

\begin{solution}
For a boost along \(x\),
\[
E'_x=E_x,\quad
\bm E'_\perp=\gamma(\bm E_\perp+\bm v\times\bm B),\qquad
B'_x=B_x,\quad
\bm B'_\perp=\gamma(\bm B_\perp-\bm v\times\bm E).
\]
Hence \(E'_z=\gamma E\) and \(B'_y=\gamma vE\) for the stated convention.
Direct substitution gives
\(\bm E'^2-\bm B'^2=E^2\) and
\(\bm E'\cdot\bm B'=0\).
\end{solution}

\begin{solution}
The only available four-vector is \(u^\mu\), and its rest-frame time component
must equal \(n_0\), so \(J^\mu=n_0u^\mu\).  Consequently
\[
n=J^0=\gamma n_0,\qquad \bm j=\gamma n_0\bm v=n\bm v.
\]
Then \(\partial_\mu J^\mu=0\) is
\(\partial_t n+\bm\nabla\cdot(n\bm v)=0\).
\end{solution}

\begin{solution}
Rest-frame isotropy gives
\(T^{\hat\mu\hat\nu}=\operatorname{diag}(\rho,p,p,p)\).  The covariant tensor
built from \(u^\mu\) and \(\eta^{\mu\nu}\) with these components is the stated
one.  For \(u^\mu=\gamma(1,v,0,0)\),
\[
T^{00}=(\rho+p)\gamma^2-p,\quad
T^{0x}=(\rho+p)\gamma^2v,\quad
T^{xx}=(\rho+p)\gamma^2v^2+p.
\]
Pressure contributes because work is required to maintain a moving stressed
volume.
\end{solution}

\begin{solution}
Writing \(u^\mu=(\cosh a\tau,\sinh a\tau)\) makes
\(u_\mu u^\mu=-1\) and the proper acceleration have magnitude \(a\).
Integration with the stated initial data yields
\[
t=\frac{\sinh a\tau}{a},\qquad
x=\frac{\cosh a\tau-1}{a}.
\]
Thus \((x+a^{-1})^2-t^2=a^{-2}\), with asymptotes
\(x+a^{-1}=\pm t\).
\end{solution}

\begin{solution}
The Lorentz algebra gives
\([K_i,K_j]=-\epsilon_{ijk}J_k\).  Hence two infinitesimal non-collinear boosts
differ in order by
\(\dd\bm\theta=-\tfrac12\,\dd\bm v_1\times\dd\bm v_2\).
Applied successively along a curved trajectory, this gives at low speed
\[
\bm\Omega_T=\frac12\,\bm a\times\bm v.
\]
The exact coefficient is
\(\gamma^2/(\gamma+1)\).
\end{solution}

\begin{solution}
Using
\([M_{\mu\nu},M_{\rho\sigma}]
=\eta_{\nu\rho}M_{\mu\sigma}-\eta_{\mu\rho}M_{\nu\sigma}
-\eta_{\nu\sigma}M_{\mu\rho}+\eta_{\mu\sigma}M_{\nu\rho}\),
one finds
\[
[J_i,J_j]=\epsilon_{ijk}J_k,\quad
[J_i,K_j]=\epsilon_{ijk}K_k,\quad
[K_i,K_j]=-\epsilon_{ijk}J_k
\]
up to the conventional factor of \(\ii\).  Substitution in the commutators
shows that \(\bm J\cdot\bm K\) and
\(\bm J^2-\bm K^2\) are Casimirs.
\end{solution}

\begin{solution}
Any pair whose order reverses must be spacelike, so \(A\)--\(C\) is certainly
spacelike; arranging the complete reversal also requires the relevant time
differences to be dominated by spatial separations.  One example is
\[
A=(0,0),\quad B=(1,3),\quad C=(2,5).
\]
The frame \(v=0\) sees \(ABC\); a sufficiently large positive boost sees
\(CBA\).  Since the transformed times are
\(t'=\gamma(t-vx)\), choosing an intermediate \(v\) between the two pairwise
crossing values produces \(ACB\).
\end{solution}
