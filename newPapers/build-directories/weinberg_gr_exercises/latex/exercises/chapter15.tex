\begin{exercises}

\begin{exercise}[Single-fluid Friedmann solutions]{Tong Cosmology, Cambridge Part III Sheet 1}
In a spatially flat universe filled by a perfect fluid with constant
\(w=p/\rho\neq-1\), solve the continuity and Friedmann equations for
\(\rho(a)\) and \(a(t)\).  Treat \(w=-1\) separately.
\end{exercise}

\begin{exercise}[Critical density and cosmic age]{Cambridge Part III Cosmology, 2011 exam}
Define \(\rho_c=3H^2/(8\pi G)\) and
\(\Omega_i=\rho_i/\rho_c\).  Derive the curvature sum rule and calculate
\(H_0t_0\) for flat matter domination and flat vacuum domination.
\end{exercise}

\begin{exercise}[Thermal densities]{Tong Cosmology, Cambridge Part III Sheet 3}
Evaluate the temperature scaling of number and energy densities for
relativistic bosons and fermions in equilibrium with negligible chemical
potential.  Derive the factor \(7/8\) for fermionic energy density.
\end{exercise}

\begin{exercise}[Recombination and the Saha equation]{Tong Cosmology, Cambridge Part III Sheet 3}
For \(p+e^-\leftrightarrow H+\gamma\), derive the Saha relation for the
ionization fraction \(X_e\).  Explain parametrically why recombination occurs
at a temperature far below the hydrogen binding energy.
\end{exercise}

\begin{exercise}[Weak-interaction freeze-out]{Cambridge Part III Cosmology, 2015 exam}
During radiation domination take
\(\Gamma_{\rm wk}\sim G_F^2T^5\) and
\(H\sim\sqrt{Gg_\ast}\,T^2\).  Estimate the freeze-out temperature and the
neutron-to-proton ratio immediately afterward.
\end{exercise}

\begin{exercise}[Primordial helium]{Tong Cosmology, Cambridge Part III Sheet 4}
Assuming essentially all surviving neutrons end in \({}^4{\rm He}\), express
the helium mass fraction \(Y_p\) in terms of the neutron-to-proton ratio
\(r=n/p\) at nucleosynthesis.  Evaluate it for \(r=1/7\).
\end{exercise}

\begin{exercise}[Jeans instability in an expanding universe]{Tong Cosmology, Cambridge Part III Sheet 4}
Derive the linear Newtonian density-contrast equation for a fluid with sound
speed \(c_s\):
\[
\ddot\delta+2H\dot\delta+
\left(\frac{c_s^2k^2}{a^2}-4\pi G\rho\right)\delta=0.
\]
Identify the Jeans wavenumber.
\end{exercise}

\begin{exercise}[Growth in matter domination]{Cambridge Part III Cosmology, 2013 exam}
Solve the pressureless perturbation equation in an Einstein--de Sitter
universe.  Find both modes and express the growing mode in terms of the scale
factor.
\end{exercise}

\begin{exercise}[Gauge-invariant curvature perturbation]{Cambridge Part III Field Theory in Cosmology, 2021 exam}
Define
\(\zeta=-\psi-H\delta\rho/\dot\rho\).  Show, using energy conservation, that
\(\dot\zeta\) vanishes on superhorizon scales for adiabatic perturbations,
neglecting anisotropic stress and gradient terms.
\end{exercise}

\begin{exercise}[Inflation and the horizon problem]{Tong Cosmology, Cambridge Part III Sheet 2}
Show that accelerated expansion is equivalent to
\(\ddot a>0\), \(w<-1/3\), and decreasing comoving Hubble radius
\((aH)^{-1}\).  Explain how this permits presently distant regions to have
been in causal contact.
\end{exercise}

\end{exercises}

\begin{solutions}

\begin{solution}
Continuity gives
\[
\rho=\rho_0a^{-3(1+w)}.
\]
The flat Friedmann equation then integrates to
\[
a(t)\propto t^{2/[3(1+w)]}
\]
for \(w\neq-1\).  For \(w=-1\), \(\rho\) is constant and
\(a(t)\propto e^{Ht}\) with \(H^2=8\pi G\rho/3\).
\end{solution}

\begin{solution}
The Friedmann equation gives
\[
1=\sum_i\Omega_i+\Omega_k,\qquad
\Omega_k=-\frac{k}{a^2H^2}.
\]
For flat dust, \(a\propto t^{2/3}\), so \(H_0t_0=2/3\).  Exact vacuum
domination has \(a\propto e^{Ht}\) and no finite big-bang age; the elapsed
age depends on the arbitrary origin of de Sitter time.
\end{solution}

\begin{solution}
With \(x=p/T\),
\[
\rho_b=\frac{\pi^2}{30}gT^4,\qquad
n_b=\frac{\zeta(3)}{\pi^2}gT^3.
\]
Replacing \((e^x-1)^{-1}\) by \((e^x+1)^{-1}\) changes the energy integral by
\(1-2^{-3}=7/8\), so
\(\rho_f=(7/8)(\pi^2/30)gT^4\); the fermion number factor is \(3/4\).
\end{solution}

\begin{solution}
Chemical equilibrium and Maxwell--Boltzmann densities give
\[
\frac{X_e^2}{1-X_e}
=\frac1{n_b}
\left(\frac{m_eT}{2\pi}\right)^{3/2}e^{-B_H/T}.
\]
Because the baryon-to-photon ratio is tiny, the high-energy tail contains
many photons per baryon.  The exponential must therefore be very small
before neutral hydrogen survives, so \(T_{\rm rec}\ll B_H\).
\end{solution}

\begin{solution}
Freeze-out occurs when \(\Gamma_{\rm wk}\sim H\):
\[
T_f\sim
\left(\frac{\sqrt{Gg_\ast}}{G_F^2}\right)^{1/3},
\]
numerically of order MeV.  Chemical equilibrium just before freeze-out gives
\[
\frac np\simeq e^{-(m_n-m_p)/T_f},
\]
followed by additional neutron beta decay before nucleosynthesis.
\end{solution}

\begin{solution}
If there are \(N_p\) protons and \(N_n=rN_p\) neutrons, helium nuclei consume
two neutrons and two protons.  Their total mass fraction is therefore
\[
Y_p=\frac{2r}{1+r}.
\]
For \(r=1/7\), \(Y_p=1/4\).
\end{solution}

\begin{solution}
Linearized continuity gives
\(\dot\delta+a^{-1}\bm\nabla\cdot\bm v=0\); Euler gives
\(\dot{\bm v}+H\bm v=-a^{-1}\bm\nabla\Phi
-c_s^2a^{-1}\bm\nabla\delta\); Poisson gives
\(\nabla^2\Phi=4\pi Ga^2\rho\delta\).  Combining them yields the stated
equation, with
\[
k_J^2=\frac{4\pi G\rho a^2}{c_s^2}.
\]
\end{solution}

\begin{solution}
For Einstein--de Sitter,
\(H=2/(3t)\) and \(4\pi G\rho=2/(3t^2)\).  Trying
\(\delta\propto t^p\) gives
\(p=2/3\) or \(p=-1\).  Thus
\[
\delta_{\rm grow}\propto t^{2/3}\propto a,\qquad
\delta_{\rm decay}\propto t^{-1}\propto a^{-3/2}.
\]
\end{solution}

\begin{solution}
The perturbed energy equation gives, after neglecting spatial gradients,
\[
\dot\zeta
=-\frac{H}{\rho+p}\,\delta p_{\rm nad},
\qquad
\delta p_{\rm nad}=\delta p-\frac{\dot p}{\dot\rho}\delta\rho.
\]
For adiabatic perturbations \(\delta p_{\rm nad}=0\), hence
\(\dot\zeta\simeq0\) outside the horizon.
\end{solution}

\begin{solution}
The acceleration equation gives
\(\ddot a/a=-4\pi G(\rho+3p)/3\), so positive acceleration requires
\(w<-1/3\).  Also
\[
\frac{\dd}{\dd t}(aH)^{-1}
=-\frac{\ddot a}{a^2H^2},
\]
so acceleration makes the comoving Hubble radius shrink.  Comoving scales
now outside it could then have begun inside it, where causal microphysics
could homogenize them, before inflation stretched them outward.
\end{solution}

\end{solutions}
