\begin{exercise}[Four-momentum on a Fock state]{I-CH02-002}
Write $\varepsilon=+1$ for bosons and $\varepsilon=-1$ for fermions, and assume
\[
  a(\mathbf p)a^\dagger(\mathbf q)
  -\varepsilon a^\dagger(\mathbf q)a(\mathbf p)
  =\delta^{D-1}(\mathbf p-\mathbf q),
  \qquad a(\mathbf p)\ket{0}=0.
\]
For
\[
  \ket{\mathbf p_1,\ldots,\mathbf p_k}
  =a^\dagger(\mathbf p_1)\cdots a^\dagger(\mathbf p_k)\ket{0},
  \qquad p_r^0=\omega_{\mathbf{p_r}},\quad p_r^2=-m^2,
\]
verify directly that
\[
  P^\mu=\int\dd^{D-1}\mathbf p\,p^\mu a^\dagger(\mathbf p)a(\mathbf p)
\]
acts as
\[
  P^\mu\ket{\mathbf p_1,\ldots,\mathbf p_k}
  =\left(\sum_{r=1}^k p_r^\mu\right)\ket{\mathbf p_1,\ldots,\mathbf p_k}.
\]
In the same calculation, give the precise distributional meaning of the claim
that $n_{\mathbf p}=a^\dagger(\mathbf p)a(\mathbf p)$ is the momentum-space
number density.
\end{exercise}
