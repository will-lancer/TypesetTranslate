\begin{exercise}[Why the Dirac adjoint contains $\gamma^0$]{I-CH05-002}
In $D=4$, let the Dirac matrices obey
\begin{equation*}
  \{\gamma^\mu,\gamma^\nu\}=-2\eta^{\mu\nu},
  \qquad
  (\gamma^\mu)^\dagger=\gamma^0\gamma^\mu\gamma^0,
  \qquad \eta=\operatorname{diag}(-1,+1,+1,+1).
\end{equation*}
A proper orthochronous Lorentz transformation acts by
$\psi'(x')=S(\Lambda)\psi(x)$, where
\begin{equation*}
  S(\Lambda)^{-1}\gamma^\mu S(\Lambda)
  =\Lambda^\mu{}_{\nu}\gamma^\nu.
\end{equation*}
Prove the pseudo-unitarity identity
\begin{equation*}
  S(\Lambda)^\dagger\gamma^0=\gamma^0S(\Lambda)^{-1}.
\end{equation*}
Use it to find the transformation law of
$\bar\psi=\psi^\dagger\gamma^0$ and to verify the Lorentz invariance of
\begin{equation*}
  \mathcal L=\bar\psi(-\ii\gamma^\mu\partial_\mu-m)\psi.
\end{equation*}
Then examine a boost and show directly why $\psi^\dagger$ cannot replace
$\bar\psi$ in this Lagrangian.
\end{exercise}
