\exerciseeditionsection{Solutions to Supplementary Problems}
\begin{enumerate}[leftmargin=2.8em,itemsep=1.2em]
\SupplementarySolution{1}{Velocity space as a hyperboloid}
Write \(u^a=(\cosh\chi,\sinh\chi\,\widehat{\bm n})\).  Restricting the
Minkowski metric to this unit hyperboloid gives
\[
 \dd\ell^2=\dd\chi^2+\sinh^2\chi\,\dd\Omega_2^2.
\]
For two velocities, \(-u_1\cdot u_2=\gamma_{\rm rel}=\cosh\chi_{\rm rel}\).
The hyperbolic geodesic distance is therefore \(\chi_{\rm rel}\).
\end{enumerate}
