\exerciseeditionsection{Solutions to Supplementary Problems}
\begin{enumerate}[leftmargin=2.8em,itemsep=1.2em]
\SupplementarySolution{1}{The Schwarzschild ISCO}
From \(\dot r^2+(1-2M/r)(1+L^2/r^2)=E^2\), the circularity conditions give
\[
 L^2=\frac{Mr^2}{r-3M},\qquad
 E=\frac{1-2M/r}{\sqrt{1-3M/r}}.
\]
The second radial derivative changes sign at \(r=6M\).  There
\(L=2\sqrt3M\), \(E=\sqrt{8/9}\), and the binding efficiency is
\(1-E\simeq0.0572\).
\end{enumerate}
