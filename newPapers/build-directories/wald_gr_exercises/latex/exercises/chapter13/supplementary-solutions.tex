\exerciseeditionsection{Solutions to Supplementary Problems}
\begin{enumerate}[leftmargin=2.8em,itemsep=1.2em]
\SupplementarySolution{1}{The transverse--traceless projector}
Let \(P_{ij}=\delta_{ij}-n_in_j\).  Then
\[
 \Lambda_{ij}{}^{kl}=P_i{}^kP_j{}^l-\frac12P_{ij}P^{kl},
 \qquad S^{\rm TT}_{ij}=\Lambda_{ij}{}^{kl}S_{kl}.
\]
Because \(P_{ij}n^j=0\) and \(P_i{}^i=2\), one obtains
\(n^iS^{\rm TT}_{ij}=0\) and \(\delta^{ij}S^{\rm TT}_{ij}=0\).  Applying
\(\Lambda\) twice changes nothing, so it is a projector onto the two
spin-2 polarizations.
\end{enumerate}
