\exerciseeditionsection{Solutions to Wald's Problems}
\ExerciseEditorialNote

\begin{enumerate}[leftmargin=2.8em,itemsep=1.2em]
\WaldSolution{1}
For an ingoing horizon crossing,
\(U=-\exp[-(t-r_*)/(4M)]\).  Kruskal coordinates are regular at the horizon.
A smooth timelike tangent cannot become tangent to the null horizon, so
\(\dd U/\dd\tau\) is finite and nonzero there.  Equal proper-time intervals
between emissions therefore give equal late-time intervals in \(U\).  At fixed
large receiver radius,
\[
 |U|=C e^{-t/(4M)}.
\]
It follows that both the photon arrival rate and the received frequency decay
as \(e^{-t/(4M)}\).

\WaldSolution{2}
\begin{enumerate}[label=\alph*),leftmargin=2em]
\item Along the world line,
\[
 u^b\nabla_b[w^a(mu_a+qA_a)]
 =m u^au^b\nabla_bw_a+q w^aF_{ab}u^b
   +q u^b\nabla_b(w^aA_a).
\]
The first term vanishes by Killing's equation.  The last two combine into
\(q u^b(\Lie_wA)_b\), which vanishes by hypothesis.

\item With \(\xi=\partial_t\), \(\psi=\partial_\phi\),
\[
 E=-\xi^a(mu_a+qA_a),\qquad
 L=\psi^a(mu_a+qA_a).
\]
On the equatorial plane these are
\(E=-mu_t+qe/r\) and \(L=mu_\phi+qae/r\).  Solving for
\(u^t,u^\phi\) and using \(u^au_a=-1\) gives
\[
 m^2r^4\dot r^2=
 [E(r^2+a^2)-aL-qer]^2
 -\Delta[m^2r^2+(L-aE)^2].
\]
Thus \(\tfrac12\dot r^2+\mathcal U_q(r)=0\), with \(\mathcal U_q\) equal to minus the
right side divided by \(2m^2r^4\).
\end{enumerate}

\WaldSolution{3}
For a static particle, the conserved energy is
\[
 E=m\sqrt{1-2M/r+e^2/r^2}+\frac{qe}{r}.
\]
At the horizon, absorption obeys
\(\delta M-\Phi_H\delta e\geq0\), where \(\Phi_H=e/r_+\).
Introduce the irreducible mass
\[
 M_{\mathrm{irr}}=\frac{r_+}{2},\qquad
 M=M_{\mathrm{irr}}+\frac{e^2}{4M_{\mathrm{irr}}}.
\]
Reversible lowering saturates the inequality and holds \(M_{\mathrm{irr}}\)
fixed.  Complete discharge can extract at most
\[
 E_{\mathrm{ext}}=M-M_{\mathrm{irr}}
 =\frac12\left(M-\sqrt{M^2-e^2}\right).
\]
The extremal limit is \(M/2\).  Since \(A=16\pi M_{\mathrm{irr}}^2\), the
area theorem supplies the same bound.

\WaldSolution{4}
For each initial hole,
\[
 M_{\mathrm{irr},i}^2=\frac12
 \left(M_i^2+\sqrt{M_i^4-J_i^2}\right).
\]
Put \(S=M_{\mathrm{irr},1}^2+M_{\mathrm{irr},2}^2\).  The area theorem
requires the final irreducible mass to obey \(M_{\mathrm{irr},f}^2\geq S\),
while axisymmetry gives \(J_f=J_1+J_2\).  The Christodoulou relation then gives
\[
 M_f\geq\left(S+\frac{(J_1+J_2)^2}{4S}\right)^{1/2}.
\]
Consequently
\[
 E_{\mathrm{rad}}\leq M_1+M_2-
 \left(S+\frac{(J_1+J_2)^2}{4S}\right)^{1/2}.
\]
Antiparallel spins reduce \(|J_1+J_2|\) and increase this upper bound.

\WaldSolution{5}
\begin{enumerate}[label=\alph*),leftmargin=2em]
\item Cartan's identity gives \(\Lie_wF=\dd(i_wF)+i_w\dd F\).  Since
\(\dd F=0\), its component form, with the stated antisymmetrization
convention, is
\[
 \Lie_wF_{ab}=-2\nabla_{[a}(F_{b]c}w^c).
\]

\item On the horizon \(\chi^a=\xi^a+\Omega_H\psi^a\).  For a mode
\(e^{-i\omega t+im\phi}\), part (a) relates the contractions with
\(\xi^a\), \(\psi^a\), and \(\chi^a\).  After averaging over a period, the
horizon flux has the form
\[
 \langle J^a\rangle\dd\Sigma_a
 =C\,\omega(\omega-m\Omega_H)|\mathcal A|^2\dd v\dd A,
 \qquad C>0.
\]
It is negative for \(0<\omega<m\Omega_H\), which is precisely the
superradiant regime.
\end{enumerate}
\end{enumerate}
