\exerciseeditionsection{Solutions to Supplementary Problems}
\begin{enumerate}[leftmargin=2.8em,itemsep=1.2em]
\SupplementarySolution{1}{The extremal Reissner--Nordstr\"om throat}
For \(|Q|=M\), \(f(r)=(1-M/r)^2\).  The proper distance is
\[
 \ell=\int_{M+\epsilon}^{r_0}\frac{\dd r}{1-M/r}
 =(r_0-M-\epsilon)+M\ln\frac{r_0-M}{\epsilon}.
\]
It diverges logarithmically as \(\epsilon\to0\), although the horizon's
areal radius remains \(M\).
\end{enumerate}
