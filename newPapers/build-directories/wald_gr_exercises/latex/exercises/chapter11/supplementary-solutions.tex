\exerciseeditionsection{Solutions to Supplementary Problems}
\begin{enumerate}[leftmargin=2.8em,itemsep=1.2em]
\SupplementarySolution{1}{Noether current and Komar charge}
For fields \(\Phi\),
\[
 \delta_\xi L=E_\Phi\Lie_\xi\Phi+\dd\Theta(\Phi,\Lie_\xi\Phi)
 =\dd(\xi\mathbin{\cdot}L).
\]
Thus on shell
\(J_\xi=\Theta(\Phi,\Lie_\xi\Phi)-\xi\mathbin{\cdot}L\) is closed.  The
local Poincar\'e lemma gives \(J_\xi=\dd Q_\xi\).  In Einstein gravity,
\(Q_\xi\) is proportional to \(\star\dd\xi\), whose integral is the Komar
charge.
\end{enumerate}
