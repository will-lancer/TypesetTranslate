\exerciseeditionsection{Solutions to Supplementary Problems}
\begin{enumerate}[leftmargin=2.8em,itemsep=1.2em]
\SupplementarySolution{1}{Particle and event horizons in power-law cosmology}
Near the big bang, \(\int_0^t t'^{-p}\dd t'\) converges for \(p<1\), giving
\(d_{\rm particle}=t/(1-p)\).  The future integral converges for \(p>1\),
giving \(d_{\rm event}=t/(p-1)\).  The borderline \(p=1\) diverges
logarithmically in both cases.
\end{enumerate}
