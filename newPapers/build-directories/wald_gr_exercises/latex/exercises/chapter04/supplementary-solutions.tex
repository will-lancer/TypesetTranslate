\exerciseeditionsection{Solutions to Supplementary Problems}
\begin{enumerate}[leftmargin=2.8em,itemsep=1.2em]
\SupplementarySolution{1}{A conserved current from a Killing field}
Symmetry and Killing's equation give
\[
 \nabla_aJ^a=(\nabla_aT^{ab})\xi_b+T^{ab}\nabla_a\xi_b
 =T^{ab}\nabla_{(a}\xi_{b)}=0.
\]
Stokes' theorem makes \(\int_\Sigma J^a n_a\dd V\) independent of \(\Sigma\)
when the two hypersurfaces bound a region and the flux through any timelike or
null side boundary vanishes.
\end{enumerate}
