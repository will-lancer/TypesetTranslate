\exerciseeditionsection{Solutions to Wald's Problems}
\ExerciseEditorialNote

\begin{enumerate}[leftmargin=2.8em,itemsep=1.2em]
\WaldSolution{1}
Taking the divergence of \(\nabla_bF^{ab}=4\pi j^a\) gives
\[
 4\pi\nabla_aj^a=\nabla_a\nabla_bF^{ab}
 =\frac12[\nabla_a,\nabla_b]F^{ab}.
\]
The commutator produces contractions of the Ricci tensor with \(F^{ab}\).
Since \(R_{ab}\) is symmetric and \(F^{ab}\) is antisymmetric, those
contractions vanish.  Hence \(\nabla_aj^a=0\).

\WaldSolution{2}
\begin{enumerate}[label=\alph*),leftmargin=2em]
\item Contracting two volume forms gives
\[
 \epsilon^{a_1\ldots a_pb_1\ldots b_{n-p}}
 \epsilon_{a_1\ldots a_pc_1\ldots c_{n-p}}
 =(-1)^s p!(n-p)!\,
 \delta^{[b_1}_{c_1}\cdots\delta^{b_{n-p}]}_{c_{n-p}}.
\]
Using this identity twice in the definition of the Hodge dual, and moving a
block of \(p\) indices past a block of \(n-p\) indices, gives
\[
 \star\star\alpha=(-1)^{s+p(n-p)}\alpha.
\]

\item For a two-form,
\((\dd F)_{abc}=3\nabla_{[a}F_{bc]}\), so the homogeneous Maxwell equation is
\(\dd F=0\).  Using \(\nabla_a\epsilon_{bcde}=0\), the sourced equation is
equivalent to
\(\dd\star F=4\pi\star j\).  Stokes's theorem then gives the displayed
curved-spacetime Gauss law.

\item In four-dimensional Lorentz signature,
\[
 T_{ab}=\frac{1}{8\pi}
 \left(F_{ac}F_b{}^c+(\star F)_{ac}(\star F)_b{}^c\right).
\]
A duality rotation acts as an ordinary rotation on the ordered pair
\((F,\star F)\).  The sum of the two quadratic terms is its Euclidean norm,
so it is unchanged and \(\widetilde T_{ab}=T_{ab}\).
\end{enumerate}

\WaldSolution{3}
\begin{enumerate}[label=\alph*),leftmargin=2em]
\item To first order in both \(h_{ab}\) and the test-body velocity,
\[
 a^i=-\Gamma^i{}_{00}-2\Gamma^i{}_{0j}v^j.
\]
With \(A_a=-\tfrac14\bar h_{ab}t^b\), negligible source time derivatives,
and vanishing spatial components of \(\bar h_{ab}\), substitution of the
linearized Christoffel symbols gives
\[
 -\Gamma^i{}_{00}=-E^i,qquad
 -2\Gamma^i{}_{0j}v^j=-4(\mathbf v\mathbin\times\mathbf B)^i.
\]
Thus \(\mathbf a=-\mathbf E-4\mathbf v\mathbin\times\mathbf B\).

\item The shell's surface mass current is
\(\mathbf K=\sigma\,\boldsymbol\omega\mathbin\times\mathbf r\), with
\(\sigma=M/(4\pi R^2)\).  The magnetostatic integral for the gravitational
vector potential is the electromagnetic shell integral with charge current
replaced by mass current.  Its interior dipole term is linear in \(\mathbf r\):
\[
 \mathbf A=\frac{M}{3R}
 \boldsymbol\omega\mathbin\times\mathbf r.
\]
Therefore
\[
 \mathbf E=0,qquad
 \mathbf B=\boldsymbol\nabla\mathbin\times\mathbf A
 =\frac{2M}{3R}\boldsymbol\omega.
\]

\item At the center, parallel propagation and \(S^0=0\) give
\[
 \frac{\dd S^i}{\dd t}=-\Gamma^i{}_{0j}S^j
 =2(\mathbf B\mathbin\times\mathbf S)^i.
\]
Hence
\[
 \frac{\dd\mathbf S}{\dd t}=\boldsymbol\Omega\mathbin\times\mathbf S,
 \qquad
 \boldsymbol\Omega=2\mathbf B
 =\frac43\frac{M}{R}\boldsymbol\omega.
\]
\end{enumerate}

\WaldSolution{4}
Expand the inverse metric and connection as
\[
 g^{ab}=\eta^{ab}-h^{ab}+h^a{}_ch^{cb}+\order(h^3),
\]
\[
 \Gamma^{(1)c}{}_{ab}
 =\frac12\eta^{cd}(\partial_ah_{bd}+\partial_bh_{ad}-\partial_dh_{ab}),
 \qquad
 \Gamma^{(2)c}{}_{ab}
 =-\frac12h^{cd}(\partial_ah_{bd}+\partial_bh_{ad}-\partial_dh_{ab}).
\]
In
\(R_{ab}=\partial_c\Gamma^c{}_{ab}-\partial_a\Gamma^c{}_{cb}
+\Gamma^c{}_{cd}\Gamma^d{}_{ab}-\Gamma^c{}_{ad}\Gamma^d{}_{cb}\),
retain \(\partial\Gamma^{(2)}\) and
\(\Gamma^{(1)}\Gamma^{(1)}\).  Product rules, index relabeling, and one
integration-by-parts rearrangement give
\[
\begin{aligned}
R^{(2)}_{ab}
={}&\frac12h^{cd}\partial_a\partial_bh_{cd}
-h^{cd}\partial_c\partial_{(a}h_{b)d}
+\frac14(\partial_ah_{cd})\partial_bh^{cd}\\
&+(\partial^dh^c{}_b)\partial_{[d}h_{c]a}
+\frac12\partial_d(h^{dc}\partial_ch_{ab})
-\frac14(\partial^ch)\partial_ch_{ab}\\
&-\left(\partial_dh^{cd}-\frac12\partial^ch\right)
\partial_{(a}h_{b)c},
\end{aligned}
\]
which is Eq.~\eqref{eq:4.4.51}.

\WaldSolution{5}
For each fixed \(b\), conservation and the stated Poincar\'e-lemma result
produce \(W_{cab}=W_{[ca]b}\) such that
\[
 T_{ab}=\partial^cW_{cab}.
\]
Symmetry of \(T_{ab}\) implies \(\partial^cW_{c[ab]}=0\).  Apply the same
lemma to this divergence-free part and absorb the resulting exact term into
\(W\).  One may thereby choose a representative satisfying the cyclic
identity \(W_{[cab]}=0\).  Applying the lemma once more to the two-form in
the indices \(c,a\) gives
\[
 W_{cab}=\partial^dU_{acbd},
\]
where antisymmetrization makes
\(U_{acbd}=U_{[ac]bd}=U_{ac[bd]}\).  Replacing \(U\) by
\(\tfrac12(U_{acbd}+U_{bdac})\) preserves \(W\) up to an exact term whose
double divergence vanishes, and imposes pair exchange symmetry.  Consequently
\[
 T_{ab}=\partial^c\partial^dU_{acbd},
 \qquad U_{acbd}=U_{[ac][bd]}=U_{bdac}.
\]

\WaldSolution{6}
\begin{enumerate}[label=\alph*),leftmargin=2em]
\item Contracting the first displayed definition on \(a,b\), the first term
is canceled by the epsilon--epsilon term after use of
Eq.~\eqref{eq:B.2.13}; every remaining contraction is a trace of the Weyl
tensor.  Since \(C^a{}_{bac}=0\),
\(T^a{}_{acd}=0\).

\item In vacuum the Weyl tensor equals the Riemann tensor.  The contracted
Bianchi identity gives
\(\nabla^aC_{abcd}=0\), while the uncontracted identity gives
\(\nabla_{[a}C_{bc]de}=0\).  Differentiate the two quadratic terms in
\(T_{abcd}\).  Terms in which the derivative hits the first Weyl factor
vanish by the contracted identity; use the uncontracted identity and pair
symmetry to cycle the remaining derivative among the three symmetric free
indices.  The terms cancel, yielding
\(\nabla^aT_{abcd}=0\).
\end{enumerate}

\WaldSolution{7}
\begin{enumerate}[label=\alph*),leftmargin=2em]
\item The second-order Bianchi identity and the linearized vacuum equation
imply \(\partial^at_{ab}=0\).  Integrate this equation over the slab bounded
by \(\Sigma\), its time translate \(\Sigma'\), and a large timelike cylinder.
The cylinder term vanishes under the stated falloff.  The two spacelike terms
are therefore equal, so \(E_\Sigma=E_{\Sigma'}\).

\item Choose \(\xi'_a\) to agree with \(\xi_a\) near \(\Sigma\) and vanish
near \(\Sigma'\), with bounded spatial support throughout.  On \(\Sigma\),
\[
 E[h+2\partial\xi]=E[h+2\partial\xi'].
\]
Part (a) moves the latter energy to \(\Sigma'\), where \(\xi'=0\), giving
\(E[h+2\partial\xi']=E[h]\).  Hence the compactly supported gauge
transformation leaves \(E\) unchanged.
\end{enumerate}

\WaldSolution{8}
Let \(s=A\cos\omega t\) be the relative displacement of the two masses, so
their positions are \(x=\pm s/2\).  The reduced mass is \(M/2\) and
\(\omega^2=2K/M\).  Wald's quadrupole includes the factor three in
Eq.~\eqref{eq:4.4.48}.  Substitution in Eq.~\eqref{eq:4.4.58} gives
\[
 \langle P\rangle=\frac{4}{15}M^2A^4\omega^6.
\]
During one period,
\[
 E_{\rm rad}=\frac{8\pi}{15}M^2A^4\omega^5.
\]
The mechanical oscillation energy is
\(E_{\rm osc}=\tfrac12KA^2=\tfrac14M\omega^2A^2\), and therefore
\[
 \boxed{\frac{E_{\rm rad}}{E_{\rm osc}}
 =\frac{128\pi}{15}E_{\rm osc}\,\omega}
\]
in geometrized units.  Restoring constants replaces the last product by
\(G E_{\rm osc}\omega/c^5\).  A nonzero equilibrium separation adds the
quadrupole term obtained by replacing \(s\) with that separation plus the
oscillatory displacement.

\WaldSolution{9}
Let \(\Omega\) be the orbital angular frequency.  For equal masses at radius
\(R\), Newtonian balance and the orbital energy are
\[
 \Omega^2=\frac{M}{4R^3},
 \qquad E=-\frac{M^2}{4R}.
\]
The trace-free quadrupole has frequency \(2\Omega\), and
Eq.~\eqref{eq:4.4.58} gives
\[
 P=\frac{128}{5}M^2R^4\Omega^6.
\]
Using \(\dot E=-P\) and differentiating \(E(\Omega)\) yields
\[
 \boxed{\dot\Omega
 =\frac{96}{5}\,2^{-1/3}M^{5/3}\Omega^{11/3}}.
\]
With units restored, the right side is multiplied by \(G^{5/3}/c^5\).
The positive sign is the inspiral chirp: loss of orbital energy raises the
frequency.
\end{enumerate}
