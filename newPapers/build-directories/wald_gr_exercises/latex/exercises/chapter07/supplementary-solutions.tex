\exerciseeditionsection{Solutions to Supplementary Problems}
\begin{enumerate}[leftmargin=2.8em,itemsep=1.2em]
\SupplementarySolution{1}{Static versus stationary geometry}
Frobenius' theorem makes the displayed condition equivalent to the existence
of hypersurfaces orthogonal to \(\xi^a\).  Choose spatial coordinates on one
such hypersurface and propagate them along the Killing flow.  Orthogonality
then gives \(g_{0i}=0\), while the Killing property gives
\(\partial_0g_{\mu\nu}=0\).  These are local static coordinates.
\end{enumerate}
