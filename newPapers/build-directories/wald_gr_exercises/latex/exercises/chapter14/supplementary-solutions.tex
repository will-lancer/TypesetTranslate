\exerciseeditionsection{Solutions to Supplementary Problems}
\begin{enumerate}[leftmargin=2.8em,itemsep=1.2em]
\SupplementarySolution{1}{A single-mode squeezed vacuum}
The cross commutators vanish, while
\([b,b^\dagger]=\cosh^2r-\sinh^2r=1\).  Inverting gives
\(a=\cosh r\,b+\sinh r\,b^\dagger\).  Therefore
\[
 \langle0_b|a^\dagger a|0_b\rangle
 =\sinh^2r\,\langle0_b|bb^\dagger|0_b\rangle
 =\sinh^2r.
\]
This is the one-mode version of particle production by Bogoliubov mixing.
\end{enumerate}
