\exerciseeditionsection{Solutions to Wald's Problems}
\ExerciseEditorialNote

\begin{enumerate}[leftmargin=2.8em,itemsep=1.2em]
\WaldSolution{1}
\begin{enumerate}[label=\alph*),leftmargin=2em]
\item The sum is finite on \(V\).  Sesquilinearity and conjugate symmetry are
termwise, while \((a,a)=\sum_i|a_i|^2>0\) for \(a\ne0\).

\item Let \(a^{(n)}=(1,1/2,\ldots,1/n,0,\ldots)\).  Its tails have squared
norm \(\sum_{i=m+1}^n i^{-2}\), so it is Cauchy.  Its only possible limit is
\((1,1/2,1/3,\ldots)\), which has infinite support and does not belong to
\(V\).  Thus \(V\) is incomplete.

\item If \(a^{(n)}\) is Cauchy in \(\ell_2\), each coordinate is Cauchy in
\(\mathbb C\); call its limit \(a_i\).  For every \(N\),
\(\sum_{i=1}^N|a_i|^2\) is bounded by the common Cauchy bound, so monotone
convergence gives \(a\in\ell_2\).  Choosing a fixed large tail and then taking
coordinate limits shows \(\|a^{(n)}-a\|_2\to0\).  Hence \(\ell_2\) is complete.
\end{enumerate}

\WaldSolution{2}
\begin{enumerate}[label=\alph*),leftmargin=2em]
\item Apply the definitions to a finite symmetric tensor
\(\psi_n\in\mathcal H^{\otimes_s n}\).  Two annihilations or two creations
commute by symmetrization.  Moving one annihilation through one creation leaves
one contraction, giving
\[
 [a(\overline\sigma),a^\dagger(\rho)]\psi_n
 =(\sigma,\rho)\psi_n.
\]
The finite-particle subspace is dense, so the stated commutators hold on their
natural common domain.

\item Decompose Fock space using
\(\mathcal H=\mathbb C\sigma\oplus\sigma^\perp\).  On a vector with exactly
\(k\) factors equal to normalized \(\sigma\), annihilation supplies
\(\sqrt{k}\) and creation restores the factor with another \(\sqrt{k}\).
Thus \(N(\sigma)\) has eigenvalue \(k\).
\end{enumerate}

\WaldSolution{3}
Propagate two out-solutions to the past and evaluate their conserved
Klein--Gordon product there.  Orthogonality of positive and negative frequency
parts, together with the negative sign of the product on the latter, gives
\[
 A^\dagger A-B^\dagger B=I,qquad
 B^\dagger\overline A=A^\dagger\overline B.
\]
Doing the same with two in-solutions gives the corresponding \(C,D\) relations.
Evaluating the product of one propagated in-solution and one propagated
out-solution on opposite asymptotic surfaces gives
\[
 A^\dagger=C,qquad B^\dagger=-\overline D.
\]
Substitution in the mixed identity gives
\((\overline D\,\overline C^{-1})^\dagger=
\overline{\overline D\,\overline C^{-1}}\), so \(E\) is symmetric.

\WaldSolution{4}
For a conformally invariant field, rescaling by its conformal weight maps every
solution on \(g_{ab}=\Omega^2\eta_{ab}\) to a Minkowski solution.  The asymptotic
time translations select the same positive-frequency subspace under this map.
Positive-frequency in-data therefore evolve entirely into positive-frequency
out-data.  The Bogoliubov mixing operators \(B\) and \(D\) vanish, and particle
creation is zero.

\WaldSolution{5}
\begin{enumerate}[label=\alph*),leftmargin=2em]
\item The inverse scattering transformation gives
\[
 a_{\mathrm{out}}(\overline\sigma)=
 a_{\mathrm{in}}(\overline{A\sigma})+
 a_{\mathrm{in}}^\dagger(B\sigma)
\]
with the conventions of the text.  Acting on the in-vacuum leaves only the
creation term.  The canonical commutator then yields
\(\langle N(\sigma)\rangle=\|B\sigma\|^2\).

\item Equation~\eqref{eq:14.3.6} gives the relative horizon-mode amplitude
\(r=e^{-\pi\omega/\kappa}\).  Combining it with
\(A^\dagger A-B^\dagger B=I\) and the transmission probability
\(|t_\omega|^2\) gives
\[
 \langle N_\omega\rangle=
 \frac{|t_\omega|^2r^2}{1-r^2}
 =\frac{|t_\omega|^2}{e^{2\pi\omega/\kappa}-1},
\]
which is Eq.~\eqref{eq:14.3.7}.
\end{enumerate}

\WaldSolution{6}
Let \(|2_f\rangle\) be a normalized two-particle wave-packet state and set
\[
 |\Psi\rangle=(1+|\epsilon|^2)^{-1/2}
 (|0\rangle+\epsilon|2_f\rangle).
\]
The normal-ordered local energy density contains a vacuum--two-particle
interference term linear in \(\epsilon\).  At a point where its coefficient is
nonzero, choose the phase of \(\epsilon\) to make it negative.  For sufficiently
small \(|\epsilon|\), this linear term dominates the positive quadratic term on
an open neighborhood.  The total Hamiltonian has no vacuum--two-particle matrix
element and gives \(\langle H\rangle=2\omega_f|\epsilon|^2/(1+|\epsilon|^2)\geq0\).

\WaldSolution{7}
\begin{enumerate}[label=\alph*),leftmargin=2em]
\item In Planck units, with \(a=\pi^2/15\),
\[
 S'(M)=4\pi M^2+\frac43(aV)^{1/4}(E-M)^{3/4}.
\]

\item Stationarity is equivalent to equal black-hole and radiation
temperatures and gives
\[
 (8\pi)^4M^4(E-M)=aV.
\]
The left side has maximum at \(M=4E/5\), so equilibrium exists iff
\[
 E^5\geq\frac{3125}{256}\frac{aV}{(8\pi)^4}.
\]
For \(V=1\,\mathrm m^3\), this is approximately
\(E_{\min}=1.69\times10^{29}\,\mathrm J\), or
\(1.88\times10^{12}\,\mathrm{kg}\) of mass-energy.  At a stationary point,
\[
 S''(M)=2\pi\frac{4E-5M}{E-M}.
\]
The root below \(4E/5\) is unstable, and the root above it is stable.

\item Compare the endpoint entropies
\(S_{\mathrm{BH}}\simeq4\pi E^2\) and
\(S_{\mathrm{rad}}=\tfrac43(aV)^{1/4}E^{3/4}\).  A black-hole configuration
is globally favored, up to the neglected finite-volume corrections, when
\[
 E^5\gtrsim\frac{aV}{(3\pi)^4}.
\]
\end{enumerate}
\end{enumerate}
