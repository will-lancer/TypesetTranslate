\exerciseeditionsection{Solutions to Supplementary Problems}
\begin{enumerate}[leftmargin=2.8em,itemsep=1.2em]
\SupplementarySolution{1}{Codazzi and the momentum constraint}
Project one normal and three tangential indices in the spacetime Riemann
tensor.  The Codazzi equation is
\(h^a{}_ih^b{}_jh^c{}_kn^dR_{abcd}=D_iK_{jk}-D_jK_{ik}\).
Contracting \(j,k\) gives the displayed identity.  Einstein's equation turns
its right side into \(8\pi T_{ab}n^ah^{bi}\), so it constrains initial
\(h_{ij},K_{ij}\) and the matter momentum density.
\end{enumerate}
