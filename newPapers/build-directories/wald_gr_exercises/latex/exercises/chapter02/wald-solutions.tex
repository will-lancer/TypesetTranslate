\exerciseeditionsection{Solutions to Wald's Problems}
\ExerciseEditorialNote

\begin{enumerate}[leftmargin=2.8em,itemsep=1.2em]
\WaldSolution{1}
\begin{enumerate}[label=\alph*),leftmargin=2em]
\item On the hemisphere \(\pm x^i>0\), the inverse of the projection chart is
obtained by restoring the omitted coordinate,
\[
 x^i=\pm\sqrt{1-\sum_{k\ne i}(x^k)^2}.
\]
On an overlap the expression under the square root is strictly positive.
Every transition map is therefore a coordinate permutation followed, when
needed, by this smooth square-root function.  Hence all overlaps are smooth.

\item Let \(N=(0,0,1)\) and \(S=(0,0,-1)\).  Stereographic projection from
\(N\) maps \(S^2\setminus\{N\}\) diffeomorphically to \(\mathbb R^2\), while
projection from \(S\) does the same for \(S^2\setminus\{S\}\).  These two
domains cover \(S^2\).  In plane coordinates \(u=(u^1,u^2)\), the inverse of
the first map is
\[
 u\longmapsto\frac{1}{1+|u|^2}
 \bigl(2u^1,2u^2,|u|^2-1\bigr),
\]
and the second has the last component reversed.  Their overlap map is
\(u\mapsto u/|u|^2\), which is smooth away from the origin.
\end{enumerate}

\WaldSolution{2}
For \(x,a\in\mathbb R^n\), apply the fundamental theorem of calculus to the
line \(a+t(x-a)\):
\[
 F(x)-F(a)=\int_0^1\frac{\dd}{\dd t}F(a+t(x-a))\,\dd t
 =\sum_{\mu=1}^n(x^\mu-a^\mu)H_\mu(x),
\]
where
\[
 H_\mu(x)=\int_0^1
 \frac{\partial F}{\partial x^\mu}(a+t(x-a))\,\dd t.
\]
Differentiation under the integral shows that every \(H_\mu\) is smooth, and
\(H_\mu(a)=\partial_\mu F(a)\), which is precisely Eq.~\eqref{eq:2.2.2}.

\WaldSolution{3}
\begin{enumerate}[label=\alph*),leftmargin=2em]
\item From \([X,Y]f=X(Yf)-Y(Xf)\), linearity is immediate.  Expanding
\([X,Y](fg)\) with the product rule cancels the two cross terms and gives
\[
 [X,Y](fg)=f[X,Y]g+g[X,Y]f.
\]
Thus the commutator is a derivation and hence a vector field.

\item Acting on any smooth function, expand the three double commutators.
Every ordered product \(XYZ\) occurs twice with opposite signs, so
\[
 [[X,Y],Z]+[[Y,Z],X]+[[Z,X],Y]=0.
\]

\item Substitution of
\([Y_\alpha,Y_\beta]=C^\gamma{}_{\alpha\beta}Y_\gamma\) into the Jacobi
identity and use of \([X,fY]=X(f)Y+f[X,Y]\) yields
\[
 \begin{split}
 0={}&Y_\alpha(C^\delta{}_{\beta\gamma})
     +Y_\beta(C^\delta{}_{\gamma\alpha})
     +Y_\gamma(C^\delta{}_{\alpha\beta})\\
 &+C^\epsilon{}_{\beta\gamma}C^\delta{}_{\alpha\epsilon}
  +C^\epsilon{}_{\gamma\alpha}C^\delta{}_{\beta\epsilon}
  +C^\epsilon{}_{\alpha\beta}C^\delta{}_{\gamma\epsilon}.
 \end{split}
\]
For constant structure functions the derivative terms vanish, leaving the
usual algebraic Jacobi relation.
\end{enumerate}

\WaldSolution{4}
\begin{enumerate}[label=\alph*),leftmargin=2em]
\item Write \(v=v^\nu\partial_\nu\) and \(w=w^\nu\partial_\nu\).  Equality of
mixed partial derivatives cancels the second-derivative terms in
\(v(wf)-w(vf)\), giving
\[
 [v,w]^\mu=v^\nu\partial_\nu w^\mu-w^\nu\partial_\nu v^\mu.
\]

\item Let \(f^\gamma(Y_\alpha)=\delta^\gamma{}_{\alpha}\).  Differentiating
this identity and using part (a) gives
\[
 (\partial_\nu f^\gamma{}_\mu-
   \partial_\mu f^\gamma{}_\nu)
 Y_\alpha^\mu Y_\beta^\nu=C^\gamma{}_{\alpha\beta}.
\]
The right side of the claimed formula has the same contraction with every
pair \(Y_\alpha,Y_\beta\).  Since these fields form a basis, the two tensors
are equal.
\end{enumerate}

\WaldSolution{5}
By Problem 4(b), vanishing commutators imply
\(\partial_\nu f^\mu{}_\rho-\partial_\rho f^\mu{}_\nu=0\).  Thus each member
of the dual coframe is locally exact: \(f^\mu=\dd y^\mu\).  The Jacobian of
the functions \(y^\mu\) is nonsingular because the \(f^\mu\) form a basis,
so they are local coordinates.  Finally,
\[
 \dd y^\mu(Y_\nu)=f^\mu(Y_\nu)=\delta^\mu{}_{\nu},
\]
which characterizes \(Y_\nu=\partial/\partial y^\nu\).

\WaldSolution{6}
\begin{enumerate}[label=\alph*),leftmargin=2em]
\item If \(\sum_\mu c_\mu f^\mu=0\), evaluation on \(e_\nu\) gives
\(c_\nu=0\).  Thus the \(n\) covectors are linearly independent and form a
basis of \(V^\vee\).

\item Applying \(f^\beta\) to
\(w-\sum_\alpha f^\alpha(w)e_\alpha\) gives zero for every \(\beta\), so the
vector vanishes.  Evaluating
\(\omega-\sum_\alpha\omega(e_\alpha)f^\alpha\) on every basis vector proves
the covector formula.

\item For a tensor \(T\), contraction is the trace of the linear map obtained
by fixing its other slots.  In components it is
\(T^{\mu a_2\ldots}{}_{\mu b_2\ldots}\).  Under a basis change, the matrix and
inverse-matrix factors on the contracted indices multiply to
\(\delta^\mu{}_{\nu}\); hence the result is basis independent.
\end{enumerate}

\WaldSolution{7}
\begin{enumerate}[label=\alph*),leftmargin=2em]
\item Choose \(v\) with \(g(v,v)\ne0\); a nondegenerate symmetric form must
possess one.  Normalize it so \(g(v,v)=\pm1\).  Its orthogonal complement has
dimension \(n-1\), and the restriction of \(g\) to it is nondegenerate.
Induction supplies an orthonormal basis there, which together with \(v\)
gives the desired basis.

\item Let \(p\) be the number of plus signs in an orthonormal basis.  The span
of its positive vectors is a \(p\)-dimensional positive-definite subspace.
Any positive-definite subspace has dimension at most \(p\): otherwise it
would intersect the \((n-p)\)-dimensional negative span nontrivially.
Therefore \(p\) is the maximal dimension of a positive-definite subspace and
is basis independent.  The same argument fixes the number of minus signs.
\end{enumerate}

\WaldSolution{8}
\begin{enumerate}[label=\alph*),leftmargin=2em]
\item With
\(x=r\sin\theta\cos\phi\), \(y=r\sin\theta\sin\phi\), and
\(z=r\cos\theta\), direct differentiation gives
\[
 \dd x^2+\dd y^2+\dd z^2
 =\dd r^2+r^2\dd\theta^2+r^2\sin^2\theta\,\dd\phi^2.
\]

\item Put \(r=(x^2+y^2)^{1/2}\).  Since the inertial azimuth is
\(\phi-\omega t\),
\[
 \dd s^2=-(1-\omega^2r^2)\dd t^2+\dd r^2+r^2\dd\phi^2
 -2\omega r^2\dd t\,\dd\phi+\dd z^2.
\]
In the order \((t,r,\phi,z)\),
\[
 (g_{\mu\nu})=
 \begin{pmatrix}
 -1+\omega^2r^2&0&-\omega r^2&0\\
 0&1&0&0\\
 -\omega r^2&0&r^2&0\\
 0&0&0&1
 \end{pmatrix},\qquad
 (g^{\mu\nu})=
 \begin{pmatrix}
 -1&0&-\omega&0\\
 0&1&0&0\\
 -\omega&0&r^{-2}-\omega^2&0\\
 0&0&0&1
 \end{pmatrix}.
\]
\end{enumerate}
\end{enumerate}
