\exerciseeditionsection{Solutions to Supplementary Problems}
\begin{enumerate}[leftmargin=2.8em,itemsep=1.2em]
\SupplementarySolution{1}{The Newtonian tidal tensor}
Keeping first derivatives of the metric perturbation gives
\(R_{0i0j}=\partial_i\partial_j\Phi\).  For slowly moving neighboring
geodesics,
\[
 \frac{\dd^2\xi^i}{\dd t^2}=-R^i{}_{0j0}\xi^j
 =-\partial^i\partial_j\Phi\,\xi^j,
\]
which is the Newtonian tidal equation.  Its trace is \(\nabla^2\Phi\), so the
vacuum field equation says that the tidal tensor is trace free.
\end{enumerate}
