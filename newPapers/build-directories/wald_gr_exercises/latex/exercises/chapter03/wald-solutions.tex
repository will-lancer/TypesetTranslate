\exerciseeditionsection{Solutions to Wald's Problems}
\ExerciseEditorialNote

\begin{enumerate}[leftmargin=2.8em,itemsep=1.2em]
\WaldSolution{1}
\begin{enumerate}[label=\alph*),leftmargin=2em]
\item Let \(\widetilde\nabla_a\) be torsion-free and write
\[
 \nabla_a\omega_b=\widetilde\nabla_a\omega_b-C^c{}_{ab}\omega_c.
\]
For a scalar, \(\nabla_af=\widetilde\nabla_af\), so
\[
 (\nabla_a\nabla_b-\nabla_b\nabla_a)f
 =-(C^c{}_{ab}-C^c{}_{ba})\nabla_cf.
\]
Thus \(T^c{}_{ab}=2C^c{}_{[ab]}\) is a tensor and has the claimed property.

\item Expanding \(X^a\nabla_a(Y^c\nabla_cf)-Y^a\nabla_a(X^c\nabla_cf)\)
and using part (a), the Hessian terms combine to
\(-T^c{}_{ab}X^aY^b\nabla_cf\).  Comparing the remaining derivation with
\([X,Y]f\) gives
\[
 T^c{}_{ab}X^aY^b=X^a\nabla_aY^c-Y^a\nabla_aX^c-[X,Y]^c.
\]

\item Let \(D_a\) be the Levi--Civita derivative and write
\(\nabla_av^c=D_av^c+K^c{}_{ab}v^b\).  With
\(K_{cab}=g_{cd}K^d{}_{ab}\), metric compatibility says
\(K_{cab}=-K_{bac}\), while the prescribed torsion says
\(T_{cab}=K_{cab}-K_{cba}\).  Solving the cyclic system gives
\[
 K_{cab}=\frac12\bigl(T_{cab}+T_{bca}-T_{abc}\bigr).
\]
These algebraic equations determine \(K\) uniquely and also construct the
required connection.  Relative to any ordinary derivative \(\partial_a\),
\[
 \Gamma^c{}_{ab}
 =\frac12g^{cd}(\partial_ag_{bd}+\partial_bg_{ad}-\partial_dg_{ab})
 +g^{cd}K_{dab},
\]
with \(K\) as above.
\end{enumerate}

\WaldSolution{2}
\begin{enumerate}[label=\alph*),leftmargin=2em]
\item The metric volume form is covariantly constant.  Hence
\[
 \nabla_{[a}(\epsilon_{b]c}\nabla^c\alpha)
 =\frac12\epsilon_{ab}\nabla_c\nabla^c\alpha=0,
\]
where the two-dimensional epsilon contraction was used.  The one-form on the
left is therefore locally exact, which produces \(\beta\).  Its Laplacian is
\[
 \nabla_a\nabla^a\beta
 =\epsilon^{ab}\nabla_a\nabla_b\alpha=0
\]
because the Hessian of a scalar is symmetric.

\item Where \(\dd\alpha\ne0\), the relation
\(\dd\beta=\star\dd\alpha\) shows that the two coordinate one-forms are
orthogonal and satisfy
\[
 g^{ab}(\nabla_a\beta)(\nabla_b\beta)
 =(-1)^s g^{ab}(\nabla_a\alpha)(\nabla_b\alpha).
\]
Thus the inverse metric is diagonal in \((\alpha,\beta)\) with entries
related by this sign.  Inverting it gives
\[
 \dd s^2=\pm\Omega^2(\alpha,\beta)
 \bigl(\dd\alpha^2+(-1)^s\dd\beta^2\bigr).
\]
\end{enumerate}

\WaldSolution{3}
\begin{enumerate}[label=\alph*),leftmargin=2em]
\item The first Bianchi identity and antisymmetry in each index pair give
\[
 R_{abcd}=R_{cdab}.
\]
For example, expand \(R_{a[bcd]}=0\), interchange the first pair in two
terms, and apply the same identity with the pairs reversed; the difference
\(R_{abcd}-R_{cdab}\) cancels term by term.

\item Regard \(R_{abcd}\) as a symmetric bilinear form on
\(\Lambda^2T_pM\), whose dimension is
\(m=n(n-1)/2\).  Pair antisymmetry and pair exchange symmetry leave
\(m(m+1)/2\) components.  The algebraic Bianchi identity removes one
independent totally antisymmetric component for every four distinct indices,
namely \(\binom n4\).  Therefore
\[
 \frac{m(m+1)}2-\binom n4
 =\frac{n^2(n^2-1)}{12}.
\]
\end{enumerate}

\WaldSolution{4}
\begin{enumerate}[label=\alph*),leftmargin=2em]
\item In two dimensions the preceding count gives one curvature component.
The tensor \(g_{a[c}g_{d]b}\) has all Riemann symmetries and is nonzero, so
\(R_{abcd}=K g_{a[c}g_{d]b}\).  Contracting \(a\) with \(c\) gives
\(R_{bd}=\tfrac12K g_{bd}\), and a second contraction gives \(K=R\).

\item In three dimensions the Riemann tensor and the symmetric Ricci tensor
both have six independent components.  The trace-adjusted map
\[
 R_{abcd}=2\bigl(g_{a[c}R_{d]b}-g_{b[c}R_{d]a}\bigr)
 -R g_{a[c}g_{d]b}
\]
has the Riemann symmetries and contracts to \(R_{ab}\), so it exhausts the
curvature.  Comparison with Eq.~\eqref{eq:3.2.28} gives \(C_{abcd}=0\).
\end{enumerate}

\WaldSolution{5}
\begin{enumerate}[label=\alph*),leftmargin=2em]
\item For a parameter \(\lambda\), suppose
\(v^a\nabla_av^b=fv^b\).  Put \(u^a=(\dd\lambda/\dd s)v^a=qv^a\).
Then
\[
 u^a\nabla_au^b=q\left(\frac{\dd q}{\dd\lambda}+fq\right)v^b.
\]
Choosing
\(q=C\exp[-\int f\,\dd\lambda]\) makes this vanish and defines an affine
parameter \(s\).

\item If \(t\) and \(s\) are affine, the preceding formula with \(f=0\)
requires \(\dd^2t/\dd s^2=0\).  Hence \(t=as+b\), with \(a\ne0\).
\end{enumerate}

\WaldSolution{6}
\begin{enumerate}[label=\alph*),leftmargin=2em]
\item The nonzero components, together with those obtained by symmetry in the
lower indices, are
\[
 \begin{gathered}
 \Gamma^r{}_{\theta\theta}=-r,\qquad
 \Gamma^r{}_{\phi\phi}=-r\sin^2\theta,\\
 \Gamma^\theta{}_{r\theta}=r^{-1},\qquad
 \Gamma^\theta{}_{\phi\phi}=-\sin\theta\cos\theta,\\
 \Gamma^\phi{}_{r\phi}=r^{-1},\qquad
 \Gamma^\phi{}_{\theta\phi}=\cot\theta.
 \end{gathered}
\]

\item The geodesic equations are
\[
 \begin{aligned}
 \ddot r-r\dot\theta^2-r\sin^2\theta\dot\phi^2&=0,\\
 \ddot\theta+2\dot r\dot\theta/r-
 \sin\theta\cos\theta\dot\phi^2&=0,\\
 \ddot\phi+2\dot r\dot\phi/r+2\cot\theta\dot\theta\dot\phi&=0.
 \end{aligned}
\]
Substituting
\((x,y,z)=r(\sin\theta\cos\phi,\sin\theta\sin\phi,\cos\theta)\)
shows that these three equations are equivalent to
\(\ddot x=\ddot y=\ddot z=0\).  Their images are straight Cartesian lines.
\end{enumerate}

\WaldSolution{7}
Write \(\sigma=\ln\Omega\).  The coordinate calculation gives
\[
 \Gamma^a{}_{bc}=\delta^a_b\partial_c\sigma+
 \delta^a_c\partial_b\sigma-\eta_{bc}\eta^{ad}\partial_d\sigma
\]
and
\[
 R=2\Omega^{-2}(\partial_t^2-\partial_x^2)\sigma,
 \qquad R_{abcd}=R\,g_{a[c}g_{d]b}.
\]
For the tetrad calculation choose
\(e^0=\Omega\,\dd t\), \(e^1=\Omega\,\dd x\).  The torsion-free Cartan
equations give
\[
 \omega^0{}_1=(\partial_x\sigma)\,\dd t+(\partial_t\sigma)\,\dd x.
\]
Taking \(\dd\omega^0{}_1\) produces the same curvature two-form and hence
the same scalar and Riemann tensor.

\WaldSolution{8}
Antisymmetry in the last two indices expands the two antisymmetrizations and
immediately gives
\[
 \omega_{\lambda\mu\nu}=3\omega_{[\lambda\mu\nu]}
 -2\omega_{[\mu\nu]\lambda}.
\]
Define
\(C_{\sigma\mu\nu}=(e_\sigma)_a[e_\mu,e_\nu]^a\).  Equation
\eqref{eq:3.4.23} and its two cyclic permutations form a linear system.  Its
solution is
\[
 \boxed{\quad
 \omega_{\lambda\mu\nu}
 =\frac12\bigl(C_{\mu\lambda\nu}+C_{\nu\mu\lambda}
 -C_{\lambda\nu\mu}\bigr).
 \quad}
\]
Replacing each commutator by the antisymmetrized derivative of the basis
vectors gives the equivalent ordinary-derivative formula.
\end{enumerate}
