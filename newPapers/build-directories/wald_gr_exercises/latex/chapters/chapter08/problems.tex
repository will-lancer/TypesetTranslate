\chapterproblems

\begin{enumerate}[label=\arabic*.,leftmargin=*]
\item Prove that for any subset \(S\) of the spacetime manifold, \(M\), we have
  \begin{enumerate}[label=\alph*),leftmargin=2em]
  \item \(I^+(I^+(S))=I^+(S)\) and
  \item \(I^+(\overline S)=I^+(S)\), where \(\overline S\) is the closure of \(S\).
  \end{enumerate}

\item Let \(S\) be any subset of the spacetime manifold \(M\). Give a proof of the
following two claims made in the text:
  \begin{enumerate}[label=\alph*),leftmargin=2em]
  \item \(J^+(S)\subseteq\overline{I^+(S)}\) and
  \item \(I^+(S)=\operatorname{int}(J^+(S))\).
  \end{enumerate}

\item Prove lemma~\ref{lem:8.3.3}.

\item Let \((M,g_{ab})\) be a globally hyperbolic spacetime with Cauchy surface
\(\Sigma\), and let \(C\subseteq\Sigma\) be closed. Show that
\(J^+(\Sigma)=D^+(C)\cup J^+(\Sigma-C)\).

\item Prove that for any closed, achronal set \(S\), we have
\[
  H^+(S)=\bigl[I^+(S)\cup S\bigr]\cap\partial I^-(D^+(S)).
\]

\item Prove proposition~\ref{prop:8.3.6}.

\item Prove theorem~\ref{thm:8.3.11}.

\item
  \begin{enumerate}[label=\alph*),leftmargin=2em]
  \item Let \((M,g_{ab})\) be a globally hyperbolic spacetime with Cauchy surface
  \(\Sigma\). Let \(C\subseteq\Sigma\) be closed but not necessarily compact. Prove
  that \(J^+(C)\) is closed.
  \item Find an example of a closed, achronal set \(S\) in Minkowski spacetime such
  that \(J^+(S)\) is not closed. (Hint: Consider a noncompact subset of a null plane.)
  \end{enumerate}
\end{enumerate}
