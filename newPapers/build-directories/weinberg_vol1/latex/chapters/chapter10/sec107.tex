\subsection[The K\"all\'en--Lehmann Representation]{The K\"all\'en--Lehmann Representation\texorpdfstring{\protect\footnotemark}{}}
\footnotetext{This section lies somewhat out of the book's main line of
development, and may be omitted in a first reading.}

We saw in Section 10.2 that the presence of one-particle intermediate
states leads to poles in Fourier transforms of matrix elements of time-ordered
products, like~\eqref{eq:10.2.1}. Multi-particle intermediate states
lead to more complicated singularities, which are difficult to describe in
general. But in the special case of a vacuum expectation value involving just
two operators, we have a convenient representation that explicitly displays
the analytic structure of the Fourier transform. In particular, this
representation may be used for propagators, where the two operators are the
fields of elementary particles. When combined with the positivity requirements
of quantum mechanics, this representation yields interesting bounds on the
asymptotic behavior of propagators and the magnitude of renormalization
constants.

Consider a complex scalar Heisenberg-picture operator $\Phi(x)$, which
may or may not be an elementary particle field. The vacuum expectation
value of a product $\Phi(x)\Phi^\dagger(y)$ may be expressed as
\begin{equation}
  \bra{\Omega}\Phi(x)\Phi^\dagger(y)\ket{\Omega}
  =\sum_n\bra{\Omega}\Phi(x)\ket{n}
  \bra{n}\Phi^\dagger(y)\ket{\Omega},
  \tag{10.7.1}\label{eq:10.7.1}
\end{equation}
where the sum runs over any complete set of states. (Here the sum over
$n$ includes integrals over continuous labels as well as sums over discrete
labels.) Choosing these states as eigenstates of the momentum four-vector
$P^\mu$, translational invariance tells us that
\begin{align}
  \bra{\Omega}\Phi(x)\ket{n}
  &=\exp(ip_n\cdot x)\bra{\Omega}\Phi(0)\ket{n},\notag\\
  \bra{n}\Phi^\dagger(y)\ket{\Omega}
  &=\exp(-ip_n\cdot y)\bra{n}\Phi^\dagger(0)\ket{\Omega}
  \tag{10.7.2}\label{eq:10.7.2}
\end{align}
and so
\begin{equation}
  \bra{\Omega}\Phi(x)\Phi^\dagger(y)\ket{\Omega}
  =\sum_n\exp\bigl(ip_n\cdot(x-y)\bigr)
  \left|\bra{\Omega}\Phi(0)\ket{n}\right|^2.
  \tag{10.7.3}\label{eq:10.7.3}
\end{equation}

It is convenient to rewrite this in terms of a \emph{spectral function}.
Note that the sum
$\sum_n\delta^4(p-p_n)|\bra{\Omega}\Phi(0)\ket{n}|^2$ is a scalar function
of the four-vector $p^\mu$, and therefore may depend only on $p^2$ and
(for $p^2\leq0$) on the step function $\theta(p^0)$. In fact, the intermediate
states in Eq.~\eqref{eq:10.7.3} all have $p^2\leq0$ and $p^0>0$, so this sum
takes the form
\begin{equation}
  \sum_n\delta^4(p-p_n)
  \left|\bra{\Omega}\Phi(0)\ket{n}\right|^2
  =(2\pi)^{-3}\theta(p^0)\rho(-p^2)
  \tag{10.7.4}\label{eq:10.7.4}
\end{equation}
with $\rho(-p^2)=0$ for $p^2>0$. (The factor $(2\pi)^{-3}$ is extracted
from $\rho$ for future convenience.) The spectral function $\rho(-p^2)$
is clearly real and positive. With this definition, we can rewrite
Eq.~\eqref{eq:10.7.3} as
\begin{align}
  \bra{\Omega}\Phi(x)\Phi^\dagger(y)\ket{\Omega}
  &=(2\pi)^{-3}\int d^4p\,\notag\\[-2pt]
  &\quad\cdot\exp\bigl[ip\cdot(x-y)\bigr]\theta(p^0)\rho(-p^2)\notag\\
  &=(2\pi)^{-3}\int d^4p\int_0^\infty d\mu^2\,\notag\\[-2pt]
  &\quad\cdot
  \exp\bigl[ip\cdot(x-y)\bigr]\theta(p^0)
  \rho(\mu^2)\delta(p^2+\mu^2).
  \tag{10.7.5}\label{eq:10.7.5}
\end{align}
Interchanging the order of integration over $p^\mu$ and $\mu^2$, this may
be expressed as
\begin{equation}
  \bra{\Omega}\Phi(x)\Phi^\dagger(y)\ket{\Omega}
  =\int_0^\infty d\mu^2\,\rho(\mu^2)\Delta_+(x-y;\mu^2),
  \tag{10.7.6}\label{eq:10.7.6}
\end{equation}
where $\Delta_+$ is the familiar function
\begin{equation}
  \Delta_+(x-y;\mu^2)\equiv(2\pi)^{-3}\int d^4p\,
  \exp\bigl[ip\cdot(x-y)\bigr]\theta(p^0)\delta(p^2+\mu^2).
  \tag{10.7.7}\label{eq:10.7.7}
\end{equation}

In just the same way, we can show that
\begin{equation}
  \bra{\Omega}\Phi^\dagger(y)\Phi(x)\ket{\Omega}
  =\int_0^\infty d\mu^2\,\bar\rho(\mu^2)\Delta_+(y-x;\mu^2)
  \tag{10.7.8}\label{eq:10.7.8}
\end{equation}
with a second spectral function $\bar\rho(\mu^2)$ defined by
\begin{equation}
  \sum_n\delta^4(p+p_n)
  \left|\bra{n}\Phi^\dagger(0)\ket{\Omega}\right|^2
  =(2\pi)^{-3}\theta(p^0)\bar\rho(-p^2).
  \tag{10.7.9}\label{eq:10.7.9}
\end{equation}

We now make use of the causality requirement, that the commutator
$[\Phi(x),\Phi^\dagger(y)]$ must vanish for space-like separations $x-y$.
The vacuum expectation value of the commutator is
\begin{align}
  \bra{\Omega}[\Phi(x),\Phi^\dagger(y)]\ket{\Omega}
  =\int_0^\infty d\mu^2\bigl(&\rho(\mu^2)\Delta_+(x-y;\mu^2)
  \notag\\[-2pt]
  &{}-\bar\rho(\mu^2)\Delta_+(y-x;\mu^2)\bigr).
  \tag{10.7.10}\label{eq:10.7.10}
\end{align}
As noted in Section 5.2, for $x-y$ space-like the function
$\Delta_+(x-y)$ does not vanish, but it does become \emph{even}. In order
for~\eqref{eq:10.7.10} to vanish for arbitrary space-like separations, it is
thus necessary that
\begin{equation}
  \rho(\mu^2)=\bar\rho(\mu^2).
  \tag{10.7.11}\label{eq:10.7.11}
\end{equation}
This is a special case of the CPT theorem, proved here without the use of
perturbation theory; for whatever states with $p^2=-\mu^2$ have the quantum
numbers of the operator $\Phi$, there must be corresponding states with
$p^2=-\mu^2$ that have the quantum numbers of the operator $\Phi^\dagger$.

Using Eq.~\eqref{eq:10.7.11}, the vacuum expectation of the time-ordered
product is
\begin{equation}
  \bra{\Omega}T\{\Phi(x)\Phi^\dagger(y)\}\ket{\Omega}
  =-i\int_0^\infty d\mu^2\,\rho(\mu^2)\Delta_F(x-y;\mu^2),
  \tag{10.7.12}\label{eq:10.7.12}
\end{equation}
where $\Delta_F(x-y;\mu^2)$ is the Feynman propagator for a spinless
particle of mass $\mu$:
\begin{align}
  -i\Delta_F(x-y;\mu^2)\equiv{}&
  \theta(x^0-y^0)\Delta_+(x-y;\mu^2)\notag\\
  &{}-\theta(y^0-x^0)\Delta_+(y-x;\mu^2).
  \tag{10.7.13}\label{eq:10.7.13}
\end{align}
Borrowing the notation introduced in Section 10.3 for complete propagators,
we introduce the momentum space function
\begin{equation}
  -i\Delta'(p)\equiv\int d^4x\,\exp\bigl[-ip\cdot(x-y)\bigr]
  \bra{\Omega}T\{\Phi(x)\Phi^\dagger(y)\}\ket{\Omega}.
  \tag{10.7.14}\label{eq:10.7.14}
\end{equation}
Recall that
\begin{equation}
  \int d^4x\,\exp\bigl[-ip\cdot(x-y)\bigr]\Delta_F(x-y;\mu^2)
  =\frac{1}{p^2+\mu^2-i\epsilon}.
  \tag{10.7.15}\label{eq:10.7.15}
\end{equation}
This yields our spectral representation~\hyperref[ref:10.9]{[9]}:
\begin{equation}
  \Delta'(p)=\int_0^\infty\rho(\mu^2)
  \frac{d\mu^2}{p^2+\mu^2-i\epsilon}.
  \tag{10.7.16}\label{eq:10.7.16}
\end{equation}

One immediate consequence of this result and the positivity of $\rho(\mu^2)$
is that $\Delta'(p)$ cannot vanish for $|p^2|\to\infty$ faster\footnote{In fact,
it is not even certain that $\Delta'(p)$ vanishes for $|p^2|\to\infty$ at all,
even though this would seem to follow from the spectral representation. The
problem arises from the interchange of the integrals over $p^\mu$ and $\mu^2$.
What is certain is that $\Delta'(p)$ is an analytic function of $-p^2$ with a
discontinuity across the positive real axis $-p^2=\mu^2$ given by
$\pi\rho(\mu^2)$, as can be shown by the methods of the next section. From
this, it follows that $\Delta'(p)$ is given by a dispersion relation with
spectral function $\rho(\mu^2)$ and possible subtractions:
\[
  \Delta'(p)=P(p^2)+(-p^2+\mu_0^2)^n\int_0^\infty
  \frac{\rho(\mu^2)}{(\mu^2+\mu_0^2)^n}
  \frac{d\mu^2}{p^2+\mu^2-i\epsilon},
\]
where $n$ is a positive integer, $\mu_0^2$ is an arbitrary positive constant,
and $P(p^2)$ is a $\mu_0^2$-dependent polynomial in $p^2$ of order $n-1$
that is absent for $n=0$.} than the bare propagator
$1/(p^2+m^2-i\epsilon)$. From time to time the suggestion is made to
include higher derivative terms in the unperturbed Lagrangian, which would
make the propagator vanish faster than $1/p^2$ for $|p^2|\to\infty$, but the
spectral representation shows that this would necessarily entail a departure
from the positivity postulates of quantum mechanics.

We can use the spectral representation together with equal-time commutation
relations to derive an interesting sum rule for the spectral function. If
$\Phi(x)$ is a conventionally normalized (not renormalized) canonical field
operator, then
\begin{equation}
  \left[\frac{\partial\Phi(\mathbf{x},t)}{\partial t},
  \Phi^\dagger(\mathbf{y},t)\right]=-i\delta^3(\mathbf{x}-\mathbf{y}).
  \tag{10.7.17}\label{eq:10.7.17}
\end{equation}
We note that
\begin{equation*}
  \left.\frac{\partial}{\partial x^0}\Delta_+(x-y)\right|_{x^0=y^0}
  =-i\delta^3(\mathbf{x}-\mathbf{y})
\end{equation*}
so the spectral representation~\eqref{eq:10.7.10} and the commutation
relations~\eqref{eq:10.7.17} together tell us that
\begin{equation}
  \int_0^\infty\rho(\mu^2)\,d\mu^2=1.
  \tag{10.7.18}\label{eq:10.7.18}
\end{equation}
This implies that for $|p^2|\to\infty$, the momentum space propagator
\eqref{eq:10.7.16} of the unrenormalized fields has the free-field asymptotic
behavior
\begin{equation*}
  \Delta'(p)\longrightarrow\frac{1}{p^2}.
\end{equation*}
This result is only meaningful within a suitable scheme for regulating
ultraviolet divergences; in perturbation theory the unrenormalized fields
have infinite matrix elements, and their propagator is ill-defined.

Now consider the possibility that there is a one-particle state $\ket{k}$ of
mass $m$ with a non-vanishing matrix element with the state
$\bra{\Omega}\Phi(0)$. Lorentz invariance requires this matrix element to take
the form
\begin{equation}
  \bra{\Omega}\Phi(0)\ket{k}
  =(2\pi)^{-3/2}\left(2\sqrt{\mathbf{k}^2+m^2}\right)^{-1/2}N,
  \tag{10.7.19}\label{eq:10.7.19}
\end{equation}
where $N$ is a constant. According to the general results of Section 10.3,
the propagator $\Delta'(p)$ of the unrenormalized fields should have a pole at
$p^2\to-m^2$ with residue $Z\equiv|N|^2>0$. That is,
\begin{equation}
  \rho(\mu^2)=Z\delta(\mu^2-m^2)+\sigma(\mu^2),
  \tag{10.7.20}\label{eq:10.7.20}
\end{equation}
where $\sigma(\mu^2)\geq0$ is the contribution of multi-particle states.
Together with Eq.~\eqref{eq:10.7.18}, this has the consequence that
\begin{equation}
  1=Z+\int_0^\infty\sigma(\mu^2)\,d\mu^2
  \tag{10.7.21}\label{eq:10.7.21}
\end{equation}
and so
\begin{equation}
  Z\leq1
  \tag{10.7.22}\label{eq:10.7.22}
\end{equation}
with the equality reached only for a free particle, for which
$\bra{\Omega}\Phi(x)$ has no matrix elements with multi-particle states.

Because $Z$ is positive, Eq.~\eqref{eq:10.7.21} can also be regarded as
providing an upper bound on the coupling of the field $\Phi$ to multi-particle
states:
\begin{equation}
  \int_0^\infty\sigma(\mu^2)\,d\mu^2\leq1
  \tag{10.7.23}\label{eq:10.7.23}
\end{equation}
with the equality reached for $Z=0$. The limit $Z=0$ has an interesting
interpretation as a condition for a particle to be composite rather than
elementary~\hyperref[ref:10.10]{[10]}. In this context, a `composite' particle
may be understood to be one whose field does not appear in the Lagrangian.
Consider such a particle, say a neutral particle of spin zero, and suppose
that its quantum numbers allow it to be destroyed by an operator $F(\Psi)$
constructed out of other fields. We can freely introduce a field $\Phi$ for
this particle by adding a term to the Lagrangian density of the
form\footnote{This is known in condensed matter physics as a
`Hubbard--Stratonovich transformation'~\hyperref[ref:10.11]{[11]}. It will be
used to introduce fields for pairs of electrons in our discussion of
superconductivity in Volume II.} $\Delta\mathcal L=(\Phi-F(\Psi))^2$,
because the path integral over $\Phi$ can be done by setting it equal to the
stationary point $\Phi=F(\Psi)$, at which $\Delta\mathcal L=0$. But suppose
instead we write $\Delta\mathcal L=\Delta\mathcal L_0+\Delta\mathcal L_1$,
where $\Delta\mathcal L_0=-\frac12\partial_\mu\Phi\partial^\mu\Phi

$\Delta\mathcal L_1=\Delta\mathcal L-\Delta\mathcal L_0$ as an interaction.
