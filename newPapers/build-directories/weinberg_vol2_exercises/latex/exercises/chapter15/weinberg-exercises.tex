\chapterbackmatter{Weinberg Exercises}

\begin{enumerate}
  \WeinbergExercise{1} Derive the Bianchi identity
  \[
    D_\mu F_{a\nu\lambda}+D_\nu F_{a\lambda\mu}
    +D_\lambda F_{a\mu\nu}=0.
  \]

  \WeinbergExercise{2} Suppose we use generalized Coulomb gauge in a non-Abelian gauge theory, taking the gauge-fixing function as $f_a=\boldsymbol\nabla\cdot\mathbf A_a$. Derive the ghost Lagrangian. What is the ghost propagator? (Take $B[f]=\exp(-i\int d^4x\,f_a f_a/2\xi)$.)

  \WeinbergExercise{3} Suppose that in electrodynamics we use a gauge-fixing function $f=\partial_\mu A^\mu+cA_\mu A^\mu$, with $c$ an arbitrary constant. Derive the ghost Lagrangian.\\
  % The source prints f_\alpha f_\alpha here even though its Abelian f is unindexed.
  (Take $B[f]=\exp(-i\int d^4x\,f^2/2\xi)$.) What is the ghost propagator?

  \WeinbergExercise{4} Show that there is no simple Lie algebra with just four generators.

  \WeinbergExercise{5} Show that if a field $\psi_\ell(x)$ belonging to a representation of a gauge group with generator matrices $t_a$ varies along a path $x^\mu=x^\mu(\tau)$ according to the differential equation
  \[
    \frac{d\psi(\tau)}{d\tau}
    =it_a\psi(\tau)A_{a\mu}(x(\tau))\frac{dx^\mu(\tau)}{d\tau},
  \]
  (with both $\psi_\ell(x)$ and $A_{a\mu}(x)$ classical c-number fields) then the change in $\psi$ around any small closed path $\gamma$ around a point $X^\mu$ is proportional to
  \[
    t_a\psi(X)F_{a\mu\nu}(X)
    \oint_\gamma x^\mu(\tau)\frac{dx^\nu(\tau)}{d\tau}\,d\tau.
  \]
  Find the constant of proportionality. Use this result to show that if $F_{a\mu\nu}(x)$ vanishes everywhere then it is possible by a gauge transformation to make $A_{a\mu}(x)$ vanish throughout at least a finite region. (Hint: Follow the analogous argument in electrodynamics, or in general relativity, as in Ref.~\hyperref[ch15-ref-6]{[6]}.)

  \WeinbergExercise{6} Applying path-integral methods to a general non-Abelian gauge theory, calculate the propagator of the gauge field $A_{a\mu}(x)$ if we choose a gauge-fixing functional $f_a=n_\mu A_a^\mu$, with $n_\mu$ arbitrary constants. (Take $B[f]=\exp(-i\int d^4x\,f_a f_a/2\xi)$.) What is the ghost Lagrangian? What is the ghost propagator? What is the ghost interaction vertex?

  \WeinbergExercise{7} Suppose we adopt BRST invariance instead of gauge invariance as a fundamental physical principle. Derive the most general Lagrangian density constructed from sums of products of fields and field derivatives of dimensionality $(\text{mass})^d$ with $d\leq4$, constructed from $A_{a\mu}$, $\omega_a$, $\omega_a^*$, $h_a$, and/or their derivatives, that is invariant (up to total derivatives) under Lorentz transformations, ghost number phase transformations ($\omega_a\longrightarrow e^{i\theta}\omega_a$, $\omega_a^*\longrightarrow e^{-i\theta}\omega_a^*$), global gauge transformations ($\epsilon_a$ constant), and BRST invariance.

  \WeinbergExercise{8} Show that the antibracket satisfies the symmetry condition \eqref{eq:15.9.15} and the Jacobi identity \eqref{eq:15.9.21}.

  \WeinbergExercise{9} Show that if a functional $O$ satisfies the condition $(O,S)=i\Delta S$ and the action $S$ satisfies the quantum master equation then the quantum average $\langle O\rangle$ is independent of the gauge-fixing functional $\Psi$.
\end{enumerate}
