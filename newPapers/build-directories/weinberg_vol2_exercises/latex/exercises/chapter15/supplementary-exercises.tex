\chapterbackmatter{Supplementary Exercises}

\begin{enumerate}
  \SupplementaryExercise{1}{Covariant-Derivative Curvature}
  {Adapted from Hong Liu, MIT OpenCourseWare 8.324, Problem Set 1, Problem 2 (2010).}
  {mit-8324-f10-ps1}
  Let \(D_\mu=\partial_\mu-iA_\mu\), where
  \(A_\mu=t_\alpha A^\alpha_\mu\).  Expand
  \([D_\mu,D_\nu]\psi\), identify the matrix-valued field strength, and show
  directly that it transforms as
  \(F_{\mu\nu}\longrightarrow UF_{\mu\nu}U^{-1}\) under a finite gauge
  transformation.

  \SupplementaryExercise{2}{Infinitesimal Plaquette Holonomy}
  {Adapted from Hong Liu, MIT OpenCourseWare 8.324, Problem Set 1, Problem 2 (2010).}
  {mit-8324-f10-ps1}
  Form the path-ordered parallel transporter around a rectangular loop based
  at \(x\), with side vectors \(a^\mu\) and \(b^\mu\).  Expand all four
  segments through second order in the side lengths and show that
  \[
    U_\square=1+iF_{\mu\nu}(x)a^\mu b^\nu+O(\ell^3).
  \]
  Explain why path ordering is essential for the commutator term in
  \(F_{\mu\nu}\).

  \SupplementaryExercise{3}{Endpoint Law for Wilson Lines}
  {Adapted from N. Beisert et al., ETH Zurich Quantum Field Theory II, Problem Set 5, Problem 5.3 (2017).}
  {eth-qft2-fs17}
  For an oriented path \(\gamma:y\to x\), define
  \[
    W_\gamma(x,y)=\mathcal P\exp\left(i\int_\gamma A_\mu\,dz^\mu\right).
  \]
  Derive the gauge-transformation law of \(W_\gamma(x,y)\).  Use it to show
  that a closed Wilson loop has a gauge-invariant trace, and that
  \(\psi^\dagger(x)W_\gamma(x,y)\psi(y)\) is gauge invariant when \(\psi\)
  transforms in the same representation.

  \SupplementaryExercise{4}{Abelian Wilson Loop and Flux}
  {Adapted from N. Beisert et al., ETH Zurich Quantum Field Theory II, Problem Set 5, Problem 5.3 (2017).}
  {eth-qft2-fs17}
  Specialize the Wilson loop to an Abelian gauge theory.  Show that path
  ordering drops out and use Stokes' theorem to express the loop as the
  exponential of the field-strength flux through a spanning surface.  State
  precisely when the result is independent of the chosen spanning surface.

  \SupplementaryExercise{5}{Covariant Bianchi Identity in Matrix Form}
  {Adapted from N. Beisert et al., ETH Zurich Quantum Field Theory II, Problem Set 5, Problem 5.3 (2017).}
  {eth-qft2-fs17}
  Starting with
  \(F=dA-iA\wedge A\) in matrix notation, prove
  \[
    D_{[\lambda}F_{\mu\nu]}=0.
  \]
  Then show that the equation transforms covariantly under a finite gauge
  transformation and reduce it to the two homogeneous Maxwell equations in
  the Abelian limit.

  \SupplementaryExercise{6}{Yang--Mills Equation and Covariant Conservation}
  {Adapted from Hong Liu, MIT OpenCourseWare 8.324, Problem Set 2, Problems 4--5 (2010).}
  {mit-8324-f10-ps2}
  Vary
  \[
    I=-\frac14\int d^4x\,F^a_{\mu\nu}F_a^{\mu\nu}
      +\int d^4x\,A^a_\mu J_a^\mu
  \]
  with respect to \(A^a_\nu\), treating \(J^\mu\) as an adjoint source.
  Obtain the Yang--Mills equation and prove that its left-hand side has
  identically vanishing covariant divergence.  What consistency condition
  must the source satisfy?

  \SupplementaryExercise{7}{Adjoint \(SU(2)\) and Its Casimir}
  {Adapted from M. Grazzini et al., University of Zurich PHY552 Quantum Field Theory II, Series 4--6 (2019).}
  {uzh-phy552-fs19}
  With \(C^a{}_{bc}=\epsilon_{abc}\), construct the three adjoint
  generators \((t^A_a)^b{}_c=-iC^b{}_{ca}\).  Verify their commutation
  relations, calculate \(\sum_a t^A_at^A_a\), and compare the result with
  the quadratic Casimir in the fundamental representation
  \(t_a=\sigma_a/2\).

  \SupplementaryExercise{8}{Equal Curvature Is Not Gauge Equivalence}
  {Adapted from Hong Liu, MIT OpenCourseWare 8.324, Problem Set 2, Problem 5 (2010).}
  {mit-8324-f10-ps2}
  Work on the \(x^1x^2\)-plane with an \(SU(2)\) basis satisfying
  \([T_a,T_b]=\epsilon_{abc}T_c\).  Compare the two connections
  \[
    A_1=aT_1,\quad A_2=aT_2,
    \qquad
    A'_1=0,\quad A'_2=a^2x^1T_3.
  \]
  Show that they have the same curvature \(F_{12}=a^2T_3\), but are not
  gauge equivalent.  Find a local gauge-covariant quantity that
  distinguishes them, and explain why this phenomenon has no Abelian
  analogue on a simply connected patch.

  \SupplementaryExercise{9}{Lorenz-Gauge Faddeev--Popov Kernel}
  {Adapted from N. Beisert et al., ETH Zurich Quantum Field Theory II, Problem Set 6, Problems 6.1--6.2 (2017).}
  {eth-qft2-fs17}
  For \(f_\alpha=\partial_\mu A_\alpha^\mu\), compute
  \(\mathcal F_{\alpha x,\beta y}=\delta f_\alpha(x)/
  \delta\lambda_\beta(y)\).  Separate it into field-independent and
  interacting pieces, derive the ghost Lagrangian after integration by
  parts, and identify the free ghost propagator.

  \SupplementaryExercise{10}{Ghost Interactions from a Nonlinear Abelian Gauge}
  {Adapted from N. Beisert et al., ETH Zurich Quantum Field Theory II, Problem Set 6, Problem 6.1 (2017).}
  {eth-qft2-fs17}
  In electrodynamics choose
  \(f[A]=\partial_\mu A^\mu+\frac{\kappa}{2}A_\mu A^\mu\).
  Compute the Faddeev--Popov determinant and its ghost representation.
  Derive the photon--ghost--antighost vertex for incoming ghost momentum
  \(q\), and explain why Abelian ghosts fail to decouple in this gauge.

  \SupplementaryExercise{11}{Ghost Decoupling in Axial Gauge}
  {Adapted from N. Beisert et al., ETH Zurich Quantum Field Theory II, Problem Set 6, Problem 6.1 (2017).}
  {eth-qft2-fs17}
  In Yang--Mills theory take \(f_\alpha=n^\mu A_{\alpha\mu}\).
  Derive the Faddeev--Popov operator.  Show that on the sharp gauge slice
  \(n\cdot A_\alpha=0\) its determinant is field independent.  State the
  assumptions under which the ghost fields can therefore be removed from
  perturbation theory.

  \SupplementaryExercise{12}{General Covariant-Gauge Propagator}
  {Adapted from David Skinner, Advanced Quantum Field Theory Example Sheet 4, Problem 3 (2018).}
  {skinner-aqft-sheet4-2018}
  Add the gauge-fixing term
  \[
    -\frac1{2\xi}(\partial_\mu A_a^\mu)^2
  \]
  to the free Yang--Mills Lagrangian.  Invert the quadratic
  momentum-space operator to obtain the
  gauge-field propagator.  Check explicitly its Feynman-gauge and
  Landau-gauge limits and its longitudinal contraction with \(p^\mu\).

  \SupplementaryExercise{13}{Momentum-Space Ghost--Gluon Vertex}
  {Adapted from N. Beisert et al., ETH Zurich Quantum Field Theory II, Problem Set 6, Problems 6.1--6.2 (2017).}
  {eth-qft2-fs17}
  Starting from
  \[
    \mathcal L_{\rm GH}
      =-\partial_\mu\omega_a^*\partial^\mu\omega_a
       +C_{abc}(\partial_\mu\omega_a^*)A_c^\mu\omega_b ,
  \]
  derive the momentum-space ghost--gluon vertex with all momenta incoming.
  Record which ghost momentum appears, and explain the extra minus sign
  associated with a closed ghost loop.

  \SupplementaryExercise{14}{Auxiliary-Field Form of Gauge Fixing}
  {Adapted from Michael Peskin, Stanford Physics 331, Problem Set 2, Problem 2 (2020).}
  {stanford-physics331-w20-set2}
  Introduce an auxiliary field \(h_a\) and replace
  \(-f_af_a/(2\xi)\) by \(h_af_a+\xi h_ah_a/2\).
  Integrate out \(h_a\) to recover the original gauge-fixing term,
  including its normalization up to a field-independent determinant.
  Explain why keeping \(h_a\) makes BRST nilpotency off shell.

  \SupplementaryExercise{15}{Off-Shell BRST Nilpotency}
  {Adapted from N. Beisert et al., ETH Zurich Quantum Field Theory II, Problem Set 6, Problem 6.2 (2017).}
  {eth-qft2-fs17}
  For
  \[
    sA_{a\mu}=(D_\mu\omega)_a,\quad
    s\omega_a=-\frac12C_{abc}\omega_b\omega_c,\quad
    s\omega_a^*=-h_a,\quad sh_a=0,
  \]
  prove \(s^2=0\) on each field without using any field equation.  Identify
  exactly where Grassmann antisymmetry and the Jacobi identity enter.

  \SupplementaryExercise{16}{A BRST-Exact Gauge-Fixing Sector}
  {Adapted from N. Beisert et al., ETH Zurich Quantum Field Theory II, Problem Set 6, Problem 6.2 (2017).}
  {eth-qft2-fs17}
  Let
  \[
    \Psi=-\int d^4x\,\omega_a^*
      \left(f_a[A]+\frac{\xi}{2}h_a\right).
  \]
  Calculate \(s\Psi\) with careful Grassmann signs.  Show both that it
  reproduces the ghost and auxiliary-field gauge-fixing action and that it
  is BRST invariant.  Finally eliminate \(h_a\).

  \SupplementaryExercise{17}{Conserved Ghost-Number Current}
  {Adapted from Carlo Becchi, Introduction to Gauge Theories, BRST and Ward Identities (1997).}
  {becchi-gauge-theories-1997}
  The Lorenz-gauge ghost Lagrangian is invariant under
  \(\omega_a\to e^{i\theta}\omega_a\),
  \(\omega_a^*\to e^{-i\theta}\omega_a^*\).
  Derive a Noether current, verify its conservation with the ghost field
  equations, and show that the corresponding charge assigns ghost numbers
  \(+1\) and \(-1\) to \(\omega\) and \(\omega^*\), respectively.

  \SupplementaryExercise{18}{Abelian BRST Ward Identity}
  {Adapted from David Skinner, Advanced Quantum Field Theory Example Sheet 4, Problem 3 (2018).}
  {skinner-aqft-sheet4-2018}
  For free electrodynamics use
  \[
    sA_\mu=\partial_\mu\omega,\qquad s\omega=0,\qquad
    s\omega^*=-h,\qquad sh=0.
  \]
  Apply BRST invariance of the vacuum to
  \(A_\mu(x)\omega^*(y)\).  Derive the resulting two-point Ward identity,
  eliminate \(h\), and verify the identity using the covariant-gauge photon
  and ghost propagators.

  \SupplementaryExercise{19}{Physical Polarizations from BRST Cohomology}
  {Adapted from Savdeep Sethi, University of Chicago Physics 445, Problem Set 5, Problem 1 (2021).}
  {uchicago-physics445-ps5-2021}
  In the free Abelian theory, determine the action of the BRST charge on
  one-particle states created by
  \[
    a_\mu^\dagger(p),\qquad
    \omega^\dagger(p),\qquad \omega^{*\dagger}(p).
  \]
  At ghost number zero, compute \(\ker Q/\operatorname{im}Q\) for
  \(p^2=0\) and show that it contains
  precisely the two transverse photon polarizations.

  \SupplementaryExercise{20}{Lie-Algebra BRST Differential}
  {Adapted from David Skinner, Advanced Quantum Field Theory Example Sheet 4, Problem 2 (2018).}
  {skinner-aqft-sheet4-2018}
  On polynomials of anticommuting variables \(\omega^a\), define
  \[
    Q=-\frac12C^a{}_{bc}\omega^b\omega^c
      \frac{\partial_L}{\partial\omega^a}.
  \]
  Compute \(Q^2\omega^a\) and prove that \(Q^2=0\) on every polynomial if
  and only if the structure constants satisfy the Jacobi identity.
  Interpret this complex as Lie-algebra cohomology.

  \SupplementaryExercise{21}{A Cubic Ghost Cocycle}
  {Adapted from David Skinner, Advanced Quantum Field Theory Example Sheet 4, Problem 2 (2018).}
  {skinner-aqft-sheet4-2018}
  For a compact simple Lie algebra with an invariant inner product, lower the
  first index on the structure constants and consider
  \[
    \Omega=\frac1{3!}C_{abc}\omega^a\omega^b\omega^c.
  \]
  Show that \(Q\Omega=0\).  Explain why this cocycle is not \(Q\)-exact,
  illustrating the argument explicitly for \(\mathfrak{su}(2)\).

  \SupplementaryExercise{22}{Elementary BV Antibrackets}
  {Adapted from J. Gomis, J. París, and S. Samuel, Antibracket, Antifields and Gauge-Theory Quantization, Section 4.1 (1995).}
  {gomis-paris-samuel-1995}
  Assign to every field \(\chi^n\) an antifield
  \(\chi_n^\ddagger\) of opposite Grassmann parity and ghost number
  \(-1-\operatorname{gh}(\chi^n)\).  From the definition
  \eqref{eq:15.9.13}, compute all elementary antibrackets among fields and
  antifields.  Then evaluate
  \((\chi^n\chi^m,\chi_k^\ddagger)\) for all parity choices and account for
  every sign.

  \SupplementaryExercise{23}{Graded Derivation Rules for the Antibracket}
  {Adapted from J. Gomis, J. París, and S. Samuel, Antibracket, Antifields and Gauge-Theory Quantization, Section 4.1 (1995).}
  {gomis-paris-samuel-1995}
  For homogeneous functionals \(F,G,H\), derive the two graded Leibniz rules
  for \((F,GH)\) and \((FG,H)\).  Use them to prove the graded symmetry
  relation
  \[
    (F,G)=-(-1)^{(\epsilon_F+1)(\epsilon_G+1)}(G,F).
  \]

  \SupplementaryExercise{24}{Jacobi Identity in BV Darboux Coordinates}
  {Adapted from J. Gomis, J. París, and S. Samuel, Antibracket, Antifields and Gauge-Theory Quantization, Section 4.1 (1995).}
  {gomis-paris-samuel-1995}
  Write the antibracket in Darboux coordinates as
  \((F,G)=F\overleftarrow\partial_AE^{AB}\overrightarrow\partial_BG\)
  with constant odd symplectic matrix \(E^{AB}\).  Expand the cyclic Jacobi
  combination, prove its cancellation, and explain what extra term would
  appear if \(E^{AB}\) were coordinate dependent.

  \SupplementaryExercise{25}{Minimal Yang--Mills BV Action}
  {\sloppy Adapted from Pavel Mnev, Lectures on Batalin--Vilkovisky Formalism, Sections 4.4--4.6 (2017).}
  {mnev-bv-2017}
  Introduce antifields for \(A^a_\mu\) and \(\omega^a\).  Construct the
  minimal BV action whose antifield derivatives generate the Yang--Mills
  BRST transformations.  Check its ghost number and parity, and verify the
  classical master equation by separating terms according to antifield
  number.

  \SupplementaryExercise{26}{Master Equation Encodes the Gauge Algebra}
  {Adapted from J. Gomis, J. París, and S. Samuel, Antibracket, Antifields and Gauge-Theory Quantization, Sections 3--4 (1995).}
  {gomis-paris-samuel-1995}
  Consider
  \[
    S=S_0[\phi]+\phi_r^\ddagger R^r{}_a(\phi)\omega^a
      -\frac12\omega_a^\ddagger C^a{}_{bc}(\phi)
       \omega^b\omega^c+\cdots .
  \]
  Expand \((S,S)=0\) by antifield number.  Show how its first three
  nontrivial conditions express gauge invariance of \(S_0\), closure of the
  gauge transformations, and the Jacobi identity.  State what the ellipsis
  must accommodate for an open algebra.

  \SupplementaryExercise{27}{Lorenz Gauge from a Gauge Fermion}
  {Adapted from J. Gomis, J. París, and S. Samuel, Antibracket, Antifields and Gauge-Theory Quantization, Section 6 (1995).}
  {gomis-paris-samuel-1995}
  Extend the minimal Yang--Mills BV action by the nonminimal pair
  \((\omega_a^*,h_a)\).  Choose
  \[
    \Psi=-\int d^4x\,\omega_a^*
      \left(\partial_\mu A_a^\mu+\frac{\xi}{2}h_a\right).
  \]
  Set every antifield equal to \(\delta\Psi/\delta\chi\), derive the
  gauge-fixed action, and integrate out \(h_a\).  Track the ghost term and
  all signs.

  \SupplementaryExercise{28}{Gauge-Fermion Shift Is Anticanonical}
  {\sloppy Adapted from Pavel Mnev, Lectures on Batalin--Vilkovisky Formalism, Sections 4.6--4.7 (2017).}
  {mnev-bv-2017}
  Let \(\Psi[\chi]\) be fermionic with ghost number \(-1\).  Prove that
  \[
    \chi^{\prime n}=\chi^n,\qquad
    \chi_n^{\ddagger\prime}
      =\chi_n^\ddagger-\frac{\delta\Psi}{\delta\chi^n}
  \]
  preserves every elementary antibracket.  Show that restricting
  \(\chi_n^{\ddagger\prime}=0\) gives the gauge-fixing prescription
  \(\chi_n^\ddagger=\delta\Psi/\delta\chi^n\).

  \SupplementaryExercise{29}{Nilpotency of the Quantum BRST Operator}
  {\sloppy Adapted from Pavel Mnev, Lectures on Batalin--Vilkovisky Formalism, Sections 4.7--4.8 (2017).}
  {mnev-bv-2017}
  Define
  \[
    \sigma F=(F,S)-i\Delta F,\qquad
    \Delta=\frac{\delta_R}{\delta\chi_n^\ddagger}
      \frac{\delta_L}{\delta\chi^n}.
  \]
  Derive \(\sigma^2F\) for arbitrary \(F\).  Prove that \(\sigma\) is
  nilpotent when \(S\) satisfies the quantum master equation, and state the
  corresponding condition for a quantum observable.

  \SupplementaryExercise{30}{Yang--Mills as a Consistent Deformation}
  {Adapted from Glenn Barnich, Friedemann Brandt, and Marc Henneaux, Local BRST Cohomology in Gauge Theories, Sections 2 and 12 (2000).}
  {barnich-brandt-henneaux-2000}
  Begin with \(N\) free Abelian vector fields and their minimal BV action.
  Seek a local, Poincaré-invariant deformation
  \(S=S_0+gS_1+g^2S_2+\cdots\) with at most one derivative in the cubic
  vertex.  Show that the first-order master equation permits the
  Yang--Mills cubic vertex with antisymmetric constants \(C_{abc}\), while
  the second-order equation requires the Jacobi identity.  Identify the
  quartic interaction supplied by \(S_2\).
\end{enumerate}
