\begin{enumerate}
  \SupplementarySolution{1}{Curvature from Parallel Transport}
  \begin{enumerate}
    \item[(a)]
    Acting on a test field,
    \[
      [D_\mu,D_\nu]\psi
        =-\ii\left(\partial_\mu A_\nu-\partial_\nu A_\mu
          -\ii[A_\mu,A_\nu]\right)\psi .
    \]
    Thus
    \[
      F_{\mu\nu}
        =\partial_\mu A_\nu-\partial_\nu A_\mu-\ii[A_\mu,A_\nu],
    \]
    which in components is Eq.~\eqref{eq:15.1.13}.  A finite gauge
    transformation is defined by requiring
    \(D'_\mu\psi'=UD_\mu\psi\) with \(\psi'=U\psi\); hence
    \(D'_\mu=UD_\mu U^{-1}\).  It follows without further expansion that
    \[
      [D'_\mu,D'_\nu]
        =U[D_\mu,D_\nu]U^{-1}
        =-\ii UF_{\mu\nu}U^{-1}.
    \]
    Comparing this with
    \([D'_\mu,D'_\nu]=-\ii F'_{\mu\nu}\) gives
    \(F'_{\mu\nu}=UF_{\mu\nu}U^{-1}\).
    \item[(b)]
    The transporter on a short displacement \(u^\mu\) from \(z\) is
    \[
      U(z+u,z)
        =\mathbf{1}+\ii u^\mu A_\mu(z)
         +\frac{\ii}{2}u^\mu u^\nu\partial_\mu A_\nu(z)
         -\frac12u^\mu u^\nu A_\mu(z)A_\nu(z)+O(u^3).
    \]
    Multiply the corresponding expressions for the four oriented sides in
    their traversal order.  The terms linear in \(u\) or \(v\) cancel.  The
    derivative terms give
    \(\ii(\partial_\mu A_\nu-\partial_\nu A_\mu)u^\mu v^\nu\), while the
    products whose order differs on the two routes give
    \([A_\mu,A_\nu]u^\mu v^\nu\).  Therefore
    \[
      U_\square
        =\mathbf{1}+\ii\bigl(\partial_\mu A_\nu-\partial_\nu A_\mu
          -\ii[A_\mu,A_\nu]\bigr)u^\mu v^\nu+O(\ell^3)
        =\mathbf{1}+\ii F_{\mu\nu}u^\mu v^\nu+O(\ell^3).
    \]
    Replacing the ordered product by an ordinary exponential would
    symmetrize the matrix factors and lose the commutator; path ordering is
    precisely what retains the non-Abelian part of the curvature.
  \end{enumerate}

  \SupplementarySolution{2}{Yang--Mills Equation and Covariant Conservation}
  \begin{enumerate}
    \item[(a)]
    Applying a covariant divergence to the field equation gives
    \[
      D_\nu J^\nu=D_\nu D_\mu F^{\mu\nu}
        =\frac12[D_\nu,D_\mu]F^{\mu\nu}.
    \]
    In the adjoint representation,
    \([D_\nu,D_\mu]X=-\ii[F_{\nu\mu},X]\), up to the convention for where
    \(g\) is absorbed.  Therefore
    \[
      D_\nu J^\nu
        =-\frac{\ii}{2}[F_{\nu\mu},F^{\mu\nu}]=0.
    \]
    The last contraction vanishes because interchanging the two field
    strengths reverses the Lie bracket while leaving the summed Lorentz
    indices unchanged.  Thus the Yang--Mills equation requires the
    covariant, rather than ordinary, conservation law
    \[
      \boxed{D_\nu J^\nu=0.}
    \]
    \item[(b)]
    Since \(\Psi'=U\Psi\) and
    \(\overline\Psi'=\overline\Psi U^{-1}\),
    \[
      J'^\nu
        =g\overline\Psi U^{-1}\gamma^\nu T^aU\Psi\,T^a
        =UJ^\nu U^{-1},
    \]
    where the final equality is the matrix form of the adjoint action on
    the components.

    The transformed covariant derivative satisfies
    \(D'_\mu=UD_\mu U^{-1}\), while
    \(F'^{\mu\nu}=UF^{\mu\nu}U^{-1}\).  Acting on the curvature therefore
    gives
    \[
      D'_\mu F'^{\mu\nu}
        =U(D_\mu F^{\mu\nu})U^{-1}.
    \]
    Both sides of the field equation transform homogeneously in the
    adjoint representation, so its covariance follows from the independent
    transformations of the matter current and curvature.
  \end{enumerate}

  \SupplementarySolution{3}{Equal Curvature Is Not Gauge Equivalence}
  \begin{enumerate}
    \item[(a)]
    If Abelian connections \(A\) and \(A'\) have the same curvature, then
    \(d(A'-A)=0\).  The Poincar\'e lemma on a contractible patch gives
    \(A'-A=\dd\lambda\), which is precisely a local Abelian gauge
    transformation.  The corresponding non-Abelian equation contains a
    commutator and does not reduce to the statement that the difference is
    a closed one-form.
    \item[(b)]
    Since the first connection is proportional to \(T_3\), its components
    commute.  Therefore
    \[
      F_{xy}=\partial_x\left(\frac{gx}{2}T_3\right)
       -\partial_y\left(-\frac{gy}{2}T_3\right)=gT_3.
    \]
    For the constant second connection the derivative terms vanish, while
    \[
      F'_{xy}=-ig[T_1,T_2]
        =-ig(\ii T_3)=gT_3.
    \]
    Thus the two curvatures agree pointwise.
    \item[(c)]
    The covariant derivative of the curvature transforms by conjugation.
    For the first connection it vanishes,
    \[
      D_xF_{xy}=D_yF_{xy}=0,
    \]
    because both \(A_\mu\) and \(F_{xy}\) lie in the \(T_3\) direction.
    For the second connection,
    \[
      D'_xF'_{xy}=-ig[T_1,gT_3]=-g^2T_2,
      \qquad
      D'_yF'_{xy}=-ig[T_2,gT_3]=g^2T_1.
    \]
    With \(\operatorname{tr}(T_aT_b)=\delta_{ab}/2\), the local
    gauge-invariant scalar
    \[
      \mathcal I=\operatorname{tr}\!\left[
        (D_xF_{xy})^2+(D_yF_{xy})^2\right]
    \]
    is zero for the first connection and \(g^4\) for the second.  Gauge
    transformations cannot change a conjugation-invariant scalar, so one
    such mismatch is enough to rule out gauge equivalence.
  \end{enumerate}

  \SupplementarySolution{4}{Gauge Transport and Curvature}
  \begin{enumerate}
    \item[(a)] Endpoint factors cancel around the quadrangle, so its
      ordered product transforms by conjugation at the base point.  Applying
      the short-segment expansion on each edge, including the translated
      connection, gives
      \[
        \mathcal W(x)=\mathbf{1}+\ii\epsilon^2m^\mu n^\nu
        \bigl(\partial_\mu A_\nu-\partial_\nu A_\mu
              \ii[A_\mu,A_\nu]\bigr)+O(\epsilon^3).
      \]
      The bracket is \(F_{\mu\nu}\), up to the orientation and covariant
      derivative conventions.
    \item[(b)] Commutativity removes path ordering in the Abelian case.
      Thus, for \(C=\partial\Sigma\),
      \[
        U[C]=\exp\left(\ii\oint_C A\right)
          =\exp\left(\ii\int_\Sigma F\right).
      \]
      Two surfaces give the same result when their union bounds a region
      with \(\dd F=0\); magnetic sources and nontrivial topology are the
      standard qualifications.
    \item[(c)] Put
      \(I(t,s)=\int_s^t \dd\tau\,\dot y^\mu A_\mu(y(\tau))\).
      Differentiating \(U=e^{\ii I}\) gives
      \[
        \partial_tU=\ii\dot y(t)\cdot A(y(t))U,\qquad
        \partial_sU=-\ii U\dot y(s)\cdot A(y(s)).
      \]
    \item[(d)] The ordered expansion is
      \[
        U(t,s)=\mathbf{1}+\sum_{r\geq1}i^r
        \int_{s<\tau_r<\cdots<\tau_1<t}
        d^r\tau\,A(\tau_1)\cdots A(\tau_r).
      \]
      Differentiating the upper boundary inserts \(A(t)\) on the left;
      differentiating the lower boundary inserts \(A(s)\) on the right
      with a minus sign.  These are precisely the two endpoint equations.
    \item[(e)] Let
      \(U'(t,s)=V(y(t))U(t,s)V^{-1}(y(s))\).  The upper endpoint equation
      is preserved provided
      \[
        A'_\mu=VA_\mu V^{-1}-\ii(\partial_\mu V)V^{-1}.
      \]
      The lower equation yields the same law and the correct right endpoint
      multiplication.  Hence the complete pair of transport equations is
      gauge covariant.
  \end{enumerate}

  \SupplementarySolution{5}{Nonlinear Lorenz and Axial Gauges}
  \begin{enumerate}
    \item[(a)] The infinitesimal Abelian gauge change gives
      \[
        \delta G=(\partial^2+\zeta A^\mu\partial_\mu)\lambda,
      \]
      so \(M=\partial^2+\zeta A\cdot\partial\).  Exponentiating
      \(\det M\) and averaging \(\Omega\) gives
      \[
        \mathcal L_{\rm gf+gh}
        =-\frac1{2\xi}
          \left(\partial\cdot A+\frac{\zeta}{2}A^2\right)^2
          \omega^*(\partial^2+\zeta A\cdot\partial)\omega .
      \]
      Its quadratic inverse is
      \[
        D_{\mu\nu}(p)=\frac{-\ii}{p^2+\ii0}
          \left[\eta_{\mu\nu}-(1-\xi)\frac{p_\mu p_\nu}{p^2}\right],
        \qquad D_{\rm gh}(p)=\frac{\ii}{p^2+\ii0}.
      \]
      The nonlinear gauge also produces
      \(\zeta\omega^*A^\mu\partial_\mu\omega\),
      \(-\zeta(\partial\cdot A)A^2/(2\xi)\), and
      \(-\zeta^2(A^2)^2/(8\xi)\).  The field-dependent ghost operator
      prevents Abelian ghost decoupling.  All three extra vertices vanish
      when \(\zeta\to0\), restoring the usual covariant gauge.
    \item[(b)] In Yang--Mills theory
      \[
        \delta(n\cdot A)^a=(n\cdot D)^{ab}\lambda^b,
        \qquad M^{ab}=(n\cdot D)^{ab}.
      \]
      Inverting the quadratic kernel gives, in one common metric convention,
      \[
        D_{\mu\nu}^{ab}(p)=\frac{-\ii\delta^{ab}}{p^2+\ii0}
        \left[\eta_{\mu\nu}
          \frac{(\xi p^2+n^2)p_\mu p_\nu}{(p\cdot n)^2}
          \frac{p_\mu n_\nu+n_\mu p_\nu}{p\cdot n}\right].
      \]
      Reversing the sign used for the gauge kinetic operator reverses the
      last two displayed signs; multiplying by that operator is the
      convention-independent check.  The ghost inverse is \(n\cdot p\),
      and its only gauge interaction is proportional to
      \(C^{abc}\omega_a^*(n\cdot A_b)\omega_c\).  On the sharp slice
      \(n\cdot A=0\), \(M=n\cdot\partial\) is field independent and the
      ghosts may be omitted.  One must still fix residual
      \(n\cdot\partial\lambda=0\) transformations and prescribe the axial
      poles \(1/(p\cdot n)\).
  \end{enumerate}

  \SupplementarySolution{6}{BRST Nilpotency and Gauge-Fermion Fixing}
  \begin{enumerate}
    \item[(a)]
    Write \(\omega=\omega^aT_a\).  Grassmann antisymmetry removes the
    symmetric generator product, so
    \[
      \omega^2=\frac12\omega^a\omega^b[T_a,T_b]
        =\frac{\ii}{2}f_{ab}{}^c\omega^a\omega^bT_c.
    \]
    The graded Leibniz rule then gives
    \[
      s^2\psi
       =ig\left(s\omega^c
          +\frac g2f_{ab}{}^c\omega^a\omega^b\right)T_c\psi.
    \]
    Hence nilpotency on every matter field requires
    \[
      \boxed{s\omega^c=-\frac g2f_{ab}{}^c\omega^a\omega^b.}
    \]
    \item[(b)]
    Applying \(s\) once more to the ghost gives
    \[
      s^2\omega^a
       =\frac{g^2}{4}
        \left(f^a{}_{bc}f^b{}_{\dd e}
             +f^a{}_{bd}f^b{}_{ec}
             +f^a{}_{be}f^b{}_{cd}\right)
        \omega^c\omega^d\omega^e=0.
    \]
    The three-ghost product performs the required total antisymmetrization;
    the coefficient vanishes by Jacobi.  Similarly,
    \[
      s^2A^a_\mu
       =\partial_\mu(s\omega^a)
        +gf^a{}_{bc}
          \left[(sA^b_\mu)\omega^c+A^b_\mu(s\omega^c)\right].
    \]
    Substitution first cancels every term containing
    \(\partial_\mu\omega\); the remaining \(A\omega\omega\) coefficient is
    the same Jacobi combination.  Thus \(s^2A^a_\mu=0\), without field
    equations.
    \item[(c)]
    Since \(\omega_a^*\) is odd, \(s\omega_a^*=-h_a\), and \(sh_a=0\),
    \[
      s\Psi=\int \dd^4x\left[
        \omega_a^*\partial^\mu(D_\mu\omega)_a
        +h_a\partial^\mu A^a_\mu+\frac{\xi}{2}h_ah_a\right].
    \]
    These are respectively the Faddeev--Popov ghost term and the auxiliary
    form of Lorenz gauge fixing.  Nilpotency gives \(s(s\Psi)=0\)
    immediately.  The algebraic field equation
    \(h_a=-\partial^\mu A_{a\mu}/\xi\) reduces the gauge-fixing sector to
    \[
      \int \dd^4x\left[
        \omega_a^*\partial^\mu(D_\mu\omega)_a
        -\frac{(\partial^\mu A_{a\mu})^2}{2\xi}\right].
    \]
    The overall minus sign in \(\Psi\) is tied to the local convention
    \(s\omega^*=-h\); reversing both signs leaves \(s\Psi\) unchanged.
  \end{enumerate}

  \SupplementarySolution{7}{Matter, an Adjoint Scalar, and BRST in \(SU(2)\)}
  Write \(A_\mu=A_\mu^aT^a\) and
  \(F_{\mu\nu}=\partial_\mu A_\nu-\partial_\nu A_\mu
  -ig[A_\mu,A_\nu]\).
  \begin{enumerate}
    \item[(a)] The covariant derivatives are
      \[
        D_\mu\psi=(\partial_\mu-igA_\mu)\psi,
        \qquad
        D_\mu\bar\psi=\partial_\mu\bar\psi+ig\bar\psi A_\mu,
      \]
      and
      \[
        D_\mu\phi=\partial_\mu\phi-ig[A_\mu,\phi],
        \qquad
        (D_\mu\phi)^a=\partial_\mu\phi^a
          +g\epsilon^{abc}A_\mu^b\phi^c.
      \]
      Requiring \(D_\mu\psi\mapsto U D_\mu\psi\) gives
      \[
        \boxed{A_\mu\longmapsto
          UA_\mu U^{-1}-\frac{\ii}{g}(\partial_\mu U)U^{-1}.}
      \]
      Substitution then gives
      \(D_\mu\bar\psi\mapsto(D_\mu\bar\psi)U^{-1}\) and
      \(D_\mu\phi\mapsto U(D_\mu\phi)U^{-1}\).  The same calculation or
      the commutator \([D_\mu,D_\nu]\) gives
      \(F_{\mu\nu}\mapsto UF_{\mu\nu}U^{-1}\).

    \item[(b)] With real couplings and the usual conserved Dirac fermion
      number and parity, the complete power-counting-renormalizable
      Euclidean Lagrangian can be written
      \[
        \begin{split}
        \mathcal L_E={}&\frac14F^a_{\mu\nu}F^a_{\mu\nu}
          +\bar\psi(\gamma_\mu D_\mu+m)\psi
          +\frac12(D_\mu\phi)^a(D_\mu\phi)^a
          +\frac12M^2\phi^a\phi^a\\
          &+\frac\lambda4(\phi^a\phi^a)^2
          +y\,\bar\psi\phi^aT^a\psi .
        \end{split}
      \]
      Cyclicity of the trace proves invariance of the gauge kinetic term:
      \[
        \operatorname{tr}(UFU^{-1}UFU^{-1})
          =\operatorname{tr}F^2.
      \]
      It proves invariance of the scalar kinetic term and potential in the
      same way.  The fermion kinetic and mass
      terms are invariant because \(D\psi\mapsto UD\psi\) and
      \(\bar\psi\mapsto\bar\psi U^{-1}\).  Finally,
      \[
        \bar\psi\phi\psi\longmapsto
        \bar\psi U^{-1}(U\phi U^{-1})U\psi
        =\bar\psi\phi\psi.
      \]
      There is no independent adjoint-scalar cubic: for
      \(\phi=\phi^a\sigma^a/2\),
      \(\operatorname{tr}\phi^3=0\), and
      \(\operatorname{tr}\phi^4=\tfrac12(\operatorname{tr}\phi^2)^2\).
      If parity or CP is not imposed, one may also include the topological
      density \(\ii\theta F^a_{\mu\nu}\widetilde F^a_{\mu\nu}\) and
      independent pseudoscalar mass and Yukawa terms
      \(i m_5\bar\psi\gamma_5\psi\) and
      \(\ii y_5\bar\psi\gamma_5\phi\psi\).  These do not alter the gauge
      invariance argument.

    \item[(c)] Gauge-related configurations are redundant directions of the
      quadratic gauge-field operator, so without gauge fixing that operator
      has zero modes and no propagator.  In the functional integral, imposing
      a gauge condition produces the Faddeev--Popov determinant.  A pair of
      Grassmann scalar fields \(c,\bar c\) exponentiates that determinant.
      A bosonic Nakanishi--Lautrup field \(B\) rewrites the quadratic
      gauge-fixing weight in a local form linear in the gauge condition; its
      algebraic equation of motion returns the usual squared gauge-fixing
      term.  This enlarged field content makes off-shell BRST nilpotence
      manifest.

    \item[(d)] Remove the parameter by writing
      \(\delta_B=\eta s\).  In vector notation in the adjoint,
      \[
        sA_\mu=D_\mu c,\qquad
        sc=-\frac g2[c,c],\qquad
        s\bar c=B,\qquad sB=0,
      \]
      where the bracket includes the structure constants and the
      Grassmann coefficients.  The last two equations immediately give
      \(s^2\bar c=s^2B=0\).  For the ghost,
      \[
        (s^2c)^a
          =\frac{g^2}{4}
            \epsilon^{a}{}_{e[b}\epsilon^{e}{}_{cd]}
            c^bc^cc^d=0,
      \]
      by the Jacobi identity.  For the gauge field,
      \[
        s^2A_\mu=D_\mu(sc)+g[sA_\mu,c]
          =-\frac g2D_\mu[c,c]+g[D_\mu c,c]=0,
      \]
      because the covariant derivative is a derivation and the two ghost
      factors are odd.

      The BRST operator is itself an odd derivation.  Hence for homogeneous
      fields \(X,Y\),
      \[
        s^2(XY)=(s^2X)Y+X(s^2Y),
      \]
      since the two cross terms cancel.  Induction proves \(s^2O=0\) for
      every polynomial \(O(A_\mu,c,\bar c,B)\).  With independent
      parameters, the successive variation is proportional to
      \(\eta_2\eta_1s^2O\), so the conclusion does not rely on setting a
      single \(\eta^2\) to zero.
  \end{enumerate}

  \SupplementarySolution{8}{The Complete Free Abelian BRST Multiplet}
  \begin{enumerate}
    \item[(a)] The Maxwell term is gauge invariant.  The remaining variation
      is
      \[
        s(h\,\partial\cdot A)+s(\omega^*\partial^2\omega)
        =h\partial^2\omega-h\partial^2\omega=0.
      \]
      Also \(s^2A_\mu=\partial_\mu s\omega=0\),
      \(s^2\omega^*=-sh=0\), and nilpotence on \(\omega,h\) is immediate.
    \item[(b)] After integrating derivatives by parts,
      \[
        S=-\frac12\int d^Dx\,\Phi^{\mathsf T}
        \begin{pmatrix}
          -\partial^2\eta^{\mu\nu}+\partial^\mu\partial^\nu
            &\partial^\mu\\
          -\partial^\nu&-\xi
        \end{pmatrix}\Phi
        +\int d^Dx\,\omega^*\partial^2\omega .
      \]
      The opposite off-diagonal signs arise from the integration by parts
      and make the block operator transpose correctly.
    \item[(c)] Acting with the quadratic operators on normalized
      time-ordered correlators gives delta functions.  Fourier inversion
      yields, with the source's overall convention,
      \[
        \int d^Dx\,e^{-\ii px}\langle\Phi(x)\Phi^{\mathsf T}(0)\rangle
        =-\frac1{p^2}
        \begin{pmatrix}
          \ii\!\left[\eta_{\mu\nu}
             -(1-\xi)\dfrac{p_\mu p_\nu}{p^2}\right]&\ii p_\mu\\
          -\ii p_\nu&0
        \end{pmatrix},
      \]
      and
      \[
        \int d^Dx\,e^{-\ii px}
          \langle\omega(x)\omega^*(0)\rangle=-\frac{\ii}{p^2}.
      \]
      Direct multiplication by the matrix in part (b) gives the identity
      block and fixes every mixed sign.
    \item[(d)] Since \(sh=0\),
      \[
        0=\langle s[h(x)\omega^*(0)]\rangle
          =-\langle h(x)h(0)\rangle .
      \]
      Likewise,
      \[
        0=\langle s[A_\mu(x)\omega^*(0)]\rangle
          =\langle\partial_\mu\omega(x)\omega^*(0)\rangle
           -\langle A_\mu(x)h(0)\rangle .
      \]
      The \(hh\) entry above is zero, while the mixed entry and the
      derivative of \(-\ii/p^2\) agree exactly, verifying both BRST Ward
      identities.
  \end{enumerate}

  \SupplementarySolution{9}{Lie-Algebra BRST Cohomology}
  \begin{enumerate}
    \item[(a)]
    Decompose \(Q=\rho+d_{\rm CE}\), where
    \(\rho=\omega^aR_a\) acts on the representation factor and
    \(d_{\rm CE}\omega^a=-\tfrac12C^a{}_{bc}\omega^b\omega^c\).
    For a constant representation vector \(v\), the graded Leibniz rule
    gives
    \[
      Q^2v
       =-\frac12C^c{}_{ab}\omega^a\omega^bR_cv
        -\omega^a\omega^bR_bR_av.
    \]
    Anticommutativity retains only the commutator in the second term:
    \[
      -\omega^a\omega^bR_bR_a
        =\frac12\omega^a\omega^b[R_a,R_b]
        =\frac12C^c{}_{ab}\omega^a\omega^bR_c,
    \]
    so the representation-dependent terms cancel.  On a ghost generator,
    \[
      Q^2\omega^a
        =\frac16C^a{}_{e[b}C^e{}_{cd]}
          \omega^b\omega^c\omega^d=0,
    \]
    by the Jacobi identity.  Since \(Q\) is an odd derivation, it therefore
    squares to zero on every \(R\)-valued polynomial.
    \item[(b)]
    In the trivial representation, the polynomial algebra is
    \(\Lambda^\bullet\mathfrak g^*\), and \(Q\) is the
    Chevalley--Eilenberg differential.  Applying it to the three-cochain
    produces four ghosts:
    \[
      Q\Omega
        =-\frac1{12}C_{abc}C^a{}_{\dd e}
          \omega^d\omega^e\omega^b\omega^c+\text{cyclic terms}.
    \]
    After total antisymmetrization, its coefficient is the invariant-metric
    form of the Jacobi identity,
    \[
      C_{a[bc}C^a{}_{\dd e]}=0,
    \]
    so \(Q\Omega=0\).

    This is the Cartan three-cocycle and represents the nonzero generator of
    \(H^3(\mathfrak g)\) for each compact simple factor.  For
    \(\mathfrak{su}(2)\), choose \(C_{abc}=\epsilon_{abc}\).  Then
    \(\Omega=\omega^1\omega^2\omega^3\), apart from normalization, and
    \[
      Q\omega^1=-\omega^2\omega^3,\quad
      Q\omega^2=-\omega^3\omega^1,\quad
      Q\omega^3=-\omega^1\omega^2.
    \]
    Every quadratic monomial has zero \(Q\)-image because a ghost is repeated.
    Therefore no degree-two polynomial can map to \(\Omega\), proving
    non-exactness explicitly.  Restriction to the \(SU(2)\) subgroup
    associated with a root gives the same obstruction in the general compact
    simple case.
  \end{enumerate}

  \SupplementarySolution{10}{Auxiliary-Field Gauge Fixing}
  \begin{enumerate}
    \item[(a)] Completing the square gives
      \[
        h\,\partial\cdot A+\frac{\xi}{2}h^2
        =\frac{\xi}{2}\left(h+\frac{\partial\cdot A}{\xi}\right)^2
          -\frac{(\partial\cdot A)^2}{2\xi}.
      \]
      Translation invariance of \(Dh\) leaves a field-independent
      determinant and produces the standard covariant-gauge term.  The
      free Abelian ghost determinant is likewise field independent.
    \item[(b)] The Maxwell term is invariant because
      \(sF_{\mu\nu}=0\).  The matter transformation is an infinitesimal
      gauge transformation, so
      \(s[\bar\psi(\ii\slashed D-m)\psi]=0\).  Finally,
      \[
        s(h\partial\cdot A)=h\partial^2\omega,\qquad
        s[\omega^*(-\partial^2)\omega]
          =-h\partial^2\omega ,
      \]
      while \(s(h^2)=0\).  The two remaining variations cancel.
    \item[(c)] Clearly \(s^2A_\mu=s^2\omega=s^2\omega^*=s^2h=0\).
      For the fermion,
      \[
        s^2\psi=-ie[(s\omega)\psi-\omega s\psi]
          =e^2\omega^2\psi=0,
      \]
      because \(\omega\) is Grassmann odd; the same calculation applies
      to \(\bar\psi\).  Thus nilpotence is algebraic and off shell.
  \end{enumerate}

  \SupplementarySolution{11}{Lorenz and Axial BRST Gauges at One Loop}
  \begin{enumerate}
    \item[(a)] The finite form of the transformation is
      \[
        A_\mu' =UA_\mu U^{-1}-\frac{\ii}{g}(\partial_\mu U)U^{-1}.
      \]
      Therefore \(D_\mu'=UD_\mu U^{-1}\), and the commutator of two
      covariant derivatives gives
      \[
        F_{\mu\nu}'=UF_{\mu\nu}U^{-1}.
      \]
      Infinitesimally,
      \(\delta F_{\mu\nu}=\ii[\alpha,F_{\mu\nu}]\).  Thus
      \[
        \delta\operatorname{tr}(F_{\mu\nu}F^{\mu\nu})
          =\ii\operatorname{tr}
            ([\alpha,F_{\mu\nu}]F^{\mu\nu}
             +F_{\mu\nu}[\alpha,F^{\mu\nu}])=0,
      \]
      by cyclicity.  Since
      \(\operatorname{tr}F^2=\tfrac12F^aF^a\) for
      \(\operatorname{tr}(T^aT^b)=\delta^{ab}/2\), this proves the stated
      invariance.

    \item[(b)] In components the transformations are
      \[
        sA_\mu^a=\partial_\mu c^a+gf^{abc}A_\mu^bc^c,
        \qquad
        sc^a=-\frac g2f^{abc}c^bc^c,
        \qquad s\bar c^a=B^a,\qquad sB^a=0.
      \]
      Nilpotence on \(\bar c\) and \(B\) is immediate.  On the ghost,
      total antisymmetrization by \(c^bc^cc^d\) gives
      \[
        s^2c^a=\frac{g^2}{4}f^a{}_{e[b}f^e{}_{cd]}
          c^bc^cc^d=0.
      \]
      On the gauge field, the graded Leibniz rule gives
      \[
        s^2A_\mu^a
          =\partial_\mu(sc^a)+gf^{abc}
             \bigl[(sA_\mu^b)c^c+A_\mu^b(sc^c)\bigr].
      \]
      The terms containing \(\partial_\mu c\) cancel pairwise.  The
      remaining coefficient of \(A_\mu^bc^cc^d\) is the same Jacobi
      combination as above, so \(s^2A_\mu^a=0\).

    \item[(c)] The Yang--Mills term is BRST invariant because a BRST
      transformation is a gauge transformation with an odd parameter.
      Nilpotence also gives
      \(s^2\{\bar c^a(L[A^a]-\xi B^a/2)\}=0\).  Therefore
      \(s\mathcal L=0\).  Applying the odd product rule once gives
      \[
        \boxed{\mathcal L=\frac14F^a_{\mu\nu}F^{a\mu\nu}
          +B^aL[A^a]-\frac\xi2B^aB^a
          -\bar c^aL[(D c)^a].}
      \]
      The algebraic equation \(B^a=L[A^a]/\xi\) yields the equivalent
      Euclidean gauge-fixing term \((L[A^a])^2/(2\xi)\).

    \item[(d)] For Lorenz gauge, the terms after eliminating \(B\) are
      \[
        \frac{1}{2\xi}(\partial^\mu A_\mu^a)^2,
        \qquad
        -\bar c^a\partial^\mu(D_\mu c)^a.
      \]
      The first makes the gauge-field quadratic operator invertible; the
      second is the Faddeev--Popov determinant and contains the
      ghost--gluon vertex.  The nonzero one-loop two-point graphs are the
      gluon bubble with two cubic vertices, the gluon tadpole from one
      quartic vertex, and the ghost bubble:

      the ghost operator is
      \[
        n^\mu D_\mu^{ab}
          =n\cdot\partial\,\delta^{ab}
             +gf^{acb}n\cdot A^c.
      \]
      On the strict axial-gauge slice, the second term vanishes.  The
      determinant is then field independent, so the ghosts decouple.
      Equivalently, the putative ghost vertex is proportional to \(n^\mu\)
      and vanishes against the axial propagator.  Consequently only the

      the ghost bubble is absent.
  \end{enumerate}

  \SupplementarySolution{12}{The Free QED BRST Charge}
  \begin{enumerate}
    \item[(a)] The gauge-invariant part changes by an ordinary gauge
      transformation.  In Lorenz gauge, the variation of the gauge-fixing
      term is canceled by that of the ghosts after one integration by
      parts.  A convenient free-theory Noether current, differing from
      other choices only by improvements and equations of motion, is
      \[
        j_Q^\mu=-(\partial_\nu A^\nu)\partial^\mu\omega
          +F^{\mu\nu}\partial_\nu\omega .
      \]
      The free equations give \(\partial_\mu j_Q^\mu=0\), and
      \(Q=\int \dd^3x\,j_Q^0\) generates the stated graded transformations.
    \item[(b)] Inserting the standard massless mode expansions and using
      the canonical brackets gives, up to a common normalization and a
      harmless overall sign,
      \[
        Q=\int\frac{\dd^3p}{(2\pi)^3\,2|\mathbf p|}
        \left[
          \omega^\dagger(\mathbf p)\,p^\mu a_\mu(\mathbf p)
          +p^\mu a_\mu^\dagger(\mathbf p)\,\omega(\mathbf p)
        \right].
      \]
      Hermiticity fixes the relative coefficient of the two terms.
    \item[(c)] Taking graded commutators with this expression yields
      \[
        [Q,a_\mu]=-p_\mu\omega,\quad
        [Q,a_\mu^\dagger]=p_\mu\omega^\dagger,\quad
        \{Q,\omega^*\}=p\cdot a,\quad
        \{Q,\omega^{*\dagger}\}=p\cdot a^\dagger,
      \]
      and \(Q\) anticommutes with both \(\omega\) and
      \(\omega^\dagger\).  Hence a state
      \(\epsilon^\mu a_\mu^\dagger\ket0\) is closed when
      \(p\cdot\epsilon=0\), while \(\epsilon^\mu\sim
      \epsilon^\mu+\alpha p^\mu\) differs by an exact state.  For null
      \(p\), the quotient has the two transverse photon polarizations.
  \end{enumerate}

  \SupplementarySolution{13}{BRST Gauge Independence and Physical States}
  \begin{enumerate}
    \item[(a)] First use the transformations as printed in the question;
      the sign of \(Q\bar c\) is immaterial for nilpotence.  For the ghost,
      \[
        Q^2c^a=\frac{g^2}{4}f^a{}_{e[b}f^e{}_{cd]}
          c^bc^cc^d=0,
      \]
      where the three ghosts perform the total antisymmetrization and the
      coefficient vanishes by Jacobi.  For the gauge field,
      \[
        Q^2A_\mu^a
          =\partial_\mu(Qc^a)+gf^a{}_{bc}
            \bigl[(QA_\mu^b)c^c+A_\mu^b(Qc^c)\bigr].
      \]
      Terms containing \(\partial_\mu c\) cancel by antisymmetry; the
      remaining \(A_\mu c c\) coefficient is again the Jacobi identity.
      Thus \(Q^2A_\mu^a=0\).

      On the fermion, the odd Leibniz rule gives
      \[
        Q^2\psi=ig\left[(Qc^a)T_a\psi-c^aT_aQ\psi\right].
      \]
      Only the commutator part of \(T_aT_b\) survives contraction with
      \(c^ac^b\), and
      \([T_a,T_b]=if_{ab}{}^cT_c\) cancels the term containing \(Qc\).
      Hence \(Q^2\psi=0\); the row-spinor calculation gives
      \(Q^2\bar\psi=0\) in the same way.  Finally,
      \(Q^2\bar c=QB=0\) and \(Q^2B=0\).  Nilpotence is off shell for every
      field.

    \item[(b)] We now adopt the explicit convention announced in the
      editorial note, \(Q\bar c_a=-B_a\), while retaining the displayed
      positive ghost term.  The gauge-invariant part
      \[
        \mathcal L_0=-\frac14F^a_{\mu\nu}F_a^{\mu\nu}
          +\bar\psi(\ii\slashed D-m)\psi
      \]
      obeys \(Q\mathcal L_0=0\).  Moreover,
      \[
        QG^a=\frac{\delta G^a}{\delta A_\mu^b}
          D_\mu^{bc}c_c.
      \]
      Since \(\bar c\) is odd and \(B\) is even,
      \[
        \begin{split}
        -Q\Psi
          &=-Q\!\left[\bar c_a
             \left(G^a-\frac\xi2B^a\right)\right]\\
          &=B_aG^a-\frac\xi2B_aB^a
            +\bar c_a\frac{\delta G^a}{\delta A_\mu^b}
              D_\mu^{bc}c_c.
        \end{split}
      \]
      Thus the full displayed Lagrangian is
      \(\mathcal L=\mathcal L_0-Q\Psi\), and
      \[
        Q\mathcal L=Q\mathcal L_0-Q^2\Psi=0.
      \]
      If instead one keeps the source rule \(Q\bar c=B\), the same proof
      goes through after reversing the ghost-term sign.  This is precisely
      the equivalent convention recorded after the exercise.

    \item[(c)] Let \(\delta\Psi\) be an arbitrary infinitesimal change of
      gauge fixing.  Absorb the inessential overall sign from part~(b) into
      the definition of \(\Psi\).  To first order, the amplitude changes by
      an insertion
      \[
        \delta_\Psi\braket{f}{\ii}
          =\ii\bra{f}
             \int \dd^4x\,\{\widehat Q,\delta\Psi\}
            \ket{\ii} .
      \]
      If
      \(\widehat Q\ket{\ii}=0\) and
      \(\bra{f}\widehat Q=0\), the two terms in the anticommutator
      vanish separately, for every \(\delta\Psi\).  If
      \(\widehat Q\) is Hermitian, the second condition is equivalent to
      \(\widehat Q\ket{f}=0\), which is the form stated in the problem.

      Conversely, because the gauge-fixing change is arbitrary, a state on
      which \(\widehat Q\) acts nontrivially would in general give a
      gauge-dependent transition amplitude.  Physical external states must
      therefore be BRST closed:
      \[
        \boxed{\widehat Q\ket{\mathrm{phys}}=0.}
      \]
      Nilpotence, \(\widehat Q^2=0\), also means that adding an exact state
      \(\widehat Q\ket{\chi}\) does not change its matrix elements with
      closed states.  Thus the physical state space is the BRST cohomology,
      the kernel of \(\widehat Q\) modulo its image.
  \end{enumerate}

  \SupplementarySolution{14}{Minimal and Quantum BV Constructions}
  \begin{enumerate}
    \item[(a)]
    With the ordering convention of Eq.~\eqref{eq:15.9.1}, the minimal action
    is
    \[
      S_{\min}=S_{\rm YM}
        +\int \dd^4x\left[
          (D_\mu\omega)_a A_a^{\ddagger\mu}
          -\frac12C_{abc}\omega_b\omega_c\omega_a^\ddagger
        \right].
    \]
    Here
    \[
      \operatorname{gh}(A^\ddagger)=-1,\quad
      \operatorname{gh}(\omega^\ddagger)=-2,
    \]
    while \(A^\ddagger\) is odd and \(\omega^\ddagger\) is even.  Every term
    in \(S_{\min}\) is therefore even and has ghost number zero.

    Sort \((S_{\min},S_{\min})\) by antifield number.  The term with no
    antifield is twice the gauge variation of \(S_{\rm YM}\), hence vanishes.
    Terms linear in \(A^\ddagger\) state
    \(s^2A_\mu=0\); terms linear in \(\omega^\ddagger\) state
    \(s^2\omega=0\).  The former is closure of the Yang--Mills gauge
    transformations, and the latter is the Jacobi identity.  These are
    exactly the calculations in Supplementary Exercise 15, so all
    coefficients vanish and \((S_{\min},S_{\min})=0\).
    \item[(b)]
    Because \(\Psi\) depends only on fields,
    \[
      (\chi^{\prime n},\chi^{\prime m})=0.
    \]
    Also,
    \[
      (\chi^{\prime n},\chi_m^{\ddagger\prime})
        =(\chi^n,\chi_m^\ddagger)
         -\left(\chi^n,\frac{\delta\Psi}{\delta\chi^m}\right)
        =\delta^n_m.
    \]
    Finally,
    \[
      (\chi_n^{\ddagger\prime},\chi_m^{\ddagger\prime})
        =-\frac{\delta^2\Psi}{\delta\chi^n\delta\chi^m}
         +\hbox{the graded-transposed second derivative}=0.
    \]
    The cancellation is graded symmetry of the Hessian of the fermionic
    functional.  Thus all fundamental brackets are preserved and the shift is
    anticanonical.  Setting the shifted antifields to zero immediately gives
    \[
      \chi_n^\ddagger=\frac{\delta\Psi}{\delta\chi^n},
    \]
    which is the BV gauge-fixing Lagrangian submanifold.
    \item[(c)]
    The compatibility of \(\Delta\) with the antibracket, the Jacobi identity,
    and \(\Delta^2=0\) give
    \[
      \sigma^2F
        =\frac12\left(F,(S,S)-2\ii\Delta S\right)
    \]
    in the right/left derivative convention of Section~15.9.  Therefore the
    quantum master equation
    \[
      (S,S)-2\ii\Delta S=0
    \]
    implies \(\sigma^2=0\).  Equivalently, the weighted measure
    \(e^{\ii S}[d\chi]\) has vanishing quantum BRST divergence.

    A quantum observable is a cohomology class of this differential:
    \[
      \sigma O=0,\qquad O\sim O+\sigma K.
    \]
    In explicit form its closure condition is
    \[
      (O,S)=\ii\Delta O.
    \]
    This is also the condition that makes its expectation value independent
    of the gauge-fixing Lagrangian submanifold.
  \end{enumerate}

  \SupplementarySolution{15}{Yang--Mills as a Consistent Deformation}
  Let \(f^a_{\mu\nu}=\partial_\mu A^a_\nu-\partial_\nu A^a_\mu\).
  A representative first-order deformation is
  \[
    \begin{split}
    S_1=\int \dd^4x\bigg[
      &-\frac12 C_{abc}f_a^{\mu\nu}A^b_\mu A^c_\nu
       +C^a{}_{bc}A_a^{\ddagger\mu}A^b_\mu\omega^c\\
      &-\frac12C^a{}_{bc}\omega_a^\ddagger
        \omega^b\omega^c\bigg].
    \end{split}
  \]
  The equation \((S_0,S_1)=0\) fixes the relative coefficients: its
  antifield-free part is the cubic vertex, while its antifield terms deform
  the Abelian gauge law and ghost transformation.  Poincaré invariance and
  integration by parts make \(C_{abc}\) totally antisymmetric with respect
  to the kinetic inner product.

  At order \(g^2\),
  \[
    (S_0,S_2)+\frac12(S_1,S_1)=0.
  \]
  The term in \((S_1,S_1)\) proportional to
  \(\omega^\ddagger\omega^3\) cannot be removed by an antifield-free
  \(S_2\); its coefficient is
  \(C^a{}_{e[b}C^e{}_{cd]}\).  Thus Jacobi is required.  When it holds, the
  remaining term is cancelled by
  \[
    S_2=-\frac14\int \dd^4x\,
      C^a{}_{bc}C_{ade}
      A_b^\mu A_c^\nu A^d_\mu A^e_\nu .
  \]
  Combining the quadratic, cubic, and quartic pieces gives
  \(-\tfrac14F^a_{\mu\nu}F_a^{\mu\nu}\) with
  \(F^a_{\mu\nu}=f^a_{\mu\nu}
  +gC^a{}_{bc}A^b_\mu A^c_\nu\): the unique nontrivial deformation is
  Yang--Mills theory, up to field redefinitions and strictly
  gauge-invariant additions.

\end{enumerate}
