  \SupplementaryExercise{1}{Fujikawa's Jacobian, Heat Kernel, and Index}
  {Inspired by Adel Bilal, Lectures on Anomalies (2008),
  Section 3.}
  {bilal-lectures-anomalies-2008-v2ch22-p01}
  \begin{enumerate}
    \item[(a)]
    Expand independent Grassmann fields in a finite orthonormal basis,
    \(\psi=\sum_na_n\phi_n\) and
    \(\bar\psi=\sum_n\bar b_n\phi_n^\dagger\).  Under
    \(\psi'=U\psi\), \(\bar\psi'=\bar\psi U\), derive the transformation
    of \(\prod_n da_n\,d\bar b_n\).  Specialize to
    \(U=1+i\alpha\gamma_5T\) and explain why the formal trace requires a
    regulator in the infinite-dimensional limit.
    \item[(b)]
    In Euclidean space regulate the chiral Jacobian with
    \[
      \mathcal A_M(x)
        =-2\,\operatorname{tr}\!
         \left[\gamma_5T e^{-\slashed D^{\,2}/M^2}\right]_{x,x}.
    \]
    Use
    \(\slashed D^{\,2}=D^2+\tfrac14[\gamma^\mu,\gamma^\nu]F_{\mu\nu}\)
    and gamma-matrix counting to show that the first finite term as
    \(M\to\infty\) is proportional to
    \(\epsilon^{\mu\nu\rho\sigma}
      \operatorname{tr}(T F_{\mu\nu}F_{\rho\sigma})\).
    Explain why the terms of orders \(M^4\) and \(M^2\) vanish.
    \item[(c)]
    Replace the exponential in part~(b) by a smooth function
    \(f(-\slashed D^{\,2}/M^2)\) satisfying
    \(f(0)=1\), \(f(\infty)=0\), and
    \(sf'(s)\to0\) at both endpoints.  Show that the coefficient of the
    four-dimensional anomaly depends only on these endpoint values, not
    on the shape of \(f\).
    \item[(d)]
    Let \(i\slashed D\phi_n=\lambda_n\phi_n\) in Euclidean space and
    \(\{\gamma_5,\slashed D\}=0\).  Prove that all nonzero eigenvalues
    occur in opposite-sign pairs whose contributions to
    \(\operatorname{Tr}\gamma_5f(\slashed D^{\,2}/M^2)\) cancel.  Show
    that the integrated regulated trace equals \(n_+-n_-\), the index of
    the Dirac operator.
  \end{enumerate}

  \SupplementaryExercise{2}{Anomalous Ward Identities and Removable Terms}
  {Inspired by Adel Bilal, Lectures on Anomalies (2008),
  Sections 4.2--4.3.}
  {bilal-lectures-anomalies-2008-v2ch22-p02}
  \begin{enumerate}
    \item[(a)]
    Suppose
    \(\partial_\mu J_5^\mu=C F_{\mu\nu}\widetilde F^{\mu\nu}\).
    Differentiate the background-field identity twice with respect to
    \(A_\nu(y)\) and \(A_\rho(z)\).  Derive the anomalous divergence of
    \(\langle T J_5^\mu(x)J^\nu(y)J^\rho(z)\rangle\), including its
    delta-function support, and give its momentum-space tensor
    structure.
    \item[(b)]
    Let the variation of an effective action be
    \(\delta_\alpha\Gamma=\mathcal G[\alpha,A]\).
    Show that if
    \(\mathcal G=\delta_\alpha B[A]\) for a local functional \(B\), the
    anomaly can be removed by a counterterm.  Explain why a local,
    finite anomaly that is not of this form is a nontrivial obstruction,
    and express the distinction as BRST exactness versus cohomology.
  \end{enumerate}

  \SupplementaryExercise{3}{Spectral Flow from Two to Four Dimensions}
  {Inspired by David Tong, Lectures on Gauge Theory (2018),
  Section 3.1.1.}
  {tong-gauge-theory-anomalies-2018-v2ch22-p03}
  \begin{enumerate}
    \item[(a)]
    Put a massless Dirac fermion of charge \(q\) on a spatial circle of
    length \(L\) in a uniform electric field \(E\).  Use
    \(\dot p=qE\) and the density of one-particle states to calculate the
    rates at which right- and left-moving levels cross zero.  Derive the
    local axial-current anomaly in \(1+1\) dimensions.
    \item[(b)]
    Place a massless Dirac fermion of charge \(q\) in parallel constant
    electric and magnetic fields.  The lowest Landau level has transverse
    degeneracy \(|qB|/(2\pi)\) per unit area.  Reduce its dynamics to the
    two-dimensional spectral-flow problem and derive
    \(\partial_\mu J_5^\mu\) in terms of
    \(\mathbf E\cdot\mathbf B\).
  \end{enumerate}

  \SupplementaryExercise{4}{Chiral Spectra, Representation Indices, and Anomaly Matching}
  {Inspired by David Tong, Lectures on Gauge Theory (2018),
  Section 3.4.1.}
  {tong-gauge-theory-anomalies-2018-v2ch22-p04}
  \begin{enumerate}
    \item[(a)]
    A four-dimensional \(U(1)\) theory contains only left-handed Weyl
    fermions with charges
    \[
      q_i=\{1,5,-7,-8,9\}.
    \]
    State the cubic gauge-anomaly and mixed gauge--gravitational anomaly
    conditions.  Verify that this spectrum satisfies both, and show that
    no gauge-invariant fermion mass bilinear can pair all five fields.
    \item[(b)]
    Define \(A(R)\) by
    \(\operatorname{tr}_R[T^a\{T^b,T^c\}]=A(R)d^{abc}\).
    Prove
    \[
      A(R\otimes S)=d(S)A(R)+d(R)A(S)
    \]
    and \(A(\overline R)=-A(R)\).  Use
    \(\mathbf3\otimes\mathbf3
      =\mathbf6\oplus\overline{\mathbf3}\)
    to find \(A(\mathbf6)\) when \(A(\mathbf3)=1\).
    \item[(c)]
    Couple a global current to a nondynamical background gauge field.
    Explain why integrating out modes above a scale \(\mu\) cannot change
    a nontrivial anomaly by a continuous amount.  Deduce the alternatives
    available to an infrared theory with a nonzero ultraviolet anomaly:
    massless degrees of freedom, spontaneous symmetry breaking with an
    anomalous Goldstone action, or a nontrivial topological sector.
  \end{enumerate}

  \SupplementaryExercise{5}{Standard-Model Anomaly Cancellation}
  {Adapted from John McGreevy, Physics 215C, Assignment 3, Problem 2 (2025).}
  {mcgreevy-215c-s25-a3-p2-v2ch22-p05}
  Treat all Standard-Model fermions as left-handed:
  \[
    Q:(\mathbf3,\mathbf2)_{1/6},\quad
    u^c:(\overline{\mathbf3},\mathbf{1})_{-2/3},\quad
    d^c:(\overline{\mathbf3},\mathbf{1})_{1/3},
  \]
  \[
    L:(\mathbf{1},\mathbf2)_{-1/2},\qquad
    e^c:(\mathbf{1},\mathbf{1})_1.
  \]
  For three current insertions associated with groups
  \(G_1,G_2,G_3\), use the left-handed-fermion convention
  \[
    \partial_\mu j_1^{A\mu}
      =\frac{\epsilon^{\mu\nu\rho\sigma}}{32\pi^2}
       F^{2B}_{\mu\nu}F^{3C}_{\rho\sigma}
       \sum_f\operatorname{tr}_{R_f}
       \left(\{T_1^A,T_2^B\}T_3^C\right),
  \]
  with conjugate representations used for the charge-conjugated
  right-handed fields.
  \begin{enumerate}
    \item[(a)] Verify cancellation of \(U(1)_Y^3\).
    \item[(b)] Show generally that triangles of type
      \(SU(N)U(1)^2\) and \(SU(N)G_1G_2\), with only one current from the
      indicated simple group, vanish.
    \item[(c)] Explain why the \(SU(3)^3\) anomaly cancels in QCD.
    \item[(d)] Prove that a perturbative \(SU(2)^3\) anomaly is impossible.
    \item[(e)] Derive the condition
      \(2Y_Q-Y_u-Y_d=0\) from \(SU(3)^2U(1)_Y\) and check it.
    \item[(f)] Derive \(Y_L+3Y_Q=0\) from
      \(SU(2)^2U(1)_Y\) and check it.
    \item[(g)] Verify cancellation of the mixed
      gravitational--\(U(1)_Y\) anomaly.
    \item[(h)] Assuming a neutral right-handed neutrino, show that these
      constraints determine the chiral hypercharges up to overall
      normalization.
    \item[(i)] Compute the corresponding anomaly coefficients for baryon
      and lepton number.  Show that each is anomalous but \(B-L\) is not.
  \end{enumerate}

  \SupplementaryExercise{6}{Characteristic Forms and BRST Descent}
  {Inspired by Jeffrey A. Harvey, TASI 2003 Lectures on Anomalies
  (2005), Sections 3.1--3.2.}
  {harvey-tasi-anomalies-2005-v2ch22-p06}
  \begin{enumerate}
    \item[(a)]
    With matrix-valued connection \(A\), curvature
    \(F=dA+A^2\), and covariant derivative \(D=d+[A,\cdot]\), prove that
    \(d\,\operatorname{Tr}F^n=0\) and that
    \(\operatorname{Tr}F^n\) is gauge invariant.  Identify the two facts
    used in the proof and explain why such forms can classify anomaly
    polynomials.
    \item[(b)]
    Suppose
    \(\delta_\alpha\Gamma=\mathcal G(\alpha,A)\) and gauge variations obey
    \([\delta_\alpha,\delta_\beta]=\delta_{[\alpha,\beta]}\).
    Derive the Wess--Zumino consistency condition on \(\mathcal G\).
    Specialize to an Abelian group.
    \item[(c)]
    Introduce a ghost \(\omega\) and a nilpotent BRST differential \(s\).
    Show that the Wess--Zumino condition is \(s\mathcal G=0\).  Prove that
    \(\mathcal G=sB\), with \(B\) local and of ghost number zero, describes
    a removable anomaly.  State the resulting cohomology group that
    classifies candidate anomalies.
    \item[(d)]
    In the convention \(F=dA+A^2\), verify
    \[
      \operatorname{Tr}F^2=dQ_3,\qquad
      Q_3=\operatorname{Tr}\left(A\,dA+\frac23A^3\right).
    \]
    Under \(sA=-D\omega\), show that
    \(sQ_3+dQ_2^{\,1}=0\) up to an exact convention-dependent
    redefinition, and find a convenient representative \(Q_2^{\,1}\).
  \end{enumerate}

  \SupplementaryExercise{7}{Bulk and Axion-String Anomaly Inflow}
  {Inspired by Jeffrey A. Harvey, TASI 2003 Lectures on Anomalies
  (2005), Sections 4.1--4.2.}
  {harvey-tasi-anomalies-2005-v2ch22-p07}
  \begin{enumerate}
    \item[(a)]
    Let a four-manifold \(M_4\) be the boundary of \(X_5\), and consider
    \[
      S_{\rm CS}=k\int_{X_5}A\wedge F\wedge F.
    \]
    Compute its variation under \(A\to A+d\alpha\).  Determine the
    boundary anomaly that it can cancel and explain why the combined
    bulk--boundary system may be gauge invariant even though neither
    part is separately invariant.
    \item[(b)]
    An axion angle \(\theta\) winds by \(2\pi\) around a string and couples
    through
    \(S_\theta=c\int\theta F\wedge F\).
    Integrate by parts away from the string to exhibit the
    Chern--Simons-like inflow current.  Show that its divergence is
    localized on the string and explain how it cancels the gauge anomaly
    of chiral string zero modes.
  \end{enumerate}

  \SupplementaryExercise{8}{The Axial Anomaly in Dimensional Regularization}
  {Adapted from Iain Stewart, MIT 8.325 Homework 6, Problem 2 (2007).}
  {mit-8325-s07-hw6-anomalies-v2ch22-p08}
  Compute the massless QED AVV triangle in
  \(d=4-\epsilon\) dimensions using a prescription for \(\gamma_5\) that
  preserves both vector Ward identities.  Show every step that turns the
  axial contraction into an evanescent numerator times a divergent
  integral, evaluate its finite limit, and prove that the result is the
  two-photon matrix element of
  \[
    \partial_\mu J^\mu_5
      =-\frac{e^2}{16\pi^2}F^{\alpha\beta}\widetilde F_{\alpha\beta}
  \]
  in the stated epsilon-tensor convention.

  \SupplementaryExercise{9}{Electroweak \(B\), \(L\), and \(B+L\) Selection Rules}
  {Adapted from Iain Stewart, MIT 8.325 Homework 6 (2007),
  Problem 3.}
  {mit-8325-s07-hw6-anomalies-v2ch22-p09}
  \begin{enumerate}
    \item[(a)]
    For \(N_g\) Standard-Model generations, compute the
    \(SU(2)_L^2U(1)_B\) anomaly coefficient using left-handed doublets.
    Normalize baryon number to \(1/3\) per quark and
    \(T(\mathbf2)=1/2\).  Explain why right-handed singlets do not enter.
    \item[(b)]
    Compute the corresponding \(SU(2)_L^2U(1)_L\) coefficient and show
    that the weak anomaly violates \(B+L\) but preserves \(B-L\).
    State the changes \(\Delta B\) and \(\Delta L\) associated with a
    unit electroweak topological transition.
    \item[(c)]
    Write the anomalous divergence of \(J^\mu_{B+L}\) in terms of
    \(W^a_{\mu\nu}\widetilde W^{a\mu\nu}\), using a convention in which
    \[
      \nu=\frac{g^2}{32\pi^2}
        \int d^4x\,W^a_{\mu\nu}\widetilde W^{a\mu\nu}
    \]
    is an integer.  Integrate the divergence to recover the selection
    rule for arbitrary integer \(\nu\).
  \end{enumerate}

  \SupplementaryExercise{10}{Anomaly Poles, Symmetric Masses, and Real Representations}
  {Inspired by John Preskill, Physics 230abc Field Theory Notes
  (1983--1984), Chapter 3, pp. 75--117.}
  {preskill-qcd-anomalies-1984-v2ch22-p10}
  \begin{enumerate}
    \item[(a)]
    A three-current correlator has an anomalous divergence that approaches
    a nonzero polynomial in external momenta.  Show that a correlator
    analytic at zero momentum cannot reproduce this Ward identity.
    Identify the necessary longitudinal pole and explain its possible
    realization by a massless fermion or Goldstone boson.
    \item[(b)]
    Suppose a set of Weyl fermions can acquire a nonzero mass without
    breaking a global symmetry \(G\).  Show that the massive states form a
    real or pseudoreal representation of \(G\), or conjugate pairs, and
    hence have zero cubic \(G^3\) anomaly.  Relate this to decoupling at
    momenta much smaller than the mass.
    \item[(c)]
    If \(R\) is real or pseudoreal, there is a unitary matrix \(U\) such
    that \(T_R^{a*}=-UT_R^aU^{-1}\).  Use this relation and cyclicity of
    the trace to prove
    \(\operatorname{tr}_R[T^a\{T^b,T^c\}]=0\).
    Explain why this eliminates perturbative \(SU(2)^3\) anomalies but
    does not by itself address a possible global \(SU(2)\) anomaly.
  \end{enumerate}

  \SupplementaryExercise{11}{Spectator Matching and a Wess--Zumino Compensator}
  {Inspired by John Preskill, Physics 230abc Field Theory Notes
  (1983--1984), Chapter 3, pp. 75--117.}
  {preskill-qcd-anomalies-1984-v2ch22-p11}
  \begin{enumerate}
    \item[(a)]
    Weakly gauge an exact global symmetry \(G\) of a confining theory and
    add spectator Weyl fermions that cancel the ultraviolet gauge anomaly.
    Assuming \(G\) is unbroken, prove that the infrared composite spectrum
    must reproduce the anomaly of the confined elementary fermions.
    \item[(b)]
    In three-color QCD with two massless flavors, normalize quark baryon
    number to \(1/3\).  Compute the ultraviolet
    \(SU(2)_L^2U(1)_B\) anomaly.  Show that a single massless
    left-handed nucleon doublet with baryon number \(1\) would match this
    perturbative anomaly, and compare the parity of the number of
    \(SU(2)_L\) doublets in the ultraviolet and infrared.
    \item[(c)]
    An Abelian effective action varies as
    \(\delta_\alpha\Gamma=C\int\alpha F\wedge F\).
    Introduce a pseudoscalar \(a\) with
    \(\delta_\alpha a=-f\alpha\).  Construct a local term that cancels
    the anomaly, and combine \(a\) with \(A\) in a gauge-invariant kinetic
    term.  What mass does the gauge field acquire if that kinetic term is
    canonically normalized?
  \end{enumerate}

  \SupplementaryExercise{12}{Heat-Kernel Derivation of the Two-Dimensional Chiral Anomaly}
  {Adapted from John McGreevy, Physics 215C, Assignment 2, Problem 2 (2025).}
  {mcgreevy-215c-s25-a2-p2-v2ch22-p12}
  For a massive Dirac fermion of charge \(q\) in \(1+1\) dimensions, take
  \[
    I=\int d^2x\,
      \overline\psi\,[i\gamma^\mu(\partial_\mu+iqA_\mu)-m]\psi,
    \qquad
    j_5^\mu=i\overline\psi\gamma^\mu\gamma_5\psi .
  \]
  Derive the classical mass contribution to
  \(\partial_\mu j_5^\mu\).  Regulate the coincident fermion trace with
  \(e^{-\slashed D^{\,2}/M^2}\), expand the heat kernel through the first
  term that survives a \(\gamma_5\) trace, and show that
  \[
    \partial_\mu j_5^\mu
      =2im\overline\psi\gamma_5\psi
       +\frac{q}{2\pi}\epsilon^{\mu\nu}F_{\mu\nu}
  \]
  for the stated current and orientation conventions.  Explain why a
  regulator preserving the vector gauge symmetry assigns the quantum
  violation to the axial current.

\end{enumerate}
