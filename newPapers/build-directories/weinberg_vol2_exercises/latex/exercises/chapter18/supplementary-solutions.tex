\chapterbackmatter{Solutions to Supplementary Exercises}

\begin{enumerate}
  \SupplementarySolution{1}{Sine--Gordon Vertex Operator under Shell RG}
  \begin{enumerate}
    \item[(a)]
    In mass units, \([x]=-1\) and \([\boldsymbol\nabla]=1\).  The kinetic
    term in two dimensions gives
    \[
      2[\phi]+2-2=0,
      \qquad\hbox{so}\qquad [\phi]=0.
    \]
    The argument of the cosine must be dimensionless, while its coefficient
    supplies the two units carried by a free-energy density.  Hence
    \[
      \boxed{[\beta_0]=0,\qquad [\lambda_0]=2.}
      \tag{18.1.S1}
    \]

    \item[(b)]
    Orthogonality of the retained and shell Fourier supports separates the
    Gaussian free energy:
    \(F_0[\phi^-+\phi^+]=F_0[\phi^-]+F_0[\phi^+]\).  Define
    \[
      Z_+=\int[\mathcal D\phi^+]\,e^{-F_0[\phi^+]},
      \qquad
      \langle\mathcal O\rangle_+
      =\frac{1}{Z_+}\int[\mathcal D\phi^+]\,\mathcal O[\phi^+]\,
        e^{-F_0[\phi^+]},
      \tag{18.1.S2}
    \]
    where every mode of \(\phi^+\) lies in \(\mathcal S_\zeta\).  The
    shell-integrated Boltzmann weight is therefore
    \[
      e^{-F'[\phi^-]}
      =e^{-F_0[\phi^-]}
        \left\langle
          \exp\!\left\{
            \lambda_0\int d^2x\,
            \cos\!\bigl[\beta_0(\phi^-+\phi^+)\bigr]
          \right\}
        \right\rangle_+ .
    \]
    Expanding the exponential and then the logarithm through first order
    gives
    \[
      F'[\phi^-]
      =F_0[\phi^-]
       -\lambda_0\int d^2x\,
        \bigl\langle
          \cos\!\bigl[\beta_0(\phi^-+\phi^+)\bigr]
        \bigr\rangle_+
       +O(\lambda_0^2),
    \]
    which is (18.1.3).

    \item[(c)]
    At coincident points, (18.1.7) becomes
    \[
      C_\zeta\equiv
      \langle[\phi^+(x)]^2\rangle_+
      =\frac{1}{2\pi}\int_{\Lambda/\zeta}^{\Lambda}\frac{dk}{k}
      =\frac{\ln\zeta}{2\pi}.
      \tag{18.1.S3}
    \]
    Since the Gaussian is centered, (18.1.5) yields
    \[
      \bigl\langle e^{\pm i\beta_0\phi^+(x)}\bigr\rangle_+
      =e^{-\beta_0^2C_\zeta/2}
      =\zeta^{-\beta_0^2/(4\pi)}.
    \]
    Combining this with (18.1.6),
    \[
      \bigl\langle
        \cos\!\bigl[\beta_0(\phi^-+\phi^+)\bigr]
      \bigr\rangle_+
      =\zeta^{-\beta_0^2/(4\pi)}\cos(\beta_0\phi^-),
    \]
    and substitution proves (18.1.4).

    Now put \(\mathbf x=\zeta\mathbf x'\) and
    \(\phi'(\mathbf x')=\phi^-(\zeta\mathbf x')\).  No field amplitude
    rescaling is required because \([\phi]=0\), so the kinetic term remains
    normalized and \(\beta'=\beta_0+O(\lambda_0^2)\).  The area element
    supplies the remaining factor:
    \[
      \lambda'=\lambda_0
        \zeta^{\,2-\beta_0^2/(4\pi)}+O(\lambda_0^2).
    \]
    With \(s=\ln\zeta\),
    \[
      \frac{d\lambda}{ds}
      =\left(2-\frac{\beta_0^2}{4\pi}\right)\lambda+O(\lambda^2).
      \tag{18.1.S4}
    \]
    Thus
    \[
      \boxed{\beta_0^2<8\pi:\ \hbox{relevant},
      \qquad
      \beta_0^2>8\pi:\ \hbox{irrelevant}.}
    \]
    At \(\beta_0^2=8\pi\) the perturbation is marginal at this order; the
    coupled higher-order flow resolves that boundary.
  \end{enumerate}

  \SupplementarySolution{2}{Wilsonian Flow with Cubic and Quartic Interactions}
  \begin{enumerate}
    \item[(a)]
    \begin{enumerate}
      \item[(i)] Choose \(\zeta>1\), split the field into retained and shell
      modes,
      \[
        \phi=\phi^-+\phi^+,
        \qquad
        |\mathbf k|<\frac{\Lambda}{\zeta}\quad\hbox{or}\quad
        \frac{\Lambda}{\zeta}<|\mathbf k|<\Lambda,
      \]
      and integrate over \(\phi^+\).  Next rescale momenta
      \(\mathbf k'=\zeta\mathbf k\), equivalently
      \(\mathbf x'=\mathbf x/\zeta\), to restore the cutoff to
      \(\Lambda\).  Finally rescale the retained field so that the kinetic
      term again has its chosen normalization.  The resulting local action
      has the same symmetry-allowed operator basis but new coefficients.
      Repetition composes these maps, and therefore traces a trajectory in
      coupling space.

      \item[(ii)] With \(s=\ln\zeta\) and dimensionless couplings
      \(u^I(s)\), define
      \[
        \beta^I(\mathbf u)=\frac{du^I}{ds}.
      \]
      At a fixed point \(\mathbf u_*\), all beta functions vanish.  For
      \(\delta u^I=u^I-u_*^I\),
      \[
        \frac{d\delta u^I}{ds}=M^I{}_{J}\delta u^J+\cdots,
        \qquad
        M^I{}_{J}=\left.
          \frac{\partial\beta^I}{\partial u^J}\right|_{\mathbf u_*}.
      \]
      If \(M V_r=y_rV_r\), the corresponding scaling field behaves as
      \(e^{y_rs}\).  The numbers \(y_r\) are the RG eigenvalues, or
      coupling scaling dimensions, in this coarse-graining convention.

      \item[(iii)] A coupling is relevant, irrelevant, or marginal when
      \(y_r\) is respectively positive, negative, or zero.  Nonlinear terms
      decide the fate of a marginal direction.  The critical surface is the
      set of microscopic couplings whose RG trajectories approach the same
      infrared fixed point.  Irrelevant differences between microscopic
      models decay along this surface, so long-distance observables depend
      only on the fixed point and its relevant scaling fields.  This is the
      RG explanation of universality.
    \end{enumerate}

    \item[(b)]
    Let \(\mathcal S\) denote the eliminated shell and write
    \[
      \begin{aligned}
        I_1&=\int_{\mathcal S}\frac{d^dq}{(2\pi)^d}\,G_0(\mathbf q),\\
        I_2(\mathbf p)&=\int\frac{d^dq}{(2\pi)^d}\,
          \chi_{\mathcal S}(\mathbf q)\chi_{\mathcal S}(\mathbf q+\mathbf p)\\
        &\hspace{27mm}\times
          G_0(\mathbf q)G_0(\mathbf q+\mathbf p),
      \end{aligned}
    \]
    where \(\chi_{\mathcal S}\) is the shell characteristic function.
    The Wilson free energy follows from the cumulant expansion
    \[
      F_{\mathrm W}[\phi^-]=F_0[\phi^-]
      +\langle F_{\mathrm{int}}\rangle_+
      -\frac12\left(
        \langle F_{\mathrm{int}}^2\rangle_+
        -\langle F_{\mathrm{int}}\rangle_+^2
      \right)+\cdots.
    \]
    \begin{enumerate}
      \item[(i)] At first order in \(g_0\), the term with two retained and
      two shell fields is
      \[
        g_0\int d^dx\,6(\phi^-)^2
          \langle(\phi^+)^2\rangle_+
        =6g_0I_1\int d^dx\,(\phi^-)^2.
      \]
      Since the mass term is \(\mu^2\phi^2/2\), this gives
      \(\delta\mu_g^2=12g_0I_1\).

      The leading quadratic term from the cubic interaction comes from two
      vertices, each supplying one retained field and two shell fields.
      There are three choices at each vertex and two connected Wick
      contractions between them.  Thus
      \[
        \begin{aligned}
        &-\frac12\lambda_0^2(3)(3)(2)
          \int d^dx\,d^dy\,
            \phi^-(x)\phi^-(y)G_+(x-y)^2\\
        &\hspace{22mm}
          =-9\lambda_0^2\int d^dx\,d^dy\,
            \phi^-(x)\phi^-(y)G_+(x-y)^2.
        \end{aligned}
      \]
      Comparing again with the quadratic normalization gives, at external
      momentum \(\mathbf p\),
      \[
        \delta\mu_\lambda^2(\mathbf p)
          =-18\lambda_0^2I_2(\mathbf p).
      \]
      Hence, before the coordinate and field rescalings,
      \[
        \boxed{\begin{aligned}
          \mu'^2={}&\mu_0^2
            +12g_0\int_{\mathcal S}\frac{d^dq}{(2\pi)^d}
              \frac{1}{\mathbf q^2+\mu_0^2}\\
          &-18\lambda_0^2
            \int_{\mathcal S}\frac{d^dq}{(2\pi)^d}
              \frac{1}{(\mathbf q^2+\mu_0^2)^2}
            +\cdots.
        \end{aligned}}
      \]
      In the last integral the zero-momentum limit was taken; more
      precisely both internal momenta must lie in \(\mathcal S\), as in
      \(I_2(\mathbf p)\).  The cubic interaction also produces the linear
      term \(3\lambda_0I_1\int d^dx\,\phi^-\).  Thus \(\phi=0\) is not a
      stationary background unless that tadpole is cancelled or the field
      is shifted; the boxed expression is the quadratic coefficient in the
      unshifted Wilson action requested here.

      \item[(ii)] Two cubic vertices can supply four external legs and be
      joined by one internal line.  Therefore the first correction to the
      quartic coupling is of order \(\lambda_0^2\):
      \begin{center}
      \begin{tikzpicture}[baseline=(current bounding box.center),scale=0.78]
        \coordinate (v1) at (-0.9,0);
        \coordinate (v2) at (0.9,0);
        \draw (-2.2,0.9)--(v1);
        \draw (-2.2,-0.9)--(v1);
        \draw (v1)--(v2);
        \draw (v2)--(2.2,0.9);
        \draw (v2)--(2.2,-0.9);
        \fill (v1) circle (2pt);
        \fill (v2) circle (2pt);
        \node[below=3pt] at (v1) {$\lambda_0$};
        \node[below=3pt] at (v2) {$\lambda_0$};
      \end{tikzpicture}
      \end{center}
      The three choices of channel are obtained by permuting the four
      external momenta.  In a sharp shell implementation the exchanged
      momentum must lie in the eliminated shell.

      \item[(iii)] When \(\lambda_0=0\), both the Gaussian part and the
      \(g_0\phi^4\) interaction are invariant under
      \(\phi\mapsto-\phi\).  Integrating out modes preserves this symmetry,
      so no \(\phi^3\) operator is generated at any order in \(g_0\).
      There is therefore no lowest nonzero order.
    \end{enumerate}
  \end{enumerate}

  \SupplementarySolution{3}{Fluctuating Membrane and the Running Bending Modulus}
  \begin{enumerate}
    \item[(a)]
    The sum \(1+(\boldsymbol\nabla h)^2\) in the unexpanded geometric free
    energy forces \(\boldsymbol\nabla h\) to be dimensionless.  Consequently
    \[
      [h]=-1.
    \]
    The tension term and bending term then give, respectively,
    \[
      [r_0]-2=0,
      \qquad
      [\lambda_0]+2-2=0.
    \]
    Therefore
    \[
      \boxed{[h]=-1,\qquad [r_0]=2,\qquad [\lambda_0]=0.}
      \tag{18.3.S1}
    \]

    \item[(b)]
    Use
    \[
      h(x)=\int_{\mathbf k}e^{i\mathbf k\cdot\mathbf x}h_{\mathbf k},
      \qquad
      \int_{\mathbf k}\equiv\int\frac{d^2k}{(2\pi)^2}.
    \]
    The quadratic part is
    \[
      F_0[h]=\frac12\int_{\mathbf k}
        \bigl(r_0\mathbf k^2+\lambda_0\mathbf k^4\bigr)
        h_{\mathbf k}h_{-\mathbf k}.
      \tag{18.3.S2}
    \]
    With \(Z_+=\int[\mathcal Dh^+]e^{-F_0[h^+]}\), define the shell average
    by
    \[
      \langle\mathcal O\rangle_+
      =Z_+^{-1}\int[\mathcal Dh^+]\,\mathcal O[h^+]e^{-F_0[h^+]}.
    \]
    Inverting the quadratic kernel in (18.3.S2) gives, for momenta in the
    shell,
    \[
      \boxed{
      \bigl\langle h_{\mathbf k_1}^+h_{\mathbf k_2}^+\bigr\rangle_+
      =\frac{(2\pi)^2\delta^{(2)}
        (\mathbf k_1+\mathbf k_2)}
        {r_0\mathbf k_1^2+\lambda_0\mathbf k_1^4}.}
      \tag{18.3.S3}
    \]
    The correlator is zero if either momentum lies outside
    \(\mathcal S_\zeta\).

    \item[(c)]
    The quartic interaction is
    \[
      F_I[h]=-\frac{5\lambda_0}{4}\int d^2x\,
        (\partial_a h\,\partial_a h)(\boldsymbol\nabla^2h)^2.
    \]
    The stated Fourier convention gives
    \[
      \begin{split}
      F_I[h]
      ={}&\frac{5\lambda_0}{4}
        \int_{\mathbf k_1}\!\cdots\!\int_{\mathbf k_4}
        (2\pi)^2\delta^{(2)}
          (\mathbf k_1+\mathbf k_2+\mathbf k_3+\mathbf k_4)\\
      &\quad{}\times
        (\mathbf k_1\!\cdot\!\mathbf k_2)
        \mathbf k_3^2\mathbf k_4^2\,
        h_{\mathbf k_1}h_{\mathbf k_2}
        h_{\mathbf k_3}h_{\mathbf k_4}.
      \end{split}
      \tag{18.3.S4}
    \]
    To renormalize the displayed bending operator, retain the two
    Laplacian fields as slow fields and contract the two gradient fields in
    the shell.  The first cumulant then contains
    \[
      \left.\langle F_I\rangle_+\right|_{(\nabla^2h^-)^2}
      =-\frac{5\lambda_0}{4}\,I_\zeta
        \int d^2x\,(\boldsymbol\nabla^2h^-)^2,
      \tag{18.3.S5}
    \]
    where
    \[
      \begin{split}
      I_\zeta
      &\equiv\bigl\langle
        (\boldsymbol\nabla h^+)^2\bigr\rangle_+\\
      &=\int_{\Lambda/\zeta<|\mathbf k|<\Lambda}
        \frac{d^2k}{(2\pi)^2}\,
        \frac{\mathbf k^2}
          {r_0\mathbf k^2+\lambda_0\mathbf k^4}\\
      &=\frac{1}{4\pi\lambda_0}
        \ln\!\left(
          \frac{r_0+\lambda_0\Lambda^2}
               {r_0+\lambda_0\Lambda^2/\zeta^2}
        \right).
      \end{split}
      \tag{18.3.S6}
    \]
    The other quadratic contraction,
    \((\boldsymbol\nabla h^-)^2
      \langle(\boldsymbol\nabla^2h^+)^2\rangle_+\),
    changes the tension; the mixed contraction is odd in shell momentum.
    Neither is part of the requested bending-only flow.

    Comparing (18.3.S5) with
    \(\lambda(\boldsymbol\nabla^2h^-)^2/2\) gives
    \[
      \lambda_{\rm W}
      =\lambda_0-\frac{5\lambda_0}{2}I_\zeta.
      \tag{18.3.S7}
    \]
    Under \(\mathbf x=\zeta\mathbf x'\), take
    \(h'(\mathbf x')=\zeta^{-1}h^-(\zeta\mathbf x')\).  The bending integral
    is invariant, consistently with \([\lambda_0]=0\), so it contributes no
    additional tree-level factor.  For an infinitesimal shell
    \(\zeta=e^{ds}\),
    \[
      I_\zeta
      =\frac{\Lambda^2}{2\pi
        (r_0+\lambda_0\Lambda^2)}\,ds+O(ds^2).
    \]
    Thus the result with the tension retained is
    \[
      \boxed{
      \frac{d\lambda}{ds}
      =-\frac{5}{4\pi}\,
        \frac{\lambda\Lambda^2}{r+\lambda\Lambda^2}.}
      \tag{18.3.S8}
    \]
    In the bending-dominated limit
    \(r/(\lambda\Lambda^2)\to0\), this reduces to the sheet's target,
    \[
      \boxed{\frac{d\lambda}{ds}\simeq-\frac{5}{4\pi}.}
    \]
    Coarse graining therefore lowers the bending modulus.  Since a smaller
    bending modulus costs less free energy for curvature, the
    long-wavelength membrane becomes more flexible.
  \end{enumerate}

  \SupplementarySolution{4}{The \(O(N)\) Nonlinear Sigma Model Near Two Dimensions}
  \begin{enumerate}
    \item[(a)] Since \(\mathbf n\) has fixed unit length, it is
    dimensionless.  Requiring the free energy to be dimensionless gives
    \[
      [g]=[\text{momentum}]^{\,2-d},
      \qquad [g]_{\mathrm{mass}}=2-d.
    \]
    Thus \(g\) is classically marginal in two dimensions.

    \item[(b)] The constraint gives
    \(\sigma=\sqrt{1-\boldsymbol\pi^2}\) in a coordinate patch on the
    target sphere.  Consequently
    \[
      \partial_i\sigma
        =-\frac{\boldsymbol\pi\cdot\partial_i\boldsymbol\pi}
          {\sqrt{1-\boldsymbol\pi^2}},
    \]
    and
    \[
      (\partial_i\mathbf n)^2
      =(\partial_i\boldsymbol\pi)^2
       +\frac{(\boldsymbol\pi\cdot
          \partial_i\boldsymbol\pi)^2}{1-\boldsymbol\pi^2}.
    \]
    Substitution proves the stated exact \(\boldsymbol\pi\)-space free
    energy.

    \item[(c)] To the order needed,
    \[
      F=\frac{1}{2g_0}\int d^dx\left[
        (\partial_i\boldsymbol\pi)^2
        +(\pi_a\partial_i\pi_a)(\pi_b\partial_i\pi_b)
        +O(\pi^6)
      \right].
    \]
    Split \(\boldsymbol\pi=\boldsymbol\pi^-+\boldsymbol\pi^+\).
    Contracting the two undifferentiated shell fields in the quartic term
    gives the retained kinetic contribution
    \[
      \frac{1}{2g_0}\int d^dx\,
        \bigl\langle\pi_a^+\pi_b^+\bigr\rangle_+
        (\partial_i\pi_a^-)(\partial_i\pi_b^-)
      =\frac{I_d}{2}\int d^dx\,
        (\partial_i\boldsymbol\pi^-)^2.
    \]
    Here
    \(\langle\pi_a^+(x)\pi_b^+(x)\rangle_+
      =g_0\delta_{ab}I_d\).
    The quadratic part of the Wilson free energy is therefore
    \[
      F_{\mathrm W}^{(2)}
      =\frac12\left(\frac{1}{g_0}+I_d\right)
        \int d^dx\,(\partial_i\boldsymbol\pi^-)^2.
    \]
    The supplied wavefunction rescaling means
    \(\boldsymbol\pi^-=A\boldsymbol\pi'\), with
    \[
      A^2=1-(N-1)g_0I_d+O(g_0^2).
    \]
    Hence
    \[
      \left(\frac{1}{g_0}+I_d\right)A^2
      =\frac{1}{g_0}+(2-N)I_d+O(g_0I_d^2).
    \]
    Finally set \(\mathbf x=\zeta\mathbf x'\).  The kinetic integral
    acquires \(\zeta^{d-2}\), giving
    \[
      \boxed{
      \frac{1}{g(\zeta)}=\zeta^{d-2}\left[
        \frac{1}{g_0}+(2-N)I_d
      \right].}
    \]
    The field-rescaling contribution is what changes the direct loop
    coefficient into \(2-N\).

    \item[(d)] With \(d=2+\epsilon\),
    \[
      I_d=\frac{\Omega_{d-1}}{(2\pi)^d}
        \int_{\Lambda/\zeta}^{\Lambda}dq\,q^{d-3}
      =\frac{\Omega_{d-1}}{(2\pi)^d}
        \Lambda^\epsilon\frac{1-e^{-\epsilon s}}{\epsilon}.
    \]
    In an infinitesimal step,
    \(dI_d/ds|_{s=0}
      =\Omega_{d-1}\Lambda^\epsilon/(2\pi)^d\).  Differentiating the
    result of part~(c) at \(s=0\) gives
    \[
      -\frac{1}{g^2}\frac{dg}{ds}
      =\frac{\epsilon}{g}
       -(N-2)\frac{\Omega_{d-1}}{(2\pi)^d}\Lambda^\epsilon,
    \]
    and therefore
    \[
      \boxed{
      \frac{dg}{ds}=-\epsilon g
       +(N-2)\frac{\Omega_{d-1}}{(2\pi)^d}
        \Lambda^\epsilon g^2+O(g^3).}
    \]
    Since \(\Omega_1=2\pi\), the coefficient near \(d=2\) is
    \(1/(2\pi)\), as quoted.

    \item[(e)] To this order the zeros are
    \[
      \boxed{g_*^{(0)}=0,\qquad
      g_*^{(c)}=
        \frac{\epsilon(2\pi)^d}
        {(N-2)\Omega_{d-1}\Lambda^\epsilon}
        \simeq\frac{2\pi\epsilon}
        {(N-2)\Lambda^\epsilon}.}
    \]
    The second zero is at positive temperature for \(N>2\).  For
    \(N=2\) it is absent at this order, while for \(N<2\) its formal value
    is negative.

    \item[(f)] At the nonzero fixed point,
    \[
      \beta'(g_*^{(c)})=-\epsilon
        +2(N-2)\frac{\Omega_{d-1}}{(2\pi)^d}
          \Lambda^\epsilon g_*^{(c)}
      =\epsilon.
    \]
    Thus a reduced-temperature displacement grows as
    \(\delta g(s)=\delta g(0)e^{\epsilon s}\).  The RG evolution stops
    at \(e^{s_\xi}\sim\xi/a\) when this displacement becomes order one,
    so
    \[
      \boxed{\xi\propto|g-g_*^{(c)}|^{-1/\epsilon},
      \qquad \nu=\frac{1}{\epsilon}.}
    \]
    For \(N>2\) and \(\epsilon>0\), the Gaussian point is the stable
    zero-temperature phase rather than the finite-temperature critical
    point.  At \(\epsilon=0\), the nonzero fixed point merges with the
    origin and the one-loop flow gives essential rather than power-law
    scaling,
    \(\xi\sim a\exp[2\pi/((N-2)g)]\); a finite \(\nu\) is then not
    defined by this approximation.
  \end{enumerate}

  \SupplementarySolution{5}{Dimensional Transmutation in the Gross--Neveu Model}
  \begin{enumerate}
    \item[(a)] The kinetic term fixes
    \([\psi]=[\bar\psi]=(D-1)/2\).  Therefore
    \([\bar\psi\psi]=D-1\), and dimensionality of the Lagrangian gives
    \[
      \boxed{[g]_{\mathrm{mass}}=D-2(D-1)=2-D.}
    \]
    The coupling is dimensionless at \(D=2\), so the massless classical
    action has no intrinsic scale there.

    \item[(b)] Applying the supplied identity gives
    \[
      S[\psi,\sigma]=\int d^Dx\left[
        \bar\psi^a(i\slashed\partial-\sigma)\psi^a
        +\frac{N}{4g}\sigma^2
      \right].
    \]
    Eliminating \(\sigma\) classically gives
    \(\sigma=2g\bar\psi^a\psi^a/N\) and exactly recovers
    \(-g(\bar\psi^a\psi^a)^2/N\), so the normalization and sign agree with
    the original action.

    Each flavor produces the same fermion determinant.  Hence
    \[
      \begin{aligned}
        Z&=\int_{\mathcal C}\mathcal D\sigma\,
          e^{iS_{\mathrm{eff}}[\sigma]},\\
        S_{\mathrm{eff}}[\sigma]
        &=N\left[
          \int d^Dx\,\frac{\sigma^2}{4g}
          -i\operatorname{Tr}\ln
            (i\slashed\partial-\sigma+i0)
        \right].
      \end{aligned}
    \]
    The trace includes spacetime and spinor indices.  Its variation gives
    the saddle equation
    \[
      \frac{\sigma(x)}{2g}
      +i\operatorname{tr}_{\mathrm{spin}}
        \langle x|(i\slashed\partial-\sigma+i0)^{-1}|x\rangle=0.
    \]
    The whole effective action is proportional to \(N\); stationary-phase
    configurations therefore dominate as \(N\to\infty\), while normalized
    \(\sigma\) fluctuations are suppressed by powers of \(N^{-1/2}\).

    For a translation-invariant saddle \(\sigma(x)=m\), Wick rotation gives
    the effective potential per flavor
    \[
      \frac{V_{\mathrm{eff}}(m)}{N}
      =\frac{m^2}{4g_B}
       -\frac{d_\gamma}{2}
        \int_{|p_E|<\Lambda}\frac{d^Dp_E}{(2\pi)^D}
          \ln(p_E^2+m^2)+\text{constant},
    \]
    where \(d_\gamma=\operatorname{tr}\mathbf1\).  The factor
    \(d_\gamma/2\) follows from
    \(\det(\slashed p_E+m)=(p_E^2+m^2)^{d_\gamma/2}\).  Thus
    \[
      m\left[
        \frac{1}{2g_B}
        -d_\gamma\int_{|p_E|<\Lambda}
          \frac{d^Dp_E}{(2\pi)^D}\frac{1}{p_E^2+m^2}
      \right]=0.
    \]

    In \(D=2\), take two-component spinors, so \(d_\gamma=2\).  For the
    nonzero saddle,
    \[
      \begin{aligned}
        \frac{1}{2g_B}&=2I_\Lambda(m),\\
        I_\Lambda(m)&=\int_{|p_E|<\Lambda}
          \frac{d^2p_E}{(2\pi)^2}\frac{1}{p_E^2+m^2}\\
        &=\frac{1}{4\pi}\ln\left(
          \frac{\Lambda^2+m^2}{m^2}\right).
      \end{aligned}
    \]
    Equivalently,
    \[
      m^2=\frac{\Lambda^2}{e^{\pi/g_B}-1}
      \underset{\Lambda\gg|m|}{\simeq}
      \Lambda^2e^{-\pi/g_B}.
    \]
    Introduce the renormalized coupling at a reference scale \(\mu\) by
    \[
      \frac{1}{g_R(\mu)}
        =\frac{1}{g_B}-\frac{2}{\pi}\ln\left(\frac{\Lambda}{\mu}\right).
    \]
    Removing the cutoff at fixed \(g_R(\mu)\) turns the gap equation into
    \[
      \boxed{|m|=\mu
        \exp\left[-\frac{\pi}{2g_R(\mu)}\right].}
    \]
    The right-hand side is RG invariant: differentiation at fixed bare
    coupling gives
    \[
      \mu\frac{dg_R}{d\mu}=-\frac{2}{\pi}g_R^2,
    \]
    and the running of \(g_R\) cancels that of \(\mu\) in the boxed
    expression.  A dimensionless classical coupling has thus been exchanged
    for a physical mass scale, which is dimensional transmutation.

    At the saddle the fermion action is
    \[
      S_{\mathrm{saddle}}=\int d^2x\left[
        \bar\psi^a(i\slashed\partial-m)\psi^a
        +\frac{Nm^2}{4g_B}
      \right].

  \SupplementarySolution{6}{Asymptotic Freedom, Unification, and \(SU(5)\)}
  \begin{enumerate}
    \item[(i)] Differentiating the supplied expression gives

    arbitrarily high renormalization scale, \(g(\mu)\to0\) as
    \(\mu\to\infty\).  At one loop this requires
    \[
      \boxed{b_0>0
      \quad\Longleftrightarrow\quad
      11N_c-N_f-\frac12N_s>0.}
    \]

    \item[(ii)] Define \(\Lambda\) as the scale where the one-loop inverse
    coupling extrapolates to zero.  The supplied formula gives
    \[
      \ln\left(\frac{\Lambda_{\mathrm{UV}}^2}{\Lambda^2}\right)
        =\frac{3(4\pi)^2}{b_0g_0^2},
      \qquad
      \boxed{\Lambda=\Lambda_{\mathrm{UV}}
        \exp\left[-\frac{3(4\pi)^2}{2b_0g_0^2}\right].}
    \]

    \[
      \frac{1}{g^2(\mu)}
        =\frac{b_0}{3(4\pi)^2}
          \ln\left(\frac{\mu^2}{\Lambda^2}\right),
      \qquad

        \frac{3(4\pi)^2}{2b_0g^2(\mu)}\right].
    \]
    The divergence at \(\mu=\Lambda\) only signals the failure of the
    one-loop weak-coupling approximation; in QCD it identifies the
    nonperturbative scale.

    fundamentals in the weak doublet \(Q_L\), plus the two singlets
    \(u_R\) and \(d_R\).  A right-handed fundamental may equivalently be
    counted as a left-handed antifundamental, which has the same quadratic
    index.  Thus
    \[
      N_f^{(3)}=4N_g,\qquad N_s^{(3)}=0,\qquad
      b_s=33-4N_g.
    \]
    Strict positivity gives \(N_g<33/4\), so
    \[
      \boxed{N_{g,\max}^{SU(3)}=8.}
    \]

    each color, and one lepton doublet.  The Standard Model Higgs is one
    complex scalar doublet:
    \[
      N_f^{(2)}=4N_g,\qquad N_s^{(2)}=1,\qquad
      b_w=22-4N_g-\frac12=\frac{43}{2}-4N_g.
    \]

    \[
      \boxed{N_{g,\max}^{SU(2)}=5.}

    \[
      \frac{1}{g_i^2(\mu)}=\frac{1}{g_i^2(M)}-b_iL.

        =\frac{3}{5g_Y^2(M)}.

    \[
      \frac{3}{5g_Y^2(\mu)}-\frac{1}{g_w^2(\mu)}
        =\left(b_w-\frac35b_Y\right)L.

    \[
      \boxed{
      \frac{1}{g_s^2(\mu)}-\frac{1}{g_w^2(\mu)}
      =\frac{b_s-b_w}{\tfrac35b_Y-b_w}\left[
        \frac{3}{5g_Y^2(\mu)}-\frac{1}{g_w^2(\mu)}
      \right].}
    \]

        0&e^{i\theta/2}U_2

    \([SU(3)\times SU(2)\times U(1)_Y]/\mathbb Z_6\); its Lie algebra is the

      \qquad \operatorname{tr}T_Y^2=\frac12.
    \]

  \SupplementarySolution{8}{Derivative Power Counting and \(\phi^6\)-Induced Quartic Flow}
  \begin{enumerate}
    \item[(a)]
    For \(\zeta>1\), decompose
    \[
      \phi=\phi^-+\phi^+,\qquad
      |\mathbf k|<\Lambda/\zeta
      \quad\hbox{or}\quad
      \Lambda/\zeta<|\mathbf k|<\Lambda .
    \]
    First integrate over \(\phi^+\), producing a Wilson free energy with
    cutoff \(\Lambda/\zeta\).  Next set
    \(\mathbf k'=\zeta\mathbf k\), or
    \(\mathbf x'=\mathbf x/\zeta\), to restore the original cutoff.
    Finally set
    \[
      \phi'(\mathbf x')
        =\zeta^{(d-2)/2}\phi^-(\zeta\mathbf x')
    \]
    so that the kinetic term is canonically normalized.  The coefficients
    of the resulting local operator expansion are the flowed parameters.

    \item[(b)]
    The kinetic term is dimensionless only if
    \[
      -d+2+2[\phi]=0.
    \]
    Therefore
    \[
      \boxed{[\phi]=\frac{d-2}{2}.}
    \]
    Comparing the mass and gradient terms gives
    \[
      \boxed{[\mu_0^2]=2}
    \]
    (and hence \([\mu_0]=1\)).

    \item[(c)]
    Interactions require a scale-dependent field renormalization
    \(Z_\phi\).  At a fixed point one writes
    \[
      \Delta_\phi=\frac{d-2+\eta}{2}.
    \]
    Here the anomalous exponent \(\eta\) is determined by the scale
    dependence of \(Z_\phi\); its sign convention depends on whether
    \(Z_\phi\) multiplies the bare or renormalized field.
    Composite operators can also mix with every operator having the same
    symmetries.  These quantum effects are absent from engineering power
    counting and explain the difference.

    Interpreting (18.8.2) as \(2n\) fields and \(2m\) derivatives, its

    \[

        d<d_{\rm c}&\hbox{relevant}\\

  \SupplementarySolution{9}{Cubic Anisotropy and Competing Fixed Points}
  \begin{enumerate}
    \item[(a)]
    Independent sign flips force every component to occur with even
    exponent, and permutations require symmetric combinations.  At degree
    two the only invariant is \(\phi^2\), which is already \(O(N)\)
    invariant.  At degree four there are two independent invariants,
    \[
      (\phi^2)^2,\qquad \sum_a\phi_a^4.
    \]
    Their difference is a genuine cubic anisotropy.  Higher power sums,
    such as \(\sum_a\phi_a^6\), give further cubic but non-\(O(N)\)
    invariants.
    \item[(b)]
    At the Gaussian point
    \([\phi]=(d-2)/2=1-\epsilon/2\).  A nonderivative monomial of degree
    \(k\) has coupling dimension \(d-k(d-2)/2\).  Apart from the common
    mass deformation, the only cubic-symmetric non-\(O(N)\) operator with
    nonnegative coupling dimension near \(d=4\) is therefore the quartic
    \(\sum_a\phi_a^4\).  Operators of degree six or higher and additional
    derivatives are irrelevant near four dimensions.
    \item[(c)]
    The one-loop four-point fish graphs can be organized by the general
    quartic-coupling tensor formula.  It is convenient to define
    \(\bar u=u/(8\pi^2)\) and \(\bar v=v/(8\pi^2)\), and then suppress the
    bars.  Minimal subtraction gives
    \[
      \boxed{\begin{aligned}
        \beta_u&=-\epsilon u+\frac{N+8}{6}u^2+uv,\\
        \beta_v&=-\epsilon v+2uv+\frac32v^2.
      \end{aligned}}
    \]
    Restoring the unscaled couplings, the quadratic numerators over
    \(48\pi^2\) are
    \((N+8)u^2+6uv\) and \(12uv+9v^2\), respectively.  The absence of
    \(v^2\) in \(\beta_u\) and of \(u^2\) in \(\beta_v\) follows because
    two purely diagonal vertices remain diagonal, whereas two
    \(O(N)\)-invariant vertices cannot generate anisotropy.
    \item[(d)]
    In the rescaled variables the four fixed points are
    \[
      \begin{aligned}
        (u_*,v_*)_{\rm G}&=(0,0),\\
        (u_*,v_*)_{O(N)}&=\left(\frac{6\epsilon}{N+8},0\right),\\
        (u_*,v_*)_{\rm I^N}&=\left(0,\frac{2\epsilon}{3}\right),\\
        (u_*,v_*)_{\rm cub}&=\left(\frac{2\epsilon}{N},
          \frac{2\epsilon(N-4)}{3N}\right).
      \end{aligned}
    \]
    When \(u=0\), the action is a sum of \(N\) independent one-component
    \(\phi^4\) theories.  Its interacting fixed point is therefore \(N\)
    decoupled copies of the Ising Wilson--Fisher theory.
    \item[(e)]
    For \(M_{ij}=\partial\beta_i/\partial g_j|_*\), positive eigenvalues
    decay toward the infrared in the convention
    \(\beta=\mu\,d/d\mu\).  The eigenvalues are
    \[
      \begin{aligned}
        \operatorname{spec}M_{\rm G}&=(-\epsilon,-\epsilon),\\
        \operatorname{spec}M_{O(N)}&=
          \left(\epsilon,\epsilon\frac{4-N}{N+8}\right),\\
        \operatorname{spec}M_{\rm I^N}&=
          \left(\epsilon,-\frac{\epsilon}{3}\right),\\
        \operatorname{spec}M_{\rm cub}&=
          \left(\epsilon,\epsilon\frac{N-4}{3N}\right).
      \end{aligned}
    \]
    Hence the \(O(N)\) point is infrared stable for \(N<4\), while the
    cubic point is stable for \(N>4\).  They merge at \(N=4\), where the
    anisotropy is marginal at this order.  The Gaussian and
    decoupled-Ising points are unstable or saddle points.  A flow sketch
    therefore has the generic critical trajectories ending at \(O(N)\)
    below four components and at the mixed cubic point above four
    components, separated from runaway regions by the stable manifolds of
    the saddles.
    \item[(f)]
    Once a trajectory leaves the stability cone of the quartic form, the
    quartic truncation cannot decide the global potential.  Positive
    sextic terms such as
    \((\phi^2)^3\), \((\phi^2)\sum_a\phi_a^4\), and
    \(\sum_a\phi_a^6\), as well as the bounded microscopic spin length,
    stabilize large fields.  A flow that runs away before reaching a
    scale-invariant fixed point is the perturbative signal of a
    fluctuation-induced first-order transition.  It is therefore a real
    change from the continuous mean-field prediction, not merely a shift
    of critical exponents.
  \end{enumerate}

  \SupplementarySolution{10}{The \(O(N)\) Model at Large \(N\)}
  \begin{enumerate}
    \item[(a)]
    The free kinetic term gives
    \([\phi]=(D-2)/2\).  Since the action is dimensionless,
    \[
      \boxed{[g]=D-4[\phi]=4-D.}
    \]
    Thus the quartic coupling is relevant below four dimensions,
    irrelevant above four dimensions, and classically marginal at \(D=4\).
    \item[(b)]
    Every quartic vertex contributes \(g/N\), whereas every freely summed
    closed \(O(N)\) index loop contributes \(N\).  The leading connected
    vacuum graphs maximize the number of index loops and scale as
    \(N^{L_{\rm index}-V}=N\).  Choosing a coupling independent of \(N\)
    would suppress interactions; omitting the \(1/N\) would make them
    overwhelm the free \(O(N)\) contribution.  The normalization \(g/N\)
    therefore produces a nontrivial saddle and an extensive
    \(O(N)\) free energy.
    \item[(c)]
    Introduce fields \(\rho\) and \(\lambda\) so that
    \(\phi^2/N=\rho\) is enforced by \(\lambda\).  Up to a choice of
    contour for \(\lambda\), the action becomes
    \[
      S_E=\frac12\int d^Dx\,\phi_a(-\partial^2+\lambda)\phi_a
      +N\int d^Dx\left[
        \frac g4\rho^2+\frac{r-\lambda}{2}\rho\right].
    \]
    Integrating out the \(N\) Gaussian fields gives
    \[
      \frac{S_{\rm eff}}N=
      \frac12\operatorname{Tr}\ln(-\partial^2+\lambda)
      +\int d^Dx\left[
        \frac g4\rho^2+\frac{r-\lambda}{2}\rho\right].
    \]
    The overall \(N\) makes the saddle exact at leading order.

    For a translation-invariant symmetric saddle put
    \(m^2=\lambda\).  With a sharp ultraviolet cutoff,
    \[
      \rho=I_D(m),\qquad m^2=r+g\rho,\qquad
      I_D(m)=\int_{|k|<\Lambda}\frac{d^Dk}{(2\pi)^D}
        \frac1{k^2+m^2}.
    \]
    At this order the \(\phi_a\) are free massive fields with propagator
    \(\delta_{ab}/(k^2+m^2)\), while fluctuations of the auxiliary fields
    begin at relative order \(1/N\).

    To display the broken phase, allow
    \(\langle\phi_1\rangle=\sqrt N\,\varphi\).  The saddle equations become
    \[
      m^2\varphi=0,\qquad
      m^2=r+g\bigl[\varphi^2+I_D(m)\bigr].
    \]
    For \(D>2\), define the cutoff-dependent critical parameter
    \[
      r_c=-gI_D(0).
    \]
    If \(r>r_c\), then \(\varphi=0\) and \(m>0\).  At \(r=r_c\), \(m=0\).
    If \(r<r_c\), then \(m=0\) and
    \(\varphi^2=(r_c-r)/g\); the \(N-1\) transverse modes are Goldstone
    bosons.

    Write \(t=r-r_c\).  On the symmetric side,
    \[
      t=m^2+g\,[I_D(0)-I_D(m)].
    \]
    The small-\(m\) behavior separates the requested regimes:
    \[
    \begin{array}{c|c}
      D>4&
      I_D(0)-I_D(m)=c_D\Lambda^{D-4}m^2+\cdots\\
      D=4&
      I_4(0)-I_4(m)=
        \dfrac{m^2}{16\pi^2}\ln\dfrac{\Lambda^2}{m^2}
        +O(m^2)\\[6pt]
      2<D<4&
      I_D(0)-I_D(m)=
        A_Dm^{D-2}+o(m^{D-2})
    \end{array}
    \]
    with
    \[
      A_D=-\frac{\Gamma(1-D/2)}{(4\pi)^{D/2}}>0 .
    \]
    Thus \(D>4\) has mean-field scaling; \(D=4\) has mean-field powers
    multiplied by logarithms; and \(2<D<4\) has interacting large-\(N\)
    scaling.  For \(D\le2\), \(I_D(0)\) is infrared divergent.  No finite
    \(r_c\) supports a massless broken saddle, in accord with the
    lower-critical-dimension obstruction.
    \item[(d)]
    Since \(\xi=m^{-1}\), the preceding gap equation gives
    \[
      \boxed{\nu=
      \begin{cases}
        \dfrac1{D-2},&2<D<4,\\[4pt]
        \dfrac12,&D>4,
      \end{cases}}
    \]
    with \(\nu=1/2\) and logarithmic corrections at \(D=4\).
    At \(D=4-\epsilon\),
    \[
      \nu=\frac1{2-\epsilon}
        =\frac12+\frac{\epsilon}{4}+O(\epsilon^2).
    \]
    The one-loop Wilson--Fisher result is
    \[
      \nu=\frac12+
        \frac{N+2}{4(N+8)}\epsilon+O(\epsilon^2),
    \]
    whose \(N\to\infty\) limit agrees exactly with the expansion above.
  \end{enumerate}

  \SupplementarySolution{11}{The \(\phi^6\) Deformation at Wilson--Fisher}
  Setting \(n=6\) gives
  \[
    \Delta_{\phi^6}=6-3\epsilon+5\epsilon+O(\epsilon^2)
      =6+2\epsilon+O(\epsilon^2).
  \]
  Hence
  \[
    y_6=d-\Delta_{\phi^6}
      =(4-\epsilon)-(6+2\epsilon)+O(\epsilon^2)
      =-2-3\epsilon+O(\epsilon^2).
  \]
  The deformation is irrelevant.  At the Gaussian point its engineering
  eigenvalue is \(d-6(d-2)/2=-2+2\epsilon\), so the interaction makes it
  still more irrelevant near four dimensions.

  This conclusion is local to the ordinary Wilson--Fisher fixed point and
  controlled by the \(4-\epsilon\) expansion.  In \(d=3\), \(\phi^6\) is
  classically marginal; after tuning the mass and quartic coupling, another
  tricritical fixed point can govern the flow.  Irrelevance at one fixed
  point does not rule out that distinct critical surface.

  \SupplementarySolution{12}{A Particle Potential and Weyl Asymptotics}
  \begin{enumerate}
    \item[(a)] With \(\varphi=\sqrt m\,q\),
      \[
        S_E=\int dt\left[\frac12\dot\varphi^2+
          V(\varphi/\sqrt m)\right].
      \]
      Under \(t=b\,t'\), invariance of the kinetic term requires
      \(\varphi(t)=b^{1/2}\varphi'(t')\).  Expanding a smooth potential,
      \(V=\sum_{\mathbf n}g_{\mathbf n}\varphi^{\mathbf n}\), gives
      \[
        g_{\mathbf n}'=b^{\,1+|\mathbf n|/2}g_{\mathbf n}.
      \]
      Every Taylor coefficient therefore grows toward long times and is
      relevant at the free-particle fixed point.  For
      \(V\sim|\varphi|^{-\alpha}\), the coupling scales as
      \(b^{1-\alpha/2}\): it is relevant for \(\alpha<2\), marginal at
      \(\alpha=2\), and irrelevant by this power count for \(\alpha>2\).
    \item[(b)] High energies correspond to short proper time.  The free
      heat trace in a box is
      \[
        Z_0(\beta)
          =L^d\int\frac{d^dp}{(2\pi)^d}
             e^{-\beta p^2/(2m)}
          =L^d\left(\frac{m}{2\pi\beta}\right)^{d/2}.
      \]
      A smooth \(V\) enters the heat-kernel expansion at lower powers of
      \(E\), so the leading coefficient is free.  Since
      \[
        Z(\beta)=\int_0^\infty e^{-\beta E}\,dN(E),
      \]
      a Tauberian inversion of \(Z\sim
      L^d(m/2\pi\beta)^{d/2}\) yields
      \[
        N(E)\sim \frac{L^d}{(2\pi)^d}
          \operatorname{Vol}(B_d)(2mE)^{d/2}.
      \]
      Equivalently, the imaginary part of
      \(\operatorname{Tr}(\omega+i0-H)^{-1}\) gives the density of states;
      representing the resolvent by a proper-time integral leads to the
      same short-time asymptotics.
  \end{enumerate}

\end{enumerate}
