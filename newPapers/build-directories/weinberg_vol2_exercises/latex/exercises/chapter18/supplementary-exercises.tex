  {Inspired by Vadim Kaplunovsky, Renormalization Group Techniques (2024).}
  {kaplunovsky-rg-techniques-2024-v2ch18-p01}
  \begin{enumerate}
    \item[(a)]
    In massive \(\lambda\phi^4\) theory, suppose a fixed-angle amplitude has
    terms \(\lambda+\lambda^2L/(16\pi^2)
    +\lambda^3L^2/(16\pi^2)^2+\cdots\), where
    \(L=\ln(E^2/m^2)\).  State the domain in which fixed-order perturbation
    theory fails even though \(\lambda/(16\pi^2)\ll1\), and explain what the
    renormalization group reorganizes.
    \item[(b)]
    Solve
    \(\mu\,d\lambda/d\mu=3\lambda^2/(16\pi^2)\) with
    \(\lambda(\mu_0)=\lambda_0\).  Expand the exact one-loop solution through
    order \(\lambda_0^3\) and identify the leading logarithm at each order.
    \item[(c)]
    For the running coupling found in part~(b), determine the
    scale at which it diverges.  If \(\lambda_0>0\), explain why fixing
    \(\lambda_0\) while taking an ultraviolet cutoff to infinity is
    incompatible with a perturbatively interacting continuum theory.
  \end{enumerate}

  \SupplementaryExercise{2}{Wilsonian Flow with Cubic and Quartic Interactions}
  {Adapted from University of Cambridge, Part III Statistical Field Theory,
  2025, Paper 303, Question 2 (printed p.~3).}
  {cambridge-part-iii-sft-2025-p303-q2-v2ch18-p02}
  Consider a real scalar field \(\phi\) in \(d\) Euclidean dimensions with
  effective free energy
  \[
    F[\phi]=\int d^dx\left[
      \frac12(\boldsymbol\nabla\phi)^2+\frac12\mu^2\phi^2+\cdots
    \right],
  \]
  where the ellipsis denotes interactions such as \(g\phi^4\).  Impose an
  ultraviolet momentum cutoff \(\Lambda\).
  \begin{enumerate}
    \item[(a)]
    \begin{enumerate}
      \item[(i)] Describe the three momentum-space RG steps.  Why do they
      define a flow on the couplings of the theory?
      \item[(ii)] Define the beta functions.  Explain how their
      linearization at a fixed point determines the scaling dimensions of
      the couplings.
      \item[(iii)] Define relevant, irrelevant, and marginal couplings.
      What is a critical surface, and how does it account for universality?
    \end{enumerate}
    \item[(b)]
    Now take the interaction terms to be
    \(\lambda\phi^3+g\phi^4\), with no factorials absorbed into the
    couplings.
    \begin{enumerate}
      \item[(i)] Starting from
      \(\mu^2=\mu_0^2\), \(\lambda=\lambda_0\), and \(g=g_0\), with
      \(\lambda_0\) and \(g_0\) small, carry out the first RG step and
      determine the contributions to \(\mu'^2\) to leading nonvanishing
      order in \(g_0\) and \(\lambda_0\).

      You may use Wick's theorem and
      \[
        \bigl\langle\phi_{\mathbf k}^{+}
          \phi_{\mathbf k'}^{+}\bigr\rangle_{+}
        =(2\pi)^d\delta^{(d)}(\mathbf k+\mathbf k')G_0(\mathbf k),
        \qquad
        G_0(\mathbf k)=\frac{1}{\mathbf k^2+\mu_0^2},
      \]
      where \(\phi^+\) contains the modes in the shell being integrated
      out and \(\langle\cdot\rangle_+\) is their Gaussian average.  You may
      leave the answer as momentum integrals.
      \item[(ii)] At what lowest order does \(\lambda_0\) correct \(g\)?
      Draw a corresponding Feynman diagram; no evaluation is required.
      \item[(iii)] If \(\lambda_0=0\), at what lowest order does \(g_0\)
      correct \(\lambda\)?  No evaluation is required.
    \end{enumerate}
  \end{enumerate}

  \SupplementaryExercise{4}{The \(O(N)\) Nonlinear Sigma Model Near Two Dimensions}
  {Adapted from University of Cambridge, Part III Statistical Field Theory,
  2021, Paper 303, Question 3 (printed p.~4).}
  {cambridge-part-iii-sft-2021-p303-q3-v2ch18-p04}
  Let \(\mathbf n(x)\) be an \(N\)-component real field constrained by
  \(\mathbf n\cdot\mathbf n=1\) in \(d\) Euclidean dimensions, with free
  energy
  \[
    F[\mathbf n]=\int d^dx\,\frac{1}{2g}
      (\partial_i n_A)(\partial_i n_A).
  \]
  Repeated indices are summed, with \(i=1,\ldots,d\) and
  \(A=1,\ldots,N\).
  \begin{enumerate}
    \item[(a)] Determine the naive engineering dimension of \(g\).
    \item[(b)] Write
    \(\mathbf n(x)=(\boldsymbol\pi(x),\sigma(x))\), where
    \(\boldsymbol\pi\) has \(N-1\) components.  Show that
    \[
      F[\boldsymbol\pi]=\int d^dx\,\frac{1}{2g}\left[
        (\partial_i\pi_a)(\partial_i\pi_a)
        +\frac{\pi_a(\partial_i\pi_a)\,
          \pi_b(\partial_i\pi_b)}{1-\boldsymbol\pi^2}
      \right],
    \]
    where \(a,b=1,\ldots,N-1\).
    \item[(c)] Expand for small \(\boldsymbol\pi\), integrate out momenta
    in the shell \(\Lambda/\zeta<|\mathbf q|<\Lambda\), and show to leading
    order that
    \[
      \frac{1}{g(\zeta)}=\zeta^{d-2}\left[
        \frac{1}{g_0}+(2-N)I_d
      \right],
      \qquad
      I_d=\int_{\Lambda/\zeta}^{\Lambda}
        \frac{d^dq}{(2\pi)^d}\frac{1}{\mathbf q^2}.
    \]
    You may use
    \[
      \bigl\langle\pi_{\mathbf k a}^{+}
        \pi_{\mathbf k'b}^{+}\bigr\rangle_+
      =g_0\delta_{ab}(2\pi)^d
        \delta^{(d)}(\mathbf k+\mathbf k')\frac{1}{\mathbf k^2},
    \]
    and the field rescaling
    \[
      \boldsymbol\pi'=\frac{\boldsymbol\pi^-}{A},
      \qquad
      A=1-\frac12(N-1)g_0I_d.
    \]
    Here \(\boldsymbol\pi^+\) and \(\boldsymbol\pi^-\) are respectively
    the shell and retained fields.
    \item[(d)] Put \(d=2+\epsilon\), with \(\epsilon\geq0\) small, and
    \(s=\ln\zeta\).  Show that
    \[
      \frac{dg}{ds}\simeq-\epsilon g
        +\frac{\Omega_{d-1}}{(2\pi)^d}\Lambda^\epsilon(N-2)g^2
      \simeq-\epsilon g+(N-2)\Lambda^\epsilon\frac{g^2}{2\pi},
    \]
    where \(\Omega_{d-1}\) is the area of the unit sphere
    \(S^{d-1}\).
    \item[(e)] Find the fixed points \(g_*\) of this equation.
    \item[(f)] Identifying \(g\) with temperature, determine the
    correlation-length exponent \(\nu\) associated with the critical fixed
    point or points.
  \end{enumerate}

  \SupplementaryExercise{5}{Dimensional Transmutation in the Gross--Neveu Model}
  {Adapted from John McGreevy, Physics 215C QFT, Spring 2025,
  Assignment 7, Problem 1 (printed p.~1).}
  {mcgreevy-215c-s25-hw7-problem1-v2ch18-p05}
  This problem uses very similar steps to the analysis of the \(O(N)\)
  model, but leads to very different physical conclusions.  It emphasizes
  the many applications of this method, illustrates the manipulations used
  in describing the BCS phenomenon, and is an application of fermionic path
  integrals.

  Consider an \(N\)-vector of fermionic spinor fields in \(D\) spacetime
  dimensions,
  \[
    Z=\int[\mathcal D\psi\,\mathcal D\bar\psi]\,e^{iS[\psi]},
    \qquad
    S[\psi]=\int d^Dx\left[
      \bar\psi^{a}i\slashed\partial\psi^{a}
      -\frac{g}{N}\bigl(\bar\psi^{a}\psi^{a}\bigr)^2
    \right],
  \]
  where \(a=1,\ldots,N\) is summed and \(g>0\).  Use the Feynman
  \(i0\) prescription.  A standalone form of the Hubbard--Stratonovich
  identity is
  \[
    \begin{aligned}
      \exp\left[-i\frac{g}{N}\int d^Dx\,X^2\right]
      &\propto\int_{\mathcal C}\mathcal D\sigma\,
        \exp\left\{i\int d^Dx\left[
          \frac{N}{4g}\sigma^2-\sigma X
        \right]\right\},\\
      X&=\bar\psi^a\psi^a,
    \end{aligned}
  \]
  with the steepest-descent contour \(\mathcal C\) fixed by the same
  \(i0\) prescription.  Regulate the saddle equation after Wick rotation
  with a rotationally invariant Euclidean momentum cutoff \(\Lambda\).
  \begin{enumerate}
    \item[(a)] At the free fixed point, determine the engineering dimension
    of \(g\) as a function of \(D\).  Show that it is classically marginal
    in \(D=2\), so the classical action is scale invariant there.
    \item[(b)] Show that this model in \(D=2\) exhibits dimensional
    transmutation in the form of a dynamically generated mass gap.  First
    use the Hubbard--Stratonovich trick to replace \(\psi^4\) by
    \(\sigma\psi^2+\sigma^2\) in the action, where \(\sigma\) is a scalar
    field, using the displayed identity to fix the normalization.  Then
    integrate out the \(\psi\) fields.  Find the saddle-point equation for
    \(\sigma\); argue that the saddle point dominates the integral for large
    \(N\).  Find a translation-invariant saddle point.  Renormalize the
    logarithmically divergent equation and exhibit the regulator-independent
    mass scale.  Plug the saddle-point configuration of \(\sigma\) back into
    the action for \(\psi\) and describe the resulting dynamics.
  \end{enumerate}

  \SupplementaryExercise{6}{Asymptotic Freedom, Unification, and \(SU(5)\)}
  {Adapted from University of Cambridge, Part III The Standard Model,
  2024, Paper 305, Question 2 (printed p.~3).}
  {cambridge-part-iii-standard-model-2024-p305-q2-v2ch18-p06}
  A gauge coupling \(g\) depends on the scale \(\mu\) according to
  \[
    \frac{1}{g^2(\mu)}=\frac{1}{g_0^2}
      -\frac{b_0}{3(4\pi)^2}
       \ln\left(\frac{\Lambda_{\mathrm{UV}}^2}{\mu^2}\right).
  \]
  Here \(\Lambda_{\mathrm{UV}}\) is the ultraviolet cutoff and \(b_0\)
  is the one-loop beta-function coefficient.  For an \(SU(N_c)\) theory
  coupled to \(N_f\) Weyl fermions (of either chirality) and \(N_s\)
  complex scalars, all in the fundamental or antifundamental
  representation,
  \[
    b_0=11N_c-N_f-\frac12N_s.
  \]
  \begin{enumerate}
    \item[(i)] What does asymptotic freedom mean, and what condition on
    \(N_f,N_s,N_c\) gives it?
    \item[(ii)] For an asymptotically free theory, define its strong-coupling
    scale \(\Lambda\).  In QCD this is conventionally denoted
    \(\Lambda_{\mathrm{QCD}}\).
    \item[(iii)] For the Standard Model with \(N_g\) generations and its
    usual one complex Higgs doublet, find the largest integer \(N_g\) for
    which the \(SU(3)\) gauge group is asymptotically free, and then do the
    same for \(SU(2)\).  Count Weyl fields, including their multiplicities
    under the spectator gauge groups.
    \item[(iv)] Denote the \(SU(3)\), \(SU(2)\), and \(U(1)_Y\) couplings
    by \(g_s,g_w,g_Y\).  Suppose there is a scale \(M\) at which
    \[
      g_s^2(M)=g_w^2(M)=\frac53g_Y^2(M).
    \]
    Show that at any \(\mu<M\),
    \[
      \frac{1}{g_s^2(\mu)}-\frac{1}{g_w^2(\mu)}
      =\frac{b_s-b_w}{\tfrac35b_Y-b_w}\left[
        \frac{3}{5g_Y^2(\mu)}-\frac{1}{g_w^2(\mu)}
      \right],
    \]
    where \(b_s,b_w,b_Y\) are the respective one-loop coefficients in the
    normalization used above.
    \item[(v)] Embed
    \(U(1)_Y\times SU(2)_w\times SU(3)_s\) in \(SU(5)\).  By comparing the
    covariant derivative on the \(\mathbf5\) of \(SU(5)\) with the
    Standard Model covariant derivative, explain the factor \(5/3\) in the
    unification relation.
  \end{enumerate}

  \SupplementaryExercise{9}{Cubic Anisotropy and Competing Fixed Points}
  {Adapted from John McGreevy, Physics 215C, Spring 2025,
  Homework 5, Problem 3.}
  {mcgreevy-215c-s25-hw5-problem3-v2ch18-p09}
  Consider an \(N\)-component Landau--Ginzburg field in
  \(d=4-\epsilon\) dimensions,
  \[
    S=\int d^dx\left[
      \frac12(\partial_\mu\phi_a)^2+\frac r2\phi_a^2
      +\frac{\mu^\epsilon}{4!}\left(
        u(\phi_a\phi_a)^2+v\sum_{a=1}^N\phi_a^4\right)\right].
  \]
  The \(v\)-interaction preserves permutations, independent sign flips,
  and quarter-turns of the components, but not the full \(O(N)\) group.
  \begin{enumerate}
    \item[(a)]
    Before using the displayed action, classify low-degree polynomials in
    \(\phi_a\) that are invariant under the cubic group but not under
    \(O(N)\).
    \item[(b)]
    Explain why \(\sum_a\phi_a^4\) is the only additional nonderivative
    interaction that can be relevant or marginal near four dimensions
    while preserving cubic symmetry.
    \item[(c)]
    Treat \(u\) and \(v\) as \(O(\epsilon)\).  Compute their one-loop beta
    functions through quadratic order.  In the notation
    \[
      \beta_u=-\epsilon u+A_1u^2+A_2uv+A_3v^2,\qquad
      \beta_v=-\epsilon v+B_1u^2+B_2uv+B_3v^2,
    \]
    verify in particular that \(A_3=B_1=0\).
    \item[(d)]
    Find the Gaussian, \(O(N)\), decoupled-Ising, and mixed cubic fixed
    points.  Explain the physical meaning of the decoupled-Ising point.
    \emph{Hint.}  When the \(O(N)\)-symmetric quartic vanishes, the action
    separates into \(N\) one-component theories.
    \item[(e)]
    Linearize the two-coupling flow at every fixed point.  Determine the
    infrared-stable fixed point separately for \(N<4\) and \(N>4\), and
    sketch the resulting flow in the \((u,v)\) plane.
    \item[(f)]
    For \(N>4\), some trajectories run toward a negative quartic direction
    and make the quartic approximation to the potential unbounded.
    Identify the operators omitted from the truncation that restore
    stability and explain what such a runaway implies for the order of the
    transition.
  \end{enumerate}

  \SupplementaryExercise{10}{The \(O(N)\) Model at Large \(N\)}
  {Adapted from John McGreevy, Physics 215C, Spring 2025,
  Homework 6, Problem 1.}
  {mcgreevy-215c-s25-hw6-problem1-v2ch18-p10}
  In \(D\)-dimensional Euclidean space take
  \[
    S_E[\boldsymbol\phi]=\int d^Dx\left[
      \frac12\partial_\mu\phi_a\partial_\mu\phi_a
      +\frac r2\phi_a\phi_a
      +\frac g{4N}(\phi_a\phi_a)^2\right],
      \qquad g>0 .
  \]
  \begin{enumerate}
    \item[(a)]
    Determine the engineering dimension of \(g\) as a function of \(D\)
    and verify that it is classically marginal in four dimensions.
    \item[(b)]
    Optionally, use index-loop counting in vacuum graphs to show why
    \(g/N\) is the useful large-\(N\) normalization and why the leading
    free energy is \(O(N)\).
    \item[(c)]
    Introduce an auxiliary field that replaces the quartic interaction by
    a quadratic term in \(\boldsymbol\phi\), integrate out the \(N\)
    components, and derive the translation-invariant saddle equations.
    Analyze the phases and critical behavior as \(r\) is varied, treating
    separately \(D>4\), \(D=4\), \(2<D<4\), and \(D\le2\).  State the
    dynamics of \(\boldsymbol\phi\) at the saddle and regulate all divergent
    integrals explicitly.

    \emph{Hint.}  For \(X=\phi_a\phi_a\), a useful Gaussian identity is
    \[
      e^{-gX^2/(4N)}
        =\sqrt{\frac{N}{\pi g}}\int_{-\infty}^{\infty}d\sigma\,
          \exp\left(-\frac{N\sigma^2}{g}+i\sigma X\right).
    \]
    \item[(d)]
    At leading order in \(1/N\), compute the correlation-length exponent
    \(\nu\).  Expand your answer near \(D=4-\epsilon\) and compare it with
    the large-\(N\) limit of the Wilson--Fisher epsilon expansion.
  \end{enumerate}

  \SupplementaryExercise{12}{A Particle Potential and Weyl Asymptotics}
  {Adapted from John McGreevy, Physics 215C, Assignment 4, Problem 2 (2025).}
  {mcgreevy-215c-s25-a4-p2-v2ch18-p12}
  A particle in \(d\) spatial dimensions has
  \[
    H=\frac{p^2}{2m}+V(q),\qquad
    Z(\beta)=\int_{q(\beta)=q(0)}\mathcal Dq\,e^{-S_E[q]}
      =\operatorname{Tr}e^{-\beta H}.
  \]
  \begin{enumerate}
    \item[(a)] Rescale to \(\varphi=\sqrt m\,q\) and perform a short-time
      Wilsonian scaling analysis about the free-particle fixed point.  Show
      that every smooth potential is relevant.  If
      \(V(q)\sim |q|^{-\alpha}\), determine the threshold at which the
      singularity changes this conclusion.
    \item[(b)] Use the ultraviolet irrelevance of the potential to explain
      the leading high-energy counting law for a particle in a box of
      volume \(L^d\):
      \[
        N(E)\sim K_dL^dE^{d/2},
        \qquad
        K_d=\frac{\operatorname{Vol}(B_d)(2m)^{d/2}}{(2\pi)^d}.
      \]
      Connect the argument to the short-\(\beta\) heat kernel and to the
      inverse-Laplace representation of the resolvent or density of states.
  \end{enumerate}

\end{enumerate}
