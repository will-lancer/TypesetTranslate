  \SupplementaryExercise{1}{Large Logarithms, Scalar Running, and the Landau Scale}
  {Inspired by Vadim Kaplunovsky, Renormalization Group Techniques (2024).}
  {kaplunovsky-rg-techniques-2024-v2ch18-p01}
  \begin{enumerate}
    \item[(a)]
    In massive \(\lambda\phi^4\) theory, suppose a fixed-angle amplitude has
    terms \(\lambda+\lambda^2L/(16\pi^2)
    +\lambda^3L^2/(16\pi^2)^2+\cdots\), where
    \(L=\ln(E^2/m^2)\).  State the domain in which fixed-order perturbation
    theory fails even though \(\lambda/(16\pi^2)\ll1\), and explain what the
    renormalization group reorganizes.
    \item[(b)]
    Solve
    \(\mu\,d\lambda/d\mu=3\lambda^2/(16\pi^2)\) with
    \(\lambda(\mu_0)=\lambda_0\).  Expand the exact one-loop solution through
    order \(\lambda_0^3\) and identify the leading logarithm at each order.
    \item[(c)]
    For the running coupling found in part~(b), determine the
    scale at which it diverges.  If \(\lambda_0>0\), explain why fixing
    \(\lambda_0\) while taking an ultraviolet cutoff to infinity is
    incompatible with a perturbatively interacting continuum theory.
  \end{enumerate}

  \SupplementaryExercise{2}{Thresholds, Simple Poles, and Characteristic Curves}
  {Inspired by Vadim Kaplunovsky, Renormalization Group Techniques (2024).}
  {kaplunovsky-rg-techniques-2024-v2ch18-p02}
  \begin{enumerate}
    \item[(a)]
    Neglect finite threshold constants and take two unit-charge Dirac
    fermions with masses \(m_1<m_2\).  Starting from \(\alpha(\mu_0)\) at
    \(\mu_0<m_1\), derive \(\alpha(E)\) for \(E>m_2\) by matching the
    one-loop inverse coupling continuously at \(m_1\) and \(m_2\).
    \item[(b)]
    In \(d=4-\epsilon\), let a dimensionless coupling obey
    \(g_B=\mu^{\rho\epsilon}[g+a g^2/\epsilon+O(g^3)]\).
    Differentiate at fixed \(g_B\) and determine the four-dimensional
    beta-function through order \(g^2\), keeping the engineering term until
    the end.
    \item[(c)]
    A dimensionless quantity \(F(L,g)\), with
    \(L=\ln(E/\mu)\), satisfies
    \((-\partial_L+\beta(g)\partial_g)F=0\),
    \(\beta(g)=b g^2\), and \(F(0,g)=g\).
    Solve by characteristics and extract the coefficient of
    \(g^{n+1}L^n\).
  \end{enumerate}

  \SupplementaryExercise{3}{Fixed-Point Stability and Scheme Independence}
  {Inspired by Vadim Kaplunovsky, Types of Renormalization Group Flows (2024).}
  {kaplunovsky-rg-flows-2024-v2ch18-p03}
  \begin{enumerate}
    \item[(a)]
    Let \(\beta(g)=-\omega(g-g_*)+c(g-g_*)^2+\cdots\), with
    \(\omega>0\).  Find the first two powers in the large-\(\mu\) expansion
    of \(g(\mu)-g_*\), including the relation between their coefficients.
    \item[(b)]
    Near a fixed point let
    \(M=\begin{pmatrix}-2&1\\0&1\end{pmatrix}\).
    Find its eigenvalues and eigenvectors, solve the linearized flow, and
    identify the set of initial conditions that reaches the fixed point as
    \(\mu\to\infty\).
    \item[(c)]
    Under a nonsingular redefinition
    \(\widetilde g^\ell=f^\ell(\mathbf g)\), prove that the linearized
    beta-function matrices at corresponding fixed points are related by a
    similarity transformation.  State precisely which assumption is needed
    for their eigenvalues to agree.
  \end{enumerate}

  \SupplementaryExercise{4}{Marginal Directions and Ultraviolet Critical Surfaces}
  {Inspired by Vadim Kaplunovsky, Types of Renormalization Group Flows (2024).}
  {kaplunovsky-rg-flows-2024-v2ch18-p04}
  \begin{enumerate}
    \item[(a)]
    Consider
    \(\dot x=-x\) and \(\dot y=a y^2\), where the dot denotes
    \(d/d\ln\mu\).  Classify the fixed point at the origin for both signs of
    \(a\), and show why linearization alone cannot classify the \(y\)
    direction.
    \item[(b)]
    A fixed point in a six-coupling theory has stability eigenvalues
    \[
      (-4,-1,-0.3,0.2,1,3).
    \]
    Determine the dimension and codimension of its ultraviolet critical
    surface and count the independent parameters of an asymptotically safe
    trajectory ending there.
    \item[(c)]
    Define \(\widehat m=m/\mu\) and suppose the mass operator has constant
    anomalous dimension \(\gamma_m\) near a fixed point.  Derive the flow of
    \(\widehat m\), solve it, and give the condition under which the mass
    becomes negligible in the ultraviolet.
  \end{enumerate}

  \SupplementaryExercise{5}{Banks--Zaks Flow and Short-Distance Forces}
  {Inspired by David Tong, Gauge Theory, Part 2 (2018).}
  {tong-gauge-rg-2018-v2ch18-p05}
  \begin{enumerate}
    \item[(a)]
    Suppose
    \(\beta(g)=-b_0g^3/(16\pi^2)
    -b_1g^5/(16\pi^2)^2+\cdots\), with \(b_0>0\) small and
    \(b_1<0\).  Find the nonzero perturbative fixed point, calculate the
    beta-function slope there, and determine whether it is ultraviolet or
    infrared attractive.
    \item[(b)]
    At a four-dimensional gauge-theory fixed point, use dimensional analysis
    to determine the static potential between external electric probes.
    If the leading irrelevant coupling has eigenvalue \(\omega>0\), derive
    the leading large-distance correction to this potential.
    \item[(c)]
    Starting from the momentum-space color-Coulomb interaction and the
    one-loop asymptotically free coupling, choose the scale that minimizes
    logarithms and derive the leading short-distance form of the potential
    between two fundamental probes for \(r\Lambda\ll1\).
  \end{enumerate}

  \SupplementaryExercise{6}{Heavy-Flavor Matching and Inclusive Running-Coupling Probes}
  {Inspired by David Tong, Gauge Theory, Part 2 (2018).}
  {tong-gauge-rg-2018-v2ch18-p06}
  \begin{enumerate}
    \item[(a)]
    Let the QCD coefficient change from \(b_0^{(n_f)}\) above a heavy-quark
    mass \(M\) to \(b_0^{(n_f-1)}\) below it.  Impose continuity of
    \(\alpha_s\) at \(M\) and derive the relation between the two
    one-loop \(\Lambda\) parameters.
    \item[(b)]
    At an energy above \(u,d,s,c,b\) thresholds but below top threshold,
    compute the tree-level ratio
    \(R=\sigma(e^+e^-\to{\rm hadrons})/
    \sigma(e^+e^-\to\mu^+\mu^-)\).
    Explain why the inclusive ratio is infrared safe while an exclusive
    colored final state is not.
    \item[(c)]
    Explain why the two-jet rate begins at order \(\alpha_s^0\) while the
    resolved three-jet rate begins at order \(\alpha_s\).  Show how measuring
    their ratio at two hard scales tests asymptotic freedom without requiring
    a prediction for the internal momentum distribution of a jet.
  \end{enumerate}

  \SupplementaryExercise{7}{The Wilson--Fisher Fixed Point and Its Thermal Exponent}
  {Inspired by David Tong, Statistical Field Theory, Section 3 (2021).}
  {tong-statistical-field-theory-2021-v2ch18-p07}
  \begin{enumerate}
    \item[(a)]
    At the Gaussian fixed point in \(d\) dimensions, determine the RG
    eigenvalue of the coupling \(g_{2n}\) multiplying
    \(\phi^{2n}\).  Classify \(\phi^4\) and \(\phi^6\) as relevant,
    marginal, or irrelevant in \(d=3\) and \(d=4\).
    \item[(b)]
    For
    \(\beta(g)=-\epsilon g+A g^2+O(g^3)\), with \(A>0\),
    find both perturbative fixed points and determine the linearized
    eigenvalue of the quartic direction at the nontrivial one.
    \item[(c)]
    In the theory with interaction \(g(\phi_n\phi_n)^2/4\), use
    \(\beta_g=-\epsilon g+(N+8)g^2/(8\pi^2)\) and
    \(\beta_{g_2}=[-2+(N+2)g/(8\pi^2)]g_2\) to calculate the thermal
    eigenvalue through first order in \(\epsilon\).
  \end{enumerate}

  \SupplementaryExercise{8}{Correlation-Length and Correction-to-Scaling Exponents}
  {Inspired by David Tong, Statistical Field Theory, Sections 3--4 (2021).}
  {tong-statistical-field-theory-2021-v2ch18-p08}
  \begin{enumerate}
    \item[(a)]
    Use the thermal eigenvalue at the Wilson--Fisher fixed point to derive
    the first-order epsilon expansion of \(\nu\).  Evaluate the result for
    \(N=1,2,3\) at \(\epsilon=1\), without claiming higher-order accuracy.
    \item[(b)]
    Linearize the quartic flow at the Wilson--Fisher fixed point and derive
    the leading correction-to-scaling exponent \(\omega\).  Explain why an
    observable generically has a relative correction proportional to
    \(\xi^{-\omega}\).
    \item[(c)]
    Use the topology and momentum dependence of one-loop two-point graphs in
    \(O(N)\) \(\phi^4\) theory to show that field-strength renormalization
    vanishes at order \(g\).  Deduce the order in \(\epsilon\) at which the
    anomalous exponent \(\eta\) first appears.
  \end{enumerate}

  \SupplementaryExercise{9}{Homogeneity, Susceptibility, and Hyperscaling}
  {Inspired by MIT OpenCourseWare 8.334, Statistical Mechanics II (2014).}
  {mit-ocw-stat-mech-ii-2014-v2ch18-p09}
  \begin{enumerate}
    \item[(a)]
    Suppose the singular free-energy density obeys
    \(f_s(t,h)=b^{-d}f_s(t b^{y_t},h b^{y_h})\).
    Choose \(b\) in terms of \(t\) and derive the susceptibility exponent
    \(\gamma_{\rm crit}\) defined by
    \(\chi(t,0)\propto |t|^{-\gamma_{\rm crit}}\).
    \item[(b)]
    Using a single diverging correlation length
    \(\xi\propto |t|^{-\nu}\), argue that the singular free-energy density
    scales as \(\xi^{-d}\).  Derive the hyperscaling relation between the
    specific-heat exponent \(\alpha_{\rm crit}\), \(d\), and \(\nu\).
    \item[(c)]
    Add a coupling \(u\) with eigenvalue \(y_u<0\) to a critical theory.
    Derive the scaling form of an observable \(X(t,u)\) and show explicitly
    how \(u\) changes amplitudes and corrections to scaling without changing
    the leading critical exponent, assuming regularity at \(u=0\).
  \end{enumerate}

  \SupplementaryExercise{10}{Coupled Scaling Fields and Marginal Corrections}
  {Inspired by MIT OpenCourseWare 8.334, Statistical Mechanics II (2014).}
  {mit-ocw-stat-mech-ii-2014-v2ch18-p10}
  \begin{enumerate}
    \item[(a)]
    Two perturbations obey
    \(d(t,v)^T/d\ell=
    \begin{pmatrix}2&1\\1&2\end{pmatrix}(t,v)^T\).
    Find the scaling fields and eigenvalues, identify the more relevant
    direction, and write the general linearized solution.
    \item[(b)]
    Let a dimensionless temperature obey
    \(dT/d\ell=-aT^2\), with \(a>0\).  Solve the flow and show that \(T\)
    is marginally irrelevant.  If the flow stops when \(T(\ell)\) becomes
    order one under reversal toward the ultraviolet, estimate the associated
    exponentially large length scale.
    \item[(c)]
    An observable obeys
    \(X(t,h)=b^{-x_X}X(t b^{y_t},h b^{y_h})\).
    Derive equivalent scaling forms obtained by fixing either the thermal or
    magnetic argument, and identify the crossover combination that must be
    held fixed near the critical point.
  \end{enumerate}

  \SupplementaryExercise{11}{The \(\phi^6\) Deformation at Wilson--Fisher}
  {Adapted from John McGreevy, Physics 215C, Assignment 4, Problem 1 (2025).}
  {mcgreevy-215c-s25-a4-p1-v2ch18-p11}
  Consider the one-component Wilson--Fisher fixed point in
  \(d=4-\epsilon\).  To first order,
  \[
    \Delta_{\phi^n}
      =n\left(1-\frac{\epsilon}{2}\right)
       +\frac{n(n-1)}6\,\epsilon+O(\epsilon^2).
  \]
  Compute the RG eigenvalue of the \(Z_2\)-even deformation
  \(\int d^dx\,g_6\phi^6\).  Decide whether it is relevant near four
  dimensions, compare it with the Gaussian answer, and explain why this
  conclusion does not exclude a separate tricritical fixed point near
  three dimensions.

  \SupplementaryExercise{12}{A Particle Potential and Weyl Asymptotics}
  {Adapted from John McGreevy, Physics 215C, Assignment 4, Problem 2 (2025).}
  {mcgreevy-215c-s25-a4-p2-v2ch18-p12}
  A particle in \(d\) spatial dimensions has
  \[
    H=\frac{p^2}{2m}+V(q),\qquad
    Z(\beta)=\int_{q(\beta)=q(0)}\mathcal Dq\,e^{-S_E[q]}
      =\operatorname{Tr}e^{-\beta H}.
  \]
  \begin{enumerate}
    \item[(a)] Rescale to \(\varphi=\sqrt m\,q\) and perform a short-time
      Wilsonian scaling analysis about the free-particle fixed point.  Show
      that every smooth potential is relevant.  If
      \(V(q)\sim |q|^{-\alpha}\), determine the threshold at which the
      singularity changes this conclusion.
    \item[(b)] Use the ultraviolet irrelevance of the potential to explain
      the leading high-energy counting law for a particle in a box of
      volume \(L^d\):
      \[
        N(E)\sim K_dL^dE^{d/2},
        \qquad
        K_d=\frac{\operatorname{Vol}(B_d)(2m)^{d/2}}{(2\pi)^d}.
      \]
      Connect the argument to the short-\(\beta\) heat kernel and to the
      inverse-Laplace representation of the resolvent or density of states.
  \end{enumerate}

\end{enumerate}
