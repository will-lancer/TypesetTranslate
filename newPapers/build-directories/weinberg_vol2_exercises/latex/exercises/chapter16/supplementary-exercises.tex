  \SupplementaryExercise{1}{Integrating Out a Momentum Shell in \(\Phi^3\) Theory}
  {Adapted from Cambridge Part III, Advanced Quantum Field Theory,
  2024 Paper 304, Question 3.}
  {cambridge-part-iii-aqft-2024-p304-q3-v2ch16-p01}
  In \(d\)-dimensional Euclidean space consider
  \[
    S[\Phi]=\int \dd^dx\left[
      \frac12\partial_\mu\Phi\,\partial^\mu\Phi+
      \frac{g}{3!}\Phi^3\right],
  \]
  with Fourier modes restricted to \(|p|<\Lambda\).  Choose
  \(0<b<1\) and split the momentum-space field as
  \[
    \widetilde\Phi(p)=
    \begin{cases}
      \widetilde\varphi(p),& |p|<b\Lambda,\\
      \widetilde\chi(p),& b\Lambda<|p|<\Lambda.
    \end{cases}
  \]
  \begin{enumerate}
    \item[(a)]
    Prove that \(\Phi(x)=\varphi(x)+\chi(x)\), where
    \(\varphi\) and \(\chi\) are the inverse Fourier transforms of the two
    fields displayed above.
    \item[(b)]
    Define the shell effective action by
    \[
      e^{-W[\varphi]}=\int\mathcal D\chi\,e^{-S[\varphi+\chi]}.
    \]
    Let
    \[
      Z_\chi[J]=\int\mathcal D\chi\,
      \exp\left[-\frac12\int \dd^dx\,(\partial\chi)^2
      -\int \dd^dx\,J\chi\right].
    \]
    Find \(S_{\rm int}[\varphi,\chi]\) and show that
    \[
      e^{-W[\varphi]}=
      e^{-S[\varphi]}
      \left.
      \exp\!\left[-S_{\rm int}
        \!\left(\varphi,-\frac{\delta}{\delta J}\right)\right]
      Z_\chi[J]\right|_{J=0}.
    \]
    Explain how this expression organizes a perturbative expansion.
    \item[(c)]
    Suppose the local part of the effective action is written
    \[
      W[\varphi]=\int \dd^dx\left[
        \frac12(\partial\varphi)^2+
        \frac{g_{\rm eff}}{3!}\varphi^3+\cdots\right],
    \]
    where wavefunction rescaling and the omitted operators may be ignored.
    Identify every diagram required for \(g_{\rm eff}\) through order
    \(g^3\), evaluate the shell integral, and show in \(d=6\) that
    \[
      g_{\rm eff}=g+Cg^3\ln b+\cdots .
    \]
    Determine \(C\).

    \emph{Hint.}  The area of the unit \((d-1)\)-sphere is
    \(2\pi^{d/2}/\Gamma(d/2)\), and
    \(\Gamma(n)=(n-1)!\) for positive integers \(n\).
  \end{enumerate}

  \SupplementaryExercise{2}{Convexity and Linear Symmetries of the Effective Action}
  {Inspired by Ian Lim, Advanced Quantum Field Theory, Section 5 (2019).}
  {lim-advanced-qft-2019-v2ch16-p02}
  \begin{enumerate}
    \item[(a)]
    Work in Euclidean signature, where the connected two-point kernel is
    positive semidefinite.  Use the inverse-Hessian relation and
    \(\Gamma[\phi_0]=-\mathcal V_4V(\phi_0)\) to show that the exact
    effective potential is convex.  Discuss zero modes and state why a
    finite-order loop expansion need not display this property.
    \item[(b)]
    Let a finite invertible matrix \(M\) leave the action and functional
    measure invariant under \(\phi\mapsto M\phi\).  Derive the induced
    transformation of \(J\), prove the corresponding identity for \(W[J]\),
    and then prove \(\Gamma[M\phi]=\Gamma[\phi]\).  Do not assume that \(M\)
    is orthogonal.
    \item[(c)]
    Suppose the action and measure are invariant under
    \(\phi\mapsto-\phi\).  Prove successively that \(Z[J]=Z[-J]\),
    \(\phi_{-J}=-\phi_J\), and \(\Gamma[-\phi]=\Gamma[\phi]\).
    Show that every odd 1PI vertex at the symmetric configuration vanishes,
    and state how the conclusion changes after expansion about a nonzero
    symmetry-breaking minimum.
  \end{enumerate}

  \SupplementaryExercise{3}{Generating Functionals and Linear Symmetries}
  {Adapted from University of Cambridge, Part III Advanced Quantum Field
  Theory, 2021 Paper 304, Question 1 (printed p.~2).}
  {cambridge-part-iii-aqft-2021-p304-q1-v2ch16-p03}
  Consider \(N\) real scalar fields
  \(\phi_a\), \(a=1,\ldots,N\), in \(d>0\) Euclidean dimensions, with an
  otherwise arbitrary classical action \(S[\phi]\).  Couple a real source
  \(J_a\) through \(-\int \dd^dx\,J_a\phi_a\), and use
  \(W[J]=-\log Z[J]\).  Thus \(W\) is the connected generator; the source
  calls it a ``Wilsonian effective action,'' but no momentum-shell
  coarse-graining is intended here.
  \begin{enumerate}
    \item[(a)]
    Write the path-integral expression for \(Z[J]\).  Explain what the source
    \(J\) means, what \(Z[J]\) generates, and how its functional derivatives
    generate correlation functions.
    \item[(b)]
    Define the classical field \(\Phi_a(x)\) and write the quantum effective
    action \(\Gamma[\Phi]\) as the Legendre transform of \(W[J]\).  Derive
    the first and second functional derivatives of \(\Gamma\), and interpret
    each of them in terms of the correlation functions and proper vertices
    introduced above.  Keep the order of all field, source, and spacetime
    indices explicit in the inverse-kernel relation.
    \item[(c)]
    Suppose \(Z[J]\) is expanded perturbatively as a sum of Feynman diagrams.
    State how \(W[J]\) and \(\Gamma[\Phi]\) are expressed as perturbative
    diagram sums, and justify both statements mathematically as well as
    diagrammatically.
    \item[(d)]
    Assume that the classical action and path-integral measure are invariant
    under
    \[
      \phi_a(x)\longmapsto\phi'_a(x)=U_{ab}\phi_b(x),
    \]
    where \(U\) is a constant invertible \(N\times N\) matrix.  Show that
    \(Z[J]\) is invariant under
    \[
      J_a(x)\longmapsto J'_a(x)=J_b(x)U_{ba},
    \]
    and use this result to prove that
    \[
      \Gamma[U\Phi]=\Gamma[\Phi],\qquad
      \Phi'_a(x)=U_{ab}\Phi_b(x).
    \]
    Do not assume that \(U\) is orthogonal.
  \end{enumerate}

  \SupplementaryExercise{4}{Four-Point Vertices, Even-Dimensional Logs, and Goldstone Modes}
  {Inspired by Mark Srednicki, Quantum Field Theory prepublication draft, Sections 17 and 21 (2006).}
  {srednicki-qft-draft-2006-v2ch16-p04}
  \begin{enumerate}
    \item[(a)]
    Obtain the complete relation between \(W_{,ijkl}\) and derivatives of
    \(\Gamma\).  Show that it contains one 1PI four-point term and three
    exchange terms made from pairs of 1PI three-point vertices.  For four
    identical scalars, identify the three inequivalent partitions of the
    external labels.
    \item[(b)]
    Define
    \[
      \mathcal I_{2n}(x)
        =\frac12\int\frac{d^{2n}k_E}{(2\pi)^{2n}}\ln(k_E^2+x).
    \]
    Differentiate sufficiently many times to obtain a convergent integral
    and show that its nonpolynomial part is proportional to
    \((-1)^n x^n\ln(x/\kappa^2)\).  Determine the coefficient and check the
    cases \(D=4\) and \(D=6\).
    \item[(c)]
    For infinitesimal \(O(N)\) rotations
    \(\delta\phi_a=\epsilon^A(T^A)_{ab}\phi_b\), use a change of variables in
    the source functional to derive
    \[
      \int \dd^4x\,\frac{\delta\Gamma}{\delta\phi_a(x)}
        (T^A)_{ab}\phi_b(x)=0.
    \]
    Differentiate this identity once and twice with respect to the field and
    state the resulting constraints on 1PI vertices at a symmetric
    background.
  \end{enumerate}

  \SupplementaryExercise{5}{Loop Counting, Supertraces, and the Derivative Expansion}
  {Inspired by Howard Haber, Physics 222, Functional Integral Derivation of the Effective Potential, pp. 1--4 (2020).}
  {ucsc-physics222-effective-potential-2020-v2ch16-p05}
  \begin{enumerate}
    \item[(a)]
    Replace the action in a functional integral by \(g^{-1}\Gamma[\phi]\).
    Show that a graph with \(I\) internal lines and \(V\) vertices carries
    \(g^{I-V}\).  Use \(L=I-V+1\) for a connected graph to derive the
    \(g^{L-1}\) loop counting of \(W_\Gamma[J,g]\).  Explain why the
    \(g\to0\) saddle retains exactly the connected trees.
    \item[(b)]
    For a translation-invariant scalar theory, expand \(\Gamma[\phi]\) in
    1PI \(n\)-point kernels about \(\phi=0\).  Set \(\phi(x)=\phi_0\) and
    prove that the effective potential is determined by the zero-momentum
    vertices.  Carefully track the spacetime-volume delta function and the
    sign in \(\Gamma[\phi_0]=-\mathcal V_4V(\phi_0)\).
    \item[(c)]
    Consider constant backgrounds in a theory whose bosonic and fermionic
    quadratic fluctuation operators have mass-squared eigenvalues
    \(m_b^2(\phi)\) and \(m_f^2(\phi)\).  Derive a compact one-loop expression
    using a graded trace over species.  State how real scalars, complex
    scalars, and Dirac fermions are weighted, and explain which polynomial
    terms are scheme dependent.
    \item[(d)]
    Let a complex scalar have
    \(V_0=m^2\Phi^*\Phi+\lambda(\Phi^*\Phi)^2/4\).  Choose a real constant
    background and rewrite the two real fluctuations as radial and angular
    modes.  Determine their field-dependent masses and the one-loop
    logarithms.  Verify explicitly that the answer depends only on
    \(\Phi^*\Phi\).
    \item[(e)]
    For a slowly varying real scalar background, write the most general
    Lorentz-invariant effective action through two derivatives as
    \[
      \Gamma[\phi]=\int \dd^4x\,
        \left[-V(\phi)-\frac12 Z(\phi)
        \partial_\mu\phi\,\partial^\mu\phi\right]+O(\partial^4).
    \]
    Explain which momentum derivatives of the 1PI two-point vertex determine
    \(Z(\phi)\), why a constant background determines only \(V\), and how a
    local field redefinition changes \(Z\) and \(V\).
  \end{enumerate}

  \SupplementaryExercise{6}{Scalar Determinants and Counterterms}
  {Inspired by Roman Jackiw, Functional Evaluation of the Effective Potential, Sections II--III (1974).}
  {jackiw-effective-potential-1974-v2ch16-p06}
  \begin{enumerate}
    \item[(a)]
    Expand a real scalar action about a constant background and retain the
    quadratic fluctuation operator \(K\).  Evaluate the Gaussian integral to
    show that the one-loop effective action is proportional to
    \(\frac{\ii}{2}\operatorname{Tr}\ln K\).  After Wick rotation, differentiate
    the four-dimensional momentum integral three times with respect to the
    field-dependent mass squared and recover the logarithmic term.
    \item[(b)]
    For
    \[
      V_0(\boldsymbol\phi)
        =\frac12m^2\boldsymbol\phi^2
         +\frac{\lambda}{4!}(\boldsymbol\phi^2)^2
    \]
    in an \(O(N)\) theory, choose the background
    \(\boldsymbol\phi=(\varphi,0,\ldots,0)\).  Diagonalize the Hessian and
    find the radial and \(N-1\) transverse mass-squared eigenvalues.  Use
    them to write the nonpolynomial part of the four-dimensional one-loop
    potential.
    \item[(c)]
    In \(D\) spacetime dimensions suppose the fluctuation mass is affine,
    \(\mu^2(\phi)=a+b\phi\).  Use power counting of
    \(\frac12\operatorname{Tr}\ln(-\partial^2+\mu^2)\) to determine the
    degree of the divergent polynomial in \(\mu^2\) for even \(D\).
    Specialize to \(D=6\) and explain why vacuum, tadpole, mass, and cubic
    counterterms form a closed set.
  \end{enumerate}

  \SupplementaryExercise{7}{Fermionic Determinants and Graded Legendre Transforms}
  {Inspired by Hong Liu, MIT OpenCourseWare 8.323, Problem Set 10, Problem 3 (2023).}
  {mit-8323-s23-ps10-v2ch16-p07}
  \begin{enumerate}
    \item[(a)]
    Evaluate a finite-dimensional Grassmann Gaussian and then extend the
    result formally to a Dirac field.  Explain why a complex fermion produces
    \(\det D\), whereas a real boson produces \((\det K)^{-1/2}\).  Reduce
    \(\operatorname{Tr}\ln(\slashed p-M)\) to a scalar logarithm and identify
    the spin multiplicity.
    \item[(b)]
    In four dimensions take a real scalar background \(\phi\) and a Dirac
    fermion with mass \(M(\phi)=m+y\phi\).  Compute the nonpolynomial
    fermion-loop contribution to \(V_{\rm eff}(\phi)\).  List all local
    counterterms required when no reflection symmetry is imposed, and show
    how the list is reduced if a symmetry forces \(M(-\phi)=-M(\phi)\).
    \item[(c)]
    Introduce Grassmann sources for fermionic fields, with every current
    written to the right of its field.  Using explicit left and right
    derivatives, derive
    \[
      \frac{\delta_R W}{\delta J_n}=\chi^n,\qquad
      \frac{\delta_L\Gamma}{\delta\chi^n}=-J_n.
    \]
    Work out the graded inverse-Hessian relation for one bosonic and one
    fermionic species and identify where ordinary matrix transposition must
    be replaced by graded transposition.
  \end{enumerate}

  \SupplementaryExercise{8}{Particle-Number Breaking and the Phase-Only Theory}
  {Adapted from University of Cambridge, Part III Applications of Quantum
  Field Theory, 2024 Paper 337, Question 1 (printed p.~2).}
  {cambridge-part-iii-applications-qft-2024-p337-q1-v2ch16-p08}
  Consider a system in \(d\) spatial dimensions with conserved particle
  number \(N\).  The possible spontaneous breaking of particle-number
  symmetry is described by a complex order parameter \(\Psi\) with
  Landau--Ginzburg Lagrangian density
  \[
    \mathcal L
      =\ii\Psi^\dagger\partial_t\Psi
       -\frac1{2m}\partial_i\Psi^\dagger\partial_i\Psi
       -r\Psi^\dagger\Psi
       -\lambda(\Psi^\dagger\Psi)^2+\cdots .
  \]
  \begin{enumerate}
    \item[(a)]
    Write the generator \(N\) in terms of \(\Psi\).  Set
    \(\Psi=\sqrt\rho\,e^{-\ii\theta}\) and obtain the effective Lagrangian
    density for \(\rho\) and \(\theta\).  What is the physical meaning of
    \(\rho\), and what momentum is conjugate to the phase \(\theta\)?  With
    \(\theta_0\equiv\int \dd^dx\,\theta\), derive an uncertainty relation
    between the spatially uniform phase and \(N\), and briefly discuss its
    consequences for spontaneous symmetry breaking.
    \item[(b)]
    At the classical level, what condition on the Landau--Ginzburg theory
    implies spontaneous symmetry breaking?  Assuming that it occurs, write
    \(\rho=\bar\rho+\delta\rho\), determine the stable value \(\bar\rho\),
    and expand \(\mathcal L\) to quadratic order in \(\delta\rho\) and
    \(\theta\).
    \item[(c)]
    Write the path integral for the quadratic theory and integrate out the
    gapped mode \(\delta\rho\).  You may use standard Gaussian-integral
    results and need retain only the leading exponential contribution.  Take
    a long-wavelength expansion and obtain the effective Lagrangian for the
    Goldstone boson.  Give its velocity in terms of the parameters of the
    original Landau--Ginzburg theory.
    \item[(d)]
    Heat the system to \(T>0\), with classical symmetry-breaking phase
    \(\langle\theta\rangle=0\).  You may assume
    \[
      \langle\theta^2\rangle
        =\frac1\chi\int\frac{\dd^dk}{(2\pi)^d}\,
          T\sum_{\omega_n}\frac1{\omega_n^2+v^2k^2},
      \qquad \omega_n=2\pi nT,
    \]
    where \(\chi\) and \(v\) are constants.  By considering a suitable
    contour integral of \(\coth(\pi z)/(z^2-a^2)\), or otherwise, prove
    \[
      \sum_{n=-\infty}^{\infty}\frac1{n^2+a^2}
        =\frac{\pi\coth(\pi a)}a.
    \]
    Hence express \(\langle\theta^2\rangle\) as a single integral over
    \(k=|\mathbf k|\).  Analyze \(d=1,2,3\): in which dimensions is the
    infrared integral finite, and in which can spontaneous symmetry breaking
    occur at \(T>0\)?  Justify the answer.
    \item[(e)]
    Subtract the temperature-independent short-distance divergence and use
    the resulting integral to estimate the temperature at which thermal
    phase fluctuations restore the symmetry in \(d=3\).
  \end{enumerate}

  \SupplementaryExercise{9}{Dimensional Transmutation and RG Invariance}
  {Inspired by Sidney Coleman and Erick Weinberg, Radiative Corrections as the Origin of Spontaneous Symmetry Breaking, Sections II--III (1973).}
  {coleman-weinberg-1973-v2ch16-p09}
  \begin{enumerate}
    \item[(a)]
    Consider a massless theory whose one-loop potential can be written
    \[
      V(\phi)=A\phi^4+B\phi^4\ln(\phi^2/\kappa^2),
      \qquad B>0.
    \]
    Find every nonzero stationary point and determine when it is a local
    minimum.  Eliminate \(A\) in favor of the nonzero minimum \(v\), rewrite
    the potential using \(v\), and explain how a dimensionless coupling has
    been traded for a mass scale.
    \item[(b)]
    Starting from the fact that the bare effective potential is independent
    of the subtraction scale, derive
    \[
      \left(\kappa\frac{\partial}{\partial\kappa}
        +\sum_i\beta_i\frac{\partial}{\partial g_i}
        -\gamma\,\phi\frac{\partial}{\partial\phi}\right)V(\phi)=0.
    \]
    Apply it to
    \(V=\lambda\phi^4/4!+B\phi^4\ln(\phi^2/\kappa^2)\) through one loop and
    find the relation among \(B\), the one-loop beta function, and the
    anomalous dimension.
  \end{enumerate}

  \SupplementaryExercise{10}{Large-\(N\) Sigma Model at Finite Temperature}
  {Adapted from University of Cambridge, Part III Applications of Quantum
  Field Theory, 2023 Paper 337, Question 1 (printed pp.~2--3).}
  {cambridge-part-iii-applications-qft-2023-p337-q1-v2ch16-p10}
  Consider the \(O(N)\) nonlinear sigma model in two spatial dimensions.  Its
  imaginary-time partition function is
  \[
    Z=\int\mathcal D\mathbf n\,\mathcal D\lambda\,e^{-S[\mathbf n,\lambda]},
  \]
  where \(\mathbf n\) is an \(N\)-component vector and \(\lambda\) is a
  Lagrange multiplier, with
  \[
    S[\mathbf n,\lambda]
      =\frac1{2vg}\int \dd^2x\int_0^{1/T}\dd\tau\,
        \left[(\partial_\tau\mathbf n)^2
          +v^2(\boldsymbol\nabla\mathbf n)^2
          +\ii\lambda(\mathbf n^2-N)\right].
  \]
  Here \(T\) is the temperature, \(v\) a velocity, and \(g\) a coupling.
  \begin{enumerate}
    \item[(a)]
    Perform the integral over \(\mathbf n\) and write
    \[
      Z=\int\mathcal D\lambda\,e^{-\widetilde S[\lambda]},
    \]
    expressing \(\widetilde S[\lambda]\) as a functional trace.
    \item[(b)]
    Explain why the large-\(N\) partition function may be evaluated by a
    saddle-point approximation.  Assuming the saddle is independent of
    space and time and writing \(\ii\lambda=m^2\), express its equation as a
    sum over Matsubara frequencies and an integral over momentum.
    \item[(c)]
    Write the zero-temperature saddle equation as an integral.  Regulate any
    high-energy or momentum divergence with a cutoff \(\Lambda\), taking
    \(\Lambda\to\infty\) wherever possible.  Explain why a solution exists
    only for \(g>g_c\), and show that
    \[
      m=\frac{4\pi v(g-g_c)}{gg_c}.
    \]
    Without solving further equations, state what happens physically when
    \(g<g_c\).
    \item[(d)]
    Now take \(T>0\), still with \(g>g_c\), and write the solution as
    \(m(T)\).  Denote the zero-temperature gap by \(\Delta=m(0)\); the
    critical coupling remains defined at \(T=0\).  For a suitable function
    \(F\), you may use
    \[
      T\sum_n F(\ii\omega_n)
        =-\frac{\ii}{4\pi}\int_{\mathcal C}\dd z\,
          \coth\!\left(\frac z{2T}\right)F(z),
    \]
    where \(\mathcal C\) runs down just to the left of the imaginary axis
    and up just to its right.  Use this contour formula to evaluate the
    Matsubara sum in the finite-temperature saddle equation.
    \item[(e)]
    Perform the momentum integral in that equation with a cutoff
    \(\Lambda\).  Remove the divergence by subtracting the zero-temperature
    saddle equation at \(g=g_c\), regulated in the same way, and then take
    \(\Lambda\to\infty\).  Derive
    \[
      m(T)=2T\sinh^{-1}\!\left(\frac12e^{\Delta/(2T)}\right).
    \]
    You may use without proof
    \[
      \int_{-\infty}^{\infty}\frac{\dd\omega}{2\pi}\,
        \frac1{v^2k^2+\omega^2}=\frac1{2kv}.
    \]
    \item[(f)]
    Find the leading temperature dependence of \(m(T)\) for
    \(T\ll\Delta\) and \(T\gg\Delta\), and interpret both limits.
    \item[(g)]
    % SOURCE NOTE: Paper 337 prints the prefactor vg/N here.  Taken together
    % with its displayed 1/(2vg) action and n^2=N constraint, a reader might
    % instead expect the unsummed component covariance to carry vg.  The
    % examination's supplied normalization is preserved without correction.
    The Matsubara Green function of each component of \(\mathbf n\) is
    \[
      G(\ii\omega_n,\mathbf k)
        =\frac{vg}{N}\,
          \frac1{v^2\mathbf k^2+\omega_n^2+m(T)^2}.
    \]
    Analytically continue this expression to the retarded function
    \(G^R(\omega,\mathbf k)\), and calculate its spectral weight
    \(\operatorname{Im}G^R(\omega,\mathbf k)\).  Explain how the result, with
    the displayed \(m(T)\), exhibits quantum-critical scaling.
  \end{enumerate}

  \SupplementaryExercise{11}{Generating Functionals, Legendre Geometry, and Split Independence}
  {Adapted from Cambridge Part III, Advanced Quantum Field Theory, 2019 Paper 304, Question 1.}
  {cambridge-part-iii-aqft-2019-p304-q1-v2ch16-p11}
  \begin{enumerate}
    \item[(a)] For a scalar action \(S[\phi]\), define a source-dependent
      partition function and give an explicit differentiation formula for
      a correlation function.
    \item[(b)] Define the connected generator \(W[J]\), mean field
      \(\Phi\), and quantum effective action \(\Gamma[\Phi]\).  Explain how
      these objects differ from the classical action and from a Wilsonian
      action obtained by integrating out high-momentum modes.
    \item[(c)] Starting from the diagram expansion of \(Z[J]\), prove that
      \(W[J]\) generates connected diagrams and that \(\Gamma[\Phi]\)
      generates one-particle-irreducible diagrams.
    \item[(d)] In zero dimensions set
      \[
        W(J;\phi_0)=-\hbar\log\int \dd\eta\,
        e^{-[S(\phi_0+\eta)+J\eta]/\hbar},\qquad
        \chi=\left.\frac{\partial W}{\partial J}\right|_{J=J_\chi},
      \]
      and define \(\Gamma(\chi;\phi_0)=W(J_\chi;\phi_0)-J_\chi\chi\).
      Without expanding in \(\hbar\), prove
      \[
        \Gamma(\eta;\phi_0)=\Gamma(\phi_0+\eta;0).
      \]
  \end{enumerate}

\end{enumerate}
