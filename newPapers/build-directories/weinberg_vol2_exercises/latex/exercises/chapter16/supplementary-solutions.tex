\begin{enumerate}
  \SupplementarySolution{1}{Integrating Out a Momentum Shell in \(\Phi^3\) Theory}
  \begin{enumerate}
    \item[(a)]
    Insert the two pieces of \(\widetilde\Phi\) into the inverse transform:
    \[
      \Phi(x)=\int_{|p|<b\Lambda}\frac{d^dp}{(2\pi)^d}
        e^{ipx}\widetilde\varphi(p)
      +\int_{b\Lambda<|p|<\Lambda}\frac{d^dp}{(2\pi)^d}
        e^{ipx}\widetilde\chi(p).
    \]
    These two terms are precisely \(\varphi(x)\) and \(\chi(x)\).
    Their Fourier supports are disjoint, so the quadratic cross term
    \(\int\partial\varphi\cdot\partial\chi\) also vanishes.
    \item[(b)]
    Expanding the cubic interaction gives
    \[
      S[\varphi+\chi]=S[\varphi]+S_{\chi,0}
        +S_{\rm int}[\varphi,\chi],
    \]
    where
    \[
      \boxed{S_{\rm int}=\int d^dx\left(
        \frac g2\varphi^2\chi+
        \frac g2\varphi\chi^2+
        \frac g{3!}\chi^3\right).}
    \]
    With the source sign used in the question,
    \(-\delta/\delta J(x)\) inserts \(\chi(x)\).  Taylor expanding the
    exponential of \(S_{\rm int}\), making this replacement, and then
    setting \(J=0\) therefore reproduces the shell integral term by term.
    Wick contractions generated by the Gaussian \(Z_\chi\) are exactly
    the internal high-momentum propagators; the uncontracted
    \(\varphi\)'s are external low-momentum legs.
    \item[(c)]
    At zero external momentum the only new local cubic graph through
    \(g^3\) is the one-loop triangle with three
    \((g/2)\varphi\chi^2\) vertices.  The tree cubic vertex supplies the
    leading \(g\).  Graphs containing a
    \(\varphi^2\chi\) vertex cannot carry a high-shell momentum when the
    external momenta are taken to zero; external-leg self-energies would
    only affect the wavefunction rescaling that the question excludes.

    Put \(V_1=(g/2)\int\varphi\chi^2\).  The connected third cumulant in
    \(-\ln\langle e^{-V_1}\rangle_\chi\) is
    \[
      \frac1{3!}\left(\frac g2\right)^3
      \int_{x,y,z}\varphi_x\varphi_y\varphi_z
      \langle\chi_x^2\chi_y^2\chi_z^2\rangle_c.
    \]
    There are eight connected Wick contractions, all forming the
    triangle.  Expanding the slowly varying fields about one point gives
    \[
      \delta W_{\varphi^3}
        =\frac{g^3}{3!}\int d^dx\,\varphi^3(x)\,I_d(b,\Lambda),
      \qquad
      I_d=\int_{b\Lambda<|k|<\Lambda}
        \frac{d^dk}{(2\pi)^d}\frac1{(k^2)^3}.
    \]
    Thus \(g_{\rm eff}=g+g^3I_d+O(g^5)\).  For \(d=6\),
    \[
      I_6=\frac{\operatorname{Area}(S^5)}{(2\pi)^6}
        \int_{b\Lambda}^{\Lambda}\frac{dk}{k}
      =\frac{\pi^3}{64\pi^6}\ln\frac1b
      =-\frac{\ln b}{64\pi^3}.
    \]
    Consequently
    \[
      \boxed{g_{\rm eff}
        =g-\frac{g^3}{64\pi^3}\ln b+\cdots,\qquad
        C=-\frac1{64\pi^3}.}
    \]
  \end{enumerate}

  \SupplementarySolution{2}{Convexity and Linear Symmetries of the Effective Action}
  \begin{enumerate}
    \item[(a)]
    In Euclidean signature, for every test function \(f_i\),
    \[
      f_iW_{,ij}f_j
        =\left\langle (f_i\chi^i-\langle f_i\chi^i\rangle)^2
          \right\rangle\ge0.
    \]
    After zero modes are regulated, \(\Delta=W^{(2)}\) is positive and
    \(-\Gamma^{(2)}=\Delta^{-1}\) is positive in the convention of this
    chapter.  For a constant scalar,
    \(\Gamma=-\mathcal V_4V\), so the zero-momentum kernel obeys
    \[
      V''(\phi)\ge0.
    \]
    With several scalars, the Hessian of \(V\) is positive semidefinite.

    A zero eigenvalue corresponds to a flat direction or to a limiting
    susceptibility and is handled by a regulated inverse or a
    Legendre--Fenchel transform.  A loop expansion about an unstable
    homogeneous saddle does not perform the global minimization implicit in
    the exact transform, so any finite truncation may remain nonconvex.
    \item[(b)]
    The source pairing is preserved by
    \[
      \phi\mapsto M\phi,\qquad J\mapsto M^{-\mathsf T}J,
    \]
    since
    \((M\phi)^{\mathsf T}M^{-\mathsf T}J=\phi^{\mathsf T}J\).
    A change of integration variables therefore gives
    \[
      W[M^{-\mathsf T}J]=W[J].
    \]
    Differentiation and the chain rule yield
    \[
      \phi_{M^{-\mathsf T}J}=M\phi_J,
      \qquad
      J_{M\phi}=M^{-\mathsf T}J_\phi.
    \]
    Consequently,
    \[
      \begin{split}
      \Gamma[M\phi]
        &=-(M\phi)^{\mathsf T}M^{-\mathsf T}J_\phi
          +W[M^{-\mathsf T}J_\phi]\\
        &=-\phi^{\mathsf T}J_\phi+W[J_\phi]
         =\Gamma[\phi].
      \end{split}
    \]
    Orthogonality was never used; invertibility and invariance of action and
    measure suffice.
    \item[(c)]
    Changing variables \(\phi\mapsto-\phi\) gives
    \[
      Z[J]=Z[-J],\qquad W[J]=W[-J].
    \]
    Differentiation yields
    \(\phi_{-J}=-\phi_J\), and inversion gives
    \(J_{-\phi}=-J_\phi\).  The Legendre transform then satisfies
    \[
      \Gamma[-\phi]
        =-(-\phi)J_{-\phi}+W[J_{-\phi}]
        =-\phi J_\phi+W[J_\phi]
        =\Gamma[\phi].
    \]
    Taking an odd number of derivatives and setting \(\phi=0\) therefore
    gives
    \[
      \Gamma^{(2r+1)}(0)=0.
    \]
    If one instead writes \(\phi=v+\eta\) about a chosen nonzero minimum,
    the reflection exchanges the expansions about \(+v\) and \(-v\);
    it is not simply \(\eta\mapsto-\eta\) within one chosen phase.  Odd
    vertices in \(\eta\) need not vanish.
  \end{enumerate}

  \SupplementarySolution{3}{Generating Functionals and Linear Symmetries}
  \begin{enumerate}
    \item[(a)]
    With the stated source sign,
    \[
      Z[J]=\int\mathcal D\phi\,
        \exp\!\left[-S[\phi]-\int d^dx\,J_a(x)\phi_a(x)\right].
    \]
    The source is a freely specifiable classical function, one for each
    field component.  Differentiating inserts fields:
    \[
      \frac{\delta^n Z}{\delta J_{a_1}(x_1)\cdots
        \delta J_{a_n}(x_n)}
        =(-1)^n Z[J]
          \left\langle\phi_{a_1}(x_1)\cdots
          \phi_{a_n}(x_n)\right\rangle_J.
    \]
    Thus the derivatives at \(J=0\), divided by \(Z[0]\) and supplied with
    the displayed sign, give all Euclidean correlation functions.
    \item[(b)]
    The classical field and Legendre transform are
    \[
      \Phi_a(x)=\frac{\delta W}{\delta J_a(x)}
        =\langle\phi_a(x)\rangle_J,
      \qquad
      \Gamma[\Phi]=W[J_\Phi]
        -\int d^dx\,J_{\Phi a}(x)\Phi_a(x).
    \]
    Varying and using \(\delta W=\int\Phi_a\,\delta J_a\) gives
    \[
      \frac{\delta\Gamma}{\delta\Phi_a(x)}=-J_{\Phi a}(x).
    \]
    A second derivative followed by the chain rule gives the ordered kernel
    identity
    \[
      \int d^dy\,
        \frac{\delta^2\Gamma}
          {\delta\Phi_a(x)\delta\Phi_b(y)}
        \frac{\delta^2W}
          {\delta J_b(y)\delta J_c(z)}
        =-\delta_{ac}\delta^{(d)}(x-z).
    \]
    Here
    \[
      \frac{\delta^2W}{\delta J_a(x)\delta J_b(y)}
        =-\langle\phi_a(x)\phi_b(y)\rangle_{J,\mathrm c}.
    \]
    Hence \(\Gamma^{(2)}\) is the inverse full connected propagator, while
    higher derivatives of \(\Gamma\) are proper, or 1PI, vertices.  The
    first derivative is the external source required to sustain the chosen
    mean field and vanishes at a physical configuration with \(J=0\).
    \item[(c)]
    Every diagram contributing to \(Z\) is a disjoint union of connected
    components.  The exponential formula for such unions says
    \[
      Z[J]=\exp C[J],
    \]
    where \(C[J]\) is the sum of connected vacuum diagrams with source
    insertions.  Therefore \(W=-C\) in the present Euclidean sign convention,
    so its derivatives generate connected correlators with the signs already
    found in part (b).

    Every connected diagram, in turn, has a unique decomposition into a
    tree whose vertices are maximal 1PI subdiagrams and whose edges are full
    connected propagators.  The Legendre transform removes these connecting
    tree edges: the inverse-Hessian identity shows that differentiation of
    \(\Gamma\) amputates the full external propagators, and any bridge would
    instead be generated by joining two lower proper vertices with
    \(-W^{(2)}\).  Consequently the perturbative expansion of \(\Gamma\) is
    precisely the sum of 1PI diagrams with external propagators removed.
    \item[(d)]
    Put \(J'_a=J_bU_{ba}\) and change variables to
    \(\chi_a=U_{ab}\phi_b\).  Invariance of the action and measure gives
    \[
      \begin{split}
      Z[J']
        &=\int\mathcal D\phi\,
          e^{-S[\phi]-\int J_bU_{ba}\phi_a}\\
        &=\int\mathcal D\chi\,
          e^{-S[\chi]-\int J_b\chi_b}=Z[J].
      \end{split}
    \]
    Thus \(W[U^{\mathsf T}J]=W[J]\).  Applying the same statement to
    \(U^{-1}\) gives \(W[U^{-\mathsf T}J]=W[J]\).  If \(J_\Phi\) produces
    \(\Phi\), differentiation of this identity shows that
    \(U^{-\mathsf T}J_\Phi\) produces \(U\Phi\).  The source pairing is
    unchanged,
    \[
      (U^{-\mathsf T}J_\Phi)_a(U\Phi)_a=J_{\Phi a}\Phi_a,
    \]
    and therefore
    \[
      \Gamma[U\Phi]
        =W[U^{-\mathsf T}J_\Phi]
          -\int (U^{-\mathsf T}J_\Phi)\cdot(U\Phi)
        =\Gamma[\Phi].
    \]
    Only invertibility, not orthogonality, was used.
  \end{enumerate}

  \SupplementarySolution{4}{Four-Point Vertices, Even-Dimensional Logs, and Goldstone Modes}
  \begin{enumerate}
    \item[(a)]
    A further current derivative of the preceding result can act on the
    1PI vertex or on any one of its three propagators.  Using the
    three-point formula for the derivative of each propagator gives
    \[
      \begin{split}
      W_{,ijkl}={}&
        \Delta_{ia}\Delta_{jb}\Delta_{kc}\Delta_{ld}
        \bigl[
          \Gamma_{abcd}
          +\Gamma_{abe}\Delta_{ef}\Gamma_{fcd}\\
          &\hspace{35mm}
          +\Gamma_{ace}\Delta_{ef}\Gamma_{fbd}
          +\Gamma_{ade}\Delta_{ef}\Gamma_{fbc}
        \bigr].
      \end{split}
    \]
    The first term is the full 1PI four-vertex.  The remaining terms split
    the labels as
    \[
      (ij|kl),\qquad(ik|jl),\qquad(il|jk),
    \]
    and join the two sides with a full propagator.  They are the three
    exchange channels.  No other connected tree with four labeled leaves
    and vertices of valence at least three exists.
    \item[(b)]
    For \(D=2n\), \(n+1\) derivatives make the integral convergent:
    \[
      \mathcal I_{2n}^{(n+1)}(x)
        =\frac12(-1)^n n!
          \int\frac{d^{2n}k_E}{(2\pi)^{2n}}
          \frac1{(k_E^2+x)^{n+1}}.
    \]
    Dimensional continuation, or direct integration of this derivative,
    gives the nonpolynomial term
    \[
      \boxed{\mathcal I_{2n}^{\rm nonpol}(x)
        =\frac{(-1)^n}{2n!(4\pi)^n}
          x^n\ln\frac{x}{\kappa^2}.}
    \]
    The omitted integration polynomial has degree \(n\) in \(x\).
    For \(n=2\),
    \[
      \mathcal I_4^{\rm nonpol}
        =\frac{x^2}{64\pi^2}\ln\frac{x}{\kappa^2},
    \]
    while for \(n=3\),
    \[
      \mathcal I_6^{\rm nonpol}
        =-\frac{x^3}{12(4\pi)^3}\ln\frac{x}{\kappa^2},
    \]
    in agreement with the four- and six-dimensional calculations.
    \item[(c)]
    Invariance of the action and measure under the infinitesimal rotation
    makes the source variation vanish:
    \[
      \int d^4x\,J_a(x)(T^A)_{ab}
        \frac{\delta W}{\delta J_b(x)}=0.
    \]
    With \(J_a=-\delta\Gamma/\delta\phi_a\), this becomes
    \[
      \boxed{\int d^4x\,\Gamma_{,a}(x)
        (T^A)_{ab}\phi_b(x)=0.}
    \]
    One field derivative gives
    \[
      \Gamma_{,ca}(y,x)(T^A)_{ab}\phi_b(x)
        +\Gamma_{,a}(y)(T^A)_{ac}=0,
    \]
    with integration over \(x\) understood in the first term.  A second
    derivative gives the corresponding sum of a three-point term contracted
    with \(T^A\phi\) and two two-point terms with \(T^A\) acting on an
    external index.

    At the symmetric background \(\phi=0\), the one-derivative identity says
    that \(\Gamma^{(2)}\) commutes with every generator.  The
    two-derivative identity similarly says that \(\Gamma^{(3)}\) is an
    invariant rank-three tensor.  More generally every 1PI vertex is an
    \(O(N)\)-invariant tensor.
  \end{enumerate}

  \SupplementarySolution{5}{Loop Counting, Supertraces, and the Derivative Expansion}
  \begin{enumerate}
    \item[(a)]
    A quadratic term in \(g^{-1}\Gamma\) has inverse proportional to \(g\),
    so every internal line supplies \(g\).  Every interaction vertex comes
    from \(g^{-1}\Gamma\) and supplies \(g^{-1}\).  A graph with \(I\)
    internal lines and \(V\) vertices therefore carries \(g^{I-V}\).
    For a connected graph,
    \[
      L=I-V+1,
    \]
    hence its weight is \(g^{L-1}\).  It follows that
    \[
      W_\Gamma[J,g]
        =\sum_{L\ge0}g^{L-1}W_\Gamma^{(L)}[J].
    \]
    The leading \(g^{-1}\) term as \(g\to0\) is the stationary-phase
    contribution.  Gaussian fluctuations start at \(g^0\), so the leading
    term contains no loops.  It is exactly the sum of connected trees made
    from the full vertices in \(\Gamma\) and their inverse quadratic kernel.
    \item[(b)]
    Translation invariance permits the expansion
    \[
      \Gamma[\phi]
        =\sum_{n=0}^{\infty}\frac1{n!}
         \int\prod_{a=1}^n d^4x_a\,
         \Gamma^{(n)}(x_1,\ldots,x_n)
         \phi(x_1)\cdots\phi(x_n).
    \]
    In momentum space each kernel contains
    \((2\pi)^4\delta^4(p_1+\cdots+p_n)\).  A constant field has Fourier
    transform
    \((2\pi)^4\phi_0\delta^4(p)\), so every external momentum is zero and the
    remaining momentum-conservation delta function becomes
    \[
      (2\pi)^4\delta^4(0)=\mathcal V_4.
    \]
    Therefore
    \[
      \Gamma[\phi_0]
        =\mathcal V_4\sum_{n=0}^{\infty}
          \frac{\Gamma^{(n)}(0,\ldots,0)}{n!}\phi_0^n
        =-\mathcal V_4V(\phi_0),
    \]
    or
    \[
      V(\phi_0)
        =-\sum_{n=0}^{\infty}
          \frac{\Gamma^{(n)}(0,\ldots,0)}{n!}\phi_0^n.
    \]
    Thus a constant-background calculation probes only zero-momentum 1PI
    vertices; momentum dependence belongs to derivative terms in
    \(\Gamma\).
    \item[(c)]
    Diagonalizing the quadratic fluctuation matrices reduces the Gaussian
    functional integral to a product of one-species determinants.  In the
    chapter's state-counting convention,
    \[
      V_1^{\rm nonpol}(\phi)
        =\frac1{64\pi^2}
          \operatorname{STr}\left[
            \mathcal M^4(\phi)
            \ln\frac{\mathcal M^2(\phi)}{\kappa^2}\right].
    \]
    A real scalar has weight \(+1\), a complex scalar has weight \(+2\),
    and a complex Dirac field has weight \(-2\): the minus sign is
    Grassmannian and the factor two counts its spin states as in
    Eq.~\eqref{eq:16.2.16}.  Internal degeneracies multiply these weights.
    Other common determinant conventions retain four Dirac components and
    redistribute a factor two; one must not mix those conventions within a
    calculation.

    The displayed logarithms are the universal nonpolynomial part at this
    order.  Local polynomials in the backgrounds, including constants
    accompanying the logarithms, depend on the regulator and subtraction
    prescription and are absorbed into renormalized couplings.
    \item[(d)]
    Choose
    \[
      \Phi=\frac1{\sqrt2}(\varphi+h+iG).
    \]
    Then
    \[
      V_0=\frac{m^2}{2}\bigl[(\varphi+h)^2+G^2\bigr]
         +\frac{\lambda}{16}
          \bigl[(\varphi+h)^2+G^2\bigr]^2.
    \]
    The quadratic eigenvalues are
    \[
      m_h^2=m^2+\frac{3\lambda}{4}\varphi^2,
      \qquad
      m_G^2=m^2+\frac{\lambda}{4}\varphi^2.
    \]
    Therefore
    \[
      V_1^{\rm nonpol}
        =\frac1{64\pi^2}\left[
          m_h^4\ln\frac{m_h^2}{\kappa^2}
          +m_G^4\ln\frac{m_G^2}{\kappa^2}\right].
    \]
    A general constant background can be rotated to the real axis.  Since
    \(\varphi^2=2\Phi^*\Phi\), both eigenvalues and hence the determinant
    depend only on the invariant \(\Phi^*\Phi\), as required by the global
    \(U(1)\) symmetry.
    \item[(e)]
    Lorentz invariance and integration by parts leave, through two
    derivatives,
    \[
      \Gamma[\phi]
        =\int d^4x\left[-V(\phi)
          -\frac12Z(\phi)(\partial\phi)^2\right]+O(\partial^4).
    \]
    Expanding about a constant background \(\phi_0\), the two-point vertex
    has the small-momentum form
    \[
      \Gamma^{(2)}(p,-p;\phi_0)
        =-V''(\phi_0)+Z(\phi_0)p^2+O(p^4),
    \]
    with overall signs adjusted by the metric convention.  Thus the
    derivative with respect to \(p^2\) at \(p=0\) determines \(Z(\phi_0)\);
    a strictly constant background probes only \(V\).

    Under a local redefinition \(\phi=f(\chi)\),
    \[
      \widetilde V(\chi)=V(f(\chi)),\qquad
      \widetilde Z(\chi)=Z(f(\chi))[f'(\chi)]^2.
    \]
    Hence \(Z\) is parametrization dependent, though observables obtained
    from the complete effective action are unchanged.
  \end{enumerate}

  \SupplementarySolution{6}{Scalar Determinants and Counterterms}
  \begin{enumerate}
    \item[(a)]
    Write the shifted field as \(\phi_0+\eta\).  Through quadratic order,
    \[
      I[\phi_0+\eta]
        =I[\phi_0]-\frac12\eta K\eta+O(\eta^3).
    \]
    The one-loop Gaussian is
    \[
      \int\mathcal D\eta\,
        e^{-i\eta K\eta/2}\propto(\det K)^{-1/2},
    \]
    so
    \[
      i\Gamma_1=-\frac12\operatorname{Tr}\ln K,
      \qquad
      \Gamma_1=\frac{i}{2}\operatorname{Tr}\ln K
    \]
    up to a background-independent normalization.  After Wick rotation the
    potential contribution is
    \[
      \mathcal I_4(x)
        =\frac12\int\frac{d^4k_E}{(2\pi)^4}\ln(k_E^2+x).
    \]
    Three derivatives give
    \[
      \mathcal I_4'''(x)
        =\int\frac{d^4k_E}{(2\pi)^4}\frac1{(k_E^2+x)^3}
        =\frac1{32\pi^2x}.
    \]
    Hence
    \[
      \mathcal I_4(x)
        =\frac{x^2}{64\pi^2}\ln\frac{x}{\kappa^2}
         +A+Bx+Cx^2.
    \]
    The last three terms are local and are fixed by renormalization
    conditions.
    \item[(b)]
    The Hessian of the classical potential is
    \[
      \frac{\partial^2V_0}{\partial\phi_a\partial\phi_b}
        =\left(m^2+\frac{\lambda}{6}\boldsymbol\phi^2\right)\delta_{ab}
         +\frac{\lambda}{3}\phi_a\phi_b.
    \]
    At \(\boldsymbol\phi=(\varphi,0,\ldots,0)\), the radial eigenvalue and
    the \(N-1\) transverse eigenvalues are
    \[
      m_R^2(\varphi)=m^2+\frac{\lambda}{2}\varphi^2,
      \qquad
      m_G^2(\varphi)=m^2+\frac{\lambda}{6}\varphi^2.
    \]
    Thus
    \[
      V_1^{\rm nonpol}(\varphi)
        =\frac1{64\pi^2}\left[
          m_R^4\ln\frac{m_R^2}{\kappa^2}
          +(N-1)m_G^4\ln\frac{m_G^2}{\kappa^2}\right].
    \]
    Polynomial terms are understood to be fixed by the chosen
    renormalization conditions.  At an exact broken minimum, the transverse
    zero modes become the Goldstone modes.
    \item[(c)]
    At large momentum, expand
    \[
      \ln(k^2+x)
        =\ln k^2+\sum_{r\ge1}\frac{(-1)^{r+1}}r
          \left(\frac{x}{k^2}\right)^r.
    \]
    In \(D=2n\), the terms through \(r=n\) are ultraviolet divergent.
    Thus the local divergent part is a polynomial of degree \(n\) in
    \(x=\mu^2\), while the first nonpolynomial term is
    \(x^n\ln x\).

    In six dimensions \(n=3\).  With \(x=a+b\phi\), the polynomial
    \(A+Bx+Cx^2+Dx^3\) becomes a general polynomial through \(\phi^3\).
    Its four coefficients renormalize, respectively, the vacuum energy,
    tadpole, mass, and cubic coupling.  No \(\phi^4\) divergence is generated
    by this one-loop determinant, in accord with six-dimensional power
    counting for the renormalizable \(\phi^3\) theory.
  \end{enumerate}

  \SupplementarySolution{7}{Fermionic Determinants and Graded Legendre Transforms}
  \begin{enumerate}
    \item[(a)]
    For Grassmann vectors,
    \[
      \int\prod_i d\bar\theta_i\,d\theta_i\,
        e^{-\bar\theta_iD_{ij}\theta_j}=\det D.
    \]
    This follows by expanding the exponential: only the term containing
    every \(\theta_i\) and \(\bar\theta_i\) survives, and its antisymmetrized
    coefficient is the determinant.  A real commuting Gaussian instead
    gives \((\det K)^{-1/2}\).  Taking logarithms therefore produces an
    opposite sign for a closed fermion loop.

    For a constant mass,
    \[
      \operatorname{tr}_{\rm spin}\ln(\slashed p-M)
        =\frac12\operatorname{tr}_{\rm spin}
          \ln[(\slashed p-M)(-\slashed p-M)].
    \]
    The product is proportional to \(p^2+M^2\) after Wick rotation.  The
    trace counts the two physical spin states in the convention of this
    chapter, so the fermion logarithm is \(-2\) times a real-scalar
    logarithm:
    \[
      V_{1,f}^{\rm nonpol}
        =-\frac{M^4}{32\pi^2}\ln\frac{M^2}{\kappa^2}.
    \]
    \item[(b)]
    The constant background replaces the fermion mass by
    \(M(\phi)=m+y\phi\).  In the normalization of the chapter,
    \[
      V_{1,f}^{\rm nonpol}(\phi)
        =-\frac{[m+y\phi]^4}{32\pi^2}
          \ln\frac{[m+y\phi]^2}{\kappa^2}.
    \]
    The divergent local part is a polynomial of degree four in \(M\), and
    hence of degree four in \(\phi\).  Without a reflection symmetry the
    scalar sector needs counterterms proportional to
    \[
      1,\quad\phi,\quad\phi^2,\quad\phi^3,\quad\phi^4.
    \]
    If a symmetry enforces \(M(-\phi)=-M(\phi)\), then \(m=0\) in this
    parametrization and the determinant is even.  Only
    \(1,\phi^2,\phi^4\) remain.  The same symmetry must also be respected by
    the regulator for this reduction to be valid.
    \item[(c)]
    With every source placed to the right of its field, variation of \(W\)
    must be written
    \[
      \delta W=\delta J_n\,
        \frac{\delta_R W}{\delta J_n}
    \]
    after the appropriate graded reordering.  Thus
    \[
      \frac{\delta_R W}{\delta J_n}=\chi^n.
    \]
    Varying
    \(\Gamma=W-\chi^nJ_n\) cancels the \(W_{,J}\delta J\) term and gives
    \[
      \frac{\delta_L\Gamma}{\delta\chi^n}=-J_n.
    \]

    Arrange one bosonic and one fermionic variable into a graded column
    \(z=(b,f)\), and define the Hessians with the derivative orientations
    just displayed.  Then their four blocks obey the supermatrix equation
    \[
      \begin{pmatrix}
        \Gamma_{bb}&\Gamma_{bf}\\
        \Gamma_{fb}&\Gamma_{ff}
      \end{pmatrix}
      \begin{pmatrix}
        W_{bb}&W_{bf}\\
        W_{fb}&W_{ff}
      \end{pmatrix}
      =-\begin{pmatrix}1&0\\0&1\end{pmatrix}.
    \]
    Explicitly these are
    \[
      \Gamma_{bb}W_{bb}+\Gamma_{bf}W_{fb}=-1,\quad
      \Gamma_{bb}W_{bf}+\Gamma_{bf}W_{ff}=0,
    \]
    together with the analogous lower-row equations.  The Koszul signs are
    contained in the definitions of the mixed left/right derivative blocks.
    Interchanging their order uses graded transposition,
    \((A^{\mathrm{sT}})_{ij}=(-1)^{\epsilon_i\epsilon_j}A_{ji}\),
    rather than ordinary transposition.
  \end{enumerate}

  \SupplementarySolution{8}{Particle-Number Breaking and the Phase-Only Theory}
  \begin{enumerate}
    \item[(a)]
    The Noether charge is the particle number,
    \[
      N=\int d^dx\,\Psi^\dagger\Psi=\int d^dx\,\rho.
    \]
    Substitution of \(\Psi=\sqrt\rho\,e^{-i\theta}\) gives
    \[
      i\Psi^\dagger\partial_t\Psi
        =\rho\,\partial_t\theta+\frac i2\partial_t\rho,
      \qquad
      \partial_i\Psi^\dagger\partial_i\Psi
        =\frac{(\boldsymbol\nabla\rho)^2}{4\rho}
          +\rho(\boldsymbol\nabla\theta)^2.
    \]
    Dropping the total derivative \(i\partial_t\rho/2\),
    \[
      \mathcal L
        =\rho\,\partial_t\theta
         -\frac{(\boldsymbol\nabla\rho)^2}{8m\rho}
         -\frac{\rho}{2m}(\boldsymbol\nabla\theta)^2
         -r\rho-\lambda\rho^2+\cdots .
    \]
    Thus \(\rho=\Psi^\dagger\Psi\) is the particle-number density and
    \(\pi_\theta=\rho\).  Canonical quantization gives
    \([\theta(\mathbf x),\rho(\mathbf y)]=i\delta^{(d)}(\mathbf x-\mathbf y)\).
    In a volume \(\mathcal V_d\),
    \[
      [\theta_0,N]=i\mathcal V_d,
      \qquad
      \Delta\theta_0\,\Delta N\ge\frac{\mathcal V_d}{2}.
    \]
    Equivalently the volume-averaged phase
    \(\bar\theta=\theta_0/\mathcal V_d\) satisfies
    \(\Delta\bar\theta\,\Delta N\ge1/2\).  A number eigenstate has no sharp
    global phase.  A symmetry-breaking state with a sharp phase must instead
    superpose number sectors; in the thermodynamic limit its macroscopic
    number fluctuations can coexist with a sharply defined average phase.
    \item[(b)]
    Stability requires \(\lambda>0\).  Since the homogeneous potential is
    \(V(\rho)=r\rho+\lambda\rho^2\) for \(\rho\ge0\), a nonzero minimum
    exists when \(r<0\), at
    \[
      \bar\rho=-\frac r{2\lambda}.
    \]
    With \(\rho=\bar\rho+\delta\rho\), the condition
    \(r+2\lambda\bar\rho=0\) cancels the linear potential term.  To quadratic
    order,
    \[
      \mathcal L_2
        =\bar\rho\,\partial_t\theta
         +\delta\rho\,\partial_t\theta
         -\lambda(\delta\rho)^2
         -\frac{(\boldsymbol\nabla\delta\rho)^2}{8m\bar\rho}
         -\frac{\bar\rho}{2m}(\boldsymbol\nabla\theta)^2.
    \]
    The first term is a total derivative in the topologically trivial sector.
    The density fluctuation is gapped, whereas the phase has no mass term.
    \item[(c)]
    Ignoring the displayed total derivative, the quadratic functional
    integral is
    \[
      Z_2=\int\mathcal D\delta\rho\,\mathcal D\theta\,
        \exp\!\left(i\int dt\,d^dx\,\mathcal L_2\right).
    \]
    After integrating the density-gradient term by parts, define
    \[
      A=2\lambda-\frac{\nabla^2}{4m\bar\rho}.
    \]
    Completing the square in
    \(-\delta\rho A\delta\rho/2+\delta\rho\,\partial_t\theta\) and carrying
    out the Gaussian integral gives
    \[
      \mathcal L_{\rm eff}
        =\frac12(\partial_t\theta)A^{-1}(\partial_t\theta)
         -\frac{\bar\rho}{2m}(\boldsymbol\nabla\theta)^2.
    \]
    At wavelengths much longer than the density correlation length,
    \[
      \mathcal L_{\rm eff}
        =\frac1{4\lambda}(\partial_t\theta)^2
         -\frac{\bar\rho}{2m}(\boldsymbol\nabla\theta)^2
         +O(\partial^4)
        =\frac\chi2\left[(\partial_t\theta)^2
          -v^2(\boldsymbol\nabla\theta)^2\right]+O(\partial^4),
    \]
    where
    \[
      \chi=\frac1{2\lambda},
      \qquad
      v^2=\frac{2\lambda\bar\rho}{m}=-\frac rm.
    \]
    Hence \(\omega=v|\mathbf k|\), the Goldstone sound mode.
    \item[(d)]
    The Mittag--Leffler expansion
    \[
      \pi\coth(\pi a)=\frac1a
        +2a\sum_{n=1}^{\infty}\frac1{a^2+n^2}
    \]
    follows, for example, by summing the residues of
    \(\coth(\pi z)/(z^2-a^2)\).  Division by \(a\) and inclusion of positive
    and negative integers proves the required identity.  With
    \(E_k=v k\), it implies
    \[
      T\sum_{\omega_n}\frac1{\omega_n^2+E_k^2}
        =\frac1{2E_k}\coth\!\left(\frac{E_k}{2T}\right).
    \]
    Therefore, with a microscopic ultraviolet cutoff \(\Lambda\),
    \[
      \langle\theta^2\rangle
        =\frac{\operatorname{Area}(S^{d-1})}
          {2\chi v(2\pi)^d}
          \int_0^\Lambda dk\,k^{d-2}
            \coth\!\left(\frac{vk}{2T}\right).
    \]
    At small \(k\), the thermal integrand behaves as \(k^{d-3}\).  It is
    infrared finite only for \(d>2\).  Thus among \(d=1,2,3\), true
    finite-temperature long-range order is possible only in \(d=3\).
    In \(d=2\) the logarithmic divergence forbids a nonzero order parameter
    even though a Berezinskii--Kosterlitz--Thouless phase with algebraic order
    may occur; the \(d=1\) divergence is stronger.  The raw continuum
    integral is ultraviolet divergent in every case, but this is unrelated
    to the infrared obstruction and is cut off by microscopic physics.
    \item[(e)]
    Subtracting the zero-temperature contribution replaces the coth by its
    thermal part:
    \[
      \delta_T\langle\theta^2\rangle
        =\frac1{\chi v}\int\frac{d^3k}{(2\pi)^3}\,
          \frac1{k(e^{vk/T}-1)}
        =\frac{T^2}{12\chi v^3}.
    \]
    Since Gaussian phase fluctuations reduce the order parameter as
    \(\langle\Psi\rangle\sim\sqrt{\bar\rho}\,
    e^{-\langle\theta^2\rangle/2}\), a phase-only estimate of restoration is
    obtained when the thermal variance becomes order unity:
    \[
      T_{\rm restore}\sim\sqrt{12\chi v^3}
        =\sqrt{\frac6\lambda}\left(-\frac rm\right)^{3/4}.
    \]
    The numerical threshold is not universal: near restoration, vortices,
    amplitude fluctuations, and the ultraviolet completion of the
    Landau--Ginzburg theory can no longer be neglected.
  \end{enumerate}

  \SupplementarySolution{9}{Dimensional Transmutation and RG Invariance}
  \begin{enumerate}
    \item[(a)]
    For \(\phi\ne0\),
    \[
      V'(\phi)
        =\phi^3\left[
          4A+B\left(4\ln\frac{\phi^2}{\kappa^2}+2\right)\right].
    \]
    A nonzero stationary point \(v\) therefore obeys
    \[
      \ln\frac{v^2}{\kappa^2}=-\frac{A}{B}-\frac12.
    \]
    Both signs \(\phi=\pm v\) occur when the field has a reflection
    symmetry.  At either point,
    \[
      V''(v)=8Bv^2>0,
    \]
    so it is a local minimum for \(B>0\).  Eliminating \(A\) gives
    \[
      V(\phi)
        =B\phi^4\left(\ln\frac{\phi^2}{v^2}-\frac12\right),
    \]
    up to an additive vacuum constant.  The dimensionless combination
    \(A/B\) has been replaced by the dimensionful scale
    \(v=\kappa\exp[-A/(2B)-1/4]\).  This is dimensional transmutation.
    \item[(b)]
    Holding bare fields and couplings fixed while differentiating the
    renormalized potential gives
    \[
      \left(\kappa\partial_\kappa
        +\sum_i\beta_i\partial_{g_i}
        -\gamma\phi\partial_\phi\right)V=0.
    \]
    The last term appears because the renormalized field changes with
    \(\kappa\).  Insert
    \[
      V=\frac{\lambda}{24}\phi^4
        +B\phi^4\ln\frac{\phi^2}{\kappa^2}.
    \]
    Through one loop, derivatives of \(B\) and of the logarithmic term with
    respect to running couplings are of higher order.  The three surviving
    contributions are
    \[
      -2B\phi^4+\frac{\beta_\lambda}{24}\phi^4
        -\frac{\gamma\lambda}{6}\phi^4=0.
    \]
    Hence
    \[
      \boxed{
        B=\frac{\beta_\lambda-4\gamma\lambda}{48}}
    \]
    at one loop.  This relation ensures that the explicit scale dependence
    of the logarithm is canceled by the implicit running of the coupling and
    field normalization.
  \end{enumerate}

  \SupplementarySolution{10}{Large-\(N\) Sigma Model at Finite Temperature}
  \begin{enumerate}
    \item[(a)]
    Integrating by parts puts the quadratic form for each component in terms
    of
    \[
      K[\lambda]=-\partial_\tau^2-v^2\nabla^2+i\lambda.
    \]
    The \(N\) Gaussian integrals give \((\det K)^{-N/2}\).  Up to a
    \(\lambda\)-independent normalization,
    \[
      \boxed{
      \widetilde S[\lambda]
        =\frac N2\operatorname{Tr}\ln K[\lambda]
         -\frac{iN}{2vg}\int d^2x\int_0^{1/T}d\tau\,\lambda.}
    \]
    A constant such as \(\operatorname{Tr}\ln(vg)\) may be included inside
    the logarithm without affecting any saddle equation.
    \item[(b)]
    Both terms in \(\widetilde S\) are proportional to \(N\), so fluctuations
    of \(\lambda\) about a stationary configuration are suppressed at large
    \(N\).  For \(i\lambda=m^2\), variation of the trace logarithm gives
    \[
      \boxed{
      \frac1{vg}
        =T\sum_{n=-\infty}^{\infty}
          \int\frac{d^2k}{(2\pi)^2}\,
          \frac1{\omega_n^2+v^2\mathbf k^2+m^2},
      \qquad \omega_n=2\pi nT.}
    \]
    This is also the constraint equation enforcing the prescribed length of
    \(\mathbf n\).
    \item[(c)]
    At zero temperature, the sum becomes a frequency integral.  With a sharp
    spatial momentum cutoff,
    \[
      \begin{split}
      \frac1{vg}
        &=\int_{|\mathbf k|<\Lambda}\frac{d^2k}{(2\pi)^2}
          \int_{-\infty}^{\infty}\frac{d\omega}{2\pi}\,
          \frac1{\omega^2+v^2\mathbf k^2+m^2}\\
        &=\frac{\sqrt{v^2\Lambda^2+m^2}-m}{4\pi v^2}.
      \end{split}
    \]
    Define the critical coupling by setting \(m=0\):
    \[
      \frac1{g_c}=\frac\Lambda{4\pi}.
    \]
    Taking \(\Lambda\to\infty\) in the difference of the two equations gives
    \[
      \frac1g=\frac1{g_c}-\frac{m}{4\pi v},
      \qquad
      \boxed{m=\frac{4\pi v(g-g_c)}{gg_c}.}
    \]
    A positive symmetric saddle therefore exists only for \(g>g_c\); this is
    the quantum-disordered phase with a finite gap.  For \(g<g_c\), the
    constant massive saddle is replaced by a condensate of \(\mathbf n\),
    which breaks \(O(N)\) to \(O(N-1)\) and produces Goldstone modes.
    \item[(d)]
    Let \(E_k=\sqrt{v^2\mathbf k^2+m(T)^2}\).  Deforming the two pieces of
    \(\mathcal C\) around the poles at \(z=\pm E_k\) gives
    \[
      T\sum_n\frac1{\omega_n^2+E_k^2}
        =\frac1{2E_k}\coth\!\left(\frac{E_k}{2T}\right)
        =\frac{1+2n_B(E_k)}{2E_k},
    \]
    where \(n_B(E)=1/(e^{E/T}-1)\).  The finite-temperature saddle equation
    is consequently
    \[
      \frac1{vg}
        =\int\frac{d^2k}{(2\pi)^2}\,
          \frac1{2E_k}\coth\!\left(\frac{E_k}{2T}\right).
    \]
    \item[(e)]
    Using \(k\,dk=E\,dE/v^2\), the cutoff equation is
    \[
      \frac1{vg}
        =\frac1{4\pi v^2}
          \int_{m(T)}^{\sqrt{v^2\Lambda^2+m(T)^2}}
            dE\,\coth\!\left(\frac E{2T}\right)
        =\frac{T}{2\pi v^2}
          \left[\ln\sinh\!\left(\frac E{2T}\right)\right]_{m(T)}^{E_\Lambda}.
    \]
    The zero-temperature critical equation, evaluated with the supplied
    frequency integral, is
    \[
      \frac1{vg_c}=\frac\Lambda{4\pi v}.
    \]
    For large \(\Lambda\),
    \(\ln\sinh(E_\Lambda/2T)=E_\Lambda/(2T)-\ln2+o(1)\).
    Subtracting the critical equation and using
    \(\Delta=4\pi v(1/g_c-1/g)\) leaves
    \[
      \frac{\Delta}{2T}
        =\ln\!\left[2\sinh\!\left(\frac{m(T)}{2T}\right)\right].
    \]
    Therefore
    \[
      \boxed{m(T)=2T\sinh^{-1}\!\left(\frac12e^{\Delta/(2T)}\right).}
    \]
    \item[(f)]
    At low temperature the argument of the inverse hyperbolic sine is large,
    and
    \[
      m(T)=\Delta+2T e^{-\Delta/T}
        +O\!\left(Te^{-2\Delta/T}\right).
    \]
    The zero-temperature quantum-disordered gap is therefore changed only by
    thermally activated corrections.  At high temperature,
    \[
      m(T)=2T\sinh^{-1}\!\left(\frac12\right)
        +\frac{\Delta}{\sqrt5}
        +O\!\left(\frac{\Delta^2}{T}\right).
    \]
    In particular, at the quantum-critical coupling \(\Delta=0\),
    \(m(T)=\Theta T\), where
    \(\Theta=2\ln[(1+\sqrt5)/2]\).  Thus temperature itself sets the inverse
    correlation length, \(\xi^{-1}=m/v\), in the quantum-critical regime.
    \item[(g)]
    Analytic continuation \(i\omega_n\to\omega+i0\) gives, with
    \(E_k^2=v^2\mathbf k^2+m(T)^2\),
    \[
      G^R(\omega,\mathbf k)
        =\frac{vg}{N}\frac1{E_k^2-(\omega+i0)^2}.
    \]
    Its imaginary part is
    \[
      \boxed{
      \operatorname{Im}G^R(\omega,\mathbf k)
        =\frac{\pi vg}{2NE_k}
          \left[\delta(\omega-E_k)-\delta(\omega+E_k)\right].}
    \]
    Finally write
    \[
      m(T)=T f(\Delta/T),
      \qquad
      f(x)=2\sinh^{-1}\!\left(\frac12e^{x/2}\right).
    \]
    Apart from the nonuniversal normalization \(vg/N\),
    \[
      G^R(\omega,\mathbf k)
        =\frac{vg}{N T^2}\,
          \frac1{(v|\mathbf k|/T)^2
            -[(\omega+i0)/T]^2+f(\Delta/T)^2}.
    \]
    It depends only on the scaling variables
    \(\omega/T\), \(v|\mathbf k|/T\), and \(\Delta/T\), which is the
    expected \(z=1\) quantum-critical form.
  \end{enumerate}

  \SupplementarySolution{11}{Generating Functionals, Legendre Geometry, and Split Independence}
  \begin{enumerate}
    \item[(a)] In Euclidean conventions one may take
      \[
        Z[J]=\int\mathcal D\phi\,
        e^{-[S[\phi]-\int J\phi]/\hbar}.
      \]
      For example,
      \[
        \langle\phi(x)\phi(y)\rangle_J
        =\frac{\hbar^2}{Z[J]}
          \frac{\delta^2Z[J]}{\delta J(x)\delta J(y)} .
      \]
      Factors of \(i\) move in the usual way in Lorentzian signature.
    \item[(b)] Define
      \(W[J]=\hbar\log Z[J]\),
      \(\Phi=\delta W/\delta J\), and
      \[
        \Gamma[\Phi]=\int J\Phi-W[J],\qquad
        \frac{\delta\Gamma}{\delta\Phi}=J .
      \]
      The classical action is the tree-level input; \(\Gamma\) includes
      all quantum fluctuations and its derivatives are proper vertices.
      A Wilsonian action instead results from integrating out only modes
      above a chosen scale and therefore retains low modes as variables.
    \item[(c)] Every vacuum graph in \(Z\) is a disjoint union of connected
      components.  The exponential formula for such unions gives
      \(Z=\exp(W/\hbar)\), so the logarithm selects connected graphs.
      Connected graphs themselves are trees of maximal 1PI subgraphs joined
      by exact propagators.  The Legendre transform removes precisely those
      tree connections: differentiating \(\Gamma\) produces the 1PI
      vertices, while reconnecting them with
      \(G=(\Gamma^{(2)})^{-1}\) reconstructs connected correlators.
    \item[(d)] Change variables from \(\eta\) to
      \(\phi=\phi_0+\eta\).  Then
      \[
        W(J;\phi_0)=W(J;0)-J\phi_0 .
      \]
      Consequently
      \[
        \chi=\partial_JW(J;\phi_0)
          =\partial_JW(J;0)-\phi_0 .
      \]
      Thus the same saddle source \(J_\chi\) is the source associated in
      the unshifted description with \(\Phi=\phi_0+\chi\).  The Legendre
      transform now gives
      \[
        \Gamma(\chi;\phi_0)
        =W(J_\chi;0)-J_\chi(\phi_0+\chi)
        =\Gamma(\phi_0+\chi;0),
      \]
      an exact change-of-variables identity with no loop expansion.
  \end{enumerate}

\end{enumerate}
