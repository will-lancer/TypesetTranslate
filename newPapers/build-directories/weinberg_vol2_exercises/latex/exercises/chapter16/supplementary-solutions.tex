\chapterbackmatter{Solutions to Supplementary Exercises}

\begin{enumerate}
  \SupplementarySolution{1}{Connected Graphs and the Quantum Action}
  \begin{enumerate}
    \item[(a)]
    Let \(C_\alpha[J]\) denote the value of a connected vacuum graph of type
    \(\alpha\), including its usual symmetry factor.  A vacuum graph with
    \(n_\alpha\) identical connected components of each type has value
    \[
      \prod_\alpha\frac{C_\alpha[J]^{n_\alpha}}{n_\alpha!}.
    \]
    The factorial divides out permutations of identical components.  Summing
    independently over every \(n_\alpha\) gives
    \[
      Z[J]
        =\prod_\alpha\sum_{n_\alpha=0}^{\infty}
          \frac{C_\alpha[J]^{n_\alpha}}{n_\alpha!}
        =\exp\left(\sum_\alpha C_\alpha[J]\right).
    \]
    Defining \(\sum_\alpha C_\alpha=iW[J]\) gives \(Z=e^{iW}\).

    A derivative of \(Z\) may act on different factors in its exponential,
    so it includes products of disconnected correlators.  In contrast,
    derivatives of the exponent select insertions within the same connected
    component.  Consequently \(W_{,i_1\cdots i_n}\) generates connected
    \(n\)-point functions, with the conventional factors of \(i\) restored
    according to the source convention.
    \item[(b)]
    Differentiate
    \(\Gamma_{ia}\Delta_{aj}=-\delta_{ij}\), where
    \(\Delta_{ij}=W_{,ij}\), with respect to \(J_k\):
    \[
      \Gamma_{iab}\Delta_{bk}\Delta_{aj}
        +\Gamma_{ia}W_{,ajk}=0.
    \]
    Multiplication by \(\Delta_{\ell i}\), together with
    \(\Delta_{\ell i}\Gamma_{ia}=-\delta_{\ell a}\), gives
    \[
      \boxed{W_{,\ell jk}
        =\Delta_{\ell i}\Delta_{ja}\Delta_{kb}\Gamma_{iab}}.
    \]
    Thus the connected three-point function is a single full 1PI cubic
    vertex with one full propagator attached to each external point.  The
    displayed plus sign would look different if the minus sign were moved
    out of the definition \(\Delta=-\Gamma^{(2)-1}\).
  \end{enumerate}

  \SupplementarySolution{2}{Convexity and Linear Symmetries of the Effective Action}
  \begin{enumerate}
    \item[(a)]
    In Euclidean signature, for every test function \(f_i\),
    \[
      f_iW_{,ij}f_j
        =\left\langle (f_i\chi^i-\langle f_i\chi^i\rangle)^2
          \right\rangle\ge0.
    \]
    After zero modes are regulated, \(\Delta=W^{(2)}\) is positive and
    \(-\Gamma^{(2)}=\Delta^{-1}\) is positive in the convention of this
    chapter.  For a constant scalar,
    \(\Gamma=-\mathcal V_4V\), so the zero-momentum kernel obeys
    \[
      V''(\phi)\ge0.
    \]
    With several scalars, the Hessian of \(V\) is positive semidefinite.

    A zero eigenvalue corresponds to a flat direction or to a limiting
    susceptibility and is handled by a regulated inverse or a
    Legendre--Fenchel transform.  A loop expansion about an unstable
    homogeneous saddle does not perform the global minimization implicit in
    the exact transform, so any finite truncation may remain nonconvex.
    \item[(b)]
    The source pairing is preserved by
    \[
      \phi\mapsto M\phi,\qquad J\mapsto M^{-\mathsf T}J,
    \]
    since
    \((M\phi)^{\mathsf T}M^{-\mathsf T}J=\phi^{\mathsf T}J\).
    A change of integration variables therefore gives
    \[
      W[M^{-\mathsf T}J]=W[J].
    \]
    Differentiation and the chain rule yield
    \[
      \phi_{M^{-\mathsf T}J}=M\phi_J,
      \qquad
      J_{M\phi}=M^{-\mathsf T}J_\phi.
    \]
    Consequently,
    \[
      \begin{split}
      \Gamma[M\phi]
        &=-(M\phi)^{\mathsf T}M^{-\mathsf T}J_\phi
          +W[M^{-\mathsf T}J_\phi]\\
        &=-\phi^{\mathsf T}J_\phi+W[J_\phi]
         =\Gamma[\phi].
      \end{split}
    \]
    Orthogonality was never used; invertibility and invariance of action and
    measure suffice.
    \item[(c)]
    Changing variables \(\phi\mapsto-\phi\) gives
    \[
      Z[J]=Z[-J],\qquad W[J]=W[-J].
    \]
    Differentiation yields
    \(\phi_{-J}=-\phi_J\), and inversion gives
    \(J_{-\phi}=-J_\phi\).  The Legendre transform then satisfies
    \[
      \Gamma[-\phi]
        =-(-\phi)J_{-\phi}+W[J_{-\phi}]
        =-\phi J_\phi+W[J_\phi]
        =\Gamma[\phi].
    \]
    Taking an odd number of derivatives and setting \(\phi=0\) therefore
    gives
    \[
      \Gamma^{(2r+1)}(0)=0.
    \]
    If one instead writes \(\phi=v+\eta\) about a chosen nonzero minimum,
    the reflection exchanges the expansions about \(+v\) and \(-v\);
    it is not simply \(\eta\mapsto-\eta\) within one chosen phase.  Odd
    vertices in \(\eta\) need not vanish.
  \end{enumerate}

  \SupplementarySolution{3}{Legendre Geometry and Nonlinear Ward Identities}
  \begin{enumerate}
    \item[(a)]
    Varying the Legendre transform while using
    \(\phi^i=W_{,i}[J_\phi]\) cancels the terms proportional to
    \(\delta J_\phi\), leaving
    \[
      \Gamma_{,i}=-J_{\phi i}.
    \]
    Differentiate this equation with respect to \(J_k\):
    \[
      \Gamma_{,ij}\frac{\delta\phi^j}{\delta J_k}
        =-\delta_i{}^k.
    \]
    Since \(\delta\phi^j/\delta J_k=W_{,jk}\),
    \[
      \boxed{\Gamma_{,ij}W_{,jk}=-\delta_i{}^k},
      \qquad W^{(2)}=-\Gamma^{(2)-1}.
    \]
    Invertibility requires that zero modes have first been removed or
    regulated and that the source-to-field map be locally one-to-one.
    \(W^{(2)}\) is the full connected propagator.  \(\Gamma^{(2)}\) is its
    full 1PI inverse kernel, with the minus sign fixed by the chapter's
    Legendre-transform convention.
    \item[(b)]
    For \(m^2(\phi)=-a\) with \(a>0\), the prescribed continuation gives
    \[
      \ln[-a-i0]=\ln a-i\pi.
    \]
    Hence
    \[
      \operatorname{Im}V_1(\phi)
        =-\frac{a^2}{64\pi}
    \]
    for the displayed \(-i0\) prescription.  Reversing the prescription
    reverses the sign.  The magnitude is associated with modes whose
    quadratic fluctuation operator has negative eigenvalues.

    Such a homogeneous configuration is not a stable saddle about which an
    ordinary vacuum loop expansion is valid.  The exact effective potential
    is defined by a constrained minimum energy and is real and convex.  The
    imaginary part therefore diagnoses decay or spinodal instability of the
    perturbative branch; it is not an imaginary exact ground-state energy.
    \item[(c)]
    During the long interval in which the source is constant, evolution of
    the adiabatically prepared lowest-energy state supplies the phase
    \(e^{-iE[J]T}\).  Switching regions contribute only terms that do not
    grow with \(T\).  Comparing with \(Z=e^{iW}\) gives
    \[
      W[J]=-E[J]T+O(T^0).
    \]
    Since \(\phi=\delta W/\delta J\), the Legendre transform is
    \[
      -\frac{\Gamma[\phi]}T
        =E[J_\phi]+\int d^3x\,J_{\phi n}(\mathbf x)
          \phi^n(\mathbf x)+O(T^{-1}).
    \]
    The right-hand side is the Lagrange-multiplier expression that minimizes
    the source-free energy over normalized states subject to
    \(\langle\Phi^n(\mathbf x)\rangle=\phi^n(\mathbf x)\).  Adiabatic
    switching selects the continuously connected lowest state; the
    \(T\to\infty\) limit removes switching effects.  The infinite-volume
    limit suppresses interfaces and interference between distinct phases.
    \item[(d)]
    Evaluate the once-differentiated Ward identity at a constant stationary
    point \(v\), where \(\Gamma_{,a}[v]=0\).  It reduces to
    \[
      \int d^4x\,\Gamma_{,ca}^{(2)}(y,x;v)
        (T^Av)_a=0.
    \]
    Translation invariance turns the integral into the zero-momentum
    component:
    \[
      \boxed{\Gamma_{ca}^{(2)}(p=0;v)(T^Av)_a=0.}
    \]
    If \(T^Av\ne0\), the exact inverse propagator has a zero eigenvector at
    zero momentum, so the spectrum contains a massless mode.  The number of
    independent modes is the dimension of the span of \(T^Av\), equivalently
    \(\dim G-\dim H\) when \(H\) is the unbroken subgroup, subject to the
    usual qualifications for spacetime or nonrelativistic symmetries.
    \item[(e)]
    Invariance under the change of variables gives, to first order,
    \[
      J_n\left\langle F^n[\widehat\chi]\right\rangle_J=0.
    \]
    Substituting
    \(J_{\chi n}=-\delta\Gamma/\delta\chi^n\) yields
    \[
      \frac{\delta\Gamma}{\delta\chi^n}
        \left\langle F^n[\widehat\chi]\right\rangle_{J_\chi}=0.
    \]
    Thus \(\Gamma\) is invariant under the generally nonlocal transformation
    \[
      \delta\chi^n
        =\epsilon\left\langle
          F^n[\widehat\chi]\right\rangle_{J_\chi}.
    \]
    For one field and \(F[\widehat\chi]=\widehat\chi^2\),
    \[
      \left\langle\widehat\chi^2(x)\right\rangle_J
        =\chi^2(x)+
         \left\langle
         [\widehat\chi(x)-\chi(x)]^2\right\rangle_J
        =\chi^2(x)+W_{,JJ}(x,x),
    \]
    up to the conventional factor of \(i\) relating the last derivative to
    the connected propagator.  This fluctuation term is precisely why a
    nonlinear symmetry of the microscopic action does not act on the mean
    field by simple substitution.
    \item[(f)]
    The action is invariant, while the source term and Jacobian change the
    integrand by
    \[
      i\epsilon\left[
        J_nF^n[\chi]+\mathcal A[\chi]\right].
    \]
    Since a change of integration variables cannot change \(Z\),
    \[
      J_n\left\langle F^n[\widehat\chi]\right\rangle_J
        +\left\langle\mathcal A[\widehat\chi]\right\rangle_J=0.
    \]
    At \(J=J_\chi\), this becomes
    \[
      \frac{\delta\Gamma}{\delta\chi^n}
        \left\langle F^n[\widehat\chi]\right\rangle_{J_\chi}
        =\left\langle\mathcal A[\widehat\chi]\right\rangle_{J_\chi}.
    \]
    The right-hand side obstructs invariance of the quantum action.  If it
    is the variation of an allowed local counterterm, redefining the
    renormalized action moves or removes it.  A local term that cannot be
    removed while preserving all required symmetries is an anomaly.
  \end{enumerate}

  \SupplementarySolution{4}{Four-Point Vertices, Even-Dimensional Logs, and Goldstone Modes}
  \begin{enumerate}
    \item[(a)]
    A further current derivative of the preceding result can act on the
    1PI vertex or on any one of its three propagators.  Using the
    three-point formula for the derivative of each propagator gives
    \[
      \begin{split}
      W_{,ijkl}={}&
        \Delta_{ia}\Delta_{jb}\Delta_{kc}\Delta_{ld}
        \bigl[
          \Gamma_{abcd}
          +\Gamma_{abe}\Delta_{ef}\Gamma_{fcd}\\
          &\hspace{35mm}
          +\Gamma_{ace}\Delta_{ef}\Gamma_{fbd}
          +\Gamma_{ade}\Delta_{ef}\Gamma_{fbc}
        \bigr].
      \end{split}
    \]
    The first term is the full 1PI four-vertex.  The remaining terms split
    the labels as
    \[
      (ij|kl),\qquad(ik|jl),\qquad(il|jk),
    \]
    and join the two sides with a full propagator.  They are the three
    exchange channels.  No other connected tree with four labeled leaves
    and vertices of valence at least three exists.
    \item[(b)]
    For \(D=2n\), \(n+1\) derivatives make the integral convergent:
    \[
      \mathcal I_{2n}^{(n+1)}(x)
        =\frac12(-1)^n n!
          \int\frac{d^{2n}k_E}{(2\pi)^{2n}}
          \frac1{(k_E^2+x)^{n+1}}.
    \]
    Dimensional continuation, or direct integration of this derivative,
    gives the nonpolynomial term
    \[
      \boxed{\mathcal I_{2n}^{\rm nonpol}(x)
        =\frac{(-1)^n}{2n!(4\pi)^n}
          x^n\ln\frac{x}{\kappa^2}.}
    \]
    The omitted integration polynomial has degree \(n\) in \(x\).
    For \(n=2\),
    \[
      \mathcal I_4^{\rm nonpol}
        =\frac{x^2}{64\pi^2}\ln\frac{x}{\kappa^2},
    \]
    while for \(n=3\),
    \[
      \mathcal I_6^{\rm nonpol}
        =-\frac{x^3}{12(4\pi)^3}\ln\frac{x}{\kappa^2},
    \]
    in agreement with the four- and six-dimensional calculations.
    \item[(c)]
    Invariance of the action and measure under the infinitesimal rotation
    makes the source variation vanish:
    \[
      \int d^4x\,J_a(x)(T^A)_{ab}
        \frac{\delta W}{\delta J_b(x)}=0.
    \]
    With \(J_a=-\delta\Gamma/\delta\phi_a\), this becomes
    \[
      \boxed{\int d^4x\,\Gamma_{,a}(x)
        (T^A)_{ab}\phi_b(x)=0.}
    \]
    One field derivative gives
    \[
      \Gamma_{,ca}(y,x)(T^A)_{ab}\phi_b(x)
        +\Gamma_{,a}(y)(T^A)_{ac}=0,
    \]
    with integration over \(x\) understood in the first term.  A second
    derivative gives the corresponding sum of a three-point term contracted
    with \(T^A\phi\) and two two-point terms with \(T^A\) acting on an
    external index.

    At the symmetric background \(\phi=0\), the one-derivative identity says
    that \(\Gamma^{(2)}\) commutes with every generator.  The
    two-derivative identity similarly says that \(\Gamma^{(3)}\) is an
    invariant rank-three tensor.  More generally every 1PI vertex is an
    \(O(N)\)-invariant tensor.
  \end{enumerate}

  \SupplementarySolution{5}{Loop Counting, Supertraces, and the Derivative Expansion}
  \begin{enumerate}
    \item[(a)]
    A quadratic term in \(g^{-1}\Gamma\) has inverse proportional to \(g\),
    so every internal line supplies \(g\).  Every interaction vertex comes
    from \(g^{-1}\Gamma\) and supplies \(g^{-1}\).  A graph with \(I\)
    internal lines and \(V\) vertices therefore carries \(g^{I-V}\).
    For a connected graph,
    \[
      L=I-V+1,
    \]
    hence its weight is \(g^{L-1}\).  It follows that
    \[
      W_\Gamma[J,g]
        =\sum_{L\ge0}g^{L-1}W_\Gamma^{(L)}[J].
    \]
    The leading \(g^{-1}\) term as \(g\to0\) is the stationary-phase
    contribution.  Gaussian fluctuations start at \(g^0\), so the leading
    term contains no loops.  It is exactly the sum of connected trees made
    from the full vertices in \(\Gamma\) and their inverse quadratic kernel.
    \item[(b)]
    Translation invariance permits the expansion
    \[
      \Gamma[\phi]
        =\sum_{n=0}^{\infty}\frac1{n!}
         \int\prod_{a=1}^n d^4x_a\,
         \Gamma^{(n)}(x_1,\ldots,x_n)
         \phi(x_1)\cdots\phi(x_n).
    \]
    In momentum space each kernel contains
    \((2\pi)^4\delta^4(p_1+\cdots+p_n)\).  A constant field has Fourier
    transform
    \((2\pi)^4\phi_0\delta^4(p)\), so every external momentum is zero and the
    remaining momentum-conservation delta function becomes
    \[
      (2\pi)^4\delta^4(0)=\mathcal V_4.
    \]
    Therefore
    \[
      \Gamma[\phi_0]
        =\mathcal V_4\sum_{n=0}^{\infty}
          \frac{\Gamma^{(n)}(0,\ldots,0)}{n!}\phi_0^n
        =-\mathcal V_4V(\phi_0),
    \]
    or
    \[
      V(\phi_0)
        =-\sum_{n=0}^{\infty}
          \frac{\Gamma^{(n)}(0,\ldots,0)}{n!}\phi_0^n.
    \]
    Thus a constant-background calculation probes only zero-momentum 1PI
    vertices; momentum dependence belongs to derivative terms in
    \(\Gamma\).
    \item[(c)]
    Diagonalizing the quadratic fluctuation matrices reduces the Gaussian
    functional integral to a product of one-species determinants.  In the
    chapter's state-counting convention,
    \[
      V_1^{\rm nonpol}(\phi)
        =\frac1{64\pi^2}
          \operatorname{STr}\left[
            \mathcal M^4(\phi)
            \ln\frac{\mathcal M^2(\phi)}{\kappa^2}\right].
    \]
    A real scalar has weight \(+1\), a complex scalar has weight \(+2\),
    and a complex Dirac field has weight \(-2\): the minus sign is
    Grassmannian and the factor two counts its spin states as in
    Eq.~\eqref{eq:16.2.16}.  Internal degeneracies multiply these weights.
    Other common determinant conventions retain four Dirac components and
    redistribute a factor two; one must not mix those conventions within a
    calculation.

    The displayed logarithms are the universal nonpolynomial part at this
    order.  Local polynomials in the backgrounds, including constants
    accompanying the logarithms, depend on the regulator and subtraction
    prescription and are absorbed into renormalized couplings.
    \item[(d)]
    Choose
    \[
      \Phi=\frac1{\sqrt2}(\varphi+h+iG).
    \]
    Then
    \[
      V_0=\frac{m^2}{2}\bigl[(\varphi+h)^2+G^2\bigr]
         +\frac{\lambda}{16}
          \bigl[(\varphi+h)^2+G^2\bigr]^2.
    \]
    The quadratic eigenvalues are
    \[
      m_h^2=m^2+\frac{3\lambda}{4}\varphi^2,
      \qquad
      m_G^2=m^2+\frac{\lambda}{4}\varphi^2.
    \]
    Therefore
    \[
      V_1^{\rm nonpol}
        =\frac1{64\pi^2}\left[
          m_h^4\ln\frac{m_h^2}{\kappa^2}
          +m_G^4\ln\frac{m_G^2}{\kappa^2}\right].
    \]
    A general constant background can be rotated to the real axis.  Since
    \(\varphi^2=2\Phi^*\Phi\), both eigenvalues and hence the determinant
    depend only on the invariant \(\Phi^*\Phi\), as required by the global
    \(U(1)\) symmetry.
    \item[(e)]
    Lorentz invariance and integration by parts leave, through two
    derivatives,
    \[
      \Gamma[\phi]
        =\int d^4x\left[-V(\phi)
          -\frac12Z(\phi)(\partial\phi)^2\right]+O(\partial^4).
    \]
    Expanding about a constant background \(\phi_0\), the two-point vertex
    has the small-momentum form
    \[
      \Gamma^{(2)}(p,-p;\phi_0)
        =-V''(\phi_0)+Z(\phi_0)p^2+O(p^4),
    \]
    with overall signs adjusted by the metric convention.  Thus the
    derivative with respect to \(p^2\) at \(p=0\) determines \(Z(\phi_0)\);
    a strictly constant background probes only \(V\).

    Under a local redefinition \(\phi=f(\chi)\),
    \[
      \widetilde V(\chi)=V(f(\chi)),\qquad
      \widetilde Z(\chi)=Z(f(\chi))[f'(\chi)]^2.
    \]
    Hence \(Z\) is parametrization dependent, though observables obtained
    from the complete effective action are unchanged.
  \end{enumerate}

  \SupplementarySolution{6}{Scalar Determinants and Counterterms}
  \begin{enumerate}
    \item[(a)]
    Write the shifted field as \(\phi_0+\eta\).  Through quadratic order,
    \[
      I[\phi_0+\eta]
        =I[\phi_0]-\frac12\eta K\eta+O(\eta^3).
    \]
    The one-loop Gaussian is
    \[
      \int\mathcal D\eta\,
        e^{-i\eta K\eta/2}\propto(\det K)^{-1/2},
    \]
    so
    \[
      i\Gamma_1=-\frac12\operatorname{Tr}\ln K,
      \qquad
      \Gamma_1=\frac{i}{2}\operatorname{Tr}\ln K
    \]
    up to a background-independent normalization.  After Wick rotation the
    potential contribution is
    \[
      \mathcal I_4(x)
        =\frac12\int\frac{d^4k_E}{(2\pi)^4}\ln(k_E^2+x).
    \]
    Three derivatives give
    \[
      \mathcal I_4'''(x)
        =\int\frac{d^4k_E}{(2\pi)^4}\frac1{(k_E^2+x)^3}
        =\frac1{32\pi^2x}.
    \]
    Hence
    \[
      \mathcal I_4(x)
        =\frac{x^2}{64\pi^2}\ln\frac{x}{\kappa^2}
         +A+Bx+Cx^2.
    \]
    The last three terms are local and are fixed by renormalization
    conditions.
    \item[(b)]
    The Hessian of the classical potential is
    \[
      \frac{\partial^2V_0}{\partial\phi_a\partial\phi_b}
        =\left(m^2+\frac{\lambda}{6}\boldsymbol\phi^2\right)\delta_{ab}
         +\frac{\lambda}{3}\phi_a\phi_b.
    \]
    At \(\boldsymbol\phi=(\varphi,0,\ldots,0)\), the radial eigenvalue and
    the \(N-1\) transverse eigenvalues are
    \[
      m_R^2(\varphi)=m^2+\frac{\lambda}{2}\varphi^2,
      \qquad
      m_G^2(\varphi)=m^2+\frac{\lambda}{6}\varphi^2.
    \]
    Thus
    \[
      V_1^{\rm nonpol}(\varphi)
        =\frac1{64\pi^2}\left[
          m_R^4\ln\frac{m_R^2}{\kappa^2}
          +(N-1)m_G^4\ln\frac{m_G^2}{\kappa^2}\right].
    \]
    Polynomial terms are understood to be fixed by the chosen
    renormalization conditions.  At an exact broken minimum, the transverse
    zero modes become the Goldstone modes.
    \item[(c)]
    At large momentum, expand
    \[
      \ln(k^2+x)
        =\ln k^2+\sum_{r\ge1}\frac{(-1)^{r+1}}r
          \left(\frac{x}{k^2}\right)^r.
    \]
    In \(D=2n\), the terms through \(r=n\) are ultraviolet divergent.
    Thus the local divergent part is a polynomial of degree \(n\) in
    \(x=\mu^2\), while the first nonpolynomial term is
    \(x^n\ln x\).

    In six dimensions \(n=3\).  With \(x=a+b\phi\), the polynomial
    \(A+Bx+Cx^2+Dx^3\) becomes a general polynomial through \(\phi^3\).
    Its four coefficients renormalize, respectively, the vacuum energy,
    tadpole, mass, and cubic coupling.  No \(\phi^4\) divergence is generated
    by this one-loop determinant, in accord with six-dimensional power
    counting for the renormalizable \(\phi^3\) theory.
  \end{enumerate}

  \SupplementarySolution{7}{Fermionic Determinants and Graded Legendre Transforms}
  \begin{enumerate}
    \item[(a)]
    For Grassmann vectors,
    \[
      \int\prod_i d\bar\theta_i\,d\theta_i\,
        e^{-\bar\theta_iD_{ij}\theta_j}=\det D.
    \]
    This follows by expanding the exponential: only the term containing
    every \(\theta_i\) and \(\bar\theta_i\) survives, and its antisymmetrized
    coefficient is the determinant.  A real commuting Gaussian instead
    gives \((\det K)^{-1/2}\).  Taking logarithms therefore produces an
    opposite sign for a closed fermion loop.

    For a constant mass,
    \[
      \operatorname{tr}_{\rm spin}\ln(\slashed p-M)
        =\frac12\operatorname{tr}_{\rm spin}
          \ln[(\slashed p-M)(-\slashed p-M)].
    \]
    The product is proportional to \(p^2+M^2\) after Wick rotation.  The
    trace counts the two physical spin states in the convention of this
    chapter, so the fermion logarithm is \(-2\) times a real-scalar
    logarithm:
    \[
      V_{1,f}^{\rm nonpol}
        =-\frac{M^4}{32\pi^2}\ln\frac{M^2}{\kappa^2}.
    \]
    \item[(b)]
    The constant background replaces the fermion mass by
    \(M(\phi)=m+y\phi\).  In the normalization of the chapter,
    \[
      V_{1,f}^{\rm nonpol}(\phi)
        =-\frac{[m+y\phi]^4}{32\pi^2}
          \ln\frac{[m+y\phi]^2}{\kappa^2}.
    \]
    The divergent local part is a polynomial of degree four in \(M\), and
    hence of degree four in \(\phi\).  Without a reflection symmetry the
    scalar sector needs counterterms proportional to
    \[
      1,\quad\phi,\quad\phi^2,\quad\phi^3,\quad\phi^4.
    \]
    If a symmetry enforces \(M(-\phi)=-M(\phi)\), then \(m=0\) in this
    parametrization and the determinant is even.  Only
    \(1,\phi^2,\phi^4\) remain.  The same symmetry must also be respected by
    the regulator for this reduction to be valid.
    \item[(c)]
    With every source placed to the right of its field, variation of \(W\)
    must be written
    \[
      \delta W=\delta J_n\,
        \frac{\delta_R W}{\delta J_n}
    \]
    after the appropriate graded reordering.  Thus
    \[
      \frac{\delta_R W}{\delta J_n}=\chi^n.
    \]
    Varying
    \(\Gamma=W-\chi^nJ_n\) cancels the \(W_{,J}\delta J\) term and gives
    \[
      \frac{\delta_L\Gamma}{\delta\chi^n}=-J_n.
    \]

    Arrange one bosonic and one fermionic variable into a graded column
    \(z=(b,f)\), and define the Hessians with the derivative orientations
    just displayed.  Then their four blocks obey the supermatrix equation
    \[
      \begin{pmatrix}
        \Gamma_{bb}&\Gamma_{bf}\\
        \Gamma_{fb}&\Gamma_{ff}
      \end{pmatrix}
      \begin{pmatrix}
        W_{bb}&W_{bf}\\
        W_{fb}&W_{ff}
      \end{pmatrix}
      =-\begin{pmatrix}1&0\\0&1\end{pmatrix}.
    \]
    Explicitly these are
    \[
      \Gamma_{bb}W_{bb}+\Gamma_{bf}W_{fb}=-1,\quad
      \Gamma_{bb}W_{bf}+\Gamma_{bf}W_{ff}=0,
    \]
    together with the analogous lower-row equations.  The Koszul signs are
    contained in the definitions of the mixed left/right derivative blocks.
    Interchanging their order uses graded transposition,
    \((A^{\mathrm{sT}})_{ij}=(-1)^{\epsilon_i\epsilon_j}A_{ji}\),
    rather than ordinary transposition.
  \end{enumerate}

  \SupplementarySolution{8}{Renormalization Conditions and the Maxwell Segment}
  \begin{enumerate}
    \item[(a)]
    Separate the calculated expression into a fixed part \(F(\phi)\),
    containing the classical terms and the chosen logarithm, and the most
    general even polynomial left by the three integrations:
    \[
      V(\phi)=F(\phi)+c_0+c_2\phi^2+c_4\phi^4.
    \]
    The three conditions determine the coefficients directly:
    \[
      c_0=-F(0),\qquad
      c_2=\frac12\bigl[m_R^2-F''(0)\bigr],\qquad
      c_4=\frac1{24}\bigl[g_R-F^{(4)}(\kappa)\bigr].
    \]
    Since
    \(\mu^2(\phi)=m^2+g\phi^2/2\), the original constants
    \(A+B\mu^2+C\mu^4\) span exactly the same basis
    \(1,\phi^2,\phi^4\), so these equations uniquely fix \(A,B,C\) whenever
    the change of basis is nonsingular.

    Moving \(\kappa\) changes the split between the logarithm and the local
    polynomial.  The renormalized parameters must run so that their sum,
    and hence physical quantities derived from the potential, are unchanged
    to the calculated order.
    \item[(b)]
    Fill a fraction \(\alpha\) of a very large volume with phase 1 and the
    rest with phase 2.  The volume averages are
    \[
      \phi=\alpha\phi_1+(1-\alpha)\phi_2,\qquad
      \mathcal E=\alpha V_1+(1-\alpha)V_2+O(A_{\rm wall}/\mathcal V_3).
    \]
    Solving for
    \(\alpha=(\phi_2-\phi)/(\phi_2-\phi_1)\) and taking the
    infinite-volume limit gives
    \[
      V_{\rm mix}(\phi)
        =\frac{\phi_2-\phi}{\phi_2-\phi_1}V_1
         +\frac{\phi-\phi_1}{\phi_2-\phi_1}V_2.
    \]
    This is the straight chord between the two homogeneous phases.  The
    interface energy scales like area and vanishes per unit volume.  When
    \(V_1=V_2\), the chord is horizontal, producing the flat Maxwell segment
    between degenerate vacua.
  \end{enumerate}

  \SupplementarySolution{9}{Dimensional Transmutation and RG Invariance}
  \begin{enumerate}
    \item[(a)]
    For \(\phi\ne0\),
    \[
      V'(\phi)
        =\phi^3\left[
          4A+B\left(4\ln\frac{\phi^2}{\kappa^2}+2\right)\right].
    \]
    A nonzero stationary point \(v\) therefore obeys
    \[
      \ln\frac{v^2}{\kappa^2}=-\frac{A}{B}-\frac12.
    \]
    Both signs \(\phi=\pm v\) occur when the field has a reflection
    symmetry.  At either point,
    \[
      V''(v)=8Bv^2>0,
    \]
    so it is a local minimum for \(B>0\).  Eliminating \(A\) gives
    \[
      V(\phi)
        =B\phi^4\left(\ln\frac{\phi^2}{v^2}-\frac12\right),
    \]
    up to an additive vacuum constant.  The dimensionless combination
    \(A/B\) has been replaced by the dimensionful scale
    \(v=\kappa\exp[-A/(2B)-1/4]\).  This is dimensional transmutation.
    \item[(b)]
    Holding bare fields and couplings fixed while differentiating the
    renormalized potential gives
    \[
      \left(\kappa\partial_\kappa
        +\sum_i\beta_i\partial_{g_i}
        -\gamma\phi\partial_\phi\right)V=0.
    \]
    The last term appears because the renormalized field changes with
    \(\kappa\).  Insert
    \[
      V=\frac{\lambda}{24}\phi^4
        +B\phi^4\ln\frac{\phi^2}{\kappa^2}.
    \]
    Through one loop, derivatives of \(B\) and of the logarithmic term with
    respect to running couplings are of higher order.  The three surviving
    contributions are
    \[
      -2B\phi^4+\frac{\beta_\lambda}{24}\phi^4
        -\frac{\gamma\lambda}{6}\phi^4=0.
    \]
    Hence
    \[
      \boxed{
        B=\frac{\beta_\lambda-4\gamma\lambda}{48}}
    \]
    at one loop.  This relation ensures that the explicit scale dependence
    of the logarithm is canceled by the implicit running of the coupling and
    field normalization.
  \end{enumerate}

  \SupplementarySolution{10}{Composite Sources and the Two-Particle-Irreducible Action}
  Differentiating the connected generator gives
  \[
    \varphi(x)=\frac{\delta W}{\delta J(x)},\qquad
    2\frac{\delta W}{\delta K(x,y)}
      =\langle\phi(x)\phi(y)\rangle
      =\varphi(x)\varphi(y)+G(x,y).
  \]
  Varying the double Legendre transform while using these definitions
  cancels the implicit variations of \(J\) and \(K\).  Because \(K\) is
  symmetric,
  \[
    \boxed{\frac{\delta\Gamma}{\delta\varphi(x)}
      =-J(x)-\int_yK(x,y)\varphi(y),\qquad
      \frac{\delta\Gamma}{\delta G(x,y)}=-\frac12K(x,y).}
  \]

  Let
  \(iD^{-1}(x,y;\varphi)=\delta^2S[\varphi]/
    [\delta\varphi(x)\delta\varphi(y)]\), with the factor of \(i\)
  adjusted if the quadratic kernel is defined directly in the exponent.
  Shifting to \(\eta=\phi-\varphi\), expanding the action, and collecting
  the Gaussian determinant gives the CJT form
  \[
    \Gamma[\varphi,G]
      =S[\varphi]
       +\frac{i}{2}\operatorname{Tr}\ln G^{-1}
       +\frac{i}{2}\operatorname{Tr}\!\bigl[D^{-1}(\varphi)G-\mathbf1\bigr]
       +\Gamma_2[\varphi,G].
  \]
  The first trace is the entropy-like determinant and the second replaces
  the free quadratic kernel by the full trial propagator.  Expanding the
  shifted interactions with propagator \(G\), every two-particle-reducible
  vacuum graph is generated by iterating these explicit trace terms and the
  stationarity equation.  The remainder \(\Gamma_2\) is therefore the sum
  of vacuum skeletons that cannot be disconnected by cutting two internal
  \(G\) lines.

  With the external sources removed,
  \[
    \frac{\delta\Gamma}{\delta\varphi}=0,\qquad
    \frac{\delta\Gamma}{\delta G}=0.
  \]
  The second equation is the self-consistent Dyson equation
  \[
    G^{-1}=D^{-1}(\varphi)-\Sigma,\qquad
    \Sigma=2i\,\frac{\delta\Gamma_2}{\delta G},
  \]
  up to the same continuation convention.  Thus \(G\) is the full
  propagator, rather than merely its coincident local value; the bilocal
  Legendre variable is what makes the construction genuinely 2PI.

  \SupplementarySolution{11}{The Exact Background-Field Effective-Action Identity}
