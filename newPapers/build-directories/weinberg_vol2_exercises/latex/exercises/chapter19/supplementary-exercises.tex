\chapterbackmatter{Supplementary Exercises}

\begin{enumerate}
  \SupplementaryExercise{1}{Counting Tangent Modes in an \(O(N)\) Vacuum}
  {Adapted from David Tong, The Standard Model (2025).}
  {tong-standard-model-broken-symmetries-2025}
  An \(O(N)\)-invariant scalar theory has a vacuum
  \(\langle\boldsymbol\phi\rangle=v\boldsymbol e_N\).
  Determine the unbroken group, the vacuum manifold, and the number of
  Goldstone bosons.  Identify the broken generators explicitly.

  \SupplementaryExercise{2}{Radial and Angular Masses in the Linear Sigma Model}
  {Adapted from David Tong, The Standard Model (2025).}
  {tong-standard-model-broken-symmetries-2025}
  For
  \(V(\boldsymbol\phi)=\lambda(\boldsymbol\phi^2-v^2)^2/4\),
  expand about
  \(\boldsymbol\phi=(\boldsymbol\pi,v+\sigma)\).
  Find the quadratic masses and explain why the result agrees with the
  symmetry-breaking pattern.

  \SupplementaryExercise{3}{The Zero Vector of an Invariant Hessian}
  {Adapted from David Tong, The Standard Model (2025).}
  {tong-standard-model-broken-symmetries-2025}
  Let \(V(\phi)\) be invariant under
  \(\delta\phi_n=i\epsilon(t\phi)_n\), and let \(v\) be a stationary
  point.  Differentiate the invariance identity and prove that \(tv\) is a
  null eigenvector of the mass-squared matrix whenever the generator is
  broken.

  \SupplementaryExercise{4}{Why Chiral Symmetry Leaves a Diagonal Group}
  {Adapted from David Tong, The Standard Model (2025).}
  {tong-standard-model-broken-symmetries-2025}
  Suppose
  \(\langle\overline q_{Rj}q_{Li}\rangle=-\Sigma\delta_{ij}\).
  Under \(SU(N_f)_L\times SU(N_f)_R\), determine the subgroup that leaves
  this condensate invariant and count the resulting Goldstone bosons.

  \SupplementaryExercise{5}{A Two-Flavor Mass Spurion}
  {Adapted from David Tong, The Standard Model (2025).}
  {tong-standard-model-broken-symmetries-2025}
  Let \(U\in SU(2)\) transform as \(U\mapsto RUL^\dagger\), and assign
  \(M\mapsto RML^\dagger\) to the quark-mass spurion.  Show that
  \(\operatorname{Tr}(M^\dagger U+U^\dagger M)\) is invariant.  Expand
  \(U=\exp(i\tau_a\pi_a/f)\) through quadratic order and show that
  \(m_u-m_d\) does not split the pion triplet at first order.

  \SupplementaryExercise{6}{The Missing Singlet Goldstone Boson}
  {Adapted from David Tong, The Standard Model (2025).}
  {tong-standard-model-broken-symmetries-2025}
  If the axial \(U(1)_A\) of three-flavor massless QCD were an exact
  spontaneously broken symmetry, how many pseudoscalar Goldstone bosons
  would be expected?  Explain why the observed octet alone signals that
  one premise of this argument fails.

  \SupplementaryExercise{7}{Metric of a Spherical Nonlinear Sigma Model}
  {Adapted from David Tong, Statistical Field Theory (2021).}
  {tong-statistical-field-theory-goldstone-2021}
  Parameterize an \(O(N+1)/O(N)\) unit vector by
  \(n=(\boldsymbol\pi/F,\sqrt{1-\boldsymbol\pi^2/F^2})\).
  Compute the target-space metric \(g_{ab}(\pi)\) in
  \(\mathcal L=-\tfrac12g_{ab}\partial_\mu\pi_a\partial^\mu\pi_b\),
  and expand it through \(O(\pi^2/F^2)\).

  \SupplementaryExercise{8}{Maurer--Cartan Forms on \(SO(3)/SO(2)\)}
  {Adapted from David Tong, Statistical Field Theory (2021).}
  {tong-statistical-field-theory-goldstone-2021}
  With \(\gamma=\exp(i\xi_at_a)\), \(a=1,2\), expand
  \(\gamma^{-1}\partial_\mu\gamma\) through quadratic order in \(\xi\).
  Separate the broken covariant derivative from the unbroken connection,
  and verify their leading transformation roles.

  \SupplementaryExercise{9}{A Soft Zero from Derivative Couplings}
  {Adapted from David Tong, Statistical Field Theory (2021).}
  {tong-statistical-field-theory-goldstone-2021}
  In an exact internal-symmetry nonlinear sigma model, every Goldstone
  interaction contains derivatives.  Explain why a tree amplitude with
  one external Goldstone momentum \(q\) vanishes as \(q\to0\), and state
  the exception caused by an external-leg pole.

  \SupplementaryExercise{10}{Weinberg Power Counting for Pure Goldstones}
  {Adapted from David Tong, Statistical Field Theory (2021).}
  {tong-statistical-field-theory-goldstone-2021}
  A connected graph has \(L\) loops and \(V_i\) vertices containing
  \(d_i\) derivatives.  Derive
  \(\nu=2+2L+\sum_iV_i(d_i-2)\) for its low-momentum order \(Q^\nu\).
  Apply the formula to a one-loop graph built only from two-derivative
  vertices.

  \SupplementaryExercise{11}{Curvature Controls the Four-Goldstone Vertex}
  {Adapted from David Tong, Statistical Field Theory (2021).}
  {tong-statistical-field-theory-goldstone-2021}
  In Riemann normal coordinates about the vacuum,
  \(g_{ab}=\delta_{ab}-R_{acbd}\pi_c\pi_d/3+\cdots\).
  Show that the leading four-Goldstone interaction is fixed by the target
  curvature.  Specialize to a sphere of radius \(F\) and state the
  crossing-symmetric massless amplitude.

  \SupplementaryExercise{12}{A Magnetic Field as a Pseudo-Goldstone Spurion}
  {Adapted from David Tong, Statistical Field Theory (2021).}
  {tong-statistical-field-theory-goldstone-2021}
  An \(O(3)\) order parameter of fixed length \(F\) is weakly perturbed by
  \(\Delta\mathcal L=hF n_3\).  Expand about the aligned vacuum and find
  the two pseudo-Goldstone masses.  Explain the dependence on \(h\).

  \SupplementaryExercise{13}{The Current Pole in Goldstone's Theorem}
  {Adapted from Stefan Scherer, Introduction to Chiral Perturbation Theory (2002).}
  {scherer-chiral-perturbation-2002}
  Let \(J^\mu_a\) be a broken conserved current and \(\Phi_b\) an operator
  with a nonzero vacuum commutator.  Use the spectral representation of
  \(\langle T J^\mu_a(x)\Phi_b(0)\rangle\) and current conservation to
  show that its Fourier transform must contain a pole at \(q^2=0\).

  \SupplementaryExercise{14}{PCAC and the One-Pion Matrix Element}
  {Adapted from Stefan Scherer, Introduction to Chiral Perturbation Theory (2002).}
  {scherer-chiral-perturbation-2002}
  Given
  \(\bra{\Omega}A^\mu_a(0)\ket{\pi_b(p)}
    =if_\pi p^\mu\delta_{ab}\),
  derive the one-pion matrix element of \(\partial_\mu A^\mu_a\) for a
  pion of mass \(m_\pi\).  State what happens in the chiral limit.

  \SupplementaryExercise{15}{Isospin Channels at Pion Threshold}
  {Adapted from Stefan Scherer, Introduction to Chiral Perturbation Theory (2002).}
  {scherer-chiral-perturbation-2002}
  Starting from
  \(A(s,t,u)=(s-m_\pi^2)/f_\pi^2\), form the \(I=0,1,2\)
  amplitudes
  \(T^0=3A(s,t,u)+A(t,s,u)+A(u,t,s)\),
  \(T^1=A(t,s,u)-A(u,t,s)\), and
  \(T^2=A(t,s,u)+A(u,t,s)\).
  Evaluate them at threshold and obtain \(a_0^0\) and \(a_0^2\) in the
  convention \(a_0^I=T^I_{\rm thr}/(32\pi)\).

  \SupplementaryExercise{16}{Recovering Goldberger--Treiman}
  {Adapted from Stefan Scherer, Introduction to Chiral Perturbation Theory (2002).}
  {scherer-chiral-perturbation-2002}
  The leading pion--nucleon interaction is
  \(-g_A\overline N\gamma^\mu\gamma_5\tau_a
  (\partial_\mu\pi_a)N/(2f_\pi)\).
  Use the on-shell Dirac equations to replace it by a pseudoscalar
  Yukawa interaction and derive the Goldberger--Treiman relation.

  \SupplementaryExercise{17}{Single-Baryon Chiral Index}
  {Adapted from Stefan Scherer, Introduction to Chiral Perturbation Theory (2002).}
  {scherer-chiral-perturbation-2002}
  Count a pion propagator as \(Q^{-2}\), a nearly on-shell nucleon
  propagator as \(Q^{-1}\), a loop as \(Q^4\), and a vertex of type \(i\)
  as \(Q^{d_i}\).  Derive the chiral index for a connected graph with
  \(E_N\) external nucleon lines and vertices containing \(n_i\) nucleon
  fields.

  \SupplementaryExercise{18}{A Soft-Pion Commutator Formula}
  {Adapted from Stefan Scherer, Introduction to Chiral Perturbation Theory (2002).}
  {scherer-chiral-perturbation-2002}
  For a local operator \(\mathcal O\), use the pion pole in the axial
  current to show schematically that
  \(\lim_{q\to0}\langle\beta;\pi^a(q)|\mathcal O|\alpha\rangle\)
  is fixed by \(\langle\beta|[Q_5^a,\mathcal O]|\alpha\rangle/f_\pi\),
  apart from external-leg pole terms.  Track the factor of \(i\).

  \SupplementaryExercise{19}{The Unique Two-Derivative Chiral Invariant}
  {Adapted from Antonio Pich, Effective Field Theory (1998).}
  {pich-effective-field-theory-1998}
  Let \(U\mapsto RUL^\dagger\) under
  \(SU(3)_R\times SU(3)_L\).  Show that, up to normalization and a total
  derivative, the unique parity-even invariant with two derivatives is
  \(\operatorname{Tr}(\partial_\mu U\partial^\mu U^\dagger)\).
  Expand it to verify canonical kinetic terms.

  \SupplementaryExercise{20}{Spurion Classification of Quark-Mass Terms}
  {Adapted from Antonio Pich, Effective Field Theory (1998).}
  {pich-effective-field-theory-1998}
  Assign the mass spurion \(\chi\mapsto R\chi L^\dagger\) and parity
  \(U\leftrightarrow U^\dagger\), \(\chi\leftrightarrow\chi^\dagger\).
  Find the unique invariant linear in \(\chi\) and containing no
  derivatives.  Explain why \(\chi\) counts as order \(Q^2\).

  \SupplementaryExercise{21}{The Pseudoscalar Gell-Mann--Okubo Relation}
  {Adapted from Antonio Pich, Effective Field Theory (1998).}
  {pich-effective-field-theory-1998}
  In the isospin limit, use
  \(m_\pi^2=2B_0\widehat m\),
  \(m_K^2=B_0(\widehat m+m_s)\), and
  \(m_\eta^2=2B_0(\widehat m+2m_s)/3\)
  to eliminate the quark masses and obtain the leading pseudoscalar
  mass relation.

  \SupplementaryExercise{22}{Quark-Mass Ratios with Dashen's Cancellation}
  {Adapted from Antonio Pich, Effective Field Theory (1998).}
  {pich-effective-field-theory-1998}
  Assume the leading electromagnetic shift is the same for
  \(K^+\) and \(\pi^+\), and vanishes for \(K^0,\pi^0\).
  Eliminate the common electromagnetic term and \(B_0\) to derive
  \(m_u/m_d\) from the four measured squared meson masses.

  \SupplementaryExercise{23}{Why One Loop Requires Fourth-Order Couplings}
  {Adapted from Antonio Pich, Effective Field Theory (1998).}
  {pich-effective-field-theory-1998}
  Use chiral power counting to show that one-loop graphs made only from
  \(\mathcal L_2\) and tree graphs with one \(\mathcal L_4\) insertion
  both contribute at order \(Q^4\).  Explain why the coefficients in
  \(\mathcal L_4\) must run with the subtraction scale.

  \SupplementaryExercise{24}{Five Mesons from the Wess--Zumino--Witten Term}
  {Adapted from Antonio Pich, Effective Field Theory (1998).}
  {pich-effective-field-theory-1998}
  Put \(U=\exp(2iB/F)\) in
  \(\int_{B_5}\operatorname{Tr}(U^{-1}dU)^5\).
  Determine the lowest power of \(B\), the number of derivatives, and the
  parity-odd tensor that appears in four-dimensional spacetime.  Explain
  why its coefficient is quantized.

  \SupplementaryExercise{25}{Vacuum Alignment in a Tilted Mexican Hat}
  {Adapted from MIT OpenCourseWare 8.325, Relativistic Quantum Field Theory III (2007).}
  {mit-ocw-qft-iii-2007}
  Add \(-\boldsymbol h\cdot\boldsymbol\phi\) to an \(O(N)\) linear sigma
  model in its broken phase.  Determine the aligned vacuum direction and,
  to first order in \(|\boldsymbol h|\), the transverse mass squared.

  \SupplementaryExercise{26}{The CCWZ Compensator as a Local Rotation}
  {Adapted from MIT OpenCourseWare 8.325, Relativistic Quantum Field Theory III (2007).}
  {mit-ocw-qft-iii-2007}
  Starting from
  \(g\gamma(\xi)=\gamma(\xi')h(\xi,g)\), show that a matter field
  transforming only under the unbroken group acquires the
  field-dependent transformation \(h(\xi,g)\).  Derive the
  inhomogeneous transformation of the unbroken Maurer--Cartan connection.

  \SupplementaryExercise{27}{A Positivity Bound for Massive Euclidean Fermions}
  {Adapted from MIT OpenCourseWare 8.325, Relativistic Quantum Field Theory III (2007).}
  {mit-ocw-qft-iii-2007}
  Let \(\slashed D\) be anti-Hermitian in Euclidean space and let all
  eigenvalues of the Hermitian flavor mass matrix satisfy
  \(|m_a|\geq m_{\min}>0\).  Prove
  \(\|(\slashed D+M)^{-1}\|\leq1/m_{\min}\), and explain why this blocks
  spontaneous breaking of exact vector flavor symmetries by a singular
  zero-mass limit.

  \SupplementaryExercise{28}{Why \(U(1)_A\) Evades Goldstone Counting}
  {Adapted from MIT OpenCourseWare 8.325, Relativistic Quantum Field Theory III (2007).}
  {mit-ocw-qft-iii-2007}
  Write the classical singlet axial current of \(N_f\) massless quarks and
  its quantum divergence in a non-Abelian gauge field.  Explain why the
  anomaly removes the ninth Goldstone prediction for \(N_f=3\).

  \SupplementaryExercise{29}{Neutral-Pion Decay from the Anomaly}
  {Adapted from MIT OpenCourseWare 8.325, Relativistic Quantum Field Theory III (2007).}
  {mit-ocw-qft-iii-2007}
  Using
  \(\mathcal L_{\pi\gamma\gamma}
    =g_{\pi\gamma\gamma}\pi^0
      \epsilon^{\mu\nu\rho\sigma}F_{\mu\nu}F_{\rho\sigma}/4\)
  with \(g_{\pi\gamma\gamma}=\alpha/(\pi f_\pi)\), derive
  \(\Gamma(\pi^0\to\gamma\gamma)\) in terms of
  \(\alpha,f_\pi,m_\pi\).

  \SupplementaryExercise{30}{An Adjoint Source and Its Residual Torus}
  {Adapted from MIT OpenCourseWare 8.325, Relativistic Quantum Field Theory III (2007).}
  {mit-ocw-qft-iii-2007}
  An \(SU(N)\) theory is spontaneously broken to the stabilizer of a unit
  vector \(v\).  Add a generic traceless Hermitian adjoint source \(A\).
  Show that vacuum alignment chooses an eigenvector of \(A\), and find
  the completely unbroken group when all eigenvalues are distinct.
\end{enumerate}
