\chapterbackmatter{Weinberg Exercises}

\begin{enumerate}
\WeinbergExercise{1} Consider a theory of a fermion field $\psi$ interacting with a scalar field $\phi$ with interactions of the form $\bar\psi\psi\phi$ and $\phi^4$. List the operators that appear in the operator product expansion of $\bar\psi(x)\psi(0)$ with coefficient functions that (judging from perturbation theory) are singular and non-vanishing for $x\to0$. Describe how you would calculate these coefficient functions to one-loop order.

\WeinbergExercise{2} Consider quantum chromodynamics with $N$ massless quarks, and define the spectral functions of the scalar and pseudoscalar quark bilinears by
\[
\begin{aligned}
&\sum_N
\bra{\Omega}\bar\psi(0)\lambda_\alpha\psi(0)\ket{\Omega}
\bra{\Omega}\bar\psi(0)\lambda_\beta\psi(0)\ket{\Omega}^{*}
\\
&\hspace{36mm}
=(2\pi)^{-3}\theta(p^0)\rho_{\alpha\beta}^{S}(-p^2),
\end{aligned}
\]
\[
\begin{aligned}
&\sum_N
\bra{\Omega}\bar\psi(0)\gamma^5\lambda_\alpha\psi(0)\ket{\Omega}
\bra{\Omega}\bar\psi(0)\gamma^5\lambda_\beta\psi(0)\ket{\Omega}^{*}
\\
&\hspace{36mm}
=(2\pi)^{-3}\theta(p^0)\rho_{\alpha\beta}^{P}(-p^2),
\end{aligned}
\]
where $\lambda_\alpha$ are a complete set of traceless Hermitian $N\cdot N$ matrices, normalized so that $\operatorname{Tr}(\lambda_\alpha\lambda_\beta)=2\delta_{\alpha\beta}$. What spectral function sum rules are satisfied by linear combinations of $\rho_{\alpha\beta}^{S}(\mu^2)$ and $\rho_{\alpha\beta}^{P}(\mu^2)$?

\WeinbergExercise{3} Derive Eq.~\eqref{eq:20.6.8} in the parton model from the formulas \hyperref[eq:8.7.7]{(8.7.7)} and \hyperref[eq:8.7.38]{(8.7.38)} for Compton scattering.

\WeinbergExercise{4} List the gauge-invariant symmetric traceless tensors of twist four in quantum chromodynamics.

\WeinbergExercise{5} In the massless scalar field theory with interaction $-g\phi^4/24$ (with $g>0$), where in the complex plane would you expect the function \eqref{eq:20.7.3} to have renormalon singularities?
\end{enumerate}
