\begin{enumerate}
  \SupplementarySolution{1}{A Free-Field Operator-Product Expansion}
  \begin{enumerate}
    \item[(a)]
    There are six choices for a single self-contraction and three pairings
    for a double self-contraction.  Therefore
    \[
      \boxed{\phi^4(x)=:\!\phi^4(x)\!:
        +6D(0):\!\phi^2(x)\!:+3D(0)^2\mathbf{1}.}
    \]
    This identity also specifies how the regulator dependence in \(D(0)\)
    is absorbed into the definition of the composite operators.
    \item[(b)]
    Normal ordering forbids contractions within either factor, so only
    cross-contractions occur.  Contracting \(k\) fields across the two
    factors gives
    \[
      O(x)O(0)
       =\sum_{k=0}^4\binom4k^2k!\,D(x)^k
          :\!\phi^{4-k}(x)\phi^{4-k}(0)\!:.
    \]
    Terms with \(k\leq1\) begin with operators of dimensions eight and six,
    respectively.  The three relevant contraction sectors are thus
    \[
      24D(x)^4\mathbf{1},\qquad
      96D(x)^3:\!\phi(x)\phi(0)\!:,\qquad
      72D(x)^2:\!\phi^2(x)\phi^2(0)\!:.
    \]
    Taylor expansion within normal ordering gives
    \[
      :\!\phi(x)\phi(0)\!:
       =:\!\phi^2\!:
        +\frac{x^\mu}{2}\partial_\mu:\!\phi^2\!:
        +x^\mu x^\nu
          \left[\frac14\partial_\mu\partial_\nu:\!\phi^2\!:
                -\frac12:\!\partial_\mu\phi\,\partial_\nu\phi\!:\right]
        +O(\Delta\geq5),
    \]
    where all operators on the right are evaluated at the origin.  Also
    \[
      :\!\phi^2(x)\phi^2(0)\!:
        =:\!\phi^4(0)\!:+O(\Delta\geq5).
    \]
    Combining these results, the complete OPE through canonical dimension
    four is
    \[
      \begin{aligned}
      O(x)O(0)={}&24D(x)^4\mathbf{1}\\
       &+96D(x)^3\Bigl[
          :\!\phi^2\!:
          +\frac{x^\mu}{2}\partial_\mu:\!\phi^2\!:
          +x^\mu x^\nu\!
            \left(\frac14\partial_\mu\partial_\nu:\!\phi^2\!:
                 -\frac12:\!\partial_\mu\phi\,\partial_\nu\phi\!:\right)
        \Bigr]\\
       &+72D(x)^2:\!\phi^4\!:+O(\Delta\geq5).
      \end{aligned}
    \]
    Thus the dimension-four tensor operators as well as the scalar
    primaries and descendants are retained.  If one imposes the free field
    equation, traces of the dimension-four tensor basis may be reduced; the
    unreduced basis above is the complete local Taylor expansion.
  \end{enumerate}

  \SupplementarySolution{2}{An \(O(N)\) OPE and the Wilson--Fisher Exponent}
  \begin{enumerate}
    \item[(a)]
    Double contractions pair the two fields at \(x\) with the two at the
    origin in \(2N\) ways, while a single contraction has four choices.
    Therefore
    \[
      \boxed{E(x)E(0)
        =2N\,G(x)^2\mathbf{1}+4G(x)E(0)+O(\dd\geq4).}
    \]
    The omitted terms include the quartic operator and descendants.
    \item[(b)]
    Contracting two quartic vertices across a short-distance shell gives
    three index channels.  Their \(O(N)\) multiplicities sum to \(N+8\);
    the symmetry factor is \(1/6\).  Combining this logarithmic term with
    the engineering dimension of \(u\) gives
    \[
      \beta(u)=-\epsilon u+\frac{N+8}{6}u^2+O(u^3).
    \]
    Besides the Gaussian point, the infrared fixed point is
    \[
      \boxed{u_*=\frac{6\epsilon}{N+8}+O(\epsilon^2).}
    \]
    \item[(c)]
    Substitution gives
    \[
      y_t=2-\frac{N+2}{N+8}\epsilon+O(\epsilon^2).
    \]
    Expanding its reciprocal,
    \[
      \boxed{\nu=\frac12+
        \frac{N+2}{4(N+8)}\epsilon+O(\epsilon^2).}
    \]
    \item[(d)]
    At \(N=1\),
    \(\nu=\tfrac12+\epsilon/12+O(\epsilon^2)\).
    The coefficient \(4\) in the free \(E\times E\) OPE determines the
    leading mixing of the thermal operator, while the index contractions
    of the quartic OPE give \(N+8\) in the coupling flow.  Evaluating that
    mixing at the zero of the beta function turns Gaussian OPE
    combinatorics into the first interacting correction to \(\nu\).
  \end{enumerate}

  \SupplementarySolution{3}{Scheme Changes and Dimensional Transmutation}
  \begin{enumerate}
    \item[(a)] The exact chain rule is
    \[
      \boxed{\bar\beta(\bar g)
        =\frac{\dd\bar g}{\dd g}\,\beta(g).}
    \]
      Hence \(\beta=\beta^i\partial_i\) is a contravariant vector on
      coupling space.
    \item[(b)] Series inversion and re-expansion give
      \[
        \bar b_2=b_2,\qquad \bar b_3=b_3,\qquad
        \bar b_4=b_4-a_2b_3+(a_3-a_2^2)b_2 .
      \]
      Thus \(a_3\) can choose \(\bar b_4\) freely when \(b_2\ne0\).
      The chain rule maps a zero \(\lambda_F\) to a zero
      \(\bar\lambda_F\).  Differentiating it and then setting
      \(\beta(\lambda_F)=0\) gives
      \[
        \bar\beta'(\bar\lambda_F)=\beta'(\lambda_F),
      \]
      so the fixed-point eigenvalue is invariant.
    \item[(c)] Expanding the inverse flow,
      \[
        \frac1{-bg^2-cg^3-\cdots}
          =-\frac1{bg^2}+\frac{c}{b^2g}+O(1),
      \]
      and integrating gives
      \[
        \log\frac{\mu}{\Lambda}
          =\frac1{bg}+\frac{c}{b^2}\log(bg)+O(g).
      \]
      Equivalently,
      \(\Lambda=\mu\exp[-\int^g \dd g'/\beta(g')]\); applying
      \(\mu d/\dd\mu\) proves \(\dd\Lambda/\dd\mu=0\).
    \item[(d)] With one dimensionless coupling and no mass parameter, the
      autonomous RG equation has one integration constant.  Dimensional
      analysis writes it as a scale \(\Lambda\), so inversion of the
      integrated flow gives
      \(g(\mu)=f(\log(\mu/\Lambda))\).  Fixing a dimensionless coupling at
      one reference scale is therefore equivalent to fixing the
      dimensionful \(\Lambda\).
  \end{enumerate}

  \SupplementarySolution{4}{Correlators of Composite Operators Made of Free Bosons}
  \begin{enumerate}
    \item[(a)]
    On a finite world sheet of area \(A\), decompose each field and source
    as
    \[
      X^\mu(\sigma)=x_0^\mu+\widetilde X^\mu(\sigma),
      \qquad
      \int \dd^2\sigma\,\widetilde X^\mu=0,
      \qquad
      J_0^\mu=\int \dd^2\sigma\,J^\mu(\sigma).
    \]
    The action contains only \(\widetilde X\).  Consequently the path
    integral factorizes:
    \[
      Z[J]=
      \frac{\int_{\mathbb R^n}\dd^nx_0\,
        e^{x_0\cdot J_0}}
      {\int_{\mathbb R^n}\dd^nx_0}\,
      \exp\left\{\frac12\int \dd^2\sigma\,\dd^2\sigma'\,
        J_\mu(\sigma)G(\sigma,\sigma')J^\mu(\sigma')\right\}.
      \tag{20.4.S1}
    \]
    For an ordinary real source the first ratio is not defined when
    \(J_0\neq0\); this is the zero-mode obstruction, not an innocuous
    normalization constant.  For neutral sources, \(J_0=0\), the same
    regulated target-space volume cancels between numerator and
    denominator, leaving
    \[
      \boxed{Z_{\rm osc}[J]=
      \exp\left\{\frac12\int J_\mu GJ^\mu\right\},
      \qquad \int J^\mu=0.}
      \tag{20.4.S2}
    \]
    For Fourier sources \(J=ij\), integration over \(x_0\) instead gives
    \((2\pi)^n\delta^{(n)}(j_0)\).  Thus the familiar delta function is
    the distributional version of the same neutrality condition.

    To verify the Green function, write the quadratic kernel as
    \(K=-(2\pi g)^{-1}\nabla^2\).  Then \(KG=1\) requires
    \(\nabla^2G=-2\pi g\,\delta^{(2)}\), and hence
    \(G=-g\log(|z-w|/\ell)\) after its zero mode is removed.

    \item[(b)]
    Insert the distributional source
    \[
      J^\mu(\sigma)=-\ii\sqrt{2\alpha'}
        \sum_i k_i^\mu\delta^{(2)}(\sigma-\sigma^{(i)}).
    \]
    The zero-mode integral gives
    \(\delta^{(n)}(\sum_i k_i)\), with constants involving
    \(2\pi\) and \(\alpha'\) absorbed into the chosen normalization of the
    noncompact measure.  Normal ordering deletes all \(i=j\)
    contractions.  The remaining terms in (20.4.S2) give
    \[
      \frac12\sum_{i\ne j}
      (-\ii\sqrt{2\alpha'}\,k_i)\mathbin{\cdot}
      (-\ii\sqrt{2\alpha'}\,k_j)\,
      [-g\log|z_i-z_j|]
      =
      2\alpha'g\sum_{i<j}k_i\mathbin{\cdot}k_j
        \log|z_i-z_j|.
    \]
    Therefore the result consistent with the stated Euclidean action is
    \[
      \boxed{
      \left\langle\prod_i
        :e^{-\ii\sqrt{2\alpha'}\,k_i\cdot X(z_i)}:
      \right\rangle
      =
      \delta^{(n)}\!\left(\sum_i k_i\right)
      \prod_{i<j}|z_i-z_j|^{\,2\alpha'g\,k_i\cdot k_j}.}
      \tag{20.4.S3}
    \]
    Equivalently, the product may be over ordered pairs \(i\ne j\) with
    exponent \(+\alpha'g\,k_i\cdot k_j\).  Thus (20.4.1), as printed in
    the source, has both undefined self-contraction factors and the
    opposite exponent sign.  The latter also conflicts with the
    two-point scaling law in part (c); it is a printed sign error in the
    conventions of this problem.

    The action is invariant under the constant shift
    \(X^\mu\mapsto X^\mu+a^\mu\).  The product of vertices acquires the
    phase
    \(\exp[-\ii\sqrt{2\alpha'}\,a\cdot\sum_i k_i]\).
    Averaging over the symmetry orbit kills the correlator unless
    \(\sum_i k_i=0\), explaining the delta function.

    \item[(c)]
    For general charges \(a_i\), the same calculation gives
    \[
      \left\langle\prod_i:\!e^{ia_iX(z_i)}\!:\right\rangle
      =
      \delta\!\left(\sum_i a_i\right)
      \prod_{i<j}|z_i-z_j|^{g a_i a_j}.
      \tag{20.4.S4}
    \]
    In the two-point function the zero mode enforces \(a=b\), and then
    \[
      \langle\mathcal O_a(z)\mathcal O_b^\dagger(0)\rangle
        =\delta(a-b)|z|^{-ga^2}.
    \]
    Comparing with \(|z|^{-2\Delta_a}\) yields
    \[
      \boxed{\Delta_a=\frac{ga^2}{2}.}
    \]
    The arbitrary length \(\ell\) drops out when the charges are neutral.

    \item[(d)]
    Single-valuedness requires
    \[
      e^{ia(X+2\pi R)}=e^{iaX},
      \qquad\Longrightarrow\qquad
      \boxed{a=\frac{m}{R},\quad m\in\mathbb Z.}
    \]
    With a normalized compact zero-mode integral,
    \[
      \frac{1}{2\pi R}\int_0^{2\pi R}\dd x_0\,
       e^{\ii x_0\sum_i m_i/R}
       =\delta_{\sum_i m_i,0}.
    \]
    Hence on the plane
    \[
      \boxed{
      \left\langle\prod_i
        :e^{\ii(m_i/R)X(z_i)}:
      \right\rangle
      =
      \delta_{\sum_i m_i,0}
      \prod_{i<j}|z_i-z_j|^{g m_i m_j/R^2}.}
    \]
    For the vertices of part (b),
    \(-\sqrt{2\alpha'}\,k_i=m_i/R\); their continuous delta function is
    replaced by the Kronecker neutrality condition.  On a world sheet
    with noncontractible cycles one must additionally sum over winding
    sectors, but that global sum does not alter this local plane result.
  \end{enumerate}

  \SupplementarySolution{5}{The Stress Tensor Is Not a Conformal Primary}
  \begin{enumerate}
    \item[(a)]
    Expand the test function about the contour center:
    \[
      \xi(z)=\xi(w)+(z-w)\partial\xi(w)
       +\frac{(z-w)^2}{2}\partial^2\xi(w)
       +\frac{(z-w)^3}{6}\partial^3\xi(w)+\cdots .
    \]
    The contour selects the coefficient of \((z-w)^{-1}\).  The three
    singular terms in the \(TT\) OPE respectively contribute
    \(c\,\partial^3\xi/12\), \(2(\partial\xi)T\), and
    \(\xi\partial T\).  Thus
    \[
      \boxed{\delta_\xi T
        =\xi\partial T+2(\partial\xi)T
          +\frac{c}{12}\partial^3\xi.}
    \]
    The last term is the obstruction to treating \(T\) as a primary of
    weight two when \(c\neq0\).

    \item[(b)]
    Put \(f(z)=z+\xi(z)\).  To first order,
    \[
      [f'(z)]^2T(f(z))
        =T+\xi\partial T+2(\partial\xi)T+O(\xi^2),
      \qquad
      \{f,z\}=\partial^3\xi+O(\xi^2).
    \]
    The finite law therefore reduces exactly to the result of part (a).
    Its consistency under successive maps follows from
    \[
      \boxed{\{h\circ f,z\}
        =[f'(z)]^2\{h,f(z)\}+\{f,z\}.}
      \tag{20.5.S1}
    \]
    This identity follows directly by differentiating \(h(f(z))\) three
    times and collecting the terms in the definition of the Schwarzian.
    It supplies the optional composition check.

    \item[(c)]
    It is clearest first to use the Euclidean cylinder coordinate
    \(u=-\ii w\), so \(z=e^u\).  Direct differentiation gives
    \[
      \{e^u,u\}=1-\frac32=-\frac12 .
    \]
    The finite transformation law then gives
    \[
      \boxed{T_{\rm cyl}(u)
        =z^2T_{\rm plane}(z)-\frac{c}{24}.}
      \tag{20.5.S2}
    \]
    In quadratic-differential notation this is
    \[
      \boxed{
      T_{\rm cyl}(u)(\dd u)^2
        =\left(T_{\rm plane}(z)-\frac{c}{24z^2}\right)(\dd z)^2.}
      \tag{20.5.S3}
    \]
    Equation (20.5.2) as printed is missing the \(z^{-2}\) multiplying
    \(c/24\).  Without it the two terms inside its parentheses have
    different conformal weights.  Returning to \(w\) simply uses
    \(\dd u=-\ii\,\dd w\); the stress-tensor components acquire the corresponding
    tensor factors.

    Now expand
    \[
      T_{\rm plane}(z)=\sum_{n\in\mathbb Z}L_nz^{-n-2}.
    \]
    Equation (20.5.S2) becomes
    \[
      T_{\rm cyl}(u)
        =\sum_nL_ne^{-nu}-\frac{c}{24}.
    \]
    The spatial integral keeps the zero mode.  Including the
    antiholomorphic sector gives, for a cylinder of circumference
    \(2\pi\),
    \[
      \boxed{H=L_0+\widetilde L_0-\frac{c+\bar c}{24}.}
    \]
    For circumference \(L\), dimensional analysis supplies the overall
    factor \(2\pi/L\).  The constant term is the universal cylinder
    Casimir energy: introducing the radius breaks plane scale invariance
    by selecting a length scale.
  \end{enumerate}

  \SupplementarySolution{6}{Deep-Inelastic Scattering from a Photon}
  \begin{enumerate}
    \item[(a)] For a quark of charge \(Q_qe\) and three colors,
      \[
      \begin{split}
        \partial_{\log Q}f_q^\gamma
          &=\frac{3Q_q^2\alpha}{\pi}
            [P_{e\leftarrow e}\otimes f_q^\gamma+
             P_{e\leftarrow\gamma}\otimes f_\gamma^\gamma],\\
        \partial_{\log Q}f_{\bar q}^\gamma
          &=\frac{3Q_q^2\alpha}{\pi}
            [P_{e\leftarrow e}\otimes f_{\bar q}^\gamma+
             P_{e\leftarrow\gamma}\otimes f_\gamma^\gamma],\\
        \partial_{\log Q}f_\gamma^\gamma
          &=\sum_q\frac{3Q_q^2\alpha}{\pi}
            [P_{\gamma\leftarrow e}\otimes
             (f_q^\gamma+f_{\bar q}^\gamma)
             +P_{\gamma\leftarrow\gamma}\otimes f_\gamma^\gamma].
      \end{split}
      \]
      Here \((P\otimes f)(x)=\int_x^1\dd z\,P(z)f(x/z)/z\).
      With \(f_\gamma^\gamma(x,Q_0)=\delta(1-x)\) and zero initial quark
      densities,
      \[
        f_q^\gamma=f_{\bar q}^\gamma
          =\frac{3Q_q^2\alpha}{2\pi}
            \log\frac{Q^2}{Q_0^2}[x^2+(1-x)^2],
      \]
      \[
        f_\gamma^\gamma=
        \left[1-\sum_q\frac{Q_q^2\alpha}{\pi}
          \log\frac{Q^2}{Q_0^2}\right]\delta(1-x).
      \]
    \item[(b)] For a photon target of momentum \(P\),
      \[
        W^{\mu\nu}=\ii\int \dd^4x\,e^{\ii qx}
        \bra{\gamma(P)}T\{J^\mu(x)J^\nu(0)\}\ket{\gamma(P)}
      \]
      decomposes into the two standard transverse tensors.  The
      leading-twist \(W_2\) expansion is
      \[
        W_2=\sum_q3Q_q^2\sum_{\substack{n\ge2\\n\ {\rm even}}}
        \frac{8(2q\cdot P)^{n-2}}{(Q^2)^n}A_{nq}(Q^2).
      \]
      The one-loop operator-mixing coefficients are
      \[
      \begin{split}
        a_{qq}^{(n)}&=-2Q_q^2\left[
          1+4\sum_{j=2}^n\frac1j-\frac{2}{n(n+1)}\right],\\
        a_{q\gamma}^{(n)}
          &=\frac{8Q_q^2(n^2+n+2)}{n(n+1)(n+2)},\\
        a_{\gamma q}^{(n)}
          &=\frac{4Q_q^2(n^2+n+2)}{n(n^2-1)},\qquad
        a_{\gamma\gamma}^{(n)}=0,
      \end{split}
      \]
      with color multiplicity in the quark-from-photon entry and
      \(\gamma^{(n)}=-(\alpha/4\pi)a^{(n)}\).
    \item[(c)] With \(Q_u=2/3\), \(Q_d=-1/3\), and \(n=2\),
      \[
        \boxed{\gamma^{(2)}=-\frac{\alpha}{4\pi}
        \begin{pmatrix}
          -64/27&0&32/27\\
          0&-16/27&8/27\\
          64/27&16/27&0
        \end{pmatrix}.}
      \]
      The RG equation for
      \((A_{2u},A_{2\dd},A_{2\gamma})^{\mathsf T}\) is governed by this matrix.
      Evolving the pointlike photon boundary vector and contracting the
      quark entries with \(3Q_q^2\) gives the second moment of
      \(F_2^\gamma\).
  \end{enumerate}

  \SupplementarySolution{7}{\(SU(2)\) Current Algebra from a Free Scalar}
  \begin{enumerate}
    \item[(a)]
    The derivative is unchanged by a constant shift, so \(J^3\) is
    single-valued.  For the vertex currents,
    \[
      :e^{\pm \ii(\phi+2\pi)}:
        =e^{\pm2\pi \ii}:e^{\pm \ii\phi}:
        =:e^{\pm \ii\phi}: .
    \]
    Hence all three currents are single-valued on the target circle.

    \item[(b)]
    Wick's theorem and (20.7.3) give
    \[
      \langle J^-(z)J^+(0)\rangle=z^{-1/R^2}.
    \]
    Since the exponent is \(2\Delta\),
    \[
      \Delta[J^\pm]=\frac{1}{2R^2}.
    \]
    Differentiating the logarithmic correlator shows that
    \[
      \langle J^3(z)J^3(0)\rangle
        =\frac{1}{R^2z^2},
    \]
    so \(\Delta[J^3]=1\) for every radius.  The extra vertices become
    currents when
    \[
      \boxed{R^2=\frac12.}
    \]

    \item[(c)]
    At this radius,
    \(\langle\phi(z)\phi(0)\rangle=-2\log z\).  Direct contractions using
    the definitions printed in (20.7.2) yield
    \[
      \begin{aligned}
        J^3(z)J^3(0)&\sim\frac{2}{z^2},\\
        J^3(z)J^\pm(0)&\sim
          \pm\frac{2J^\pm(0)}{z},\\
        J^+(z)J^-(0)&\sim
          \frac{1}{z^2}+\frac{J^3(0)}{z}+\cdots,\\
        J^-(z)J^+(0)&\sim
          \frac{1}{z^2}-\frac{J^3(0)}{z}+\cdots .
      \end{aligned}
      \tag{20.7.S1}
    \]
    For example,
    \[
      J^+(z)J^-(0)
       =z^{-2}:\!e^{\ii[\phi(z)-\phi(0)]}\!:
       =\frac{1}{z^2}+\frac{\ii\partial\phi(0)}{z}+\cdots .
    \]
    Equations (20.7.S1) do not obey (20.7.4) with the same \(J^3\) and
    with structure constants \(\epsilon^{abc}\).  This is the promised
    factor-of-two mismatch in the printed conventions.

    A convention-consistent Cartan--Weyl basis is
    \[
      H\equiv\frac{J^3}{2}=\frac{\ii}{2}\partial\phi,
      \qquad E^\pm\equiv J^\pm .
    \]
    Then
    \[
      \begin{aligned}
        H(z)H(0)&\sim\frac{1}{2z^2},&
        H(z)E^\pm(0)&\sim\pm\frac{E^\pm(0)}{z},\\
        E^+(z)E^-(0)&\sim
          \frac{1}{z^2}+\frac{2H(0)}{z},&
        E^-(z)E^+(0)&\sim
          \frac{1}{z^2}-\frac{2H(0)}{z}.
      \end{aligned}
      \tag{20.7.S2}
    \]
    Defining \(E^\pm=K^1\pm \ii K^2\) and \(H=K^3\), these relations become
    \[
      \boxed{K^a(z)K^b(0)\sim
        \frac{k\delta^{ab}}{2z^2}
        +\ii\epsilon^{abc}\frac{K^c(0)}{z}+\cdots,
        \qquad k=1.}
      \tag{20.7.S3}
    \]
    This is the standard \(SU(2)_1\) convention with
    \(\operatorname{tr}(t^at^b)=\delta^{ab}/2\).
    If instead the currents are rescaled to
    \(\widehat K^a=\sqrt2K^a\) so their double pole is
    \(\delta^{ab}/z^2\), the simple pole becomes
    \(\ii\sqrt2\,\epsilon^{abc}\widehat K^c/z\).
    The printed formula (20.7.4) takes the double pole from this rescaled
    convention but the simple pole from (20.7.S3), so no uniform
    rescaling gives all of its coefficients simultaneously.

    \item[(d)]
    For a general OPE
    \[
      A^a(z)A^b(w)\sim
        \frac{\kappa\delta^{ab}}{(z-w)^2}
        +if^{abc}\frac{A^c(w)}{z-w},
    \]
    radial quantization gives
    \[
      [A_m^a,A_n^b]
      =\frac{1}{(2\pi \ii)^2}
        \oint_0\dd w\,w^n\oint_wdz\,z^m A^a(z)A^b(w).
    \]
    The simple pole produces \(if^{abc}A^c_{m+n}\).  Since
    \(\mathop{\rm Res}_{z=w}z^m/(z-w)^2=mw^{m-1}\), the double pole
    produces \(\kappa m\delta^{ab}\delta_{m+n,0}\).
    Applying this mechanically to the source's printed OPE yields its
    printed algebra (20.7.5).

    In the consistent basis (20.7.S3), however, the modes obey
    \[
      \boxed{
      [K_m^a,K_n^b]
        =\ii\epsilon^{abc}K_{m+n}^c
          +\frac{k}{2}m\delta^{ab}\delta_{m+n,0},
      \qquad k=1.}
      \tag{20.7.S4}
    \]
    Equivalently,
    \[
      \begin{aligned}
        [H_m,H_n]&=\frac{m}{2}\delta_{m+n,0},\\
        [H_m,E_n^\pm]&=\pm E_{m+n}^\pm,\\
        [E_m^+,E_n^-]&=2H_{m+n}+m\delta_{m+n,0}.
      \end{aligned}
    \]
    Setting \(m=n=0\) removes every central term and leaves the ordinary
    \(SU(2)\) Lie algebra.
  \end{enumerate}

  \SupplementarySolution{8}{Constraints from Unitarity}
  Let \(\ket{h}\) be a primary of positive norm.  For every positive
  integer \(n\),
  \[
    \begin{aligned}
      \|L_{-n}\ket{h}\|^2
        &=\bra{h}L_nL_{-n}\ket{h}\\
        &=\bra{h}[L_n,L_{-n}]\ket{h}\\
        &=\left[2nh+\frac{c}{12}(n^3-n)\right]
          \braket{h}{h} .
    \end{aligned}
    \tag{20.8.S1}
  \]
  Here the term with \(L_{-n}L_n\) vanishes because \(L_n\ket{h}=0\).
  Positivity at \(n=1\) gives
  \[
    0\leq\|L_{-1}\ket{h}\|^2
      =2h\braket{h}{h},
    \qquad\Longrightarrow\qquad
    \boxed{h\geq0.}
  \]

  Now take the vacuum, for which \(h=0\) and \(L_2\ket{0}=0\).
  Equation (20.8.S1) at \(n=2\) gives
  \[
    \|L_{-2}\ket{0}\|^2
      =\bra{0}[L_2,L_{-2}]\ket{0}
      =\frac{c}{2}\braket{0}{0}.
  \]
  Positive semidefiniteness therefore implies \(c\geq0\).  The state
  \(L_{-2}\ket{0}\) is the state created by the stress tensor.  In a
  nontrivial unitary CFT it is nonzero in the physical Hilbert space, so
  positive definiteness strengthens the result to
  \[
    \boxed{c>0.}
  \]
  If \(c=0\), the stress-tensor state is null.  After quotienting null
  states, the Virasoro stress-tensor sector is trivial; under the usual
  CFT assumptions this is the trivial theory (with possible
  topological or otherwise degenerate exceptions).  Thus unitarity by
  itself first proves \(c\geq0\); the qualifier \emph{nontrivial} is what
  makes the strict inequality precise.  The antiholomorphic sector gives
  the same conclusions for \(\bar h\) and \(\bar c\).

  \SupplementarySolution{9}{Borel Singularities and Renormalon Ambiguities}
  \begin{enumerate}
    \item[(a)]
    Direct summation gives
    \[
      B(z)=K\sum_{n=0}^\infty\left(\frac za\right)^n
        =\boxed{\frac{K}{1-z/a}},
    \]
    with a pole at \(z=a\).  If \(a>0\), the pole lies on the integration
    contour \(z\in[0,\infty)\), so upper and lower contour prescriptions
    differ.  If \(a<0\), the singularity is on the negative axis and the
    positive-coupling integral is unambiguous, although the perturbative
    series still has zero radius of convergence.
    \item[(b)]
    For \(f_n\sim(-1)^nn!A^n\),
    \[
      B(z)\sim\sum_n(-Az)^n=\frac1{1+Az},
    \]
    so the leading singularity is \(z=-1/A\).  For
    \(f_n\sim n!A^n\),
    \[
      B(z)\sim\frac1{1-Az},
    \]
    with a singularity at \(z=+1/A\).  The alternating series is Borel
    summable along the positive axis; the fixed-sign series requires a
    lateral or principal-value prescription because its pole lies on the
    integration contour.
    \item[(c)]
    The two lateral contours differ by a closed contour around the pole.
    Since the residue of \(B(z)=R/(z_0-z)\) at \(z_0\) is \(-R\),
    \[
      (gF)_+-(gF)_-
        =-2\pi \ii R\,e^{-z_0/g},
    \]
    up to the orientation convention.  Thus the magnitude of the
    ambiguity in \(F\) is
    \[
      \boxed{\left|\Delta F\right|
        =\frac{2\pi|R|}{g}e^{-z_0/g}.}
    \]
    No power series in \(g\) can reproduce the exponential, which is why
    perturbation theory alone cannot select the contour.
  \end{enumerate}

  \SupplementarySolution{10}{Renormalons, Optimal Truncation, and OPE Cancellation}
  \begin{enumerate}
    \item[(a)]
    In the conventional QCD Borel variable, a singularity at \(u=p/2\)
    produces an exponential factor
    \[
      \exp\!\left[-\frac{u}{\beta_0\alpha(Q)}\right]
        =\exp\!\left[-\frac{p}{2\beta_0\alpha(Q)}\right]
        \sim\left(\frac{\Lambda}{Q}\right)^p.
    \]
    Therefore its contour ambiguity has precisely the size
    \[
      \boxed{\Delta C_{\rm pert}\sim(\Lambda/Q)^p.}
    \]
    Consistency requires an OPE operator whose matrix element supplies a
    power correction of the same order.  The Wilson-coefficient and
    matrix-element prescription dependences can then cancel.
    \item[(b)]
    Consecutive terms have ratio
    \[
      \frac{T_{n+1}}{T_n}\simeq(n+1)A\alpha.
    \]
    Their magnitude decreases until
    \[
      \boxed{n_*\simeq\frac1{A\alpha}.}
    \]
    Stirling's formula gives a smallest term, up to a power prefactor, of
    order
    \[
      |T_{n_*}|\sim
        \exp\!\left[-\frac1{A\alpha}\right].
    \]
    The nearest Borel singularity is at \(z_0=1/A\), so this is
    \(e^{-z_0/\alpha}\), the same nonperturbative scale as the lateral
    Borel ambiguity.  Adding terms beyond \(n_*\) worsens the approximation.
    \item[(c)]
    Write a physical quantity schematically as
    \[
      \mathcal A=C_0^{(\mathcal P)}(Q)
        +\frac{C_d(Q,\mu)}{Q^d}
         \langle O_d(\mu)\rangle_{\mathcal P}+\cdots,
    \]
    where \(\mathcal P\) labels a Borel/subtraction prescription.  Changing
    it shifts the perturbative coefficient by
    \(\delta C_0\sim(\Lambda/Q)^d\).  The separation between short and long
    distances also shifts the renormalized matrix element so that
    \[
      \boxed{\delta C_0+
        \frac{C_d}{Q^d}\delta\langle O_d\rangle=0.}
    \]
    Only the sum is physical.  A perturbative coefficient or condensate
    quoted separately is therefore incomplete unless its common
    prescription is specified.
  \end{enumerate}

\end{enumerate}
