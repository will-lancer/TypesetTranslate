  \SupplementaryExercise{1}{A Free-Field Operator-Product Expansion}
  {Adapted from Hong Liu, MIT 8.324 Assignment 6, Problem 3 (2010).}
  {mit-8324-f10-ps6-ope-rg-v2ch20-p01}
  Consider a real scalar field in four-dimensional Euclidean space,
  \[
    S_E=\frac12\int d^4x\,
      \left[(\partial_\mu\phi)(\partial_\mu\phi)+m^2\phi^2\right],
    \qquad
    D(x)=\langle\phi(x)\phi(0)\rangle_0,
  \]
  and understand \(D(0)\) in the same regulator used to define composite
  operators.
  \begin{enumerate}
    \item[(a)]
    Express \(\phi^4(x)\) completely in terms of normal-ordered local
    operators.
    \item[(b)]
    Let \(O(x)=:\!\phi^4(x)\!:\).  As \(|x|\to0\), expand
    \(O(x)O(0)\) in a complete basis of normal-ordered local operators
    built from \(\phi\) and its derivatives.  Determine every coefficient
    multiplying an operator of canonical dimension at most four, including
    derivative descendants and tensor operators.
  \end{enumerate}

  \SupplementaryExercise{2}{An \(O(N)\) OPE and the Wilson--Fisher Exponent}
  {Adapted from John McGreevy, Physics 215C, Homework 5, Problem 2 (2025).}
  {mcgreevy-215c-s25-hw5-q2-v2ch20-p02}
  \begin{enumerate}
    \item[(a)]
    For \(N\) free real scalars with
    \(\langle\phi_i(x)\phi_j(0)\rangle=\delta_{ij}G(x)\), let
    \(E=:\!\phi_i\phi_i\!:\).  Use Wick's theorem to determine the
    coefficients of \(\mathbf{1}\) and \(E\) in \(E(x)E(0)\).
    \item[(b)]
    Perturb the theory in \(d=4-\epsilon\) by
    \(\lambda(\phi_i\phi_i)^2/4!\) and define
    \(u=\lambda/(8\pi^2)\).  Use the one-loop OPE contractions to obtain
    \[
      \beta(u)=-\epsilon u+\frac{N+8}{6}u^2+O(u^3)
    \]
    and find the nontrivial fixed point.
    \item[(c)]
    The thermal perturbation \(tE/2\) has eigenvalue
    \(y_t=2-(N+2)u/6+O(u^2)\).  Evaluate it at the fixed point and derive
    the correlation-length exponent \(\nu=1/y_t\) through
    \(O(\epsilon)\).
    \item[(d)]
    Check the \(N=1\) result and explain how the free OPE data control both
    operator mixing and the leading interacting critical exponent.
  \end{enumerate}

  \SupplementaryExercise{3}{Scheme Changes and Dimensional Transmutation}
  {Adapted from Hong Liu, MIT 8.324 Assignment 6, Problem 4 (2010).}
  {mit-8324-f10-ps6-ope-rg-v2ch20-p03}
  \begin{enumerate}
    \item[(a)] Treat the beta function as a flow on coupling space.  Under
      an invertible redefinition \(\bar\lambda(\lambda)\), derive its exact
      transformation law and identify its tensor character.
    \item[(b)] Let
      \[
        \beta(\lambda)=b_2\lambda^2+b_3\lambda^3+b_4\lambda^4+\cdots,
        \qquad
        \bar\lambda=\lambda+a_2\lambda^2+a_3\lambda^3+\cdots .
      \]
      Show that \(\bar b_2=b_2\) and \(\bar b_3=b_3\), exhibit the freedom
      to choose \(\bar b_4\), prove that fixed points map to fixed points,
      and compare \(\beta'\) and \(\bar\beta'\) at a fixed point.
    \item[(c)] For
      \[
        \mu\frac{dg}{d\mu}=-bg^2-cg^3-dg^4-\cdots ,
      \]
      derive
      \[
        \log\frac{\mu}{\Lambda}
          =\frac1{bg(\mu)}+\frac{c}{b^2}\log[bg(\mu)]+O(g),
      \]
      and prove that \(\Lambda\) is RG invariant.
    \item[(d)] In a renormalizable theory whose only parameter is one
      dimensionless running coupling, show generally that
      \(g(\mu)=f(\log(\mu/\Lambda))\).  Explain dimensional transmutation.
  \end{enumerate}

  \SupplementaryExercise{4}{Correlators of Composite Operators Made of Free Bosons}
  {Adapted from John McGreevy, Physics 215C QFT, Spring 2025,
  Assignment 9, Problem 2 (printed pp.~1--3).}
  {mcgreevy-215c-s25-hw9-problem2-v2ch20-p04}
  Consider \(n\) free bosons \(X^\mu\) in two Euclidean dimensions, with
  \[
    S[X]=\frac{1}{4\pi g}\int d^2\sigma\,
      \partial_aX_\mu\,\partial^aX^\mu ,
    \qquad z=\sigma^1+i\sigma^2 .
  \]
  Until part (d), take the target to be \(\mathbb R^n\).  For a
  noncompact target the constant \(g\) can be absorbed into \(X^\mu\),
  but retaining it makes normalization conventions visible; in part (d)
  it becomes physical.  Use
  \[
    \nabla^2\log|z-w|=2\pi\delta^{(2)}(z-w),\qquad
    G(z,w)\equiv\langle X^\mu(z)X^\nu(w)\rangle
      =-g\,\delta^{\mu\nu}\log\frac{|z-w|}{\ell}
  \]
  on the subspace orthogonal to the constant mode.
  \begin{enumerate}
    \item[(a)]
    Compute the normalized Euclidean generating functional
    \[
      \begin{aligned}
        Z[J]&\equiv
        \left\langle
          \exp\left\{\int d^2\sigma\,J^\mu X_\mu\right\}
        \right\rangle\\
        &=Z_0^{-1}\int[dX]\,e^{-S[X]}\\
        &\qquad\times
          \exp\left\{\int d^2\sigma\,J^\mu X_\mu\right\},\\
        Z_0&\equiv\int[dX]\,e^{-S[X]} .
      \end{aligned}
    \]
    The source parenthetical prints
    \(Z_0^{-1}\equiv Z[J=0]\) and asks the reader not to worry about the
    path-integral normalization.  With the explicit definition above,
    \(Z_0\) is the unnormalized vacuum integral and the normalized
    functional obeys \(Z[0]=1\); this states the intended normalization
    without hiding the printed inverse.
    The Euclidean kinetic operator has a zero-momentum kernel.  Separate
    that mode and do its integral explicitly; state precisely what
    neutrality condition is needed for the remaining Gaussian formula.
    You may instead derive the result with Wick's theorem.

    This theory describes the propagation of featureless strings in flat
    target space \(\mathbb R^n\): \(X^\mu(\sigma)\) gives the target-space
    position assigned to the world-sheet point \(\sigma\).  It is also a
    conformal field theory, and a massless scalar in two dimensions has
    engineering dimension zero.

    \item[(b)]
    For arbitrary target-space vectors \(k_i^\mu\), points
    \(\sigma^{(i)}\), and a parameter \(\alpha'\) with dimensions
    \([X^2/g]\), evaluate
    \[
      \left\langle
        \prod_{i=1}^{N}
        :\!\exp\!\left[-i\sqrt{2\alpha'}\,
          k_i\mathbin{\cdot}X(\sigma^{(i)})\right]\!:
      \right\rangle ,
      \qquad
      z_i\equiv\sigma^{(i)}_1+i\sigma^{(i)}_2 .
    \]
    Normal ordering means that Wick contractions within one pair of
    colons are omitted.  Explain the symmetry origin of the delta
    function enforcing target-space momentum conservation.

    The source prints the target formula
    \[
      \delta^{(n)}\!\left(\sum_i k_i\right)
      \prod_{i,j=1}^{N}
        |z_i-z_j|^{-\alpha' g\,k_i\cdot k_j}.
      \tag{20.4.1}
    \]
    Preserve this comparison, but check its sign and its \(i=j\) factors
    against the action, Green function, and normal-ordering prescription
    stated above; give the convention-consistent form and identify any
    printed mismatch rather than silently repairing it.

    This calculation is the central ingredient in the Veneziano
    amplitude for tree-level scattering of bosonic strings.

    \item[(c)]
    For
    \[
      \mathcal O_a(z)\equiv:\!e^{iaX(z)}\!:
    \]
    show that
    \[
      \Delta_a=\frac{ga^2}{2},\qquad
      \left\langle\mathcal O_a(z)\mathcal O_b^\dagger(0)\right\rangle
        =\delta(a-b)\frac{1}{|z|^{2\Delta_a}},
    \]
    up to the conventional normalization of the noncompact zero-mode
    measure.  Notice that these correlators do not describe particle
    propagation: \(\mathcal O_a\) creates a power-law excitation of the
    CFT soup.

    \item[(d)]
    Now take one field with compact target
    \[
      X\simeq X+2\pi R .
    \]
    Which \(a\) label single-valued operators
    \(:\!e^{iaX}\!:\)?  How is the \(N\)-vertex result in part (b)
    modified?
  \end{enumerate}

  \SupplementaryExercise{5}{The Stress Tensor Is Not a Conformal Primary}
  {Adapted from John McGreevy, Physics 215C QFT, Spring 2025,
  Assignment 9, Problem 3 (printed p.~3).}
  {mcgreevy-215c-s25-hw9-problem3-v2ch20-p05}
  In a two-dimensional CFT, take the holomorphic stress-tensor OPE and
  contour convention to be
  \[
  \begin{split}
    T(z)T(w)\sim{}&
      \frac{c/2}{(z-w)^4}
      +\frac{2T(w)}{(z-w)^2}\\
      &+\frac{\partial T(w)}{z-w}+\cdots ,
  \end{split}
  \]
  \[
    \delta_\xi T(w)=\frac{1}{2\pi i}\oint_w dz\,
      \xi(z)T(z)T(w).
  \]
  \begin{enumerate}
    \item[(a)]
    Show that under the infinitesimal conformal transformation
    \(z\mapsto z+\xi(z)\),
    \[
      \delta_\xi T(w)
        =(\xi\partial+2\partial\xi)T(w)
          +\frac{c}{12}\partial^3\xi(w).
      \tag{20.5.1}
    \]

    \item[(b)]
    For a finite conformal transformation \(z\mapsto f(z)\), show that
    (20.5.1) is the infinitesimal version of
    \[
      T(z)\longmapsto
        [\partial f(z)]^2T(f(z))+\frac{c}{12}\{f,z\},
      \qquad
      \{f,z\}\equiv
        \frac{(\partial f)(\partial^3 f)
          -\tfrac32(\partial^2f)^2}{(\partial f)^2}.
    \]
    The last expression is the Schwarzian derivative.

    \emph{Optional.} Verify directly that the extra term has the correct
    composition law under
    \(z\mapsto f(z)\mapsto h(f(z))\).

    \item[(c)]
    The conformal map from the cylinder to the plane is
    \(z=e^{-iw}\).  Use part (b) to obtain the relation between the
    cylinder and plane stress tensors and then show that
    \[
      H=\int\frac{d\sigma}{2\pi}\,T_{\tau\tau}
       =L_0+\widetilde L_0-\frac{c+\bar c}{24}.
    \]
    The source prints the intermediate quadratic-differential formula
    \[
      T_{\rm cyl}(w)(dw)^2
        =\left(T_{\rm plane}(z)-\frac{c}{24}\right)(dz)^2.
      \tag{20.5.2}
    \]
    Check it dimensionally and by direct substitution.  If a factor of
    \(z\) is missing, display the corrected formula explicitly rather
    than silently altering (20.5.2).

    The answer has a simple physical interpretation.  The cylinder
    introduces a world-sheet length scale through its radius; the
    resulting \(c\)-dependent Casimir energy is physically observable in
    appropriate systems.
  \end{enumerate}

  \SupplementaryExercise{7}{\(SU(2)\) Current Algebra from a Free Scalar}
  {Adapted from John McGreevy, Physics 215C QFT, Spring 2025,
  Assignment 9, Problem 4 (printed pp.~4--5).}
  {mcgreevy-215c-s25-hw9-problem4-v2ch20-p07}
  Consider a compact free boson \(\phi\simeq\phi+2\pi\) in
  \(D=1+1\), with the source action
  \[
    S[\phi]=\frac{R^2}{8\pi}\int dx\,dt\,
      \partial_\mu\phi\,\partial^\mu\phi .
    \tag{20.7.1}
  \]
  Equation (20.7.1) is quoted with the source's real-time sign; the
  holomorphic two-point function below fixes the Euclidean OPE
  convention.  The redefinition \(\widetilde\phi=R\phi\) removes \(R\)
  from the kinetic term,
  \[
    S[\widetilde\phi]=\frac{1}{8\pi}\int dx\,dt\,
      \partial_\mu\widetilde\phi\,\partial^\mu\widetilde\phi ,
    \qquad
    \widetilde\phi\simeq\widetilde\phi+2\pi R,
  \]
  so \(R\) is the target-space radius.  Relative to the notation used in
  the source lectures, \(\pi T=R^2\); in string conventions the source
  uses \(\alpha'=2\).

  At a special \(SU(2)\) radius, new operators of dimensions \((1,0)\)
  and \((0,1)\), charged under the currents \(\partial_\pm\phi\), join
  those currents to form an \(SU(2)_L\times SU(2)_R\) chiral algebra.
  These operators involve winding sectors, in which \(\phi\) winds around
  its target circle as one traverses the spatial circle.  Work with the
  holomorphic part \(\phi(z)=\phi_R(z)\); the antiholomorphic calculation
  is identical with bars added.  Define
  \[
    J^\pm(z)\equiv:\!e^{\pm i\phi(z)}\!:,
    \qquad
    J^3(z)\equiv i\partial\phi(z).
    \tag{20.7.2}
  \]
  Normal ordering omits Wick contractions within a single pair of
  colons.
  \begin{enumerate}
    \item[(a)]
    Show that \(J^3\) and \(J^\pm\) are single-valued under
    \(\phi\mapsto\phi+2\pi\).

    \item[(b)]
    Compute the scaling dimensions of \(J^3\) and \(J^\pm\).  Recall that
    for a holomorphic operator
    \[
      \langle\mathcal O^\dagger(z)\mathcal O(0)\rangle
        \sim\frac{1}{z^{2\Delta}},
      \qquad
      \langle\phi(z)\phi(0)\rangle=-\frac{1}{R^2}\log z .
      \tag{20.7.3}
    \]
    Find the radius at which \(J^\pm\) have dimension one.

    \item[(c)]
    The source defines
    \[
      J^\pm\equiv\frac{1}{\sqrt2}(J^1\pm iJ^2)
    \]
    and asks for the current OPE
    \[
      J^a(z)J^b(0)\sim
        \frac{k\delta^{ab}}{z^2}
        +i\epsilon^{abc}\frac{J^c(0)}{z}+\cdots ,
      \qquad k=1.
      \tag{20.7.4}
    \]
    Reproduce the Wick contractions from (20.7.2)--(20.7.3) and track
    every normalization.  The printed source combines two common
    generator normalizations in (20.7.2) and (20.7.4); retain the printed
    target (20.7.4), identify the factor-of-two mismatch, and then state a
    convention-consistent level-one \(SU(2)\) OPE explicitly.

    \item[(d)]
    \emph{Bonus tedium.}  Introduce modes for each dimension-one current,
    \[
      J^a(z)=\sum_{n\in\mathbb Z}J_n^a z^{-n-1}.
    \]
    Starting from the printed OPE convention (20.7.4), show formally that
    \[
      [J_m^a,J_n^b]
        =i\epsilon^{abc}J_{m+n}^c
          +mk\delta^{ab}\delta_{m+n,0},
      \qquad k=1,
      \tag{20.7.5}
    \]
    and then give the mode algebra in the convention-consistent
    normalization found in part (c).  The \(m=0\) modes obey the ordinary
    \(SU(2)\) Lie algebra.

    \emph{Hint.} Use
    \(J_n^a=(2\pi i)^{-1}\oint dz\,z^nJ^a(z)\) and evaluate the
    radially ordered double contour.  The source points to a linked
    follow-up problem for further hints and string-theory applications.
  \end{enumerate}

  \SupplementaryExercise{8}{Constraints from Unitarity}
  {Adapted from John McGreevy, Physics 215C QFT, Spring 2025,
  Assignment 9, Problem 5 (printed p.~5).}
  {mcgreevy-215c-s25-hw9-problem5-v2ch20-p08}
  In the holomorphic sector of a unitary two-dimensional CFT, use
  \[
    [L_m,L_n]=(m-n)L_{m+n}
      +\frac{c}{12}(m^3-m)\delta_{m+n,0},
    \qquad
    L_n^\dagger=L_{-n}.
    \tag{20.8.1}
  \]
  A primary state of positive norm obeys
  \[
    L_0\ket{h}=h\ket{h},\qquad
    L_n\ket{h}=0\quad(n>0),
  \]
  and the conformal vacuum obeys \(L_n\ket{0}=0\) for \(n\geq-1\).
  Show that unitarity requires \(h\geq0\) for every primary and
  \(c>0\) for every nontrivial unitary CFT.  State carefully what
  unitarity alone says before null states are quotiented out, and explain
  the exceptional \(c=0\) case.

  \emph{Hint.} Evaluate
  \(\bra{h}[L_n,L_{-n}]\ket{h}\), first for \(n=1\), and then apply
  the same calculation to the vacuum with \(n=2\).

  \SupplementaryExercise{9}{Borel Singularities and Renormalon Ambiguities}
  {Inspired by M. Beneke, Renormalons (1999), Sections 1--2.}
  {beneke-renormalons-1999-v2ch20-p09}
  \begin{enumerate}
    \item[(a)]
    For coefficients \(f_n=K\,n!\,a^{-n}\), compute the Borel transform
    \(B(z)=\sum_nf_nz^n/n!\).  Locate its singularity and determine whether
    the positive-coupling Borel integral is ambiguous for positive and
    negative \(a\).
    \item[(b)]
    Compare series with large-order behavior
    \(f_n\sim(-1)^n n!A^n\) and \(f_n\sim n!A^n\), with \(A>0\).
    Locate their leading Borel singularities and explain which series is
    Borel summable without a contour prescription.
    \item[(c)]
    If \(B(z)=R/(z_0-z)\) with \(z_0>0\), evaluate the difference between
    contours passing above and below the pole in the Borel integral.
    Show its nonperturbative dependence on the coupling.
  \end{enumerate}

  \SupplementaryExercise{10}{Renormalons, Optimal Truncation, and OPE Cancellation}
  {Inspired by M. Beneke, Renormalons (1999), Sections 3 and 6.}
  {beneke-renormalons-1999-v2ch20-p10}
  \begin{enumerate}
    \item[(a)]
    In an asymptotically free theory,
    \(\Lambda/Q\sim\exp[-1/(2\beta_0\alpha(Q))]\).
    Show how a Borel singularity at \(u=p/2\) produces an ambiguity of
    order \((\Lambda/Q)^p\), and identify the required OPE power correction.
    \item[(b)]
    For \(f_n\sim n!A^n\), estimate the order \(n_*\) of the smallest term
    in \(\sum_nf_n\alpha^n\) at small positive \(\alpha\).  Estimate the
    irreducible remainder and relate it to the nearest Borel singularity.
    \item[(c)]
    Explain why a physical quantity cannot retain the contour ambiguity of
    a perturbative Wilson coefficient.  Show schematically how a change in
    the prescription for a dimension-\(d\) coefficient is cancelled by
    the corresponding prescription dependence of an OPE matrix element.
  \end{enumerate}
