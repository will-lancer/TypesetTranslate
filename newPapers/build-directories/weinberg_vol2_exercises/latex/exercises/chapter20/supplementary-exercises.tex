\chapterbackmatter{Supplementary Exercises}

\begin{enumerate}
  \SupplementaryExercise{1}{The Free-Scalar Short-Distance Product}
  {Adapted from Hong Liu, MIT 8.324 Assignment 6, Problem 3 (2010).}
  {mit-8324-f10-ps6-ope-rg}
  In four-dimensional Euclidean free scalar theory with
  \(\langle\phi(x)\phi(0)\rangle=1/(4\pi^2x^2)\), use Wick's theorem to
  find the OPE of \(\phi(x)\phi(0)\) through operators of dimension two,
  including the first derivative descendant.

  \SupplementaryExercise{2}{A Normal-Ordered Square OPE}
  {Adapted from Hong Liu, MIT 8.324 Assignment 6, Problem 3 (2010).}
  {mit-8324-f10-ps6-ope-rg}
  Let \(O(x)=:\!\phi^2(x)\!:\) in the same free theory.  Determine the
  coefficients of \(\mathbf1\), \(:\!\phi^2(0)\!:\), and
  \(\partial_\mu:\!\phi^2(0)\!:\) in \(O(x)O(0)\) through the first
  descendant order.

  \SupplementaryExercise{3}{Dimension-Four Operators in a Free OPE}
  {Adapted from Hong Liu, MIT 8.324 Assignment 6, Problem 3 (2010).}
  {mit-8324-f10-ps6-ope-rg}
  For \(O(x)=:\!\phi^4(x)\!:\), use cross-contractions to list all scalar
  normal-ordered operators of canonical dimension at most four that
  occur in \(O(x)O(0)\), and determine the leading coefficient of each.

  \SupplementaryExercise{4}{A Wilson Coefficient from Soft-State Matching}
  {Adapted from Hong Liu, MIT 8.324 Assignment 6, Problems 3--4 (2010).}
  {mit-8324-f10-ps6-ope-rg}
  Suppose \(A(x)B(0)\sim C_1(x)\mathbf1+C_{\phi^2}(x)[\phi^2]_R(0)\).
  Explain how vacuum and two-particle matrix elements determine both
  coefficients, and identify the subtraction needed before extracting
  \(C_{\phi^2}\).

  \SupplementaryExercise{5}{Beta Functions under a Coupling Redefinition}
  {Adapted from Hong Liu, MIT 8.324 Assignment 6, Problem 4 (2010).}
  {mit-8324-f10-ps6-ope-rg}
  Under an analytic change
  \(\bar g=g+a_2g^2+a_3g^3+\cdots\), derive the transformation of the beta
  function.  Show that the first two nonzero coefficients of
  \(\beta(g)=b_2g^2+b_3g^3+b_4g^4+\cdots\) are invariant.

  \SupplementaryExercise{6}{Dimensional Transmutation}
  {Adapted from Hong Liu, MIT 8.324 Assignment 6, Problem 4 (2010).}
  {mit-8324-f10-ps6-ope-rg}
  Solve \(\mu\,dg/d\mu=-bg^2\), with \(b>0\), and define an RG-invariant
  scale \(\Lambda\).  Express the running coupling in terms of
  \(\mu/\Lambda\) and explain why a dimensionless parameter has been
  replaced by a dimensional one.

  \SupplementaryExercise{7}{Transversality of a Current Correlator}
  {Adapted from A. V. Radyushkin, Introduction to QCD Sum Rule Approach
  (2001), Sections 3--4.}
  {radyushkin-qcd-sum-rules-2001}
  For a conserved Hermitian current \(J^\mu\), show that its vacuum
  correlator has the form
  \(\Pi^{\mu\nu}(q)=(q^\mu q^\nu-q^2\eta^{\mu\nu})\Pi(q^2)\), up to
  contact terms.  State the positivity property of its spectral density.

  \SupplementaryExercise{8}{Large-Momentum Expansion of a Dispersion Relation}
  {Adapted from A. V. Radyushkin, Introduction to QCD Sum Rule Approach
  (2001), Sections 2--3.}
  {radyushkin-qcd-sum-rules-2001}
  Starting from
  \(\Pi(Q^2)=\int_0^\infty ds\,\rho(s)/(s+Q^2)\), expand for large
  Euclidean \(Q^2\).  Relate the first three inverse powers of \(Q^2\) to
  moments of \(\rho\), and state when this termwise expansion is valid.

  \SupplementaryExercise{9}{A Finite-Energy Sum Rule}
  {Adapted from A. V. Radyushkin, Introduction to QCD Sum Rule Approach
  (2001), Sections 3--4.}
  {radyushkin-qcd-sum-rules-2001}
  Let \(\Pi(s)\) be analytic except for a cut on the positive real axis.
  Apply Cauchy's theorem to \(s^n\Pi(s)\) on a circle \(|s|=s_0\) and
  derive a finite-energy sum rule for
  \(\int_0^{s_0}ds\,s^n\rho(s)\).

  \SupplementaryExercise{10}{Borel Suppression of a Continuum}
  {Adapted from A. V. Radyushkin, Introduction to QCD Sum Rule Approach
  (2001), Sections 2--4.}
  {radyushkin-qcd-sum-rules-2001}
  Apply the QCD-sum-rule Borel operator to
  \(1/(s+Q^2)\) and show that it produces an exponential weight.
  Explain why this suppresses a high-energy continuum relative to a
  narrow low-mass resonance and removes polynomial subtraction terms.

  \SupplementaryExercise{11}{A Pole-plus-Continuum Spectral Model}
  {Adapted from A. V. Radyushkin, Introduction to QCD Sum Rule Approach
  (2001), Sections 2--4.}
  {radyushkin-qcd-sum-rules-2001}
  Take
  \(\rho(s)=f_R^2\delta(s-m_R^2)+\theta(s-s_0)\rho_{\rm pert}(s)\).
  Write its Borel-weighted sum rule and show how differentiating with
  respect to \(1/M^2\) can isolate \(m_R^2\) as a ratio of moments.

  \SupplementaryExercise{12}{Condensate Dimension and Power Suppression}
  {Adapted from A. V. Radyushkin, Introduction to QCD Sum Rule Approach
  (2001), Sections 3--5.}
  {radyushkin-qcd-sum-rules-2001}
  A correlator of two dimension-three currents has scalar invariant
  \(\Pi(Q^2)\) of dimension zero.  Determine the large-\(Q^2\) power
  multiplying a vacuum condensate of dimension \(d\), and compare the
  dimension-four gluon and dimension-six four-quark contributions.

  \SupplementaryExercise{13}{The Inclusive Hadronic Tensor}
  {Adapted from Iain Stewart, MIT 8.325 QFT III, Homework 7, Problem 3
  and course DIS material (2007).}
  {mit-8325-s07-dis}
  Construct the most general symmetric unpolarized hadronic tensor from
  \(p^\mu,q^\mu,\eta^{\mu\nu}\).  Impose current conservation and show
  that only two independent scalar structure functions remain.

  \SupplementaryExercise{14}{Bjorken Support from Parton Kinematics}
  {Adapted from Iain Stewart, MIT 8.325 QFT III, Homework 7, Problem 3
  and course DIS material (2007).}
  {mit-8325-s07-dis}
  A massless parton carries momentum \(xp\) and absorbs a spacelike
  momentum \(q\).  Use the final-state on-shell condition to show that
  \(x=x_B=-Q^2/(2p\cdot q)\), and explain why physical partons require
  \(0<x_B\leq1\).

  \SupplementaryExercise{15}{Parton-Model Structure Functions}
  {Adapted from Iain Stewart, MIT 8.325 QFT III, Homework 7, Problem 3
  and course DIS material (2007).}
  {mit-8325-s07-dis}
  For spin-\(\tfrac12\) quarks of charge \(Q_f\) with distributions
  \(q_f(x)\) and \(\bar q_f(x)\), derive the leading-order electromagnetic
  expression for \(F_2(x)\).  State the corresponding expression for
  \(F_1(x)\).

  \SupplementaryExercise{16}{The Callan--Gross Diagnostic}
  {Adapted from Iain Stewart, MIT 8.325 QFT III, Homework 7, Problem 3
  and course DIS material (2007).}
  {mit-8325-s07-dis}
  Show that scattering from free massless Dirac partons gives
  \(F_L\equiv F_2-2xF_1=0\).  Explain how a nonzero leading-order
  longitudinal structure function would diagnose constituents whose
  spin or interactions differ from this idealization.

  \SupplementaryExercise{17}{Even Spins in the Forward OPE}
  {Adapted from Iain Stewart, MIT 8.325 QFT III, Homework 7, Problem 3
  and course DIS material (2007).}
  {mit-8325-s07-dis}
  Expand the forward Compton amplitude in powers of
  \(\omega=2p\cdot q/Q^2\).  Use crossing symmetry to show why only even
  powers occur for electromagnetic DIS, and identify the corresponding
  spins of local OPE operators.

  \SupplementaryExercise{18}{Moments and Twist}
  {Adapted from Iain Stewart, MIT 8.325 QFT III, Homework 7, Problem 3
  and course DIS material (2007).}
  {mit-8325-s07-dis}
  A symmetric traceless spin-\(s\) operator has dimension \(d\) and twist
  \(\tau=d-s\).  By dimensional analysis show that its DIS contribution
  is suppressed relative to twist two by
  \(Q^{-(\tau-2)}\), apart from logarithms.

  \SupplementaryExercise{19}{Valence-Number Sum Rules}
  {Adapted from the Particle Data Group, Structure Functions review
  (2025), Sections 18.1--18.2.}
  {pdg-structure-functions-2025}
  Define \(q_v(x)=q(x)-\bar q(x)\).  State and justify the proton sum
  rules for the integrals of \(u_v,d_v\), and all other light-flavor
  valence distributions.  Identify the conserved local operators behind
  these relations.

  \SupplementaryExercise{20}{The Momentum Sum Rule}
  {Adapted from the Particle Data Group, Structure Functions review
  (2025), Sections 18.1--18.3.}
  {pdg-structure-functions-2025}
  Write the quark, antiquark, and gluon momentum sum rule at a
  factorization scale \(\mu\).  Explain why QCD evolution changes the
  individual terms but cannot change their sum.

  \SupplementaryExercise{21}{Testing a Plus Distribution}
  {Adapted from the Particle Data Group, Structure Functions review
  (2025), Section 18.2.}
  {pdg-structure-functions-2025}
  Using
  \(\int_0^1dx\,f(x)/(1-x)_+
    =\int_0^1dx\,[f(x)-f(1)]/(1-x)\),
  evaluate the zeroth and first moments of
  \((1+x^2)/(1-x)_+\).  Explain the role of an accompanying
  \(\delta(1-x)\) term.

  \SupplementaryExercise{22}{Mellin Moments Diagonalize a Convolution}
  {Adapted from the Particle Data Group, Structure Functions review
  (2025), Section 18.2.}
  {pdg-structure-functions-2025}
  For
  \((P\otimes f)(x)=\int_x^1dy\,P(x/y)f(y)/y\), prove that its
  \(n\)th Mellin moment equals the product of the \(n\)th moments of
  \(P\) and \(f\).  State the resulting simplification of DGLAP evolution.

  \SupplementaryExercise{23}{Conservation Constraints on Splitting Kernels}
  {Adapted from the Particle Data Group, Structure Functions review
  (2025), Section 18.2.}
  {pdg-structure-functions-2025}
  Derive the Mellin-moment constraints on nonsinglet and singlet splitting
  kernels that preserve valence number and total momentum.  Explain why
  these constraints require virtual \(\delta(1-x)\) contributions.

  \SupplementaryExercise{24}{Recognizing Higher Twist}
  {Adapted from the Particle Data Group, Structure Functions review
  (2025), Sections 18.2--18.3.}
  {pdg-structure-functions-2025}
  Suppose a structure-function moment is fitted by
  \(M(Q^2)=M_{\rm LT}(Q^2)+H/Q^2+K/Q^4\).
  Assign the twists of the two power corrections, and explain why they
  become most visible at modest \(Q^2\) or near \(x=1\).

  \SupplementaryExercise{25}{A Factorial Series and Its Borel Pole}
  {Adapted from M. Beneke, Renormalons (1999), Sections 1--2.}
  {beneke-renormalons-1999}
  For coefficients \(f_n=K\,n!\,a^{-n}\), compute the Borel transform
  \(B(z)=\sum_nf_nz^n/n!\).  Locate its singularity and determine whether
  the positive-coupling Borel integral is ambiguous for positive and
  negative \(a\).

  \SupplementaryExercise{26}{Alternating and Fixed-Sign Divergence}
  {Adapted from M. Beneke, Renormalons (1999), Sections 1--2.}
  {beneke-renormalons-1999}
  Compare series with large-order behavior
  \(f_n\sim(-1)^n n!A^n\) and \(f_n\sim n!A^n\), with \(A>0\).
  Locate their leading Borel singularities and explain which series is
  Borel summable without a contour prescription.

  \SupplementaryExercise{27}{An Ambiguity from a Positive-Axis Pole}
  {Adapted from M. Beneke, Renormalons (1999), Sections 2--3.}
  {beneke-renormalons-1999}
  If \(B(z)=R/(z_0-z)\) with \(z_0>0\), evaluate the difference between
  contours passing above and below the pole in the Borel integral.
  Show its nonperturbative dependence on the coupling.

  \SupplementaryExercise{28}{Renormalon Position and OPE Power}
  {Adapted from M. Beneke, Renormalons (1999), Sections 3 and 6.}
  {beneke-renormalons-1999}
  In an asymptotically free theory,
  \(\Lambda/Q\sim\exp[-1/(2\beta_0\alpha(Q))]\).
  Show how a Borel singularity at \(u=p/2\) produces an ambiguity of
  order \((\Lambda/Q)^p\), and identify the required OPE power correction.

  \SupplementaryExercise{29}{Optimal Truncation of an Asymptotic Series}
  {Adapted from M. Beneke, Renormalons (1999), Sections 1--2.}
  {beneke-renormalons-1999}
  For \(f_n\sim n!A^n\), estimate the order \(n_*\) of the smallest term
  in \(\sum_nf_n\alpha^n\) at small positive \(\alpha\).  Estimate the
  irreducible remainder and relate it to the nearest Borel singularity.

  \SupplementaryExercise{30}{Cancellation of a Renormalon Ambiguity}
  {Adapted from M. Beneke, Renormalons (1999), Sections 3 and 6.}
  {beneke-renormalons-1999}
  Explain why a physical quantity cannot retain the contour ambiguity of
  a perturbative Wilson coefficient.  Show schematically how a change in
  the prescription for a dimension-\(d\) coefficient is cancelled by
  the corresponding prescription dependence of an OPE matrix element.
\end{enumerate}
