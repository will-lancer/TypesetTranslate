  \SupplementaryExercise{1}{The Zinn--Justin Identity and Linearized Cohomology}
  {Inspired by Daniel H.\ T.\ Franco and Olivier Piguet, Algebraic Renormalization (2013).}
  {franco-piguet-algebraic-renormalization-2013-v2ch17-p01}
  \begin{enumerate}
    \item[(a)]
    Couple an external field \(K_n\) to every nonlinear BRST variation
    \(\Delta^n=s\chi^n\), and define
    \(\Gamma[\chi,K]=W[J,K]-\int d^4x\,\chi^nJ_n\).
    Perform a BRST change of variables in the functional integral and derive
    \[
      \int d^4x\,
        \frac{\delta_R\Gamma}{\delta K_n}
        \frac{\delta_L\Gamma}{\delta\chi^n}=0 .
    \]
    Identify the assumptions about the measure and boundary terms.
    \item[(b)]
    If a field \(\chi^n\) has dimension \(d_n\), ghost number \(\gamma_n\),
    and parity \(\epsilon_n\), determine the dimension, ghost number, and
    parity of the source \(K_n\) coupled to \(s\chi^n\).  Apply the result to
    \(A_\mu,\omega,\omega^*,h,\psi\), and a real scalar \(\phi\).  Use the
    table to explain why no source is needed for \(h\).
    \item[(c)]
    For a bosonic functional \(S\) obeying \((S,S)=0\), define
    \(\mathcal B_SX=(S,X)\).  Use the graded Jacobi identity to prove
    \(\mathcal B_S^2=0\).  Show that a local counterterm \(\Sigma\) preserves
    the master equation through first order in an infinitesimal parameter
    precisely when \(\mathcal B_S\Sigma=0\).
  \end{enumerate}

  \SupplementaryExercise{2}{Loopwise Master Equations and Trivial Counterterms}
  {Inspired by Daniel H.\ T.\ Franco and Olivier Piguet, Algebraic Renormalization (2013).}
  {franco-piguet-algebraic-renormalization-2013-v2ch17-p02}
  \begin{enumerate}
    \item[(a)]
    Expand \(\Gamma=\sum_{N\geq0}\Gamma_N\), with
    \(\Gamma_0=S_R\), in the Zinn--Justin equation.  Derive the equation at
    loop order \(N\).  Assuming that all subdivergences below this order have
    been removed, prove that the remaining local divergence satisfies
    \((S_R,\Gamma_{N,\infty})=0\).
    \item[(b)]
    Suppose a divergent local functional has the form
    \(\Sigma=\mathcal B_SF=(S,F)\), where \(F\) has ghost number \(-1\).
    Show that it is generated by an infinitesimal anticanonical
    transformation of fields and sources.  Explain why such a divergence
    renormalizes field coordinates or the gauge-fixing fermion rather than a
    physical coupling.
    \item[(c)]
    In Lorenz gauge the antighost occurs only through
    \(\partial_\mu\omega_\alpha^*\), apart from its prescribed source
    coupling to \(h_\alpha\).  Derive the Ward identity associated with
    \(\omega_\alpha^*(x)\mapsto\omega_\alpha^*(x)+c_\alpha\).  Use it,
    power counting, and ghost number to exclude a divergent term quadratic
    in the source \(K_\alpha^*\) coupled to \(s\omega_\alpha^*\).
  \end{enumerate}

  \SupplementaryExercise{3}{Scalar BRST Cohomology and Power Counting}
  {Inspired by Andrei O.\ Barvinsky et al., Renormalization of Gauge Theories in the Background-Field Approach, Sections 2--3 (2017).}
  {barvinsky-et-al-background-renormalization-2017-v2ch17-p03}
  \begin{enumerate}
    \item[(a)]
    Consider
    \(f_\alpha=\partial_\mu A_\alpha^\mu+
    a_{\alpha\beta\gamma}A_\beta^\mu A_{\gamma\mu}\).
    Explain why antighost translations are lost.  Verify that both
    \(s(\omega_\alpha^*A_{\beta\mu}A_\gamma^\mu)\) and
    \(s(\omega_\alpha^*\omega_\beta^*\omega_\gamma)\) are local,
    dimension-four, ghost-number-zero interactions.  Why must the second
    structure be admitted if the gauge is to be closed under
    renormalization?
    \item[(b)]
    Let \(s\phi=i\omega^\alpha T_\alpha\phi\) and
    \(s\omega_\alpha=-C_{\alpha\beta\gamma}
    \omega_\beta\omega_\gamma/2\).  Calculate \(s^2\phi\), keeping the
    Grassmann signs explicit, and prove that it vanishes if and only if the
    \(T_\alpha\) furnish a representation of the gauge algebra.
    \item[(c)]
    Add a dimension-one scalar to the field content of Section 17.2 and a
    source \(K_{\phi}\) for its BRST variation.  List every possible
    dimension-four monomial containing at least two \(K_\phi\)'s before
    imposing ghost number.  Show that none has ghost number zero.  Repeat
    the argument for mixed products of \(K_\phi\) with the gauge, ghost, and
    fermion BRST sources.
  \end{enumerate}

  \SupplementaryExercise{4}{Slavnov Stability under Renormalization}
  {Inspired by Andrei O.\ Barvinsky et al., Renormalization of Gauge Theories in the Background-Field Approach, Sections 4--5 (2017).}
  {barvinsky-et-al-background-renormalization-2017-v2ch17-p04}
  Suppose the regularized \(N\)-loop effective action violates the
  Zinn--Justin equation by a local ghost-number-one functional
  \(\mathcal A_N\).  Use the Jacobi identity to derive the consistency
  condition \(\mathcal B_{S_R}\mathcal A_N=0\).  Explain when a local
  counterterm restores the identity and identify the cohomological
  obstruction.

  \SupplementaryExercise{5}{Invariant Scalar and Yukawa Counterterms}
  {Inspired by Olivier Piguet and Silvio Sorella, Algebraic Renormalization, sections on Yang--Mills theories and invariant interactions (1995).}
  {piguet-sorella-algebraic-renormalization-1995-v2ch17-p05}
  \begin{enumerate}
    \item[(a)]
    For real scalars in a representation \(T_\alpha\), write the most
    general renormalizable potential.  Derive the infinitesimal
    gauge-invariance conditions on its tadpole, mass, cubic, and quartic
    tensors.  Show directly that these conditions are preserved under
    changes of scalar basis and therefore describe the complete physical
    scalar counterterm space.
    \item[(b)]
    Let left- and right-handed fermions transform with generators
    \(t_\alpha^L,t_\alpha^R\), and let real scalars transform with
    \(T_\alpha\).  For
    \(-\bar\psi_LY^i\phi_i\psi_R+\mathrm{H.c.}\), derive the linear identity
    obeyed by \(Y^i\) under an infinitesimal gauge transformation.  Check
    that any divergent Yukawa vertex satisfying the Slavnov identity has
    the same tensor form.
  \end{enumerate}

  \SupplementaryExercise{6}{Background Splitting, Gauge Fixing, and Determinants}
  {Inspired by David Skinner, Quantum Field Theory II, Chapter 8 (2018).}
  {skinner-qft2-ch8-2018-v2ch17-p06}
  \begin{enumerate}
    \item[(a)]
    Split \(\mathcal A_\mu=A_\mu+A'_\mu\).  Write a background
    transformation for which \(A_\mu\) transforms as a connection and
    \(A'_\mu\) homogeneously, and a quantum transformation for which \(A_\mu\)
    is fixed and the sum \(\mathcal A_\mu\) transforms as a connection.
    Verify explicitly that both transformations have the same action on
    \(\mathcal A_\mu\).
    \item[(b)]
    With \(f_\alpha=\bar D^\mu A'_{\alpha\mu}\), calculate its variation
    under the background transformation defined in part~(a).  Prove
    that \(f_\alpha\) transforms in the adjoint representation and that
    \(f_\alpha f_\alpha\) is invariant.
    \item[(c)]
    Vary \(f_\alpha=\bar D^\mu A'_{\alpha\mu}\) under a quantum gauge
    transformation, replace the parameter by \(\omega_\alpha\), and derive
    the background-gauge ghost operator.  Write the ghost Lagrangian both
    before and after a background-covariant integration by parts.
    \item[(d)]
    Take
    \[
      \Psi=-\int d^4x\,\omega_\alpha^*
        \left(\bar D^\mu A'_{\alpha\mu}+\frac{\xi}{2}h_\alpha\right).
    \]
    Apply the quantum BRST transformation, eliminate \(h_\alpha\), and
    recover the background gauge-fixing and ghost terms.  Check off-shell
    nilpotence and background gauge invariance.
    \item[(e)]
    Assuming an invariant functional measure, perform an infinitesimal
    background transformation in the functional integral.  Derive the Ward
    identity obeyed by \(\Gamma[A]\).  Differentiate it twice about
    \(A=0\) and show that the background two-point vertex is transverse.
    \item[(f)]
    Evaluate finite-dimensional Gaussian integrals to determine the powers
    of determinants produced by a real boson, a complex Grassmann field,
    and a Faddeev--Popov ghost pair.  Apply the result to the one-loop
    background-field functional and explain the relative signs of gauge,
    fermion, and ghost traces.
  \end{enumerate}

  \SupplementaryExercise{7}{Background Transversality and Coupling Renormalization}
  {Inspired by Laurence F.\ Abbott, The Background Field Method Beyond One Loop, Sections 2--3 (1981).}
  {abbott-background-field-1981-v2ch17-p07}
  \begin{enumerate}
    \item[(a)]
    Suppose the divergent pure-background action for a simple group is
    \(-\tfrac14L_AF^2\).  Rescale the background to make its kinetic term
    canonical.  Prove that the cubic and quartic parts of the same \(F^2\)
    operator force \(g^R=g(1+L_A)^{-1/2}\), or
    \(Z_gZ_A^{1/2}=1\).
    \item[(b)]
    Let
    \(\Gamma_{\mu\nu}^{\alpha\beta}(q)\) be the full 1PI background
    two-point function.  Use the background Ward identity and Lorentz
    covariance to determine its tensor form.  Explain why this statement
    survives subdivergence subtraction and why it does not imply that every
    individual quantum-gauge or ghost diagram is transverse.
  \end{enumerate}

  \SupplementaryExercise{8}{Group Theory and the Gauge Beta Function}
  {Inspired by David Tong, Gauge Theory, Part 2, Section 2.4 (2018).}
  {tong-gauge-theory-2018-v2ch17-p08}
  \begin{enumerate}
    \item[(a)]
    Let a quadratic operator in a constant background be
    \(M=M_0+M_1+M_2\), where \(M_j\) contains \(j\) powers of the background.
    Expand \(\operatorname{Tr}\ln M\) through fourth order and show that its
    quartic part is
    \[
      \operatorname{Tr}\!\left[
        -\frac12(M_0^{-1}M_2)^2
        +(M_0^{-1}M_1)^2M_0^{-1}M_2
        -\frac14(M_0^{-1}M_1)^4\right],
    \]
    using cyclicity to combine equivalent orderings.
    \item[(b)]
    Starting from
    \(C_{\gamma\alpha\beta}C_{\delta\alpha\beta}
      =g^2C_1\delta_{\gamma\delta}\) and
    \({\rm Tr}\{t_\gamma,t_\delta\}
      =g^2C_2\delta_{\gamma\delta}\),
    relate \(C_1,C_2\) to the conventional adjoint Casimir and matter Dynkin
    index.  For \(SU(N)\), verify \(C_1=N\) and
    \(C_2=n_f/2\) for \(n_f\) Dirac fundamentals.
    \item[(c)]
    Consider Dirac fermions and complex scalars in representations \(r\) of
    a simple group, with
    \({\rm tr}_r(T^aT^b)=T(r)\delta^{ab}\).  Show that one-loop vacuum
    polarizations add over species and express the total matter index as a
    sum of \(T(r)\).  Evaluate the sums for \(SU(N)\) matter in the
    fundamental and adjoint representations.
    \item[(d)]
    Expand the background ghost operator
    \(\Delta_0=-\bar D^2\) through second order in \(A_\mu\).  Evaluate the
    bubble and seagull contributions to
    \(\operatorname{Tr}\ln\Delta_0\), show that their logarithmic part is
    transverse, and then include the closed-Grassmann-loop sign with which
    the ghost determinant enters the one-loop effective action.
    \item[(e)]
    In background Feynman gauge use the quadratic operators
    \[
      (\Delta_1)_{\mu\nu}
        =-\bar D^2\eta_{\mu\nu}-2\,{\rm ad}(F_{\mu\nu}),
      \qquad \Delta_0=-\bar D^2 .
    \]
    Expand
    \(\tfrac12\operatorname{Tr}\ln\Delta_1
      -\operatorname{Tr}\ln\Delta_0\) through \(A^2\).  Separate the
    covariant-Laplacian and spin-coupling terms and derive the combined
    one-loop coefficient \(11C_A/3\).
    \item[(f)]
    Assemble the gauge, Dirac-fermion, complex-scalar, and real-scalar
    contributions for a simple gauge group.  Write \(b_0\) in terms of
    \(C_A\) and representation indices.  Specialize to \(SU(N)\) with
    \(n_f\) Dirac fundamentals, \(n_s\) complex fundamental scalars, and
    \(n_{\rm adj}\) real adjoint scalars.
    \item[(g)]
    For the \(SU(N)\) matter content specified in part~(f), determine
    the inequality for \(b_0>0\).  Find the largest integer \(n_f\) allowed
    when \(n_s=n_{\rm adj}=0\), and discuss separately the case in which
    \(11N/2\) is an integer.  Interpret the result in terms of screening and
    antiscreening.
  \end{enumerate}

  \SupplementaryExercise{9}{A Complex Scalar in the Gauge Beta Function}
  {Adapted from Daniel Harlow, \emph{Relativistic Quantum Field Theory III},
  Section 7.8, Problem 1 (2026).}
  {harlow-qft3-2026-sec7p8-problem1-v2ch17-p09}
  Let \(G\) be a compact simple gauge group and let a massless complex
  scalar \(\Phi\) transform in a representation \(r\), with Hermitian
  generators \(T^a\) normalized by
  \(\operatorname{tr}_r(T^aT^b)=T(r)\delta^{ab}\).  Use canonically
  normalized gauge fields and
  \[
    \mathcal L_\Phi=-(D_\mu\Phi)^\dagger D^\mu\Phi,\qquad
    D_\mu=\partial_\mu-igA_\mu^aT^a .
  \]
  Work in dimensional regularization with \(d=4-2\epsilon\) and minimal
  subtraction.  Derive the scalar--scalar--gauge and
  scalar--scalar--gauge--gauge vertices, compute both one-loop scalar
  contributions to the gauge two-point function, and extract the scalar
  contribution to \(Z_3\) and to the one-loop beta function.

  \emph{Hint.}  Continue to define the gauge coupling through a fermionic
  gauge vertex,
  \[
    g=\frac{Z_2\sqrt{Z_3}}{Z_1}\,g_0.
  \]
  The scalar changes \(Z_3\) but not the previously computed \(Z_1\) and
  \(Z_2\).  In the massless theory the seagull integral is scaleless in
  dimensional regularization.

  \SupplementaryExercise{10}{From Counterterms to Asymptotic Freedom}
  {Inspired by David J.\ Gross and Frank Wilczek, Ultraviolet Behavior of Non-Abelian Gauge Theories (1973).}
  {gross-wilczek-asymptotic-freedom-1973-v2ch17-p10}
  Starting from
  \(g_R=g[1+g^2b_0\ln(\Lambda/\mu)/(16\pi^2)+O(g^4)]\), hold the bare
  coupling and cutoff fixed and derive the one-loop beta function.
  Integrate it between two scales.  Show that \(b_0>0\) gives logarithmic
  weakening at high energy and determine the associated
  renormalization-group invariant scale.

  \SupplementaryExercise{11}{Schwinger--Dyson, Ward--Takahashi, and QED Renormalization}
  {Adapted from Cambridge Part III, Advanced Quantum Field Theory, 2022 Paper 304, Question 3.}
  {cambridge-part-iii-aqft-2022-p304-q3-v2ch17-p11}
  \begin{enumerate}
    \item[(a)]
    For QED with
    \(\mathcal L=-F_{\mu\nu}F^{\mu\nu}/4
      -\overline\psi(\gamma^\mu D_\mu+m)\psi\),
    verify the global \(U(1)\) symmetry and derive conservation of
    \(j^\mu=\overline\psi\gamma^\mu\psi\).
    \item[(b)]
    Make the corresponding infinitesimal, spacetime-dependent change of
    fermion variables in the source-free functional integral.  Derive the
    Schwinger--Dyson identity
    \[
      \partial_\mu
      \langle Tj^\mu(x)\psi(x_1)\overline\psi(x_2)\rangle
       =-\bigl[\delta^{(4)}(x-x_1)-\delta^{(4)}(x-x_2)\bigr]
        \langle T\psi(x_1)\overline\psi(x_2)\rangle,
    \]
    with the conventional continuation factors stated explicitly.
    \item[(c)]
    Fourier transform and amputate the identity in part~(b) to prove
    \[
      q_\mu\Gamma^\mu(p+q,p)
        =S^{-1}(p+q)-S^{-1}(p).
    \]
    Take \(q\to0\) and identify \(\Gamma^\mu(p,p)\).
    \item[(d)]
    Compare bare and renormalized functions to show \(Z_1=Z_2\), determine
    which renormalization constant controls the electric charge, and state
    what part of the argument remains nonperturbatively valid.
  \end{enumerate}

\end{enumerate}
