  \SupplementaryExercise{1}{The Zinn--Justin Identity and Linearized Cohomology}
  {Inspired by Daniel H.\ T.\ Franco and Olivier Piguet, Algebraic Renormalization (2013).}
  {franco-piguet-algebraic-renormalization-2013-v2ch17-p01}
  \begin{enumerate}
    \item[(a)]
    Couple an external field \(K_n\) to every nonlinear BRST variation
    \(\Delta^n=s\chi^n\), and define
    \(\Gamma[\chi,K]=W[J,K]-\int d^4x\,\chi^nJ_n\).
    Perform a BRST change of variables in the functional integral and derive
    \[
      \int d^4x\,
        \frac{\delta_R\Gamma}{\delta K_n}
        \frac{\delta_L\Gamma}{\delta\chi^n}=0 .
    \]
    Identify the assumptions about the measure and boundary terms.
    \item[(b)]
    If a field \(\chi^n\) has dimension \(d_n\), ghost number \(\gamma_n\),
    and parity \(\epsilon_n\), determine the dimension, ghost number, and
    parity of the source \(K_n\) coupled to \(s\chi^n\).  Apply the result to
    \(A_\mu,\omega,\omega^*,h,\psi\), and a real scalar \(\phi\).  Use the
    table to explain why no source is needed for \(h\).
    \item[(c)]
    For a bosonic functional \(S\) obeying \((S,S)=0\), define
    \(\mathcal B_SX=(S,X)\).  Use the graded Jacobi identity to prove
    \(\mathcal B_S^2=0\).  Show that a local counterterm \(\Sigma\) preserves
    the master equation through first order in an infinitesimal parameter
    precisely when \(\mathcal B_S\Sigma=0\).
  \end{enumerate}

  \SupplementaryExercise{2}{Loopwise Master Equations and Trivial Counterterms}
  {Inspired by Daniel H.\ T.\ Franco and Olivier Piguet, Algebraic Renormalization (2013).}
  {franco-piguet-algebraic-renormalization-2013-v2ch17-p02}
  \begin{enumerate}
    \item[(a)]
    Expand \(\Gamma=\sum_{N\geq0}\Gamma_N\), with
    \(\Gamma_0=S_R\), in the Zinn--Justin equation.  Derive the equation at
    loop order \(N\).  Assuming that all subdivergences below this order have
    been removed, prove that the remaining local divergence satisfies
    \((S_R,\Gamma_{N,\infty})=0\).
    \item[(b)]
    Suppose a divergent local functional has the form
    \(\Sigma=\mathcal B_SF=(S,F)\), where \(F\) has ghost number \(-1\).
    Show that it is generated by an infinitesimal anticanonical
    transformation of fields and sources.  Explain why such a divergence
    renormalizes field coordinates or the gauge-fixing fermion rather than a
    physical coupling.
    \item[(c)]
    In Lorenz gauge the antighost occurs only through
    \(\partial_\mu\omega_\alpha^*\), apart from its prescribed source
    coupling to \(h_\alpha\).  Derive the Ward identity associated with
    \(\omega_\alpha^*(x)\mapsto\omega_\alpha^*(x)+c_\alpha\).  Use it,
    power counting, and ghost number to exclude a divergent term quadratic
    in the source \(K_\alpha^*\) coupled to \(s\omega_\alpha^*\).
  \end{enumerate}

  \SupplementaryExercise{3}{Scalar BRST Cohomology and Power Counting}
  {Inspired by Andrei O.\ Barvinsky et al., Renormalization of Gauge Theories in the Background-Field Approach, Sections 2--3 (2017).}
  {barvinsky-et-al-background-renormalization-2017-v2ch17-p03}
  \begin{enumerate}
    \item[(a)]
    Consider
    \(f_\alpha=\partial_\mu A_\alpha^\mu+
    a_{\alpha\beta\gamma}A_\beta^\mu A_{\gamma\mu}\).
    Explain why antighost translations are lost.  Verify that both
    \(s(\omega_\alpha^*A_{\beta\mu}A_\gamma^\mu)\) and
    \(s(\omega_\alpha^*\omega_\beta^*\omega_\gamma)\) are local,
    dimension-four, ghost-number-zero interactions.  Why must the second
    structure be admitted if the gauge is to be closed under
    renormalization?
    \item[(b)]
    Let \(s\phi=i\omega^\alpha T_\alpha\phi\) and
    \(s\omega_\alpha=-C_{\alpha\beta\gamma}
    \omega_\beta\omega_\gamma/2\).  Calculate \(s^2\phi\), keeping the
    Grassmann signs explicit, and prove that it vanishes if and only if the
    \(T_\alpha\) furnish a representation of the gauge algebra.
    \item[(c)]
    Add a dimension-one scalar to the field content of Section 17.2 and a
    source \(K_{\phi}\) for its BRST variation.  List every possible
    dimension-four monomial containing at least two \(K_\phi\)'s before
    imposing ghost number.  Show that none has ghost number zero.  Repeat
    the argument for mixed products of \(K_\phi\) with the gauge, ghost, and
    fermion BRST sources.
  \end{enumerate}

  \SupplementaryExercise{4}{Gauge Fields with Adjoint-Scalar and Fundamental-Fermion Matter}
  {Adapted from University of Cambridge, Part III Advanced Quantum
  Field Theory, 2023 Paper 304, Question 3 (printed p.~4).}
  {cambridge-part-iii-aqft-2023-p304-q3-v2ch17-p04}
  A Yang--Mills vector field \(A_\mu\) has field strength
  \[
    F_{\mu\nu}=\partial_\mu A_\nu-\partial_\nu A_\mu
      -ig[A_\mu,A_\nu].
  \]
  \begin{enumerate}
    \item[(a)] State how \(F_{\mu\nu}\) transforms under gauge
      transformations and show that the Lagrangian
      \[
        \mathcal L=-\frac14\operatorname{Tr}
          (F_{\mu\nu}F^{\mu\nu})
      \]
      is gauge invariant.  Briefly explain why gauge fixing is necessary
      to find the propagator.
    \item[(b)] A scalar field \(\phi(x)\) in the adjoint representation
      transforms under the gauge symmetry as
      \[
        \phi\longmapsto U^\dagger\phi U.
      \]
      Show that the covariant derivative of \(\phi\) is given by
      \[
        D_\mu\phi=\partial_\mu\phi-ig[A_\mu,\phi].
      \]
      By writing
      \[
        A_\mu=A_\mu^a(x)T_a,\qquad
        \phi(x)=\phi^a(x)T_a,\qquad
        U(x)=\exp\!\bigl(i\alpha^a(x)T_a\bigr),
      \]
      where \(T_a\) are generators of the gauge group, find the
      infinitesimal form (to first order in \(\alpha^a(x)\)) of the gauge
      transformations of \(\phi^a(x)\) in terms of the fields and the
      structure constants of the gauge group.
    \item[(c)] Suppose we also have a fermion \(\psi(x)\), which transforms
      in the fundamental representation of the gauge group.
      \begin{enumerate}
        \item[(i)] Write down the general gauge-invariant Lagrangian in
          \(d\) dimensions, up to and including terms quartic in the fields
          \(\psi(x)\), \(\phi(x)\), and \(A_\mu(x)\), that is invariant
          under the transformations
          \(\phi\mapsto-\phi\) and \(\psi\mapsto-\psi\).  Clearly state
          the mass dimension of any coupling constants you introduce and
          comment on how each term is expected to evolve under
          renormalization-group flow.
        \item[(ii)] Draw the Feynman diagrams that contribute to the
          four-scalar 1PI correlation function through one loop.  You do
          not have to evaluate the diagrams or include counterterms.
      \end{enumerate}
  \end{enumerate}

  \SupplementaryExercise{6}{Background Splitting, Gauge Fixing, and Determinants}
  {Inspired by David Skinner, Quantum Field Theory II, Chapter 8 (2018).}
  {skinner-qft2-ch8-2018-v2ch17-p06}
  \begin{enumerate}
    \item[(a)]
    Split \(\mathcal A_\mu=A_\mu+A'_\mu\).  Write a background
    transformation for which \(A_\mu\) transforms as a connection and
    \(A'_\mu\) homogeneously, and a quantum transformation for which \(A_\mu\)
    is fixed and the sum \(\mathcal A_\mu\) transforms as a connection.
    Verify explicitly that both transformations have the same action on
    \(\mathcal A_\mu\).
    \item[(b)]
    With \(f_\alpha=\bar D^\mu A'_{\alpha\mu}\), calculate its variation
    under the background transformation defined in part~(a).  Prove
    that \(f_\alpha\) transforms in the adjoint representation and that
    \(f_\alpha f_\alpha\) is invariant.
    \item[(c)]
    Vary \(f_\alpha=\bar D^\mu A'_{\alpha\mu}\) under a quantum gauge
    transformation, replace the parameter by \(\omega_\alpha\), and derive
    the background-gauge ghost operator.  Write the ghost Lagrangian both
    before and after a background-covariant integration by parts.
    \item[(d)]
    Take
    \[
      \Psi=-\int d^4x\,\omega_\alpha^*
        \left(\bar D^\mu A'_{\alpha\mu}+\frac{\xi}{2}h_\alpha\right).
    \]
    Apply the quantum BRST transformation, eliminate \(h_\alpha\), and
    recover the background gauge-fixing and ghost terms.  Check off-shell
    nilpotence and background gauge invariance.
    \item[(e)]
    Assuming an invariant functional measure, perform an infinitesimal
    background transformation in the functional integral.  Derive the Ward
    identity obeyed by \(\Gamma[A]\).  Differentiate it twice about
    \(A=0\) and show that the background two-point vertex is transverse.
    \item[(f)]
    Evaluate finite-dimensional Gaussian integrals to determine the powers
    of determinants produced by a real boson, a complex Grassmann field,
    and a Faddeev--Popov ghost pair.  Apply the result to the one-loop
    background-field functional and explain the relative signs of gauge,
    fermion, and ghost traces.
  \end{enumerate}

  \SupplementaryExercise{7}{Adjoint Scalars, Gauge Covariance, and BRST Gauge Independence}
  {Adapted from Cambridge Part III, Advanced Quantum Field Theory, 2025
  Paper 304, Question 4.}
  {cambridge-part-iii-aqft-2025-p304-q4-v2ch17-p07}
  In the volume's mostly-plus convention, consider \(SU(N)\) Yang--Mills
  theory coupled to a scalar in the adjoint representation,
  \[
    \mathcal L=-\frac12\operatorname{Tr}(F_{\mu\nu}F^{\mu\nu})
      -\operatorname{Tr}(D_\mu\phi D^\mu\phi)
      -\frac1{3!}\operatorname{Tr}(\phi^3),
  \]
  where
  \[
    D_\mu\phi=\partial_\mu\phi-ig[A_\mu,\phi],\qquad
    F_{\mu\nu}=\partial_\mu A_\nu-\partial_\nu A_\mu
      -ig[A_\mu,A_\nu].
  \]
  The fields \(A_\mu\) and \(\phi\) are traceless Hermitian matrices and
  transform as
  \[
    A_\mu\longmapsto UA_\mu U^\dagger
      +\frac{i}{g}U\partial_\mu U^\dagger,\qquad
    \phi\longmapsto U\phi U^\dagger,\qquad U\in SU(N).
  \]
  The source writes the scalar kinetic term with the opposite metric
  signature; only that signature translation has been made here.
  \begin{enumerate}
    \item[(a)]
    Prove that \(D_\mu\phi\) transforms covariantly.  Show that
    \[
      [D_\mu,D_\nu]\phi=-ig[F_{\mu\nu},\phi],
    \]
    and use this result to deduce the covariant transformation law of
    \(F_{\mu\nu}\).  Explain why every term in \(\mathcal L\) is gauge
    invariant.
    \item[(b)]
    Introduce generators satisfying
    \[
      [T_a,T_b]=if_{ab}{}^cT_c,\qquad
      \operatorname{Tr}(T_aT_b)=\frac12\delta_{ab},
      \qquad
      \{T_a,T_b\}=\frac1N\delta_{ab}\mathbf 1+d_{ab}{}^cT_c,
    \]
    where \(d_{abc}=d_{ab}{}^e\delta_{ec}\) is constant and totally
    symmetric.  With \(\phi=\phi^aT_a\), express the complete cubic scalar
    interaction in terms of \(f_{ab}{}^c\), \(d_{abc}\), and the component
    fields \(\phi^a\); in particular, display why one of the two invariant
    tensors drops out.
    \item[(c)]
    Introduce a Grassmann-odd matrix ghost \(\omega\), an antighost
    \(\omega^*\), and an auxiliary field \(h\).  Let \(s\) denote the BRST
    differential, with
    \[
      s\omega=ig\omega^2,\qquad s\omega^*=-h,\qquad sh=0,
    \]
    and write the gauge-fixed action as
    \[
      S_\Psi=\int d^4x\,\mathcal L+s\Psi .
    \]
    \begin{enumerate}
      \item[(i)] Derive the BRST transformations of \(A_\mu\) and \(\phi\)
      from their finite gauge transformations.
      \item[(ii)] Explain why \(S_\Psi\) is BRST invariant.
      \item[(iii)] Let \(\Psi\mapsto\Psi+\delta\Psi\) be an infinitesimal
      change of gauge.  Assuming a BRST-invariant functional measure and
      no boundary term in field space, prove that normalized correlation
      functions of gauge-invariant operators do not depend on this choice.
    \end{enumerate}
  \end{enumerate}

  \SupplementaryExercise{8}{Fundamental and Adjoint Matter in a Gluon Self-Energy}
  {Adapted from Cambridge Part III, Advanced Quantum Field Theory, 2024
  Paper 304, Question 4.}
  {cambridge-part-iii-aqft-2024-p304-q4-v2ch17-p08}
  Consider \(SU(N)\) gauge fields \(A_\mu=A_\mu^aT_a\), coupled to
  fundamental fermions \(\psi_i\), with
  \[
    \mathcal L=-\frac14\operatorname{Tr}(F_{\mu\nu}F^{\mu\nu})
      -\overline\psi(\gamma^\mu\partial_\mu+m)\psi
      +J_a^\mu A_\mu^a .
  \]
  This is the volume's mostly-plus rewriting of the source's
  \(\overline\psi(i\slashed\partial-m)\psi\) convention.  Take
  \[
    A_\mu\longmapsto UA_\mu U^\dagger
      +\frac{i}{g}U\partial_\mu U^\dagger,\qquad
    \psi\longmapsto U\psi,\qquad
    U=\exp(i\alpha^a(x)T_a),
  \]
  Write \(F_{\mu\nu}=F_{\mu\nu}^aT_a\) and normalize the generators by
  \(\operatorname{Tr}(T_aT_b)=\delta_{ab}/2\).
  \begin{enumerate}
    \item[(a)]
    Starting only from the finite transformation of \(A_\mu\), derive the
    finite transformation of \(F_{\mu\nu}\).  Then find its infinitesimal
    component form
    \[
      \delta F_{\mu\nu}^a=f_{bc}{}^aF_{\mu\nu}^b\alpha^c,
    \]
    defining the constants \(f_{bc}{}^a\) directly from the generator
    commutator and stating any index convention used.
    \item[(b)]
    Gauge invariance requires the matrix current \(J^\mu=J_a^\mu T_a\) to
    transform as \(J^\mu\mapsto UJ^\mu U^\dagger\).  Derive this statement
    and determine \(J_a^\mu\) explicitly as a fermion bilinear in the
    conventions displayed above.
    \item[(c)]
    Replace the fundamental fermion by a matrix-valued fermion
    \(\Psi=\Psi^aT_a\) in the adjoint representation, transforming as
    \(\Psi\mapsto U\Psi U^\dagger\).  Find the infinitesimal component
    transformation of \(\Psi\), construct its covariant derivative, and
    give a gauge-invariant Yang--Mills--fermion Lagrangian.
    \item[(d)]
    Draw every connected diagram contributing to the gluon propagator
    through one loop: the tree propagator and the fermion, ghost, and
    pure-gauge one-loop graphs, including the graph containing one
    four-gluon vertex.  No loop evaluation is required.  Explain precisely
    which color factor changes when the fermion is moved from the
    fundamental to the adjoint representation, and compare the two factors
    for \(SU(N)\).
  \end{enumerate}

  \SupplementaryExercise{9}{A Complex Scalar in the Gauge Beta Function}
  {Adapted from Daniel Harlow, \emph{Relativistic Quantum Field Theory III},
  Section 7.8, Problem 1 (2026).}
  {harlow-qft3-2026-sec7p8-problem1-v2ch17-p09}
  Let \(G\) be a compact simple gauge group and let a massless complex
  scalar \(\Phi\) transform in a representation \(r\), with Hermitian
  generators \(T^a\) normalized by
  \(\operatorname{tr}_r(T^aT^b)=T(r)\delta^{ab}\).  Use canonically
  normalized gauge fields and
  \[
    \mathcal L_\Phi=-(D_\mu\Phi)^\dagger D^\mu\Phi,\qquad
    D_\mu=\partial_\mu-igA_\mu^aT^a .
  \]
  Work in dimensional regularization with \(d=4-2\epsilon\) and minimal
  subtraction.  Derive the scalar--scalar--gauge and
  scalar--scalar--gauge--gauge vertices, compute both one-loop scalar
  contributions to the gauge two-point function, and extract the scalar
  contribution to \(Z_3\) and to the one-loop beta function.

  \emph{Hint.}  Continue to define the gauge coupling through a fermionic
  gauge vertex,
  \[
    g=\frac{Z_2\sqrt{Z_3}}{Z_1}\,g_0.
  \]
  The scalar changes \(Z_3\) but not the previously computed \(Z_1\) and
  \(Z_2\).  In the massless theory the seagull integral is scaleless in
  dimensional regularization.

  \SupplementaryExercise{10}{The One-Loop Three-Gluon Counterterm}
  {Adapted from Daniel Harlow, \emph{Relativistic Quantum Field Theory III},
  Section 7.8, Problem 2, with the required setup from Sections 6 and
  7.1--7.7 (2026).}
  {harlow-qft3-2026-sec7p8-problem2-v2ch17-p10}
  Let \(G\) be compact and simple.  For fermions in a representation \(r\),
  use
  \[
    \operatorname{tr}_r(T^aT^b)=T(r)\delta^{ab},\qquad
    T^aT^a=C_2(r)\mathbf1,\qquad
    f^{acd}f^{bcd}=C_A\delta^{ab}.
  \]
  In Feynman gauge, with \(d=4-2\epsilon\), take the bare gauge-fixed
  Lagrangian in the volume's mostly-plus conventions to be
  \[
  \begin{split}
    \mathcal L_0={}&-\frac14 f_{0\,\mu\nu}^af_0^{a\,\mu\nu}
      -\overline\psi(\gamma^\mu D_\mu+m_0)\psi
      -\frac12(\partial^\mu\widehat a_\mu^a)^2
      -(\partial^\mu b^a)(D_\mu c)^a,\\
    f_{0\,\mu\nu}^a={}&\partial_\mu\widehat a_\nu^a
      -\partial_\nu\widehat a_\mu^a
      +g_0f^{abc}\widehat a_\mu^b\widehat a_\nu^c,\\
    D_\mu\psi={}&(\partial_\mu-ig_0\widehat a_\mu^aT^a)\psi,
    \qquad
    (D_\mu c)^a=\partial_\mu c^a+g_0f^{abc}\widehat a_\mu^bc^c .
  \end{split}
  \]
  All momenta at a vertex are incoming.  The rules needed below are
  \[
  \begin{gathered}
    S(p)=\frac{i(\slashed p+im_0)}{p^2+m_0^2-i0},\qquad
    \Delta_{\mu\nu}^{ab}(p)=\frac{-i\delta^{ab}\eta_{\mu\nu}}{p^2-i0},
    \qquad
    \Theta^{ab}(p)=\frac{-i\delta^{ba}}{p^2-i0},\\
    V_{\bar\psi\psi a}^{a\mu}=-ig_0T^a\gamma^\mu,\qquad
    V_{bca}^{abc,\mu}(p)=g_0f^{abc}p^\mu,
  \end{gathered}
  \]
  where \(p\) on the ghost vertex is the outgoing antighost momentum.  For
  three gluons define
  \[
    V_{\mu\nu\rho}(p,q,r)
      =(q-p)_\rho\eta_{\mu\nu}
       +(p-r)_\nu\eta_{\mu\rho}
       +(r-q)_\mu\eta_{\nu\rho},
  \]
  so the vertex is \(g_0f^{abc}V_{\mu\nu\rho}(p,q,r)\).  For ordered gluon
  legs \(a\mu\), \(b\nu\), \(c\rho\), \(d\sigma\), the four-gluon vertex is
  \[
  \begin{split}
   -ig_0^2\bigl[{}
    &f^{eab}f^{ecd}(\eta_{\mu\rho}\eta_{\nu\sigma}
                    -\eta_{\mu\sigma}\eta_{\nu\rho})\\
    +{}&f^{eac}f^{ebd}(\eta_{\mu\nu}\eta_{\rho\sigma}
                    -\eta_{\mu\sigma}\eta_{\nu\rho})\\
    +{}&f^{ead}f^{ebc}(\eta_{\mu\nu}\eta_{\rho\sigma}
                    -\eta_{\mu\rho}\eta_{\nu\sigma})\bigr].
  \end{split}
  \]

  Define renormalized fields and the coupling by
  \[
    \widetilde a_\mu^a=Z_3^{-1/2}\widehat a_\mu^a,\qquad
    \widetilde\psi=Z_2^{-1/2}\psi,\qquad
    g=\frac{Z_2\sqrt{Z_3}}{Z_1}g_0,\qquad \xi_0=Z_3\xi.
  \]
  The preceding one-loop calculation in the source gives, in minimal
  subtraction,
  \[
  \begin{split}
    \delta Z_1&=-\frac{g^2}{16\pi^2\epsilon}
       \bigl(C_2(r)+C_A\bigr),\\
    \delta Z_2&=-\frac{g^2}{16\pi^2\epsilon}C_2(r),\\
    \delta Z_3&=\frac{g^2}{16\pi^2\epsilon}
       \left(\frac53C_A-\frac43n_FT(r)\right),
  \end{split}
  \]
  where \(Z_j=1+\delta Z_j\) and \(n_F\) counts Dirac fermions.  If
  \(Z_{1,3g}\) removes the intrinsic 1PI divergence of the three-gluon
  vertex, then the bare and renormalized connected functions have the
  structure
  \[
    \langle\widehat a\widehat a\widehat a\rangle
      =\frac{Z_3^3}{Z_{1,3g}}g_0\,{\cal F}_{\rm fin},
    \qquad
    \langle\widetilde a\widetilde a\widetilde a\rangle
      =\frac{Z_3^{3/2}}{Z_{1,3g}}g_0\,{\cal F}_{\rm fin}
      =\frac{Z_3Z_1}{Z_{1,3g}Z_2}g\,{\cal F}_{\rm fin}.
  \]
  Thus finiteness predicts
  \[
    Z_{1,3g}=\frac{Z_3Z_1}{Z_2},\qquad
    \delta Z_{1,3g}=\delta Z_1+\delta Z_3-\delta Z_2.
  \]
