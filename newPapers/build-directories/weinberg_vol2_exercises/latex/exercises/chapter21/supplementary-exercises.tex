  \SupplementaryExercise{1}{Charges, Goldstone's Theorem, and the Abelian Higgs Mechanism}
  {Adapted from Cambridge Part III, The Standard Model, 2022 Paper 305, Question 3.}
  {cambridge-part-iii-standard-model-2022-p305-q3-v2ch21-p01}
  Let \(N\) scalar fields transform under a compact continuous group as
  \(\delta\phi_m=i\alpha^a(T_a)_{mn}\phi_n\), with conserved currents
  \(J^\mu_a\).
  \begin{enumerate}
    \item[(a)] Show that
      \(Q_a=\int d^3x\,J^0_a\) generates the transformation of the fields.
    \item[(b)] Define a local order parameter that distinguishes a
      symmetry-preserving vacuum from one that spontaneously breaks
      generator \(a\).
    \item[(c)] Use \([H,Q_a]=0\) to explain both why excitations related by
      a broken symmetry need not be energy-degenerate and why
      \(Q_a\ket0\), when nonzero, has the vacuum energy.
    \item[(d)] State and prove Goldstone's theorem, displaying the
      zero-mass pole created by the broken current.
    \item[(e)] Gauge a \(U(1)\) symmetry acting on one complex scalar.
      Write the most general CP-even renormalizable Lagrangian after
      canonical normalization, count its independent physical parameters,
      and give the stability and symmetry-breaking range.  Expand about a
      minimum; determine every physical mass and the cubic and quartic
      interactions.  Count degrees of freedom and explain the Higgs
      mechanism.
  \end{enumerate}

  \SupplementaryExercise{2}{Unitarity Gauge for General Higgsing}
  {Adapted from Daniel Harlow, \emph{Relativistic Quantum Field Theory III},
  Section 10.7, Problem 1 (2026).}
  {harlow-qft3-2026-sec10p7-problem1-v2ch21-p02}
  Let a compact gauge group \(G\) act on a scalar multiplet through
  \(D(g)\), with Hermitian Lie-algebra generators \(\tau_a\).  Consider
  \[
    \mathcal L=-\frac14\sum_a\frac1{g_a^2}
      F_{\mu\nu}^aF^{a\mu\nu}
      -(D_\mu\phi)^\dagger D^\mu\phi-V(\phi),
    \qquad
    D_\mu\phi=\partial_\mu\phi-i\tau_aA_\mu^a\phi ,
  \]
  and write
  \[
    \langle\phi\rangle=\frac v{\sqrt2},\qquad
    H=\{h\in G:D(h)v=v\}.
  \]
  On each gauge orbit choose \(g_\phi(x)\) to maximize
  \[
    f_\phi(g)=\operatorname{Re}\!\left[\phi^\dagger(x)D(g)v\right],
  \]
  and define the unitary-gauge fields
  \[
    \widetilde\phi=D(g_\phi^{-1})\phi,\qquad
    \widetilde A_\mu=g_\phi^{-1}A_\mu g_\phi+
      i g_\phi^{-1}\partial_\mu g_\phi .
  \]
  \begin{enumerate}
    \item[(a)]
    Derive the stationarity condition
    \[
      \operatorname{Im}(\widetilde\phi^\dagger\tau_av)=0 .
    \]
    Show also that a configuration equal to \(v\) may be represented by
    \(\widetilde\phi=v\), so the gauge choice preserves the stated vacuum.
    \item[(b)]
    Prove that \(g_\phi h\) is another maximum for every \(h(x)\in H\).
    Interpret this as a residual local \(H\) symmetry.  For an
    \(H\)-invariant decomposition
    \(\mathfrak g=\mathfrak h\oplus\mathfrak h^\perp\), determine how the
    unbroken and broken components of \(\widetilde A_\mu\) transform under
    this residual symmetry.
    \item[(c)]
    A global maximum need not be unique on \(G/H\), but perturbation theory
    only needs a unique local maximum near the vacuum.  Expand
    \[
      f_{\widetilde\phi}(e^{i\theta^aT_a})
        =\operatorname{Re}\!
        \left[\widetilde\phi^\dagger
          e^{i\theta^a\tau_a}v\right]
    \]
    to second order in \(\theta\).  Show for
    \(\widetilde\phi\) sufficiently close to \(v\) that the maximum is
    strict on \(G/H\), and that a flat direction obeys
    \(\theta^a\tau_av=0\), hence lies in \(\mathfrak h\).
    \item[(d)]
    Parameterize the remaining scalar fluctuation by
    \[
      \widetilde\phi=\frac1{\sqrt2}(v+h).
    \]
    Rewrite the Lagrangian in terms of \(h\) and \(\widetilde A_\mu\),
    using the condition from part~(a).  For canonically normalized vector
    fields \(\widehat A_\mu^a=\widetilde A_\mu^a/g_a\), prove that
    \[
      (\mu^2)_{ab}=\frac{g_ag_b}{2}\,
        v^\dagger\{\tau_a,\tau_b\}v .
    \]
    Show that this matrix is real, symmetric, and positive semidefinite,
    and characterize its null vectors in terms of \(\mathfrak h\).
    \item[(e)]
    Specialize to \(G=U(1)\) and a scalar of integer charge \(p\).  Find
    \(g_\phi\), \(\widetilde\phi\), \(\widetilde A_\mu\), the residual
    subgroup, and the vector mass.  Compare with the usual Abelian Higgs
    description of unitary gauge.
    \item[(f)]
    Finally take \(G=SU(2)\times U(1)_Y\), with a Higgs doublet of
    hypercharge \(-1/2\) and
    \[
      \langle\phi\rangle=\frac1{\sqrt2}
        \begin{pmatrix}v\\0\end{pmatrix},\qquad v>0 .
    \]
    Show that unitary gauge leaves one real Higgs field and that, in the
    basis \((A^1,A^2,A^3,B)\), the vector mass matrix is
    \[
      \mu^2=\frac{v^2}{4}
      \begin{pmatrix}
        g_2^2&0&0&0\\
        0&g_2^2&0&0\\
        0&0&g_2^2&-g_1g_2\\
        0&0&-g_1g_2&g_1^2
      \end{pmatrix}.
    \]
  \end{enumerate}

  \SupplementaryExercise{3}{Goldstone's Theorem and Fundamental \(SU(3)\) Higgsing}
  {Adapted from Cambridge Part III, The Standard Model, 2021 Paper 305,
  Question 3.}
  {cambridge-part-iii-standard-model-2021-p305-q3-v2ch21-p03}
  \begin{enumerate}
    \item[(a)]
    State and prove Goldstone's theorem using both classical and quantum
    arguments.  You may use without proof that conserved charges \(Q^a\)
    are quantum operators such that
    \[
      [\phi_i,Q^a]=iT^a_{ij}\phi_j,
    \]
    with \(\phi_i\) scalar fields transforming in the representation of
    the algebra generators \(T^a_{ij}\).

    \item[(b)]
    Consider an \(SU(3)\) gauge theory with gauge field \(A_\mu(x)\)
    coupled to a scalar field \(\Phi\) in the fundamental representation.
    \begin{enumerate}
      \item[(i)]
      State how \(A_\mu(x)\) transforms under an \(SU(3)\) gauge
      transformation and prove that the field strength transforms
      covariantly.
      \item[(ii)]
      Write down the most general renormalisable Lagrangian determining
      the couplings of \(\Phi\) to the gauge fields, including kinetic
      terms for both fields and the scalar potential for \(\Phi\).
      \item[(iii)]
      Show that the symmetry may be broken by a non-vanishing vacuum
      expectation value for \(\Phi\).  Determine the pattern of symmetry
      breaking and verify Goldstone's theorem.
      \item[(iv)]
      Show explicitly how the Goldstone modes become the longitudinal
      modes of the original gauge field to give it a mass.  Are there
      other massive states in this theory?
    \end{enumerate}
  \end{enumerate}

  \SupplementaryExercise{4}{Neutrino Masses and a Hypercharge-One Scalar Triplet}
  {Adapted from Cambridge Part III, The Standard Model, 2024 Paper 305,
  Question 4.}
  {cambridge-part-iii-standard-model-2024-p305-q4-v2ch21-p04}
  List the fermions in one generation of the Standard Model, including
  the right-handed neutrino, and give their representations under the
  gauge group.  Explain why it is not possible to write down mass terms
  for the fermions but how, with judiciously chosen quantum numbers for
  the Higgs boson, it is possible for them all to gain a mass through
  Yukawa terms.

  Explain how the small observed value of the neutrino mass may be due to
  a large Majorana mass for the right-handed neutrino.  How is it possible
  to capture this same physics in the absence of a right-handed neutrino?

  Suppose that the Standard Model is extended to include a complex scalar
  field \(\phi\), transforming in the \(\mathbf3\) of \(SU(2)\), a
  singlet under \(SU(3)\), and with hypercharge \(Y=+1\).  What are the
  electric charges of the \(\phi\) field after electroweak symmetry
  breaking?  Write
  \[
    \phi=\frac12
    \begin{pmatrix}
      \phi^3 & \phi^1-i\phi^2\\
      \phi^1+i\phi^2 & -\phi^3
    \end{pmatrix},
    \qquad \phi^1,\phi^2,\phi^3\in\mathbb C,
  \]
  Find the components of \(\phi\) that are uncharged under
  electromagnetism.  Write down a gauge-invariant Yukawa coupling between
  \(\phi\) and the fermions of the Standard Model.  Show that if \(\phi\)
  condenses without breaking electromagnetism then the left-handed
  neutrino gets a Majorana mass.

  \SupplementaryExercise{5}{The Abelian Higgs Model and a Two-Scalar Vacuum}
  {Adapted from Cambridge Part III, The Standard Model, 2025 Paper 305,
  Question 1.}
  {cambridge-part-iii-standard-model-2025-p305-q1-v2ch21-p05}
  The Abelian Higgs model consists of a \(U(1)\) gauge field \(A_\mu\)
  coupled to a complex scalar field \(\phi\).  The action is
  \[
  \begin{gathered}
    S=\int d^4x\left[
      -\frac14F_{\mu\nu}F^{\mu\nu}
      +(D_\mu\phi)^\dagger D^\mu\phi
      -\frac{\lambda}{2}\bigl(\phi^\dagger\phi-v^2\bigr)^2
    \right],\\
    F_{\mu\nu}=\partial_\mu A_\nu-\partial_\nu A_\mu,\\
    D_\mu=\partial_\mu-ieA_\mu .
  \end{gathered}
  \]
  \emph{Source-convention note.}  This displayed action retains the
  paper's mostly-minus metric convention; the solutions translate the
  physical conclusions to this volume's mostly-plus convention.
  \begin{enumerate}
    \item[(i)]
    By working in a suitable gauge, determine the masses and spins of the
    particle excitations in this theory.
    \item[(ii)]
    Show that the equation of motion for the gauge field takes the form
    \[
      \partial_\mu F^{\mu\nu}=J^\nu
    \]
    for some current \(J^\nu\) that you should compute.  Show that, when
    restricted to configurations such that \(|\phi|=v\), the curl of the
    spatial current \(\boldsymbol\nabla\times\mathbf J\) is related to the
    magnetic field \(\mathbf B\).  Hence show that, in time-independent
    backgrounds, the magnetic field obeys
    \[
      \nabla^2\mathbf B=\frac1{\xi^2}\mathbf B .
    \]
    for some penetration depth \(\xi\) that you should compute.  Explain
    the physical significance of this result for superconductivity.
    \item[(iii)]
    Consider now a \(U(1)\) gauge field coupled to two complex scalar
    fields \(\phi_i\), \(i=1,2\), with action
    \[
      S=\int d^4x\left[
        -\frac14F_{\mu\nu}F^{\mu\nu}
        +\sum_{i=1}^2(D_\mu\phi_i)^\dagger D^\mu\phi_i
        -\frac{\lambda}{2}
          \left(\sum_{i=1}^2\phi_i^\dagger\phi_i-v^2\right)^2
      \right].
    \]
    What is the global symmetry of the theory?  Show that there is a
    combination of the gauge symmetry and global symmetry that is
    unbroken in the ground state.  What is the spectrum of particle
    masses?  Confirm that the number of massless scalars is in agreement
    with expectations from Goldstone's theorem.  What is the manifold of
    gauge-inequivalent ground states?
  \end{enumerate}

  \SupplementaryExercise{6}{Precision Electroweak Widths and Loop Diagnostics}
  {Inspired by the Particle Data Group, Electroweak Model and
  Constraints on New Physics (2025), Sections 10.4.6--10.4.7.}
  {pdg-electroweak-2025-v2ch21-p06}
  \begin{enumerate}
    \item[(a)]
    Neglecting the final-state lepton mass and radiative corrections,
    derive
    \[
      \Gamma(Z\to f\bar f)
        =N_c\frac{G_Fm_Z^3}{6\pi\sqrt2}
         \bigl[(g_V^f)^2+(g_A^f)^2\bigr].
    \]
    Specialize to a charged lepton and explain why its small vector
    coupling makes parity violation large.
    \item[(b)]
    A new fermion doublet has masses \(m_1,m_2\) and color multiplicity
    \(N_c\).  Its custodial-symmetry-breaking contribution is proportional
    to
    \[
      \Delta m^2=m_1^2+m_2^2
        -\frac{4m_1^2m_2^2}{m_1^2-m_2^2}\ln\frac{m_1}{m_2}.
    \]
    Show that it vanishes for degenerate masses and expand it for a small
    splitting \(m_1-m_2=\delta m\).
    \item[(c)]
    Neglect the Higgs term and use the one-loop result
    \[
      a_\ell^{\rm EW}
        =\frac{G_Fm_\ell^2}{8\sqrt2\pi^2}
         \left[\frac53+\frac13(1-4s_W^2)^2\right].
    \]
    Explain the \(m_\ell^2\) scaling and estimate the ratio
    \(a_\mu^{\rm EW}/a_e^{\rm EW}\).
  \end{enumerate}

  \SupplementaryExercise{8}{Unified Multiplets, Proton Decay, and Thresholds}
  {Inspired by the Particle Data Group, Grand Unified Theories (2025),
  Section 92.5.}
  {pdg-guts-2025-v2ch21-p08}
  \begin{enumerate}
    \item[(a)]
    New light matter shifts the one-loop coefficients by
    \(\Delta b_i\).  Show that if
    \(\Delta b_1=\Delta b_2=\Delta b_3\) in unified normalization, the
    differences \(\alpha_i^{-1}-\alpha_j^{-1}\) and the inferred
    unification scale are unchanged, while \(\alpha_U\) changes.  Explain
    the connection with complete unified multiplets.
    \item[(b)]
    Integrating out a heavy \(X\) gauge boson of mass \(M_X\) and unified
    coupling \(g_U\) gives a four-fermion coefficient of order
    \(g_U^2/M_X^2\).  By dimensional analysis derive the scaling
    \(\Gamma_p\sim\alpha_U^2m_p^5/M_X^4\) and
    \(\tau_p\sim M_X^4/(\alpha_U^2m_p^5)\).
    \item[(c)]
    A heavy multiplet component \(r\) of mass \(M_r\) contributes
    \[
      \delta_i^{(r)}
        =\frac{b_i^{(r)}}{2\pi}\ln\frac{M_U}{M_r}
    \]
    to \(\alpha_i^{-1}\).  Show which combinations of the
    \(\delta_i\) shift the predicted unification scale and which shift the
    unified coupling.  Explain why a degenerate complete multiplet cannot
    repair a mismatch among the three running couplings.
  \end{enumerate}

  \SupplementaryExercise{9}{\(R_\xi\)-Gauge Feynman Rules for the Abelian Higgs Model}
  {Adapted from Daniel Harlow, \emph{Relativistic Quantum Field Theory II},
  Section 13.7, Problem 4 (Fall 2024).}
  {harlow-qft2-2024-sec13p7-problem4-v2ch21-p09}
  In this problem we will discuss the Feynman rules for the Abelian Higgs
  model.  For standalone reference, take
  \[
    \mathcal L=-\frac14F^{\mu\nu}F_{\mu\nu}
      -(D_\mu\phi)^*D^\mu\phi
      +m^2|\phi|^2-\frac{\lambda}{4}|\phi|^4,
    \qquad D_\mu\phi=(\partial_\mu-iqA_\mu)\phi ,
  \]
  with \(m^2,\lambda>0\), and choose the real positive vacuum
  \(\phi_0=\sqrt{2m^2/\lambda}\).  For this purpose it is easier not to
  use a radial parametrization of \(\phi\); instead take
  \[
    \phi=\phi_0+\frac1{\sqrt2}(\zeta+i\chi),
  \]
  where \(\zeta\) and \(\chi\) are real fields.
  \begin{enumerate}
    \item[(a)]
    Expand the Lagrangian in terms of \(A_\mu,\zeta,\chi\), and show that
    its quadratic part contains a mixing term between \(A_\mu\) and
    \(\chi\).
    \item[(b)]
    Unitary gauge in this model is not good in perturbation theory, as it
    makes the photon propagator unpleasant.  It is better to choose the
    gauge-fixing condition to cancel the cross term found above.  Show
    that this is accomplished by adding
    \[
      \mathcal L_{\mathrm{gf}}
        =-\frac1{2\xi}
          \left(\partial_\mu A^\mu-q\sqrt2\,\xi\phi_0\chi\right)^2 .
    \]
    Show that it cancels the quadratic mixing term.  Argue that the
    Faddeev--Popov determinant is field independent and may be ignored.
    \item[(c)]
    Find the propagators for \(A_\mu,\zeta,\chi\) as functions of \(\xi\).
    \emph{Source note.}  The printed problem asks for the sequence
    \(A,\zeta,\text{ and }\zeta\); the second \(\zeta\) is understood as
    \(\chi\), the third field defined above.
    \item[(d)]
    Draw all the interaction vertices.  You need not work out the vertex
    factors, but you can if you want to.
  \end{enumerate}

  \SupplementaryExercise{10}{Diagrammatic Understanding of the BCS Instability}
  {Adapted from John McGreevy, Physics 215C, Homework 8, Problem 1
  (Spring 2025).}
  {mcgreevy-215c-s25-hw8-problem1-v2ch21-p10}
  \begin{enumerate}
    \item[(a)]
    Recall that only the four-fermion interactions with special kinematics
    are marginal.  Keeping only these interactions, show that cactus
    diagrams (like this:
    \[
      % A cactus is a stack of tadpoles, each one growing out of the top of
      % the one below.  Every loop therefore closes on a single vertex; open
      % arcs running alongside a stem would be a different graph.
      \begin{tikzpicture}[baseline=-0.35ex,x=0.55cm,y=0.55cm,
          line width=0.55pt,
          vtx/.style={circle,fill,inner sep=1.1pt}]
        \draw (-1.35,0)--(1.35,0);
        \draw (0,0) .. controls (1.33,1.20) and (-1.33,1.20) .. (0,0);
        \draw (0,0.9) .. controls (1.33,2.10) and (-1.33,2.10) .. (0,0.9);
        \draw (0,1.8) .. controls (1.33,3.00) and (-1.33,3.00) .. (0,1.8);
        \node[vtx] at (0,0) {}; \node[vtx] at (0,0.9) {};
        \node[vtx] at (0,1.8) {};
      \end{tikzpicture})
    \]
    dominate.
    \item[(b)]
    To sum the cacti, we can make bubbles with a corrected propagator.
    Argue that this correction to the propagator is innocuous and can be
    ignored.
    \item[(c)]
    Armed with these results, compute diagrammatically the Cooper-channel
    susceptibility (two-particle Green's function),
    \[
      \chi(\omega_0)=
      \left\langle T\,
        \psi^\dagger_{\mathbf k,\omega_3,\downarrow}
        \psi^\dagger_{-\mathbf k,\omega_4,\uparrow}
        \psi_{\mathbf p,\omega_1,\downarrow}
        \psi_{-\mathbf p,\omega_2,\uparrow}
      \right\rangle
    \]
    as a function of \(\omega_0=\omega_1+\omega_2\), the frequencies of
    the incoming particles.  Think of \(\chi\) as a two-point function of
    the Cooper-pair field
    \(\Phi=\epsilon_{\alpha\beta}\psi_\alpha\psi_\beta\) at zero
    momentum.  Sum the geometric series in terms of a one-loop integral
    kernel.
    \item[(d)]
    Do the integrals.  In the loops, restrict the range of momenta to
    \(|\epsilon(\mathbf k)|<E_D\), the Debye energy, since it is electrons
    with these energies that experience attractive interactions.

    Consider for simplicity a round Fermi surface.  For doing integrals
    of functions singular near a round Fermi surface, approximate the
    dispersion relation as
    \[
      \epsilon(\mathbf k)\simeq v_F(|\mathbf k|-k_F),
      \qquad
      d^dk\simeq k_F^{d-1}\frac{d\xi}{v_F}\,d\Omega_{d-1}.
    \]
    I recommend doing the frequency integral first (by residues).
    \item[(e)]
    Show that when \(V<0\) is attractive, \(\chi(\omega_0)\) has a pole.
    Does it represent a bound state?  Interpret this pole in the
    two-particle Green's function as indicating an instability of the
    Fermi liquid to superconductivity.  Compare the location of the pole
    to the energy \(E_{\mathrm{BCS}}\) where the Cooper-channel
    interaction becomes strong.
    \item[(f)]
    \emph{Cooper problem.}  [Optional.]  We can compare this result to
    Cooper's influential analysis of the problem of two electrons
    interacting with each other in the presence of an inert Fermi sea.
    Consider a state with two electrons with antipodal momenta and
    opposite spin,
    \[
      \ket{\psi}=\sum_{\mathbf k}a_{\mathbf k}
        \psi^\dagger_{\mathbf k\uparrow}
        \psi^\dagger_{-\mathbf k\downarrow}\ket{F},
      \qquad
      \ket{F}=\prod_{|\mathbf k|<k_F}
        \psi^\dagger_{\mathbf k\uparrow}
        \psi^\dagger_{\mathbf k\downarrow}\ket{0} ,
    \]
    and the Hamiltonian
    \[
      H=\sum_{\mathbf k}\epsilon_{\mathbf k}
        \psi^\dagger_{\mathbf k\sigma}\psi_{\mathbf k\sigma}
       +\sum_{\mathbf k,\mathbf k'}V_{\mathbf k,\mathbf k'}
        \psi^\dagger_{\mathbf k\sigma}\psi_{\mathbf k\sigma}
        \psi^\dagger_{\mathbf k'\sigma'}\psi_{\mathbf k'\sigma'} .
    \]
    The Schrödinger equation is
    \[
      (\omega-2\epsilon_{\mathbf k})a_{\mathbf k}
        =\sum_{\mathbf k'}V_{\mathbf k,\mathbf k'}a_{\mathbf k'} .
    \]
    Take
    \[
      V_{\mathbf k,\mathbf k'}=Vw_{\mathbf k'}^*w_{\mathbf k},
      \qquad
      w_{\mathbf k}=
      \begin{cases}
        1,&0<\epsilon_{\mathbf k}<E_D,\\
        0,&\text{otherwise},
      \end{cases}
    \]
    and define \(C\equiv\sum_{\mathbf k}w_{\mathbf k}^*a_{\mathbf k}\).
    Show that
    \begin{equation}
      1=V\sum_{\mathbf k}
        \frac{|w_{\mathbf k}|^2}{\omega-2\epsilon_{\mathbf k}} .
      \tag{1}
    \end{equation}
    For attractive \(V\), find the bound state.  Compare (14) with the
    condition for the pole in the pair susceptibility.

    \emph{Source note.}  The parent problem's reference to (14) points to
    the displayed condition (1); there is no equation (14) in the
    two-page assignment.
  \end{enumerate}

\end{enumerate}
