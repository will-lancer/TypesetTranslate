\chapterbackmatter{Supplementary Exercises}

\begin{enumerate}
  \SupplementaryExercise{1}{The Mass Matrix as a Gram Matrix}
  {Adapted from M. J. D. Hamilton, The Higgs Boson for Mathematicians
  (2015), Sections 4--7.}
  {hamilton-higgs-mathematicians-2015}
  For real scalar fields with vacuum \(v\), define
  \(\mu^2_{\alpha\beta}=-(t_\alpha v)\cdot(t_\beta v)\).
  Prove that \(\mu^2\) is positive semidefinite and that its kernel is
  precisely the Lie algebra of the stabilizer of \(v\).  Relate its rank
  to the number of massive gauge bosons.

  \SupplementaryExercise{2}{The Abelian Higgs Spectrum}
  {Adapted from M. J. D. Hamilton, The Higgs Boson for Mathematicians
  (2015), Sections 5--7.}
  {hamilton-higgs-mathematicians-2015}
  Consider
  \[
    \mathcal L=-\frac14F_{\mu\nu}F^{\mu\nu}
      -|D_\mu\Phi|^2-\lambda
       \left(|\Phi|^2-\frac{v^2}{2}\right)^2,
    \qquad D_\mu=\partial_\mu-iqA_\mu.
  \]
  Expand \(\Phi=(v+h+i\chi)/\sqrt2\).  Find the tree-level masses of
  \(A_\mu,h,\chi\) in unitarity gauge and count physical degrees of
  freedom before and after symmetry breaking.

  \SupplementaryExercise{3}{Orbit Directions and Eaten Modes}
  {Adapted from M. J. D. Hamilton, The Higgs Boson for Mathematicians
  (2015), Sections 5--7.}
  {hamilton-higgs-mathematicians-2015}
  Let a compact group \(G\) act orthogonally on a real scalar
  representation and let \(H\) stabilize \(v\).  Show that the tangent
  space to the vacuum orbit \(Gv\) is spanned by \(t_\alpha v\) and has
  dimension \(\dim G-\dim H\).  Explain why exactly this many scalar
  directions supply longitudinal vector polarizations.

  \SupplementaryExercise{4}{Residual Gauge Freedom in Unitarity Gauge}
  {Adapted from M. J. D. Hamilton, The Higgs Boson for Mathematicians
  (2015), Section 6.}
  {hamilton-higgs-mathematicians-2015}
  Unitarity gauge removes fluctuations tangent to the orbit through
  \(v\).  Determine which infinitesimal gauge transformations preserve
  this condition.  Explain why massless gauge fields still require an
  ordinary gauge-fixing condition and why a global unitary-gauge slice
  may fail where the scalar modulus vanishes.

  \SupplementaryExercise{5}{Poles in an \(R_\xi\) Gauge}
  {Adapted from M. J. D. Hamilton, The Higgs Boson for Mathematicians
  (2015), Sections 6--7.}
  {hamilton-higgs-mathematicians-2015}
  In a broken Abelian theory use
  \(f=\partial_\mu A^\mu-\xi m\chi\).  Derive the vector, Goldstone, and
  ghost propagator denominators.  Identify their physical and
  gauge-dependent poles and show how the limit \(\xi\to\infty\) recovers
  the unitary-gauge vector propagator.

  \SupplementaryExercise{6}{A General Tree-Level \(\rho\) Formula}
  {Adapted from M. J. D. Hamilton, The Higgs Boson for Mathematicians
  (2015), Sections 7--9.}
  {hamilton-higgs-mathematicians-2015}
  Scalar multiplets have weak isospin \(T_i\), hypercharge \(Y_i\), and
  neutral vacuum values \(v_i\), so \(T_{3i}=-Y_i\).  For complex
  multiplets derive
  \[
    \rho_{\rm tree}
      =\frac{\sum_i [T_i(T_i+1)-Y_i^2]|v_i|^2}
             {\sum_i2Y_i^2|v_i|^2}.
  \]
  Evaluate it for a doublet with \((T,Y)=(\tfrac12,\tfrac12)\) and
  explain why a \(Y=0\) triplet needs separate treatment in the
  denominator.

  \SupplementaryExercise{7}{Why a Chiral Mass Is Forbidden}
  {Adapted from Wentao Cui, MIT 8.323 Recitation 10 (2023),
  Section 1.1.}
  {mit-8323-s23-rec10-ew-higgs}
  Let \(\psi_L\) carry charges \((Q_L,Y_L)\) under
  \(U(1)_L\times U(1)_Y\), while \(\psi_R\) carries
  \((Q_R,Y_R)\).  Determine the transformation of
  \(\bar\psi_L\psi_R\) and state the two conditions for a bare Dirac mass.
  Apply them to \((Q_L,Y_L)=(-\tfrac12,-\tfrac12)\) and
  \((Q_R,Y_R)=(0,-1)\).

  \SupplementaryExercise{8}{Solving for Higgs Charges}
  {Adapted from Wentao Cui, MIT 8.323 Recitation 10 (2023),
  Sections 1.1--1.2.}
  {mit-8323-s23-rec10-ew-higgs}
  Add a complex scalar \(H\) and a Yukawa term
  \(yH\bar\psi_L\psi_R+\mathrm{H.c.}\).
  Determine the charges of \(H\) that make this interaction invariant
  for the fermion assignments in Exercise S.21.7.  Show explicitly that
  a nonzero \(\langle H\rangle\) produces a Dirac mass.

  \SupplementaryExercise{9}{An Unbroken Combination of Two \(U(1)\)s}
  {Adapted from Wentao Cui, MIT 8.323 Recitation 10 (2023),
  Sections 1.1--1.2.}
  {mit-8323-s23-rec10-ew-higgs}
  A scalar of charges \((q_1,q_2)\) under two Abelian groups acquires an
  expectation value \(v/\sqrt2\).  Find the massive and massless
  canonically normalized gauge-field combinations when the couplings are
  \(g_1,g_2\), and identify the generator that annihilates the vacuum.

  \SupplementaryExercise{10}{Fermion Mass and Higgs Coupling}
  {Adapted from Wentao Cui, MIT 8.323 Recitation 10 (2023),
  Section 1.2.}
  {mit-8323-s23-rec10-ew-higgs}
  With
  \(H=(v+h)/\sqrt2\) in unitarity gauge and Yukawa interaction
  \(-y\bar\psi_LH\psi_R+\mathrm{H.c.}\), derive the fermion mass and the
  \(h\bar\psi\psi\) coupling.  Express the latter entirely in terms of
  the physical mass and \(v\).

  \SupplementaryExercise{11}{Electric Charge from \(T_3+Y\)}
  {Adapted from Wentao Cui, MIT 8.323 Recitation 10 (2023),
  Sections 1.1--1.2.}
  {mit-8323-s23-rec10-ew-higgs}
  Use \(Q=T_3+Y\) to assign hypercharges to the lepton doublet
  \(L=(\nu_L,e_L)\), the singlet \(e_R\), and a Higgs doublet
  \(H=(H^+,H^0)\).  Verify that the charged-lepton Yukawa interaction
  \(-y_e\bar LHe_R+\mathrm{H.c.}\) is gauge invariant.

  \SupplementaryExercise{12}{The Dimension-Five Neutrino Operator}
  {Adapted from Wentao Cui, MIT 8.323 Recitation 10 (2023),
  Section 1.2.}
  {mit-8323-s23-rec10-ew-higgs}
  Without a right-handed neutrino, construct the lowest-dimension
  gauge-invariant operator made from two lepton doublets and two Higgs
  doublets that gives a neutrino mass.  Determine its canonical
  dimension and the neutrino mass after symmetry breaking in terms of
  its coefficient \(c/\Lambda\).

  \SupplementaryExercise{13}{Neutral-Current Couplings}
  {Adapted from the Particle Data Group, Electroweak Model and
  Constraints on New Physics (2025), Sections 10.1 and 10.4.6.}
  {pdg-electroweak-2025}
  Starting from \(Q=T_3+Y\) and the \(W^3\)--\(B\) rotation, show that
  the \(Z\) coupling to a fermion is
  \[
    -\frac{g}{2\cos\theta_W}\,
    \bar f\gamma^\mu(g_V^f-g_A^f\gamma^5)fZ_\mu,
    \quad
    g_V^f=T_3^f-2Q_f\sin^2\theta_W,\quad
    g_A^f=T_3^f.
  \]
  Evaluate \(g_V,g_A\) for a charged lepton and a neutrino.

  \SupplementaryExercise{14}{The Tree-Level Mass Relation}
  {Adapted from the Particle Data Group, Electroweak Model and
  Constraints on New Physics (2025), Section 10.1.}
  {pdg-electroweak-2025}
  Expand \(|D_\mu H|^2\) about
  \(\langle H\rangle=(0,v/\sqrt2)^{\mathsf T}\).  Derive
  \(m_W=gv/2\), \(m_Z=v\sqrt{g^2+g'^2}/2\), the massless photon, and
  \(m_W=m_Z\cos\theta_W\).

  \SupplementaryExercise{15}{Determining the Electroweak Scale}
  {Adapted from the Particle Data Group, Electroweak Model and
  Constraints on New Physics (2025), Sections 10.1 and 10.3.2.}
  {pdg-electroweak-2025}
  Integrate out the \(W\) at momenta much smaller than \(m_W\) and match
  to the Fermi interaction.  Show that
  \(G_F/\sqrt2=g^2/(8m_W^2)=1/(2v^2)\), and evaluate \(v\) using
  \(G_F=1.1663787\times10^{-5}\,\mathrm{GeV}^{-2}\).

  \SupplementaryExercise{16}{A Leptonic \(Z\) Width}
  {Adapted from the Particle Data Group, Electroweak Model and
  Constraints on New Physics (2025), Sections 10.4.6--10.4.7.}
  {pdg-electroweak-2025}
  Neglecting the final-state lepton mass and radiative corrections,
  derive
  \[
    \Gamma(Z\to f\bar f)
      =N_c\frac{G_Fm_Z^3}{6\pi\sqrt2}
       \bigl[(g_V^f)^2+(g_A^f)^2\bigr].
  \]
  Specialize to a charged lepton and explain why its small vector
  coupling makes parity violation large.

  \SupplementaryExercise{17}{A Heavy Doublet and the \(T\) Parameter}
  {Adapted from the Particle Data Group, Electroweak Model and
  Constraints on New Physics (2025), Section 10.6.}
  {pdg-electroweak-2025}
  A new fermion doublet has masses \(m_1,m_2\) and color multiplicity
  \(N_c\).  Its custodial-symmetry-breaking contribution is proportional
  to
  \[
    \Delta m^2=m_1^2+m_2^2
      -\frac{4m_1^2m_2^2}{m_1^2-m_2^2}\ln\frac{m_1}{m_2}.
  \]
  Show that it vanishes for degenerate masses and expand it for a small
  splitting \(m_1-m_2=\delta m\).

  \SupplementaryExercise{18}{The One-Loop Electroweak Scale of \(a_\ell\)}
  {Adapted from the Particle Data Group, Electroweak Model and
  Constraints on New Physics (2025), Sections 10.3.3 and 10.4.}
  {pdg-electroweak-2025}
  Neglect the Higgs term and use the one-loop result
  \[
    a_\ell^{\rm EW}
      =\frac{G_Fm_\ell^2}{8\sqrt2\pi^2}
       \left[\frac53+\frac13(1-4s_W^2)^2\right].
  \]
  Explain the \(m_\ell^2\) scaling and estimate the ratio
  \(a_\mu^{\rm EW}/a_e^{\rm EW}\).

  \SupplementaryExercise{19}{One Standard-Model Family in \(SU(5)\)}
  {Adapted from the Particle Data Group, Grand Unified Theories (2025),
  Section 92.2.}
  {pdg-guts-2025}
  Show that the left-handed fields of one Standard-Model family fit into
  \(\overline{\mathbf5}\oplus\mathbf{10}\) of \(SU(5)\).  Decompose
  these representations under
  \(SU(3)_C\times SU(2)_L\times U(1)_Y\) and identify the Standard-Model
  fields in each component.

  \SupplementaryExercise{20}{The \(5/3\) Hypercharge Normalization}
  {Adapted from the Particle Data Group, Grand Unified Theories (2025),
  Sections 92.1--92.2.}
  {pdg-guts-2025}
  In the fundamental of \(SU(5)\), take
  \(Y=\operatorname{diag}(-\tfrac13,-\tfrac13,-\tfrac13,
  \tfrac12,\tfrac12)\).  Compare \(\operatorname{Tr}Y^2\) with
  \(\operatorname{Tr}(T_3^2)=1/2\) for a canonically normalized
  \(SU(2)\) generator and derive \(g_1^2=(5/3)g'^2\).

  \SupplementaryExercise{21}{Solving One-Loop Unification}
  {Adapted from the Particle Data Group, Grand Unified Theories (2025),
  Section 92.5.}
  {pdg-guts-2025}
  Suppose
  \[
    \alpha_i^{-1}(\mu)
      =\alpha_U^{-1}+\frac{b_i}{2\pi}\ln\frac{M_U}{\mu}.
  \]
  Given two measured inverse couplings, solve algebraically for
  \(\ln(M_U/\mu)\) and \(\alpha_U^{-1}\).  State the consistency test
  supplied by the third coupling.

  \SupplementaryExercise{22}{Why Complete Multiplets Preserve a Meeting}
  {Adapted from the Particle Data Group, Grand Unified Theories (2025),
  Section 92.5.}
  {pdg-guts-2025}
  New light matter shifts the one-loop coefficients by
  \(\Delta b_i\).  Show that if
  \(\Delta b_1=\Delta b_2=\Delta b_3\) in unified normalization, the
  differences \(\alpha_i^{-1}-\alpha_j^{-1}\) and the inferred
  unification scale are unchanged, while \(\alpha_U\) changes.  Explain
  the connection with complete unified multiplets.

  \SupplementaryExercise{23}{Dimension-Six Proton Decay}
  {Adapted from the Particle Data Group, Grand Unified Theories (2025),
  Section 92.6.}
  {pdg-guts-2025}
  Integrating out a heavy \(X\) gauge boson of mass \(M_X\) and unified
  coupling \(g_U\) gives a four-fermion coefficient of order
  \(g_U^2/M_X^2\).  By dimensional analysis derive the scaling
  \(\Gamma_p\sim\alpha_U^2m_p^5/M_X^4\) and
  \(\tau_p\sim M_X^4/(\alpha_U^2m_p^5)\).

  \SupplementaryExercise{24}{Threshold Corrections}
  {Adapted from the Particle Data Group, Grand Unified Theories (2025),
  Section 92.5.}
  {pdg-guts-2025}
  A heavy multiplet component \(r\) of mass \(M_r\) contributes
  \[
    \delta_i^{(r)}
      =\frac{b_i^{(r)}}{2\pi}\ln\frac{M_U}{M_r}
  \]
  to \(\alpha_i^{-1}\).  Show which combinations of the
  \(\delta_i\) shift the predicted unification scale and which shift the
  unified coupling.  Explain why a degenerate complete multiplet cannot
  repair a mismatch among the three running couplings.

  \SupplementaryExercise{25}{London Screening}
  {Adapted from Martin Greiter, Is Electromagnetic Gauge Invariance
  Spontaneously Violated in Superconductors? (2005), Sections V--VI.}
  {greiter-superconductivity-2005}
  Take the static free-energy density
  \[
    \mathcal F=\frac{\mathbf B^2}{2}
      +\frac{\rho_s}{2}
       \left(\boldsymbol\nabla\theta-\frac{q}{\hbar}\mathbf A\right)^2.
  \]
  Derive the supercurrent and combine it with Maxwell's equation to show
  that a magnetic field obeys
  \((\nabla^2-\lambda^{-2})\mathbf B=0\).  Find \(\lambda\).

  \SupplementaryExercise{26}{Flux Quantization for Charge \(q\)}
  {Adapted from Martin Greiter, Is Electromagnetic Gauge Invariance
  Spontaneously Violated in Superconductors? (2005), Sections VI--VII.}
  {greiter-superconductivity-2005}
  Deep inside a superconducting ring the current vanishes.  Use
  single-valuedness of an order parameter of charge \(q\) to derive the
  allowed magnetic fluxes.  Recover the conventional flux quantum for
  \(q=-2e\).

  \SupplementaryExercise{27}{The ac Josephson Relation}
  {Adapted from Martin Greiter, Is Electromagnetic Gauge Invariance
  Spontaneously Violated in Superconductors? (2005), Section IX.}
  {greiter-superconductivity-2005}
  Let the junction energy be
  \(-E_J\cos\delta\), where \(\delta\) is the gauge-invariant phase
  difference of condensates with charge \(q\).  Derive the current-phase
  relation and use a constant voltage difference \(\Delta V\) to find
  the oscillation frequency.

  \SupplementaryExercise{28}{A Phase Slip Changes the Fluxoid}
  {Adapted from Martin Greiter, Is Electromagnetic Gauge Invariance
  Spontaneously Violated in Superconductors? (2005), Sections VII--VIII.}
  {greiter-superconductivity-2005}
  Explain why the winding number of a nonvanishing complex order
  parameter cannot change continuously.  Show what must happen locally
  during a phase slip and determine the associated change of flux in a
  ring with condensate charge \(q\).

  \SupplementaryExercise{29}{Several Commensurate Condensates}
  {Adapted from Martin Greiter, Is Electromagnetic Gauge Invariance
  Spontaneously Violated in Superconductors? (2005), Sections VI--IX.}
  {greiter-superconductivity-2005}
  Condensates carry charges \(q_i=n_iq_0\), where the integers \(n_i\)
  have greatest common divisor \(d\).  Determine the residual subgroup
  of the compact \(U(1)\), the smallest common nonzero flux, and the
  fundamental Josephson frequency allowed when all tunneling channels
  are present.

  \SupplementaryExercise{30}{Incommensurate Charges and Relative Phases}
  {Adapted from Martin Greiter, Is Electromagnetic Gauge Invariance
  Spontaneously Violated in Superconductors? (2005), Sections VI--IX.}
  {greiter-superconductivity-2005}
  Two condensates have charges \(q_1,q_2\) with irrational
  \(q_1/q_2\).  Determine the intersection of their flux lattices.
  Construct a gauge-invariant relative phase, and explain what happens
  to flux quantization and to the junction current if no interaction
  locks this relative phase.
\end{enumerate}
