  \SupplementaryExercise{1}{Charges, Goldstone's Theorem, and the Abelian Higgs Mechanism}
  {Adapted from Cambridge Part III, The Standard Model, 2022 Paper 305, Question 3.}
  {cambridge-part-iii-standard-model-2022-p305-q3-v2ch21-p01}
  Let \(N\) scalar fields transform under a compact continuous group as
  \(\delta\phi_m=i\alpha^a(T_a)_{mn}\phi_n\), with conserved currents
  \(J^\mu_a\).
  \begin{enumerate}
    \item[(a)] Show that
      \(Q_a=\int d^3x\,J^0_a\) generates the transformation of the fields.
    \item[(b)] Define a local order parameter that distinguishes a
      symmetry-preserving vacuum from one that spontaneously breaks
      generator \(a\).
    \item[(c)] Use \([H,Q_a]=0\) to explain both why excitations related by
      a broken symmetry need not be energy-degenerate and why
      \(Q_a\ket0\), when nonzero, has the vacuum energy.
    \item[(d)] State and prove Goldstone's theorem, displaying the
      zero-mass pole created by the broken current.
    \item[(e)] Gauge a \(U(1)\) symmetry acting on one complex scalar.
      Write the most general CP-even renormalizable Lagrangian after
      canonical normalization, count its independent physical parameters,
      and give the stability and symmetry-breaking range.  Expand about a
      minimum; determine every physical mass and the cubic and quartic
      interactions.  Count degrees of freedom and explain the Higgs
      mechanism.
  \end{enumerate}

  \SupplementaryExercise{2}{Unitarity Gauge for General Higgsing}
  {Adapted from Daniel Harlow, \emph{Relativistic Quantum Field Theory III},
  Section 10.7, Problem 1 (2026).}
  {harlow-qft3-2026-sec10p7-problem1-v2ch21-p02}
  Let a compact gauge group \(G\) act on a scalar multiplet through
  \(D(g)\), with Hermitian Lie-algebra generators \(\tau_a\).  Consider
  \[
    \mathcal L=-\frac14\sum_a\frac1{g_a^2}
      F_{\mu\nu}^aF^{a\mu\nu}
      -(D_\mu\phi)^\dagger D^\mu\phi-V(\phi),
    \qquad
    D_\mu\phi=\partial_\mu\phi-i\tau_aA_\mu^a\phi ,
  \]
  and write
  \[
    \langle\phi\rangle=\frac v{\sqrt2},\qquad
    H=\{h\in G:D(h)v=v\}.
  \]
  On each gauge orbit choose \(g_\phi(x)\) to maximize
  \[
    f_\phi(g)=\operatorname{Re}\!\left[\phi^\dagger(x)D(g)v\right],
  \]
  and define the unitary-gauge fields
  \[
    \widetilde\phi=D(g_\phi^{-1})\phi,\qquad
    \widetilde A_\mu=g_\phi^{-1}A_\mu g_\phi+
      i g_\phi^{-1}\partial_\mu g_\phi .
  \]
  \begin{enumerate}
    \item[(a)]
    Derive the stationarity condition
    \[
      \operatorname{Im}(\widetilde\phi^\dagger\tau_av)=0 .
    \]
    Show also that a configuration equal to \(v\) may be represented by
    \(\widetilde\phi=v\), so the gauge choice preserves the stated vacuum.
    \item[(b)]
    Prove that \(g_\phi h\) is another maximum for every \(h(x)\in H\).
    Interpret this as a residual local \(H\) symmetry.  For an
    \(H\)-invariant decomposition
    \(\mathfrak g=\mathfrak h\oplus\mathfrak h^\perp\), determine how the
    unbroken and broken components of \(\widetilde A_\mu\) transform under
    this residual symmetry.
    \item[(c)]
    A global maximum need not be unique on \(G/H\), but perturbation theory
    only needs a unique local maximum near the vacuum.  Expand
    \[
      f_{\widetilde\phi}(e^{i\theta^aT_a})
        =\operatorname{Re}\!
        \left[\widetilde\phi^\dagger
          e^{i\theta^a\tau_a}v\right]
    \]
    to second order in \(\theta\).  Show for
    \(\widetilde\phi\) sufficiently close to \(v\) that the maximum is
    strict on \(G/H\), and that a flat direction obeys
    \(\theta^a\tau_av=0\), hence lies in \(\mathfrak h\).
    \item[(d)]
    Parameterize the remaining scalar fluctuation by
    \[
      \widetilde\phi=\frac1{\sqrt2}(v+h).
    \]
    Rewrite the Lagrangian in terms of \(h\) and \(\widetilde A_\mu\),
    using the condition from part~(a).  For canonically normalized vector
    fields \(\widehat A_\mu^a=\widetilde A_\mu^a/g_a\), prove that
    \[
      (\mu^2)_{ab}=\frac{g_ag_b}{2}\,
        v^\dagger\{\tau_a,\tau_b\}v .
    \]
    Show that this matrix is real, symmetric, and positive semidefinite,
    and characterize its null vectors in terms of \(\mathfrak h\).
    \item[(e)]
    Specialize to \(G=U(1)\) and a scalar of integer charge \(p\).  Find
    \(g_\phi\), \(\widetilde\phi\), \(\widetilde A_\mu\), the residual
    subgroup, and the vector mass.  Compare with the usual Abelian Higgs
    description of unitary gauge.
    \item[(f)]
    Finally take \(G=SU(2)\times U(1)_Y\), with a Higgs doublet of
    hypercharge \(-1/2\) and
    \[
      \langle\phi\rangle=\frac1{\sqrt2}
        \begin{pmatrix}v\\0\end{pmatrix},\qquad v>0 .
    \]
    Show that unitary gauge leaves one real Higgs field and that, in the
    basis \((A^1,A^2,A^3,B)\), the vector mass matrix is
    \[
      \mu^2=\frac{v^2}{4}
      \begin{pmatrix}
        g_2^2&0&0&0\\
        0&g_2^2&0&0\\
        0&0&g_2^2&-g_1g_2\\
        0&0&-g_1g_2&g_1^2
      \end{pmatrix}.
    \]
  \end{enumerate}

  \SupplementaryExercise{3}{Chiral Charges and Unbroken Gauge Generators}
  {Inspired by Wentao Cui, MIT 8.323 Recitation 10 (2023),
  Section 1.1.}
  {mit-8323-s23-rec10-ew-higgs-v2ch21-p03}
  \begin{enumerate}
    \item[(a)]
    Let \(\psi_L\) carry charges \((Q_L,Y_L)\) under
    \(U(1)_L\times U(1)_Y\), while \(\psi_R\) carries
    \((Q_R,Y_R)\).  Determine the transformation of
    \(\bar\psi_L\psi_R\) and state the two conditions for a bare Dirac mass.
    Apply them to \((Q_L,Y_L)=(-\tfrac12,-\tfrac12)\) and
    \((Q_R,Y_R)=(0,-1)\).
    \item[(b)]
    Add a complex scalar \(H\) and a Yukawa term
    \(yH\bar\psi_L\psi_R+\mathrm{H.c.}\).
    Determine the charges of \(H\) that make this interaction invariant
    for the fermion assignments in part~(a).  Show explicitly that
    a nonzero \(\langle H\rangle\) produces a Dirac mass.
    \item[(c)]
    A scalar of charges \((q_1,q_2)\) under two Abelian groups acquires an
    expectation value \(v/\sqrt2\).  Find the massive and massless
    canonically normalized gauge-field combinations when the couplings are
    \(g_1,g_2\), and identify the generator that annihilates the vacuum.
  \end{enumerate}

  \SupplementaryExercise{4}{Yukawa Masses, Electric Charge, and the Neutrino Operator}
  {Inspired by Wentao Cui, MIT 8.323 Recitation 10 (2023),
  Section 1.2.}
  {mit-8323-s23-rec10-ew-higgs-v2ch21-p04}
  \begin{enumerate}
    \item[(a)]
    With
    \(H=(v+h)/\sqrt2\) in unitarity gauge and Yukawa interaction
    \(-y\bar\psi_LH\psi_R+\mathrm{H.c.}\), derive the fermion mass and the
    \(h\bar\psi\psi\) coupling.  Express the latter entirely in terms of
    the physical mass and \(v\).
    \item[(b)]
    Use \(Q=T_3+Y\) to assign hypercharges to the lepton doublet
    \(L=(\nu_L,e_L)\), the singlet \(e_R\), and a Higgs doublet
    \(H=(H^+,H^0)\).  Verify that the charged-lepton Yukawa interaction
    \(-y_e\bar LHe_R+\mathrm{H.c.}\) is gauge invariant.
    \item[(c)]
    Without a right-handed neutrino, construct the lowest-dimension
    gauge-invariant operator made from two lepton doublets and two Higgs
    doublets that gives a neutrino mass.  Determine its canonical
    dimension and the neutrino mass after symmetry breaking in terms of
    its coefficient \(c/\Lambda\).
  \end{enumerate}

  \SupplementaryExercise{5}{Neutral Currents, Mass Relations, and the Electroweak Scale}
  {Inspired by the Particle Data Group, Electroweak Model and
  Constraints on New Physics (2025), Sections 10.1 and 10.4.6.}
  {pdg-electroweak-2025-v2ch21-p05}
  \begin{enumerate}
    \item[(a)]
    Starting from \(Q=T_3+Y\) and the \(W^3\)--\(B\) rotation, show that
    the \(Z\) coupling to a fermion is
    \[
      -\frac{g}{2\cos\theta_W}\,
      \bar f\gamma^\mu(g_V^f-g_A^f\gamma_5)fZ_\mu,
      \quad
      g_V^f=T_3^f-2Q_f\sin^2\theta_W,\quad
      g_A^f=T_3^f.
    \]
    Evaluate \(g_V,g_A\) for a charged lepton and a neutrino.
    \item[(b)]
    Expand \(|D_\mu H|^2\) about
    \(\langle H\rangle=(0,v/\sqrt2)^{\mathsf T}\).  Derive
    \(m_W=gv/2\), \(m_Z=v\sqrt{g^2+g'^2}/2\), the massless photon, and
    \(m_W=m_Z\cos\theta_W\).
    \item[(c)]
    Integrate out the \(W\) at momenta much smaller than \(m_W\) and match
    to the Fermi interaction.  Show that
    \(G_F/\sqrt2=g^2/(8m_W^2)=1/(2v^2)\), and evaluate \(v\) using
    \(G_F=1.1663787\times10^{-5}\,\mathrm{GeV}^{-2}\).
  \end{enumerate}

  \SupplementaryExercise{6}{Precision Electroweak Widths and Loop Diagnostics}
  {Inspired by the Particle Data Group, Electroweak Model and
  Constraints on New Physics (2025), Sections 10.4.6--10.4.7.}
  {pdg-electroweak-2025-v2ch21-p06}
  \begin{enumerate}
    \item[(a)]
    Neglecting the final-state lepton mass and radiative corrections,
    derive
    \[
      \Gamma(Z\to f\bar f)
        =N_c\frac{G_Fm_Z^3}{6\pi\sqrt2}
         \bigl[(g_V^f)^2+(g_A^f)^2\bigr].
    \]
    Specialize to a charged lepton and explain why its small vector
    coupling makes parity violation large.
    \item[(b)]
    A new fermion doublet has masses \(m_1,m_2\) and color multiplicity
    \(N_c\).  Its custodial-symmetry-breaking contribution is proportional
    to
    \[
      \Delta m^2=m_1^2+m_2^2
        -\frac{4m_1^2m_2^2}{m_1^2-m_2^2}\ln\frac{m_1}{m_2}.
    \]
    Show that it vanishes for degenerate masses and expand it for a small
    splitting \(m_1-m_2=\delta m\).
    \item[(c)]
    Neglect the Higgs term and use the one-loop result
    \[
      a_\ell^{\rm EW}
        =\frac{G_Fm_\ell^2}{8\sqrt2\pi^2}
         \left[\frac53+\frac13(1-4s_W^2)^2\right].
    \]
    Explain the \(m_\ell^2\) scaling and estimate the ratio
    \(a_\mu^{\rm EW}/a_e^{\rm EW}\).
  \end{enumerate}

  \SupplementaryExercise{7}{A Standard-Model Family inside \(SU(5)\)}
  {Inspired by the Particle Data Group, Grand Unified Theories (2025),
  Section 92.2.}
  {pdg-guts-2025-v2ch21-p07}
  \begin{enumerate}
    \item[(a)]
    Show that the left-handed fields of one Standard-Model family fit into
    \(\overline{\mathbf5}\oplus\mathbf{10}\) of \(SU(5)\).  Decompose
    these representations under
    \(SU(3)_C\times SU(2)_L\times U(1)_Y\) and identify the Standard-Model
    fields in each component.
    \item[(b)]
    In the fundamental of \(SU(5)\), take
    \(Y=\operatorname{diag}(-\tfrac13,-\tfrac13,-\tfrac13,
    \tfrac12,\tfrac12)\).  Compare \(\operatorname{Tr}Y^2\) with
    \(\operatorname{Tr}(T_3^2)=1/2\) for a canonically normalized
    \(SU(2)\) generator and derive \(g_1^2=(5/3)g'^2\).
    \item[(c)]
    Suppose
    \[
      \alpha_i^{-1}(\mu)
        =\alpha_U^{-1}+\frac{b_i}{2\pi}\ln\frac{M_U}{\mu}.
    \]
    Given two measured inverse couplings, solve algebraically for
    \(\ln(M_U/\mu)\) and \(\alpha_U^{-1}\).  State the consistency test
    supplied by the third coupling.
  \end{enumerate}

  \SupplementaryExercise{8}{Unified Multiplets, Proton Decay, and Thresholds}
  {Inspired by the Particle Data Group, Grand Unified Theories (2025),
  Section 92.5.}
  {pdg-guts-2025-v2ch21-p08}
  \begin{enumerate}
    \item[(a)]
    New light matter shifts the one-loop coefficients by
    \(\Delta b_i\).  Show that if
    \(\Delta b_1=\Delta b_2=\Delta b_3\) in unified normalization, the
    differences \(\alpha_i^{-1}-\alpha_j^{-1}\) and the inferred
    unification scale are unchanged, while \(\alpha_U\) changes.  Explain
    the connection with complete unified multiplets.
    \item[(b)]
    Integrating out a heavy \(X\) gauge boson of mass \(M_X\) and unified
    coupling \(g_U\) gives a four-fermion coefficient of order
    \(g_U^2/M_X^2\).  By dimensional analysis derive the scaling
    \(\Gamma_p\sim\alpha_U^2m_p^5/M_X^4\) and
    \(\tau_p\sim M_X^4/(\alpha_U^2m_p^5)\).
    \item[(c)]
    A heavy multiplet component \(r\) of mass \(M_r\) contributes
    \[
      \delta_i^{(r)}
        =\frac{b_i^{(r)}}{2\pi}\ln\frac{M_U}{M_r}
    \]
    to \(\alpha_i^{-1}\).  Show which combinations of the
    \(\delta_i\) shift the predicted unification scale and which shift the
    unified coupling.  Explain why a degenerate complete multiplet cannot
    repair a mismatch among the three running couplings.
  \end{enumerate}

  \SupplementaryExercise{9}{London Screening, Flux Quantization, and Josephson Dynamics}
  {Inspired by Martin Greiter, Is Electromagnetic Gauge Invariance
  Spontaneously Violated in Superconductors? (2005), Sections V--VI.}
  {greiter-superconductivity-2005-v2ch21-p09}
  \begin{enumerate}
    \item[(a)]
    Take the static free-energy density
    \[
      \mathcal F=\frac{\mathbf B^2}{2}
        +\frac{\rho_s}{2}
         \left(\boldsymbol\nabla\theta-\frac{q}{\hbar}\mathbf A\right)^2.
    \]
    Derive the supercurrent and combine it with Maxwell's equation to show
    that a magnetic field obeys
    \((\nabla^2-\lambda^{-2})\mathbf B=0\).  Find \(\lambda\).
    \item[(b)]
    Deep inside a superconducting ring the current vanishes.  Use
    single-valuedness of an order parameter of charge \(q\) to derive the
    allowed magnetic fluxes.  Recover the conventional flux quantum for
    \(q=-2e\).
    \item[(c)]
    Let the junction energy be
    \(-E_J\cos\delta\), where \(\delta\) is the gauge-invariant phase
    difference of condensates with charge \(q\).  Derive the current-phase
    relation and use a constant voltage difference \(\Delta V\) to find
    the oscillation frequency.
  \end{enumerate}

  \SupplementaryExercise{10}{Phase Slips and Multicomponent Condensates}
  {Inspired by Martin Greiter, Is Electromagnetic Gauge Invariance
  Spontaneously Violated in Superconductors? (2005), Sections VII--VIII.}
  {greiter-superconductivity-2005-v2ch21-p10}
  \begin{enumerate}
    \item[(a)]
    Explain why the winding number of a nonvanishing complex order
    parameter cannot change continuously.  Show what must happen locally
    during a phase slip and determine the associated change of flux in a
    ring with condensate charge \(q\).
    \item[(b)]
    Condensates carry charges \(q_i=n_iq_0\), where the integers \(n_i\)
    have greatest common divisor \(d\).  Determine the residual subgroup
    of the compact \(U(1)\), the smallest common nonzero flux, and the
    fundamental Josephson frequency allowed when all tunneling channels
    are present.
    \item[(c)]
    Two condensates have charges \(q_1,q_2\) with irrational
    \(q_1/q_2\).  Determine the intersection of their flux lattices.
    Construct a gauge-invariant relative phase, and explain what happens
    to flux quantization and to the junction current if no interaction
    locks this relative phase.
  \end{enumerate}

\end{enumerate}
