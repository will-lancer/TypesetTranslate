\chapterappendix{B}{A List of Homotopy Groups}
\phantomsection\label{app:23.B}

This appendix presents a list of homotopy groups~\hyperref[ch23-ref-40]{[40]}
for various manifolds. Here $\mathbb{Z}$ denotes the group of the integers,
with group composition defined by addition, so that zero is the unit element.
Also, $\mathbb{Z}_n$ is the group of the integers modulo $n$. The trivial
group, consisting of the element 0, is denoted 0. Homotopy groups for direct
products of manifolds may be obtained from the homotopy groups of the
manifolds themselves by the product rule:
\[
 \pi_n(\mathcal M_1\times\mathcal M_2)
 =\pi_n(\mathcal M_1)\times\pi_n(\mathcal M_2).
\]

\medskip
\noindent\textbf{Spheres}
\[
 \pi_n(S_m)=0\qquad\text{for }n<m
\]
\[
 \pi_n(S_n)=\mathbb{Z}
\]
\[
 \pi_{n+1}(S_n)=\mathbb{Z}_2
 \qquad\text{except }\pi_2(S_1)=0;\quad\pi_3(S_2)=\mathbb{Z}
\]
\[
 \pi_{n+2}(S_n)=\mathbb{Z}_2
 \qquad\text{except }\pi_3(S_1)=0
\]
\[
 \begin{aligned}
 \pi_{n+3}(S_n)&=\mathbb{Z}_{24}
 \qquad\text{except }\pi_4(S_1)=0;\quad
 \pi_5(S_2)=\mathbb{Z}_2;\quad\pi_6(S_3)=\mathbb{Z}_{12};\\
 \pi_7(S_4)&=\mathbb{Z}\times\mathbb{Z}_{12}
 \end{aligned}
\]
\[
 \pi_n(S_1)=0\qquad\text{except }\pi_1(S_1)=\mathbb{Z}
\]

\medskip
\noindent\textbf{Lie Group Manifolds}
\[
 \pi_1(G)=
 \begin{cases}
  \mathbb{Z}   & G=U(1)\\
  \mathbb{Z}_2 & G=SO(n)\quad(n\geq3)\\
  0            & \text{other simple compact connected Lie groups}
 \end{cases}
\]
\[
 \pi_2(G)=0\qquad G\text{ any compact connected Lie group}
\]
\[
 \pi_3(G)=\mathbb{Z}\qquad
 G\text{ any compact connected simple Lie group}
\]
\[
 \pi_4(G)=
 \begin{cases}
  \mathbb{Z}_2\times\mathbb{Z}_2
    & G=SO(4),\ \operatorname{Spin}(4)\\
  \mathbb{Z}_2
    & G=USp(2n),\ SU(2),\ SO(3),\ \operatorname{Spin}(5),\ SO(5)\\
  0
    & G=SU(n)\ (n\geq3),\ SO(n)\ (n\geq6),\ G_2,\ F_4,\ E_n
 \end{cases}
\]
\[
 \pi_{4n+2}\bigl(USp(2n)\bigr)=
 \begin{cases}
  \mathbb{Z}_{(2n+2)!}  & n\text{ even}\\
  \mathbb{Z}_{2(2n+2)!} & n\text{ odd}
 \end{cases}
 \qquad
 \pi_{2n}\bigl(SU(n)\bigr)=\mathbb{Z}_{n!}
\]

\medskip
\noindent\textbf{Bott Periodicity Theorems}

For $n\geq(k-1)/4$, $k\geq2$,
\[
 \pi_k\bigl(USp(2n)\bigr)=
 \begin{cases}
  \mathbb{Z}   & k=3,7\pmod 8\\
  \mathbb{Z}_2 & k=4,5\pmod 8\\
  0            & k=0,1,2,6\pmod 8
 \end{cases}
\]
For $n\geq k+2$, $k\geq2$,
\[
 \pi_k\bigl(SO(n)\bigr)=
 \begin{cases}
  \mathbb{Z}   & k=3,7\pmod 8\\
  \mathbb{Z}_2 & k=0,1\pmod 8\\
  0            & k=2,4,5,6\pmod 8
 \end{cases}
\]
For $n\geq(k+1)/2$, $k\geq2$,
\[
 \pi_k\bigl(SU(n)\bigr)=
 \begin{cases}
  \mathbb{Z} & k\text{ odd}\\
  0          & k\text{ even}
 \end{cases}
\]

\medskip
\noindent\textbf{Coset Spaces}

For any Lie group $G$ and any Lie subgroup $H\subset G$,
\[
 \pi_2(G/H)=\ker\{\pi_1(H)\mapsto\pi_1(G)\}.
\]
That is, $\pi_2(G/H)$ is the subgroup of $\pi_1(H)$ that maps into the
trivial element of $\pi_1(G)$ when $H$ is embedded in $G$. As a special
case,
\[
 \pi_2(G/H)=\pi_1(H)\qquad\text{for }\pi_1(G)=0.
\]
