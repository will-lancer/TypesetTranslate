\chapterproblems

\begin{enumerate}[label=\arabic*.,leftmargin=*]
\item Show that the Robertson--Walker metric, Eq.~\eqref{eq:5.1.11}, can be
expressed in the form
\begin{equation*}
  \dd s^2=-\dd\tau^2+a^2(\tau)\left[
    \frac{\dd r^2}{1-kr^2}+r^2\bigl(\dd\theta^2+\sin^2\theta\,\dd\phi^2\bigr)
  \right].
\end{equation*}
What portion of the 3-sphere (\(k=+1\)) is covered by these coordinates?

\item Derive Einstein's equations, Eqs.~\eqref{eq:5.2.14} and \eqref{eq:5.2.15},
for the 3-sphere (\(k=+1\)) and hyperboloid (\(k=-1\)) cases.

\item
\begin{enumerate}[label=\alph*),leftmargin=2.5em]
\item Consider the modified Einstein's equation \eqref{eq:5.2.17} with cosmological
constant \(\Lambda\). Write out the analogs of Eqs.~\eqref{eq:5.2.14} and
\eqref{eq:5.2.15} with the \(\Lambda\)-terms. Show that static solutions of these
equations are possible if and only if \(k=+1\) (3-sphere) and \(\Lambda>0\).
(These solutions are called \emph{Einstein static universes}.) For a dust filled
Einstein universe (\(P=0\)), relate the ``radius of the universe'' \(a\) to the density
\(\rho\). Examine small perturbations from this ``equilibrium'' value of \(a\), and
show that the Einstein static universe is unstable.

\par\medskip

\item Consider the modified Einstein equation with \(\Lambda>0\) and \(T_{ab}=0\).
Obtain the spatially homogeneous, isotropic solution in the case \(k=0\). (The
resulting spacetime actually is spacetime homogeneous and isotropic and is known as
the \emph{de Sitter spacetime}. The solutions with \(k=\pm1\) correspond to different
choices of spacelike hypersurfaces in this spacetime. See Hawking and Ellis 1973 for
further details.)
\end{enumerate}

\item Derive the cosmological redshift formula \eqref{eq:5.3.6} by the following
argument:
\begin{enumerate}[label=\alph*),leftmargin=2.5em]
\item Show that \(\nabla_a u_b=(\dot a/a)h_{ab}\), where \(h_{ab}\) is defined by
Eq.~\eqref{eq:5.1.10} and \(\dot a\equiv \dd a/\dd\tau\).

\item Show that along any null geodesic we have
\(\dd\omega/\dd\lambda=-k^a k^b\nabla_a u_b=-(\dot a/a)\omega^2\), where \(\lambda\)
is an affine parameter along the geodesic.

\item Show that the result of (b) yields Eq.~\eqref{eq:5.3.6}.
\end{enumerate}

\item Consider a radial \((\dd\theta/\dd\lambda=\dd\phi/\dd\lambda=0)\) null geodesic
propagating in a Robertson--Walker cosmology, Eq.~\eqref{eq:5.1.11}.
\begin{enumerate}[label=\alph*),leftmargin=2.5em]
\item Show that for all three spatial geometries the change in the coordinate \(\psi\)
of the ray, between times \(\tau_1\) and \(\tau_2\), is
\begin{equation*}
  \Delta\psi=\int_{\tau_1}^{\tau_2}\frac{\dd\tau}{a(\tau)}.
\end{equation*}
[Here, in the flat case (\(k=0\)), \(\psi\) is defined to be the ordinary radial
coordinate, \(\psi=(x^2+y^2+z^2)^{1/2}\).]

\item Show that in the dust filled spherical model, a light ray emitted at the big bang
travels precisely all the way around the universe by the time of the ``big crunch.''

\item Show that in the radiation filled spherical model, a light ray emitted at the big
bang travels precisely halfway around the universe by the time of the ``big crunch.''
\end{enumerate}
\end{enumerate}
