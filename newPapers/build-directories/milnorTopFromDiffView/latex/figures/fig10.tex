% Figure 10: A nonzero vector field on the 1-sphere
\begin{tikzpicture}[line cap=round,line join=round,>=Stealth,scale=0.9]
  \draw[thick] (0,0) circle (1.25);
  \foreach \a in {0,45,...,315} {
    \draw[->] ({1.25*cos(\a)},{1.25*sin(\a)})
      -- ({1.25*cos(\a)-0.38*sin(\a+14)},{1.25*sin(\a)+0.38*cos(\a+14)});
  }
  \node at (0,-1.7) {$S^1$};
\end{tikzpicture}
