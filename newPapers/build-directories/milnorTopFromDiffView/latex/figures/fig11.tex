% Figure 11: Attempts for n = 2
\begin{tikzpicture}[line cap=round,line join=round,>=Stealth,scale=0.9,
    midarrow/.style={postaction={decorate,
      decoration={markings,mark=at position #1 with {\arrow{Stealth}}}}}]
  % left sphere: meridian flow from pole to pole (two zeros)
  \begin{scope}[shift={(-2.0,0)}]
    \draw[thick] (0,0) circle (1.15);
    \foreach \k in {-0.78,-0.42,0,0.42,0.78} {
      \draw[thin,midarrow=0.55] (0,1.15)
        .. controls ({\k*1.6},0.5) and ({\k*1.6},-0.5) .. (0,-1.15);
    }
    \fill (0,1.15) circle (1.6pt);
    \node[above] at (0,1.2) {\footnotesize zero};
    \fill (0,-1.15) circle (1.6pt);
    \node[below] at (0,-1.2) {\footnotesize zero};
  \end{scope}
  % right sphere: dipole flow with a single zero
  \begin{scope}[shift={(2.0,0)}]
    \draw[thick] (0,0) circle (1.15);
    \foreach \r in {0.28,0.56,0.86} {
      \draw[thin,midarrow=0.3,midarrow=0.8] (0,{1.15-\r}) circle (\r);
    }
    \fill (0,1.15) circle (1.6pt);
  \end{scope}
\end{tikzpicture}
