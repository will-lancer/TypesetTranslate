% Figure 4 (Brouwer): f(x) is the point of the boundary nearer x
% on the line through x and g(x)
\begin{tikzpicture}[line cap=round,line join=round,>=Stealth,scale=0.9]
  \draw[thick] (0,0) circle (1.7);
  \coordinate (F) at ({1.7*cos(192)},{1.7*sin(192)});
  \coordinate (X) at (-0.55,-0.13);
  \coordinate (G) at (0.6,0.075);
  \draw[thick] (F) -- (G);
  \fill (F) circle (1.7pt);
  \node[left] at ({1.7*cos(192)-0.06},{1.7*sin(192)}) {$f(x)$};
  \fill (X) circle (1.7pt);
  \node[above] at (-0.55,-0.03) {$x$};
  \fill (G) circle (1.7pt);
  \node[above] at (0.62,0.17) {$g(x)$};
\end{tikzpicture}
