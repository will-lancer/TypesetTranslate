% Figure 7: Deforming the unit ball
\begin{tikzpicture}[line cap=round,line join=round,>=Stealth,scale=0.9]
  % left: square grid with inscribed circle
  \draw[very thin,gray!70] (0,0) grid[step=0.4] (2.4,2.4);
  \draw[thick] (0,0) rectangle (2.4,2.4);
  \draw[thick] (1.2,1.2) circle (1.2);
  \fill (2.4,1.2) circle (1.6pt);
  \node at (2.62,0.95) {\footnotesize $c$};
  % squiggly arrow F_t
  \draw[->,thick,decorate,decoration={snake,amplitude=0.5mm,segment length=3mm,post length=1.5mm}]
    (2.95,1.2) -- (4.15,1.2);
  \node at (3.55,1.6) {$F_t$};
  % right: warped grid
  \begin{scope}[shift={(4.6,0)}]
    \foreach \i in {1,...,5} {
      \draw[very thin,gray!70] ({0.4*\i},0)
        .. controls ({0.4*\i+0.22*sin(85*\i)},0.8) and ({0.4*\i-0.22*sin(85*\i)},1.6)
        .. ({0.4*\i},2.4);
    }
    \foreach \j in {1,...,5} {
      \draw[very thin,gray!70] (0,{0.4*\j})
        .. controls (0.8,{0.4*\j+0.22*cos(70*\j)}) and (1.6,{0.4*\j-0.22*cos(70*\j)})
        .. (2.4,{0.4*\j});
    }
    \draw[thick] (0,0) rectangle (2.4,2.4);
  \end{scope}
\end{tikzpicture}
