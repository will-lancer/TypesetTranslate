The pair $(N, \mathfrak{v})$ is called a \emph{framed submanifold} of $M$. Two framed submanifolds $(N, \mathfrak{v})$ and $(N', \mathfrak{w})$ are \emph{framed cobordant} if there exists a cobordism $X \subset M \times [0, 1]$ between $N$ and $N'$ and a framing $\mathfrak{u}$ of $X$, so that
\[
u^i(x, t) = (v^i(x), 0) \quad \text{for } (x, t) \in N \times [0, \epsilon)
\]
\[
u^i(x, t) = (w^i(x), 0) \quad \text{for } (x, t) \in N' \times (1 - \epsilon, 1].
\]
Again this is an equivalence relation.

Now consider a smooth map $f : M \to S^p$ and a regular value $y \in S^p$. The map $f$ induces a framing of the manifold $f^{-1}(y)$ as follows: Choose a positively oriented basis $\mathfrak{v} = (v^1, \ldots, v^p)$ for the tangent space $T(S^p)_y$. For each $x \in f^{-1}(y)$ recall from Lemma~2 of \S2 that
\[
\dd f_x : TM_x \to T(S^p)_y
\]
maps the subspace $Tf^{-1}(y)_x$ to zero and maps its orthogonal complement $Tf^{-1}(y)_x^\perp$ isomorphically onto $T(S^p)_y$. Hence there is a unique vector
\[
w^i(x) \in Tf^{-1}(y)_x^\perp \subset TM_x
\]
that maps into $v^i$ under $\dd f_x$. It will be convenient to use the notation $\mathfrak{w} = f^*\mathfrak{v}$ for the resulting framing $w^1(x), \ldots, w^p(x)$ of $f^{-1}(y)$.

\textsc{Definition.} This framed manifold $(f^{-1}(y), f^*\mathfrak{v})$ will be called the \emph{Pontryagin manifold} associated with $f$.

Of course $f$ has many Pontryagin manifolds, corresponding to different choices of $y$ and $\mathfrak{v}$, but they all belong to a single framed cobordism class:

\noindent\textbf{Theorem A.} \textit{If $y'$ is another regular value of $f$ and $\mathfrak{v}'$ is a positively oriented basis for $T(S^p)_{y'}$, then the framed manifold $(f^{-1}(y'), f^*\mathfrak{v}')$ is framed cobordant to $(f^{-1}(y), f^*\mathfrak{v})$.}

\noindent\textbf{Theorem B.} \textit{Two mappings from $M$ to $S^p$ are smoothly homotopic if and only if the associated Pontryagin manifolds are framed cobordant.}

\noindent\textbf{Theorem C.} \textit{Any compact framed submanifold $(N, \mathfrak{w})$ of codimension $p$ in $M$ occurs as Pontryagin manifold for some smooth mapping $f : M \to S^p$.}

Thus the homotopy classes of maps are in one-one correspondence with the framed cobordism classes of submanifolds.

The proof of Theorem~A will be very similar to the arguments in \S\S4 and~5. It will be based on three lemmas.

\noindent\textbf{Lemma 1.} \textit{If $\mathfrak{v}$ and $\mathfrak{v}'$ are two different positively oriented bases at $y$, then the Pontryagin manifold $(f^{-1}(y), f^*\mathfrak{v})$ is framed cobordant to $(f^{-1}(y), f^*\mathfrak{v}')$.}

\noindent\textit{Proof.}
Choose a smooth path from $\mathfrak{v}$ to $\mathfrak{v}'$ in the space of all positively oriented bases for $T(S^p)_y$. This is possible since this space of bases


% PDF page 57

can be identified with the space $GL^+(p, \mathbb{R})$ of matrices with positive determinant, and hence is connected. Such a path gives rise to the required framing of the cobordism $f^{-1}(y) \times [0, 1]$.

By abuse of notation we will often delete reference to $f^*\mathfrak{v}$ and speak simply of ``the framed manifold $f^{-1}(y)$.''

\noindent\textbf{Lemma 2.} \textit{If $y$ is a regular value of $f$, and $z$ is sufficiently close to $y$, then $f^{-1}(z)$ is framed cobordant to $f^{-1}(y)$.}

\noindent\textit{Proof.}
Since the set $f(C)$ of critical values is compact, we can choose $\epsilon > 0$ so that the $\epsilon$-neighborhood of $y$ contains only regular values. Given $z$ with $\|z - y\| < \epsilon$, choose a smooth one-parameter family of rotations (i.e.\ an isotopy) $r_t : S^p \to S^p$ so that $r_1(y) = z$, and so that
\begin{enumerate}
\item $r_t$ is the identity for $0 \le t < \epsilon'$,
\item $r_t$ equals $r_1$ for $1 - \epsilon' < t \le 1$, and
\item each $r_t^{-1}(z)$ lies on the great circle from $y$ to $z$, and hence is a regular value of $f$.
\end{enumerate}

Define the homotopy
\[
F : M \times [0, 1] \to S^p
\]
by $F(x, t) = r_t f(x)$. For each $t$ note that $z$ is a regular value of the composition
\[
r_t \circ f : M \to S^p.
\]
It follows a fortiori that $z$ is a regular value for the mapping $F$. Hence
\[
F^{-1}(z) \subset M \times [0, 1]
\]
is a framed manifold and provides a framed cobordism between the framed manifolds $f^{-1}(z)$ and $(r_1 \circ f)^{-1}(z) = f^{-1} r_1^{-1}(z) = f^{-1}(y)$. This proves Lemma~2.

\noindent\textbf{Lemma 3.} \textit{If $f$ and $g$ are smoothly homotopic and $y$ is a regular value for both, then $f^{-1}(y)$ is framed cobordant to $g^{-1}(y)$.}

\noindent\textit{Proof.}
Choose a homotopy $F$ with
\[
F(x, t) = f(x) \quad \text{for } 0 \le t < \epsilon,
\]
\[
F(x, t) = g(x) \quad \text{for } 1 - \epsilon < t \le 1.
\]
Choose a regular value $z$ for $F$ which is close enough to $y$ so that $f^{-1}(z)$ is framed cobordant to $f^{-1}(y)$ and so that $g^{-1}(z)$ is framed cobordant to $g^{-1}(y)$. Then $F^{-1}(z)$ is a framed manifold and provides a framed cobordism between $f^{-1}(z)$ and $g^{-1}(z)$. This proves Lemma~3.
