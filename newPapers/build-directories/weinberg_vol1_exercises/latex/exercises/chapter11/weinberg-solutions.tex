\chapterbackmatter{Solutions to Weinberg Exercises}
\ExerciseEditorialNote

\begin{enumerate}
\WeinbergSolution{1}
Both scalar-QED graphs must be retained: the loop with two three-point
vertices and the seagull graph.  Their non-transverse pieces cancel.  With
the normalization
\[
 \Pi_s^{\mu\nu}(q)
 =(q^2\eta^{\mu\nu}-q^\mu q^\nu)\pi_s(q^2),
 \qquad \pi_s(0)=0,
\]
the result for one charged complex scalar is
\[
 \boxed{\;
 \pi_s(q^2)=\frac{e^2}{16\pi^2}
 \int_0^1dx\,(1-2x)^2
 \ln\left(1+\frac{q^2x(1-x)}{m_s^2}\right) .
 \;}
\]
For comparison, the coefficient of the large-\(q^2\) logarithm is
\(e^2/(48\pi^2)\), one quarter of the coefficient from a Dirac lepton.
Dimensional regularization gives the corresponding divergent part
\[
 \boxed{\qquad
 (Z_3-1)_{\rm div}
 =\frac{e^2}{24\pi^2}\frac{1}{d-4}.
 \qquad}
\]
As in the fermion calculation, finite terms in \(Z_3\) depend on the
definition of the \(d\)-dimensional scale, while \(\pi_s(q^2)\) above does
not.

For \(q^2\ll m_s^2\),
\[
 \pi_s(q^2)
 =\frac{e^2q^2}{16\pi^2m_s^2}
 \int_0^1dx\,(1-2x)^2x(1-x)+O(q^4)
 =\frac{e^2q^2}{480\pi^2m_s^2}+O(q^4).
\]
The local potential shift therefore satisfies
\[
 \int d^3r\,\Delta V(\mathbf r)
 =-Ze^2\pi_s'(0)=-\frac{Z\alpha^2}{30m_s^2}.
\]
Using
\(\lvert\psi_{n0}(0)\rvert^2=(Z\alpha m_e)^3/(\pi n^3)\), we obtain
\[
 \boxed{\;
 \Delta E_{n s}^{(s)}
 =-\frac{Z^4\alpha^5m_e^3}{30\pi n^3m_s^2},
 \qquad
 \Delta E_{2s}^{(s)}
 =-\frac{\alpha^5m_e^3}{240\pi m_s^2}\quad(Z=1).
 \;}
\]
The assumed hierarchy \(m_s\gg Z\alpha m_e\) is precisely what permits the
replacement of the short-range polarization potential by this contact term.

\WeinbergSolution{2}
Let \(r=m_\phi/m_e\).  The one-loop vertex graph with a scalar and two
electron propagators is ultraviolet finite in its Pauli form factor.  After
combining denominators and using the on-shell Dirac equations, it gives
\[
 \boxed{\;
 \delta a_e\equiv\frac{g_e-2}{2}
 =\frac{g^2}{8\pi^2}\int_0^1dz\,
 \frac{(1-z)^2(1+z)}
 {(1-z)^2+zr^2}.
 \;}
\]
Thus the scalar correction to the magnetic moment is
\(\delta\mu_e=(e/2m_e)\delta a_e\).  It is positive.  Useful checks are
\[
 \delta a_e\longrightarrow\frac{3g^2}{16\pi^2}
 \quad(r\to0),
 \qquad
 \delta a_e=\frac{g^2}{8\pi^2}\frac{m_e^2}{m_\phi^2}
 \left[\ln\frac{m_\phi^2}{m_e^2}-\frac76
+O\left(\frac{m_e^2}{m_\phi^2}\ln r\right)\right]
 \quad(r\gg1).
\]

For completeness, the wave-function effect can be displayed without hiding
the subtraction convention.  Put
\[
 C_d=\frac{2}{4-d}-\gamma+\ln4\pi,\qquad
 \Delta(x,p^2)=xm_\phi^2+(1-x)m_e^2+x(1-x)p^2 .
\]
With the self-energy convention
\[
 S'^{-1}(p)=i\sl p+m_e-\Sigma_\phi(p),
\]
\[
 \Sigma_\phi(p)=-\frac{g^2}{16\pi^2}\int_0^1dx\,
 \bigl[-ix\sl p+m_e\bigr]
 \left[C_d-\ln\frac{\Delta(x,p^2)}{\mu^2}\right],
\]
terms vanishing as \(d\to4\) being omitted, the on-shell residue condition
is
\[
 Z_2-1=-i\left.
 \frac{\partial\Sigma_\phi}{\partial\sl p}\right|_{i\sl p=-m_e}.
\]
Writing
\(\Delta_{\rm os}(x)=xm_\phi^2+(1-x)^2m_e^2\), this becomes
\[
 \boxed{\;
 Z_2-1=\frac{g^2}{16\pi^2}\int_0^1dx
 \left[
 x\left(C_d-\ln\frac{\Delta_{\rm os}(x)}{\mu^2}\right)
 +
 \frac{2m_e^2x(1-x)(1+x)}{\Delta_{\rm os}(x)}
 \right].
 \;}
\]
Unlike the Pauli form factor, \(Z_2\) contains an ultraviolet pole and is
scheme dependent.

\WeinbergSolution{3}
It is clearest to state the answer in terms of renormalized one-particle
irreducible functions.  For a four-momentum \(P\), define
\[
 D(P)=P^2+m_\phi^2-i\epsilon
\]
and define the renormalized self-energy \(\Pi_R\) by the on-shell conditions
\[
 \Pi_R(-m_\phi^2)=0,\qquad
 \Pi_R'(-m_\phi^2)=0.
\]
The one-loop building blocks are
\[
 i\Pi_{\rm loop}(P)
 =\frac12(-ig)^2\int\frac{d^dk}{(2\pi)^d}
 \frac{i}{D(k)}\frac{i}{D(P-k)},
\]
\[
 i\Gamma_{\triangle}(P_1,P_2,P_3)
 =(-ig)^3\int\frac{d^4k}{(2\pi)^4}
 \frac{i}{D(k)}
 \frac{i}{D(k+P_1)}
 \frac{i}{D(k-P_2)} .
\]
The triangle is ultraviolet finite in four dimensions.  Its finite
renormalized version \(\Gamma_R\) includes the chosen finite definition of
\(g\).  The box for a cyclic ordering \((P_1,P_2,P_3,P_4)\) is
\[
 iB(P_1,P_2,P_3,P_4)
 =(-ig)^4\int\frac{d^4k}{(2\pi)^4}
 \prod_{j=0}^{3}\frac{i}{D(k+K_j)},
\]
where \(K_0=0\), \(K_1=P_1\), \(K_2=P_1+P_2\), and
\(K_3=P_1+P_2+P_3\).  It too is ultraviolet finite.

Let all external momenta be incoming and put
\[
 P_s=p_1+p_2,\qquad P_t=p_1+p_3,\qquad P_u=p_1+p_4.
\]
Through order \(g^4\), the amputated on-shell amplitude is
\[
\begin{aligned}
 i\mathcal M={}&
 \sum_{X=s,t,u}\left[
 (-ig)^2\frac{i}{D(P_X)}
 +(-ig)^2\frac{i}{D(P_X)}
 i\Pi_R(P_X)\frac{i}{D(P_X)}
 \right]\\
 &+\sum_{X=s,t,u}
 \left[
 2(-ig)\frac{i}{D(P_X)}\,i\Gamma_R(P_X)
 \right]
 +iB(s,t)+iB(s,u)+iB(t,u)+O(g^6).
\end{aligned}
\]
Here \(\Gamma_R(P_X)\) denotes the triangle correction at either end of the
exchange line; the factor two counts the two ends.  The three \(B\)'s are
the inequivalent box orderings.  External self-energy insertions are absent
after LSZ because the pole residue is unity.  The only ultraviolet
divergence at this order lies in the two-point subgraph and is removed by
the mass and field-strength counterterms used in \(\Pi_R\).  This expression
is the complete one-loop \(S\)-matrix element, with the overall
momentum-conserving delta function supplied by the usual
\(\OutBra{p_3,p_4}S\InKet{p_1,p_2}\) normalization.

\WeinbergSolution{4}
Use the same self-energy convention as in Problem 2.  The mass counterterm
is fixed by the pole condition,
\[
 \delta m_e^{(\phi)}
 =-\left.\Sigma_\phi(p)\right|_{i\sl p=-m_e}.
\]
On the mass shell,
\[
 \Delta_{\rm os}(x)=xm_\phi^2+(1-x)^2m_e^2,\qquad
 [-ix\sl p+m_e]_{i\sl p=-m_e}=m_e(1+x),
\]
and hence
\[
 \boxed{\;
 \delta m_e^{(\phi)}
 =\frac{g^2m_e}{16\pi^2}\int_0^1dx\,(1+x)
 \left[
 C_d-\ln\frac{xm_\phi^2+(1-x)^2m_e^2}{\mu^2}
 \right].
 \;}
\]
This is the bare on-shell counterterm in dimensional regularization.  In
the \(\overline{\rm MS}\) scheme the bracket \(C_d\) is subtracted; in an
on-shell scheme the full expression is used so that the dressed propagator
has its pole at the measured mass.  The ultraviolet pole has coefficient
\[
 \left.\delta m_e^{(\phi)}\right|_{\rm div}
 =\frac{3g^2m_e}{32\pi^2}C_d,
\]
which has the required mass dimension and vanishes with \(m_e\), as demanded
by the chiral symmetry restored when the electron mass is zero.
\end{enumerate}
