\SupplementaryExercise{1}{A Complex Gaussian and Its Pairings}
  {(Adapted from D. Harlow, \emph{Quantum Field Theory I}, 2024, Section 7.8, Homework Problem 5.)}
  {harlow-qft1-2024-s7p8-problem5-v1ch04-s01}
Let \(x\) be an \(n\)-component complex vector, let \(A\) be a positive
real symmetric matrix, and let \(B\) and \(\widetilde B\) be independent
complex vectors.  With the componentwise convention
\[
 \dd x\,\dd x^*\equiv\prod_{a=1}^n \dd x_a\,\dd x_a^*,
 \qquad
 \dd x_a\,\dd x_a^*=-2\ii\,d(\operatorname{Re}x_a)\,
 d(\operatorname{Im}x_a),
\]
show that
\[
 \int \dd x\,\dd x^*\,
 e^{-x^\dagger A x+B^{\mathsf T}x+\widetilde B^{\mathsf T}x^*}
 =\frac{1}{\det\!\left(\dfrac{\ii A}{2\pi}\right)}
 e^{\widetilde B^{\mathsf T}A^{-1}B}.
\]
Use this identity to evaluate the normalized Gaussian correlators
\[
 \left\langle
 x_{i_1}\cdots x_{i_M}
 x^*_{j_1}\cdots x^*_{j_N}
 \right\rangle_A
 \equiv
 \frac{\displaystyle\int \dd x\,\dd x^*\,
 e^{-x^\dagger A x}
 x_{i_1}\cdots x_{i_M}x^*_{j_1}\cdots x^*_{j_N}}
 {\displaystyle\int \dd x\,\dd x^*\,e^{-x^\dagger A x}}.
\]
Prove that the answer vanishes unless \(M=N\), and that when \(M=N\)
it is the sum over every pairing in which each unstarred variable is joined
to one starred variable.  State the factor contributed by each joined pair.

The source prints \(j_M\), rather than \(j_N\), as the final starred index
on the right-hand side of its correlator formula even though its left-hand
side contains \(N\) starred variables.  Diagnose this mismatch before giving
the corrected general formula.  For \(M=N=2\), the two allowed arrow
patterns are
\[
\begin{tikzpicture}[baseline=-0.4ex,scale=.78,
  every node/.style={font=\small},
  pair/.style={->,>=stealth,thick}]
 \node (x1) at (0,0.5) {$x_{i_1}$};
 \node (x2) at (0,-0.5) {$x_{i_2}$};
 \node (y1) at (2.0,0.5) {$x^*_{j_1}$};
 \node (y2) at (2.0,-0.5) {$x^*_{j_2}$};
 \draw[pair] (x1)--(y1);
 \draw[pair] (x2)--(y2);
 \node at (2.75,0) {$+$};
 \node (u1) at (3.5,0.5) {$x_{i_1}$};
 \node (u2) at (3.5,-0.5) {$x_{i_2}$};
 \node (v1) at (5.5,0.5) {$x^*_{j_1}$};
 \node (v2) at (5.5,-0.5) {$x^*_{j_2}$};
 \draw[pair] (u1)--(v2);
 \draw[pair] (u2)--(v1);
\end{tikzpicture}
\]

\SupplementaryExercise{2}{Generating Connected Contributions and Their Volume Scaling}
  {(Inspired by D. Harlow, \emph{Quantum Field Theory I}, 2024, Section 7.4 | D. Harlow, \emph{Quantum Field Theory I}, 2024, Section 7.5 | D. Harlow, \emph{Quantum Field Theory I}, 2024, Sections 7.4--7.5.)}
  {harlow-qft1-linked-2024-curated-parent-2}
\begin{enumerate}
\item
Suppose each connected diagram type \(C\) has value \(V_C\), and a
disconnected diagram contains \(m_C\) identical copies of each type.
Show that its combinatorial weight is
\(\prod_C V_C^{m_C}/m_C!\), and prove that the sum over all diagrams is
\(\exp(\sum_CV_C)\).
\item
Let \(Z[J]\) be normalized by \(Z[0]=1\), and define \(W[J]=\log Z[J]\).
Show through fourth order in \(J\) that functional derivatives of \(W\)
at \(J=0\) give connected correlation functions, while derivatives of
\(Z\) give the full functions.
\item
In a translationally invariant theory in a periodic box of volume \(V\),
show that a connected momentum-space amplitude carries one momentum
Kronecker delta and scales as \(V\), whereas a contribution with \(C\)
disconnected components scales as \(V^C\).  Explain why the logarithm is
extensive.
\end{enumerate}

\SupplementaryExercise{3}{Vacuum Cancellation, Connectedness, and Momentum Delta Functions}
  {(Inspired by D. Tong, \emph{Lectures on Quantum Field Theory}, 2006, Section 3.7.1 | D. Tong, \emph{Lectures on Quantum Field Theory}, 2006, Sections 3.5.4 and 3.7.1 | D. Tong, \emph{Lectures on Quantum Field Theory}, 2006, Sections 3.4.1 and 3.5.4.)}
  {tong-qft-connected-2006-curated-parent-1}
\begin{enumerate}
\item
Use the Gell-Mann--Low ratio for an \(n\)-point function to prove that all
vacuum bubbles cancel between numerator and denominator.  Distinguish a
vacuum bubble from a diagram disconnected only with respect to external
points.
\item
Give graph-theoretic definitions of a connected graph and a
one-particle-irreducible graph.  Construct a connected two-point graph
that becomes disconnected when one internal line is cut, and explain why
it contributes to a connected correlator.
\item
For a diagram with \(V\) vertices, \(I\) internal lines, \(L\) loop
momenta, and \(C\) connected components, prove
\(V-I+L=C\).  Use momentum conservation at every vertex to show that
exactly \(C\) independent momentum delta functions remain.
\end{enumerate}

\SupplementaryExercise{4}{Microcausality, Statistics, and Cluster Decomposition}
  {(Adapted from D. Harlow, \emph{Quantum Field Theory I}, 2024, Section 6.6, Homework Problem 5.)}
  {harlow-qft1-2024-s6p6-problem5-v1ch04-s04}
Suppose that at spacelike separation a nontrivial local field obeys
\[
 \Phi(x)\Phi^\dagger(y)
 \mathbin{\pm}\Phi^\dagger(y)\Phi(x)=0.
\]
Show that the same choice of sign must occur in both of the relations
\[
 \Phi(x)\Phi(y)\mathbin{\pm}\Phi(y)\Phi(x)=0,
 \qquad
 \Phi^\dagger(x)\Phi^\dagger(y)
 \mathbin{\pm}\Phi^\dagger(y)\Phi^\dagger(x)=0
\]
at spacelike separation.

\emph{Hint.}  Write the first same-field relation provisionally as
\[
 \Phi(x)\Phi(y)+s\Phi(y)\Phi(x)=0,
 \qquad s=\pm1,
\]
and determine \(s\) by considering the norm of
\(\Phi(x)\Phi(y)\ket\Omega\).  You will need the Reeh--Schlieder
property and the large-separation limit
\[
 \left\langle
 \Phi^\dagger(x)\Phi(x)\Phi^\dagger(y)\Phi(y)
 \right\rangle
 \longrightarrow
 \left\langle\Phi^\dagger(x)\Phi(x)\right\rangle
 \left\langle\Phi^\dagger(y)\Phi(y)\right\rangle
 \quad\text{as }(x-y)^2\to+\infty.
\]
This is cluster decomposition: connected correlators of local operators
decay at large separation.  Here it is being applied to the connected
two-point function of the composite local operator \(\Phi^\dagger\Phi\).

\SupplementaryExercise{5}{Wick's Theorem for Three Fields}
  {(Adapted from B. Mistlberger and K. Zhou, Stanford PHYS-330, Problem Set 4, 2022, Problem 1.)}
  {zhou-stanford-phys330-ps4-2022-curated-parent-1}
You may use Wick's theorem for a product of two free real scalar fields.
\begin{enumerate}
\item
For a free real scalar, show separately that
\(T\{\phi(x_1)\phi(x_2)\}\) and
\(:\!\phi(x_1)\phi(x_2)\!:\) are unchanged under
\(x_1\leftrightarrow x_2\).  Use
\[
 T\{\phi(x_1)\phi(x_2)\}
 =:\!\phi(x_1)\phi(x_2)\!:+D_F(x_1-x_2)
\]
to prove \(D_F(x_1-x_2)=D_F(x_2-x_1)\).
\item
By separating each field into creation and annihilation parts, prove
\[
\begin{split}
T\{\phi(x_1)\phi(x_2)\phi(x_3)\}
={}&:\!\phi(x_1)\phi(x_2)\phi(x_3)\!:
 +\phi(x_1)D_F(x_2-x_3)\\
&+\phi(x_2)D_F(x_3-x_1)
 +\phi(x_3)D_F(x_1-x_2).
\end{split}
\]
\end{enumerate}

\SupplementaryExercise{6}{Cubic-Theory Vacuum Graphs and Normalized Correlators}
  {(Adapted from B. Mistlberger and K. Zhou, Stanford PHYS-330, Problem Set 4, 2022, Problem 2.)}
  {zhou-stanford-phys330-ps4-2022-curated-parent-2}
Consider a real scalar field of mass \(m\) with
\[
 \mathcal L_0=-\frac12(\partial_\mu\phi)(\partial^\mu\phi)
 -\frac12m^2\phi^2,
 \qquad
 \mathcal L_{\rm int}=\frac{\lambda}{3!}\phi^3.
\]
Let \(\ket0\) be the free vacuum,
\(\mathcal H_{\rm int}=-\mathcal L_{\rm int}\), and \(H_I(t)\) the
interaction-picture Hamiltonian.
\begin{enumerate}
\item
Use Wick's theorem to evaluate
\[
 \bra0T\exp\!\left[-\ii\int \dd t\,H_I(t)\right]\ket0
\]
through order \(\lambda^2\), as an explicit sum of position-space
propagator integrals.  Terms containing \(D_F(0)\) should be retained.
\item
Draw every distinct vacuum diagram at orders \(\lambda^3\) and
\(\lambda^4\).  No evaluation or symmetry-factor calculation is required
in this part.
\item
For the interacting-vacuum two-point function
\[
 \bra\Omega T\{\phi_H(x)\phi_H(y)\}\ket\Omega
 =
 \frac{\bra0T\{\phi_I(x)\phi_I(y)
 \exp[-\ii\int \dd t\,H_I(t)]\}\ket0}
 {\bra0T\exp[-\ii\int \dd t\,H_I(t)]\ket0},
\]
draw every contributing diagram through order \(\lambda^4\) and give
its symmetry factor.  Include diagrams with more than one component
when every component touches \(x\) or \(y\), but omit vacuum components.
As a check, thirteen diagrams in the complete list are fully connected,
meaning that every field is connected to both external points.
\end{enumerate}

\SupplementaryExercise{7}{A Classical Source and Coherent In--Out States}
  {(Adapted from A. Guth, MIT 8.323, Problem Set 5, 2008, Problem 3.)}
  {mit-823-s08-ps5-curated-parent-1}
For a quantized scalar field driven by a fixed classical source \(j(x)\),
suppose the in- and out-fields obey
\[
 \phi_{\rm out}(x)=S^{-1}\phi_{\rm in}(x)S,\qquad
 a_{\rm out}(\mathbf p)=S^{-1}a_{\rm in}(\mathbf p)S.
\]
Write
\[
 \mu=\int\frac{\dd^3p}{(2\pi)^3}\,
       \frac{|\widetilde j(p)|^2}{2\omega_{\mathbf p}},
 \qquad
 \widetilde j(p)=\int \dd^4y\,e^{\ii p\cdot y}j(y),
\]
and take
\[
 S=e^{-\mu/2}e^Fe^G,
\]
\[
 F=\int\frac{\dd^3p}{(2\pi)^3\sqrt{2\omega_{\mathbf p}}}\,
       \widetilde j(p)a_{\rm in}^\dagger(\mathbf p),
 \qquad
 G=\int\frac{\dd^3p}{(2\pi)^3\sqrt{2\omega_{\mathbf p}}}\,
       \widetilde j(-p)a_{\rm in}(\mathbf p).
\]
\begin{enumerate}
\item
Define \(F'\) and \(G'\) by replacing the in-operators in \(F,G\) with
out-operators.  Prove
\[
 S=e^{-\mu/2}e^{F'}e^{G'}
\]
and hence
\[
 \InKet{0}=e^{-\mu/2}e^{F'}\OutKet{0}.
\]
\item
For one harmonic oscillator,
\[
 H=\frac{p^2}{2m}+\frac{m\omega^2q^2}{2},
\]
make the canonical rescaling
\(p=\sqrt{m\omega}\,\bar p\),
\(q=\bar q/\sqrt{m\omega}\), drop the bars, and use
\[
 a^\dagger=\frac{q-\ii p}{\sqrt2},\qquad
 a=\frac{q+\ii p}{\sqrt2}.
\]
For the unnormalized coherent state
\(\ket z=e^{za^\dagger}\ket0\), show that it is an eigenstate of \(a\)
and determine the eigenvalue.
\item
Compute \(\braket{z_2}{z_1}\).
\item
Using expectation values normalized by \(\braket z z\), find
\(\langle q\rangle_z\) and \(\langle p\rangle_z\).
\item
Compute \((\Delta q)^2\) and \((\Delta p)^2\), and show
\(\Delta q\,\Delta p=1/2\).
\end{enumerate}

\SupplementaryExercise{8}{Recovering a Classical Field from an Impulse}
  {(Adapted from B. Mistlberger and K. Zhou, Stanford PHYS-330, Problem Set 3, 2022, Problem 2.)}
  {zhou-stanford-phys330-ps3-2022-curated-parent-1}
Let a free real scalar field couple to a classical source through
\[
 H(t)=H_0-\int \dd^3x\,\phi(\mathbf x,t)J(\mathbf x,t).
\]
For an operator with no explicit Schr\"odinger-picture time dependence,
use the Heisenberg equation \(\dot O=\ii[H,O]\).
\begin{enumerate}
\item
Derive the sourced Klein--Gordon equation in the local mostly-plus
convention,
\[
 (\Box-m^2)\phi(x)=-J(x).
\]
\item
Starting in the vacuum, apply
\[
 J(\mathbf x,t)=\delta(t)j(\mathbf x).
\]
With \(0^\pm\) denoting the limits just after and before the impulse,
show
\[
 \phi(\mathbf x,0^+)=\phi(\mathbf x,0^-),\qquad
 \pi(\mathbf x,0^+)=\pi(\mathbf x,0^-)+j(\mathbf x).
\]
\item
For
\[
 \phi(\mathbf x,0)=
 \int\frac{\dd^3p}{(2\pi)^3\sqrt{2\omega_{\mathbf p}}}
 \left[a(\mathbf p,0)e^{\ii\mathbf p\cdot\mathbf x}
 +a^\dagger(\mathbf p,0)e^{-\ii\mathbf p\cdot\mathbf x}\right],
\]
show
\[
 a(\mathbf p,0^+)=a(\mathbf p,0^-)
 +\frac{\ii\,\widetilde j(\mathbf p)}{\sqrt{2\omega_{\mathbf p}}},
\qquad
 \widetilde j(\mathbf p)=\int \dd^3x\,
 e^{-\ii\mathbf p\cdot\mathbf x}j(\mathbf x).
\]
\item
Show that the state is a multimode coherent state with respect to
\(a(\mathbf p,0^+)\), and evaluate its produced-particle momentum
density
\(\langle a^\dagger(\mathbf p,0^+)a(\mathbf q,0^+)\rangle\).
\end{enumerate}

\SupplementaryExercise{9}{Relativistic Propagators and Causal Support}
  {(Adapted from B. Mistlberger and K. Zhou, Stanford PHYS-330, Problem Set 3, 2022, Problem 1.)}
  {zhou-stanford-phys330-ps3-2022-curated-parent-2}
For a free real scalar, define
\[
 D(z)=\bra0\phi(x)\phi(y)\ket0,\qquad z=x-y.
\]
\begin{enumerate}
\item
Starting from the free-field expansion, show
\[
 D(z)=\int\frac{\dd^4p}{(2\pi)^3}\,
 \delta(p^2+m^2)\theta(p^0)e^{\ii p\cdot z}.
\]
\item
For a proper orthochronous Lorentz transformation, prove
\(D(\Lambda z)=D(z)\).  In particular, verify the invariance of
\(\dd^4p\) and \(\delta(p^2+m^2)\theta(p^0)\).
\item
At spacelike separation \(z^2>0\), derive
\[
 D(z)=\frac{m}{4\pi^2\sqrt{z^2}}\,
 K_1\!\left(m\sqrt{z^2}\right)
\]
or verify it by numerical integration, and extract the large-distance
exponential falloff.  Standard tables or computer algebra may be used.

Although \(D(z)\) is nonzero at spacelike separation, relativistic
locality concerns the propagation of a change in the state, not the
existence of vacuum correlations.
\item
Show
\[
 \bra0[\phi(x),\phi(y)]\ket0=0
\]
when \(x-y\) is spacelike.
\item
Now take
\[
 \mathcal L=-\frac12(\partial_\mu\phi)(\partial^\mu\phi)
 -\frac12m^2\phi^2+\phi(x)J(x).
\]
Derive
\[
 (\Box-m^2)\phi(x)=-J(x).
\]
\item
Show that a particular solution is
\[
 \phi(x)=\ii\int \dd^4y\,G(x-y)J(y),
\]
provided
\[
 (\Box-m^2)G(z)=\ii\delta^4(z),
\]
and state the freedom to add a homogeneous solution.
\item
Using the residue theorem for the \(p^0\) integral, evaluate
\[
\begin{split}
G_R(z)&=\lim_{\epsilon\to0^+}\int\frac{\dd^4p}{(2\pi)^4}
\frac{-\ii\,e^{\ii p\cdot z}}
{-(p^0+\ii\epsilon)^2+\mathbf p^2+m^2},\\
G_A(z)&=\lim_{\epsilon\to0^+}\int\frac{\dd^4p}{(2\pi)^4}
\frac{-\ii\,e^{\ii p\cdot z}}
{-(p^0-\ii\epsilon)^2+\mathbf p^2+m^2},\\
G_F(z)&=\lim_{\epsilon\to0^+}\int\frac{\dd^4p}{(2\pi)^4}
\frac{-\ii\,e^{\ii p\cdot z}}{p^2+m^2-\ii\epsilon},
\end{split}
\]
and prove
\[
\begin{split}
G_R(z)&=\theta(z^0)\,[D(z)-D(-z)],\\
G_A(z)&=\theta(-z^0)\,[D(-z)-D(z)],\\
G_F(z)&=\theta(z^0)D(z)+\theta(-z^0)D(-z).
\end{split}
\]
\item
For a field that vanishes before the source acts, use \(G_R\) to show
that the response vanishes at spacelike separation from the source.
\item
Does \(G_F\) vanish at spacelike separation?  Explain why its answer is
consistent with causal signaling.
\end{enumerate}

\SupplementaryExercise{10}{The Force Between Static Sources}
  {(Adapted from B. Mistlberger and K. Zhou, Stanford PHYS-330, Problem Set 4, 2022, Problem 4.)}
  {zhou-stanford-phys330-ps4-2022-curated-parent-3}
Consider a free real scalar field coupled to a source,
\[
 \mathcal L=-\frac12(\partial_\mu\phi)(\partial^\mu\phi)
 -\frac12m^2\phi^2+\phi(x)J(x).
\]
\begin{enumerate}
\item
Using Wick's theorem, prove
\[
 \bra0T\exp\!\left(\ii\int \dd^4x\,\phi(x)J(x)\right)\ket0
 =
 \exp\!\left[-\frac12\int \dd^4x\,\dd^4y\,
 J(x)D_F(x-y)J(y)\right].
\]
\item
Take two point sources, separated by \(\mathbf R\), that remain on for a
time \(T\):
\[
 J(x)=g f(t)
 \left[\delta^3(\mathbf x)+\delta^3(\mathbf x-\mathbf R)\right],
\qquad
 f(t)=
 \begin{cases}
 1,&-T/2<t<T/2,\\
 0,&\text{otherwise}.
 \end{cases}
\]
Discard the \(D_F(0)\) self-energy terms and evaluate the remaining
vacuum amplitude when \(T\gg R\gg m^{-1}\).
\item
Match the result to \(\exp[-\ii V(R)T]\), determine \(V(R)\), and decide
whether the force is attractive or repulsive.
\end{enumerate}
\end{enumerate}
