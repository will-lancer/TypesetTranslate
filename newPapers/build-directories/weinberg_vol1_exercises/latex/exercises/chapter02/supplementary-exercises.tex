\chapterbackmatter{Supplementary Exercises}

\begin{enumerate}
\SupplementaryExercise{1}{Invariant Mass-Shell Measure}
{(Adapted from MIT OpenCourseWare, 8.323, Spring 2023, Problem Set 1,
Problem 2.)}{mit-823-s23-ps1}
For \(m\geq0\), prove that
\[
 d^4p\,\delta(p^2+m^2)\theta(p^0)
   =\frac{d^3\mathbf p}{2E_{\mathbf p}},
 \qquad E_{\mathbf p}=\sqrt{\mathbf p^2+m^2},
\]
after the \(p^0\) integration, and explain precisely which component of the
Lorentz group preserves the positive-energy sheet.

\SupplementaryExercise{2}{The Invariant Delta Kernel}
{(Adapted from MIT OpenCourseWare, 8.323, Spring 2023, Problem Set 1,
Problem 2.)}{mit-823-s23-ps1}
Let \(p\) and \(q\) lie on the same positive-energy mass shell.  Show that
\(E_{\mathbf p}\delta^3(\mathbf p-\mathbf q)\) is Lorentz invariant.  Deduce
the covariant momentum-state normalization and compare it with the
\(\delta^3\)-normalization used in Section 2.5.

\SupplementaryExercise{3}{A Lorentz-Normalized Wave Packet}
{(Adapted from MIT OpenCourseWare, 8.323, Spring 2023, Problem Set 1,
Problem 2.)}{mit-823-s23-ps1}
A scalar one-particle wave packet is represented by a function
\(\psi(\mathbf p)\) on the positive mass shell.  Find an invariant inner
product, determine the scalar transformation law for \(\psi\), and verify
directly that the norm is frame independent.

\SupplementaryExercise{4}{Boost Jacobian on Shell}
{(Adapted from MIT OpenCourseWare, 8.323, Spring 2023, Problem Set 1,
Problem 2.)}{mit-823-s23-ps1}
For a boost of velocity \(u\widehat{\mathbf z}\), calculate the Jacobian
\(\partial(p'_x,p'_y,p'_z)/\partial(p_x,p_y,p_z)\) on the mass shell.
Use it to prove \(d^3\mathbf p'/E_{\mathbf p'}=
d^3\mathbf p/E_{\mathbf p}\) without invoking a four-dimensional delta
function.

\SupplementaryExercise{5}{Poincar\'e Algebra from Differential Operators}
{(Adapted from M. Gaberdiel, ETH Z\"urich QFT I, HS 2011, Sheet 4,
Exercise 2.)}{eth-qft1-hs11-sheet4}
Acting on scalar functions of spacetime, take
\[
 P^\mu=-i\partial^\mu,\qquad
 J^{\mu\nu}=-i(x^\mu\partial^\nu-x^\nu\partial^\mu).
\]
Derive all three covariant Poincar\'e commutators.  Check the signs by
comparing with Eqs.~\eqref{eq:2.4.12} and \eqref{eq:2.4.14}.

\SupplementaryExercise{6}{Rotations and Boosts as a Closed Algebra}
{(Adapted from M. Gaberdiel, ETH Z\"urich QFT I, HS 2011, Sheet 4,
Exercise 2.)}{eth-qft1-hs11-sheet4}
Starting from the covariant Poincar\'e algebra, define
\(J_i=\frac12\epsilon_{ijk}J^{jk}\) and \(K_i=J^{i0}\).  Derive the
commutators among \(\mathbf J,\mathbf K,\mathbf P,H\), including
\([K_i,K_j]\), \([K_i,P_j]\), and \([K_i,H]\).

\SupplementaryExercise{7}{Scalar Generators in Momentum Space}
{(Adapted from M. Gaberdiel, ETH Z\"urich QFT I, HS 2011, Sheet 4,
Exercises 1--3.)}{eth-qft1-hs11-sheet4}
On scalar wavefunctions with inner product
\[
 \braket{\phi}{\psi}
 =\int\frac{d^3\mathbf p}{2E_{\mathbf p}}\,
   \phi(\mathbf p)^*\psi(\mathbf p),
\]
show that one may take
\[
 P^\mu=p^\mu,\quad
 \mathbf J=-i\,\mathbf p\times\nabla_{\mathbf p},\quad
 K_i=iE_{\mathbf p}\frac{\partial}{\partial p_i}.
\]
Verify Hermiticity on a suitable dense domain and check the nontrivial
Poincar\'e commutators.

\SupplementaryExercise{8}{Finite Translations of Momentum Eigenstates}
{(Adapted from M. Gaberdiel, ETH Z\"urich QFT I, HS 2011, Sheet 4,
Exercise 4.)}{eth-qft1-hs11-sheet4}
Exponentiate the translation generators to find
\(U(\mathbf 1,a)\ket{p,\sigma}\).  Then use the Poincar\'e group
multiplication law to prove
\[
 U(\Lambda)U(\mathbf1,a)U(\Lambda)^{-1}
 =U(\mathbf1,\Lambda a).
\]

\SupplementaryExercise{9}{Pauli--Lubanski Orthogonality}
{(Adapted from S. Gieseke, C. Borschensky, and C. B. Duncan, KIT,
Exercise Sheet 5, 2021, Problem 1.)}{kit-ettp-ws21-sheet5}
Define
\[
 \mathcal W_\mu=\frac12\epsilon_{\mu\nu\rho\sigma}
 J^{\nu\rho}P^\sigma.
\]
Prove \(\mathcal W_\mu P^\mu=0\) and
\([\mathcal W_\mu,P^\nu]=0\) using only the Poincar\'e algebra.

\SupplementaryExercise{10}{Spin from the Rest-Frame Casimir}
{(Adapted from S. Gieseke, C. Borschensky, and C. B. Duncan, KIT,
Exercise Sheet 5, 2021, Problem 1.)}{kit-ettp-ws21-sheet5}
For a massive irreducible representation, evaluate \(\mathcal W^\mu\) in
the rest frame \(p^\mu=(m,\mathbf0)\).  Show that
\(\mathcal W^2=m^2j(j+1)\) in the mostly-plus metric and explain why the
answer is frame independent.

\SupplementaryExercise{11}{The Pauli--Lubanski Pseudovector}
{(Adapted from S. Gieseke, C. Borschensky, and C. B. Duncan, KIT,
Exercise Sheet 5, 2021, Problem 1.)}{kit-ettp-ws21-sheet5}
Show directly that \(\mathcal W^\mu\) is unchanged by translations and
transforms as a pseudovector under the full Lorentz group.  Conclude that
\(\mathcal W^2\) commutes with all Poincar\'e generators.

\SupplementaryExercise{12}{What the Two Casimirs Do and Do Not Distinguish}
{(Adapted from P. Woit, \emph{Quantum Field Theory for Mathematicians},
Chapter 11, 2026.)}{woit-qftmath-ch11}
List the eigenvalues of \(P^2\) and \(\mathcal W^2\) for massive particles
of spin \(0,\frac12,1\).  Then show that for a finite-helicity massless
representation \(\mathcal W^\mu=\lambda P^\mu\), so both Casimirs vanish
for every helicity.  What extra datum distinguishes the massless
representations?

\SupplementaryExercise{13}{The Massive Little Group}
{(Adapted from P. Woit, \emph{Quantum Field Theory for Mathematicians},
Chapter 11, 2026.)}{woit-qftmath-ch11}
Find every proper orthochronous Lorentz transformation that fixes
\(k^\mu=(m,0,0,0)\).  Identify the little group and use its unitary
irreducible representations to recover the \(2j+1\) spin states of a
massive particle.

\SupplementaryExercise{14}{The Null Little-Group Algebra}
{(Adapted from J. Preskill, Physics 236c notes, Chapter 1, 1990.)}
{preskill-ph236c-ch1}
For \(k^\mu=(\kappa,0,0,\kappa)\), verify that
\[
 A=J_2+K_1,\qquad B=-J_1+K_2,\qquad J_3
\]
annihilate \(k^\mu\) in the vector representation.  Derive their
commutators and identify the resulting Lie algebra.

\SupplementaryExercise{15}{Why Finite Helicity Sets Null Translations to Zero}
{(Adapted from J. Preskill, Physics 236c notes, Chapter 1, 1990.)}
{preskill-ph236c-ch1}
Suppose the commuting null-translation generators \(A,B\) of the massless
little group have a simultaneous eigenstate with
\(a^2+b^2=\rho^2>0\).  Show that rotations generate a continuous circle of
orthogonal eigenstates.  Explain why a finite set of helicity states
therefore requires \(A=B=0\).

\SupplementaryExercise{16}{The Wigner Phase of a Helicity State}
{(Adapted from J. Preskill, Physics 236c notes, Chapter 1, 1990.)}
{preskill-ph236c-ch1}
Assuming the null translations act trivially, derive the Lorentz
transformation law of a massless state \(\ket{p,\lambda}\) in terms of the
rotation angle of \(L^{-1}(\Lambda p)\Lambda L(p)\).  Check the result for
a rotation through \(\theta\) about \(\mathbf p\).

\SupplementaryExercise{17}{Parity, Time Reversal, and Helicity}
{(Adapted from J. Preskill, Physics 205abc notes, Chapter 3, 1986--87.)}
{preskill-ph205-ch3}
Using that spin is an axial vector, determine the action of \(P\) and \(T\)
on the helicity \(\lambda=\mathbf J\cdot\widehat{\mathbf p}\).  State the
particle-content condition imposed by parity invariance when
\(\lambda\ne0\), and explain why time reversal imposes no analogous
opposite-helicity partner.

\SupplementaryExercise{18}{Normalization from a Standard Boost}
{(Adapted from P. Woit, \emph{Quantum Field Theory for Mathematicians},
Chapter 11, 2026.)}{woit-qftmath-ch11}
Let \(\ket{p,\sigma}=N(p)U(L(p))\ket{k,\sigma}\) and normalize momentum
states with \(\delta^3(\mathbf p-\mathbf p')\).  Determine \(N(p)\) for
massive and massless standard momenta, and rederive the square-root factor
in the Lorentz transformation law.

\SupplementaryExercise{19}{A Wigner Rotation from Perpendicular Boosts}
{(Adapted from J. Preskill, Physics 236c notes, Chapter 1, 1990.)}
{preskill-ph236c-ch1}
A rest momentum is first boosted by rapidity \(\xi\) along \(y\), then by
rapidity \(\eta\) along \(z\).  Show that the little-group element obtained
after removing the final rotation-free boost is a rotation about \(x\),
with
\[
 \tan\frac{\Omega}{2}
 =\tanh\frac{\eta}{2}\tanh\frac{\xi}{2}.
\]
Discuss the reversal of the angle under an active/passive change of
convention.

\SupplementaryExercise{20}{Spin-One Mixing under the Wigner Rotation}
{(Adapted from J. Preskill, Physics 236c notes, Chapter 1, 1990.)}
{preskill-ph236c-ch1}
For a spin-one particle initially in a \(J_z\) eigenstate, compute the
probabilities for all three \(J_z\) outcomes after a rotation
\(R_x(\Omega)\).  Give separate answers for initial
\(\sigma=0\) and \(\sigma=+1\).

\SupplementaryExercise{21}{The Double Cover \(SL(2,\mathbb C)\)}
{(Adapted from V. Kaplunovsky, \emph{Finite Multiplets of the Spin(3,1)
Group}, 2016, Exercises 3--5.)}{kaplunovsky-sl2c-2016}
Associate the Hermitian matrix \(X=x^0\mathbf1+x^i\sigma_i\) with a
four-vector.  For \(A\in SL(2,\mathbb C)\), show that
\(X\mapsto AXA^\dagger\) induces a proper orthochronous Lorentz
transformation.  Determine the kernel of the map.

\SupplementaryExercise{22}{Two Commuting Angular-Momentum Algebras}
{(Adapted from V. Kaplunovsky, \emph{Finite Multiplets of the Spin(3,1)
Group}, 2016, Exercise 1.)}{kaplunovsky-sl2c-2016}
Define
\[
 \mathbf A=\frac12(\mathbf J+i\mathbf K),\qquad
 \mathbf B=\frac12(\mathbf J-i\mathbf K).
\]
Show separately that \(\mathbf A\) and \(\mathbf B\) obey angular-momentum
commutators, and show that the two sets commute.  Explain the labels
\((j_A,j_B)\) for finite-dimensional Lorentz representations and why
nontrivial finite-dimensional representations cannot be unitary.

\SupplementaryExercise{23}{A Four-Vector as a Bispinor}
{(Adapted from V. Kaplunovsky, \emph{Finite Multiplets of the Spin(3,1)
Group}, 2016, Exercises 4 and 7.)}{kaplunovsky-sl2c-2016}
Show that \(V_{\alpha\dot\alpha}=V^\mu(\sigma_\mu)_{\alpha\dot\alpha}\)
transforms in the \((\frac12,\frac12)\) representation.  Recover the
Lorentz-vector transformation and account for all four complex
components.

\SupplementaryExercise{24}{The Sign of a \(2\pi\) Rotation}
{(Adapted from V. Kaplunovsky, \emph{Finite Multiplets of the Spin(3,1)
Group}, 2016, Exercises 3--5.)}{kaplunovsky-sl2c-2016}
Follow the path \(A(\theta)=\exp(-i\theta\sigma_3/2)\) in
\(SL(2,\mathbb C)\) from \(\theta=0\) to \(2\pi\) and \(4\pi\).  Determine
its image in the Lorentz group and its action on a spin-\(j\) state.
Explain the origin of projective Lorentz representations.

\SupplementaryExercise{25}{The Galilean Mass Cocycle}
{(Adapted from E. Annigoni and V. Moretti, arXiv:1205.1372, Section 2,
2012.)}{annigoni-moretti-1205.1372}
For a Schr\"odinger wavefunction of mass \(m\), use
\[
 (U(\mathbf v)\psi)(\mathbf x,t)
 =e^{im(\mathbf v\cdot\mathbf x-\mathbf v^2t/2)}
  \psi(\mathbf x-\mathbf v t,t)
\]
to compare a spatial translation with a boost.  Extract the central
commutator \([K_i,P_j]\).

\SupplementaryExercise{26}{The Bargmann Group Commutator Loop}
{(Adapted from E. Annigoni and V. Moretti, arXiv:1205.1372, Section 2,
2012.)}{annigoni-moretti-1205.1372}
Evaluate the operator corresponding to a translation by \(\mathbf a\), a
boost by \(\mathbf v\), and their inverses in the group-commutator order.
Although the net spacetime transformation is the identity, show that the
state acquires a mass-dependent phase.

\SupplementaryExercise{27}{Mass Superselection and Its Central-Extension Escape}
{(Adapted from E. Annigoni and V. Moretti, arXiv:1205.1372, Sections 1--2,
2012.)}{annigoni-moretti-1205.1372}
Apply the Bargmann loop to a coherent superposition of two different
masses.  Derive the usual mass-superselection conclusion.  Then explain
how treating mass as a central operator of an enlarged symmetry group
changes the interpretation.

\SupplementaryExercise{28}{The Galilean Algebra as a Poincar\'e Contraction}
{(Adapted from E. Annigoni and V. Moretti, arXiv:1205.1372, Sections 2
and 5, 2012.)}{annigoni-moretti-1205.1372}
Restore \(c\), write \(P^0=Mc+H_{\rm NR}/c+O(c^{-3})\), and rescale the
relativistic boosts by \(K_i^{\rm G}=J^{i0}/c\).  Take \(c\to\infty\) in
the Poincar\'e algebra and identify the central charge.

\SupplementaryExercise{29}{Massive Spin in \(2+1\) Dimensions}
{(Adapted from P. Woit, \emph{Quantum Field Theory for Mathematicians},
Chapter 11, 2026; specialized to \(2+1\) dimensions.)}
{woit-qftmath-ch11}
Determine the connected little group of \(k^\mu=(m,0,0)\).  Classify its
one-dimensional unitary representations for \(SO(2)\), its double cover,
and its universal cover.  Show how a reflection acts on the spin label.

\SupplementaryExercise{30}{Massless Little-Group Characters in \(2+1\)
Dimensions}
{(Adapted from J. Preskill, Physics 236c notes, Chapter 1, 1990;
specialized to \(2+1\) dimensions.)}{preskill-ph236c-ch1}
For \(k^\mu=(\kappa,0,\kappa)\), find the connected Lorentz transformations
that leave \(k^\mu\) fixed.  Classify their unitary irreducible
representations, determine the effect of a reflection, and identify the
representation that gives the usual single massless polarization.
\end{enumerate}
