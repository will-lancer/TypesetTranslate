\SupplementaryExercise{1}{Invariant On-Shell Measures and Lorentz-Normalized Wave Packets}
  {(Adapted from MIT OpenCourseWare, 8.323, Spring 2023, Problem Set 1, Problem 2.)}
  {mit-823-s23-ps1-curated-parent-1}
\begin{enumerate}
\item
Let \(\Lambda\) be a Lorentz transformation.  Prove directly from the
defining property of the Dirac delta that
\[
 \delta^4\!\bigl(\Lambda(p-q)\bigr)=\delta^4(p-q).
\]
State where the absolute value of the determinant enters the argument.
\item
For \(m\geq0\), prove that
\[
 d^4p\,\delta(p^2+m^2)\theta(p^0)
   =\frac{d^3\mathbf p}{2\omega_{\mathbf p}},
 \qquad \omega_{\mathbf p}=\sqrt{\mathbf p^2+m^2},
\]
after the \(p^0\) integration, and explain precisely which component of the
Lorentz group preserves the positive-energy sheet.
\item
Let \(p\) and \(q\) lie on the same positive-energy mass shell.  Show that
\(\omega_{\mathbf p}\delta^3(\mathbf p-\mathbf q)\) is Lorentz invariant.  Deduce
the covariant momentum-state normalization and compare it with the
\(\delta^3\)-normalization used in Section 2.5.
\item
A scalar one-particle wave packet is represented by a function
\(\psi(\mathbf p)\) on the positive mass shell.  Find an invariant inner
product, determine the scalar transformation law for \(\psi\), and verify
directly that the norm is frame independent.
\item
For a boost of velocity \(u\widehat{\mathbf z}\), calculate the Jacobian
\(\partial(p'_x,p'_y,p'_z)/\partial(p_x,p_y,p_z)\) on the mass shell.
Use it to prove \(d^3\mathbf p'/\omega_{\mathbf p'}=
d^3\mathbf p/\omega_{\mathbf p}\) without invoking a four-dimensional delta
function.
\end{enumerate}

\SupplementaryExercise{3}{The Poincar\'e Group Commutator and Its Lie Algebra}
  {(Adapted from University of Cambridge, Mathematical Tripos Part III, 2022, Paper 302, \emph{Symmetries, Fields and Particles}, Question 2, printed p.~3.)}
  {cambridge-partiii-2022-sfp-paper302-q2}

A Poincar\'e transformation
\[
 x^\mu\longmapsto x'{}^\mu=\Lambda^\mu{}_{\nu}x^\nu+a^\mu
\]
is written as \((\Lambda,a)\), where \(\Lambda^\mu{}_{\nu}\) is a
Lorentz transformation.
\begin{enumerate}
\item[(i)] Compute \((L,b)(\Lambda,a)\).
\item[(ii)] Determine \((\Lambda,a)^{-1}\) in terms of \(\Lambda\) and
\(a\).
\item[(iii)] Defining
\[
 (\Lambda_c,a_c):=[(L,b),(\Lambda,a)],
\]
where the brackets denote the group commutator, find \(\Lambda_c\) and
\(a_c\) in terms of \(L,\Lambda,b,a\).
\item[(iv)] Define infinitesimal Lorentz transformations by
\[
 \Lambda:=I+\omega_\Lambda+O(\omega_\Lambda^2),\qquad
 L:=I+\omega_L+O(\omega_L^2).
\]
Find \(a_c\) to first order in \(\omega_L\) and first order in
\(\omega_\Lambda\).
\item[(v)] Put \(\Lambda_c:=I+\omega_c+O(\omega_c^2)\), neglect terms
of order \(\omega_L^2\) and \(\omega_\Lambda^2\), and find \(\omega_c\)
in terms of \(\omega_L\) and \(\omega_\Lambda\).
\item[(vi)] Write the defining relation for \(\Lambda^\mu{}_{\nu}\) in
terms of the Minkowski metric \(\eta_{\rho\sigma}\), and hence derive a
symmetry relation for \(\omega_\Lambda\) under interchange of its
indices.
\item[(vii)] Introduce a unitary operator corresponding to a Poincar\'e
transformation, \(U[(\Lambda,a)]\), imposing
\[
 U[(\Lambda_2,a_2)]U[(\Lambda_1,a_1)]
 =U[(\Lambda_2,a_2)(\Lambda_1,a_1)].
\]
Expanding in infinitesimal \(\omega\) and \(a\), use the source
normalization
\[
 U[(\Lambda(\omega),a)]
 :=I+\frac12\omega^{\mu\nu}M_{\mu\nu}+a^\mu P_\mu+\cdots,
 \qquad M_{\mu\nu}=-M_{\nu\mu}.
\]
Retaining constant, linear, and mixed bilinear terms in
\(\omega_\Lambda,\omega_L,a,b\), but neglecting pure squares and higher
orders, show that
\[
 U[(\Lambda_c(\omega_c),a_c)]
 =I+\left[
 \frac12\omega_L^{\mu\nu}M_{\mu\nu}+b^\mu P_\mu,
 \frac12\omega_\Lambda^{\rho\sigma}M_{\rho\sigma}+a^\rho P_\rho
 \right]+\cdots.
\]
\item[(viii)] Hence derive the Poincar\'e algebra: determine
\([P_\mu,P_\nu]\), \([M_{\mu\sigma},P_\rho]\), and
\([M_{\mu\nu},M_{\rho\sigma}]\).
\end{enumerate}

\SupplementaryExercise{4}{Pauli--Lubanski Spin and the Massive Casimir}
  {(Adapted from S. Gieseke, C. Borschensky, and C. B. Duncan, KIT, Exercise Sheet 5, 2021, Problem 1.)}
  {kit-ettp-ws21-sheet5-curated-parent-1}
\begin{enumerate}
\item
With
\[
 P^\mu=-i\partial^\mu,\qquad
 J^{\mu\nu}=-i(x^\mu\partial^\nu-x^\nu\partial^\mu),
\]
derive \([J^{\mu\nu},P^\rho]\).  Then define
\[
 \mathcal W_\mu=\frac12\epsilon_{\mu\nu\rho\sigma}
 J^{\nu\rho}P^\sigma.
\]
Prove \(\mathcal W_\mu P^\mu=0\) and
\([\mathcal W_\mu,P^\nu]=0\).  Finally derive the supplied
Pauli--Lubanski commutator, translated to the conventions of this volume,
\[
 [\mathcal W_\mu,\mathcal W_\nu]
 =-i\epsilon_{\mu\nu\rho\sigma}\mathcal W^\rho P^\sigma .
\]
\item
For a massive irreducible representation, evaluate \(\mathcal W^\mu\) in
the rest frame \(p^\mu=(m,\mathbf0)\).  Show that
\(\mathcal W^2=m^2j(j+1)\) in the mostly-plus metric and explain why the
answer is frame independent.
\item
Establish the algebraic identity
\[
 \mathcal W^2
 =-\frac12J_{\mu\nu}J^{\mu\nu}P^2
  +J_{\mu\rho}J_\nu{}^\rho P^\mu P^\nu .
\]
Use it, together with the Poincar\'e algebra, to show directly that
\(P^2\) commutes with every Poincar\'e generator and that
\([\mathcal W^2,P^\mu]=0\).  As in the source problem, do not perform the
tedious direct expansion of
\([\mathcal W^2,J^{\mu\nu}]\); obtain that commutator in the next part.
\item
Show directly that \(\mathcal W^\mu\) is unchanged by translations and
transforms as a pseudovector under the full Lorentz group.  Reconcile this
finite-transformation argument with the preceding algebra and conclude
that \([\mathcal W^2,J^{\mu\nu}]=0\).
\end{enumerate}

\SupplementaryExercise{7}{Finite-Dimensional \(SU(2)\) Representations and Charge Conjugation}
  {(Adapted from University of Cambridge, Mathematical Tripos Part III, 2023, Paper 302, \emph{Symmetries, Fields and Particles}, Question 2, printed p.~3.)}
  {cambridge-partiii-2023-sfp-paper302-q2}

Consider a finite-dimensional irreducible representation \(d\) of the
\(SU(2)\) Lie algebra.  Denote its Hermitian generators by
\(J_1,J_2,J_3\), with
\[
 [J_i,J_j]=i\epsilon_{ijk}J_k.
\]
Let \(I\) be the maximum weight of \(d\), and let \(v_m\) be an
eigenvector of \(d(J_3)\), so that
\[
 d(J_3)v_m=mv_m.
\]
\begin{enumerate}
\item[(a)] Construct the \(J_3\)-eigenvector basis for the vector space
on which \(d\) acts, showing that its dimension is \(2I+1\).
\emph{Hint:} use \(J_\pm=J_1\pm iJ_2\).
\item[(b)] Write \(\lVert v_m\rVert=N_m\), so that
\(v_m^\dagger v_m=N_m^2\), and define normalized eigenvectors by
\[
 \ket{I,m}:=\frac1{N_m}v_m.
\]
Show that
\[
 \ket{I,m}=A(I,m)\bigl(d(J_-)\bigr)^{I-m}\ket{I,I},
\]
and determine the numerical factor \(A(I,m)\).
\emph{Hint:} note that \(J_-^\dagger=J_+\).
\item[(c)] An operator \(C\) maps the \(SU(2)\) generators according to
\[
 CJ_1C^{-1}=-J_1,\qquad
 CJ_2C^{-1}=+J_2,\qquad
 CJ_3C^{-1}=-J_3.
\]
Show that the set \(\{J_i'\}\), where \(J_i':=CJ_iC^{-1}\), also
generates an \(SU(2)\) algebra.
\item[(d)] Because the up and down quarks have very light masses, the
strong force has an approximate \(SU(2)\) symmetry called isospin
symmetry; for this problem, assume it is exact.  The three pion states
form an \(I=1\) triplet
\[
 \bigl\{\ket{\pi^-},\ket{\pi^0},\ket{\pi^+}\bigr\},
\]
corresponding respectively to \(m=-1,0,1\).  Given that
\(C\ket{\pi^0}=\ket{\pi^0}\), how are \(C\ket{\pi^+}\) and
\(C\ket{\pi^-}\) related to the original states?
\end{enumerate}

\SupplementaryExercise{12}{The Heisenberg Group and Its Schr\"odinger Representation}
  {(Adapted from University of Cambridge, Mathematical Tripos Part III, 2024, Paper 302, \emph{Symmetries, Fields and Particles}, Question 1, printed p.~2.)}
  {cambridge-partiii-2024-sfp-paper302-q1}

\begin{enumerate}
\item[(a)] For each set and binary operation below, determine whether it
forms a group.  If it does, verify the group axioms; if it does not, give
an axiom that fails.
  \begin{enumerate}
  \item[(i)] The integers under addition, \((\mathbb Z,+)\).
  \item[(ii)] The nonzero integers under multiplication,
  \((\mathbb Z^*,\times)\).
  \item[(iii)] The set \(\mathcal L\) of linear functions under
  composition, where
  \[
   \mathcal L=\{f:\mathbb R\to\mathbb R\mid
   f(x)=ux+b;\ b\in\mathbb R,\ u\in\mathbb R,\ u\ne0\}.
  \]
  \end{enumerate}
\item[(b)] Consider the matrix Lie group
\[
 G=\left\{
 \begin{pmatrix}1&q&s\\0&1&r\\0&0&1\end{pmatrix}
 \mathrel{\Big|}q,r,s\in\mathbb R
 \right\},
\]
with matrix multiplication as the group operation.

  \item[(i)] Find the centre \(Z(G)\).
  \item[(ii)] Find the Lie algebra \(L(G)\).  Choose a basis and determine
  the corresponding structure constants.

  \item[(iv)] Explain whether \(L(G)\) is simple, semisimple, or neither.

 V=\{\psi:\mathbb R\to\mathbb C\mid(\psi,\psi)=1\},\qquad

 g=\begin{pmatrix}1&q&s\\0&1&r\\0&0&1\end{pmatrix},\qquad

\]
Show that \(D\) is a unitary representation of \(G\).  If \(\psi(x)\)

\SupplementaryExercise{14}{Parity Symmetry of Scalar Field Theory}
  {(Adapted from J. McGreevy, Physics 215A QFT, Fall 2023, Assignment 9, Problem 8, printed pp.~4--5.)}
  {mcgreevy-215a-f23-hw09-p8}

Under parity, \(\mathbf x\mapsto\mathbf x'=-\mathbf x\), and a real
Klein--Gordon field transforms as
\[
 P\phi(t,\mathbf x)P^{-1}=\eta_P\phi(t,-\mathbf x),
 \qquad \eta_P=\pm1,
\]
where \(P\) is unitary, \(\eta_P\) is the intrinsic parity of \(\phi\),
and \(P\ket0=\ket0\).  For completeness, use
\[
 \mathcal L_0=-\frac12\partial_\mu\phi\,\partial^\mu\phi
              -\frac12m^2\phi^2,
\]
and the mode conventions
\[
 \phi(t,\mathbf x)=\int\!\frac{d^3\mathbf k}{(2\pi)^3\sqrt{2\omega_{\mathbf k}}}
 \left(a_{\mathbf k}e^{-i\omega_{\mathbf k}t+i\mathbf k\cdot\mathbf x}
 +a_{\mathbf k}^\dagger e^{i\omega_{\mathbf k}t-i\mathbf k\cdot\mathbf x}\right),
 \quad
 [a_{\mathbf k},a_{\mathbf q}^\dagger]=(2\pi)^3\delta^3(\mathbf k-\mathbf q).
\]
\begin{enumerate}
\item[(a)] Show that parity preserves the free Lagrangian, though not the
Lagrangian density at a fixed point, for both values of \(\eta_P\).
\item[(b)] Show that an arbitrary \(n\)-particle state transforms as
\[
 P\ket{\mathbf k_1,\ldots,\mathbf k_n}
 =\eta_P^n\ket{-\mathbf k_1,\ldots,-\mathbf k_n}.
\]
\item[(c)] Here is an explicit realization of the parity operator.  Let
\[
 P_1=\exp\!\left[-\frac{i\pi}{2}\sum_{\mathbf k}
 a_{\mathbf k}^\dagger a_{\mathbf k}\right],\qquad
 P_2=\exp\!\left[\frac{i\eta_P\pi}{2}\sum_{\mathbf k}
 a_{\mathbf k}^\dagger a_{-\mathbf k}\right].
\]
Show that
\[
 P_1a_{\mathbf k}P_1^{-1}=ia_{\mathbf k},\qquad
 P_2a_{\mathbf k}P_2^{-1}=-i\eta_Pa_{-\mathbf k}.
\]
\emph{Hint.} Use the Campbell--Baker--Hausdorff formula
\[
 e^{i\alpha A}Be^{-i\alpha A}
 =\sum_{n=0}^{\infty}\frac{(i\alpha)^n}{n!}B_n,
 \qquad B_0=B,\quad B_n=[A,B_{n-1}]\quad(n\ge1).
\]
Show that \(P=P_1P_2\) is unitary and satisfies the parity transformation
above.  (The source cross-reference reads ``(24)'' here, while the
displayed transformation is its Eq.~(14).)
\item[(d)] Now take a complex scalar field.  Using the preceding results
and the particle current
\[
 j^\mu=i\left(\Phi^\dagger\partial^\mu\Phi
 -(\partial^\mu\Phi^\dagger)\Phi\right),
\]
find the action of parity on \(j^\mu\), namely
\(j^\mu\mapsto Pj^\mu P^{-1}\).  Extend the action of \(P\) from the real
field to the complex field as needed.
\end{enumerate}
\end{enumerate}
