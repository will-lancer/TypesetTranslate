\chapterbackmatter{Solutions to Weinberg Exercises}
\ExerciseEditorialNote

\begin{enumerate}
\WeinbergSolution{1}
Write
\[
  p^\mu=(E,0,p,0),\qquad
  E=\sqrt{p^2+m^2},\qquad
  \sinh\xi=\frac{p}{m}.
\]
If \(\mathcal O'\) moves with velocity \(v\widehat{\mathbf z}\), its
coordinates are obtained with the passive boost \(B_z(-\eta)\), where
\(\tanh\eta=v\).  Thus
\[
  p'^\mu=B_z(-\eta)p^\mu
    =(\gamma_vE,0,p,-\gamma_vvE),
  \qquad \gamma_v=(1-v^2)^{-1/2}.
\]
The transformed state is not, in general, an eigenstate of the old
\(z\)-component of spin.  With rotation-free standard boosts \(L(p)\), its
spin is acted on by the Wigner rotation
\[
  R_W=L^{-1}(p')B_z(-\eta)L(p).
\]
For the two perpendicular boosts here, \(R_W\) is a rotation about the
\(x\)-axis.  Choosing its positive sense by the displayed definition,
\[
  \tan\frac{\Omega}{2}
    =\tanh\frac{\eta}{2}\tanh\frac{\xi}{2},
  \qquad
  \tan\Omega=\frac{\sinh\eta\,\sinh\xi}
  {\cosh\eta+\cosh\xi}.
\]
(Changing from the passive to the active convention reverses the sign of
\(\Omega\).)  Equation \eqref{eq:2.5.23} therefore gives
\[
  U(B_z(-\eta))\ket{p\widehat{\mathbf y},\sigma}
  =\sqrt{\frac{p'^0}{E}}
    \sum_{\sigma'=-1}^{1}
    D^{(1)}_{\sigma'\sigma}\!\left(R_x(\Omega)\right)
    \ket{\mathbf p',\sigma'}.
\]
The square-root is absent if covariantly normalized momentum states are
used.  Hence \(\mathcal O'\) sees the momentum above and a coherent
superposition of the three \(z\)-spin values, rather than the original
single value \(\sigma\).

\WeinbergSolution{2}
Take the original four-momentum and polarization to be
\[
  p^\mu=(\omega,0,\omega,0),\qquad
  \epsilon^\mu=(0,0,0,1).
\]
The same passive boost as in the preceding solution gives
\[
  p'^\mu=(\gamma_v\omega,0,\omega,-\gamma_vv\omega),
  \qquad
  \epsilon'^\mu=(-\gamma_vv,0,0,\gamma_v).
\]
They obey \(p'\cdot\epsilon'=0\) and
\(\epsilon'^2=1\).  The time component of \(\epsilon'\) is not a new
physical polarization; it is removable by a gauge transformation.  Adding
\((v/\omega)p'^\mu\) produces the temporal-gauge representative
\[
  \widetilde\epsilon^\mu
   =\epsilon'^\mu+\frac{v}{\omega}p'^\mu
   =(0,0,v,\gamma_v^{-1}).
\]
Indeed,
\[
  \widehat{\mathbf p}'=(0,\gamma_v^{-1},-v),\qquad
  \widetilde{\boldsymbol\epsilon}=(0,v,\gamma_v^{-1}),
\]
so \(\widehat{\mathbf p}'\cdot\widetilde{\boldsymbol\epsilon}=0\) and
\(\widetilde{\boldsymbol\epsilon}^{\,2}=1\).  Thus \(\mathcal O'\) sees a
photon moving obliquely in the \(yz\)-plane, with a transverse linear
polarization rotated in that plane.  No longitudinal photon state has been
created.

\WeinbergSolution{3}
Parameterize a Galilean transformation by
\((R,\mathbf v,\mathbf a,b)\), acting on spacetime as
\[
  t'=t+b,\qquad
  \mathbf x'=R\mathbf x+\mathbf v\,t+\mathbf a.
\]
Composition gives a semidirect product: rotations act on
\(\mathbf v,\mathbf a\), while a boost followed by a time translation
also shifts the spatial translation.  Differentiating the multiplication
law at the identity gives, before central terms,
\[
\begin{gathered}
 [J_i,J_j]=\ii\epsilon_{ijk}J_k,\quad
 [J_i,K_j]=\ii\epsilon_{ijk}K_k,\quad
 [J_i,P_j]=\ii\epsilon_{ijk}P_k,\\
 [K_i,H]=\ii P_i,\qquad
 [K_i,K_j]=[P_i,P_j]=[P_i,H]=[J_i,H]=0.
\end{gathered}
\]
A ray representation may multiply the group law by a two-cocycle.
Rotational covariance and the Jacobi identities eliminate every
nontrivial central tensor except a scalar in the boost--translation
bracket.  After constant shifts of generators remove coboundary terms, the
most general algebra in three spatial dimensions is therefore
\[
  [K_i,P_j]=\ii\delta_{ij}M,\qquad
  [M,H]=[M,P_i]=[M,K_i]=[M,J_i]=0,
\]
together with the preceding commutators.  A possible constant in \(H\), or
constant vector terms in \(\mathbf P\) or \(\mathbf K\), are removed by
redefining those generators.  Terms such as
\([K_i,K_j]=\ii\epsilon_{ijk}C_k\) would require a central vector
\(\mathbf C\); the Jacobi identity with \(J_i\) forces it to vanish in
three dimensions.  Thus the sole non-removable central charge is \(M\),
identified in an irreducible representation with the total mass.

\WeinbergSolution{4}
Equation \eqref{eq:2.4.9} says that translations leave \(P^\mu\) unchanged
and a Lorentz transformation rotates it as a four-vector.  Consequently
\[
  U(\Lambda,a)P_\mu P^\mu U^{-1}(\Lambda,a)
  =\eta_{\rho\sigma}P^\rho P^\sigma=P_\mu P^\mu.
\]
For the Pauli--Lubanski operator, use Eqs.~\eqref{eq:2.4.8} and
\eqref{eq:2.4.9}.  Under a translation,
\[
  J^{\nu\rho}\longmapsto
  J^{\nu\rho}-a^\nu P^\rho+a^\rho P^\nu.
\]
The extra contribution to \(W_\mu\) contains the contraction of
\(\epsilon_{\mu\nu\rho\lambda}\) with the symmetric product
\(P^\rho P^\lambda\), and hence vanishes.  Under a proper Lorentz
transformation the invariance of the Levi--Civita tensor gives
\[
  U(\Lambda,a)W_\mu U^{-1}(\Lambda,a)
  =\Lambda_\mu{}^\nu W_\nu.
\]
Therefore
\[
  U(\Lambda,a)W_\mu W^\mu U^{-1}(\Lambda,a)
  =W_\mu W^\mu.
\]
Both operators commute with every \(U(\Lambda,a)\), as required.  For an
improper Lorentz transformation \(W_\mu\) is a pseudovector, but its square
is still invariant.

\WeinbergSolution{5}
Choose the standard massive momentum \(k^\mu=(m,0,0)\).  Its connected
little group in \(2+1\) dimensions is \(SO(2)\), generated by the single
spatial rotation \(J\).  Every complex unitary irreducible representation
of \(SO(2)\) is one-dimensional:
\[
  D_s(R(\theta))=e^{is\theta}.
\]
For the group \(SO(2,1)\) itself, single-valuedness requires
\(s\in\mathbb Z\).  For its double cover one obtains integer or
half-integer \(s\), while for the universal cover \(s\) may be any real
number; the latter possibility is the origin of anyonic spin.

With standard boosts \(L(p)\), a one-particle state is denoted simply
\(\ket{p,s}\), with no \(2s+1\) spin degeneracy, and
\[
  U(\Lambda)\ket{p,s}
   =\sqrt{\frac{(\Lambda p)^0}{p^0}}\,
     e^{is\theta(\Lambda,p)}\ket{\Lambda p,s},
  \qquad
  L^{-1}(\Lambda p)\Lambda L(p)=R(\theta).
\]
A spatial reflection reverses the orientation of the plane and conjugates
\(J\) to \(-J\); time reversal is antiunitary and also reverses \(J\).
Thus either \(P\) or \(T\) carries the representation \(s\) into
\(-s\).  If either inversion is a symmetry, a particle with \(s\ne0\)
must have a degenerate partner of spin \(-s\) (unless the two are already
the same representation).  The combined operation \(PT\) preserves \(s\).

\WeinbergSolution{6}
For a standard null momentum \(k^\mu=(\kappa,0,\kappa)\), the connected
little group in \(2+1\) dimensions is the one-parameter group of null
rotations, isomorphic to \((\mathbb R,+)\).  Its unitary irreducible
representations are characters
\[
  D_\rho(S(\alpha))=e^{\ii\rho\alpha},
  \qquad \rho\in\mathbb R.
\]
Accordingly, the most general irreducible massless one-particle state is
\(\ket{p,\rho}\), and transforms as
\[
 U(\Lambda)\ket{p,\rho}
 =\sqrt{\frac{(\Lambda p)^0}{p^0}}\,
   e^{\ii\rho\alpha(\Lambda,p)}\ket{\Lambda p,\rho}.
\]
There is no transverse rotation group \(SO(d-2)\) in \(d=3\), so there is
no ordinary helicity label.  The particle representations normally used in
local field theory take the null-rotation subgroup trivially,
\(\rho=0\); they then have one state at each momentum.

An orientation- or time-reversing element conjugates the null-rotation
generator to its negative.  Hence \(P\) or \(T\) maps
\(\rho\mapsto-\rho\).  A full \(O(2,1)\) representation with
\(\rho\ne0\) must contain the pair \((\rho,-\rho)\).  For the usual
\(\rho=0\) representation, each inversion acts within the same
one-particle species, by a phase for the unitary \(P\) and by an
antiunitary phase convention for \(T\).
\end{enumerate}
