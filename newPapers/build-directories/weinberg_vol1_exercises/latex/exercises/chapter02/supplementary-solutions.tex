\SupplementarySolution{1}{Invariant On-Shell Measures and Lorentz-Normalized Wave Packets}
\begin{enumerate}
\item
For any nonsingular real matrix \(A\),
\(\delta^4(Ax)=\delta^4(x)/|\det A|\), as follows by testing both sides
against a smooth compactly supported function and changing variables.
The Lorentz condition gives
\[
 (\det\Lambda)^2=1,
\]
so \(|\det\Lambda|=1\), including for improper transformations.  Taking
\(x=p-q\) proves the stated invariance of the four-dimensional delta.
\item
For a proper Lorentz transformation,
\(\lvert\det\Lambda\rvert=1\), so \(d^4p\) is invariant; \(p^2\) is a
scalar, so is \(\delta(p^2+m^2)\).  On the positive-energy shell,
\[
 \delta(p^2+m^2)
 =\delta\!\left(-(p^0)^2+\mathbf p^2+m^2\right)
 =\frac{\delta(p^0-\omega_{\mathbf p})+\delta(p^0+\omega_{\mathbf p})}
 {2\omega_{\mathbf p}}.
\]
Multiplication by \(\theta(p^0)\) selects the first root, and hence
\[
 \int d^4p\,\delta(p^2+m^2)\theta(p^0)F(p)
 =\int\frac{d^3\mathbf p}{2\omega_{\mathbf p}}F(\omega_{\mathbf p},\mathbf p).
\]
A proper orthochronous transformation maps every future-directed causal
vector to another such vector.  It therefore preserves the sign of \(p^0\).
Time reversal interchanges the two energy sheets, so the positive-sheet
measure alone is not invariant under that disconnected component.
\item
The invariant on-shell measure is \(d\mu(p)=d^3\mathbf p/\omega_{\mathbf p}\),
apart from an irrelevant constant.  Its delta kernel \(\Delta(p,q)\) is
defined by
\[
 f(p)=\int d\mu(q)\,\Delta(p,q)f(q).
\]
Comparison with the ordinary three-dimensional delta function gives
\[
 \Delta(p,q)=\omega_{\mathbf q}\delta^3(\mathbf q-\mathbf p)
            =\omega_{\mathbf p}\delta^3(\mathbf p-\mathbf q).
\]
Because both \(d\mu\) and the defining identity are invariant, \(\Delta\)
is invariant.

The common covariant convention is
\[
 \braket{p',\sigma'}{p,\sigma}_{\rm cov}
 =2\omega_{\mathbf p}(2\pi)^3
 \delta_{\sigma'\sigma}\delta^3(\mathbf p-\mathbf p').
\]
The convention of Eq.~\eqref{eq:2.5.19} omits the factor
\(2\omega_{\mathbf p}(2\pi)^3\).  The compensating factor
\(\sqrt{(\Lambda p)^0/p^0}\) then appears in the state transformation law.
\item
Use
\[
 \braket{\phi}{\psi}
 =\int\frac{d^3\mathbf p}{2\omega_{\mathbf p}}\,
   \phi(\mathbf p)^*\psi(\mathbf p).
\]
For a scalar representation,
\[
 (U(\Lambda)\psi)(\mathbf p)
 =\psi(\boldsymbol{\Lambda^{-1}p}),
\]
where the energy component of \(p\) is fixed by the positive mass shell.
Then, with \(q=\Lambda^{-1}p\),
\[
 \|U(\Lambda)\psi\|^2
 =\int\frac{d^3\mathbf p}{2\omega_{\mathbf p}}
   |\psi(\boldsymbol{\Lambda^{-1}p})|^2
 =\int\frac{d^3\mathbf q}{2\omega_{\mathbf q}}|\psi(\mathbf q)|^2.
\]
The last equality is precisely the invariance established in S.2.1.
Thus the scalar pullback is unitary without an additional Jacobian factor.
\item
For the passive boost
\[
 E'=\gamma(E-up_z),\qquad
 p'_z=\gamma(p_z-uE),\qquad p'_x=p_x,\quad p'_y=p_y.
\]
Since \(\partial E/\partial p_z=p_z/E\),
\[
 \frac{\partial p'_z}{\partial p_z}
 =\gamma\left(1-u\frac{p_z}{E}\right)
 =\frac{\gamma(E-up_z)}{E}=\frac{E'}{E}.
\]
The transverse coordinates are unchanged and their mixed derivatives do
not alter the triangular determinant, so
\[
 \det\frac{\partial\mathbf p'}{\partial\mathbf p}
 =\frac{E'}{E}.
\]
Consequently \(d^3\mathbf p'=(E'/E)d^3\mathbf p\), which immediately yields
\(d^3\mathbf p'/E'=d^3\mathbf p/E\).
\end{enumerate}

\SupplementarySolution{2}{The Poincar\'e Algebra from Infinitesimal Generators}
\begin{enumerate}
\item
The elementary relations
\[
 [\partial^\mu,x^\nu]=\eta^{\mu\nu},\qquad
 [\partial^\mu,\partial^\nu]=[x^\mu,x^\nu]=0
\]
give \([P^\mu,P^\nu]=0\).  Applying the product rule once gives
\[
 [J^{\mu\nu},P^\rho]
 =i\bigl(\eta^{\mu\rho}P^\nu-\eta^{\nu\rho}P^\mu\bigr),
\]
which is equivalent to Eq.~\eqref{eq:2.4.13}.  Applying it twice gives
\[
\begin{split}
 [J^{\mu\nu},J^{\rho\sigma}]
 =i(&\eta^{\mu\rho}J^{\nu\sigma}
 -\eta^{\nu\rho}J^{\mu\sigma}\\
 &+\eta^{\nu\sigma}J^{\mu\rho}
 -\eta^{\mu\sigma}J^{\nu\rho}),
\end{split}
\]
equivalent to Eq.~\eqref{eq:2.4.12}.  No field equation was used: these
relations express the action of infinitesimal spacetime isometries on
scalar functions.
\item
Substituting
\(J_i=\frac12\epsilon_{ijk}J^{jk}\), \(K_i=J^{i0}\), and \(H=P^0\)
into the covariant algebra yields
\[
\begin{gathered}
 [J_i,J_j]=i\epsilon_{ijk}J_k,\qquad
 [J_i,K_j]=i\epsilon_{ijk}K_k,\qquad
 [K_i,K_j]=-i\epsilon_{ijk}J_k,\\
 [J_i,P_j]=i\epsilon_{ijk}P_k,\qquad
 [J_i,H]=0,\qquad
 [K_i,P_j]=i\delta_{ij}H,\qquad
 [K_i,H]=iP_i,\\
 [P_i,P_j]=[P_i,H]=0.
\end{gathered}
\]
The minus sign in \([K_i,K_j]\) is the algebraic signature of the
noncompact Lorentz boosts.  The relations involving \(H\) also show why a
boost does not preserve an energy eigenspace.
\end{enumerate}

\SupplementarySolution{3}{The Poincar\'e Group Commutator and Its Lie Algebra}
We use the group-commutator convention
\([g,h]=ghg^{-1}h^{-1}\).
\begin{enumerate}
\item[(i)] Applying \((\Lambda,a)\) first and \((L,b)\) second gives
\[
 x\longmapsto L(\Lambda x+a)+b=L\Lambda x+La+b,
\]
and hence
\[
 (L,b)(\Lambda,a)=(L\Lambda,La+b).
\]
This is the semidirect-product law: the Lorentz factor acts on the
translation before the translations are added.

\item[(ii)] Requiring the product with \((\Lambda,a)\) to equal
\((I,0)\) gives
\[
 (\Lambda,a)^{-1}=(\Lambda^{-1},-\Lambda^{-1}a).
\]

\item[(iii)] Successive use of the composition law gives
\[
\begin{split}
 (\Lambda_c,a_c)
 &=(L,b)(\Lambda,a)(L^{-1},-L^{-1}b)
   (\Lambda^{-1},-\Lambda^{-1}a),\\
 \Lambda_c&=L\Lambda L^{-1}\Lambda^{-1},\\
 a_c&=La+b-L\Lambda L^{-1}b
      -L\Lambda L^{-1}\Lambda^{-1}a.
\end{split}
\]
The result also makes clear that translations fail to commute with
Lorentz transformations even though the translation subgroup itself is
abelian.

\item[(iv)] To the required orders,
\[
\begin{aligned}
 L\Lambda L^{-1}
 &=I+\omega_\Lambda+[\omega_L,\omega_\Lambda]+\cdots,\\
 L\Lambda L^{-1}\Lambda^{-1}
 &=I+[\omega_L,\omega_\Lambda]+\cdots.
\end{aligned}
\]
Substitution in the exact answer from part (iii) yields
\[
 a_c=\omega_La-\omega_\Lambda b
 -[\omega_L,\omega_\Lambda](a+b)+\cdots.
\]
When all four infinitesimal parameters \(\omega_L,\omega_\Lambda,a,b\)
are counted as first order, as in part (vii), the last displayed
commutator term is cubic.  The bilinear truncation is therefore
\[
 a_c=\omega_La-\omega_\Lambda b+O(3).
\]

\item[(v)] The Lorentz component found above immediately gives
\[
 \omega_c=[\omega_L,\omega_\Lambda]
 =\omega_L\omega_\Lambda-\omega_\Lambda\omega_L.
\]

\item[(vi)] A Lorentz matrix obeys
\[
 \Lambda^{\mathsf T}\eta\Lambda=\eta,
 \qquad
 \eta_{\rho\sigma}\Lambda^\rho{}_{\mu}
 \Lambda^\sigma{}_{\nu}=\eta_{\mu\nu}.
\]
At first order,
\[
 \omega_\Lambda^{\mathsf T}\eta+\eta\omega_\Lambda=0.
\]
Thus, after lowering the first index,
\[
 (\omega_\Lambda)_{\mu\nu}
 :=\eta_{\mu\rho}(\omega_\Lambda)^\rho{}_{\nu}
 =-(\omega_\Lambda)_{\nu\mu}.
\]

\item[(vii)] Define
\[
 A=\frac12\omega_L^{\mu\nu}M_{\mu\nu}+b^\mu P_\mu,
 \qquad
 B=\frac12\omega_\Lambda^{\rho\sigma}M_{\rho\sigma}+a^\rho P_\rho.
\]
The representation property turns the group commutator into
\[
 U[(L,b)]U[(\Lambda,a)]U[(L,b)]^{-1}U[(\Lambda,a)]^{-1}.
\]
Multiplying the four infinitesimal expansions, or using the
Baker--Campbell--Hausdorff formula, gives
\[
 (I+A)(I+B)(I-A)(I-B)=I+[A,B]+O(3).
\]
On the other hand, parts (iv)--(v) give
\[
 U[(\Lambda_c,a_c)]
 =I+\frac12[\omega_L,\omega_\Lambda]^{\mu\nu}M_{\mu\nu}
 +(\omega_La-\omega_\Lambda b)^\mu P_\mu+O(3),
\]
which is precisely the asserted identity.

\item[(viii)] Equating independently the coefficients of
\(a^\mu b^\nu\), \(\omega_L^{\mu\nu}a^\rho\), and
\(\omega_L^{\mu\nu}\omega_\Lambda^{\rho\sigma}\) gives, in the
anti-Hermitian generator normalization supplied in the source,
\[
 [P_\mu,P_\nu]=0,
\]
\[
 [M_{\mu\nu},P_\rho]
 =\eta_{\nu\rho}P_\mu-\eta_{\mu\rho}P_\nu,
\]
and
\[
\begin{split}
 [M_{\mu\nu},M_{\rho\sigma}]
 ={}&\eta_{\nu\rho}M_{\mu\sigma}
    -\eta_{\mu\rho}M_{\nu\sigma}\\
   &+\eta_{\mu\sigma}M_{\nu\rho}
    -\eta_{\nu\sigma}M_{\mu\rho}.
\end{split}
\]
The absence of an explicit \(i\) follows from the source expansion
\(U=I+\frac12\omega M+aP+\cdots\): unitarity makes these generators
anti-Hermitian.  To return to the Hermitian convention used elsewhere in
this volume, set
\[
 \widehat M_{\mu\nu}=-iM_{\mu\nu},\qquad
 \widehat P_\mu=-iP_\mu.
\]
Then
\[
 [\widehat M_{\mu\nu},\widehat P_\rho]
 =i(\eta_{\mu\rho}\widehat P_\nu
    -\eta_{\nu\rho}\widehat P_\mu),
\]
and
\[
\begin{split}
 [\widehat M_{\mu\nu},\widehat M_{\rho\sigma}]
 =i(&\eta_{\mu\rho}\widehat M_{\nu\sigma}
    -\eta_{\nu\rho}\widehat M_{\mu\sigma}\\
   &+\eta_{\nu\sigma}\widehat M_{\mu\rho}
    -\eta_{\mu\sigma}\widehat M_{\nu\rho}),
\end{split}
\]
in agreement with the Chapter~2 convention.
\end{enumerate}

\SupplementarySolution{4}{Pauli--Lubanski Spin and the Massive Casimir}
\begin{enumerate}
\item
The differential operators give
\[
 [J^{\mu\nu},P^\rho]
 =i\bigl(\eta^{\mu\rho}P^\nu-\eta^{\nu\rho}P^\mu\bigr).
\]
Antisymmetry alone gives
\[
 \mathcal W_\mu P^\mu
 =\frac12\epsilon_{\mu\nu\rho\sigma}
 J^{\nu\rho}P^\sigma P^\mu=0,
\]
because the momentum operators commute, making
\(P^\sigma P^\mu\) symmetric.  Next,
\[
 [\mathcal W_\mu,P^\lambda]
 =\frac12\epsilon_{\mu\nu\rho\sigma}
 [J^{\nu\rho},P^\lambda]P^\sigma.
\]
Inserting the Poincar\'e commutator produces two terms, each containing a
symmetric product of momenta contracted with an antisymmetric pair of
indices.  They cancel, so
\([\mathcal W_\mu,P^\lambda]=0\).

For the remaining commutator, expand both Pauli--Lubanski operators, move
momenta through one another, and use \([J,J]\) plus the result just proved.
Terms with two momenta cancel in antisymmetric pairs, leaving
\[
 [\mathcal W_\mu,\mathcal W_\nu]
 =-i\epsilon_{\mu\nu\rho\sigma}\mathcal W^\rho P^\sigma .
\]
\item
At \(p^\mu=(m,\mathbf0)\), the antisymmetry of
\(\epsilon_{\mu\nu\rho\sigma}\) gives
\[
 \mathcal W^0=0,\qquad
 \boldsymbol{\mathcal W}=\pm m\mathbf J,
\]
where the overall sign depends on the placement of indices in the
Levi--Civita convention and is irrelevant to the square.  Thus, on a
spin-\(j\) rest state,
\[
 \mathcal W^2=m^2\mathbf J^2=m^2j(j+1).
\]
The next part proves directly that \(\mathcal W^2\) is a Poincar\'e
Casimir, so its eigenvalue is the same in every inertial frame and
throughout an irreducible massive multiplet.
\item
Contracting
\(\epsilon_{\mu\nu\rho\sigma}\epsilon^{\mu\alpha\beta\gamma}\)
and expanding the generalized Kronecker delta gives
\[
 \mathcal W^2
 =-\frac12J_{\mu\nu}J^{\mu\nu}P^2
  +J_{\mu\rho}J_\nu{}^\rho P^\mu P^\nu .
\]
The translation commutators of \(P^2\) vanish immediately.  In
\([J^{\alpha\beta},P^2]\), the two terms from the Leibniz rule cancel after
interchanging the contracted momentum indices.  Thus \(P^2\) is a
Casimir.

The result of the first part gives
\[
 [\mathcal W^2,P^\lambda]
 =\mathcal W^\mu[\mathcal W_\mu,P^\lambda]
  +[\mathcal W^\mu,P^\lambda]\mathcal W_\mu=0.
\]
The remaining Lorentz commutator is most cleanly obtained from the finite
transformation law in the next part, avoiding the direct expansion
explicitly waived in the source problem.
\item
Under a translation,
\[
 J^{\nu\rho}\mapsto J^{\nu\rho}-a^\nu P^\rho+a^\rho P^\nu.
\]
The induced change in \(\mathcal W_\mu\) vanishes because it is
proportional to \(\epsilon_{\mu\nu\rho\sigma}P^\rho P^\sigma\).
For a Lorentz transformation, substituting the tensor law for \(J\), the
vector law for \(P\), and the determinant identity for the Levi--Civita
tensor gives
\[
 \mathcal W'_\mu=(\det\Lambda)\Lambda_\mu{}^\nu\mathcal W_\nu.
\]
It is a vector for proper transformations and changes the additional sign
appropriate to a pseudovector for improper ones.  In either case
\(\mathcal W'^2=\mathcal W^2\).  Its translation invariance and Lorentz
scalar character are precisely the finite group statements whose
infinitesimal versions are
\([\mathcal W^2,P^\mu]=[\mathcal W^2,J^{\mu\nu}]=0\), completing the
Casimir result.
\end{enumerate}

\SupplementarySolution{5}{Massive Little-Group Representations and Normalization}
\begin{enumerate}
\item
For a massive positive-energy representation,
\[
 P^2=-m^2,\qquad
 \mathcal W^2=m^2j(j+1).
\]
Thus \(j=0,\frac12,1\) give respectively
\[
 \mathcal W^2=0,\qquad
 \mathcal W^2=\frac34m^2,\qquad
 \mathcal W^2=2m^2.
\]
For a finite-helicity massless representation the null translations of
the little group vanish.  The remaining relation is
\(\mathcal W^\mu=\lambda P^\mu\), and hence
\[
 P^2=0,\qquad \mathcal W^2=\lambda^2P^2=0
\]
for every \(\lambda\).  The missing datum is the one-dimensional
representation of the rotational part of the little group---the helicity
\(\lambda\), including its sign.
\item
If \(\Lambda k=k\) for \(k=(m,\mathbf0)\), then
\(\Lambda^\mu{}_0=\delta^\mu{}_0\).  The Lorentz condition forces
\(\Lambda^0{}_i=0\) and leaves a spatial matrix \(R\) satisfying
\(R^{\mathsf T}R=\mathbf{1}\).  Properness gives \(\det R=1\), so
\[
 \Lambda=\begin{pmatrix}1&0\\0&R\end{pmatrix},
 \qquad R\in SO(3).
\]
The unitary irreducible representations of its double cover \(SU(2)\) are
labeled by \(j=0,\frac12,1,\ldots\) and have dimension \(2j+1\).  Inducing
each such little-group representation over the positive mass shell gives
the states \(\ket{p,\sigma}\), \(\sigma=-j,\ldots,j\).
\item
Unitarity first gives
\[
 \braket{p',\sigma'}{p,\sigma}
 =|N(p)|^2\delta_{\sigma'\sigma}
 \frac{p^0}{k^0}\delta^3(\mathbf p'-\mathbf p).
\]
Thus \(\delta^3\)-normalization requires
\[
 N(p)=\sqrt{\frac{k^0}{p^0}},
\]
up to a momentum-independent phase.  For a massive standard momentum
\(k^0=m\); for a massless standard momentum \(k^0=\kappa\), whose arbitrary
fixed value cancels from physical transformation laws.

Applying \(U(\Lambda)\), inserting
\(\mathbf{1}=U(L(\Lambda p))U(L(\Lambda p))^{-1}\), and using the ratio of
the two normalization factors gives
\[
 \frac{N(p)}{N(\Lambda p)}
 =\sqrt{\frac{(\Lambda p)^0}{p^0}}.
\]
This is the square-root multiplying the little-group matrix in
Eq.~\eqref{eq:2.5.23}.
\end{enumerate}

\SupplementarySolution{6}{Finite-Helicity Representations of the Null Little Group}
\begin{enumerate}
\item
Acting with the vector-representation matrices on
\(k=(\kappa,0,0,\kappa)\) shows
\((J_2+K_1)k=(-J_1+K_2)k=J_3k=0\).  The Lorentz algebra gives
\[
 [J_3,A]=iB,\qquad
 [J_3,B]=-iA,\qquad
 [A,B]=0.
\]
Thus \(A,B\) are two commuting translations rotated into one another by
\(J_3\).  This is the Lie algebra
\(\mathfrak{iso}(2)=\mathbb R^2\rtimes\mathfrak{so}(2)\), the algebra of
the two-dimensional Euclidean group.  It is the connected little group of
a null momentum in \(3+1\) dimensions.
\item
Choose a simultaneous eigenstate
\(\ket{a,b}\).  A rotation through \(\theta\) about the momentum sends its
eigenvalue pair to
\[
 (a_\theta,b_\theta)
 =(a\cos\theta+b\sin\theta,-a\sin\theta+b\cos\theta).
\]
If \(\rho^2=a^2+b^2>0\), these pairs sweep a full circle.  Distinct
eigenvalues of commuting Hermitian operators label orthogonal generalized
eigenstates, so the representation contains a continuous infinity of
polarizations.  A finite-dimensional polarization space is possible only
for \(\rho=0\), which means \(A=B=0\).  The remaining \(SO(2)\) character
is \(e^{i\lambda\theta}\) and supplies the helicity.
\item
Let
\[
 W(\Lambda,p)=L^{-1}(\Lambda p)\Lambda L(p).
\]
When null translations act trivially, only the rotation angle
\(\theta(\Lambda,p)\) of this little-group element remains.  With the
\(\delta^3\)-normalized states of Chapter 2,
\[
 U(\Lambda)\ket{p,\lambda}
 =\sqrt{\frac{(\Lambda p)^0}{p^0}}\,
 e^{i\lambda\theta(\Lambda,p)}
 \ket{\Lambda p,\lambda}.
\]
For a spatial rotation by \(\theta\) about \(\mathbf p\), the standard
boost may be chosen so that \(W\) is the same rotation.  Momentum is
unchanged and the state acquires \(e^{i\lambda\theta}\), confirming that
\(\lambda\) is the angular momentum along the direction of motion.
\end{enumerate}

\SupplementarySolution{7}{Finite-Dimensional \(SU(2)\) Representations and Charge Conjugation}
For readability we suppress the representation symbol \(d\) on the
generators.
\begin{enumerate}
\item[(a)] The algebra implies
\[
 [J_3,J_\pm]=\pm J_\pm,
 \qquad [J_+,J_-]=2J_3.
\]
Thus \(J_+v_m\) and \(J_-v_m\), when nonzero, have weights \(m+1\) and
\(m-1\).  Choose a highest-weight vector \(v_I\) with
\(J_+v_I=0\).  The quadratic Casimir
\[
 J^2=J_3^2+\frac12(J_+J_-+J_-J_+)
\]
commutes with all three generators.  Acting on the highest-weight vector
shows that its eigenvalue is \(I(I+1)\).  Positivity of norms then gives
\[
\begin{aligned}
 \lVert J_-\ket{I,m}\rVert^2&=(I+m)(I-m+1),\\
 \lVert J_+\ket{I,m}\rVert^2&=(I-m)(I+m+1).
\end{aligned}
\]
The descending ladder therefore terminates at \(m=-I\), so
\(2I\in\mathbb Z_{\ge0}\).  Irreducibility says that this ladder exhausts
the representation:
\[
 m=I,I-1,\ldots,-I,
 \qquad \dim d=2I+1.
\]

\item[(b)] With phases chosen so that the ladder coefficients are
positive,
\[
 J_-\ket{I,m}
 =\sqrt{(I+m)(I-m+1)}\ket{I,m-1}.
\]
Put \(n=I-m\).  Repeated lowering from the highest-weight state gives
\[
 (J_-)^n\ket{I,I}
 =\left[\prod_{k=0}^{n-1}(2I-k)(k+1)\right]^{1/2}
 \ket{I,I-n}.
\]
The product is
\[
 \prod_{k=0}^{n-1}(2I-k)(k+1)
 =\frac{(2I)!\,n!}{(2I-n)!}
 =\frac{(2I)!(I-m)!}{(I+m)!}.
\]
Consequently
\[
 \boxed{A(I,m)=
 \sqrt{\frac{(I+m)!}{(2I)!(I-m)!}}}\,.
\]

\item[(c)] The transformed generators are
\[
 J_1'=-J_1,\qquad J_2'=J_2,\qquad J_3'=-J_3.
\]
Directly,
\[
 [J_1',J_2']=iJ_3',\qquad
 [J_2',J_3']=iJ_1',\qquad
 [J_3',J_1']=iJ_2'.
\]
Equivalently, conjugation preserves every commutator:
\([J_i',J_j']=C[J_i,J_j]C^{-1}=i\epsilon_{ijk}J_k'\).
It is also useful below that
\[
 CJ_+C^{-1}=-J_-,\qquad CJ_-C^{-1}=-J_+.
\]

\item[(d)] Fix the relative phases of the triplet by
\[
 J_+\ket{\pi^0}=\sqrt2\ket{\pi^+},
 \qquad
 J_-\ket{\pi^0}=\sqrt2\ket{\pi^-}.
\]
Using \(C\ket{\pi^0}=\ket{\pi^0}\), conjugating the first relation gives
\[
 \sqrt2\,C\ket{\pi^+}
 =CJ_+C^{-1}\ket{\pi^0}
 =-J_-\ket{\pi^0}
 =-\sqrt2\ket{\pi^-}.
\]
The second relation similarly gives
\[
 \boxed{C\ket{\pi^+}=-\ket{\pi^-},\qquad
 C\ket{\pi^-}=-\ket{\pi^+}}.
\]
The common minus sign depends on the stated ladder-phase convention;
independent rephasings of the charged basis vectors can move it between
the ladder relations and the action of \(C\), without changing the fact
that charge conjugation exchanges the two states.
\end{enumerate}

\SupplementarySolution{8}{A Wigner Rotation from Non-Collinear Boosts}
\begin{enumerate}
\item
Represent the two boosts by their rapidity matrices and factor
\[
 B_z(\eta)B_y(\xi)=L(p_{\rm final})R_x(\Omega),
\]
where \(L(p_{\rm final})\) is the unique rotation-free boost taking the
rest momentum to the final momentum.  Comparing the off-diagonal entries
(equivalently, using the polar decomposition of the corresponding
\(SL(2,\mathbb C)\) matrices) gives
\[
 \tan\frac{\Omega}{2}
 =\frac{\sinh(\eta/2)\sinh(\xi/2)}
 {\cosh(\eta/2)\cosh(\xi/2)}
 =\tanh\frac{\eta}{2}\tanh\frac{\xi}{2}.
\]
The axis is parallel to the oriented cross product of the second and first
boost directions, here the \(x\)-axis up to convention.  Replacing an
active boost of states by the passive boost of coordinates inverts one
boost and therefore reverses \(\Omega\), but leaves every mixing
probability unchanged.
\item
The absolute squares of the spin-one rotation matrix elements are
independent of whether the transverse rotation axis is chosen as \(x\) or
\(y\).  For an initial \(\sigma=0\),
\[
 P(0\to0)=\cos^2\Omega,\qquad
 P(0\to+1)=P(0\to-1)=\frac12\sin^2\Omega.
\]
For an initial \(\sigma=+1\),
\[
\begin{aligned}
 P(+1\to+1)&=\cos^4\frac{\Omega}{2},\\
 P(+1\to0)&=\frac12\sin^2\Omega,\\
 P(+1\to-1)&=\sin^4\frac{\Omega}{2}.
\end{aligned}
\]
Each set sums to one.  The phases of the amplitudes depend on the choice of
\(x\)- versus \(y\)-axis and on active/passive conventions, but these
probabilities do not.
\end{enumerate}

\SupplementarySolution{9}{The Lorentz Group through \(SL(2,\mathbb C)\)}
\begin{enumerate}
\item
The determinant is
\[
 \det X=(x^0)^2-\mathbf x^2=-x_\mu x^\mu.
\]
For \(X'=AXA^\dagger\),
\[
 \det X'=\det A\,\det X\,\det A^\dagger=\det X,
\]
so the induced real linear map preserves the Minkowski norm.  Hermitian
\(X\) remains Hermitian, so real four-vectors map to real four-vectors.
Continuity from \(A=\mathbf{1}\) places the image in the proper
orthochronous component.

If \(AXA^\dagger=X\) for every Hermitian \(X\), then \(A\) commutes with
the full matrix algebra and is proportional to \(\mathbf{1}\).  The
determinant-one condition and the action on \(X=\mathbf{1}\) leave
\(A=\pm\mathbf{1}\).  Thus
\[
 SL(2,\mathbb C)/\{\pm\mathbf{1}\}\simeq SO^+(3,1).
\]
\item
Using
\([J_i,J_j]=i\epsilon_{ijk}J_k\),
\([J_i,K_j]=i\epsilon_{ijk}K_k\), and
\([K_i,K_j]=-i\epsilon_{ijk}J_k\), one finds
\[
 [A_i,A_j]=i\epsilon_{ijk}A_k,\qquad
 [B_i,B_j]=i\epsilon_{ijk}B_k,\qquad
 [A_i,B_j]=0.
\]
After complexification, finite-dimensional irreducible representations are
therefore tensor products labeled by
\((j_A,j_B)\), of dimension \((2j_A+1)(2j_B+1)\).

The Lorentz group is noncompact.  A faithful nontrivial finite-dimensional
unitary representation would have compact image in a unitary group, which
cannot represent the unbounded boost subgroup faithfully.  Correspondingly,
\(\mathbf A\) and \(\mathbf B\) are not separately Hermitian in these
finite multiplets.  Unitary particle representations instead act on
infinite-dimensional momentum-space Hilbert spaces.
\end{enumerate}

\SupplementarySolution{10}{Bispinors and the Sign of a \(2\pi\) Rotation}
\begin{enumerate}
\item
Write \(V=V^\mu\sigma_\mu\).  Under
\(A\in SL(2,\mathbb C)\),
\[
 V_{\alpha\dot\alpha}\longmapsto
 A_\alpha{}^\beta
 (A^\dagger)^{\dot\beta}{}_{\dot\alpha}
 V_{\beta\dot\beta},
\]
which is the tensor product of the undotted
\((\frac12,0)\) and dotted \((0,\frac12)\) representations:
\((\frac12,\frac12)\).  Expanding the resulting matrix uniquely in the
basis \(\{\sigma_\mu\}\) defines \(V'^\mu=\Lambda^\mu{}_\nu V^\nu\), and
S.2.9(a) proves that \(\Lambda\) is Lorentz.

A general bispinor has \(2\times2=4\) complex components, matching a
complex four-vector.  Imposing Hermiticity on \(V\) reduces these to four
real components and gives an ordinary real Lorentz vector.
\item
At \(\theta=2\pi\),
\[
 A(2\pi)=e^{-i\pi\sigma_3}=-\mathbf{1},
\]
whereas at \(4\pi\), \(A(4\pi)=+\mathbf{1}\).  Both
\(\pm\mathbf{1}\) induce the same Lorentz transformation, so the image path
at \(2\pi\) is already the identity rotation although the lifted path in
\(SL(2,\mathbb C)\) has not closed.

In the spin-\(j\) representation,
\[
 U(R(2\pi))=(-1)^{2j}\mathbf{1},\qquad U(R(4\pi))=\mathbf{1}.
\]
Thus half-integer spin gives a representation of the Lorentz group only up
to a sign, or equivalently an ordinary representation of its double cover.
This is the topological source of the familiar projective representation.
\end{enumerate}

\SupplementarySolution{11}{The Bargmann Cocycle and Mass Superselection}
\begin{enumerate}
\item
Let \(T(\mathbf a)\psi(\mathbf x,t)=\psi(\mathbf x-\mathbf a,t)\).
Using the stated boost law in the two possible orders gives
\[
 U(\mathbf v)T(\mathbf a)
 =e^{im\mathbf v\cdot\mathbf a}
   T(\mathbf a)U(\mathbf v).
\]
The extra phase cannot be seen in the spacetime action of the Galilei
group; it is a projective multiplier.  Expanding both sides to first order
in \(\mathbf v\) and \(\mathbf a\), with the generator conventions of
Chapter 2, yields
\[
 [K_i,P_j]=i\delta_{ij}M,\qquad M\ket{\psi}=m\ket{\psi}.
\]
The mass is central because the phase is a scalar and commutes with every
spacetime transformation.
\item
The relation in S.2.11(a) immediately gives
\[
 U(\mathbf v)T(\mathbf a)U(\mathbf v)^{-1}T(\mathbf a)^{-1}
 =e^{im\mathbf v\cdot\mathbf a}\mathbf{1}.
\]
The associated sequence of coordinate transformations has zero net boost
and zero net translation, so it is the identity in the unextended Galilei
group.  Quantum mechanically it is a nontrivial element of the center.
Infinitesimally, the exponent is just the central
boost--translation commutator.
\item
For
\(\ket{\Psi}=c_1\ket{\psi_{m_1}}+c_2\ket{\psi_{m_2}}\), the loop gives
\[
 \ket{\Psi}\longmapsto
 c_1e^{im_1\mathbf v\cdot\mathbf a}\ket{\psi_{m_1}}
 +c_2e^{im_2\mathbf v\cdot\mathbf a}\ket{\psi_{m_2}}.
\]
Unless \(m_1=m_2\), this changes the relative phase even though the
underlying spacetime transformation is the identity.  Requiring the
identity to preserve every ray therefore forbids coherent superpositions
of different masses: Bargmann's mass-superselection rule.

In the centrally extended Galilei group, however, the loop is not declared
to be the identity; it is \(e^{i\alpha M}\).  If \(M\) is treated as an
operator with an additional conjugate degree of freedom, different mass
sectors are ordinary eigenspaces of the center.  The algebra alone then
does not forbid their superposition; a superselection rule requires an
additional physical restriction on observables or accessibility of that
central degree of freedom.
\end{enumerate}

\SupplementarySolution{12}{The Heisenberg Group and Its Schr\"odinger Representation}
\begin{enumerate}
\item[(a)]
  \begin{enumerate}
  \item[(i)] The integers are closed under addition, addition is
  associative, \(0\) is the identity, and the inverse of \(n\) is
  \(-n\).  Thus \((\mathbb Z,+)\) is a group.
  \item[(ii)] Multiplication is associative and closed on
  \(\mathbb Z^*\), with identity \(1\), but an element such as \(2\) has
  no multiplicative inverse in \(\mathbb Z^*\).  Therefore
  \((\mathbb Z^*,\times)\) is not a group.
  \item[(iii)] Write \(f_{u,b}(x)=ux+b\).  Then
  \[
   f_{u,b}\circ f_{u',b'}=f_{uu',\,ub'+b},
  \]
  which remains in \(\mathcal L\).  Composition is associative, the
  identity is \(f_{1,0}\), and
  \[
   f_{u,b}^{-1}=f_{u^{-1},-b/u}.
  \]
  Hence \((\mathcal L,\circ)\) is the affine group of the line.
  \end{enumerate}

\item[(b)] Denote the displayed matrix by \(g(q,r,s)\).  Direct matrix
multiplication gives
\[
 g(q,r,s)g(q',r',s')
 =g(q+q',r+r',s+s'+qr'),
\]
and therefore
\[
 g(q,r,s)^{-1}=g(-q,-r,-s+qr).
\]
  \begin{enumerate}
  \item[(i)] Two elements commute precisely when
  \(qr'=q'r\).  For this to hold for every \(q',r'\), one must have
  \(q=r=0\), while \(s\) is unrestricted.  Thus
  \[
   Z(G)=\{g(0,0,s)\mid s\in\mathbb R\}.
  \]

  \item[(ii)] The Lie algebra consists of strictly upper-triangular
  matrices
  \[
   X(q,r,s)=
   \begin{pmatrix}0&q&s\\0&0&r\\0&0&0\end{pmatrix}
   =qQ+rR+sS,
  \]
  with
  \[
   Q=E_{12},\qquad R=E_{23},\qquad S=E_{13}.
  \]
  The only nonzero basis brackets are
  \[
   [Q,R]=S,\qquad [R,Q]=-S.
  \]
  Hence \(f_{QR}{}^S=1\), \(f_{RQ}{}^S=-1\), and all other structure
  constants vanish.

  \item[(iii)] Since \(X^2=qrS\) and \(X^3=0\),
  \[
   e^{X(q,r,s)}=I+X+\frac12X^2
   =g\!\left(q,r,s+\frac12qr\right).
  \]
  Every \(g(q,r,s_0)\) therefore has the unique logarithm
  \[
   X\!\left(q,r,s_0-\frac12qr\right).
  \]
  Thus the exponential map is one-to-one and onto.  It identifies the
  underlying manifold of \(G\) continuously with \(\mathbb R^3\), so
  \(G\) is connected (indeed, simply connected).

  \item[(iv)] The one-dimensional space \(\mathbb R\,S\) is a nonzero
  central ideal, so \(L(G)\) is not simple.  Moreover,
  \[
   [L(G),L(G)]=\mathbb R\,S,\qquad
   [\mathbb R\,S,\mathbb R\,S]=0.
  \]
  The algebra is nilpotent, its Killing form is degenerate, and it is
  not a direct sum of simple ideals.  It is therefore neither simple nor
  semisimple.
  \end{enumerate}

\item[(c)] The condition \((\psi,\psi)=1\) in the source describes the
unit sphere rather than a vector space; the representation acts linearly
on the full Hilbert space \(L^2(\mathbb R)\) and preserves that unit
sphere.  Let \(g=g(q,r,s)\) and \(g'=g(q',r',s')\).  Then
\[
\begin{split}
 (D(g)D(g')\psi)(x)
 &=e^{-i(s+s')}e^{irx}e^{ir'(x-q)}
   \psi(x-q-q')\\
 &=e^{-i(s+s'+qr')}e^{i(r+r')x}
   \psi(x-q-q')\\
 &=(D(gg')\psi)(x).
\end{split}
\]
Thus \(D\) respects the exact matrix-group multiplication law.  It also
preserves the inner product:
\[
\begin{split}
 (D(g)\phi,D(g)\psi)
 &=\int_{-\infty}^{\infty}dx\,
   \phi(x-q)^*\psi(x-q)\\
 &=\int_{-\infty}^{\infty}dy\,\phi(y)^*\psi(y)
 = (\phi,\psi).
\end{split}
\]
Since \(D(g^{-1})=D(g)^{-1}\), this is a unitary representation.

With \(\widehat p=-i\,d/dx\) and \(\widehat x\) multiplication by \(x\),
the operator can be written
\[
 D(g)=e^{-is}e^{ir\widehat x}e^{-iq\widehat p}.
\]
Thus \(q\) translates position, \(r\) translates momentum, and \(s\) is
an overall phase.  The extra term \(qr'\) in the group law is the central
phase forced by the canonical noncommutativity of position and momentum;
this is the Schr\"odinger representation of the Heisenberg group.
\end{enumerate}

\SupplementarySolution{13}{The Poincar\'e Group in \(1+2\) Dimensions}
Throughout this solution, scalar products use the source metric
\[
 \eta_{\mathrm s}=\operatorname{diag}(+1,-1,-1).
\]
\begin{enumerate}
\item[(a)] The Lorentz condition is
\[
 \Lambda^{\mathsf T}\eta_{\mathrm s}\Lambda=\eta_{\mathrm s}.
\]
Taking determinants gives
\[
 (\det\Lambda)^2=1,
 \qquad \det\Lambda=\pm1.
\]
The norm of the zeroth column gives
\[
 (\Lambda^0{}_{0})^2-(\Lambda^1{}_{0})^2
 -(\Lambda^2{}_{0})^2=1,
\]
and therefore \(|\Lambda^0{}_{0}|\ge1\).  The proper component has
\(\det\Lambda=+1\), and the orthochronous component containing the
identity has \(\Lambda^0{}_{0}\ge1\).

\item[(b)] Map a Minkowski vector to the real symmetric matrix
\[
 X=\varphi(x)=x^0\tau_0+x^1\tau_1+x^2\tau_2
 =\begin{pmatrix}x^0+x^1&x^2\\x^2&x^0-x^1\end{pmatrix}.
\]
Its determinant is exactly the Minkowski norm:
\[
 \det X=(x^0)^2-(x^1)^2-(x^2)^2=x^2.
\]
For \(A\in SL(2,\mathbb R)\), define
\[
 X' = AXA^{\mathsf T}=\varphi(x').
\]
The result is again real and symmetric, and
\[
 \det X'=\det A\,\det X\,\det A^{\mathsf T}=\det X.
\]
Thus \(x'=\Lambda(A)x\) for a Lorentz transformation.  Moreover,
\[
 \varphi(\Lambda(AB)x)=ABX(AB)^{\mathsf T}
 =A(BXB^{\mathsf T})A^{\mathsf T},
\]
so \(\Lambda(AB)=\Lambda(A)\Lambda(B)\).

For an explicit formula, write
\[
 A=\begin{pmatrix}a&b\\c&d\end{pmatrix},
 \qquad ad-bc=1.
\]
Then
\[
 \Lambda(A)=
 \begin{pmatrix}
 \frac12(a^2+b^2+c^2+d^2)&
 \frac12(a^2-b^2+c^2-d^2)&ab+cd\\
 \frac12(a^2+b^2-c^2-d^2)&
 \frac12(a^2-b^2-c^2+d^2)&ab-cd\\
 ac+bd&ac-bd&ad+bc
 \end{pmatrix}.
\]
Continuity places the image in \(SO^+(2,1)\).  The differential at the
identity is an isomorphism of the three-dimensional Lie algebras, so the
image is all of the connected group.  The kernel is
\[
 \ker\Lambda=\{I,-I\}.
\]
Hence
\[
 SO^+(2,1)\simeq SL(2,\mathbb R)/\{\pm I\},
\]
and the map is a two-to-one covering homomorphism, not an isomorphism.

\item[(c)] For the full Lorentz group,
\[
 \operatorname{ISO}(2,1)=\mathbb R^{1,2}\rtimes O(2,1),
\]
while its identity component is
\[
 \operatorname{ISO}^+(2,1)=\mathbb R^{1,2}\rtimes SO^+(2,1).
\]
The group law is
\[
 (a,\Lambda)(b,L)=(a+\Lambda b,\Lambda L).
\]
Thus translations form a normal subgroup, but Lorentz transformations
act nontrivially on them; this is a semidirect rather than a direct
product.  In momentum space that action organizes momenta into Lorentz
orbits.  For a representative momentum \(k\), its little group is
\[
 G_k=\{\Lambda\in SO^+(2,1)\mid\Lambda k=k\}.
\]
Wigner's construction combines a translation character
\(e^{-ia\cdot p}\) on a Lorentz orbit with a unitary irreducible
representation of \(G_k\).

\item[(d)] A future-directed massive orbit has \(p^2=m^2\), \(p^0>0\),
and representative
\[
 k=(m,0,0).
\]
Its connected little group is the spatial rotation group \(SO(2)\).
Every irreducible unitary representation is a character
\[
 D_s(R(\theta))=e^{is\theta}.
\]
For the group \(SO(2)\) itself, single-valuedness requires
\(s\in\mathbb Z\); the double cover permits
\(s\in\tfrac12\mathbb Z\); and the universal cover has
\(\theta\in\mathbb R\), allowing arbitrary real \(s\).  The last
possibility is the group-theoretic origin of anyonic spin in
\(2+1\) dimensions.

Choose a standard Lorentz transformation \(L(p)k=p\).  The induced
massive representation may be written
\[
 U(a,\Lambda)\ket{p,s}
 =e^{-ia\cdot\Lambda p}
  e^{is\Omega(\Lambda,p)}\ket{\Lambda p,s},
\]
up to the normalization factor dictated by the choice of momentum-state
normalization, where
\[
 L(\Lambda p)^{-1}\Lambda L(p)=R(\Omega(\Lambda,p)).
\]
Thus a massive irreducible representation is labeled by \(m>0\) and the
allowed spin \(s\).

\item[(e)] Take the future-directed null representative
\(k=(\kappa,\kappa,0)\).  In the \(SL(2,\mathbb R)\) description it is
\[
 \varphi(k)=\begin{pmatrix}2\kappa&0\\0&0\end{pmatrix}.
\]
Its connected stabilizer is represented by
\[
 A(u)=\begin{pmatrix}1&u\\0&1\end{pmatrix},
 \qquad u\in\mathbb R.
\]
Using the explicit map from part (b), the corresponding Lorentz matrix is
\[
 N(u)=
 \begin{pmatrix}
 1+\frac12u^2&-\frac12u^2&u\\
 \frac12u^2&1-\frac12u^2&u\\
 u&-u&1
 \end{pmatrix},
 \qquad N(u)k=k,
\]
and \(N(u)N(v)=N(u+v)\).  Hence the connected massless little group is
\((\mathbb R,+)\).  Its irreducible unitary representations are the
one-dimensional characters
\[
 D_\rho(N(u))=e^{i\rho u},
 \qquad \rho\in\mathbb R.
\]
Together with the null momentum orbit and translation character, each
\(\rho\) induces a massless Poincar\'e representation.  The conventional
single-polarization massless particle described by a finite-component
local field uses the trivial character \(\rho=0\); nonzero \(\rho\) gives
the nontrivial null-rotation family.
\end{enumerate}

\SupplementarySolution{14}{Parity Symmetry of Scalar Field Theory}
\begin{enumerate}
\item[(a)] Differentiating the parity transformation gives
\[
\begin{aligned}
 P\,\partial_0\phi(t,\mathbf x)P^{-1}
 &=\eta_P\,\partial_0\phi(t,-\mathbf x),\\
 P\,\partial_i\phi(t,\mathbf x)P^{-1}
 &=-\eta_P\,\partial_i\phi(t,-\mathbf x).
\end{aligned}
\]
The two signs from each quadratic term cancel, as does
\(\eta_P^2=1\), so
\[
 P\mathcal L_0(t,\mathbf x)P^{-1}=\mathcal L_0(t,-\mathbf x).
\]
The density at a fixed point is generally not unchanged.  The
Lagrangian is, because a change of variables in the spatial integral
gives
\[
 P L_0(t)P^{-1}
 =\int d^3\mathbf x\,\mathcal L_0(t,-\mathbf x)
 =\int d^3\mathbf y\,\mathcal L_0(t,\mathbf y)
 =L_0(t).
\]
This works for either intrinsic parity.

\item[(b)] Comparing the transformed mode expansion with
\(\eta_P\phi(t,-\mathbf x)\) gives
\[
 Pa_{\mathbf k}P^{-1}=\eta_Pa_{-\mathbf k},
 \qquad
 Pa_{\mathbf k}^\dagger P^{-1}=\eta_Pa_{-\mathbf k}^\dagger.
\]
For
\[
 \ket{\mathbf k_1,\ldots,\mathbf k_n}
 =a_{\mathbf k_1}^\dagger\cdots
  a_{\mathbf k_n}^\dagger\ket0,
\]
vacuum invariance then gives
\[
 P\ket{\mathbf k_1,\ldots,\mathbf k_n}
 =\eta_P^n\ket{-\mathbf k_1,\ldots,-\mathbf k_n}.
\]

\item[(c)] Let
\[
 N=\sum_{\mathbf q}a_{\mathbf q}^\dagger a_{\mathbf q},
 \qquad
 B=\sum_{\mathbf q}a_{\mathbf q}^\dagger a_{-\mathbf q}.
\]
The canonical algebra gives
\[
 [N,a_{\mathbf k}]=-a_{\mathbf k},
\]
so the supplied Campbell--Baker--Hausdorff series sums to
\[
 P_1a_{\mathbf k}P_1^{-1}
 =e^{i\pi/2}a_{\mathbf k}=ia_{\mathbf k}.
\]
For the second operator,
\[
 [B,a_{\mathbf k}]=-a_{-\mathbf k},\qquad
 [B,[B,a_{\mathbf k}]]=a_{\mathbf k},
\]
and the nested commutators alternate.  Therefore
\[
 e^{i\alpha B}a_{\mathbf k}e^{-i\alpha B}
 =\cos\alpha\,a_{\mathbf k}
 -i\sin\alpha\,a_{-\mathbf k}.
\]
Putting \(\alpha=\eta_P\pi/2\) gives
\[
 P_2a_{\mathbf k}P_2^{-1}=-i\eta_Pa_{-\mathbf k}.
\]
Both \(N\) and \(B\) are Hermitian: for \(B\), take the adjoint and
relabel \(\mathbf q\mapsto-\mathbf q\).  Hence \(P_1\), \(P_2\), and
their product are unitary.  Finally,
\[
 Pa_{\mathbf k}P^{-1}
 =P_1(-i\eta_Pa_{-\mathbf k})P_1^{-1}
 =\eta_Pa_{-\mathbf k}.
\]
The adjoint relation follows similarly, and substitution in the field
expansion proves
\[
 P\phi(t,\mathbf x)P^{-1}=\eta_P\phi(t,-\mathbf x).
\]

\item[(d)] Extend parity to a complex field by
\[
 P\Phi(t,\mathbf x)P^{-1}=\eta_P\Phi(t,-\mathbf x),
 \qquad
 P\Phi^\dagger(t,\mathbf x)P^{-1}
 =\eta_P^*\Phi^\dagger(t,-\mathbf x).
\]
In oscillator language this is
\[
 Pa_{\mathbf k}P^{-1}=\eta_Pa_{-\mathbf k},
 \qquad
 Pb_{\mathbf k}P^{-1}=\eta_P^*b_{-\mathbf k},
\]
for particle and antiparticle annihilators.  The phases cancel in the
current.  With
\[
 \mathcal P^\mu{}_{\nu}=\operatorname{diag}(1,-1,-1,-1),
\]
the result is the vector transformation law
\[
 \boxed{
 Pj^\mu(t,\mathbf x)P^{-1}
 =\mathcal P^\mu{}_{\nu}j^\nu(t,-\mathbf x)}.
\]
Equivalently,
\[
 Pj^0(t,\mathbf x)P^{-1}=j^0(t,-\mathbf x),
 \qquad
 P\mathbf j(t,\mathbf x)P^{-1}=-\mathbf j(t,-\mathbf x).
\]
Thus charge density is parity even and the spatial particle current is
parity odd, as required for a Lorentz vector.
\end{enumerate}
\end{enumerate}
