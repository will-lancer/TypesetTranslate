\chapterbackmatter{Solutions to Weinberg Exercises}
\ExerciseEditorialNote

\begin{enumerate}
\WeinbergSolution{1}
Write
\[
 f(\beta)=\frac1\beta\ln\frac{1+\beta}{1-\beta}.
\]
In the center-of-mass frame the four signed charges
\(c_n=\eta_ne_n/e\), in the order \(e^+,e^-,\pi^+,\pi^-\), are
\((-1,+1,+1,-1)\).  At \(90^\circ\) every electron--pion relative speed is
the same, so the four mixed terms in the double sum in Eq.~(13.2.8)
cancel.  The two opposite-particle pairs remain, including both orderings,
and the diagonal terms give \(f(0)=2\).  Hence
\[
 A=\frac{\alpha}{\pi}\{f(\beta_e)+f(\beta_\pi)-4\},\qquad
 \beta_a=\frac{2v_a}{1+v_a^2},\quad
 v_a=\sqrt{1-\frac{4m_a^2}{s}},
 \quad s=(1\ {\rm GeV})^2 .
\]
Numerically \(f(\beta_e)\simeq30.31\), \(f(\beta_\pi)\simeq7.80\), and
\(A\simeq7.9\times10^{-2}\).

There is no separate single-photon resolution in the measurement stated in
the question, so set \(E=E_T\) in Eq.~(13.3.17).  The inclusive rate is
\[
 \boxed{\quad
 \Gamma(E_T)=F(1;A)\left(\frac{E_T}{\Lambda}\right)^A\Gamma^\Lambda
 \ \propto\ E_T^{\,0.079}.
 \quad}
\]
The unphysical separator \(\Lambda\) is cancelled by the hard rate
\(\Gamma^\Lambda\).  Also \(F(1;A)=1-\pi^2A^2/12+O(A^3)\), so it does not
alter the requested power of \(E_T\).

\WeinbergSolution{2}
For each massive external particle define the on-shell soft residue \(g_n\)
of \(J\).  With \(\eta_n=+1\) for an outgoing leg and \(-1\) for an incoming
leg, emission of one scalar of momentum \(q=\omega(1,\widehat{\mathbf q})\)
has the universal limit
\[
 \mathcal M_{\beta\alpha}(q)=
 \left[\sum_n\frac{\eta_ng_n}{p_n\cdot q-\ii\eta_n0}\right]
 \mathcal M_{\beta\alpha}+O(\omega^0).
\]
The leading term for \(N\) scalars is the product of \(N\) such factors.
Real-emission phase space and the virtual soft graphs therefore exponentiate
exactly as in the photon calculation, with
\[
 A_\phi=\frac{1}{2(2\pi)^3}\int \dd\Omega_{\widehat{\mathbf q}}\,
 \left|\sum_n\frac{\eta_ng_n}
 {p_n^0-\mathbf p_n\cdot\widehat{\mathbf q}}\right|^2\geq0.
\]
Equivalently,
\[
 A_\phi=-\frac1{8\pi^2}\sum_{n,m}
 \frac{\eta_n\eta_mg_ng_m}{(p_n\cdot p_m)\beta_{nm}}
 \ln\frac{1+\beta_{nm}}{1-\beta_{nm}},
\]
with the coincident-velocity value understood by continuity.

After summing every multiplicity whose total scalar energy is below \(E_T\),
and including virtual scalars below \(\Lambda\), the regulator-independent
answer is
\[
 \boxed{\quad
 \Gamma_\phi(E_T)=F(1;A_\phi)
 \left(\frac{E_T}{\Lambda}\right)^{A_\phi}\Gamma_\phi^\Lambda .
 \quad}
\]
Here \(\Gamma_\phi^\Lambda\) contains the hard radiative corrections.  A
temporary scalar mass or lower energy cutoff cancels between its real and
virtual appearances.

\WeinbergSolution{3}
Let \(\omega\) collectively denote the incident and outgoing photon
energies.  The low-energy theorem is most compactly stated by introducing
the target's on-shell electromagnetic vertex through first order in the
momentum transfer,
\[
 \langle p',\sigma'|J^0(0)|p,\sigma\rangle
   =e\,\delta_{\sigma'\sigma}+O(k^2),
\]
\[
 \langle p',\sigma'|\mathbf J(0)|p,\sigma\rangle
   =\left[\frac{e(\mathbf p'+\mathbf p)}{2m}
   +\frac{\ii\mu}{j}\,\mathbf J\times(\mathbf p'-\mathbf p)\right]_
   {\sigma'\sigma}+O(k^2).
\]
Form the two elastic pole graphs with this vertex, one with intermediate
momentum \(p+q\) and the crossed one with momentum \(p-q'\), and add the
contact term required by the two Ward identities.  Denote the resulting
gauge-invariant tensor by \(B^{\mu\nu}(e,m,\mu,j;q',q)\).  This gives the
following explicit prescription for the term asked for:
\[
 \boxed{\quad
 \mathcal M=
 -\frac{e^2}{m}\,
 \boldsymbol\epsilon'{}^*\!\cdot\boldsymbol\epsilon
 \left.
 \omega\frac{\partial}{\partial\omega}
 \bigl[\epsilon'{}_\mu^*B^{\mu\nu}
 (e,m,\mu,j;\omega q',\omega q)\epsilon_\nu\bigr]
 \right|_{\omega=0}
 +O(\omega^2).
 \quad}
\]
This is often called the Born form of the subleading Low--Gell-Mann--Goldberger
theorem.  It is a calculational formula: the displayed current fixes every
pole residue, while the Ward identities fix the seagull.  Thus the entire
\(O(\omega)\) coefficient depends only on \(e,m,\mu\) and the spin matrices;
charge radii, polarizabilities, and all other target structure first enter
at \(O(\omega^2)\).  For a spinless target the magnetic term is absent.

\WeinbergSolution{4}
Take the decaying fermion at rest, \(p^\mu=(M,\mathbf0)\), and let
\(q^\mu=(\omega,\mathbf q)\).  The soft attachment to the incoming fermion
gives
\[
 \mathcal M_V\simeq g\,\frac{p\cdot\epsilon(q)}{p\cdot q}\,\mathcal M_0.
\]
For the two transverse polarizations \(p\cdot\epsilon_T=0\) at this order.
For the longitudinal polarization,
\[
 \epsilon_L^\mu=\frac1m(|\mathbf q|,\omega\widehat{\mathbf q}),\qquad
 \left|\frac{p\cdot\epsilon_L}{p\cdot q}\right|^2
 =\frac{\mathbf q^2}{m^2\omega^2}.
\]
Neglecting recoil corrections of relative order \(E/W\), factorization of
the remaining decay phase space therefore gives
\[
 \frac{\Gamma_V(E)}{\Gamma_0}
 =\frac{g^2}{4\pi^2m^2}\int_0^{\sqrt{E^2-m^2}}
 \dd q\,\frac{q^4}{(q^2+m^2)^{3/2}} .
\]
In the range specified in the question this is
\[
 \boxed{\quad
 \Gamma_V(E)=\Gamma_0\,\frac{g^2}{8\pi^2}\frac{E^2}{m^2}
 \left[1+O\left(\frac{m^2}{E^2}\ln\frac Em,\frac EW\right)\right].
 \quad}
\]
The power enhancement, rather than a soft logarithm, comes from emission of
the longitudinal mode by a current that is not conserved by the decay
interaction.

\WeinbergSolution{5}
Put \(x_r=p\cdot q_r\), so that \(\sum_rx_r=0\), and use
\[
 \frac1{x-\ii0}=\ii\int_0^\infty \dd t\,e^{-\ii xt}.
\]
For a fixed permutation the \(N-1\) denominators in Eq.~(13.6.8) become
integrals over the successive positive time separations between \(N\)
ordered insertions.  Summing over all permutations partitions the whole
space of relative insertion times: away from coincident times, exactly one
ordering applies.  The common translation drops out because
\(\sum_rx_r=0\).  Fourier transformation over the remaining \(N-1\)
unrestricted relative times then gives
\[
 (2\pi \ii)^{N-1}\delta(x_1)\cdots\delta(x_{N-1}).
\]
Consequently, as a distribution on the hyperplane \(\sum_rx_r=0\),
\[
 \boxed{\quad
 \sum_{\pi\in S_N}\prod_{s=1}^{N-1}
 \frac1{x_{\pi(1)}+\cdots+x_{\pi(s)}-\ii0}
 =(2\pi \ii)^{N-1}\prod_{r=1}^{N-1}\delta(x_r),
 \quad}
\]
which is Eq.~(13.6.8).  Coincident-time boundaries have measure zero and
are already assigned correctly by the common \(i0\) prescription.
\end{enumerate}
