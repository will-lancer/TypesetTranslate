\chapterbackmatter{Supplementary Exercises}

\begin{enumerate}
\SupplementaryExercise{1}{Soft Power Counting in the QED Vertex}
  {(Adapted from Hong Liu and MIT OpenCourseWare, 8.324 Lecture 16)}
  {mit-8324-f10-lecture16-ch13}
In the one-loop on-shell QED vertex, let the photon momentum \(k\) become
soft while every external invariant remains fixed.  Power count the measure,
the photon propagator, and the two neighboring electron propagators.  Show
that the soft region is logarithmically divergent in four dimensions.

\SupplementaryExercise{2}{Why the Inclusive Rate Is Observable}
  {(Adapted from Hong Liu and MIT OpenCourseWare, 8.324 Lecture 16)}
  {mit-8324-f10-lecture16-ch13}
Explain why a detector with resolution \(E\) cannot distinguish a hard
charged final state from that state accompanied by photons of energy below
\(E\).  Write the order-\(e^2\) inclusive probability and identify the
real--virtual cancellation that makes it finite.

\SupplementaryExercise{3}{The Eikonal Factor from an External Fermion}
  {(Adapted from Hong Liu and MIT OpenCourseWare, 8.324 Lecture 16)}
  {mit-8324-f10-lecture16-ch13}
Attach a photon of momentum \(q\) to an incoming or outgoing on-shell
fermion.  Use the Dirac equation to derive the leading factor
\(\eta e\,p\cdot\epsilon/(p\cdot q-i\eta0)\), including the sign \(\eta\).

\SupplementaryExercise{4}{Gauge Terms in the Polarization Sum}
  {(Adapted from Hong Liu and MIT OpenCourseWare, 8.324 Lecture 16)}
  {mit-8324-f10-lecture16-ch13}
Show that replacing the physical photon polarization sum by a covariant
sum changes a soft-emission probability only by terms proportional to
\(\sum_n\eta_ne_n\).  Explain precisely why charge conservation removes
those terms.

\SupplementaryExercise{5}{The Soft Spectrum and Mean Multiplicity}
  {(Adapted from Hong Liu and MIT OpenCourseWare, 8.324 Lecture 16)}
  {mit-8324-f10-lecture16-ch13}
Suppose the angularly integrated one-photon probability is
\(dP=A\,d\omega/\omega\) for \(\lambda<\omega<E\).  Find the mean photon
number and the mean emitted energy.  Explain why only the first quantity
diverges as \(\lambda\to0\).

\SupplementaryExercise{6}{Real--Virtual Regulator Cancellation}
  {(Adapted from Hong Liu and MIT OpenCourseWare, 8.324 Lecture 16)}
  {mit-8324-f10-lecture16-ch13}
If virtual soft photons multiply a rate by
\((\lambda/\Lambda)^A\), sum independent real photons with individual
energies below \(E\), without imposing a total-energy constraint.  Show
that the regulator cancels and determine the remaining power of
\(E/\Lambda\).

\SupplementaryExercise{7}{Isolating the Photon-Mass Log in \(F_1\)}
  {(Adapted from Nicolai Lang, University of Stuttgart, QFT Problem Set 10)}
  {stuttgart-qft-ps10-2023-ch13}
Regulate the on-shell QED vertex by a photon mass \(\lambda\).  After
Feynman parametrization, identify the parameter-space endpoint that
produces the infrared divergence of the Dirac form factor \(F_1(q^2)\), and
extract the coefficient of \(\ln\lambda\) without evaluating finite terms.

\SupplementaryExercise{8}{An Endpoint Change of Variables}
  {(Adapted from Nicolai Lang, University of Stuttgart, QFT Problem Set 10)}
  {stuttgart-qft-ps10-2023-ch13}
For the three-denominator vertex simplex, set
\(x=uv,\ y=u(1-v),\ z=1-u\).  Compute the Jacobian and show how the soft
corner becomes \(u\to1\).  Use the new variables to separate the
logarithmically singular radial integral from a finite angular integral.

\SupplementaryExercise{9}{Positivity of the Vertex Infrared Function}
  {(Adapted from Nicolai Lang, University of Stuttgart, QFT Problem Set 10)}
  {stuttgart-qft-ps10-2023-ch13}
For spacelike transfer write the coefficient of the photon-mass logarithm
as an integral over one Feynman parameter.  Show that it is nonnegative,
vanishes at zero transfer after charge normalization, and grows
logarithmically at large spacelike momentum.

\SupplementaryExercise{10}{Soft Bremsstrahlung in Elastic Scattering}
  {(Adapted from Nicolai Lang, University of Stuttgart, QFT Problem Set 10)}
  {stuttgart-qft-ps10-2023-ch13}
For an electron deflected from momentum \(p\) to \(p'\), derive the
soft-radiation factor
\[
 e^2\int_{\lambda<\omega<E}\frac{d^3q}{(2\pi)^3\,2\omega}
 \left(\frac{p'}{p'\cdot q}-\frac{p}{p\cdot q}\right)^2 .
\]
Perform the energy integral and leave the finite angular coefficient in
Lorentz-invariant form.

\SupplementaryExercise{11}{Cancellation at Order \(\alpha\)}
  {(Adapted from Nicolai Lang, University of Stuttgart, QFT Problem Set 10)}
  {stuttgart-qft-ps10-2023-ch13}
Combine the infrared part of the on-shell vertex correction with the
bremsstrahlung probability of Exercise S.13.10.  Show explicitly that the
coefficient of \(\ln\lambda\) cancels and state which logarithm of the
experimental resolution survives.

\SupplementaryExercise{12}{Ultraviolet and Infrared Regulators Differ}
  {(Adapted from Nicolai Lang, University of Stuttgart, QFT Problem Set 10)}
  {stuttgart-qft-ps10-2023-ch13}
Compare dimensional regularization, a Pauli--Villars subtraction, and a
small photon mass in the electron self-energy.  Identify which endpoint
each regulator controls.  Explain why an ultraviolet counterterm cannot
remove a dependence on experimental soft resolution.

\SupplementaryExercise{13}{Collinear On-Shell Splitting Kinematics}
  {(Adapted from Mark Srednicki, Quantum Field Theory, Chapter 26)}
  {srednicki-qft-infrared-ch13}
In massless six-dimensional \(\phi^3\) theory, let an energetic momentum
\(P\) split into \(p=zP+k_\perp+\cdots\) and
\(q=(1-z)P-k_\perp+\cdots\).  Use the on-shell conditions to find the
virtuality of the parent and identify the collinear singular limit.

\SupplementaryExercise{14}{Collinear Two-Body Phase Space}
  {(Adapted from Mark Srednicki, Quantum Field Theory, Chapter 26)}
  {srednicki-qft-infrared-ch13}
Factorize the phase space for a nearly collinear massless pair in six
space-time dimensions.  Determine the powers of \(k_\perp\) and \(z(1-z)\),
and combine them with the squared splitting propagator to show that the
transverse integral is logarithmic.

\SupplementaryExercise{15}{A Detector-Defined Unresolved Splitting}
  {(Adapted from Mark Srednicki, Quantum Field Theory, Chapter 26)}
  {srednicki-qft-infrared-ch13}
Model a detector that merges two massless particles whenever their invariant
mass is below \(s_{\rm res}\).  Translate this criterion into a bound on
\(k_\perp^2\), and explain why the exclusive one-particle probability is
not infrared safe while the merged probability can be.

\SupplementaryExercise{16}{Extracting the Collinear Logarithm}
  {(Adapted from Mark Srednicki, Quantum Field Theory, Chapter 26)}
  {srednicki-qft-infrared-ch13}
Give the scalar a small mass \(m\) as a collinear regulator.  Evaluate the
singular part of the unresolved splitting integral and show that it has
the form \(C\ln(s_{\rm res}/m^2)\).  Express \(C\) as an integral over the
momentum fraction.

\SupplementaryExercise{17}{Inclusive Cancellation in Six-Dimensional
\(\phi^3\)}
  {(Adapted from Mark Srednicki, Quantum Field Theory, Chapter 26)}
  {srednicki-qft-infrared-ch13}
Show schematically, with the same normalization on both sides of the cut,
that the collinear logarithm from a virtual external-leg correction cancels
the logarithm from unresolved real splitting.  State the completeness sum
that is essential to the cancellation.

\SupplementaryExercise{18}{Jet Safety and the KLN Sum}
  {(Adapted from Mark Srednicki, Quantum Field Theory, Chapter 26)}
  {srednicki-qft-infrared-ch13}
Define a measurement function \(F_n(p_1,\ldots,p_n)\).  Derive the condition
on \(F_{n+1}\) when two momenta become collinear that makes an inclusive
rate collinear safe.  Test the condition for particle multiplicity and for
total energy flow.

\SupplementaryExercise{19}{A Gauge-Invariant Dirac Dressing}
  {(Adapted from Daniel Harlow, Lecture Notes for Physics 8.324)}
  {harlow-qft2-soft-qed-2024-ch13}
Let
\[
 \Psi_D(x)=\exp\left[-ie\int d^3y\,
 \frac{\partial_iA_i(x^0,\mathbf y)}{4\pi|\mathbf x-\mathbf y|}\right]\psi(x).
\]
Using \(\nabla^2(4\pi|\mathbf x-\mathbf y|)^{-1}
=-\delta^3(\mathbf x-\mathbf y)\), prove that \(\Psi_D\) is gauge invariant.

\SupplementaryExercise{20}{The Coulomb Field of a Dressed Electron}
  {(Adapted from Daniel Harlow, Lecture Notes for Physics 8.324)}
  {harlow-qft2-soft-qed-2024-ch13}
Use the canonical commutator between the electric field and vector potential
to compute the electric field created by \(\Psi_D^\dagger(x)\).  Show that
the dressing supplies the Coulomb field required by Gauss's law.

\SupplementaryExercise{21}{Why a Charged Pole Becomes a Branch Point}
  {(Adapted from Daniel Harlow, Lecture Notes for Physics 8.324)}
  {harlow-qft2-soft-qed-2024-ch13}
Assume the charged two-point function in time behaves at long times as
\(e^{-imt}t^{-\gamma}\), with \(\gamma>0\).  Fourier transform its singular
part and show that the isolated pole is replaced by a branch behavior
\((p^0-m+i0)^{\gamma-1}\).  Relate this to the absence of a bare charged
one-particle state.

\SupplementaryExercise{22}{The Logarithmic Soft-Photon Cloud}
  {(Adapted from Daniel Harlow, Lecture Notes for Physics 8.324)}
  {harlow-qft2-soft-qed-2024-ch13}
For a coherent dressing with mode amplitude
\(\alpha_\lambda(\mathbf q)\propto
p\cdot\epsilon_\lambda/(p\cdot q)\), compute the photon number and energy
between \(\lambda\) and \(E\).  Show that the number is logarithmic while
the energy remains finite as \(\lambda\to0\).

\SupplementaryExercise{23}{Dressed States versus Inclusive Rates}
  {(Adapted from Daniel Harlow, Lecture Notes for Physics 8.324)}
  {harlow-qft2-soft-qed-2024-ch13}
Compare two cures for the vanishing bare charged-particle \(S\)-matrix:
coherent asymptotic dressings and a sum over unresolved photons.  Show at
leading eikonal order that both use the same classical current and explain
which experimental question each construction answers.

\SupplementaryExercise{24}{The Soft Region of the One-Loop Vertex}
  {(Adapted from Daniel Harlow, Lecture Notes for Physics 8.324)}
  {harlow-qft2-soft-qed-2024-ch13}
Replace both internal electron propagators in the one-loop vertex by their
eikonal limits.  Derive the scalar soft integral
\[
 -ie^2p\cdot p'\int\frac{d^4k}{(2\pi)^4}
 \frac1{(k^2+\lambda^2-i0)(p\cdot k-i0)(p'\cdot k+i0)}
\]
and explain why its logarithm exponentiates.

\SupplementaryExercise{25}{The Leading Soft-Photon Factor}
  {(Adapted from Andrew Strominger, Lectures on the Infrared Structure of
  Gravity and Gauge Theory)}
  {strominger-infrared-lectures-2017-ch13}
Starting from external-line attachments, prove the leading soft theorem
\[
 \mathcal M_{n+1}(q,\epsilon)=
 \left[\sum_k\eta_k e_k\frac{p_k\cdot\epsilon}{p_k\cdot q}\right]
 \mathcal M_n+O(q^0).
\]
State clearly how incoming and outgoing legs are represented.

\SupplementaryExercise{26}{Charge Conservation from Soft Gauge Invariance}
  {(Adapted from Andrew Strominger, Lectures on the Infrared Structure of
  Gravity and Gauge Theory)}
  {strominger-infrared-lectures-2017-ch13}
Replace \(\epsilon^\mu\) by \(q^\mu\) in the leading soft factor.  Prove
that invariance under this replacement is equivalent to
\(\sum_k\eta_ke_k=0\), and interpret the result as the global Ward identity.

\SupplementaryExercise{27}{Crossing an External Charged Leg}
  {(Adapted from Andrew Strominger, Lectures on the Infrared Structure of
  Gravity and Gauge Theory)}
  {strominger-infrared-lectures-2017-ch13}
Cross an outgoing charged particle to an incoming antiparticle.  Track the
momentum, charge, and \(\eta\) signs and prove that its contribution to the
soft factor is unchanged under the all-outgoing convention.

\SupplementaryExercise{28}{The Large-Gauge Ward Identity}
  {(Adapted from Andrew Strominger, Lectures on the Infrared Structure of
  Gravity and Gauge Theory)}
  {strominger-infrared-lectures-2017-ch13}
Split a large-gauge charge into hard and soft pieces,
\(Q_\varepsilon=Q_\varepsilon^{H}+Q_\varepsilon^{S}\).  Insert
\(\OutBra{\beta}(Q_\varepsilon^+S-SQ_\varepsilon^-)\InKet{\alpha}=0\)
and show how the soft-charge matrix element reproduces the leading
soft-photon theorem.

\SupplementaryExercise{29}{A Frequency Pole and Electromagnetic Memory}
  {(Adapted from Andrew Strominger, Lectures on the Infrared Structure of
  Gravity and Gauge Theory)}
  {strominger-infrared-lectures-2017-ch13}
Show that a radiative field with Fourier behavior
\(\widetilde A(\omega)=C/(\omega+i0)+O(\omega^0)\) has a step between early
and late retarded times.  Relate the step coefficient to the soft factor
and identify the corresponding electromagnetic memory.

\SupplementaryExercise{30}{Angular Distribution of the Soft Cloud}
  {(Adapted from Andrew Strominger, Lectures on the Infrared Structure of
  Gravity and Gauge Theory)}
  {strominger-infrared-lectures-2017-ch13}
For specified massive external momenta and charges, derive
\[
 \frac{dN}{d\omega\,d\Omega}
 =\frac1{2(2\pi)^3\omega}
 \sum_\lambda\left|
 \sum_k\eta_ke_k\frac{p_k\cdot\epsilon_\lambda}{p_k\cdot\widehat q}
 \right|^2 .
\]
Show that it is nonnegative and invariant under
\(\epsilon_\lambda\to\epsilon_\lambda+c_\lambda q\).
\end{enumerate}
