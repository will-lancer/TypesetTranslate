\SupplementaryExercise{1}{The Scalar Feynman Propagator and Oriented Charged Lines}
  {(Adapted from Hong Liu and MIT OpenCourseWare, 8.323 Assignment 4, Problem 1.)}
  {mit-823-s23-ps4-ch6-curated-parent-1}
\begin{enumerate}
\item
For a free real scalar field, prove from translation and Lorentz invariance
that the Feynman function depends only on \((x-y)^2\), apart from the
boundary-value information at timelike separation.  Explain why proper
orthochronous Lorentz transformations preserve that information.
\item
Starting from the four-momentum representation in the mostly-plus
convention, act with \(\Box-m^2\) on \(\Delta_F(x-y)\).  Derive the
delta-function source and state precisely which sign follows from the
chapter's Fourier convention.
\item
Perform the \(q^0\) contour integral in the scalar Feynman propagator.
Show explicitly that positive-frequency propagation occurs for
\(x^0>y^0\) and negative-frequency propagation for \(x^0<y^0\).
\item
For a free complex scalar field, evaluate the two correlators
\[
 \bra{\Omega}T\{\phi(x)\phi^\dagger(y)\}\ket{\Omega},
 \qquad
 \bra{\Omega}T\{\phi(x)\phi(y)\}\ket{\Omega}.
\]
Relate the answer to the orientation assigned to an internal charged-scalar
line.
\end{enumerate}

\SupplementaryExercise{2}{A Linear Source and Functional Differentiation and Why an Adiabatic Source Makes No Particles}
  {(Adapted from Hong Liu and MIT OpenCourseWare, 8.323 Assignment 4, Problem 2.)}
  {mit-823-s23-ps4-ch6-curated-parent-2}
\begin{enumerate}
\item
Couple a free real scalar to a compactly supported source \(J(x)\).
Expand the vacuum functional through second order in \(J\), then show
that two functional derivatives at \(J=0\) recover the time-ordered
two-point function with the correct factors of \(i\).
\item
Show that the one-particle production amplitude from a classical source is
proportional to its Fourier transform on the positive-energy mass shell.
Use this fact to explain quantitatively why an infinitely slow source
turn-on and turn-off produces no massive particles.
\end{enumerate}

In \(\phi^4\) theory, count the contractions that connect four specified
external fields to one interaction vertex.  Show how the count cancels

\SupplementaryExercise{4}{The One-Vertex Tadpole Factor and Cubic Self-Energy Factor}
  {(Inspired by MIT OpenCourseWare, 8.323 Recitation 6, Section 1.2.)}
  {mit-823-s23-rec6-ch6-curated-parent-2}
\begin{enumerate}
\item

\(\lambda\phi^4/4!\) theory, count all Wick contractions and derive the

\item

\SupplementaryExercise{7}{The Free Dirac Propagator, Its Residues, and Fermionic Signs}
  {(Adapted from Hong Liu and MIT OpenCourseWare, 8.323 Assignment 9, Problem 2.)}
  {mit-823-s23-ps9-ch6-curated-parent-1}
\begin{enumerate}
\item
Starting from the free Dirac mode expansion and spin sums, show that the
time-ordered contraction of \(\psi_\ell(x)\) with \(\bar\psi_m(y)\) is a
first-order differential operator acting on the scalar Feynman propagator.
Translate the result to the chapter's gamma-matrix convention.
\item
Apply the free Dirac operator to the propagator obtained in S.6.13.
Derive the spinor identity times \(\delta^4(x-y)\), carefully retaining
the contact term produced when the derivative acts on time ordering.
\item
Find the momentum-space factors for contracting
\(\partial_\mu\psi(x)\) with \(\bar\psi(y)\), and \(\psi(x)\) with
\(\partial_\mu\bar\psi(y)\), when momentum flows from \(y\) to \(x\).
Explain the relative sign.
\item
Factor the Dirac propagator denominator into its two energy poles.
Evaluate the numerator at each pole and show that the residues reproduce
the positive-energy \(u\)-spinor and negative-energy \(v\)-spinor
completeness sums.
\item
Derive the minus sign in the \(y^0>x^0\) branch of
\(T\{\psi(x)\bar\psi(y)\}\).  Show that omitting it prevents the two
energy-pole residues from combining into a Lorentz-covariant propagator.
\item
Follow the index contractions around a closed fermion loop with \(n\)
bosonic insertions.  Demonstrate why the matrices form a trace and why
one additional minus sign remains relative to a bosonic loop.
\end{enumerate}

\SupplementaryExercise{8}{The Scalar Pole Prescription}
  {(Inspired by David Tong, Lectures on Quantum Field Theory, Section 2.7.1.)}
  {tong-qft-feynman-rules-2006-curated-parent-1}
Locate the two scalar propagator poles in the complex energy plane under
the chapter's \(-i0\) denominator convention.  Explain how their
displacement implements vacuum boundary conditions rather than retarded
boundary conditions.

\SupplementaryExercise{9}{Three Channels from One Cubic Interaction and Why a Quartic Vertex Has No Factorial}
  {(Inspired by David Tong, Lectures on Quantum Field Theory, Sections 3.4--3.5 | David Tong, Lectures on Quantum Field Theory, Sections 3.3--3.4.)}
  {tong-qft-feynman-rules-2006-curated-parent-2}
\begin{enumerate}
\item
For identical scalar \(2\to2\) scattering in \(g\phi^3/3!\) theory,
enumerate the inequivalent connected tree graphs and write their
invariant denominators in terms of the mostly-plus variables \(s,t,u\).

\item

\SupplementaryExercise{10}{Reading a Yukawa Fermion Line Backwards and Crossing a Fermion into an Antifermion}
  {(Inspired by David Tong, Lectures on Quantum Field Theory, Sections 3.4 and 5.7 | David Tong, Lectures on Quantum Field Theory, Sections 3.5 and 5.7.)}
  {tong-qft-feynman-rules-2006-curated-parent-3}
\begin{enumerate}
\item
For scalar emission and absorption by a fermion, write the ordered product
of external spinors, coupling matrices, and propagators by following the
fermion arrow backward.  Verify the ordering explicitly with spinor
indices at two vertices.
\item
Begin with a tree-level Yukawa amplitude having an incoming fermion.
Cross that line to an outgoing antifermion by reversing its momentum and
replacing its wavefunction.  Determine the resulting spinor chain and
state which overall signs are conventional.
\end{enumerate}

\SupplementaryExercise{13}{First Coefficients of a Quartic Integral and Automorphisms and Symmetry Factors}
  {(Inspired by Daniel Harlow, Lecture Notes for Physics 8.323, Section 7.3 | Daniel Harlow, Lecture Notes for Physics 8.323, Sections 7.3--7.4.)}
  {harlow-qft1-diagrams-2024-curated-parent-2}
\begin{enumerate}
\item
For the normalized zero-dimensional integral with interaction
\(\lambda x^4/4!\), expand through order \(\lambda^2\).  Compute the
Gaussian moments, divide by the vacuum normalization, and identify the
connected two-point contributions diagrammatically.
\item
Choose a scalar graph with two identical internal lines joining the same
vertices.  Starting from labeled Wick contractions, show that exchanging
those lines is an automorphism and produces a divisor \(2!\) in the
graph's symmetry factor.
\end{enumerate}

\SupplementaryExercise{15}{Amputation of an Off-Shell Scalar Leg and Normalized Gell-Mann--Low Expansion}
  {(Inspired by Daniel Harlow, Lecture Notes for Physics 8.323, Section 7.5 | Daniel Harlow, Lecture Notes for Physics 8.323, Sections 7.4--7.5.)}

least one displayed operator survive the vacuum denominator.
\end{enumerate}
