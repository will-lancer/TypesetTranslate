\SupplementarySolution{1}{One Phase-Space Time Slice and Integrating Out a Quadratic Momentum}
\begin{enumerate}
\item
Let \(\delta t>0\), \(q'=q(t+\delta t)\), and \(q=q(t)\).  To the
accuracy needed in a time-sliced product,
\[
 e^{-iH\delta t}
 =e^{-iP^2\delta t/(2m)}e^{-iV(Q)\delta t}
 +O(\delta t^2).
\]
Inserting \(\mathbf{1}=\int dp\,\ket p\bra p\), with
\(\braket{q}{p}=(2\pi)^{-1/2}e^{ipq}\), gives
\[
 \braket{q'}{e^{-iH\delta t}q}
 =\int\frac{dp}{2\pi}
 \exp\left\{ip(q'-q)-i\delta t
 \left[\frac{p^2}{2m}+V(q)\right]\right\}+O(\delta t^2).
\]
The potential is evaluated at the right endpoint of the ket because of the
displayed operator ordering.  A midpoint prescription instead represents
Weyl ordering; the distinction matters for nontrivial products of \(P\)
and \(Q\), but vanishes here in the continuum limit.
\item
For each slice, complete the square:
\[
 p\frac{\Delta q}{\delta t}-\frac{p^2}{2m}
 =-\frac1{2m}\left(p-m\frac{\Delta q}{\delta t}\right)^2
 +\frac m2\left(\frac{\Delta q}{\delta t}\right)^2.
\]
Thus
\[
 \int\frac{dp}{2\pi}e^{i\delta t[p\Delta q/\delta t-p^2/(2m)]}
 =\left(\frac{m}{2\pi i\delta t}\right)^{1/2}
 e^{im(\Delta q)^2/(2\delta t)}.
\]
Multiplying the slices and taking their continuum limit yields
\[
 K(q_f,t_f;q_i,t_i)
 =\int_{q_i}^{q_f}\mathcal Dq\,
 \exp\left\{i\int_{t_i}^{t_f}dt
 \left[\frac m2\dot q^2-V(q)\right]\right\}.
\]
The coefficient of \(p^2\) is the constant \(1/m\), so the product of
Gaussian determinants depends only on the slicing and total time.  A
coordinate-dependent kinetic coefficient would instead leave a
field-dependent measure factor.
\end{enumerate}

\SupplementarySolution{2}{The Schrödinger Equation from One Step}
Write \(y=x-\xi\) in
\[
 \psi(x,t+\delta t)=
 \left(\frac{m}{2\pi i\delta t}\right)^{1/2}
 \int d\xi\,
 e^{im\xi^2/(2\delta t)}
 e^{-i\delta t V(x-\xi)}\psi(x-\xi,t).
\]
The oscillatory Gaussian localizes \(\xi=O(\sqrt{\delta t})\).  Expanding
the last two factors and using
\[
 \langle\xi\rangle=0,\qquad
 \langle\xi^2\rangle=\frac{i\delta t}{m},
\]
gives
\[
 \psi(x,t+\delta t)=\psi
 +\frac{i\delta t}{2m}\partial_x^2\psi
 -i\delta t\,V(x)\psi+O(\delta t^{3/2}).
\]
After subtracting \(\psi(x,t)\), dividing by \(\delta t\), and multiplying
by \(i\), one obtains
\[
 i\partial_t\psi=
 \left[-\frac1{2m}\partial_x^2+V(x)\right]\psi.
\]
The normalization of the short-time kernel is essential: it makes the
zeroth Gaussian moment equal to one.
Replacing \(V(x-\xi)\) by \(V(x)\) changes the integrand by a term linear in
\(\xi\), whose Gaussian average vanishes, and by terms of order
\(\delta t^2\).  Thus either consistent endpoint convention gives the
displayed equation at this order; differences first matter in the
higher-order definition of the discretized path integral.

\SupplementarySolution{3}{The Free Kernel, Composition, and Dirichlet Fluctuations}
\begin{enumerate}
\item
Successive convolution of the \(N\) identical Gaussian slices gives
\[
 K_0(x_f,T;x_i,0)=
 \left(\frac{m}{2\pi iT}\right)^{1/2}
 \exp\left[\frac{im(x_f-x_i)^2}{2T}\right],
\qquad T>0.
\]
Equivalently, this is the momentum integral
\[
 \int\frac{dp}{2\pi}
 e^{ip(x_f-x_i)-ip^2T/(2m)}.
\]
As \(T\to0^+\), the latter representation converges distributionally to
\(\delta(x_f-x_i)\), fixing the branch of the square root.  In units with
\(\hbar=1\), \(m/T\) has dimension length\(^{-2}\), so the prefactor has
dimension length\(^{-1}\), as required for a one-dimensional position
kernel.
\item
Write
\[
 K_0(x_2,T;x_0,0)=A(T)
 e^{im(x_2-x_0)^2/(2T)}.
\]
Composition at \(T=T_1+T_2\) requires
\[
 A(T)=A(T_2)A(T_1)
 \left(\frac{2\pi iT_1T_2}{mT}\right)^{1/2}.
\]
Hence \(B(T)\equiv A(T)\sqrt T\) is constant, and substitution back into
the same equation gives \(B=\sqrt{m/(2\pi i)}\).  Therefore
\[
 A(T)=\left(\frac{m}{2\pi iT}\right)^{1/2}.
\]
The sign and phase of the root are chosen continuously from \(T=0^+\);
with this choice \(K_0\) approaches the delta distribution rather than its
negative.
\item
Set \(x=x_{\rm cl}+\eta\), where
\(\eta(0)=\eta(T)=0\).  The fluctuation action is
\[
 S_{\rm fl}=\frac m2\int_0^Tdt\,\dot\eta^2
 =\frac12\int_0^Tdt\,\eta(-m\partial_t^2)\eta.
\]
The normalized eigenfunctions and eigenvalues are
\[
 u_n(t)=\sqrt{\frac2T}\sin\frac{n\pi t}{T},
\qquad
 \lambda_n=m\left(\frac{n\pi}{T}\right)^2,
\qquad n=1,2,\ldots.
\]
The Gaussian fluctuation integral is formally
\((\det[-m\partial_t^2])^{-1/2}\).  Comparing determinant ratios at two
times, or using zeta regularization with \(\zeta(0)=-1/2\), leaves the
scaling \(T^{-1/2}\).  The absolute constant is not fixed by the continuum
determinant alone; the short-time delta normalization supplies
\(\sqrt{m/(2\pi i)}\).
\end{enumerate}

\SupplementarySolution{4}{The Cubic Anharmonic Oscillator through Order \(\lambda^2\)}
\begin{enumerate}
\item
After integrating by parts, write
\[
 S_0[x]=-\frac12\int dt\,x(t)Kx(t),
 \qquad K=\partial_t^2+\omega^2.
\]
The boundary value appropriate to the source convention in the problem is
\[
 D(\tau)=\lim_{\epsilon\to0^+}\int\frac{dE}{2\pi}\,
 \frac{i\,e^{-iE\tau}}{\omega^2-E^2+i\epsilon}.
\]
For \(\tau>0\), closing below selects the pole near \(E=-\omega\); for
\(\tau<0\), closing above selects the pole near \(E=+\omega\).  The
residues give
\[
 D(\tau)=\frac{1}{2\omega}e^{i\omega|\tau|}.
\]
Equivalently, this follows from the homogeneous equation away from the
origin and the jump
\[
 \dot D(0^+)-\dot D(0^-)=i,
 \qquad KD(\tau)=i\delta(\tau).
\]
Thus \(K^{-1}=-iD\).  The exponent in the free functional is
\[
 \frac i2\,xKx-iJx
 =\frac i2(x-K^{-1}J)K(x-K^{-1}J)
 -\frac i2\,JK^{-1}J,
\]
where repeated time arguments are integrated.  Translating the Gaussian
measure leaves its source-free determinant unchanged, and therefore
\[
 \boxed{\;
 Z_0[J]=Z_0[0]\exp\left[-\frac12\int dt\,dt'\,
 J(t)D(t-t')J(t')\right].
 \;}
\]
\item
In the chosen convention, insertion of \(x(t)\) is
\(i\delta/\delta J(t)\).  Since
\(-iS_\lambda=+i\lambda\int x^3/3!\), the factors of \(i\) combine to give
\[
\begin{split}
 Z[J]
 &=\exp\left[
 \frac{\lambda}{3!}\int dt\,
 \frac{\delta^3}{\delta J(t)^3}\right]Z_0[J]\\
 &=\sum_{n=0}^{\infty}\frac1{n!}
 \left(\frac{\lambda}{3!}\right)^n
 \prod_{r=1}^{n}\left[
 \int dt_r\,\frac{\delta^3}{\delta J(t_r)^3}\right]Z_0[J].
\end{split}
\]
This is a formal perturbation series, not a convergent definition of the
real cubic oscillator.  For either sign of real \(\lambda\), the cubic
potential is unbounded in one direction, so the real-contour functional
integral itself requires a contour prescription.  Moreover, the number of
Wick contractions grows factorially.  These facts are compatible with each
fixed-order coefficient being finite.
\item
In position space, every internal line contributes \(D(t-t')\), every
cubic vertex contributes \(i\lambda\int dt\), and each graph is divided by
its automorphism factor.  Vacuum components cancel in normalized
correlators, and only graphs connected to both external insertions enter
the connected two-point function.

There is no order-\(\lambda\) term because it would require pairing five
oscillator coordinates.  At order \(\lambda^2\), there are two connected
topologies: the two-line self-energy graph and the graph in which both
external legs meet one vertex while the other vertex carries a tadpole.
Each has symmetry factor \(1/2\).  Define
\[
 G_{\rm conn}(t_1,t_2)
 =\bra{\Omega}T\{x(t_1)x(t_2)\}\ket{\Omega}_{\rm conn},
 \qquad D_{a,b}=D(a-b).
\]
Then
\[
\begin{aligned}
 G_{\rm conn}(t_1,t_2)
 ={}&D_{t_1,t_2}
 +\frac{(i\lambda)^2}{2}\int du\,dv\,
 D_{t_1,u}D_{u,v}^{\,2}D_{v,t_2}\\
 &+\frac{(i\lambda)^2}{2}\int du\,dv\,
 D_{t_1,u}D_{t_2,u}D_{u,v}D_{v,v}
 +O(\lambda^4).
\end{aligned}
\]
The nominal order-\(\lambda^3\) term also vanishes because an even
correlator is even under the simultaneous replacement
\(x\mapsto-x\), \(\lambda\mapsto-\lambda\).

There is no ultraviolet divergence.  In energy space an oscillator
propagator falls as \(E^{-2}\).  The self-energy loop therefore behaves as
\(\int dE\,E^{-4}\), while the tadpole behaves as
\(\int dE\,E^{-2}\); both converge in one integration dimension.  The
\(i\epsilon\) prescription is still needed to define the oscillatory
large-time integrals.  This fixed-order ultraviolet finiteness does not
make the full perturbation series convergent.
\end{enumerate}

\SupplementarySolution{5}{Vacuum Survival under an External Force}
The forced action remains Gaussian.  Completing its functional square gives
\[
 \frac{Z[f]}{Z[0]}
 =\exp\left[-\frac{i}{2}\int dt\,dt'\,
 f(t)G_F(t-t')f(t')\right],
\]
with the convention for \(G_F\) in S.9.7.  Only the delta-function part of
the pole prescription contributes to the modulus.  For
\(\widetilde f(E)=\int dt\,e^{iEt}f(t)\) and real \(f\),
\[
 \boxed{\quad
 P_{0\to0}=\left|\frac{Z[f]}{Z[0]}\right|^2
 =\exp\left[-\frac{|\widetilde f(\omega)|^2}{2\omega}\right].
 \quad}
\]
The exponent is the mean occupation number of the coherent final state.
It is dimensionless, nonnegative, and vanishes for an adiabatic force with
no Fourier support at the oscillator frequency.

\SupplementarySolution{6}{The Free Scalar Source Functional and Wick Pairings by Differentiation}
\begin{enumerate}
\item
After integrating by parts, the free action is quadratic.  In momentum
space it reads
\[
 S_0+S_J=
 -\frac12\int\frac{d^4p}{(2\pi)^4}
 \widetilde\phi(-p)(p^2+m^2-i0)\widetilde\phi(p)
 +\int\frac{d^4p}{(2\pi)^4}\widetilde J(-p)\widetilde\phi(p).
\]
Shift
\(\widetilde\phi(p)\mapsto\widetilde\phi(p)
-(p^2+m^2-i0)^{-1}\widetilde J(p)\).  Translation invariance of the
measure then yields
\[
 \boxed{\quad
 \frac{Z_0[J]}{Z_0[0]}
 =\exp\left[\frac i2\int\frac{d^4p}{(2\pi)^4}
 \frac{\widetilde J(-p)\widetilde J(p)}
 {p^2+m^2-i0}\right].
 \quad}
\]
The positive infinitesimal is fixed by the vacuum boundary functionals,
not by the algebraic completion of the square.
\item
Let
\[
 G_F(x-y)\equiv
 \left.\frac1{i^2Z_0[0]}
 \frac{\delta^2Z_0[J]}{\delta J(x)\delta J(y)}
 \right|_{J=0}.
\]
Because the exponent of \(Z_0[J]\) is quadratic, four derivatives can be
nonzero at \(J=0\) only if they differentiate it in pairs.  Therefore
\[
 \begin{split}
 \langle T\phi_1\phi_2\phi_3\phi_4\rangle_0={}&
 G_{F,12}G_{F,34}+G_{F,13}G_{F,24}
 +G_{F,14}G_{F,23}.
 \end{split}
\]
For \(2n\) fields, every complete pairing occurs once.  The number of
terms is
\[
 (2n-1)!!=\frac{(2n)!}{2^n n!}.
\]
An odd number of derivatives leaves one source unpaired and hence gives
zero at \(J=0\).
\end{enumerate}

\SupplementarySolution{7}{The Propagator Is a Green Function}
\begin{enumerate}
\item
Set \(\tau=x^0-y^0\) and
\[
 C(x,y)=\bra{\Omega}[\phi(x),\phi(y)]\ket{\Omega}.
\]
Both fields obey the homogeneous equation, so
\((\Box_x-m^2)C=0\).  The only possible terms in
\((\Box_x-m^2)[\theta(\tau)C]\) therefore come from time derivatives of
the step function.  At equal time,
\[
 C(x,y)\big|_{\tau=0}=0,\qquad
 \partial_{x^0}C(x,y)\big|_{\tau=0}
 =-i\delta^d(\mathbf x-\mathbf y).
\]
The first derivative of \(\theta C\) has no contact term.  Differentiating
once more and remembering that \(\Box\) contains
\(-\partial_{x^0}^2\) gives
\[
\begin{split}
 (\Box_x-m^2)D_R(x-y)
 &=-\delta(\tau)\,
 \partial_{x^0}C(x,y)\big|_{\tau=0}\\
 &=i\delta^{d+1}(x-y).
\end{split}
\]
Hence
\[
 \boxed{a=i.}
\]
\item
Varying the sourced action gives
\[
 (\Box-m^2)\phi(x)=-j(x).
\]
Since the preceding part established
\((\Box-m^2)D_R=i\delta^{d+1}\), the solution with no response before the
source acts is
\[
 \boxed{\;
 \phi(x)=\phi_0(x)+i\int d^{d+1}y\,D_R(x-y)j(y).
 \;}
\]
Indeed, applying the wave operator to the second term gives
\(i^2j=-j\), and the step function in \(D_R\) enforces the retarded
boundary condition.

To display the sourced modes explicitly, define
\[
 \widetilde j(t,\mathbf p)
 =\int d^d\mathbf y\,e^{-i\mathbf p\cdot\mathbf y}j(t,\mathbf y).
\]
The free expansion gives
\[
 D_R(t,\mathbf x)
 =-i\theta(t)\int\frac{d^d\mathbf p}{(2\pi)^d}\,
 \frac{\sin(E_{\mathbf p}t)}{E_{\mathbf p}}\,
 e^{i\mathbf p\cdot\mathbf x}.
\]
Therefore the complete retarded solution is
\[
\begin{split}
 \phi(x)={}&\phi_0(x)
 +\int\frac{d^d\mathbf p}{(2\pi)^d}\,
 e^{i\mathbf p\cdot\mathbf x}
 \int_{-\infty}^{x^0}dy^0\,
 \frac{\sin[E_{\mathbf p}(x^0-y^0)]}{E_{\mathbf p}}\,
 \widetilde j(y^0,\mathbf p).
\end{split}
\]
For real \(j\), the response is Hermitian as required.
\item
Write \(D=\theta(\tau)W_>+\theta(-\tau)W_<\), where
\[
 W_>(x,y)=\bra{\Omega}\phi(x)\phi(y)\ket{\Omega},
 \qquad
 W_<(x,y)=\bra{\Omega}\phi(y)\phi(x)\ket{\Omega}.
\]
Away from \(\tau=0\), the wave operator annihilates both terms.  The
coefficient generated by the first time derivative is
\(W_>-W_<=\bra{\Omega}[\phi(x),\phi(y)]\ket{\Omega}\), which vanishes at
equal time.  The second derivative leaves
\[
 \delta(\tau)
 \bra{\Omega}[\partial_{x^0}\phi(x),\phi(y)]\ket{\Omega}
 =-i\delta(\tau)\delta^d(\mathbf x-\mathbf y).
\]
The minus sign in
\(\Box=\nabla^2-\partial_{x^0}^2\) reverses this sign.  Thus
\[
 \boxed{\;
 (\Box_x-m^2)D(x-y)=i\delta^{d+1}(x-y).
 \;}
\]
The retarded and time-ordered functions have different boundary
conditions, but they invert the same differential operator with the same
normalization.
\item
The path-integral statement follows from functional integration by parts.
For the free action and the usual vacuum boundary prescription,
\[
\begin{split}
 0
 &=\int\mathcal D\phi\,
 \frac{\delta}{\delta\phi(x)}
 \left[\phi(y)e^{iS_0[\phi]}\right]\\
 &=Z[0]\delta^{d+1}(x-y)
 +iZ[0]\,
 \bra{\Omega}T\!\left\{
 \phi(y)(\Box_x-m^2)\phi(x)\right\}\ket{\Omega}.
\end{split}
\]
Solving this identity reproduces the contact term in the preceding part.
More generally,
\[
\begin{split}
 &\bra{\Omega}T\!\left\{
 (\Box_x-m^2)\phi(x)\prod_{r=1}^{n}\phi(x_r)
 \right\}\ket{\Omega}\\
 &\hspace{2em}
 =i\sum_{r=1}^{n}\delta^{d+1}(x-x_r)
 \bra{\Omega}T\!\left\{
 \prod_{\substack{s=1\\s\ne r}}^{n}\phi(x_s)
 \right\}\ket{\Omega}.
\end{split}
\]
With a source, \((\Box-m^2)\phi\) on the left is replaced by
\((\Box-m^2)\phi+j\).  Thus the field equation holds inside a correlator
except where the varied field meets another insertion.  Those coincident
points are precisely the contact terms.
\end{enumerate}

\SupplementarySolution{8}{The Scalar Vacuum Wave Functional}
For each \(\mathbf p\), the scalar Hamiltonian is an oscillator of frequency
\(E_{\mathbf p}=\sqrt{\mathbf p^2+m^2}\).  The product of their ground
states is
\[
 \Psi_0[\phi]=\mathcal N
 \exp\left[-\frac12\int d^3p\,
 E_{\mathbf p}\widetilde\phi(\mathbf p)
 \widetilde\phi(-\mathbf p)\right].
\]
In position space the kernel is
\[
 \mathcal E(\mathbf x-\mathbf y)=
 \int\frac{d^3p}{(2\pi)^3}
 E_{\mathbf p}e^{i\mathbf p\cdot(\mathbf x-\mathbf y)},
\]
so it is nonlocal even though the Hamiltonian density is local.  Gaussian
integration gives the equal-time covariance
\[
 \langle\widetilde\phi(\mathbf p)
 \widetilde\phi(\mathbf q)\rangle_0
 =\frac{\delta^3(\mathbf p+\mathbf q)}{2E_{\mathbf p}}.
\]
This inverse-frequency factor agrees with the canonical mode expansion.
With a momentum cutoff and finite spatial volume,
\(\mathcal N\) is the product
\(\prod_{\mathbf p}(E_{\mathbf p}/\pi)^{1/4}\), with the usual adjustment
for the reality relation between \(\mathbf p\) and \(-\mathbf p\).  This
regulator-dependent constant cancels between normalized matrix elements,
whereas the kernel \(E_{\mathbf p}\) and its inverse covariance remain.

\SupplementarySolution{9}{Vacuum Boundaries and the Normalized Ratio and No-Production Probability from a Source}
\begin{enumerate}
\item
On a finite interval the path integral contains the boundary factors
\(\Psi_0[\phi_i]\Psi_0[\phi_f]^*\).  Thus
\[
 \langle T\phi(x_1)\cdots\phi(x_n)\rangle
 =\lim_{T\to\infty}
 \frac{\int_{\!-T}^{T}\mathcal D\phi\,
 \Psi_0[\phi_f]^*\phi(x_1)\cdots\phi(x_n)
 e^{iS[\phi]}\Psi_0[\phi_i]}
 {\int_{\!-T}^{T}\mathcal D\phi\,
 \Psi_0[\phi_f]^*e^{iS[\phi]}\Psi_0[\phi_i]}.
\]
Equivalently, adiabatically replace \(H\) by \((1-i\epsilon)H\).  Excited
components at the two ends are exponentially damped, leaving the vacuum.
The denominator removes the vacuum-to-vacuum amplitude, including its
phase and disconnected vacuum bubbles, so the normalized expectation of
\(\mathbf{1}\) is exactly one.
\item
The normalized vacuum amplitude is
\[
 \OutBra{\Omega}\InKet{\Omega;J}
 =\exp\left[\frac i2\int\frac{d^4p}{(2\pi)^4}
 \frac{\widetilde J(-p)\widetilde J(p)}
 {p^2+m^2-i0}\right].
\]
Using
\((u-i0)^{-1}=\operatorname{PV}(1/u)+i\pi\delta(u)\), the principal-value
part is a phase.  Hence
\[
 \boxed{\quad
 P_0=
 \left|\OutBra{\Omega}\InKet{\Omega;J}\right|^2
 =\exp\left[-\int\frac{d^3p}{(2\pi)^3\,2E_{\mathbf p}}
 |\widetilde J(E_{\mathbf p},\mathbf p)|^2\right].
 \quad}
\]
The exponent is minus the mean number of produced quanta.  Only the
on-shell source appears, so a sufficiently slow source produces no
particles.
\end{enumerate}

\SupplementarySolution{10}{Low-Order Correlators in \(\lambda\phi^4\)}
\begin{enumerate}
\item
At first order,
\[
 \langle T\phi(x)\phi(y)\rangle
 =\frac{\langle T\phi(x)\phi(y)
 [1-i\lambda\int d^4z\,\phi(z)^4/4!]\rangle_0}
 {\langle T[1-i\lambda\int d^4z\,\phi(z)^4/4!]\rangle_0}.
\]
For the connected tadpole, choose the vertex field joined to \(x\) in
\(4\) ways and that joined to \(y\) in \(3\) ways; the remaining two pair.
Thus \(12/4!=1/2\), and
\[
 G_F(x-y)-\frac{i\lambda}{2}\int d^4z\,
 G_F(x-z)G_F(0)G_F(z-y)+O(\lambda^2).
\]
The other numerator contraction pairs \(x\) with \(y\) and contracts the
four vertex fields among themselves in three ways.  It is exactly the
free two-point function times the first-order denominator and therefore
cancels.

At second order the three non-vacuum topologies are
\[
\begin{split}
 &\frac{(-i\lambda)^2}{6}\int dz\,dw\,
 G(x-z)G(z-w)^3G(w-y),\\
 &\frac{(-i\lambda)^2}{4}\int dz\,dw\,
 G(x-z)G(z-w)G(w-y)G(0)^2,\\
 &\frac{(-i\lambda)^2}{4}\int dz\,dw\,
 G(x-z)G(y-z)G(z-w)^2G(0),
\end{split}
\]
with distinct external attachments included.  The coefficients follow
by pairing the half-edges at the two labeled vertices; every omitted
pairing has a vacuum component and cancels against \(Z[0]\).
\item
The first connected contribution is
\[
 -\frac{i\lambda}{4!}\int d^4z\,
 \langle T\phi(x_1)\phi(x_2)\phi(x_3)\phi(x_4)\phi(z)^4\rangle_0.
\]
Connectivity at tree level requires each external field to contract with
a different field at \(z\).  There are \(4!\) bijections, canceling the
factor in the interaction.  Consequently
\[
 \boxed{\quad
 G^{(4)}_{\rm conn}(x_1,\ldots,x_4)
 =-i\lambda\int d^4z\,
 \prod_{a=1}^{4}G_F(x_a-z)+O(\lambda^2).
 \quad}
\]
Contractions that pair external fields among themselves are disconnected;
pure vacuum factors cancel against \(Z[0]\).

At order \(\lambda^2\), the three partitions of the external labels give
the \(s,t,u\) fish graphs, each with factor \(1/2\):
\[
 \frac{(-i\lambda)^2}{2}\int dz\,dw\,
 G(x_a-z)G(x_b-z)G(z-w)^2G(x_c-w)G(x_d-w).
\]
The other connected class consists of four external-leg tadpole
dressings: three external legs meet one vertex, one line joins the
second, and that vertex carries the remaining external leg and a
tadpole.  Its factor is \(1/2\).  The vertex-degree constraints show
that these classes exhaust order \(\lambda^2\).
\end{enumerate}

\SupplementarySolution{11}{Vacuum Energy and Connected Vacuum Graphs}
\begin{enumerate}
\item
Translation invariance leaves one unconstrained vertex integral in each
vacuum graph.  Consequently
\[
 Z[0]=e^{iW_0},\qquad
 W_0=-\mathcal V_4\mathcal E_{\rm vac},\qquad
 \mathcal V_4=\int d^4x.
\]
\item
Let \(C_r\) denote the complete weight of one connected vacuum-graph type
\(r\), including its coupling, integrals, and inverse symmetry factor.
A vacuum diagram containing \(n_r\) indistinguishable copies of each type
has weight
\(\prod_r C_r^{n_r}/n_r!\).  Summing over all multiplicities gives
\[
 Z[0]=\prod_r\sum_{n_r=0}^{\infty}\frac{C_r^{n_r}}{n_r!}
 =\prod_r e^{C_r}
 =\exp\left(\sum_r C_r\right).
\]
Therefore
\[
 \log Z[0]=\sum_r C_r.
\]
The factorials are precisely those obtained by permuting identical
connected components.  Thus the logarithm retains each connected vacuum
graph once, with the symmetry factor already contained in \(C_r\).
\item
For \(\lambda\phi^4/4!\), the one-vertex figure-eight has factor \(1/8\).
At second order, the connected four-line basketball has factor \(1/48\),
and the topology with two joining lines and one tadpole at each vertex
has factor \(1/16\).  Thus
\[
\begin{split}
 iW_0={}&\frac{-i\lambda}{8}\int dx\,G(0)^2\\
 &+(-i\lambda)^2\int dx\,dy\left[
 \frac{G(x-y)^4}{48}
 +\frac{G(x-y)^2G(0)^2}{16}\right]+O(\lambda^3).
\end{split}
\]
Dividing by \(\mathcal V_4\) and using part 1 yields the vacuum-energy
shift, with the overall sign fixed by the propagator convention.
\end{enumerate}

\SupplementarySolution{12}{Cancellation of Arbitrary Vacuum Bubbles}
Every diagram contributing to the unnormalized insertion functional can be
decomposed uniquely into a part \(D\) containing all external insertions
and a multiset \(V\) of vacuum components.  The sum over \(V\) is independent
of the external points and equals the vacuum functional:
\[
 \widetilde G_n(x_1,\ldots,x_n)
 =Z[0]\,G_n^{\rm no\ vac}(x_1,\ldots,x_n).
\]
Dividing by \(Z[0]\) therefore gives
\[
 G_n=\frac{\widetilde G_n}{Z[0]}
 =G_n^{\rm no\ vac}.
\]
This argument uses only component factorization and the exponential
counting of indistinguishable vacuum graphs, not the form or order of the
polynomial interaction.  It consequently holds to every perturbative
order.

\SupplementarySolution{13}{A Wave Functional as a Euclidean Boundary and A Parity-Twisted Euclidean Trace}
\begin{enumerate}
\item
Insert a field basis at the initial boundary:
\[
 \braket{\phi_f}{e^{-TH}\psi_i}
 =\int\mathcal D\phi_i\,K_E[\phi_f,T;\phi_i,0]\Psi_i[\phi_i]
 =\int_{\phi(T)=\phi_f}\mathcal D\phi\,
 e^{-S_E[\phi]}\Psi_i[\phi(0)].
\]
The spectral expansion is
\[
 \sum_n e^{-E_nT}\Psi_n[\phi_f]
 \braket{n}{\psi_i}.
\]
If the initial state has nonzero vacuum overlap, then as \(T\to\infty\)
\[
 e^{E_0T}\braket{\phi_f}{e^{-TH}\psi_i}
 \longrightarrow
 \Psi_0[\phi_f]\braket{\Omega}{\psi_i}.
\]
Thus a semi-infinite Euclidean path integral prepares the ground-state wave
functional, while changing its far-boundary weight changes only the
overlap constant.
\item
Since \(P\ket x=\ket{-x}\),
\[
 \operatorname{Tr}(Pe^{-TH})
 =\int dx\,\braket{x}{Pe^{-TH}x}
 =\int dx\,\braket{-x}{e^{-TH}x}.
\]
The corresponding Euclidean paths obey the twisted boundary condition
\[
 x(T)=-x(0).
\]
For the oscillator, parity of the \(n\)th energy eigenstate is
\((-1)^n\), so
\[
 \operatorname{Tr}(Pe^{-TH})
 =\sum_{n=0}^{\infty}(-1)^n
 e^{-\omega T(n+1/2)}
 =\frac1{2\cosh(\omega T/2)}.
\]
Integrating the quoted oscillator heat kernel with endpoints \(x\) and
\(-x\) gives the same Gaussian result, verifying the twisted path-integral
boundary condition.
\end{enumerate}

\SupplementarySolution{14}{Pfaffian Covariance and Canonical Form}
For \(\theta=N\theta'\), the top Grassmann monomial transforms as
\(\theta_1\cdots\theta_{2N}=\det(N)\theta'_1\cdots\theta'_{2N}\).
Berezin integration must therefore transform inversely:
\[
 d^{2N}\theta=\det(N)^{-1}d^{2N}\theta'.
\]
Now evaluate
\[
 \int d^{2N}\theta\,
 e^{\frac12\theta^{\mathsf T}A\theta}
 =\operatorname{Pf}(A)
\]
after the change of variables.  The exponent contains
\(N^{\mathsf T}AN\), while the measure contributes the inverse determinant,
so
\[
 \operatorname{Pf}(A)
 =\det(N)^{-1}\operatorname{Pf}(N^{\mathsf T}AN).
\]
Rearrangement proves
\[
 \boxed{\operatorname{Pf}(N^{\mathsf T}AN)
 =\det(N)\operatorname{Pf}(A)}.
\]
For one pair,
\(A=\left(\begin{smallmatrix}0&a\\-a&0\end{smallmatrix}\right)\), so
\(\operatorname{Pf}(A)=a\).  The formula then follows immediately for a
diagonal rescaling and fixes the sign on the component continuously
connected to the identity; polynomiality extends it to every invertible
\(N\).

There is a congruence transformation for which
\[
 N^{\mathsf T}AN=\bigoplus_{r=1}^N
 \begin{pmatrix}0&a_r\\-a_r&0\end{pmatrix}.
\]
The Grassmann Gaussian factorizes, so this canonical matrix has
Pfaffian \(\prod_ra_r\) and determinant \(\prod_ra_r^2\).
Using congruence covariance on both quantities proves
\[
 \operatorname{Pf}(A)^2=\det A,\qquad
 \operatorname{Pf}(A)=\pm\sqrt{\det A}.
\]
Continuity from an oriented canonical basis fixes the sign.

\SupplementarySolution{15}{An Exact Four-Grassmann Interaction}
Choose \(d^4\theta=d\theta_4d\theta_3d\theta_2d\theta_1\),
\(\int d^4\theta\,\theta_1\theta_2\theta_3\theta_4=1\), and write
\[
 S=\frac12A_{ab}\theta_a\theta_b
 +\lambda\theta_1\theta_2\theta_3\theta_4.
\]
Only the top monomial contributes.  Since the quadratic term squared
supplies \(\operatorname{Pf}(A)\theta_1\theta_2\theta_3\theta_4\),
\[
 Z=\int d^4\theta\,e^{-S}
 =\operatorname{Pf}(A)-\lambda.
\]
No higher term exists because every Grassmann variable squares to zero.
In diagram language, \(\operatorname{Pf}(A)\) is the sum over the three
signed Gaussian pairings, while the single quartic vacuum vertex has
weight \(-\lambda\).  There are no multi-vertex graphs, so the diagrammatic
enumeration is already exact.
Thus \(Z\to\operatorname{Pf}(A)\) when \(\lambda\to0\), while for \(A=0\)
only the saturated quartic monomial survives and \(Z=-\lambda\), confirming
both limiting cases with the chosen measure orientation.

\SupplementarySolution{16}{Bosonic and Fermionic Determinant Powers and Fermionic Wick's Theorem from a Pfaffian}
\begin{enumerate}
\item
For a positive real symmetric \(K\),
\[
 \int d^Nx\,e^{-\frac12x^{\mathsf T}Kx}
 =(2\pi)^{N/2}(\det K)^{-1/2}.
\]
For independent complex Grassmann pairs and an arbitrary invertible \(A\),
\[
 \int d^N\bar\theta\,d^N\theta\,
 e^{-\bar\theta A\theta}=\det A.
\]
Finally, for \(2N\) real Grassmann variables and antisymmetric \(B\),
\[
 \int d^{2N}\theta\,
 e^{\frac12\theta^{\mathsf T}B\theta}
 =\operatorname{Pf}(B),\qquad
 \operatorname{Pf}(B)^2=\det B.
\]
Commuting Gaussians integrate inverse widths, whereas Berezin integration
extracts the highest polynomial coefficient; this reverses the determinant
power.  Antisymmetry of \(B\) is required because the symmetric part
multiplies an antisymmetric variable product and vanishes.
\item
For antisymmetric invertible \(A\), introduce real Grassmann sources:
\[
 Z[\eta]=\int d^{2N}\theta\,
 e^{\frac12\theta^{\mathsf T}A\theta+\eta^{\mathsf T}\theta}
 =\operatorname{Pf}(A)
 e^{\frac12\eta^{\mathsf T}A^{-1}\eta},
\]
with the source and differential orders held fixed.  Four left
derivatives give
\[
 \langle\theta_a\theta_b\theta_c\theta_d\rangle
 =(A^{-1})_{ab}(A^{-1})_{cd}
 -(A^{-1})_{ac}(A^{-1})_{bd}
 +(A^{-1})_{ad}(A^{-1})_{bc}.
\]
The middle pairing is odd because forming it requires one interchange of
Grassmann objects.  With \(2N\) insertions, differentiation produces the
Pfaffian of the restricted covariance matrix, equivalently the signed sum
over all complete pairings.
\end{enumerate}

\SupplementarySolution{17}{Berezin Gaussians, Fermionic Sources, and Determinant Signs}
\begin{enumerate}
\item
Fix
\[
 \int d\bar\theta\,d\theta\,\theta\bar\theta=1.
\]
Because \((\bar\theta\theta)^2=0\),
\[
 e^{-\bar\theta a\theta}
 =1-\bar\theta a\theta
 =1+a\theta\bar\theta,
\]
and hence
\[
 \int d\bar\theta\,d\theta\,
 e^{-\bar\theta a\theta}=a.
\]
Under \(\theta\to\theta+\xi\) and
\(\bar\theta\to\bar\theta+\bar\xi\), all new terms have degree lower than
two in the integration variables or differ by a Berezin total derivative.
Their integrals vanish.  Thus the integral is translation invariant even
though Berezin integration is algebraic differentiation rather than an
ordinary limiting sum.
\item
Change variables so that \(A\) is triangular; Berezin measures transform
with the inverse Jacobian, so the result is basis independent.  For a
diagonal kernel,
\[
 \prod_{i=1}^N\int d\bar\theta_i\,d\theta_i\,
 e^{-a_i\bar\theta_i\theta_i}
 =\prod_i a_i=\det A.
\]
Therefore, generally,
\[
 \boxed{\quad
 \int\prod_i(d\bar\theta_i\,d\theta_i)\,
 e^{-\bar\theta_iA_{ij}\theta_j}=\det A.
 \quad}
\]
The exponential terminates, and the integral selects the coefficient
containing every \(\theta_i\) and \(\bar\theta_i\).  That coefficient is the
antisymmetrized product defining \(\det A\), rather than its reciprocal.
\item
Differentiate the identity \(Z(A)=\det A\):
\[
 \frac{\partial Z}{\partial A_{lk}}
 =\det A\,(A^{-1})_{kl}.
\]
On the integral side,
\[
 \frac{\partial Z}{\partial A_{lk}}
 =\int d\bar\theta\,d\theta\,
 (-\bar\theta_l\theta_k)e^{-\bar\theta A\theta}
 =\int d\bar\theta\,d\theta\,
 \theta_k\bar\theta_l e^{-\bar\theta A\theta}.
\]
Consequently
\[
 \boxed{\quad
 \int d\bar\theta\,d\theta\,
 \theta_k\bar\theta_l e^{-\bar\theta A\theta}
 =\det A\,(A^{-1})_{kl}.
 \quad}
\]
The transpose-looking order is fixed by differentiating \(A_{lk}\), not
by a convention imposed afterward.
\item
Introduce sources and complete the square:
\[
 \frac{Z[\bar\eta,\eta]}{Z[0]}
 =\exp(\bar\eta A^{-1}\eta).
\]
Taking ordered source derivatives corresponding to
\(\theta_i\bar\theta_j\theta_k\bar\theta_l\) yields
\[
 \boxed{\quad
 \langle\theta_i\bar\theta_j\theta_k\bar\theta_l\rangle
 =(A^{-1})_{ij}(A^{-1})_{kl}
 -(A^{-1})_{il}(A^{-1})_{kj}.
 \quad}
\]
The second contraction crosses the middle pair.  Moving the associated
Grassmann derivatives next to each other requires one odd interchange,
which supplies the minus sign.  This is the finite-dimensional origin of
the sign rule for interchanging fermion lines.
\item
Let
\[
 \mathcal D(p)=i\gamma^\mu p_\mu+m-i0
\]
and use the source convention
\[
 -i\bar\psi\mathcal D\psi-i\bar\psi\eta-i\bar\eta\psi
\]
in the exponent.  The shifts
\(\psi\to\psi+\mathcal D^{-1}\eta\) and
\(\bar\psi\to\bar\psi+\bar\eta\mathcal D^{-1}\) give
\[
 \frac{Z[\bar\eta,\eta]}{Z[0]}
 =\exp\left(i\int\frac{d^4p}{(2\pi)^4}
 \bar\eta(-p)\mathcal D^{-1}(p)\eta(p)\right).
\]
Since
\[
 \mathcal D^{-1}(p)=
 \frac{-i\gamma\cdot p+m}
 {p^2+m^2-i0},
\]
ordered source differentiation reproduces
\[
 \langle T\psi(p)\bar\psi(-p)\rangle
 =-i\,\frac{-i\gamma\cdot p+m}
 {p^2+m^2-i0}
\]
for the displayed exponent convention.  The pole is fixed by the
fermionic vacuum wave functional just as in the scalar case.
\item
For a background bilinear, the fermionic integral is
\[
 Z[\Gamma]\propto\det(\mathcal D+\Gamma)
 =\det\mathcal D\,
 \exp\left[\operatorname{Tr}\log
 (1+\mathcal D^{-1}\Gamma)\right].
\]
Expanding,
\[
 \operatorname{Tr}\log(1+\mathcal D^{-1}\Gamma)
 =\sum_{n=1}^{\infty}\frac{(-1)^{n+1}}n
 \operatorname{Tr}(\mathcal D^{-1}\Gamma)^n.
\]
The trace closes the chain of \(n\) propagators and vertices into one loop;
\(1/n\) removes cyclic choices of its starting vertex.  Relative to the
bosonic expansion, the alternating determinant series differs by one
overall minus sign for every connected closed chain.  Equivalently, closing
the chain requires an odd cyclic reordering of Grassmann fields.  Thus each
closed fermion loop carries the standard extra factor \(-1\).
\end{enumerate}

\SupplementarySolution{18}{The Complete Gaussian Generating Function}
\begin{enumerate}
\item
First set \(j=0\) and write \(I(a)=\int_{\mathbb R}dx\,e^{-ax^2/2}\).
For \(a>0\),
\[
 I(a)^2=\int_{\mathbb R^2}dx\,dy\,e^{-a(x^2+y^2)/2}
 =2\pi\int_0^\infty r\,dr\,e^{-ar^2/2}
 =\frac{2\pi}{a}.
\]
Positivity fixes \(I(a)=\sqrt{2\pi/a}\).  Completing the square,
\[
 -\frac a2x^2+jx
 =-\frac a2\left(x-\frac ja\right)^2+\frac{j^2}{2a},
\]
and translation of the integration variable gives
\[
 \boxed{Z_a(j)=\sqrt{\frac{2\pi}{a}}\,
 e^{j^2/(2a)}}.
\]
For a real contour, absolute convergence requires \(a>0\).
\item
Choose an orthogonal matrix \(O\) with
\(O^{\mathsf T}AO=\operatorname{diag}(a_1,\ldots,a_N)\), where every
\(a_i>0\).  With \(x=Oy\) and \(J'=O^{\mathsf T}J\), the absolute
Jacobian is \(|\det O|=1\).  The integral factors into the result of part
(a):
\[
\begin{aligned}
 Z_A(J)
 &=\prod_{i=1}^N
 \sqrt{\frac{2\pi}{a_i}}\,
 \exp\left(\frac{J_i'^2}{2a_i}\right)\\
 &=\boxed{\frac{(2\pi)^{N/2}}{\sqrt{\det A}}\,
 \exp\left(\frac12J^{\mathsf T}A^{-1}J\right)}.
\end{aligned}
\]
\item
Normalized moments follow from source differentiation:
\[
 \langle x_i x_j\rangle
 =\left.\frac{1}{Z_A(0)}
 \frac{\partial^2Z_A}{\partial J_i\partial J_j}\right|_{J=0}
 =(A^{-1})_{ij}.
\]
Because the mean vanishes, the second derivative of \(\log Z_A\) gives the
same answer.  In general, derivatives of \(\log Z\) generate connected
cumulants, while derivatives of \(Z/Z[0]\) generate full moments.  Replacing
indices by space-time points and \(A\) by a quadratic kernel gives
\[
 \frac{Z[J]}{Z[0]}
 =\exp\left(\frac12\int d^dx\,d^dy\,
 J(x)A^{-1}(x,y)J(y)\right),
\]
with the appropriate factors of \(i\) after continuation to the
real-time convention.
\end{enumerate}

\SupplementarySolution{19}{The Characteristic Function of a Gaussian Quantum Coordinate}
In the position representation, the normalized vacuum probability density
is a centered Gaussian,
\[
 |\Psi_0(q)|^2=
 \frac{1}{\sqrt{2\pi\sigma^2}}
 e^{-q^2/(2\sigma^2)},
\qquad
 \sigma^2=\bra{\Omega}q^2\ket{\Omega}.
\]
The source formula of S.9.18 with \(j=iK\) gives
\[
 \boxed{\bra{\Omega}e^{iKq}\ket{\Omega}
 =e^{-K^2\sigma^2/2}}.
\]
For an operator derivation, write
\(q=c(a+a^\dagger)\), where
\([a,a^\dagger]=1\) and \(c^2=\sigma^2\).  Since
\([iKca^\dagger,iKca]=K^2c^2\) is central, the
Baker--Campbell--Hausdorff formula yields
\[
 e^{iKq}
 =e^{-K^2c^2/2}e^{iKca^\dagger}e^{iKca}.
\]
The rightmost exponential acts trivially on \(\ket{\Omega}\), and every
nonconstant term in the left exponential is killed by
\(\bra{\Omega}\).  This reproduces the Gaussian answer and fixes the
coefficient without choosing a position basis.

If the mean is \(\bar q\), write \(q=\bar q+\delta q\) with
\(\langle\delta q\rangle=0\).  The characteristic function becomes
\[
 \boxed{\langle e^{iKq}\rangle
 =e^{iK\bar q}
 \exp\left[-\frac{K^2}{2}
 \langle(q-\bar q)^2\rangle\right]}.
\]
\end{enumerate}
