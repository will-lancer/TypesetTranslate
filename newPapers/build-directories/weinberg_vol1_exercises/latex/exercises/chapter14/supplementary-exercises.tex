\SupplementaryExercise{1}{From the Dirac Equation to the Pauli Hamiltonian}
  {(Inspired by Michael Dine, Physics 217 Homework 1 Solutions.)}
  {dine-ph217-background-field-2013-ch14-curated-parent-1}
\begin{enumerate}
\item
For a charge \(q\) in a static electromagnetic background, write a positive
energy Dirac spinor as \(e^{-i(m+\varepsilon)t}(\varphi,\chi)^T\).
Derive the coupled equations for \(\varphi\) and \(\chi\), solve for \(\chi\)
through relative order \(1/m\), and obtain the leading nonrelativistic
Hamiltonian for \(\varphi\).
\item
Use
\((\boldsymbol\sigma\cdot\boldsymbol\pi)^2
=\boldsymbol\pi^2-q\boldsymbol\sigma\cdot\mathbf B\), where
\(\boldsymbol\pi=\mathbf p-q\mathbf A\), to reduce the result of Exercise
S.14.1.  Identify the orbital and spin couplings to a weak magnetic field
and check gauge covariance.
\item
Compare the spin term in the Pauli Hamiltonian with
\(-\boldsymbol\mu\cdot\mathbf B\), taking
\(\mathbf S=\boldsymbol\sigma/2\).  Determine the tree-level gyromagnetic
ratio and explain which coefficient is changed by the Pauli form factor
\(F_2(0)\).
\end{enumerate}

\SupplementaryExercise{2}{Darwin, Spin--Orbit, and Coulombic Power Counting}
  {(Inspired by Michael Dine, Physics 217 Homework 1 Solutions.)}
  {dine-ph217-background-field-2013-ch14-curated-parent-2}
\begin{enumerate}
\item
Continue the Foldy--Wouthuysen reduction in a static electric field through
order \(m^{-2}\).  Isolate the spin-independent term proportional to
\(\boldsymbol\nabla\cdot\mathbf E\), and evaluate its first-order energy
shift for a Coulomb field in a hydrogenic \(n\ell m\) state.
\item
Extract the spin--orbit operator at order \(m^{-2}\) for a central
electrostatic potential \(V(r)=qA^0(r)\).  Express it in terms of
\(\mathbf L\cdot\mathbf S\), and show explicitly where the factor associated
with Thomas precession appears.
\item
For a hydrogenic state with principal quantum number of order unity, assign
orders in \(Z\alpha\) to \(r^{-1}\), \(\mathbf p\), the binding energy,
\(\mathbf E\), the Darwin operator, and the spin--orbit operator.  Use these
assignments to predict the order of the fine-structure energy shift.
\end{enumerate}

\SupplementaryExercise{3}{Feynman--Hellmann Relations for Hydrogenic Matrix Elements}
  {(Inspired by MIT OpenCourseWare 8.06, Problem Set 2.)}
  {mit-806-s18-ps2-hydrogen-ch14-curated-parent-1}
\begin{enumerate}
\item
Let \(H(\lambda)\) have a normalized, nondegenerate eigenstate
\(\lvert n(\lambda)\rangle\).  Derive the Feynman--Hellmann theorem by
differentiating the eigenvalue equation.  State what must be done inside a
degenerate eigenspace.
\item
Apply Exercise S.14.7 to
\(H(\kappa)=\mathbf p^2/(2m)-\kappa/r\).  Starting only from the exact
Coulomb energy \(E_n=-m\kappa^2/(2n^2)\), find
\(\langle n\ell m|r^{-1}|n\ell m\rangle\) and check the virial theorem.
\item
Add \(\lambda/r^2\) to the Coulomb Hamiltonian and absorb it into an
effective angular momentum.  Differentiate the resulting energy at
\(\lambda=0\) to obtain \(\langle r^{-2}\rangle_{n\ell}\), including its
dependence on \(\ell\).
\end{enumerate}

\SupplementaryExercise{4}{The Stark Effect and Polarizability}
  {(Inspired by MIT OpenCourseWare 8.06, Problem Set 2.)}
  {mit-806-s18-ps2-hydrogen-ch14-curated-parent-2}
\begin{enumerate}
\item
Place hydrogen in a uniform static electric field
\(\boldsymbol{\mathcal E}=\mathcal E\widehat{\mathbf z}\), so that
\(H'=e\mathcal E z\).  Prove that the \(1s\) energy has no term linear in
\(\mathcal E\), using both parity and the angular integral.
\item
Write the second-order \(1s\) Stark shift as
\(-\alpha_E\mathcal E^2/2\).  Use completeness and the fact that every
dipole-accessible excitation energy is at least \(E_2-E_1\) to derive an
upper bound on \(\alpha_E\).  Evaluate the needed expectation
\(\langle1s|z^2|1s\rangle\).
\item
Ignoring fine structure and the Lamb shift, diagonalize \(e\mathcal E z\)
within the four orbital states of the \(n=2\) hydrogen manifold.  Identify
the states that mix, compute the nonzero matrix element, and give all
first-order shifts.
\end{enumerate}

\SupplementaryExercise{5}{Einstein Coefficients and Electric-Dipole Radiation}
  {(Inspired by David Tong, Topics in Quantum Mechanics, Chapter 4.)}
  {tong-topicsqm-atomic-radiation-ch14-curated-parent-1}
\begin{enumerate}
\item
For two atomic levels in a thermal photon bath, include absorption,
stimulated emission, and spontaneous emission.  Combine detailed balance
with the Planck distribution to derive the relation between the Einstein
\(A\) and \(B\) coefficients, including level degeneracies.
\item
Starting from \(H_I=-q\,\mathbf x\cdot\mathbf E(0)\) in the dipole
approximation, apply Fermi's golden rule, sum the two photon polarizations,
and integrate over directions.  Derive the total E1 rate in natural units.
\item
Treat the three Cartesian components of \(\mathbf x\) as a parity-odd
rank-one tensor.  Use parity and angular-momentum addition to derive the E1
selection rule for orbital angular momentum and explain why
\(\ell=0\to\ell'=0\) is forbidden.
\end{enumerate}

\SupplementaryExercise{6}{Polarization Rules, the \(2p\) Lifetime, and Stark Quenching}
  {(Inspired by David Tong, Topics in Quantum Mechanics, Chapter 4.)}
  {tong-topicsqm-atomic-radiation-ch14-curated-parent-2}
\begin{enumerate}
\item
Rewrite \(\mathbf x\cdot\boldsymbol\epsilon\) in spherical components
\(x_q\) with \(q=0,\pm1\).  Use the Wigner--Eckart theorem to obtain the
magnetic-quantum-number rule and associate \(\pi\) and \(\sigma^\pm\)
polarizations with the allowed changes.
\item
Evaluate the radial and angular matrix elements for \(2p\to1s+\gamma\).
Insert them in the E1 formula, simplify the answer to a rational multiple of
\(\alpha^5m_e\), and estimate the lifetime.
\item
A weak dc electric field mixes \(2s\) with \(2p,m=0\).  Including their
physical separation \(\Delta_{\rm LS}=E_{2s}-E_{2p}\), find the admixture
coefficient and the induced one-photon width of \(2s\).  State the
perturbative condition on the field.
\end{enumerate}

\SupplementaryExercise{7}{Atomic Scales and the Uehling Contact Interaction}
  {(Inspired by Charles Thorn, QFT Problem Set 13 Solutions.)}
  {thorn-qft-ps13-lamb-2019-ch14-curated-parent-1}
\begin{enumerate}
\item
Use the Coulomb virial theorem to identify the momentum, radius, and binding
energy scales of a hydrogenic atom.  Then show why a one-loop vacuum
polarization produced by an electron of mass \(m_e\) can be expanded at
small momentum transfer when \(Z\alpha\ll1\).
\item
Starting from the on-shell-subtracted one-loop representation
\[
 \Pi_R(-\mathbf q^2)=\frac{2\alpha}{\pi}\int_0^1dx\,x(1-x)
 \ln\!\left[1+\frac{x(1-x)\mathbf q^2}{m_e^2}\right],
\]
derive the leading term for \(|\mathbf q|\ll m_e\).
\item
Insert the result of part (b) into the Coulomb kernel
\(-4\pi Z\alpha/\mathbf q^2\).  Fourier transform the leading correction
and identify the corresponding local operator in the atomic Hamiltonian.
\end{enumerate}

\SupplementaryExercise{8}{Vacuum-Polarization Shifts of Hydrogenic Levels}
  {(Inspired by Charles Thorn, QFT Problem Set 13 Solutions.)}
  {thorn-qft-ps13-lamb-2019-ch14-curated-parent-2}
\begin{enumerate}
\item
Evaluate the first-order shift produced by \(C\delta^3(\mathbf r)\) in a
hydrogenic \(n\ell m\) state.  Prove that it vanishes for \(\ell>0\), and
derive \(|\psi_{ns}(0)|^2\) including its \(n\) dependence.
\item
Combine the low-momentum polarization and contact interaction from
S.14.7(b)--(c) with S.14.8(a) to calculate the leading electronic
vacuum-polarization shift of a hydrogenic \(ns\) level.  Determine its
dependence on \(Z\), \(\alpha\), \(m_e\), and \(n\), and compare its
parametric order with the electron self-energy Lamb shift.
\item
Specialize part (b) to hydrogen \(2s\).  Determine the sign from
charge screening, estimate the answer in MHz, and explain why this
contribution is much smaller than the observed \(2s_{1/2}-2p_{1/2}\)
splitting.
\end{enumerate}

\SupplementaryExercise{9}{Bound-State Poles, Self-Energies, and Feshbach Projection}
  {(Inspired by V. Antonelli et al., Effective Lagrangians in Bound State Calculations.)}
  {antonelli-et-al-bound-state-eft-2000-ch14-curated-parent-1}
\begin{enumerate}
\item
For a charged field in a time-independent external potential, Fourier
transform its two-point function only in the time difference.  Insert a
complete set of exact energy eigenstates and show that a stable bound state
produces a simple pole.  Determine the factorized residue.
\item
Expand the Dyson equation to first order in a proper self-energy
\(\Sigma(E)\) near a nondegenerate bound-state pole.  Show that the
coefficient of the double pole determines the energy shift and give the
corresponding expression for a degenerate multiplet.
\item
Let \(P\) project onto one unperturbed bound state and \(Q=1-P\).  Starting
from \(G(E)=(E-H_0-V)^{-1}\), eliminate the \(Q\) sector and derive the exact
energy-dependent effective Hamiltonian in the \(P\) sector.  Expand it
through third order in \(V\).
\end{enumerate}

\SupplementaryExercise{10}{Bound-State EFT Power Counting and Matching}
  {(Inspired by V. Antonelli et al., Effective Lagrangians in Bound State Calculations.)}
  {antonelli-et-al-bound-state-eft-2000-ch14-curated-parent-2}
\begin{enumerate}
\item
For a heavy particle in \(V_C=-\lambda/r\), rescale
\(\mathbf r=\boldsymbol\rho/(M\lambda)\).  Derive the scaling of the
normalized wave function, a Coulomb insertion, a reduced resolvent, and a
local operator \(C_{2n}\nabla^{2n}\delta^3(\mathbf r)\).  Use the result to
organize bound-state perturbation theory.
\item
Suppose a relativistic amplitude for scattering from a weak static source
has the low-momentum form
\(\mathcal M(\mathbf p',\mathbf p)=c_0+c_2(\mathbf p'-\mathbf p)^2/M^2+\cdots\).
Construct the corresponding nonrelativistic potential through \(M^{-2}\),
including the normalization conversion, and state how its coefficients are
fixed.
\item
Consider two effective Hamiltonians related by
\(H'\!=e^{iS}He^{-iS}\), with \(S\) local and perturbatively small.  Show
that their potentials differ by terms proportional to the leading equation
of motion and that off-shell Green functions change.  Prove nevertheless
that the positions of their bound-state poles agree order by order when all
induced operators are retained.
\end{enumerate}
\end{enumerate}
