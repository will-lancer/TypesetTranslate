\SupplementaryExercise{1}{From the Dirac Equation to the Pauli Hamiltonian}
  {(Inspired by Michael Dine, Physics 217 Homework 1 Solutions.)}
  {dine-ph217-background-field-2013-ch14-curated-parent-1}
\begin{enumerate}
\item
For a charge \(q\) in a static electromagnetic background, write a positive
energy Dirac spinor as \(e^{-\ii(m+\varepsilon)t}(\varphi,\chi)^{\mathsf T}\).
Derive the coupled equations for \(\varphi\) and \(\chi\), solve for \(\chi\)
through relative order \(1/m\), and obtain the leading nonrelativistic
Hamiltonian for \(\varphi\).
\item
Use
\((\boldsymbol\sigma\cdot\boldsymbol\pi)^2
=\boldsymbol\pi^2-q\boldsymbol\sigma\cdot\mathbf B\), where
\(\boldsymbol\pi=\mathbf p-q\mathbf A\), to reduce the result of Exercise
S.14.1.  Identify the orbital and spin couplings to a weak magnetic field
and check gauge covariance.
\item
Compare the spin term in the Pauli Hamiltonian with
\(-\boldsymbol\mu\cdot\mathbf B\), taking
\(\mathbf S=\boldsymbol\sigma/2\).  Determine the tree-level gyromagnetic
ratio and explain which coefficient is changed by the Pauli form factor
\(F_2(0)\).
\end{enumerate}

\SupplementaryExercise{2}{Darwin, Spin--Orbit, and Coulombic Power Counting}
  {(Inspired by Michael Dine, Physics 217 Homework 1 Solutions.)}
  {dine-ph217-background-field-2013-ch14-curated-parent-2}
\begin{enumerate}
\item
Continue the Foldy--Wouthuysen reduction in a static electric field through
order \(m^{-2}\).  Isolate the spin-independent term proportional to
\(\boldsymbol\nabla\cdot\mathbf E\), and evaluate its first-order energy
shift for a Coulomb field in a hydrogenic \(n\ell m\) state.
\item
Extract the spin--orbit operator at order \(m^{-2}\) for a central
electrostatic potential \(V(r)=qA^0(r)\).  Express it in terms of
\(\mathbf L\cdot\mathbf S\), and show explicitly where the factor associated
with Thomas precession appears.
\item
For a hydrogenic state with principal quantum number of order unity, assign
orders in \(Z\alpha\) to \(r^{-1}\), \(\mathbf p\), the binding energy,
\(\mathbf E\), the Darwin operator, and the spin--orbit operator.  Use these
assignments to predict the order of the fine-structure energy shift.
\end{enumerate}

\SupplementaryExercise{3}{Feynman--Hellmann Relations for Hydrogenic Matrix Elements}
  {(Inspired by MIT OpenCourseWare 8.06, Problem Set 2.)}
  {mit-806-s18-ps2-hydrogen-ch14-curated-parent-1}
\begin{enumerate}
\item
Let \(H(\lambda)\) have a normalized, nondegenerate eigenstate
\(\ket{n(\lambda)}\).  Derive the Feynman--Hellmann theorem by
differentiating the eigenvalue equation.  State what must be done inside a
degenerate eigenspace.
\item
Apply Exercise S.14.7 to
\(H(\kappa)=\mathbf p^2/(2m)-\kappa/r\).  Starting only from the exact
Coulomb energy \(E_n=-m\kappa^2/(2n^2)\), find
\(\langle n\ell m|r^{-1}|n\ell m\rangle\) and check the virial theorem.
\item
Add \(\lambda/r^2\) to the Coulomb Hamiltonian and absorb it into an
effective angular momentum.  Differentiate the resulting energy at
\(\lambda=0\) to obtain \(\langle r^{-2}\rangle_{n\ell}\), including its
dependence on \(\ell\).
\end{enumerate}

\SupplementaryExercise{4}{The Stark Effect and Polarizability}
  {(Inspired by MIT OpenCourseWare 8.06, Problem Set 2.)}
  {mit-806-s18-ps2-hydrogen-ch14-curated-parent-2}
\begin{enumerate}
\item
Place hydrogen in a uniform static electric field
\(\boldsymbol{\mathcal E}=\mathcal E\widehat{\mathbf z}\), so that
\(H'=e\mathcal E z\).  Prove that the \(1s\) energy has no term linear in
\(\mathcal E\), using both parity and the angular integral.
\item
Write the second-order \(1s\) Stark shift as
\(-\alpha_E\mathcal E^2/2\).  Use completeness and the fact that every
dipole-accessible excitation energy is at least \(E_2-E_1\) to derive an
upper bound on \(\alpha_E\).  Evaluate the needed expectation
\(\langle1s|z^2|1s\rangle\).
\item
Ignoring fine structure and the Lamb shift, diagonalize \(e\mathcal E z\)
within the four orbital states of the \(n=2\) hydrogen manifold.  Identify
the states that mix, compute the nonzero matrix element, and give all
first-order shifts.
\end{enumerate}

\SupplementaryExercise{5}{Einstein Coefficients and Electric-Dipole Radiation}
  {(Inspired by David Tong, Topics in Quantum Mechanics, Chapter 4.)}
  {tong-topicsqm-atomic-radiation-ch14-curated-parent-1}
\begin{enumerate}
\item
For two atomic levels in a thermal photon bath, include absorption,
stimulated emission, and spontaneous emission.  Combine detailed balance
with the Planck distribution to derive the relation between the Einstein
\(A\) and \(B\) coefficients, including level degeneracies.
\item
Starting from \(H_I=-q\,\mathbf x\cdot\mathbf E(0)\) in the dipole
approximation, apply Fermi's golden rule, sum the two photon polarizations,
and integrate over directions.  Derive the total E1 rate in natural units.
\item
Treat the three Cartesian components of \(\mathbf x\) as a parity-odd
rank-one tensor.  Use parity and angular-momentum addition to derive the E1
selection rule for orbital angular momentum and explain why
\(\ell=0\to\ell'=0\) is forbidden.
\end{enumerate}

\SupplementaryExercise{6}{Polarization Rules, the \(2p\) Lifetime, and Stark Quenching}
  {(Inspired by David Tong, Topics in Quantum Mechanics, Chapter 4.)}

\item

\item

\SupplementaryExercise{7}{Vacuum-Polarization Contribution to the Lamb Shift}
  {(Adapted from Charles B. Thorn, Quantum Field Theory Problem Set 13, Problem 48.)}
  {thorn-qft-ps13-2019-p48}
After charge renormalization, the one-loop relation between the electric
displacement and electric field, through first order in \(e^2\), is
\[
 \widetilde{\mathbf E}(k)=
 \left[
 1+\frac{e^2}{2\pi^2}\int_0^1\dd x\,x(1-x)
 \ln\!\left(1+x(1-x)\frac{\mathbf k^2}{m_e^2}\right)
 \right]\widetilde{\mathbf D}(k).
\]
Consequently the potential energy of the electron in hydrogen is
modified from its lowest-order value \(-e^2/(4\pi r)\).  Calculate this
modification in lowest-order nonrelativistic perturbation theory.
\begin{enumerate}
\item
Explain why the momentum transfers obey
\(\mathbf k^2/m_e^2\ll1\) and
\((k^0)^2\ll\mathbf k^2\).
\item
Use those estimates to expand the logarithm and evaluate the parameter
integral.  Show that the additional potential has the form
\[
 \delta V(\mathbf r)=-V_1\delta^3(\mathbf r)
\]
and determine \(V_1\).
\item
Explain why this perturbation affects only \(s\)-levels
\((\ell=0)\).
\item
Calculate the energy shift of the hydrogen \(2s\) state in
electron-volts.  Natural units have been used above; restore
\(\hbar\) and \(c\) at a convenient late stage to check the dimensions.
\end{enumerate}

\SupplementaryExercise{8}{The Magnetic Moment of a Dirac Fermion}
  {(Adapted from J. McGreevy, Physics 215A Quantum Field Theory, Fall 2023, Assignment 10, Problem 5.)}
  {mcgreevy-215a-f23-hw10-p5}
Consider the interaction Hamiltonian density
\[
 \mathcal H_I=-\ii q\,\bar\Psi\gamma^\mu\Psi A_\mu,
\]
where the vector potential \(A_\mu\) is a fixed classical
electromagnetic background.  The factor \(-\ii\) translates the source
interaction to the volume's gamma-matrix convention.  Define
one-particle states by
\[
 \bra{0}\Psi(x)\ket{\mathbf p,s}
 =e^{-\ii p\cdot x}u_s(p).
\]
The goal is to show that, in the rest frame, the one-particle
Hamiltonian contains a magnetic-dipole term proportional to
\(\mathbf S\cdot\mathbf B\), where
\(\mathbf S=\boldsymbol\sigma/2\).
\begin{enumerate}
\item
The charge \(q\) is real.  What condition on \(A_\mu\) is required for
\[
 H_I=\int \dd^3x\,\mathcal H_I
\]
to be Hermitian?
\item
As a bonus, determine how \(A_\mu\) must transform under parity \(P\)
and charge conjugation \(C\) for \(H_I\) to be invariant.  How do
\(\mathbf E\) and \(\mathbf B\) transform?  Explain why these symmetries
allow a magnetic dipole moment but forbid an electric dipole moment.
\item
Let
\[
 \gamma^{\mu\nu}\equiv\frac12[\gamma^\mu,\gamma^\nu].
\]
In the nonrelativistic limit, show that
\[
 \bar u(p')\gamma^{\mu\nu}u(p)F_{\mu\nu}
 =a\,\xi'^\dagger\boldsymbol\sigma\cdot\mathbf B\,\xi
\]
and determine \(a\).  In a Weyl basis compatible with the volume
conventions, use
\[
 u(p)\longrightarrow\sqrt{m_\Psi}\binom{\xi}{\xi},
 \qquad
 u(p')\longrightarrow\sqrt{m_\Psi}\binom{\xi'}{\xi'}.
\]
\item
Take \(A_\mu\) to describe a magnetic field \(\mathbf B\) that is
uniform in space and time.  Show in the nonrelativistic limit that
\[
 \bra{\mathbf p',s'}H_I\ket{\mathbf p,s}
 =(2\pi)^3\delta^3(\mathbf p-\mathbf p')\,
 h(\xi',\xi,\mathbf B)
 +\text{terms independent of spin}.
\]
Find \(h\), using the Gordon identity if helpful.  Convert to
one-particle states with nonrelativistic normalization and interpret
the result as a gyromagnetic ratio \(g_{\rm mag}=2\).
\item
Optionally determine how the answer changes after adding
\[
 \Delta\mathcal H
 =\frac{c}{M}\bar\Psi F_{\mu\nu}
 [\gamma^\mu,\gamma^\nu]\Psi.
\]
Include the condition on \(c\) required for this Hamiltonian term to be
Hermitian in the stated gamma-matrix convention.
\end{enumerate}

\SupplementaryExercise{9}{Bubble-Chain Bound States in \(2+1\) Dimensions}
  {(Adapted from J. McGreevy, Physics 215B Quantum Field Theory, Winter 2025, Assignment 8, Problem 1.)}
  {mcgreevy-215b-w25-hw08-p1}
Besides a one-particle delta function at \(s=m^2\) and a continuum
above \(s=(2m)^2\), the spectral density of an operator that creates a
massive particle can contain bound-state delta functions with
\(m^2<s<(2m)^2\).  To see how such a pole can arise, work in
\(D=2+1\) dimensions with the mostly-plus form of the source action
\[
 S[\phi]=\int \dd^3x\left[
 -\frac12\partial_\mu\phi\,\partial^\mu\phi
 -\frac12m^2\phi^2-\frac{g}{4!}\phi^4\right],
\qquad m,g\in\mathbb R.
\]
Consider both signs of \(g\), setting aside vacuum stability; as the
source notes, a small stabilizing \(\phi^6\) interaction may be imagined
without affecting this calculation.
\begin{enumerate}
\item
For elastic \(\phi\phi\to\phi\phi\) scattering, let
\(s=E_T^2\), where \(E_T\) is the total center-of-mass energy.  Compute
\(\mathcal M(s)\) in the bubble-chain approximation defined by
\[
\begin{gathered}
\begin{tikzpicture}[baseline=-0.55ex,scale=0.62]
 \coordinate (v) at (0,0);
 \fill (v) circle (2pt);
 \draw (-0.8,0.55)--(v)--(0.8,0.55);
 \draw (-0.8,-0.55)--(v)--(0.8,-0.55);
\end{tikzpicture}
\quad+\quad
\begin{tikzpicture}[baseline=-0.55ex,scale=0.62]
 \coordinate (a) at (-0.48,0);
 \coordinate (b) at (0.48,0);
 \fill (a) circle (2pt); \fill (b) circle (2pt);
 \draw (-1.15,0.55)--(a); \draw (-1.15,-0.55)--(a);
 \draw (b)--(1.15,0.55); \draw (b)--(1.15,-0.55);
 \draw (a) to[bend left=48] (b);
 \draw (a) to[bend right=48] (b);
\end{tikzpicture}
\quad+\quad
\begin{tikzpicture}[baseline=-0.55ex,scale=0.62]
 \coordinate (a) at (-0.78,0);
 \coordinate (b) at (0,0);
 \coordinate (c) at (0.78,0);
 \fill (a) circle (2pt); \fill (b) circle (2pt); \fill (c) circle (2pt);
 \draw (-1.42,0.55)--(a); \draw (-1.42,-0.55)--(a);
 \draw (c)--(1.42,0.55); \draw (c)--(1.42,-0.55);
 \draw (a) to[bend left=45] (b);
 \draw (a) to[bend right=45] (b);
 \draw (b) to[bend left=45] (c);
 \draw (b) to[bend right=45] (c);
\end{tikzpicture}
\quad+\ \cdots .
\end{gathered}
\]
These are respectively the contact term, the one-bubble term, the
two-bubble term, and their continuation.  Do not justify the
approximation or the convergence of the series.  The loop may be left
as a Feynman-parameter integral.
\item
Verify the optical theorem explicitly within this approximation.  In
the center-of-mass frame in three spacetime dimensions,
\[
 \frac{\dd\sigma}{\dd\theta}
 =\frac{|\mathcal M|^2}{32\pi pE_T^2},
\]
where \(p\) is the magnitude of either spatial momentum.
\item
Look for a real pole at \(s=M_B^2\) below threshold,
\[
 0<M_B^2<4m^2.
\]
Determine the signs and range of \(g\) for which the bubble-chain
approximation predicts such a bound state.  An analytic formula for
\(M_B\) is not required.
\end{enumerate}

\SupplementaryExercise{10}{Nonrelativistic Forces from QED and Pseudoscalar Exchange}
  {(Adapted from J. McGreevy, Physics 215A Quantum Field Theory, Fall 2023, Assignment 10, Problem 6.)}
  {mcgreevy-215a-f23-hw10-p6}
\begin{enumerate}
\item
Starting from QED, derive the repulsive \(1/r^2\) force between two
nonrelativistic electrons.
\item
Consider a massive Dirac field \(\Psi\) and a massive pseudoscalar
\(\phi\), interacting through
\[
 V_5=g_5\bar\Psi\gamma_5\Psi\,\phi.
\]
\begin{enumerate}
\item Assign parity transformations to both fields and prove that the
interaction is parity invariant.
\item List all propagator and vertex rules needed at leading order.
\item Draw and evaluate every leading-order diagram for
\(\Psi\Psi\to\Psi\Psi\), including the exchange required for identical
fermions.
\item Take \(m_\Psi\gg|\mathbf p|\) and derive the effective
nonrelativistic interaction Hamiltonian.  If the naive leading term
vanishes, continue to the first nonzero term, as emphasized in the
source.
\end{enumerate}
\end{enumerate}
