\chapterbackmatter{Weinberg Exercises}

\begin{enumerate}
\WeinbergExercise{1} Consider a charged scalar particle of mass $m\ne0$, described by a
field $\phi(x)$ whose only interaction is with an external time-independent
electromagnetic field $\mathcal{A}^{\mu}(x)$. Let $\ket{N}$ be a complete set
of normalized one-boson or one-antiboson states with energies $E_N$, and
define
\[
 u_N(\mathbf{x})e^{-\ii E_Nt}
   =\bra{\Omega}\phi(\mathbf{x},t)\ket{N},
 \qquad
 v_N(\mathbf{x})e^{\ii E_Nt}
   =\bra{N}\phi(\mathbf{x},t)\ket{\Omega},
\]
where $\ket{\Omega}$ is the vacuum. Show that the $u_N$ and $v_N$ together
form a complete set, and give formulas for the coefficients of $u_N$ and
$v_N$ in the expansion of a general function $f(\mathbf{x})$.

\WeinbergExercise{2} Suppose we include radiative corrections in the theory of Problem 1.
Let $\ii\Pi^*(x,y)$ be the sum of all diagrams with one incoming and one
outgoing charged scalar line (excluding final scalar propagators) to order
$\alpha$. Derive a formula for the shift in the energies $E_N$ of the
one-boson states due to these radiative corrections, in terms of
$u_N(\mathbf{x})$ and $\Pi^*(x,y)$.

\WeinbergExercise{3} Use the results of Section 14.3 to calculate the radiative decay rate
of the $2p$ states of hydrogen.

\WeinbergExercise{4} Suppose that the electron has an interaction with a light scalar field
$\phi$, of the form $g\phi\bar\psi_e\psi_e$. Suppose that the scalar mass
$m_\phi$ is in the range
$(Z\alpha)^2m_e\ll m_\phi\ll Z\alpha m_e$. Calculate the change in the
energy of the $1s$ state of hydrogenic atoms due to this interaction.

\WeinbergExercise{5} Carry out the calculation of Problem 4 for $m_\phi=0$.
\end{enumerate}
