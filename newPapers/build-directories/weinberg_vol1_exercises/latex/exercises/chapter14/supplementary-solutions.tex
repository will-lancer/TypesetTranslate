\SupplementarySolution{1}{From the Dirac Equation to the Pauli Hamiltonian}
\begin{enumerate}
\item
With \(\boldsymbol\pi=\mathbf p-q\mathbf A\), the two-component equations are
\[
 (\varepsilon-qA^0)\varphi=\boldsymbol\sigma\cdot\boldsymbol\pi\,\chi,
 \qquad
 (2m+\varepsilon-qA^0)\chi
 =\boldsymbol\sigma\cdot\boldsymbol\pi\,\varphi .
\]
Thus
\[
 \chi=\frac{\boldsymbol\sigma\cdot\boldsymbol\pi}{2m}\varphi
 +O(m^{-2}),
\]
and the upper component obeys
\[
 \ii\partial_t\varphi=
 \left[m+qA^0+
 \frac{(\boldsymbol\sigma\cdot\boldsymbol\pi)^2}{2m}\right]\varphi
 +O(m^{-2}).
\]
The rest energy may be dropped.  The small component is suppressed by the
particle velocity, as required for a consistent nonrelativistic expansion.
\item
The covariant momenta satisfy
\([\pi_i,\pi_j]=\ii q\epsilon_{ijk}B_k\).  Hence
\[
 (\boldsymbol\sigma\cdot\boldsymbol\pi)^2
 =\boldsymbol\pi^2-q\boldsymbol\sigma\cdot\mathbf B
\]
and
\[
 \boxed{H_{\rm P}=qA^0+\frac{\boldsymbol\pi^2}{2m}
 -\frac{q}{2m}\boldsymbol\sigma\cdot\mathbf B .}
\]
For a spatially constant field, the term linear in \(\mathbf A\) inside
\(\boldsymbol\pi^2\) is the orbital magnetic coupling, while the last term
is the spin coupling.  Both \(\boldsymbol\pi\) and the wave function
transform covariantly, so the result does not depend on the gauge used for
\(\mathbf A\).
\item
Writing the spin term as
\[
 -\frac{q}{m}\mathbf S\cdot\mathbf B
 =-\boldsymbol\mu\cdot\mathbf B,\qquad
 \boldsymbol\mu=g_{\rm mag}\frac{q}{2m}\mathbf S,
\]
gives
\[
 \boxed{g_{\rm mag}=2}
\]
at tree level.  For an on-shell electromagnetic vertex written with Dirac
and Pauli form factors, charge normalization gives \(F_1(0)=1\), whereas
\[
 g_{\rm mag}=2\,[F_1(0)+2mF_2(0)]
\]
in the dimensionful convention of Chapter 14.  Thus the radiative
anomalous moment is carried by \(F_2(0)\), not by a change of the electric
charge.
\end{enumerate}

\SupplementarySolution{2}{Darwin, Spin--Orbit, and Coulombic Power Counting}
\begin{enumerate}
\item
The spin-independent order-\(m^{-2}\) term in the
Foldy--Wouthuysen Hamiltonian is
\[
 H_D=-\frac{q}{8m^2}\boldsymbol\nabla\cdot\mathbf E.
\]
For an electron in the field of a point nucleus,
\(q\boldsymbol\nabla\cdot\mathbf E=-4\pi Z\alpha\delta^3(\mathbf r)\).
Therefore
\[
 H_D=\frac{\pi Z\alpha}{2m^2}\delta^3(\mathbf r),\qquad
 \delta E_D=\frac{\pi Z\alpha}{2m^2}|\psi_{n\ell m}(0)|^2.
\]
Using
\(|\psi_{n\ell m}(0)|^2=(Z\alpha m)^3
\delta_{\ell0}/(\pi n^3)\), one obtains
\[
 \boxed{\delta E_D=\frac{(Z\alpha)^4m}{2n^3}\delta_{\ell0}.}
\]
The contact form makes the restriction to \(s\) waves manifest.
\item
The electric spin term is
\[
 H_{\rm SO}=-\frac{q}{4m^2}
 \boldsymbol\sigma\cdot(\mathbf E\times\mathbf p).
\]
Since \(V=qA^0\), \(\mathbf E=-\boldsymbol\nabla A^0\), and
\(\boldsymbol\nabla V=(\dd V/\dd r)\widehat{\mathbf r}\),
\[
 \boxed{H_{\rm SO}=
 \frac{1}{2m^2r}\frac{\dd V}{\dd r}\,\mathbf L\cdot\mathbf S .}
\]
A naive rest-frame magnetic-field argument would give twice this
coefficient.  The factor \(1/2\) is the Thomas factor; in the Dirac
reduction it is already included by the consistent elimination of the
small component.
\item
The virial theorem gives
\[
 |\mathbf p|\sim Z\alpha m,\qquad
 r^{-1}\sim Z\alpha m,\qquad
 E_{\rm bind}\sim (Z\alpha)^2m .
\]
For a Coulomb field, \(q|\mathbf E|\sim Z\alpha/r^2\), so
\[
 q|\mathbf E|\sim(Z\alpha)^3m^2,\qquad
 \frac{q\,\boldsymbol\nabla\cdot\mathbf E}{m^2}
 \sim\frac{q\,\mathbf E\cdot\mathbf p}{m^2}
 \sim(Z\alpha)^4m .
\]
Both Darwin and spin--orbit effects therefore begin at
\(\boxed{m(Z\alpha)^4}\), two powers of \(Z\alpha\) below the leading
binding energy.  The same counting applies to the relativistic kinetic
correction \(-\mathbf p^4/(8m^3)\).
\end{enumerate}

\SupplementarySolution{3}{Feynman--Hellmann Relations for Hydrogenic Matrix Elements}
\begin{enumerate}
\item
Differentiate
\(H(\lambda)\ket{n}=E_n\ket{n}\), multiply on the left by
\(\bra{n}\), and use normalization.  The terms containing
\(\partial_\lambda\ket{n}\) cancel, leaving
\[
 \boxed{\frac{\dd E_n}{\dd\lambda}
 =\left\langle n\left|\frac{\partial H}{\partial\lambda}\right|n\right\rangle.}
\]
For a degenerate level, one must first diagonalize
\(P(\partial_\lambda H)P\) in the degenerate subspace \(P\).  Its
eigenvalues are the possible first derivatives of the split levels.
\item
Here
\(\partial H/\partial\kappa=-1/r\), while
\[
 \frac{\partial E_n}{\partial\kappa}
 =-\frac{m\kappa}{n^2}.
\]
Feynman--Hellmann therefore yields
\[
 \boxed{\left\langle\frac1r\right\rangle_{n\ell}
 =\frac{m\kappa}{n^2}.}
\]
It follows that
\(\langle V\rangle=-m\kappa^2/n^2=2E_n\) and
\(\langle T\rangle=-E_n\), which is exactly the Coulomb virial theorem.
\item
The centrifugal coefficient becomes
\[
 \frac{\ell(\ell+1)}{2m}+\lambda
 =\frac{\ell_\lambda(\ell_\lambda+1)}{2m},
\quad
 \left.\frac{\dd\ell_\lambda}{\dd\lambda}\right|_0
 =\frac{m}{\ell+1/2}.
\]
With \(n=n_r+\ell+1\),
\[
 E(\lambda)=-\frac{m\kappa^2}
 {2[n_r+\ell_\lambda+1]^2}.
\]
Differentiating at zero auxiliary coupling gives
\[
 \boxed{\left\langle\frac1{r^2}\right\rangle_{n\ell}
 =\frac{m^2\kappa^2}{n^3(\ell+1/2)}.}
\]
The stronger \(\ell\) dependence reflects the centrifugal suppression near
the origin.
\end{enumerate}

\SupplementarySolution{4}{The Stark Effect and Polarizability}
\begin{enumerate}
\item
The first-order shift is
\[
 \delta E_{1s}^{(1)}=e\mathcal E\langle1s|z|1s\rangle.
\]
The \(1s\) wave function is even under parity whereas \(z\) is odd, so the
integrand is odd and the matrix element vanishes.  Directly,
\(\int \dd\Omega\,\cos\theta=0\).  Hence
\[
 \boxed{\delta E_{1s}^{(1)}=0.}
\]
The leading Stark effect is quadratic because the perturbation admixes odd
\(p\) states into the even ground state.
\item
Second-order perturbation theory gives
\[
 \alpha_E=2e^2\sum_{n\ne1}
 \frac{|\langle n|z|1s\rangle|^2}{E_n-E_1}.
\]
All nonzero dipole matrix elements have
\(E_n-E_1\geq E_2-E_1=3m\alpha^2/8\), including the continuum.  Completeness
and \(\langle z\rangle=0\) therefore imply
\[
 \alpha_E\leq
 \frac{2e^2\langle1s|z^2|1s\rangle}{E_2-E_1}.
\]
For \(\psi_{1s}\propto e^{-r/a_0}\),
\(\langle r^2\rangle=3a_0^2\), so \(\langle z^2\rangle=a_0^2\).  Thus
\[
 \boxed{\alpha_E\leq\frac{16e^2a_0^2}{3m\alpha^2}
 =\frac{64\pi}{3}a_0^3}
\]
in Heaviside--Lorentz units.
\item
Parity makes every diagonal element vanish.  The \(m=\pm1\) states cannot
couple to \(2s\), and the only nonzero orbital matrix element is
\[
 \langle2s|z|2p,m=0\rangle=-3a_0
\]
up to a phase convention.  The \(2s,2p_0\) block is therefore
\[
 \begin{pmatrix}0&-3ea_0\mathcal E\\
 -3ea_0\mathcal E&0\end{pmatrix}.
\]
Its eigenvectors are the equal even--odd combinations, with shifts
\[
 \boxed{\delta E^{(1)}=\pm3ea_0\mathcal E,}
\]
while \(2p,m=\pm1\) remain unshifted at first order.
\end{enumerate}

\SupplementarySolution{5}{Einstein Coefficients and Electric-Dipole Radiation}
\begin{enumerate}
\item
For lower and upper populations \(N_1,N_2\), equilibrium requires
\[
 N_1B_{12}u(\omega)=N_2[B_{21}u(\omega)+A_{21}].
\]
Using \(N_2/N_1=(g_2/g_1)e^{-\beta\omega}\) and
\[
 u(\omega)=\frac{\omega^3}{\pi^2(e^{\beta\omega}-1)}
\]
forces the coefficients of the constant and Boltzmann-dependent pieces to
agree:
\[
 \boxed{g_1B_{12}=g_2B_{21},\qquad
 \frac{A_{21}}{B_{21}}=\frac{\omega^3}{\pi^2}.}
\]
Spontaneous emission is therefore required for thermal equilibrium with
the full Planck spectrum.
\item
For a photon of momentum \(\mathbf k\) and polarization \(\lambda\), the
matrix element is proportional to
\(q\sqrt{\omega/2}\,
\boldsymbol\epsilon_\lambda^*\cdot\mathbf x_{fi}\).
Golden-rule phase space fixes \(\omega=E_i-E_f\).  Using
\[
 \sum_\lambda\epsilon_\lambda^i\epsilon_\lambda^{j*}
 =\delta^{ij}-\widehat k^i\widehat k^j,\qquad
 \int \dd\Omega(\delta^{ij}-\widehat k^i\widehat k^j)
 =\frac{8\pi}{3}\delta^{ij},
\]
gives
\[
 \boxed{\Gamma_{i\to f}^{\rm E1}
 =\frac{q^2}{3\pi}\omega^3|\mathbf x_{fi}|^2
 =\frac{4\alpha}{3}\omega^3|\mathbf x_{fi}|^2}
\]
for \(|q|=e\).
\item
Because \(\mathbf x\) has odd parity, a nonzero matrix element requires
\((-1)^{\ell'}=-(-1)^\ell\); hence \(\ell+\ell'\) is odd.  A rank-one
operator also satisfies the triangle rule
\(|\ell-1|\leq\ell'\leq\ell+1\).  Combining the two conditions gives
\[
 \boxed{\Delta\ell=\pm1.}
\]
For \(\ell=\ell'=0\), the triangle condition itself reads
\(|0-1|\leq0\), which is false; equivalently, the relevant rank-one
Clebsch--Gordan coefficient vanishes.
\end{enumerate}

\SupplementarySolution{6}{Polarization Rules, the \(2p\) Lifetime, and Stark Quenching}
\begin{enumerate}
\item
The spherical components \(x_q\) form a rank-one tensor, so
\[
 \langle\ell'm'|x_q|\ell m\rangle
 \propto C^{\ell'm'}_{\ell m,\,1q}.
\]
The Clebsch--Gordan coefficient vanishes unless \(m'=m+q\).  Therefore
\[
 \boxed{\Delta m=0,\pm1.}
\]
The \(q=0\) component corresponds to linear, or \(\pi\), polarization along
the quantization axis.  The \(q=\pm1\) components correspond to the two
circular \(\sigma^\pm\) polarizations, with the sign fixed by the spherical
basis convention.
\item
The energy gap and radial integral are
\[
 \omega=\frac{3}{8}\alpha^2m_e,\qquad
 \int_0^\infty \dd r\,r^3R_{10}R_{21}
 =\frac{256a_0}{81\sqrt6}.
\]
For any initial magnetic sublevel, the angular sum gives one third of the
squared radial integral.  Inserting this in the spontaneous-emission
formula from S.14.5(b) yields
\[
 \boxed{\Gamma_{2p\to1s}
 =\frac{2^8}{3^8}\alpha^5m_e
 =6.27\times10^8\ {\rm s}^{-1},\qquad
 \tau_{2p}=1.60\ {\rm ns}.}
\]
Rotational invariance ensures the same total width for all three \(2p\)
magnetic substates.
\item
To first order in the dc field,
\[
 \ket{\widetilde{2s}}
 =\ket{2s}+c\,\ket{2p,0},\qquad
 c=\frac{\langle2p,0|e\mathcal Ez|2s\rangle}{\Delta_{\rm LS}}
 =-\frac{3ea_0\mathcal E}{\Delta_{\rm LS}}.
\]
The \(2p\) component decays by its allowed E1 transition, so
\[
 \boxed{\Gamma_{\widetilde{2s}}
 =\left(\frac{3ea_0\mathcal E}{\Delta_{\rm LS}}\right)^2
 \Gamma_{2p}.}
\]
This perturbative expression requires
\(3ea_0|\mathcal E|\ll|\Delta_{\rm LS}|\).  It explains why even a modest
laboratory electric field can quench the otherwise long-lived \(2s\) level.
\end{enumerate}

\SupplementarySolution{7}{Vacuum-Polarization Contribution to the Lamb Shift}
\begin{enumerate}
\item
The Coulomb virial estimate gives
\[
 |\mathbf p|\sim m_e\alpha,\qquad
 r\sim(m_e\alpha)^{-1},\qquad
 |E_{\rm bind}|\sim m_e\alpha^2.
\]
The electron wave functions therefore restrict a typical spatial
momentum transfer to
\(|\mathbf k|\sim m_e\alpha\), while an energy transfer is only
\(|k^0|\sim m_e\alpha^2\).  Consequently
\[
 \frac{\mathbf k^2}{m_e^2}=O(\alpha^2)\ll1,
 \qquad
 \frac{(k^0)^2}{\mathbf k^2}=O(\alpha^2)\ll1.
\]
\item
Expanding the supplied logarithm gives
\[
\begin{split}
 \widetilde{\mathbf E}(k)
 &=\left[
 1+\frac{e^2}{2\pi^2}\frac{\mathbf k^2}{m_e^2}
 \int_0^1\dd x\,x^2(1-x)^2
 +O(\mathbf k^4/m_e^4)\right]\widetilde{\mathbf D}(k)\\
 &=\left[
 1+\frac{\alpha}{15\pi}\frac{\mathbf k^2}{m_e^2}
 +O(\mathbf k^4/m_e^4)\right]\widetilde{\mathbf D}(k),
\end{split}
\]
where \(\int_0^1x^2(1-x)^2\dd x=1/30\) and
\(e^2=4\pi\alpha\).  Applied to the proton Coulomb field, the
momentum-independent correction to the electron's potential energy is
\[
 \delta\widetilde V(\mathbf k)
 =-\frac{4\alpha^2}{15m_e^2}.
\]
Thus
\[
 \boxed{\delta V(\mathbf r)
 =-V_1\delta^3(\mathbf r),\qquad
 V_1=\frac{4\alpha^2}{15m_e^2}.}
\]
\item
First-order perturbation theory gives
\[
 \delta E_{n\ell m}
 =-V_1|\psi_{n\ell m}(0)|^2.
\]
A regular Coulomb radial wave function behaves as
\(R_{n\ell}(r)\propto r^\ell\) near the origin, so
\(\psi_{n\ell m}(0)=0\) unless \(\ell=0\).  Only \(s\)-levels shift.
\item
For the hydrogen \(2s\) state,
\[
 R_{20}(r)
 =\frac{1}{\sqrt{2a_0^3}}
 \left(1-\frac{r}{2a_0}\right)e^{-r/(2a_0)},
 \qquad
 a_0=\frac{1}{m_e\alpha},
\]
and hence
\[
 |\psi_{2s}(0)|^2=\frac{1}{8\pi a_0^3}.
\]
The shift is
\[
 \boxed{\delta E_{2s}
 =-\frac{V_1}{8\pi a_0^3}
 =-\frac{m_e\alpha^5}{30\pi}
 \simeq-1.1\times10^{-7}\ {\rm eV}.}
\]
Equivalently, \(\delta E/h\simeq-27\ {\rm MHz}\).  The observed
\(2s-2p\) Lamb splitting is about \(4\times10^{-6}\ {\rm eV}\), is
opposite in sign, and is dominated by the electron self-energy rather
than this smaller vacuum-polarization contribution.
\end{enumerate}

\SupplementarySolution{8}{The Magnetic Moment of a Dirac Fermion}
\begin{enumerate}
\item
With \(\beta=\ii\gamma^0\), the volume current
\[
 j^\mu=-\ii\bar\Psi\gamma^\mu\Psi
\]
is Hermitian.  Since \(q\) is real, \(H_I\) is Hermitian precisely when
the classical four-potential is real:
\[
 \boxed{A_\mu(x)^*=A_\mu(x).}
\]
\item
A polar four-vector transforms under parity as
\[
 A^0(t,\mathbf x)\longmapsto A^0(t,-\mathbf x),
 \qquad
 \mathbf A(t,\mathbf x)\longmapsto-\mathbf A(t,-\mathbf x),
\]
so
\[
 \mathbf E\overset{P}{\longmapsto}-\mathbf E,\qquad
 \mathbf B\overset{P}{\longmapsto}+\mathbf B.
\]
The Dirac vector current transforms in the same way, making
\(j^\mu A_\mu\) parity even.  Under charge conjugation the vector
current is odd, so invariance requires
\[
 A_\mu\overset{C}{\longmapsto}-A_\mu,
 \qquad
 \mathbf E\overset{C}{\longmapsto}-\mathbf E,\qquad
 \mathbf B\overset{C}{\longmapsto}-\mathbf B.
\]
Spin is an axial vector.  Therefore
\(\mathbf S\cdot\mathbf B\) is parity even, whereas
\(\mathbf S\cdot\mathbf E\) is parity odd.  A magnetic dipole coupling
is allowed, but an electric dipole coupling would violate parity
(and, in the relativistic theory, the corresponding \(CP\) symmetry).
\item
In the Weyl basis used in the prompt, the spatial commutators are
\[
 \gamma^{ij}
 =\ii\varepsilon^{ijk}
 \begin{pmatrix}\sigma^k&0\\0&\sigma^k\end{pmatrix},
 \qquad
 F_{ij}=-\varepsilon_{ij\ell}B^\ell.
\]
The terms containing \(F_{0\ii}\) are velocity suppressed at rest.
Consequently
\[
 \gamma^{\mu\nu}F_{\mu\nu}
 \longrightarrow
 -2\ii
 \begin{pmatrix}
 \boldsymbol\sigma\cdot\mathbf B&0\\
 0&\boldsymbol\sigma\cdot\mathbf B
 \end{pmatrix}.
\]
Using the two rest spinors and
\(\bar u=u^\dagger\beta\) gives
\[
 \boxed{
 \bar u(p')\gamma^{\mu\nu}u(p)F_{\mu\nu}
 =-4\ii m_\Psi\,
 \xi'^\dagger\boldsymbol\sigma\cdot\mathbf B\,\xi,
 \qquad a=-4\ii m_\Psi.}
\]
The imaginary coefficient is a consequence of the volume's
anti-Hermitian vector gamma convention; the physical current contains
the compensating factor \(-\ii\).
\item
The Gordon identity in the same convention may be written
\[
 -\ii\bar u(p')\gamma^\mu u(p)
 =\frac{1}{2m_\Psi}\bar u(p')
 \left[(p'+p)^\mu+\gamma^{\mu\nu}(p'-p)_\nu\right]u(p).
\]
The first term gives the spin-independent convection current.  The
second term, integrated against a uniform background field, gives
\[
 \bra{\mathbf p',s'}H_I\ket{\mathbf p,s}
 =(2\pi)^3\delta^3(\mathbf p-\mathbf p')
 \left[
 -q\,\xi'^\dagger\boldsymbol\sigma\cdot\mathbf B\,\xi
 +\text{spin-independent terms}\right]
\]
for relativistically normalized one-particle states.  Dividing by
\(\sqrt{2\omega_{\mathbf p'}\,2\omega_{\mathbf p}}\to2m_\Psi\) gives the
nonrelativistic kernel
\[
 \boxed{
 h_{\rm spin}
 =-\frac{q}{2m_\Psi}
 \xi'^\dagger\boldsymbol\sigma\cdot\mathbf B\,\xi.}
\]
Thus
\[
 H_{\rm spin}
 =-\frac{q}{m_\Psi}\mathbf S\cdot\mathbf B
 =-g_{\rm mag}\frac{q}{2m_\Psi}\mathbf S\cdot\mathbf B,
 \qquad
 \boxed{g_{\rm mag}=2.}
\]
\item
The bilinear
\(\bar\Psi F_{\mu\nu}[\gamma^\mu,\gamma^\nu]\Psi\)
is anti-Hermitian in the stated convention.  Hence
\[
 \boxed{c^*=-c.}
\]
Write \(c=ic_H\) with \(c_H\) real.  Since
\([\gamma^\mu,\gamma^\nu]=2\gamma^{\mu\nu}\), part (c) gives, after
conversion to nonrelativistic normalization,
\[
 \Delta h_{\rm spin}
 =\frac{4c_H}{M}\,
 \xi'^\dagger\boldsymbol\sigma\cdot\mathbf B\,\xi.
\]
The magnetic coefficient becomes
\[
 -\frac{q}{2m_\Psi}+\frac{4c_H}{M}
 =-\frac{g_{\rm mag}q}{4m_\Psi},
\]
or
\[
 \boxed{
 g_{\rm mag}
 =2-\frac{16c_Hm_\Psi}{qM}.}
\]
An opposite convention for the sign of the added Pauli operator
correspondingly reverses the sign of the shift; its invariant content
is a correction of order \(m_\Psi/M\).
\end{enumerate}

\SupplementarySolution{9}{Bubble-Chain Bound States in \(2+1\) Dimensions}
\begin{enumerate}
\item
Include the factor \(1/2\) that exchanges the two identical lines in
each bubble and define
\[
 B(s)=\frac{1}{16\pi}\int_0^1
 \frac{\dd x}{\sqrt{m^2-x(1-x)s-\ii0}}.
\]
This follows by combining the two propagators with a Feynman parameter,
shifting the loop momentum, and using the elementary three-dimensional
Euclidean integral.  Below threshold it can also be written
\[
 B(s)=\frac{1}{8\pi\sqrt{s}}\,
 \operatorname{artanh}\!\left(\frac{\sqrt{s}}{2m}\right),
\qquad 0<s<4m^2.
\]
With the tree amplitude normalized as \(-g\), the displayed chain is
\[
 \mathcal M(s)
 =-g+g^2B(s)-g^3B(s)^2+\cdots
 =\boxed{-\frac{g}{1+gB(s)}}.
\]
No assertion about convergence of the geometric series is needed; the
last expression defines the bubble-chain resummation.
\item
For \(s>4m^2\), continuation through the two-particle cut gives
\[
 \operatorname{Im}B(s)=\frac{1}{16\sqrt{s}}.
\]
Consequently
\[
 \operatorname{Im}\mathcal M(s)
 =\frac{g^2\operatorname{Im}B(s)}
 {|1+gB(s)|^2}
 =\frac{|\mathcal M(s)|^2}{16E_T}.
\]
This is exactly the two-body optical theorem.  To see it from the
cross section supplied in the problem, avoid double counting identical
final scalars by integrating over an angular range of length \(\pi\)
(equivalently integrate over \(2\pi\) and divide by \(2!\)):
\[
 \sigma_{\rm el}
 =\frac{|\mathcal M|^2}{32pE_T^2}.
\]
Multiplication by the center-of-mass flux \(4pE_T\) gives
\[
 4pE_T\,\sigma_{\rm el}
 =\frac{|\mathcal M|^2}{8E_T}
 =2\,\operatorname{Im}\mathcal M,
\]
which verifies unitarity within the approximation.
\item
Below threshold \(B(s)\) is real, positive, and monotone, with
\[
 B(0)=\frac{1}{16\pi m},\qquad
 \lim_{s\to4m^2-}B(s)=+\infty.
\]
A pole requires
\[
 1+gB(M_B^2)=0.
\]
It is therefore possible only for attractive coupling \(g<0\).
The value \(-1/g\) lies strictly inside the range of \(B(s)\) for
\[
 \boxed{-16\pi m<g<0.}
\]
At \(g=-16\pi m\) the pole has reached \(M_B^2=0\), and for more
negative \(g\) there is no pole in the requested interval
\(0<M_B^2<4m^2\).
\end{enumerate}

\SupplementarySolution{10}{Nonrelativistic Forces from QED and Pseudoscalar Exchange}
\begin{enumerate}
\item
For momentum transfer \(q^\mu\simeq(0,\mathbf q)\), the temporal Dirac
current dominates:
\[
 \bar u(\mathbf p')\gamma^0u(\mathbf p)\simeq2m_\Psi.
\]
The direct one-photon graph therefore gives
\[
 \mathcal M_{\rm NR}
 \simeq\frac{4m_\Psi^2e^2}{\mathbf q^2}.
\]
Removing the two relativistic external-state normalization factors
leaves the Born kernel
\[
 \widetilde V(\mathbf q)=\frac{e^2}{\mathbf q^2}.
\]
Fourier transformation yields
\[
 V(r)=\frac{e^2}{4\pi r}=\frac{\alpha}{r},
\qquad
 \mathbf F=-\boldsymbol\nabla V
 =\frac{\alpha}{r^2}\widehat{\mathbf r}.
\]
The force is outward and hence repulsive.  The identical-fermion
exchange graph antisymmetrizes the two-particle state but does not
alter this long-range direct force between separated wave packets.
\item
\begin{enumerate}
\item
Let
\[
 \mathsf P\,\Psi(t,\mathbf x)\,\mathsf P^{-1}
 =\eta_P\beta\Psi(t,-\mathbf x),\qquad
 \mathsf P\,\phi(t,\mathbf x)\,\mathsf P^{-1}
 =-\phi(t,-\mathbf x),
\]
where \(\beta=\ii\gamma^0\) in the volume's convention and
\(|\eta_P|=1\).  Since
\(\beta^{-1}\gamma_5\beta=-\gamma_5\),
\(\bar\Psi\gamma_5\Psi\) is a pseudoscalar.  Its product with the
pseudoscalar field \(\phi\) is parity even.
\item
Taking \(\mathcal L_{\rm int}=-g_5\bar\Psi\gamma_5\Psi\phi\), a
consistent mostly-plus set of rules is
\[
 S_F(k)=\frac{-\ii\slashed k+m_\Psi}
 {k^2+m_\Psi^2-\ii0},\qquad
 \Delta_F(k)=\frac{-\ii}{k^2+m_\phi^2-\ii0},\qquad
 \Gamma_5=-ig_5\gamma_5,
\]
together with momentum conservation at the vertex and a minus sign
for exchanging identical external fermions.  An overall phase of
\(\Gamma_5\) changes if the Hermitian interaction is written with an
explicit \(i\); none of the following force laws changes.
\item
For
\(\Psi(p_1)\Psi(p_2)\to\Psi(p_3)\Psi(p_4)\), the two leading graphs are
the direct and exchange scalar-exchange diagrams:
\[
\begin{split}
 \ii\mathcal M_t
 &=\bigl[\bar u(p_3)\Gamma_5u(p_1)\bigr]\,
   \Delta_F(p_1-p_3)\,
   \bigl[\bar u(p_4)\Gamma_5u(p_2)\bigr],\\
 \ii\mathcal M_u
 &=-\bigl[\bar u(p_4)\Gamma_5u(p_1)\bigr]\,
   \Delta_F(p_1-p_4)\,
   \bigl[\bar u(p_3)\Gamma_5u(p_2)\bigr].
\end{split}
\]
The relative minus sign is fixed by Fermi statistics.  These two
expressions are the complete leading-order relativistic answer.
\item
At zero three-momentum,
\(\bar u\gamma_5u=0\).  Keeping the first nonzero term gives, with
\(\mathbf q=\mathbf p'-\mathbf p\),
\[
 \bar u(\mathbf p')\gamma_5u(\mathbf p)
 =-\xi'^\dagger\boldsymbol\sigma\cdot\mathbf q\,\xi
 +O(|\mathbf p|^3/m_\Psi^2).
\]
After converting relativistic external states to nonrelativistic
normalization, the direct interaction is
\[
 \boxed{\widetilde V_5(\mathbf q)
 =-\frac{g_5^2}{4m_\Psi^2}
 \frac{(\boldsymbol\sigma_1\cdot\mathbf q)
       (\boldsymbol\sigma_2\cdot\mathbf q)}
      {\mathbf q^2+m_\phi^2}.}
\]
Thus the naive momentum-independent force vanishes; the first force is
spin dependent and suppressed by \(1/m_\Psi^2\).

For completeness, let
\[
 Y(r)=\frac{e^{-m_\phi r}}{4\pi r},\qquad
 S_{12}=3(\boldsymbol\sigma_1\cdot\widehat{\mathbf r})
          (\boldsymbol\sigma_2\cdot\widehat{\mathbf r})
          -\boldsymbol\sigma_1\cdot\boldsymbol\sigma_2.
\]
Fourier transformation gives the effective coordinate-space
Hamiltonian
\[
\begin{split}
 V_5(\mathbf r)
 =\frac{g_5^2}{12m_\Psi^2}\Bigg\{&
 (\boldsymbol\sigma_1\cdot\boldsymbol\sigma_2)
 \left[m_\phi^2Y(r)-\delta^3(\mathbf r)\right]\\
 &+S_{12}
 \left(m_\phi^2+\frac{3m_\phi}{r}+\frac{3}{r^2}\right)Y(r)
 \Bigg\}.
\end{split}
\]
For identical particles this kernel acts on an antisymmetric
two-fermion wave function; that antisymmetrization is the
nonrelativistic counterpart of the \(u\)-channel graph.
