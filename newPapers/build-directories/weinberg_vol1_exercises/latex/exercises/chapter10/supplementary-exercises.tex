\SupplementaryExercise{1}{Positive Spectral Measures, Pole Weights, and Sum Rules}
  {(Inspired by Mark Srednicki, \emph{Quantum Field Theory}, Chapter 13 | Mark Srednicki, \emph{Quantum Field Theory}, Chapters 13 and 15.)}
  {srednicki-qft-spectral-2006-ch10-curated-parent-1}
\begin{enumerate}
\item
For a Hermitian scalar field with vanishing vacuum expectation value,
insert exact energy--momentum eigenstates into its Wightman two-point
function.  Derive a Lorentz-invariant spectral density, prove that it is
nonnegative, and identify the part produced by a stable one-particle state.
\item
Suppose a scalar field interpolates a stable particle of mass \(m\), while
the lightest multiparticle state with the same quantum numbers has invariant
mass \(M_{\rm th}\).  Determine the support of the pole and continuum pieces
of its spectral density and the resulting analytic structure of the exact
propagator.
\item
Starting from the K\"all\'en--Lehmann representation of a canonically
normalized scalar field, take the equal-time derivative of its commutator.
Derive the normalization sum rule for the spectral density and use it to
bound the isolated pole residue.
\end{enumerate}

\SupplementaryExercise{2}{Euclidean Positivity, Fermion Parity, and Ghost Tests}
  {(Inspired by Mark Srednicki, \emph{Quantum Field Theory}, Chapters 13 and 15 | Mark Srednicki, \emph{Quantum Field Theory}, Chapter 51 | Mark Srednicki, \emph{Quantum Field Theory}, Chapters 13 and 51.)}
  {srednicki-qft-spectral-2006-ch10-curated-parent-2}
\begin{enumerate}
\item
Set \(Q^2=p_E^2>0\) in a scalar spectral representation and regard
\(D_E(Q^2)\) as a function of \(Q^2\).  Prove that
\((-1)^n d^nD_E/d(Q^2)^n\geq0\) for every nonnegative integer \(n\).
Explain which assumption makes these inequalities possible.
\item
Resolve the rest-frame intermediate states in a Dirac two-point function
into the two parity projectors.  Derive the two-function fermionic spectral
representation and prove the pointwise inequality relating its scalar and
vector spectral densities.
\item
Consider the proposed scalar propagator
\[
 D(p)=\frac{1}{p^2+m^2-i0}
 -\frac{c}{p^2+M^2-i0},
 \qquad M>m,\quad c>0.
\]
Determine its spectral density and decide whether it can be the exact
two-point function of a positive-metric local scalar field.  State how the
answer changes if the second sign is positive.
\end{enumerate}

\SupplementaryExercise{3}{Local and Momentum-Space Ward--Takahashi Identities}
  {(Inspired by David Skinner, \emph{Symmetries in Quantum Field Theory}, Section 6.)}
  {skinner-aqft-ward-2017-ch10-curated-parent-1}
\begin{enumerate}
\item
For a complex scalar with current \(j^\mu\), derive the local Ward--Takahashi
identity for
\(\langle Tj^\mu(x)\Phi(y)\Phi^\dagger(z)\rangle\) directly from a
position-dependent phase change in the path integral.  Display every
contact term and its sign.
\item
Fourier transform the identity of the preceding part, factor the full
external scalar propagators from the connected three-point function, and
derive the relation between the amputated current vertex and the inverse
exact propagator.
\item
Take the zero-current-momentum limit of the scalar Ward--Takahashi identity.
Show that the vertex at zero momentum transfer is a momentum derivative of
the exact inverse propagator.  Explain why this relates field and vertex
renormalization constants.
\end{enumerate}

\SupplementaryExercise{4}{Neutral Insertions and Vanishing Contact Terms and Charge Conservation as a Correlator Selection Rule}
  {(Inspired by David Skinner, \emph{Symmetries in Quantum Field Theory}, Section 6.)}
  {skinner-aqft-ward-2017-ch10-curated-parent-2}
\begin{enumerate}
\item
Let \(O_1,\ldots,O_n\) be local operators with charges \(q_1,\ldots,q_n\).
Derive the Ward identity for one current insertion and these operators.
Show that all universal contact terms disappear when every \(O_i\) is
neutral, and distinguish this statement from conservation at coincident
points.
\item
Integrate the local Ward identity over the current insertion point, assuming
that the surface term vanishes.  Prove that a vacuum correlator of charged
operators is zero unless their total charge vanishes.  Check the conclusion
for a correlator containing \(r\) fields \(\Phi\) and \(s\) fields
\(\Phi^\dagger\).
\end{enumerate}

\SupplementaryExercise{5}{Current Conservation after LSZ Reduction}
  {(Inspired by David Skinner, \emph{Symmetries in Quantum Field Theory}, Section 6.)}
  {skinner-aqft-ward-2017-ch10-curated-parent-3}
Apply LSZ reduction to a time-ordered correlator with one electromagnetic
current and otherwise charged-field insertions.  Show why the Ward contact
terms lack the simultaneous external-particle poles and prove that replacing
an external photon polarization by its momentum gives zero.

\SupplementaryExercise{6}{Two Form Factors for an Elastic Dirac Current and Charge and Magnetic Moment at Zero Transfer}
  {(Inspired by David Tong, \emph{Lectures on Quantum Field Theory}, Sections 4.4 and 6.5.)}
  {tong-qft-form-factors-2006-ch10-curated-parent-1}
\begin{enumerate}
\item
Use Lorentz covariance, parity, the on-shell Dirac equations, and current
conservation to reduce an elastic spin-\(\tfrac12\) electromagnetic current
matrix element to two independent form factors.  Give both a Dirac--Pauli
basis and a basis using \(p+p'\).
\item
Take the nonrelativistic, small-momentum-transfer limit of the Dirac--Pauli
current vertex.  Identify the electric charge and magnetic moment in terms
of \(F_1(0)\) and \(F_2(0)\), and state the value for a minimally coupled
point Dirac particle at tree level.
\end{enumerate}

\SupplementaryExercise{7}{Why a Transition Charge Must Vanish and Opposite-Parity Electromagnetic Transitions}
  {(Inspired by David Tong, \emph{Lectures on Quantum Field Theory}, Sections 4.4 and 6.5 | David Tong, \emph{Lectures on Quantum Field Theory}, Sections 4.3--4.4.)}
  {tong-qft-form-factors-2006-ch10-curated-parent-2}
\begin{enumerate}
\item
Consider a conserved vector current between stable spin-\(\tfrac12\) states
of unequal masses.  Construct a manifestly transverse two-form-factor
basis.  By demanding regularity at \(q^2=0\), prove that the transition
charge form factor vanishes there.
\item
Repeat the unequal-mass current analysis when the initial and final states
have opposite intrinsic parity.  Determine where \(\gamma_5\) must occur
and show explicitly why the mass sum, rather than the mass difference,
appears in the current-conservation relation.
\end{enumerate}

\SupplementaryExercise{8}{The Scalar Charge-Radius Expansion and Transverse Decomposition of Vacuum Polarization}
  {(Inspired by David Tong, \emph{Lectures on Quantum Field Theory}, Sections 2.2 and 6.1 | David Tong, \emph{Lectures on Quantum Field Theory}, Sections 6.1--6.4.)}
  {tong-qft-form-factors-2006-ch10-curated-parent-3}
\begin{enumerate}
\item
Derive the most general conserved electromagnetic current between equal-mass
spin-zero states.  Normalize its form factor at zero momentum transfer and
show how the coefficient of its small-spacelike-momentum expansion defines
the mean-square charge radius.
\item
Starting with the most general Lorentz-covariant rank-two function of one
momentum, impose the electromagnetic Ward identity.  Derive the transverse
form of the vacuum-polarization tensor and show from Dyson resummation why
the photon remains massless when the renormalized scalar function is finite
at \(q^2=0\).
\end{enumerate}

\SupplementaryExercise{9}{Subtracted Dispersion Relations and Positivity}
  {(Inspired by Bernd A. Kniehl, \emph{Dispersion Relations in Loop Calculations}, Section 2 | Bernd A. Kniehl, \emph{Dispersion Relations in Loop Calculations}, Sections 2--3.)}
  {kniehl-dispersion-hepph9607255-ch10-curated-parent-1}
\begin{enumerate}
\item
Let \(F(z)\) be analytic away from a cut \(z\geq s_0\), real below the cut,
and no worse than a constant at infinity.  Derive a once-subtracted
dispersion relation at a spacelike point \(z_0<s_0\) in terms of
\(\operatorname{Im}F(s+i0)\).
\item
Assume \(F(z)=O(|z|^N\log |z|)\) on a large circle.  Determine a sufficient
number of subtractions for its dispersion integral.  Show explicitly how
changing a subtraction point changes only a polynomial and not the
discontinuity.
\item
Suppose a once-subtracted amplitude has a nonnegative absorptive part on its
physical cut.  Differentiate its dispersive representation at a real point
below threshold and prove positivity statements for all derivatives of
order two and higher.  State precisely why the first derivative may require
separate care.
\end{enumerate}

\SupplementaryExercise{10}{Bubble Reconstruction and Discontinuities}
  {(Inspired by Bernd A. Kniehl, \emph{Dispersion Relations in Loop Calculations}, Sections 3--4 | Bernd A. Kniehl, \emph{Dispersion Relations in Loop Calculations}, Section 4 | Bernd A. Kniehl, \emph{Dispersion Relations in Loop Calculations}, Sections 2--3.)}
  {kniehl-dispersion-hepph9607255-ch10-curated-parent-2}
\begin{enumerate}
\item
For two equal-mass scalar propagators, use two-body phase space to determine
the threshold dependence of the one-loop bubble's discontinuity.  Insert it
into a once-subtracted dispersion relation and recover a one-dimensional
representation of the renormalized bubble.
\item
Use the tree cross-section for a timelike photon to produce a charged Dirac
pair to find the absorptive part of the one-loop photon vacuum-polarization
scalar.  Write a subtraction at \(q^2=0\) and determine the first term of
the low-momentum expansion.
\item
Starting from \(S^\dagger S=1\), derive the forward optical theorem and
express it as the discontinuity of a forward amplitude across its physical
cut.  Explain how this relation supplies exactly the input needed for a
dispersive reconstruction of a loop amplitude.
\end{enumerate}

\SupplementaryExercise{11}{Compton Gauge Cancellation before Squaring and Forward Thomson Limit}
  {(Inspired by Daniel Harlow, \emph{Lecture Notes for Physics 8.324}, Sections 11.3--11.6.)}
  {harlow-qft2-forward-qed-2024-ch10-curated-parent-1}
\begin{enumerate}
\item
Write the two tree diagrams for spinor Compton scattering.  Replace the
incoming photon polarization by its momentum and use inverse-propagator
identities to prove cancellation before imposing a gauge or summing over
polarizations.
\item
Evaluate the spin-averaged forward Compton amplitude for an electron in the
limit \(\omega/m\to0\).  Show that the apparent pole terms combine to give
the finite Thomson amplitude and recover the corresponding low-energy total
cross-section.
\end{enumerate}

\SupplementaryExercise{12}{Integrating the Klein--Nishina Distribution}
  {(Inspired by Daniel Harlow, \emph{Lecture Notes for Physics 8.324}, Section 11.6.)}

\SupplementaryExercise{13}{Forward Compton Crossing and Dispersion}
  {(Inspired by Daniel Harlow, \emph{Lecture Notes for Physics 8.324}, Section 11.6 | Daniel Harlow, \emph{Lecture Notes for Physics 8.324}, Sections 11.3--11.6.)}
  {harlow-qft2-forward-qed-2024-ch10-curated-parent-3}
\begin{enumerate}
\item
Derive the physical photon polarization sum using an arbitrary reference
vector.  Prove that it may be replaced by the covariant metric tensor in a
squared amplitude only after the full current obeys its Ward identity, and
give a counterexample using a single Compton diagram.
\item
Decompose the forward Compton amplitude into spin-independent and
spin-dependent scalar functions.  Determine their parity under
\(\omega\mapsto-\omega\) from photon crossing and infer which subtraction
polynomials are allowed for each function.
\item
Use the optical theorem, crossing, the renormalized Thomson constant, and the
tree Klein--Nishina total cross-section to write the complete
spin-independent forward electron Compton amplitude through order \(e^4\).
Determine its leading low-energy imaginary part and explain why the
zero-energy photon threshold prevents an ordinary term-by-term Taylor
expansion of the dispersive integral.
\end{enumerate}
\end{enumerate}
