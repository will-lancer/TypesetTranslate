\SupplementaryExercise{1}{Positive Spectral Measures, Pole Weights, and Sum Rules}
  {(Inspired by Mark Srednicki, \emph{Quantum Field Theory}, Chapter 13 | Mark Srednicki, \emph{Quantum Field Theory}, Chapters 13 and 15.)}
  {srednicki-qft-spectral-2006-ch10-curated-parent-1}

\item
For a Hermitian scalar field with vanishing vacuum expectation value,
insert exact energy--momentum eigenstates into its Wightman two-point

nonnegative, and identify the part produced by a stable one-particle state.
\item
Suppose a scalar field interpolates a stable particle of mass \(m\), while

propagator.
\item
Starting from the K\"all\'en--Lehmann representation of a canonically
normalized scalar field, take the equal-time derivative of its commutator.
Derive the normalization sum rule for the spectral density and use it to
bound the isolated pole residue.
\end{enumerate}

\SupplementaryExercise{2}{The Thermal Lehmann Representation}
  {(Adapted from John McGreevy, \emph{Physics 215B Quantum Field Theory}, Winter 2025, Assignment 7, Problem 3, p.~1.)}
  {mcgreevy-215b-w25-hw07-p3}
In the vacuum, a scalar operator has the time-ordered two-point function
\[
 -iD(x)=\bra{0}T\{O(x)O^\dagger(0)\}\ket{0}.
\]
Derive the corresponding spectral representation when the system is in
thermal equilibrium at nonzero temperature \(T=\beta^{-1}\), starting from
\[
 \begin{split}
 -iD_\beta(x)
 &\equiv \operatorname{tr}\!\left(
       \frac{e^{-\beta H}}{Z_\beta}
       T\{O(x)O^\dagger(0)\}\right)\\
 &=\frac{1}{Z_\beta}\sum_n e^{-\beta E_n}
   \bra{n}T\{O(x)O^\dagger(0)\}\ket{n},
 \qquad Z_\beta\equiv\operatorname{tr}e^{-\beta H}.
 \end{split}
\]
Make the thermal spectral weight and its detailed-balance relation explicit,
and verify that the limit \(\beta\to\infty\) reproduces the vacuum
representation.  You may take \(O=O^\dagger\).

\SupplementaryExercise{4}{Charge Conjugation and Odd-Photon Selection Rules}
  {(Adapted from University of Cambridge, \emph{Mathematical Tripos Part III, Quantum Field Theory, Paper 301}, 2024, Question 3, p.~4.)}
  {cambridge-partiii-2024-qft-paper301-q3}
\begin{enumerate}
\item
Scalar QED consists of a complex scalar \(\phi\) of mass \(M\) and an
electromagnetic field \(A_\mu\), coupled only by minimal coupling with
coupling \(e\).  Write its Lagrangian \(\mathcal L_{\mathrm{SQED}}\).
Likewise write \(\mathcal L_{\mathrm{QED}}\) for a Dirac field \(\psi\) of
mass \(m\) minimally coupled to \(A_\mu\).  Give the local \(U(1)\) gauge
transformations of both theories and derive, in each case, the Noether
current \(j^\mu\) for the global symmetry obtained by taking the gauge
parameter constant.
\item
Let the linear, unitary charge-conjugation operator \(\widehat C\) act on the
scalar field by
\[
 \widehat C^{-1}\phi(x)\widehat C=\phi^\dagger(x),
 \qquad
 \widehat C^{-1}\phi^\dagger(x)\widehat C=\phi(x),
 \qquad \widehat C^\dagger\widehat C=\mathbf 1.
\]
Determine the necessary transformation of \(A_\mu\), and show that
\(\mathcal L_{\mathrm{SQED}}\) is invariant.  Operator-ordering issues such
as normal ordering may be ignored.
\item
For the Dirac field, take
\[
 \widehat C^{-1}\psi(x)\widehat C=\mathsf C\,\bar\psi(x)^{\mathsf T},
 \qquad
 \widehat C^{-1}\bar\psi(x)\widehat C=\psi(x)^{\mathsf T}\mathsf C,
\]
where the \(4\times4\) matrix \(\mathsf C\) obeys
\[
 \mathsf C^{\mathsf T}=\mathsf C^{-1}=-\mathsf C,
 \qquad
 \mathsf C^{-1}(\gamma^\mu)^{\mathsf T}\mathsf C=-\gamma^\mu.
\]
Using the transformation of \(A_\mu\) found in the preceding part, prove
that \(\mathcal L_{\mathrm{QED}}\) is invariant.  You may use
\[
 \bar\psi M\psi=-\psi^{\mathsf T}M^{\mathsf T}\bar\psi^{\mathsf T}
\]
for any spinor matrix \(M\), and again may ignore normal-ordering issues.
\item
Assume that the vacuum \(\ket{\Omega}\) of either theory is invariant under
charge conjugation.  Without using Feynman diagrams, prove to all orders in
\(e\) that, for odd \(n\),
\[
 \bra{\Omega}T\{A^{\mu_1}(x_1)\cdots A^{\mu_n}(x_n)\}\ket{\Omega}=0,
 \qquad
 \bra{\Omega}T\{j^{\mu_1}(x_1)\cdots j^{\mu_n}(x_n)\}\ket{\Omega}=0.
\]
What do these identities imply physically for photon scattering in scalar
and spinor QED?
\end{enumerate}

\SupplementaryExercise{10}{Muon Vacuum Polarization from the Optical Theorem}
  {(Adapted from John McGreevy, \emph{Physics 215B Quantum Field Theory}, Winter 2025, Assignment 7, Problem 4, p.~2.)}
  {mcgreevy-215b-w25-hw07-p4}
For general initial and final states \(a\) and \(b\), the source sheet
states the optical theorem as
\[
 -i\bigl[\mathcal M(a\to b)-\mathcal M(b\to a)\bigr]
 =\sum_f\int d\Phi_f\,
   \mathcal M(b\to f)^*\mathcal M(a\to f).
\]
\emph{Editorial source note.}  The sheet omits complex conjugation from the
second amplitude on the left.  The identity following from
\(S^\dagger S=\mathbf 1\) contains \(\mathcal M(b\to a)^*\) there.  Keep
track of this printed omission explicitly and state which form is used in
the calculation.
\begin{enumerate}
\item
In QED with electrons of charge \(Q_e\) and muons of charge \(Q_\mu\),
consider forward \(e^-e^+\to e^-e^+\) scattering.  Isolate the contribution
to its imaginary part proportional to \(Q_e^2Q_\mu^2\).  Draw the relevant
Feynman diagram and compute the absorptive part
\(\operatorname{Im}\Pi_\mu\) of the muon vacuum polarization as a function
of the physical timelike invariant
\[
 s\equiv-(k_1+k_2)^2>0,
\]
where \(k_1,k_2\) are the incoming electron and positron momenta.  (The
minus sign is the translation of the source's mostly-minus definition to
this volume's metric.)  Results from the tree calculation of
\(e^+e^-\to\mu^+\mu^-\) may be reused.  Check that the imaginary part is
independent of the ultraviolet cutoff.
\item
Use the optical theorem and
\[
 \sigma(e^+e^-\to\mu^+\mu^-)\geq0
\]
to show that every Feynman diagram containing a closed fermion loop must
carry a minus sign.  With that sign included, verify the optical theorem for
the contribution calculated in the preceding part.
\end{enumerate}

\SupplementaryExercise{11}{The Complete Compton Trace}
  {(Adapted from Daniel Harlow, \emph{Lecture Notes for Physics 8.324: Relativistic Quantum Field Theory II}, Fall 2024, Section 11.6, Problem 3, p.~133, with Eqs.~(11.38), (11.45)--(11.47), pp.~129--131.)}
  {harlow-qft2-fall2024-sec11-6-p3}
Consider tree-level Compton scattering
\(e^-(k_1)+\gamma(k_2)\to e^-(k_1')+\gamma(k_2')\).  The direct and crossed
electron-exchange diagrams are both included, with
\[
 k_1+k_2=k_1'+k_2',\qquad
 k_1^2=k_1'^2=-m^2,\qquad k_2^2=k_2'^2=0.
\]
For their gauge-invariant sum, current conservation permits the physical
photon polarization sums to be replaced by
\[
 \sum_\lambda\epsilon_\mu(k,\lambda)
 \epsilon_\nu(k,\lambda)^*\longrightarrow\eta_{\mu\nu},
\]
and the electron spin sums are
\(\sum_\sigma u(k,\sigma)\bar u(k,\sigma)=\slashed{k}+im\).
Consequently Harlow's Eq.~(11.46), translated only by replacing the charge
\(q\) by this volume's \(e\), is
\[
\begin{split}
 \sum_{\sigma,\sigma'}|\widetilde{\mathcal M}_{c}|^2
={}&\frac{e^4}{4}\eta_{\mu\alpha}\eta_{\nu\beta}
 \operatorname{tr}\!\Bigg[(\slashed{k}_1'+im)
 \left(
 \frac{\gamma^\nu\slashed{k}_2\gamma^\mu
       +2k_1^\mu\gamma^\nu}{k_1\cdot k_2}
 +\frac{\gamma^\mu\slashed{k}_2'\gamma^\nu
       -2k_1^\nu\gamma^\mu}{k_1\cdot k_2'}
 \right)\\
 &\qquad\quad\times(\slashed{k}_1+im)
 \left(
 \frac{\gamma^\alpha\slashed{k}_2\gamma^\beta
       +2k_1^\alpha\gamma^\beta}{k_1\cdot k_2}
 +\frac{\gamma^\beta\slashed{k}_2'\gamma^\alpha
       -2k_1^\beta\gamma^\alpha}{k_1\cdot k_2'}
 \right)\Bigg].
\end{split}
\]
Evaluate this trace and prove
\[
 \begin{split}
 \frac14\sum_{\sigma,\sigma'}|\widetilde{\mathcal M}_{c}|^2
 =2e^4\Bigg[&
 \frac{k_1\cdot k_2'}{k_1\cdot k_2}
 +\frac{k_1\cdot k_2}{k_1\cdot k_2'}\\
 &+2m^2\left(\frac1{k_1\cdot k_2'}-\frac1{k_1\cdot k_2}\right)
 +m^4\left(\frac1{k_1\cdot k_2'}-\frac1{k_1\cdot k_2}\right)^2
 \Bigg].
 \end{split}
\]
You should first use the \(d\)-dimensional contraction identities
\[
\begin{gathered}
 \gamma^\mu\gamma_\mu=d,\qquad
 \gamma^\mu\gamma^\alpha\gamma_\mu=-(d-2)\gamma^\alpha,\\
 \gamma^\mu\gamma^\alpha\gamma^\beta\gamma_\mu
 =(d-4)\gamma^\alpha\gamma^\beta+4\eta^{\alpha\beta},\\
 \gamma^\mu\gamma^\alpha\gamma^\beta\gamma^\gamma\gamma_\mu
 =-2\gamma^\gamma\gamma^\beta\gamma^\alpha
 -(d-4)\gamma^\alpha\gamma^\beta\gamma^\gamma,
\end{gathered}
\]
so that no trace of more than four gamma matrices need be evaluated.
