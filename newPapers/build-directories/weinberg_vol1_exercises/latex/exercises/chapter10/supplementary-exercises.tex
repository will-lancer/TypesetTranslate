\SupplementaryExercise{2}{The Thermal Lehmann Representation}
  {(Adapted from John McGreevy, \emph{Physics 215B Quantum Field Theory}, Winter 2025, Assignment 7, Problem 3, p.~1.)}
  {mcgreevy-215b-w25-hw07-p3}

\[
 -iD(x)=\bra{0}T\{O(x)O^\dagger(0)\}\ket{0}.
\]
Derive the corresponding spectral representation when the system is in
thermal equilibrium at nonzero temperature \(T=\beta^{-1}\), starting from
\[
 \begin{split}
 -iD_\beta(x)
 &\equiv \operatorname{tr}\!\left(
       \frac{e^{-\beta H}}{Z_\beta}
       T\{O(x)O^\dagger(0)\}\right)\\
 &=\frac{1}{Z_\beta}\sum_n e^{-\beta E_n}
   \bra{n}T\{O(x)O^\dagger(0)\}\ket{n},

 \end{split}
\]

and verify that the limit \(\beta\to\infty\) reproduces the vacuum
representation.  You may take \(O=O^\dagger\).

\SupplementaryExercise{3}{The Ward--Takahashi Identity in Euclidean QED}
  {(Adapted from University of Cambridge, \emph{Mathematical Tripos Part III, Advanced Quantum Field Theory, Paper 304}, 2022, Question 3, p.~4.)}
  {cambridge-partiii-2022-aqft-paper304-q3}
For this problem only, work in Euclidean spacetime, with indices contracted
using \(\delta_{\mu\nu}\) and Euclidean gamma matrices satisfying
\(\{\gamma_E^\mu,\gamma_E^\nu\}=2\delta^{\mu\nu}\).  The QED Lagrangian is
\[
 \mathcal L_E=\frac14F_{\mu\nu}F^{\mu\nu}
 +\bar\psi(\slashed D_E+m)\psi,
 \qquad
 F_{\mu\nu}=\partial_\mu A_\nu-\partial_\nu A_\mu,
 \qquad
 \slashed D_E=\gamma_E^\mu(\partial_\mu-ieA_\mu).
\]
\begin{enumerate}
\item
Show that the Lagrangian is invariant under the global \(U(1)\)
transformations
\[
 \psi(x)\longmapsto e^{i\alpha}\psi(x),\qquad
 \bar\psi(x)\longmapsto\bar\psi(x)e^{-i\alpha},\qquad
 A_\mu(x)\longmapsto A_\mu(x).
\]
Use this invariance to show that the current
\[
 j^\mu(x)=\bar\psi(x)\gamma_E^\mu\psi(x)
\]
is conserved.
\item
Derive the Schwinger--Dyson equation
\[
 \partial_\mu
 \left\langle j^\mu(x)\psi(x_1)\bar\psi(x_2)\right\rangle
 =-\left[\delta^{(4)}(x-x_1)-\delta^{(4)}(x-x_2)\right]
 \left\langle\psi(x_1)\bar\psi(x_2)\right\rangle.
\]
\item
Use the preceding result to derive the Ward--Takahashi identity involving
the exact vertex function \(V_3^\mu\) and exact fermion propagator
\(G(\slashed p)\):
\[
 iq_\mu V_3^\mu(q,p_1,p_2)
 =ie\left[G^{-1}(\slashed p_1)
 -G^{-1}(\slashed p_1-\slashed q)\right],
 \qquad p_2=p_1-q.
\]
Here \(\slashed p=\gamma_E^\mu p_\mu\), and the charge \(e\) is included
in the normalization of \(V_3^\mu\).
\item
Explain what the Ward--Takahashi identity implies for renormalization of
terms in the Lagrangian, for example in the on-shell renormalization
scheme.  Comment on what, if any, conclusions continue to hold
nonperturbatively.
\end{enumerate}

\[
 \widehat C^{-1}\phi(x)\widehat C=\phi^\dagger(x),
 \qquad
 \widehat C^{-1}\phi^\dagger(x)\widehat C=\phi(x),
 \qquad \widehat C^\dagger\widehat C=\mathbf 1.
\]

 \widehat C^{-1}\psi(x)\widehat C=\mathsf C\,\bar\psi(x)^{\mathsf T},

\]
where the \(4\times4\) matrix \(\mathsf C\) obeys

 \mathsf C^{\mathsf T}=\mathsf C^{-1}=-\mathsf C,

 \qquad
 \bra{\Omega}T\{j^{\mu_1}(x_1)\cdots j^{\mu_n}(x_n)\}\ket{\Omega}=0.
\]

and spinor QED?

\SupplementaryExercise{5}{Scalar and Pseudoscalar Decays from Schwinger--Dyson and LSZ}
  {(Adapted from University of Cambridge, \emph{Mathematical Tripos Part III, Quantum Field Theory, Paper 301}, 2025, Question 3, pp.~4--5.)}
  {cambridge-partiii-2025-qft-paper301-q3}
Consider in four dimensions a massive real scalar field \(\phi(x)\) and a
massive Dirac field \(\psi(x)\).  In the volume's mostly-plus conventions,
the free Lagrangian is
\[
 \mathcal L_0=-\frac12\partial_\mu\phi\,\partial^\mu\phi
 -\frac12M^2\phi^2-\bar\psi(\gamma^\mu\partial_\mu+m)\psi,
\]
where \(M\) and \(m\) are the scalar and fermion masses.
\begin{enumerate}
\item
Consider first \(\mathcal L_0+\mathcal L_1\), with
\[
 \mathcal L_1=-\lambda\phi\bar\psi\psi.
\]
\begin{enumerate}
\item
Write down the classical equations of motion.  State, without derivation,
the three Schwinger--Dyson equations for this theory.
\item
Following the LSZ approach, derive the \(S\)-matrix element for the decay
\(\phi\to\psi\bar\psi\) by defining the asymptotic states and identifying
the correlation function that enters the matrix element.
\item
To leading order in \(\lambda\), evaluate the connected contribution to
the correlation function appearing in \(\phi\to\psi\bar\psi\) by using
the Schwinger--Dyson equations.  Hence evaluate the \(S\)-matrix element
to leading order and identify the corresponding invariant amplitude
\(\mathcal A_{s,s'}\).
\item
Compute the spin-summed/averaged squared matrix element
\[
 \mathcal P\equiv\frac14\sum_{s,s'=1}^{2}
 \left|\mathcal A_{s,s'}\right|^2,
\]
where \(s\) and \(s'\) are the spins of the final particles.
\end{enumerate}
\item
Now consider instead \(\mathcal L_0+\mathcal L_2\), with
\[
 \mathcal L_2=-ig\phi\bar\psi\gamma_5\psi,
 \qquad \gamma_5=i\gamma^0\gamma^1\gamma^2\gamma^3.
\]
What is the \(S\)-matrix element for \(\phi\to\psi\bar\psi\)?  You need
not repeat the derivation: state the appropriate modification of the
answer in part (a)(iii).  Evaluate the spin-summed/averaged squared matrix
element in this case.  Could an experiment distinguish the effects of
\(\mathcal L_1\) and \(\mathcal L_2\)?
\end{enumerate}

You may use without proof the following free-field mode expansions and
spinor identities.  With \(p^0>0\), define
\[
 \widetilde\omega_{\mathbf p}^{2}=M^2+\mathbf p^2,
 \qquad \omega_{\mathbf p}^{2}=m^2+\mathbf p^2,
 \qquad \widehat p\equiv-i\gamma^\mu p_\mu.
\]
The last definition translates the source's mostly-minus slash to the
volume convention.  The mode expansions are
\[
 \phi(x)=\int\frac{d^3p}{(2\pi)^3}\frac1{\sqrt{2\widetilde\omega_{\mathbf p}}}
 \left[a_{\mathbf p}e^{ip\cdot x}
 +a_{\mathbf p}^\dagger e^{-ip\cdot x}\right],
\]
\[
 \psi(x)=\int\frac{d^3p}{(2\pi)^3}\frac1{\sqrt{2\omega_{\mathbf p}}}
 \sum_{r=1}^{2}\left[
 b^r_{\mathbf p}u^r(\mathbf p)e^{ip\cdot x}
 +c^{r\dagger}_{\mathbf p}v^r(\mathbf p)e^{-ip\cdot x}
 \right].
\]
For on-shell \(p\),
\[
 (\widehat p-m)u^s(\mathbf p)=0,
 \qquad (\widehat p+m)v^s(\mathbf p)=0,
\]
\[
 \sum_{s=1}^{2}u^s(\mathbf p)\bar u^s(\mathbf p)=\widehat p+m,
 \qquad
 \sum_{s=1}^{2}v^s(\mathbf p)\bar v^s(\mathbf p)=\widehat p-m,
\]
and
\[
 \begin{aligned}
 u^{s'\dagger}(\mathbf p)u^s(\mathbf p)
 &=v^{s'\dagger}(\mathbf p)v^s(\mathbf p)
 =2\omega_{\mathbf p}\delta^{ss'},\\
 \bar u^{s'}(\mathbf p)u^s(\mathbf p)
 &=-\bar v^{s'}(\mathbf p)v^s(\mathbf p)
 =2m\delta^{ss'},\\
 u^{s\dagger}(\mathbf p)v^{s'}(-\mathbf p)
 &=v^{s\dagger}(\mathbf p)u^{s'}(-\mathbf p)=0.
 \end{aligned}
\]

\SupplementaryExercise{7}{The Ordered Time Integrals in the General LSZ Pole}
  {(Adapted from Daniel Harlow, \emph{Lecture Notes for Physics 8.323:
  Relativistic Quantum Field Theory I} (2024), Section 9.5, Homework
  Problem 4, printed p.~120, with Eqs.~(9.44)--(9.53), printed
  pp.~115--116.)}
  {harlow-qft1-2024-sec9-5-p4}
Let \(O_1^{b_1},\ldots,O_M^{b_M}\) and
\(O_1'{}^{a_1},\ldots,O_N'{}^{a_N}\) be Heisenberg operators in a quantum
field theory with a scattering description, transforming in irreducible
Lorentz representations.  With
\(O(k)=\int d^dx\,e^{-ik\cdot x}O(x)\), introduce the regulated Fourier
transform
\[
\begin{split}
G_\epsilon={}&
 \langle\Omega|T\{O_N'{}^{a_N}(k_N')\cdots O_1'{}^{a_1}(k_1')
 O_1^{b_1\dagger}(k_1)\cdots O_M^{b_M\dagger}(k_M)\}|\Omega\rangle_\epsilon
\\
:={}&\int d^dx_1\cdots d^dx_{M+N}\,
 e^{-i(k_1\cdot x_1+\cdots+k_M\cdot x_M
       +k_1'\cdot x_{M+1}+\cdots+k_N'\cdot x_{M+N})}
\\[-2pt]
&\quad\times
 \langle\Omega|T\{O_N'{}^{a_N}(x_{M+N})\cdots
 O_1'{}^{a_1}(x_{M+1})O_1^{b_1\dagger}(x_1)\cdots
 O_M^{b_M\dagger}(x_M)\}|\Omega\rangle
\\[-2pt]
&\qquad\times e^{-\epsilon(t_{\max}-t_{\min})}.
\end{split}
\tag{9.44}
\]
Here \(t_{\min}\) and \(t_{\max}\) are the least and greatest of the
times \(t_1,\ldots,t_{M+N}\).  The incoming and outgoing on-shell limits
are
\[
 k_i^0\longrightarrow-\omega_{n_i,\mathbf k_i},
 \qquad
 k_j'{}^0\longrightarrow\omega_{n_j',\mathbf k_j'}.
\tag{9.45}
\]
The pole of interest comes from the ordered region
\[
 t_1\leq t_2\leq\cdots\leq t_{M+N}.
\tag{9.46}
\]
Insert complete sets of scattering states, writing the \(r\)-particle
states as \(\InKet{\alpha_r}\) and
\(\OutKet{\beta_r}\).  The contribution from this ordered
region is
\[
\begin{split}
G_\epsilon\supset{}&
 \int_{t_1\leq\cdots\leq t_{M+N}}
 d^dx_1\cdots d^dx_{M+N}
 \int d\alpha_1\cdots d\alpha_M\,d\beta_1\cdots d\beta_N
\\
&\times e^{-i(k_1\cdot x_1+\cdots+k_M\cdot x_M
 +k_1'\cdot x_{M+1}+\cdots+k_N'\cdot x_{M+N})
 -\epsilon(t_{M+N}-t_1)}
\\
&\times\langle\Omega|O_N'{}^{a_N}(x_{M+N})
 |\beta_1;\mathrm{out}\rangle\cdots
 \langle\beta_{N-1};\mathrm{out}|O_1'{}^{a_1}(x_{M+1})
 |\beta_N;\mathrm{out}\rangle
\\
&\times\langle\beta_N;\mathrm{out}|\alpha_M;\mathrm{in}\rangle
 \langle\alpha_M;\mathrm{in}|O_1^{b_1\dagger}(x_1)
 |\alpha_{M-1};\mathrm{in}\rangle\cdots
 \langle\alpha_1;\mathrm{in}|O_M^{b_M\dagger}(x_M)|\Omega\rangle.
\end{split}
\tag{9.48}
\]
Use translation invariance in the form
\[
 \langle\gamma|O(x)|\delta\rangle
 =e^{-i(p_\gamma-p_\delta)\cdot x}
  \langle\gamma|O(0)|\delta\rangle.
\tag{9.49}
\]
After extracting the matrix elements at the origin, the spacetime
dependence is therefore
\[
\begin{split}
G_\epsilon\supset{}&
 \int d\alpha_1\cdots d\alpha_M\,d\beta_1\cdots d\beta_N\,
 \langle\beta_N;\mathrm{out}|\alpha_M;\mathrm{in}\rangle
 \int_{t_1\leq\cdots\leq t_{M+N}}d^dx_1\cdots d^dx_{M+N}
\\
&\times\exp\!\Bigl\{-i\bigl[
 (k_1+p_{\alpha_M}-p_{\alpha_{M-1}})\cdot x_1+\cdots
 +(k_M+p_{\alpha_1})\cdot x_M
\\[-2pt]
&\hspace{60pt}
 +(k_1'-p_{\beta_N}+p_{\beta_{N-1}})\cdot x_{M+1}+\cdots
 +(k_N'-p_{\beta_1})\cdot x_{M+N}\bigr]
\\[-2pt]
&\hspace{60pt}
 -\epsilon(t_{M+N}-t_1)\Bigr\}
\\
&\times\langle\Omega|O_N'{}^{a_N}(0)|\beta_1;\mathrm{out}\rangle
 \cdots\langle\beta_{N-1};\mathrm{out}|O_1'{}^{a_1}(0)
 |\beta_N;\mathrm{out}\rangle
\\
&\times\langle\alpha_M;\mathrm{in}|O_1^{b_1\dagger}(0)
 |\alpha_{M-1};\mathrm{in}\rangle\cdots
 \langle\alpha_1;\mathrm{in}|O_M^{b_M\dagger}(0)|\Omega\rangle.
\end{split}
\tag{9.50}
\]
The spatial integrations give delta functions.  The remaining nested time
integral is
\[
\begin{split}
\mathcal T:={}&\int_{-\infty}^{\infty}dt_1\,
 e^{i(k_1^0+p_{\alpha_M}^0-p_{\alpha_{M-1}}^0-i\epsilon)t_1}
 \int_{t_1}^{\infty}dt_2\,
 e^{i(k_2^0+p_{\alpha_{M-1}}^0-p_{\alpha_{M-2}}^0)t_2}
\\[-2pt]
&\times\cdots
 \int_{t_{M-1}}^{\infty}dt_M\,
 e^{i(k_M^0+p_{\alpha_1}^0)t_M}
\\
&\times\int_{t_M}^{\infty}dt_{M+1}\,
 e^{i(k_1'{}^0-p_{\beta_N}^0+p_{\beta_{N-1}}^0)t_{M+1}}
 \cdots
\\[-2pt]
&\times\int_{t_{M+N-2}}^{\infty}dt_{M+N-1}\,
 e^{i(k_{N-1}'{}^0-p_{\beta_2}^0+p_{\beta_1}^0)t_{M+N-1}}
\\
&\times\int_{t_{M+N-1}}^{\infty}dt_{M+N}\,
 e^{i(k_N'{}^0-p_{\beta_1}^0+i\epsilon)t_{M+N}}.
\end{split}
\tag{9.51}
\]
Define
\[
 k_{\mathrm{tot}}=k_1+\cdots+k_M,
 \qquad
 k'_{\mathrm{tot}}=k_1'+\cdots+k_N'.
\tag{9.53}
\]
Evaluate the integrations in \(\mathcal T\), going from right to left, and
show that
\[
\begin{split}
\mathcal T={}&2\pi\delta\!\left(
 k_{\mathrm{tot}}^0+k_{\mathrm{tot}}'{}^0
 +p_{\alpha_M}^0-p_{\beta_N}^0\right)
\\
&\times
 \frac{-i}{k_1^0+p_{\alpha_M}^0-p_{\alpha_{M-1}}^0-i\epsilon}
 \frac{-i}{k_1^0+k_2^0+p_{\alpha_M}^0-p_{\alpha_{M-2}}^0-i\epsilon}
\\[-2pt]
&\times\cdots
 \frac{-i}{k_{\mathrm{tot}}^0-k_M^0+p_{\alpha_M}^0-p_{\alpha_1}^0-i\epsilon}
\\
&\times
 \frac{i}{k_{\mathrm{tot}}'{}^0-p_{\beta_N}^0+i\epsilon}
 \cdots
 \frac{i}{k_{N-1}'{}^0+k_N'{}^0-p_{\beta_2}^0+i\epsilon}
 \frac{i}{k_N'{}^0-p_{\beta_1}^0+i\epsilon}.
\end{split}
\tag{9.52}
\]
Be prepared to use the final delta function to rewrite the poles involving
the incoming energies \(k_i^0\).

\SupplementaryExercise{11}{The Complete Compton Trace}
  {(Adapted from Daniel Harlow, \emph{Lecture Notes for Physics 8.324: Relativistic Quantum Field Theory II}, Fall 2024, Section 11.6, Problem 3, p.~133, with Eqs.~(11.38), (11.45)--(11.47), pp.~129--131.)}
  {harlow-qft2-fall2024-sec11-6-p3}
Consider tree-level Compton scattering
\(e^-(k_1)+\gamma(k_2)\to e^-(k_1')+\gamma(k_2')\).  The direct and crossed
electron-exchange diagrams are both included, with
\[
 k_1+k_2=k_1'+k_2',\qquad
 k_1^2=k_1'^2=-m^2,\qquad k_2^2=k_2'^2=0.
\]
For their gauge-invariant sum, current conservation permits the physical
photon polarization sums to be replaced by
\[
 \sum_\lambda\epsilon_\mu(k,\lambda)
 \epsilon_\nu(k,\lambda)^*\longrightarrow\eta_{\mu\nu},
\]
and the electron spin sums are
\(\sum_\sigma u(k,\sigma)\bar u(k,\sigma)=\slashed{k}+im\).
Consequently Harlow's Eq.~(11.46), translated only by replacing the charge
\(q\) by this volume's \(e\), is
\[
\begin{split}
 \sum_{\sigma,\sigma'}|\widetilde{\mathcal M}_{c}|^2
={}&\frac{e^4}{4}\eta_{\mu\alpha}\eta_{\nu\beta}
 \operatorname{tr}\!\Bigg[(\slashed{k}_1'+im)
 \left(
 \frac{\gamma^\nu\slashed{k}_2\gamma^\mu
       +2k_1^\mu\gamma^\nu}{k_1\cdot k_2}
 +\frac{\gamma^\mu\slashed{k}_2'\gamma^\nu
       -2k_1^\nu\gamma^\mu}{k_1\cdot k_2'}
 \right)\\
 &\qquad\quad\times(\slashed{k}_1+im)
 \left(
 \frac{\gamma^\alpha\slashed{k}_2\gamma^\beta
       +2k_1^\alpha\gamma^\beta}{k_1\cdot k_2}
 +\frac{\gamma^\beta\slashed{k}_2'\gamma^\alpha
       -2k_1^\beta\gamma^\alpha}{k_1\cdot k_2'}
 \right)\Bigg].
\end{split}
\]
Evaluate this trace and prove
\[
 \begin{split}
 \frac14\sum_{\sigma,\sigma'}|\widetilde{\mathcal M}_{c}|^2
 =2e^4\Bigg[&
 \frac{k_1\cdot k_2'}{k_1\cdot k_2}
 +\frac{k_1\cdot k_2}{k_1\cdot k_2'}\\
 &+2m^2\left(\frac1{k_1\cdot k_2'}-\frac1{k_1\cdot k_2}\right)
 +m^4\left(\frac1{k_1\cdot k_2'}-\frac1{k_1\cdot k_2}\right)^2
 \Bigg].
 \end{split}
\]
You should first use the \(d\)-dimensional contraction identities
\[
\begin{gathered}
 \gamma^\mu\gamma_\mu=d,\qquad
 \gamma^\mu\gamma^\alpha\gamma_\mu=-(d-2)\gamma^\alpha,\\
 \gamma^\mu\gamma^\alpha\gamma^\beta\gamma_\mu
 =(d-4)\gamma^\alpha\gamma^\beta+4\eta^{\alpha\beta},\\
 \gamma^\mu\gamma^\alpha\gamma^\beta\gamma^\gamma\gamma_\mu
 =-2\gamma^\gamma\gamma^\beta\gamma^\alpha
 -(d-4)\gamma^\alpha\gamma^\beta\gamma^\gamma,
\end{gathered}
\]
so that no trace of more than four gamma matrices need be evaluated.

\SupplementaryExercise{13}{Callan--Symanzik Evolution and Asymptotic Freedom}
  {(Adapted from University of Cambridge, \emph{Mathematical Tripos Part III,
  Advanced Quantum Field Theory}, Paper 304, 2021, Question 3, printed
  p.~4.)}
  {cambridge-partiii-2021-aqft-paper304-q3}
Consider a theory of a single scalar field with one dimensionless coupling
\(g\) and no mass parameters.  Let
\[
 G^{(n)}(x_1,\ldots,x_n)
 =\langle\phi(x_1)\cdots\phi(x_n)\rangle_{\mathrm{conn}}
\]
be the renormalized connected \(n\)-point function.
\begin{enumerate}
\item
Explain why this correlation function depends on an additional scale
\(\mu\), define \(\beta(g)\) and \(\gamma(g)\), and derive
\[
 \left(\mu\frac{\partial}{\partial\mu}
 +\beta(g)\frac{\partial}{\partial g}+n\gamma(g)\right)
 G^{(n)}(x_1,\ldots,x_n)=0.
\tag{*}
\]
\item
What is the significance of the beta function and of a fixed point
\(g_*\)?  What does the sign of \(\beta(g)\) say about the behavior of the
coupling as \(\mu\) is increased from a reference scale \(\mu_0\)?
\item
Take as given that the renormalized propagator obeys
\[
 \int d^4x\,e^{ip\cdot x}G^{(2)}(x,0)
 =\frac1{p^2}C\!\left(\frac{p^2}{\mu^2},g(\mu)\right).
\]
For \(n=2\), solve (*) in the form
\[
 C\!\left(\frac{p^2}{\mu^2},g(\mu)\right)
 =f(\mu)
 C\!\left(\frac{p^2}{\mu_0^2},g(\mu_0)\right)
\]
for arbitrary \(\mu_0\) and unspecified running coupling \(g(\mu)\).
Show that \(f(\mu)=e^{h(\mu)}\), where
\[
 h(\mu)\propto\int_{g(\mu_0)}^{g(\mu)}dg\,
 \frac{\gamma(g)}{\beta(g)}.
\]
\emph{Hint.}  Use \(t=\log(\mu/\mu_0)\).
\item
Now assume
\[
 \beta(g)=-bg^3,
 \qquad b>0.
\]
Find every fixed point and describe its nature.  Express \(g(\mu)\) in
terms of \(g(\mu_0)\) and \(\mu/\mu_0\).  Let \(\Lambda\) be the scale at
which this expression makes \(g(\mu)\) diverge; rewrite \(g(\mu)\) in
terms of \(\mu\) and \(\Lambda\).
\item
Assuming further that \(\gamma(g)=cg^2\), evaluate
\[
 \frac{C(p^2/\mu^2,g(\mu))}{C(1,g(p))}.
\]
\end{enumerate}
\end{enumerate}
