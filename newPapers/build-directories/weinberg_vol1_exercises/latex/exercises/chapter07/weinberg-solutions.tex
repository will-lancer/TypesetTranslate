\chapterbackmatter{Solutions to Weinberg Exercises}
\ExerciseEditorialNote

\begin{enumerate}
\WeinbergSolution{1}
Only the symmetric part of \(f_{mn}\) contributes, so we take
\(f_{mn}=f_{nm}\), with inverse \(f^{mn}\).  The conjugate fields are
\[
 \Pi_n=\frac{\partial\mathcal L}{\partial\dot\Phi^n}
       =f_{nm}(\Phi)\dot\Phi^m ,
 \qquad
 \dot\Phi^n=f^{nm}(\Phi)\Pi_m .
\]
Canonical quantization imposes, at equal times,
\[
 [\Phi^m(\mathbf x),\Pi_n(\mathbf y)]
 =i\delta^m{}_n\delta^3(\mathbf x-\mathbf y),\qquad
 [\Phi^m,\Phi^n]=[\Pi_m,\Pi_n]=0 .
\]
The Legendre transform gives
\[
 H=\frac12\int d^3x\left[
 \Pi_m f^{mn}(\Phi)\Pi_n+
 f_{mn}(\Phi)\boldsymbol\nabla\Phi^m\mathbin{\cdot}
                       \boldsymbol\nabla\Phi^n\right].
\]
At the quantum level the first term is to be given a fixed Hermitian
ordering; for example, one may use the displayed sandwich ordering after
replacing \(f\) by a Hermitian multiplication operator.

Choose coordinates near \(\Phi=0\) for which \(f_{mn}(0)=\delta_{mn}\),
and put \(h_{mn}(\phi)=f_{mn}(\phi)-\delta_{mn}\).  The free theory is a
set of massless scalars with \(\pi_n=\dot\phi^n\).  At \(t=0\), and hence
at every interaction-picture time, the interaction is
\[
 \boxed{\;
 H_I(t)=\frac12\int d^3x\left[
 \dot\phi^m\bigl(f^{-1}_{mn}(\phi)-\delta_{mn}\bigr)\dot\phi^n
 +
 h_{mn}(\phi)\boldsymbol\nabla\phi^m\mathbin{\cdot}
                    \boldsymbol\nabla\phi^n\right].\;}
\]
Here \(f^{-1}_{mn}\) denotes the matrix inverse with its indices placed
using \(\delta_{mn}\).  The occurrence of \(f^{-1}\), rather than
\(-h\), is the characteristic result of doing the Legendre transform
before replacing interacting fields by free ones.

\WeinbergSolution{2}
It is convenient to replace the free Dirac density by the equivalent
Hermitian density
\[
 \mathcal L_{\rm D}
 =-\frac12\sum_i\bar\Psi^i\gamma^\rho
       \overleftrightarrow{\partial_\rho}\Psi^i
  -\sum_i m_i\bar\Psi^i\Psi^i ,
 \qquad
 A\overleftrightarrow{\partial_\rho}B
 =A\partial_\rho B-(\partial_\rho A)B .
\]
This differs from the usual first-order form only by a four-divergence.
Applying the Belinfante construction, and noting that the nonderivative
interaction contributes only through \(\eta^{\mu\nu}\mathcal L\), gives
\[
 \boxed{\begin{aligned}
 \Theta^{\mu\nu}={}&
 \sum_n\partial^\mu\Phi^n\partial^\nu\Phi^n\\
 &-\frac14\sum_i\bar\Psi^i\left(
 \gamma^\mu\overleftrightarrow{\partial^\nu}
 +\gamma^\nu\overleftrightarrow{\partial^\mu}\right)\Psi^i
 +\eta^{\mu\nu}\mathcal L .
 \end{aligned}}\tag{1}
\]
The first line is already symmetric.  The improvement supplied by the
Dirac spin current converts the canonical fermion tensor into the second
line, which is manifestly symmetric.  Direct differentiation of (1),
followed by use of the scalar and adjoint Dirac equations, gives
\(\partial_\mu\Theta^{\mu\nu}=0\).  The terms obtained by differentiating
\(\mathcal L_1(\Phi,\Psi,\bar\Psi)\) cancel against the interaction terms
in those equations.  Surface terms do not change
\(\int d^3x\,\Theta^{0\nu}=P^\nu\).

\WeinbergSolution{3}
For the real scalars,
\[
 \frac{\partial\mathcal L}{\partial(\partial_\mu\Phi^n)}
 =-\partial^\mu\Phi^n .
\]
For the Dirac fields it is simplest to use the unsymmetrized free
Lagrangian; the answer obtained from the Hermitian form differs only by
an identically conserved improvement.  Equation~\eqref{eq:7.3.46} then
gives
\[
 \boxed{\;
 J^\mu=
 i\sum_{n,m}\partial^\mu\Phi^n\,t^n{}_m\Phi^m
 +i\sum_{i,j}\bar\Psi^i\gamma^\mu\tau^i{}_j\Psi^j .\;}
\]
Reality of the scalar transformation requires \(it\) to be a real
antisymmetric matrix, while invariance of the Dirac inner product
requires the corresponding flavor generator to be Hermitian.  These
properties also make \(J^\mu\) Hermitian.  Differentiating the displayed
current and inserting the field equations leaves precisely the
variation of \(\mathcal L_1\) under the assumed symmetry; that variation
vanishes, so \(\partial_\mu J^\mu=0\).  Its charge agrees with
Eq.~\eqref{eq:7.3.40} and therefore generates the stated transformations
by equal-time commutators.

\WeinbergSolution{4}
Write \(V_0=-V^0\), and use Euclidean notation for spatial indices.  The
nonzero canonical momenta are
\[
 \pi=(D_0\Phi)^\dagger,\qquad
 \pi^\dagger=D_0\Phi,\qquad
 \Pi_i=F_{0i}=\dot V_i-\partial_iV_0 ,
 \qquad \Pi_0=0 .
\]
Thus \(\Pi_0=0\) is a primary constraint.  The velocities are
\[
 \dot\Phi=\pi^\dagger+igV_0\Phi,\qquad
 \dot\Phi^\dagger=\pi-igV_0\Phi^\dagger,\qquad
 \dot V_i=\Pi_i+\partial_iV_0 .
\]
Define the canonical charge density
\[
 \rho\equiv ig\bigl(\pi\Phi-\pi^\dagger\Phi^\dagger\bigr).
\]
After integrating \(\boldsymbol\Pi\cdot\boldsymbol\nabla V_0\) by parts,
the terms involving \(V_0\) are
\[
 V_0(\rho-\boldsymbol\nabla\mathbin{\cdot}\boldsymbol\Pi)
 -\frac12m^2V_0^2 .
\]
Its equation is the secondary constraint
\[
 V_0=\frac{\rho-\boldsymbol\nabla\cdot\boldsymbol\Pi}{m^2}.
\]
Eliminating \(V_0\) gives the reduced Hamiltonian
\[
 \begin{split}
 H=\int d^3x\Bigl[&
 \pi\pi^\dagger+(D_i\Phi)^\dagger D_i\Phi
 +\mathcal H(\Phi^\dagger\Phi)
 +\frac12\boldsymbol\Pi^2+\frac14F_{ij}F_{ij}
 +\frac12m^2\mathbf V^2\\
 &+\frac1{2m^2}
 \bigl(\boldsymbol\nabla\cdot\boldsymbol\Pi-\rho\bigr)^2\Bigr].
 \end{split}\tag{2}
\]
The remaining independent fields have the usual equal-time canonical
commutators.  Equivalently, one may retain \(V^0,\Pi_0\) and use the
Dirac brackets of the two second-class constraints.

To identify the interaction picture, write
\(\mathcal H(z)=M^2z+\mathcal H_I(z)\).  Let
\(\phi,v^\mu,\boldsymbol\pi_v\) be the corresponding free complex-scalar
and Proca fields, with
\[
 \rho_I=ig\bigl(\dot\phi^\dagger\phi-\dot\phi\,\phi^\dagger\bigr),
 \qquad
 \boldsymbol\nabla\cdot\boldsymbol\pi_v=-m^2v_0 .
\]
Subtracting the free Hamiltonian from (2) yields
\[
 \boxed{\begin{aligned}
 H_I(t)=\int d^3x\Bigl\{&
 \mathcal H_I(\phi^\dagger\phi)
 +ig\,v_\mu\!\left[
 \phi^\dagger\partial^\mu\phi
 -(\partial^\mu\phi^\dagger)\phi\right]\\
 &+g^2\mathbf v^2\phi^\dagger\phi
 +\frac{\rho_I^2}{2m^2}\Bigr\}.
 \end{aligned}}\tag{3}
\]
The last term is the local contact interaction forced by eliminating the
auxiliary time component.  It would be missed by setting
\(H_I=-\int d^3x\,\mathcal L_I\).

\WeinbergSolution{5}
Varying the action with respect to the metric, or equivalently applying
the Belinfante construction, gives
\[
 \boxed{\begin{aligned}
 \Theta^{\mu\nu}={}&
 (D^\mu\Phi)^\dagger D^\nu\Phi
 +(D^\nu\Phi)^\dagger D^\mu\Phi\\
 &+F^{\mu\rho}F^\nu{}_\rho
 +m^2V^\mu V^\nu+\eta^{\mu\nu}\mathcal L .
 \end{aligned}}\tag{4}
\]
It is symmetric by inspection.  The vector equation,
\[
 \partial_\rho F^{\rho\mu}-m^2V^\mu
 =ig\left[\Phi^\dagger D^\mu\Phi
 -(D^\mu\Phi)^\dagger\Phi\right],
\]
together with the scalar equation and its adjoint, verifies
\(\partial_\mu\Theta^{\mu\nu}=0\).  In particular, \(\Theta^{00}\)
contains the positive electric, magnetic, scalar-gradient, and Proca
mass terms when the constraints are imposed.

For the rigid phase symmetry, Noether's formula gives
\[
 \boxed{\;
 J^\mu=i\left[(D^\mu\Phi)^\dagger\Phi
                  -\Phi^\dagger D^\mu\Phi\right].\;}
\tag{5}
\]
The vector is inert, so it supplies no additional term.  The scalar
equation
\[
 D_\mu D^\mu\Phi-\mathcal H'(\Phi^\dagger\Phi)\Phi=0
\]
and its adjoint make the divergence of (5) vanish.  The source on the
right side of the vector equation is \(-gJ^\mu\), so its divergence also
implies current conservation because the divergence of
\(\partial_\rho F^{\rho\mu}\) vanishes identically and the Proca equation
implies the corresponding subsidiary condition.

\WeinbergSolution{6}
Write the Poisson bracket as \(\{\, ,\,\}\), put
\(C_{MN}=\{\chi_M,\chi_N\}\), and denote its inverse by \(C^{MN}\).
The two identities needed in expanding a nested Dirac bracket are
\[
 \{A,C^{MN}\}
 =-C^{MP}\{A,C_{PQ}\}C^{QN},
\tag{6}
\]
and, by the Poisson Jacobi identity,
\[
 \{A,C_{PQ}\}
 =\{\{A,\chi_P\},\chi_Q\}
  +\{\chi_P,\{A,\chi_Q\}\}.
\tag{7}
\]
Now substitute
\[
 \{A,B\}_{\rm D}
 =\{A,B\}-\{A,\chi_M\}C^{MN}\{\chi_N,B\}
\]
twice into
\(\sum_{\rm cyc}\{A,\{B,C\}_{\rm D}\}_{\rm D}\).
The terms with no \(C^{MN}\) vanish by the ordinary Jacobi identity.
Use (6) on every term in which the outer bracket differentiates an
inverse matrix, and then use (7).  Terms in which the outer bracket hits
\(A,B,\) or \(C\) cancel cyclically; the remaining terms pair with
opposite signs after interchanging the dummy constraint indices and
using \(C^{MN}=-C^{NM}\).  Hence
\[
 \sum_{\rm cyc}\{A,\{B,C\}_{\rm D}\}_{\rm D}=0 .
\]
This proof uses only the invertibility of the second-class constraint
matrix.  As a useful local check, in Maskawa--Nakajima coordinates the
Dirac bracket is the ordinary Poisson bracket of the unconstrained
pairs, Eq.~\eqref{eq:7.6.25}, for which Jacobi is immediate.

\WeinbergSolution{7}
Let \(\chi'_A\) be another regular set of functions defining the same
constraint submanifold.  The implicit-function theorem implies locally
\[
 \chi'_A=M_A{}^B(z)\chi_B+O(\chi^2),
\]
where \(M\) is nonsingular on the submanifold.  Therefore, with weak
equality meaning equality after the constraints are imposed,
\[
 \{F,\chi'_A\}\simeq M_A{}^B\{F,\chi_B\},
\qquad
 C'_{AB}\simeq M_A{}^C C_{CD}M_B{}^D .
\]
The inverse matrix consequently obeys
\[
 (C'^{-1})^{AB}\simeq
 (M^{-1})_C{}^A C^{CD}(M^{-1})_D{}^B .
\]
Substitution into the correction term gives
\[
 \begin{split}
 &\{F,\chi'_A\}(C'^{-1})^{AB}\{\chi'_B,G\}\\
 &\hspace{2cm}\simeq
 \{F,\chi_C\}C^{CD}\{\chi_D,G\}.
 \end{split}
\]
Thus
\[
 \boxed{\quad \{F,G\}'_{\rm D}\simeq\{F,G\}_{\rm D}.\quad}
\]
The qualifier is essential: a Dirac bracket is the Poisson structure
induced on the constraint submanifold, and two extensions away from that
submanifold may differ by terms proportional to the constraints.  All
brackets of physical observables, and hence all commutators obtained
after imposing the constraints, are independent of the defining
functions.
\end{enumerate}
