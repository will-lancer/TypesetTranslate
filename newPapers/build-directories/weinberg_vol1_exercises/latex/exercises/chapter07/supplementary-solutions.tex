\SupplementarySolution{1}{Canonical Transformations and Noether Charges}
\begin{enumerate}
\item
The momentum is
\[
 \Pi=\frac{\partial\mathcal L}{\partial\dot\phi}
     =Z(\phi)\dot\phi ,
 \qquad \dot\phi=\frac{\Pi}{Z(\phi)} .
\]
The Legendre transform must be carried out before any perturbative
expansion:
\[
 \mathcal H=\Pi\dot\phi-\mathcal L
 =\frac{\Pi^2}{2Z(\phi)}
 +\frac12Z(\phi)(\boldsymbol\nabla\phi)^2+U(\phi).
\]
Thus a field-dependent coefficient of \(\dot\phi^2\) is inverted in the
momentum part of the Hamiltonian but not in the gradient part.  If
\(Z=1+h\), the exact inverse \(1/(1+h)\), rather than merely \(1-h\),
generates the full tower of momentum interactions.
\item
Let \(L'=L+dF(\phi)/dt\).  Then
\[
 \Pi'=\frac{\partial L'}{\partial\dot\phi}
 =\Pi+F'(\phi).
\]
The coordinates are unchanged and the momenta are translated:
\[
 Q'=\phi,\qquad \Pi'=\Pi+F'(Q').
\]
This is canonical because
\(\Pi'dQ'=\Pi\,dQ+dF\), so the symplectic two-form is unchanged.
The transformed Hamiltonian is
\[
 H'(Q',\Pi')=H\bigl(Q',\Pi'-F'(Q')\bigr).
\]
Hamilton's equations for \(Q'\) and \(\Pi'\), after substituting this
relation, are exactly the original equations for \(\phi\) and \(\Pi\).
The action changes only by endpoint values of \(F\).
\item
For
\(\delta\phi_1=-\epsilon\phi_2\) and
\(\delta\phi_2=\epsilon\phi_1\), Noether's formula gives
\[
 j^\mu=\phi_2\partial^\mu\phi_1
       -\phi_1\partial^\mu\phi_2 .
\]
The equations
\(\Box\phi_a-m^2\phi_a-2U'(\phi_1^2+\phi_2^2)\phi_a=0\)
make \(\partial_\mu j^\mu=0\).  Since
\(\pi_a=\dot\phi_a\),
\[
 Q=\int d^3x\,j^0
 =\int d^3x\,(\pi_2\phi_1-\pi_1\phi_2).
\]
The canonical algebra yields
\([Q,\phi_1]=i\phi_2\) and
\([Q,\phi_2]=-i\phi_1\).  Therefore
\(\delta\phi_a=i\epsilon[Q,\phi_a]\) reproduces the stated rotation.
\end{enumerate}

\SupplementarySolution{2}{Improved Stress Tensors and Derivative Couplings}
\begin{enumerate}
\item
Set
\[
 I^{\mu\nu}
 =c(\partial^\mu\partial^\nu-\eta^{\mu\nu}\Box)\phi^2 .
\]
Commutativity of partial derivatives gives the off-shell identity
\[
 \partial_\mu I^{\mu\nu}
 =c(\Box\partial^\nu-\partial^\nu\Box)\phi^2=0.
\]
Thus \(\Theta^{\mu\nu}+I^{\mu\nu}\) is conserved whenever
\(\Theta^{\mu\nu}\) is.  Moreover,
\(\int d^3x\,I^{0\nu}\) is a spatial surface term after separating the
time and spatial derivatives.  It vanishes for fields and first
derivatives that decrease sufficiently rapidly at spatial infinity.
The improvement changes local densities but neither the conserved
four-momentum nor the translation generators.
\item
For
\(\mathcal L_{\rm D}=-\bar\psi(\gamma^\rho\partial_\rho+m)\psi\),
the Lorentz generator is
\[
 \mathcal J^{\mu\nu}=-\frac{i}{4}[\gamma^\mu,\gamma^\nu].
\]
The canonical tensor is not symmetric because the field carries this
spin current.  Inserting the corresponding current into the Belinfante
formula~\eqref{eq:7.4.11}, and using the Dirac equation plus its adjoint,
gives the equivalent Hermitian expression
\[
 \Theta^{\mu\nu}
 =-\frac14\bar\psi\left(
 \gamma^\mu\overleftrightarrow{\partial^\nu}
 +\gamma^\nu\overleftrightarrow{\partial^\mu}\right)\psi
 +\eta^{\mu\nu}\mathcal L_{\rm D}.
\]
It is symmetric by inspection and conserved on shell.  The improvement
is a spatial divergence in the integrated charges, so the four-momentum
is unchanged.
\item
The conjugate momentum is
\[
 \Pi=\dot\phi+gJ^0,\qquad \dot\phi=\Pi-gJ^0 .
\]
The exact Legendre transform gives
\[
 \mathcal H
 =\frac12(\Pi-gJ^0)^2
 +\frac12(\boldsymbol\nabla\phi)^2
 -g\mathbf J\cdot\boldsymbol\nabla\phi .
\]
Taking the free momentum in the interaction picture to be
\(\Pi=\dot\phi\), the interaction density is
\[
 \mathcal H_I
 =-gJ^0\dot\phi-g\mathbf J\cdot\boldsymbol\nabla\phi
 +\frac12g^2(J^0)^2 .
\]
The last term comes from completing the square in the velocity.  Since
it is absent from \(-\mathcal L_I\), a derivative interaction cannot in
general be converted to an interaction Hamiltonian by changing only an
overall sign.
\end{enumerate}

\SupplementarySolution{3}{Complex-Scalar Phase Current and Charge as a Canonical Generator}
\begin{enumerate}
\item
Every Lorentz index in
\(-\partial_\mu\phi^\dagger\partial^\mu\phi
-V(\phi^\dagger\phi)\) is contracted, and both fields transform as
scalars.  Independent variation gives
\[
 \Box\phi-V'(\phi^\dagger\phi)\phi=0,\qquad
 \Box\phi^\dagger-V'(\phi^\dagger\phi)\phi^\dagger=0.
\]
\item
The momenta and Hamiltonian density are
\[
 \pi=\dot\phi^\dagger,\qquad \pi^\dagger=\dot\phi,\qquad
 \mathcal H=\pi\pi^\dagger+\nabla\phi^\dagger\cdot\nabla\phi
 +V(\phi^\dagger\phi).
\]
Hamilton's equations reproduce the two equations in part 1.
\item
Under
\(\delta\phi=i\epsilon\phi\) and
\(\delta\phi^\dagger=-i\epsilon\phi^\dagger\), the chapter's Noether
normalization gives
\[
 j^\mu=i\left[(\partial^\mu\phi^\dagger)\phi
              -\phi^\dagger\partial^\mu\phi\right].
\]
Its divergence is
\[
 \partial_\mu j^\mu
 =i\left[(\Box\phi^\dagger)\phi-\phi^\dagger\Box\phi\right].
\]
The field equations are
\[
 \Box\phi-V'(\phi^\dagger\phi)\phi=0,\qquad
 \Box\phi^\dagger-V'(\phi^\dagger\phi)\phi^\dagger=0 .
\]
Substitution makes the two remaining terms identical with opposite
signs, proving \(\partial_\mu j^\mu=0\).
\item
The canonical momenta are
\[
 \pi=\dot\phi^\dagger,\qquad \pi^\dagger=\dot\phi .
\]
Consequently the time component of the current in S.7.7 is
\[
 j^0=i(\phi^\dagger\pi^\dagger-\pi\phi),\qquad
 Q=\int d^3x\,j^0 .
\]
Using
\([\,\phi(\mathbf x),\pi(\mathbf y)\,]
=i\delta^3(\mathbf x-\mathbf y)\) and the analogous daggered relation
gives
\[
 [Q,\phi]=-\phi,\qquad [Q,\phi^\dagger]=+\phi^\dagger .
\]
Weinberg's convention
\(\delta\phi=i\epsilon\phi\) is therefore
\(\delta\phi=-i\epsilon[Q,\phi]\); the conjugate field acquires the
opposite phase.  The sign is tied to the chosen definition of \(j^\mu\).
\end{enumerate}

\SupplementarySolution{4}{Energy Density of an Interacting Complex Scalar and Stress-Tensor Conservation Component by Component}
\begin{enumerate}
\item
Metric variation, or the translation Noether procedure, gives
\[
 \Theta^{\mu\nu}
 =(\partial^\mu\phi^\dagger)\partial^\nu\phi
 +(\partial^\nu\phi^\dagger)\partial^\mu\phi
 +\eta^{\mu\nu}\mathcal L .
\]
Because \(\partial^0=-\partial_0\) and \(\eta^{00}=-1\),
\[
 \Theta^{00}
 =\dot\phi^\dagger\dot\phi
  +\boldsymbol\nabla\phi^\dagger\cdot\boldsymbol\nabla\phi
  +V(\phi^\dagger\phi).
\]
The Legendre transform with
\(\pi=\dot\phi^\dagger\) and \(\pi^\dagger=\dot\phi\) gives precisely
the same density.  Thus the energy from the symmetric tensor equals the
canonical Hamiltonian, including the interacting potential.
\item
Raising and lowering the spatial index does not change its sign, and
\[
 P_i=\int d^3x\,\Theta^0{}_i
 =-\int d^3x\left(
 \pi\,\partial_i\phi+\pi^\dagger\partial_i\phi^\dagger\right).
\]
Only the momentum conjugate to a displayed field contributes to its
commutator.  The equal-time algebra therefore gives
\[
 [P_i,\phi(\mathbf x)]=i\partial_i\phi(\mathbf x),\qquad
 [P_i,\phi^\dagger(\mathbf x)]
 =i\partial_i\phi^\dagger(\mathbf x).
\]
By the Leibniz rule the same relation holds for every local polynomial
without explicit position dependence.  This confirms directly that
the integrated stress tensor has the normalization required of the
spatial-translation generator.
\item
Differentiate the tensor of S.7.9:
\[
 \begin{split}
 \partial_\mu\Theta^{\mu\nu}
 ={}&(\Box\phi^\dagger)\partial^\nu\phi
 +(\partial^\nu\phi^\dagger)\Box\phi\\
 &-\partial^\nu\phi^\dagger
   V'(\phi^\dagger\phi)\phi
 -\phi^\dagger V'(\phi^\dagger\phi)\partial^\nu\phi
 \end{split}
\]
after the second-derivative kinetic terms have canceled against
\(\partial^\nu[-\partial_\rho\phi^\dagger\partial^\rho\phi]\).
Using
\(\Box\phi=V'\phi\) and
\(\Box\phi^\dagger=V'\phi^\dagger\), with the signs inherited from the
mostly-plus Lagrangian, cancels the remaining potential derivative
terms.  No special choice of \(\nu\) was made, so the cancellation holds
for energy conservation and for each of the three momentum components.
\end{enumerate}

\SupplementarySolution{5}{Current Conservation with a Covariant Derivative}
The rigid phase current is
\[
 j^\mu=i\left[(D^\mu\phi)^\dagger\phi
              -\phi^\dagger D^\mu\phi\right].
\]
It is gauge invariant, so its covariant and ordinary divergences agree.
Using the product rule in gauge-covariant form,
\[
 \partial_\mu j^\mu
 =i\left[(D_\mu D^\mu\phi)^\dagger\phi
         -\phi^\dagger D_\mu D^\mu\phi\right].
\]
The scalar equation and its adjoint set both second derivatives equal
to \(V'(\phi^\dagger\phi)\) times the corresponding field.  The two
terms cancel.  The vector potential therefore changes the current's
local expression but does not turn the global conservation equation
into a merely covariant one.

\SupplementarySolution{6}{Primary and Secondary Constraints in Hamiltonian Dynamics}
\begin{enumerate}
\item
The velocity Hessian is
\[
 W_{ab}=\frac{\partial^2L}{\partial\dot q_a\partial\dot q_b}
 =\begin{pmatrix}1&0\\0&0\end{pmatrix},
\]
so it cannot be inverted.  The momenta are
\[
 p_1=\dot q_1-q_2,\qquad p_2=0,
\]
and \(\chi_1=p_2\simeq0\) is primary.  Solving only the first relation
gives \(\dot q_1=p_1+q_2\).  The canonical and total Hamiltonians are
\[
 H_c=\frac12p_1^2+p_1q_2+\frac12\mu^2q_2^2
     +\frac12\omega^2q_1^2,\qquad
 H_T=H_c+u\,p_2 .
\]
The arbitrary multiplier \(u(t)\) represents the velocity not fixed by
the singular Legendre map.
\item
Preservation of the primary constraint gives
\[
 \dot p_2=\{p_2,H_T\}
 =-(p_1+\mu^2q_2)\simeq0,
\]
so \(\chi_2=p_1+\mu^2q_2\simeq0\).  Their constraint matrix has
\[
 \{\chi_1,\chi_2\}=\{p_2,p_1+\mu^2q_2\}=-\mu^2 ,
\]
which is invertible for \(\mu\ne0\); the pair is second class.  Finally,
\[
 \dot\chi_2=\{\chi_2,H_T\}
 =-\omega^2q_1+\mu^2u\simeq0
\]
fixes \(u=\omega^2q_1/\mu^2\).  No new constraint appears.  Thus the
Dirac--Bergmann algorithm both discovers the missing restriction and
determines the previously arbitrary velocity.
\item
For \(\chi_1=q_2,\chi_2=p_2\),
\[
 C_{AB}=\begin{pmatrix}0&1\\-1&0\end{pmatrix},
 \qquad
 C^{AB}=\begin{pmatrix}0&-1\\1&0\end{pmatrix}.
\]
The variables \(q_1,p_1\) have zero Poisson brackets with both
constraints, so
\[
 \{q_1,p_1\}_{\rm D}=1,\qquad
 \{q_1,q_1\}_{\rm D}=\{p_1,p_1\}_{\rm D}=0.
\]
Direct substitution also gives
\[
 \{q_2,F\}_{\rm D}=\{p_2,F\}_{\rm D}=0
\]
for every \(F\), including
\(\{q_2,p_2\}_{\rm D}=0\).  Hence the constrained pair can be imposed
strongly, leaving precisely the ordinary canonical algebra of
\((q_1,p_1)\).
\end{enumerate}

\SupplementarySolution{7}{Dirac Brackets and the Reduced Poisson Geometry}
\begin{enumerate}
\item
By definition,
\[
 \{\chi_A,F\}_{\rm D}
 =\{\chi_A,F\}
 -\{\chi_A,\chi_B\}C^{BC}\{\chi_C,F\}.
\]
Since
\(\{\chi_A,\chi_B\}=C_{AB}\) and
\(C_{AB}C^{BC}=\delta_A{}^C\), the second term is exactly
\(-\{\chi_A,F\}\).  Thus
\[
 \boxed{\{\chi_A,F\}_{\rm D}=0}
\]
without first imposing the constraints.  This strong identity is why
the second-class equations may be used before or after computing a
Dirac bracket, and why quantization by replacing Dirac brackets with
commutators is consistent with setting the constraint operators to
zero.
\item
Let \(P^{IJ}\) be the Poisson tensor and
\(\chi_{A,I}=\partial_I\chi_A\).  The Dirac tensor is
\[
 P_{\rm D}^{IJ}
 =P^{IJ}-P^{IK}\chi_{A,K}C^{AB}\chi_{B,L}P^{LJ}.
\]
Contracting with a constraint gradient gives
\[
 P_{\rm D}^{IJ}\chi_{C,J}
 =P^{IJ}\chi_{C,J}
 -P^{IK}\chi_{A,K}C^{AB}C_{BC}=0.
\]
It therefore removes every component normal to the constraint surface.
For functions whose Hamiltonian vector fields are tangent to the
surface, the subtracted normal component vanishes, and \(P_{\rm D}\)
acts as the inverse of the pullback symplectic form.  This identifies
the Dirac bracket with the intrinsic Poisson bracket of the reduced
phase space.
\item
Let
\(\chi'_1=e^f\chi_1\) and \(\chi'_2=e^{-f}\chi_2\).  Product rules give
\[
 \{F,\chi'_1\}=e^f\{F,\chi_1\}+e^f\chi_1\{F,f\},
\]
and the analogous expression for \(\chi'_2\).  Hence, on the common
surface,
\[
 \{F,\chi'_A\}\simeq M_A{}^B\{F,\chi_B\},
\qquad M=\operatorname{diag}(e^f,e^{-f}).
\]
All derivatives of \(f\) in \(C'_{AB}\) multiply at least one
constraint, so \(C'\simeq MCM^T\) and
\((C')^{-1}\simeq(M^T)^{-1}C^{-1}M^{-1}\).  The matrices \(M\) cancel in
the Dirac correction.  Therefore
\(\{F,G\}'_{\rm D}\simeq\{F,G\}_{\rm D}\), including the derivative
terms that disappear only after restriction.
\end{enumerate}

\SupplementarySolution{8}{Equation-of-Motion and Boundary Operators}
\begin{enumerate}
\item
For an action depending on \(\phi\) and its first derivatives,
\[
 \delta S=\int d^4x\left[
 \frac{\partial\mathcal L}{\partial\phi}\delta\phi
 +\frac{\partial\mathcal L}{\partial(\partial_\mu\phi)}
  \partial_\mu\delta\phi\right].
\]
With \(\delta\phi=\epsilon F\), integration by parts yields
\[
 \delta S=\epsilon\int d^4x\,
 \frac{\delta S}{\delta\phi(x)}F(x)
 +\epsilon\int d^4x\,\partial_\mu\left[
 \frac{\partial\mathcal L}{\partial(\partial_\mu\phi)}F\right].
\]
The same statement holds for derivative-dependent \(F\); further
integrations by parts only enlarge the displayed boundary current.
Thus the bulk shift is proportional to the Euler--Lagrange functional,
which characterizes a redundant coupling.
\item
Take
\[
 \mathcal L_0=-\frac12(\partial\phi)^2-\frac12m^2\phi^2
              -\frac{\lambda}{4!}\phi^4 ,
\]
for which
\[
 \frac{\delta S_0}{\delta\phi}
 =\Box\phi-m^2\phi-\frac{\lambda}{6}\phi^3 .
\]
Suppose the interaction contains
\(+\epsilon\phi^3(\Box-m^2)\phi\).  Under
\(\phi\mapsto\phi-\epsilon\phi^3\),
\[
 \delta\mathcal L_0
 =-\epsilon\phi^3(\Box-m^2)\phi
  +\frac{\epsilon\lambda}{6}\phi^6
\]
up to a divergence.  The equation-of-motion operator cancels, while a
local \(\phi^6\) interaction is induced.  Terms of order
\(\epsilon^2\) lie beyond this first-order change of variables and must
be retained only in a calculation at the corresponding higher order.
\item
A total divergence,
\(\Delta\mathcal L=\partial_\mu K^\mu\), changes the action only by a
boundary integral.  With fixed boundary data it has no bulk effect and
requires no change of variables.  By contrast,
\[
 \Delta\mathcal L=F(\phi,\partial\phi)
 \frac{\delta S_0}{\delta\phi}
\]
is generally not itself a divergence off shell.  It vanishes only after
using the leading equation of motion.  The local redefinition
\(\phi\mapsto\phi-\epsilon F\) removes it to first order while also
generating other operators.  Equating the two mechanisms without those
induced terms would incorrectly treat an on-shell relation as an
off-shell identity.
\end{enumerate}

\SupplementarySolution{9}{Field Redefinitions, External Poles, and Order Bookkeeping}
\begin{enumerate}
\item
Seek
\[
 \phi=\phi'+\epsilon A\phi'^2+\epsilon^2B\phi'^3+O(\epsilon^3).
\]
Substitution into
\(\phi'=\phi+\epsilon\phi^2+\epsilon^2a\phi^3\) gives
\[
 0=\epsilon(A+1)\phi'^2
 +\epsilon^2(B+2A+a)\phi'^3.
\]
Thus \(A=-1\), \(B=2-a\), and
\[
 \phi=\phi'-\epsilon\phi'^2
       +\epsilon^2(2-a)\phi'^3+O(\epsilon^3).
\]
Local admissibility requires a nonzero functional Jacobian.  Pointwise,
\(\partial\phi'/\partial\phi
=1+2\epsilon\phi+3a\epsilon^2\phi^2\ne0\) in the field region under
consideration.  Perturbatively this holds near the identity.
\item
Near the physical one-particle pole, the Fourier transform of the new
two-point function has the spectral form
\[
 \widetilde G_{\phi'}(p)
 =\frac{Z'}{p^2+m_{\rm phys}^2-i0}
  +\text{terms regular at the pole}.
\]
The composite term \(\epsilon\phi^2\) changes the overlap
\(\bra{\Omega}\phi'(0)\ket{p}\), and hence changes \(Z'\), but it cannot
move the pole because the mass is a property of the Hamiltonian
spectrum.  Amputation multiplies by the inverse pole and divides each
external leg by \(\sqrt{Z'}\).  The overlap change then cancels, leaving
the same on-shell scattering observable provided \(Z'\ne0\).  Off-shell
correlators need not agree.
\item
For
\(\phi\mapsto\phi+\epsilon F\), Taylor expansion of the action gives
\[
 S[\phi+\epsilon F]
 =S[\phi]+\epsilon\int F\frac{\delta S}{\delta\phi}
 +\frac{\epsilon^2}{2}\int d^4x\,d^4y\,
 F(x)\frac{\delta^2S}{\delta\phi(x)\delta\phi(y)}F(y)+\cdots .
\]
The linear term can cancel a chosen dimension-six equation-of-motion
operator.  The quadratic term contains two powers of the suppressed
coefficient and therefore contributes at the dimension-eight order in
the same power counting.  Omitting it is consistent in a calculation
truncated at first order in \(\epsilon\), but not in a second-order
calculation, where it is required for equivalence.
\end{enumerate}

\SupplementarySolution{10}{Geometry, Hamiltonian, and Current of the \(O(N)\) Sigma Model}
\begin{enumerate}
\item
Write
\[
 n^A=(\pi^a,\sigma),\qquad
 \sigma=\sqrt{1-\boldsymbol\pi^2}.
\]
Then
\[
 \partial_\mu\sigma
 =-\frac{\pi_a\partial_\mu\pi^a}
         {\sqrt{1-\boldsymbol\pi^2}},
\]
and hence
\[
 \partial_\mu n^A\partial^\mu n^A
 =g_{ab}(\pi)\partial_\mu\pi^a\partial^\mu\pi^b,
\qquad
 g_{ab}=\delta_{ab}
 +\frac{\pi_a\pi_b}{1-\boldsymbol\pi^2}.
\]
The apparent interaction is therefore the metric induced on the unit
sphere.  The coordinate patch fails at
\(\boldsymbol\pi^2=1\), where the eliminated component \(\sigma\)
vanishes, even though the sphere itself is regular.
\item
The rank-one inverse is
\[
 g^{ab}=\delta^{ab}-\pi^a\pi^b,
\qquad g_{ac}g^{cb}=\delta_a{}^b .
\]
For
\[
 \mathcal L=-\frac1{2e^2}g_{ab}
 \partial_\mu\pi^a\partial^\mu\pi^b,
\]
the canonical momentum is
\[
 p_a=\frac1{e^2}g_{ab}\dot\pi^b .
\]
Solving for the velocity and performing the Legendre transform gives
\[
 \mathcal H
 =\frac{e^2}{2}p_ag^{ab}p_b
 +\frac1{2e^2}g_{ab}
 \boldsymbol\nabla\pi^a\cdot\boldsymbol\nabla\pi^b .
\]
The inverse target metric necessarily appears in the momentum term,
just as in the general nonlinear sigma-model Hamiltonian.
\item
For a rotation in the \(AB\)-plane,
\(\delta n^A=\epsilon n^B\) and
\(\delta n^B=-\epsilon n^A\).  Noether's theorem gives, up to the
orientation convention,
\[
 j^\mu_{AB}=\frac1{e^2}
 \left(n^A\partial^\mu n^B-n^B\partial^\mu n^A\right).
\]
The constrained equation has the form
\[
 \Box n^A=\lambda(x)n^A,
\]
because the force normal to the sphere is supplied by a Lagrange
multiplier.  Therefore
\[
 \partial_\mu j^\mu_{AB}
 =\frac1{e^2}(n^A\Box n^B-n^B\Box n^A)=0.
\]
The unknown multiplier cancels, so conservation follows without solving
for it.
\end{enumerate}

\SupplementarySolution{11}{Target-Space Covariance, Curvature, and Stereographic Coordinates}
\begin{enumerate}
\item
Let
\(\pi^a=\pi^a(\xi)\).  Then
\[
 \partial_\mu\pi^a
 =\frac{\partial\pi^a}{\partial\xi^\alpha}
  \partial_\mu\xi^\alpha .
\]
Substitution into the kinetic term gives the transformed metric
\[
 g'_{\alpha\beta}(\xi)
 =\frac{\partial\pi^a}{\partial\xi^\alpha}
  \frac{\partial\pi^b}{\partial\xi^\beta}g_{ab}(\pi(\xi)).
\]
Thus
\[
 g_{ab}\partial_\mu\pi^a\partial^\mu\pi^b
 =g'_{\alpha\beta}
  \partial_\mu\xi^\alpha\partial^\mu\xi^\beta.
\]
The spacetime Lagrangian is unchanged: it is a scalar under
reparametrizations of the target.  Only the coordinate representation
of the field-space metric changes.
\item
In angular coordinates the unit two-sphere has
\[
 ds^2=d\theta^2+\sin^2\theta\,d\varphi^2 .
\]
The nonzero Christoffel symbols include
\(\Gamma^\theta_{\varphi\varphi}=-\sin\theta\cos\theta\) and
\(\Gamma^\varphi_{\theta\varphi}=\cot\theta\).  They give
\[
 R_{\theta\varphi\theta\varphi}=\sin^2\theta,
\]
which is nonzero away from the coordinate poles.  Curvature is a tensor,
so an invertible field redefinition cannot make it vanish.  A globally
constant kinetic metric would have zero Riemann tensor.  One may flatten
the metric at a single field value by normal coordinates, but not over
an open region of the sphere.
\item
With \(r^2=u^2+v^2\), choose
\[
 n=\frac1{1+r^2}(2u,2v,1-r^2).
\]
Direct differentiation gives
\[
 dn\cdot dn=\frac{4(du^2+dv^2)}{(1+r^2)^2}.
\]
For the mostly-plus sigma-model density, the canonical momenta are
\[
 p_u=\frac{4\dot u}{e^2(1+r^2)^2},\qquad
 p_v=\frac{4\dot v}{e^2(1+r^2)^2}.
\]
The exact Hamiltonian density is
\[
 \mathcal H
 =\frac{e^2(1+r^2)^2}{8}(p_u^2+p_v^2)
 +\frac{2}{e^2(1+r^2)^2}
 \left[(\boldsymbol\nabla u)^2+(\boldsymbol\nabla v)^2\right].
\]
The stereographic conformal factor is inverted only in the momentum
term, as required by the Legendre transform.
\end{enumerate}
\end{enumerate}
