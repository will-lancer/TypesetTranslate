\SupplementaryExercise{1}{Canonical Transformations and Noether Charges}
  {(Inspired by David Tong, Lectures on Quantum Field Theory, Sections 1.1 and 1.4 | David Tong, Lectures on Quantum Field Theory, Sections 1.3.1 and 1.3.4.)}
  {tong-qft-classical-fields-ch7-curated-parent-1}
\begin{enumerate}
\item
For
\(\mathcal L=\frac12Z(\phi)\dot\phi^2
-\frac12Z(\phi)(\boldsymbol\nabla\phi)^2-U(\phi)\),
derive the conjugate momentum and Hamiltonian density without expanding
in \(Z(\phi)-1\).
\item
Add \(\partial_tF(\phi)\) to a scalar Lagrangian.  Determine how the
canonical momentum changes, exhibit the corresponding canonical
transformation, and prove that Hamilton's equations are unchanged.
\item
For two equal-mass real scalars with an interaction
\(U(\phi_1^2+\phi_2^2)\), derive the \(SO(2)\) Noether current and verify
that its charge generates both infinitesimal field rotations.
\end{enumerate}

\SupplementaryExercise{2}{Improved Stress Tensors and Derivative Couplings}
  {(Inspired by David Tong, Lectures on Quantum Field Theory, Sections 1.3.2--1.3.3 | David Tong, Lectures on Quantum Field Theory, Section 1.4.)}
  {tong-qft-classical-fields-ch7-curated-parent-2}
\begin{enumerate}
\item
Show that adding
\((\partial^\mu\partial^\nu-\eta^{\mu\nu}\Box)\phi^2\)
to a scalar energy--momentum tensor preserves conservation and total
four-momentum under standard falloff conditions.
\item
Starting from the canonical tensor of a free Dirac field, calculate the
spin current and show that the Belinfante improvement produces a
symmetric conserved energy--momentum tensor.
\item
For
\(\mathcal L=\frac12\dot\phi^2-\frac12(\boldsymbol\nabla\phi)^2
+gJ^0\dot\phi+g\mathbf J\cdot\boldsymbol\nabla\phi\),
perform the Legendre transform and identify the quadratic local term
that prevents \(H_I=-L_I\).
\end{enumerate}

\SupplementaryExercise{3}{Complex-Scalar Phase Current and Charge as a Canonical Generator}
  {(Adapted from Hong Liu and MIT OpenCourseWare, 8.323 Assignment 1, Problem 3.)}
  {mit-823-s23-ps1-ch7-curated-parent-1}
\begin{enumerate}
\item
For
\(\mathcal L=-\partial_\mu\phi^\dagger\partial^\mu\phi
-V(\phi^\dagger\phi)\),
derive the rigid \(U(1)\) current and verify its conservation directly
from the two Euler--Lagrange equations.
\item
Express the complex-scalar \(U(1)\) charge in terms of the two canonical
momenta, then use equal-time commutators to show that it generates the
opposite phases of \(\phi\) and \(\phi^\dagger\).
\end{enumerate}

\SupplementaryExercise{4}{Energy Density of an Interacting Complex Scalar and Stress-Tensor Conservation Component by Component}
  {(Adapted from Hong Liu and MIT OpenCourseWare, 8.323 Assignment 1, Problem 4.)}
  {mit-823-s23-ps1-ch7-curated-parent-2}
\begin{enumerate}
\item
Derive the symmetric stress tensor for a complex scalar with arbitrary
\(V(\phi^\dagger\phi)\), and show that its \(00\) component equals the
canonical Hamiltonian density in the mostly-plus convention.
\item
Integrate the \(0i\) component of the complex-scalar stress tensor and
prove, using the canonical algebra, that the resulting operator
generates spatial translations of both fields.
\item
Calculate \(\partial_\mu\Theta^{\mu\nu}\) for the interacting
complex-scalar theory, keeping every potential derivative explicit,
and demonstrate the cancellation separately for each free index
\(\nu\).
\end{enumerate}

\SupplementaryExercise{5}{Current Conservation with a Covariant Derivative}
  {(Inspired by Hong Liu and MIT OpenCourseWare, 8.323 Assignment 1, Problems 3--4.)}
  {mit-823-s23-ps1-ch7-curated-parent-3}
Replace ordinary derivatives by \(D_\mu=\partial_\mu-igV_\mu\) in the
complex-scalar theory.  Derive the gauge-covariant phase current and

\SupplementaryExercise{6}{Primary and Secondary Constraints in Hamiltonian Dynamics}
  {(Inspired by Brian P. Dolan, Constrained Dynamics in the Hamiltonian Formalism, Sections 2 and 4 | Brian P. Dolan, Constrained Dynamics in the Hamiltonian Formalism, Sections 4 and 6 | Brian P. Dolan, Constrained Dynamics in the Hamiltonian Formalism, Sections 3--4.)}
  {dolan-constrained-dynamics-2112.12216-curated-parent-1}
\begin{enumerate}
\item
For
\(L=\frac12(\dot q_1-q_2)^2-\frac12\mu^2q_2^2
-\frac12\omega^2q_1^2\), compute the velocity Hessian, find the primary
constraint, and construct the total Hamiltonian with its multiplier.
\item
Continue S.7.6(a) by demanding preservation of the primary constraint.
Find the secondary constraint, classify the complete pair, and
determine the multiplier when consistency is imposed again.
\item
On a two-pair phase space impose \(\chi_1=q_2\) and \(\chi_2=p_2\).
Compute every fundamental Dirac bracket and show explicitly that only
\((q_1,p_1)\) remains dynamical.
\end{enumerate}

\SupplementaryExercise{7}{Dirac Brackets and the Reduced Poisson Geometry}
  {(Inspired by Brian P. Dolan, Constrained Dynamics in the Hamiltonian Formalism, Section 4 | Brian P. Dolan, Constrained Dynamics in the Hamiltonian Formalism, Sections 3--4 | Brian P. Dolan, Constrained Dynamics in the Hamiltonian Formalism, Sections 4 and 6.)}
  {dolan-constrained-dynamics-2112.12216-curated-parent-2}
\begin{enumerate}
\item
For arbitrary second-class constraints, prove directly from the inverse
constraint matrix that \(\{\chi_A,F\}_{\rm D}=0\) for every phase-space
function \(F\).
\item
Write the Dirac bracket as a projected Poisson tensor.  Show that the
projection annihilates every constraint gradient and restricts to the
ordinary symplectic inverse on tangent vectors.
\item
Replace constraints \((\chi_1,\chi_2)\) by
\((e^{f(z)}\chi_1,e^{-f(z)}\chi_2)\).  Prove that their Dirac brackets
agree on the common constraint surface, including all derivative terms.
\end{enumerate}

\SupplementaryExercise{8}{Equation-of-Motion and Boundary Operators}
  {(Inspired by Iain Stewart and MIT OpenCourseWare, 8.851 Lecture 3, Field Redefinitions.)}
  {mit-8851-s13-field-redefinitions-curated-parent-1}
\begin{enumerate}
\item
For \(\phi\mapsto\phi+\epsilon F(\phi,\partial\phi)\), derive the
first-order change of the action as an Euler--Lagrange functional times
\(F\), plus an explicit boundary term.
\item
In a massive scalar theory, use
\(\phi\mapsto\phi-\epsilon\phi^3\) to eliminate an interaction
\(\epsilon\phi^3(\Box-m^2)\phi\) through first order and list the
induced nonderivative operators.
\item
Compare a total four-divergence with an operator proportional to the
leading field equation.  Explain precisely why the first changes no
action while the second requires a field redefinition.
\end{enumerate}

\SupplementaryExercise{9}{Field Redefinitions, External Poles, and Order Bookkeeping}
  {(Inspired by Iain Stewart and MIT OpenCourseWare, 8.851 Lecture 3, Field Redefinitions.)}
  {mit-8851-s13-field-redefinitions-curated-parent-2}
\begin{enumerate}
\item
Construct the inverse of
\(\phi'=\phi+\epsilon\phi^2+\epsilon^2a\phi^3\)
through order \(\epsilon^2\), and identify the condition that makes this
change of field variables locally admissible.
\item
Suppose \(\phi'=\phi+\epsilon\phi^2\) has nonzero vacuum-to-one-particle
overlap.  Show from the one-particle pole of its two-point function why
amputated scattering observables are unchanged after residue adjustment.
\item
If a first-order field redefinition removes one dimension-six operator,
show why its quadratic action shift produces dimension-eight
interactions and why discarding them is consistent only at fixed order.
\end{enumerate}

\SupplementaryExercise{10}{Geometry, Hamiltonian, and Current of the \(O(N)\) Sigma Model}
  {(Inspired by David Tong, Lectures on Statistical Field Theory, Section 4.3 | David Tong, Lectures on Statistical Field Theory, Sections 4.1 and 4.3.)}
  {tong-sft-sigma-models-2017-curated-parent-1}
\begin{enumerate}
\item
Solve \(n^An^A=1\) by
\(n^A=(\pi^a,\sqrt{1-\boldsymbol\pi^2})\) and derive the exact
field-space metric multiplying \(\partial_\mu\pi^a\partial^\mu\pi^b\).
\item
Invert the sphere metric obtained in S.7.10(a), calculate the canonical
momenta, and write the exact Hamiltonian density in terms of
\(\boldsymbol\pi\) and its conjugates.
\item
For the constrained field \(n^An^A=1\), derive the current associated
with rotations in the \(AB\)-plane and verify its conservation using
the constrained equation of motion.
\end{enumerate}

\SupplementaryExercise{11}{Target-Space Covariance, Curvature, and Stereographic Coordinates}
  {(Inspired by David Tong, Lectures on Statistical Field Theory, Section 4.3.)}
  {tong-sft-sigma-models-2017-curated-parent-2}
\begin{enumerate}
\item
Under an invertible reparametrization
\(\pi^a=\pi^a(\xi)\), derive the transformation law of the sigma-model
metric and show directly that the Lagrangian remains a target-space
scalar.
\item
For the two-sphere target metric, compute one nonzero component of the
Riemann tensor and explain why no field redefinition can make the
kinetic term globally free.
\item
Parametrize \(S^2\) by stereographic fields \((u,v)\), derive the
conformally flat target metric, and obtain the canonical Hamiltonian
including its exact field-dependent inverse factor.
\end{enumerate}
