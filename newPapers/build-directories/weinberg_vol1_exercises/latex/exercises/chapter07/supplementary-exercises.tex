\SupplementaryExercise{1}{Symmetries, Currents, Charges, and Lorentz Boosts}
  {(Adapted from University of Cambridge, Part III Quantum Field Theory, Paper 301, 2023, Question 1.)}
  {cambridge-partiii-2023-qft-paper301-q1}
Consider a relativistic theory of one real scalar field \(\phi(x)\),
with Lagrangian density \(\mathcal L\) and action
\[
 S=\int d^4x\,\mathcal L.
\]
\begin{enumerate}
\item
Define a continuous symmetry of \(S\), a conserved current, and its
conserved charge.  Explain carefully how these three notions are
related, including the assumptions under which the existence of any
one implies either of the other two.
\item
Define the energy--momentum tensor \(T^{\rho\sigma}\).  For the
mostly-plus form of the source theory,
\[
 \mathcal L=-\frac12\partial_\mu\phi\,\partial^\mu\phi-V(\phi),
\]
derive an explicit formula valid for arbitrary \(V\).  Explain why
\(T^{\rho\sigma}\) supplies four conserved currents and find their
charges.
\item
Starting from invariance of the action under infinitesimal Lorentz
transformations, show that
\[
 \partial_\mu\left(x^\rho T^{\mu\sigma}
                   -x^\sigma T^{\mu\rho}\right)=0.
\]
\item
Hence prove that the three-vector
\[
 V^i\equiv\frac{d}{dt}\int d^3x\,x^iT^{00}
\]
is constant in time.
\end{enumerate}

\SupplementaryExercise{2}{The Maxwell Stress Tensor and Metric Response}
  {(Adapted from J. McGreevy, Physics 215A Quantum Field Theory, Fall 2023, Assignment 3, Problem 1; lightly edited there from Peskin and Schroeder, Problem 2.1.)}
  {mcgreevy-215a-f23-hw03-p1}
Classical electromagnetism follows from
\[
 S[A]=\int d^4x\left(
 -\frac14F_{\mu\nu}F^{\mu\nu}-j^\mu A_\mu\right),
\qquad
 F_{\mu\nu}=\partial_\mu A_\nu-\partial_\nu A_\mu.
\]
\begin{enumerate}
\item
Treat the four components \(A_\mu(x)\) as the dynamical variables and
derive the Euler--Lagrange equations
\[
 \frac{\delta S[A]}{\delta A_\mu(x)}=0.
\]
Write them in standard vector form using
\[
 E^i=-F^{0i},\qquad
 \varepsilon^{ijk}B^k=-F^{ij}.
\]
In the source, the two resulting equations are called Gauss's and
Faraday's laws; more precisely, variation of \(A_\mu\) gives Gauss's
law and the Amp\`ere--Maxwell law, while Faraday's law is the Bianchi
identity.
\item
Set \(j^\mu=0\).  Construct the canonical energy--momentum tensor and
show explicitly that it is neither symmetric nor gauge invariant.
The source starts from its canonical convention
\[
 (T_{\rm src})^\mu{}_\nu
 =\frac{\partial\mathcal L}
        {\partial(\partial_\mu A_\rho)}
   \partial_\nu A_\rho-\mathcal L\delta^\mu{}_\nu.
\]
Improve it according to
\[
 \widetilde T_{\rm src}^{\mu\nu}
 =T_{\rm src}^{\mu\nu}+\partial_\lambda K_{\rm src}^{\lambda\mu\nu},
\qquad
 K_{\rm src}^{\lambda\mu\nu}=F^{\mu\lambda}A^\nu.
\]
This is the source's canonical-tensor convention.  With the sign
convention for \(T^\mu{}_\nu\) used in this volume, the translated
improvement is
\[
 \widetilde T^{\mu\nu}
 =T^{\mu\nu}-\partial_\lambda
 \left(F^{\mu\lambda}A^\nu\right).
\]
Explain why the added term is identically conserved and does not change
the global charges.  Prove that \(\widetilde T^{\mu\nu}\) is symmetric
and gauge invariant, and recover
\[
 \mathcal E=\frac12(\mathbf E^2+\mathbf B^2),
\qquad
 \mathbf S=\mathbf E\times\mathbf B
\]
as the electromagnetic energy and momentum densities.
\item
As a bonus, couple the theory to a background metric:
\[
 S[A,g]=\int d^4x\,\sqrt{|g|}
 \left(-\frac14F_{\mu\nu}F_{\rho\sigma}
 g^{\mu\rho}g^{\nu\sigma}+j^\mu A_\mu\right).
\]
The displayed source uses the opposite sign for its source term here;
the comparison below is made at \(j^\mu=0\), so this mismatch is
irrelevant.  The source's mostly-minus metric-response convention has
a plus sign; in the volume's mostly-plus convention, compare the result
of
\[
 T^{(g)}_{\mu\nu}
 =\left.-\frac{2}{\sqrt{|g|}}
 \frac{\delta S[A,g]}{\delta g^{\mu\nu}}\right|_{g_{\mu\nu}=\eta_{\mu\nu}}
\]
with part (b), using
\[
 \frac{\delta g^{\mu\nu}(x)}{\delta g_{\rho\sigma}(y)}
 =-g^{\mu\rho}g^{\nu\sigma}\delta^D(x-y),
\qquad
 \frac{\delta\det g(x)}{\delta g_{\mu\nu}(y)}
 =\delta^D(x-y)\det g\,g^{\mu\nu}.
\]
The source points to its accompanying notes for proofs of these two
identities; they may be taken as supplied here.
Explain why the metric-response tensor is automatically symmetric and
gauge invariant.
\end{enumerate}

\SupplementaryExercise{3}{A Complex Scalar Field}
  {(Adapted from Hong Liu and MIT OpenCourseWare, 8.323 Assignment 1, Problem 3.)}
  {mit-823-s23-ps1-ch7-curated-parent-1}
Consider a complex scalar field \(\phi(x)\) with
\[
 S=\int d^4x\left[
 -\partial_\mu\phi^\dagger\partial^\mu\phi
 -V(\phi^\dagger\phi)\right].
\]
Treat \(\phi\) and \(\phi^\dagger\), rather than their real and
imaginary parts, as independent dynamical variables.
\begin{enumerate}
\item
Check that the action is Lorentz invariant and derive the two
Euler--Lagrange equations.
\item
Find the momenta canonically conjugate to \(\phi\) and
\(\phi^\dagger\), and construct the Hamiltonian \(H\).
\item
The action is invariant under the rigid transformation
\[
 \phi\longmapsto e^{i\alpha}\phi,\qquad
 \phi^\dagger\longmapsto e^{-i\alpha}\phi^\dagger,
\]
where \(\alpha\) is constant.  Use Noether's theorem to find the
corresponding conserved current \(j^\mu\) and charge \(Q\).
\item
Use the equations of motion from part (a) to verify directly that
\(\partial_\mu j^\mu=0\).
\end{enumerate}

\SupplementaryExercise{4}{The Energy--Momentum Tensor of a Complex Scalar}
  {(Adapted from Hong Liu and MIT OpenCourseWare, 8.323 Assignment 1, Problem 4.)}
  {mit-823-s23-ps1-ch7-curated-parent-2}
Continue with the complex-scalar theory
\[
 S=\int d^4x\left[
 -\partial_\mu\phi^\dagger\partial^\mu\phi
 -V(\phi^\dagger\phi)\right].
\]
\begin{enumerate}
\item
Under a spacetime translation
\[
 x^\mu\longmapsto x'^\mu=x^\mu+a^\mu,
\qquad
 \phi'(x')=\phi(x),
\]
show that the action is invariant under
\(\phi(x)\mapsto\phi'(x)\), and likewise for \(\phi^\dagger\).
\item
Write the infinitesimal transformations of \(\phi\) and
\(\phi^\dagger\), and use Noether's theorem to derive the four
conserved translation currents \(T^{\mu\nu}\).
\item
For a time translation, the conserved charge is
\[
 H=\int d^3x\,T^{00},
\]
while a spatial translation gives
\[
 P^i=\int d^3x\,T^{0i}.
\]
Write \(H\) and \(P^i\) explicitly and compare \(H\) with the
Hamiltonian found in S.7.3(b).
\item
Use the field equations from S.7.3(a) to verify directly that
\(\partial_\mu T^{\mu\nu}=0\).
\end{enumerate}

\SupplementaryExercise{5}{Non-Abelian Currents of Two Complex Scalars}
  {(Adapted from J. McGreevy, Physics 215A Quantum Field Theory, Fall 2023, Assignment 6, Problem 2.)}
  {mcgreevy-215a-f23-hw06-p2}
Consider two complex scalar fields
\(\Phi_{\alpha}\), \(\alpha=1,2\), with the mostly-plus form of the
source action
\[
 S[\Phi_\alpha]=\int d^dx\,dt\left[
 -\frac12\partial_\mu\Phi_\alpha^\dagger
             \partial^\mu\Phi_\alpha
 -V(\Phi_\alpha^\dagger\Phi_\alpha)\right].
\]
Let \(\Pi_\alpha\) and \(\Pi_\alpha^\dagger\) be the canonical momenta,
and consider the three objects written in the source as
\[
 Q^i\equiv\frac12\int d^dx\,
 i\left(\Pi_\alpha^\dagger
 \sigma^i_{\alpha\beta}\Phi_\beta^\dagger\right)
 +\text{h.c.},
\qquad i=1,2,3,
\]
where \(\sigma^i\) are the Pauli matrices.
\begin{enumerate}
\item
Determine the transformations generated by the three charges and prove
that they are symmetries of \(S\).
\item
Optionally verify directly that \([Q^i,H]=0\).
\item
Evaluate \([Q^i,Q^j]\), show that the charges are non-Abelian, and
identify the resulting familiar Lie algebra.
\item
Derive the Noether currents \(J_\mu^i\) for the transformations in
part (a), and prove that
\[
 Q^i=\int d^dx\,J^{0i}
\]
agrees with the supplied canonical expression.  (The mostly-minus
source writes the time component with a lower index.)
\item
Generalize the construction to \(N\) complex scalar fields.
\end{enumerate}

  {(Inspired by Brian P. Dolan, Constrained Dynamics in the Hamiltonian Formalism, Sections 2 and 4 | Brian P. Dolan, Constrained Dynamics in the Hamiltonian Formalism, Sections 4 and 6 | Brian P. Dolan, Constrained Dynamics in the Hamiltonian Formalism, Sections 3--4.)}

\(L=\frac12(\dot q_1-q_2)^2-\frac12\mu^2q_2^2
-\frac12\omega^2q_1^2\), compute the velocity Hessian, find the primary

\item

\SupplementaryExercise{7}{Dirac Brackets and the Reduced Poisson Geometry}
  {(Inspired by Brian P. Dolan, Constrained Dynamics in the Hamiltonian Formalism, Section 4 | Brian P. Dolan, Constrained Dynamics in the Hamiltonian Formalism, Sections 3--4 | Brian P. Dolan, Constrained Dynamics in the Hamiltonian Formalism, Sections 4 and 6.)}
  {dolan-constrained-dynamics-2112.12216-curated-parent-2}
\begin{enumerate}
\item
For arbitrary second-class constraints, prove directly from the inverse
constraint matrix that \(\{\chi_A,F\}_{\rm D}=0\) for every phase-space
function \(F\).
\item
Write the Dirac bracket as a projected Poisson tensor.  Show that the
projection annihilates every constraint gradient and restricts to the
ordinary symplectic inverse on tangent vectors.
\item
Replace constraints \((\chi_1,\chi_2)\) by
\((e^{f(z)}\chi_1,e^{-f(z)}\chi_2)\).  Prove that their Dirac brackets
agree on the common constraint surface, including all derivative terms.
\end{enumerate}

\SupplementaryExercise{8}{Equation-of-Motion and Boundary Operators}
  {(Inspired by Iain Stewart and MIT OpenCourseWare, 8.851 Lecture 3, Field Redefinitions.)}
  {mit-8851-s13-field-redefinitions-curated-parent-1}
\begin{enumerate}
\item
For \(\phi\mapsto\phi+\epsilon F(\phi,\partial\phi)\), derive the
first-order change of the action as an Euler--Lagrange functional times
\(F\), plus an explicit boundary term.
\item
In a massive scalar theory, use
\(\phi\mapsto\phi-\epsilon\phi^3\) to eliminate an interaction
\(\epsilon\phi^3(\Box-m^2)\phi\) through first order and list the
induced nonderivative operators.
\item
Compare a total four-divergence with an operator proportional to the
leading field equation.  Explain precisely why the first changes no
action while the second requires a field redefinition.
\end{enumerate}

\SupplementaryExercise{9}{Field Redefinitions, External Poles, and Order Bookkeeping}
  {(Inspired by Iain Stewart and MIT OpenCourseWare, 8.851 Lecture 3, Field Redefinitions.)}
  {mit-8851-s13-field-redefinitions-curated-parent-2}
\begin{enumerate}
\item
Construct the inverse of
\(\phi'=\phi+\epsilon\phi^2+\epsilon^2a\phi^3\)
through order \(\epsilon^2\), and identify the condition that makes this
change of field variables locally admissible.
\item
