\chapterbackmatter{Weinberg Exercises}

\begin{enumerate}
\WeinbergExercise{1} List all the renormalizable (or superrenormalizable) Lorentz-invariant
terms in the Lagrangian of a single scalar field for space-time
dimensionalities 2, 3, and 6.

\WeinbergExercise{2} Show how the overlapping divergence in the electron self-energy is
cancelled in quantum electrodynamics.

\WeinbergExercise{3} Consider the theory of a scalar field $\phi$ and spinor field $\psi$,
with interaction Hamiltonian $g\phi\bar\psi\psi$. Write the one-loop part of
the scalar self-energy function $\Pi^*(q)$ as a divergent polynomial in
$p^\mu$, plus an explicit convergent integral.

\WeinbergExercise{4} Suppose that the quantum electrodynamics of electrons and photons is
actually an effective field theory, derived by integrating out unknown
particles of mass $M\gg m_e$. Assume gauge invariance and Lorentz invariance,
but not invariance under C, P, or T. What are the non-renormalizable terms in
the Lagrangian of leading order in $1/M$? Of next to leading order?
\end{enumerate}
