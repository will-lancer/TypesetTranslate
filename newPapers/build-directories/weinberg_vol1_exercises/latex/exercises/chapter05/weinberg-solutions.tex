\chapterbackmatter{Solutions to Weinberg Exercises}
\ExerciseEditorialNote

\begin{enumerate}
\WeinbergSolution{1}
Write \(k=(m,\mathbf0)\), \(q=\Lambda p\), and
\[
 W(\Lambda,p)=L^{-1}(q)\Lambda L(p).
\]
The defining identity is therefore
\[
 L(q)W(\Lambda,p)=\Lambda L(p).
\]
Starting from Eq.~\eqref{eq:5.1.21} and using the zero-momentum
intertwining relation \eqref{eq:5.1.23}, we obtain
\[
\begin{split}
 \sum_{\bar\sigma}u_\ell(\mathbf q,\bar\sigma)
 D^{(j)}_{\bar\sigma\sigma}(W)
 &=
 \sqrt{\frac{m}{q^0}}\,
 \left[D(L(q))u(\mathbf0)D^{(j)}(W)\right]_{\ell\sigma}\\
 &=
 \sqrt{\frac{m}{q^0}}\,
 \left[D(L(q))D(W)u(\mathbf0)\right]_{\ell\sigma}\\
 &=
 \sqrt{\frac{m}{q^0}}\,
 \left[D(\Lambda)D(L(p))u(\mathbf0)\right]_{\ell\sigma}\\
 &=
 \sqrt{\frac{p^0}{q^0}}\,
 \sum_{\bar\ell}D_{\ell\bar\ell}(\Lambda)
 u_{\bar\ell}(\mathbf p,\sigma).
\end{split}
\]
This is precisely Eq.~\eqref{eq:5.1.19}.  The same calculation, with
\(v(\mathbf0)D^{(j)*}(W)=D(W)v(\mathbf0)\), proves
Eq.~\eqref{eq:5.1.20}.  Thus the rest-frame rotation intertwiners and one
standard boost determine coefficient functions satisfying the full Lorentz
covariance conditions.

\WeinbergSolution{2}
Let \(u_\ell(\mathbf p,s)\) be the spin-\(\tfrac12\) Dirac coefficient
functions of Section 5.5, and let \(e^\mu(\mathbf p,r)\) be the massive
spin-one polarization vectors of Section 5.3.  Couple the two rotation
representations at rest and then boost:
\[
 u_\ell^\mu(\mathbf p,\sigma)=
 \sum_{r=-1}^{1}\sum_{s=\pm1/2}
 \braket{1r,\tfrac12s}{\tfrac32\sigma}\,
 e^\mu(\mathbf p,r)u_\ell(\mathbf p,s).
\]
The Clebsch--Gordan coefficient projects \(1\otimes\tfrac12\) onto
spin \(\tfrac32\); covariance of the two factors then gives the vector-spinor
transformation law.  The creation coefficients are fixed by the Majorana
relation, so the causal field may be written
\[
 \psi^\mu_\ell(x)=\frac1{(2\pi)^{3/2}}\sum_\sigma\int d^3p\,
 \left[u^\mu_\ell(\mathbf p,\sigma)a_{\mathbf p,\sigma}e^{ip\cdot x}
 {}+v^\mu_\ell(\mathbf p,\sigma)a^\dagger_{\mathbf p,\sigma}e^{-ip\cdot x}\right],
\]
with \(v\) the charge conjugate of \(u\).  It obeys
\[
 (\gamma^\rho\partial_\rho+m)\psi^\mu=0,\qquad
 \gamma_\mu\psi^\mu=0,\qquad
 \partial_\mu\psi^\mu=0,
\]
where the last equation follows from the first two.  Self-charge-conjugacy
gives the reality condition
\[
 \psi^\mu=-\beta\mathcal C\,\psi^{\mu *}.
\]

For a compact form of the spin sum define, on \(p^2=-m^2\),
\[
 \Theta^{\mu\nu}(p)=\eta^{\mu\nu}+\frac{p^\mu p^\nu}{m^2},
 \qquad
 \Gamma^\mu(p)=\Theta^\mu{}_\rho(p)\gamma^\rho,
\]
\[
 Q_{3/2}^{\mu\nu}(p)=
 \Theta^{\mu\nu}(p)-\frac13\Gamma^\mu(p)\Gamma^\nu(p).
\]
This is transverse, gamma-traceless on the physical Dirac subspace, and at
rest is the familiar projector
\(\delta^{ij}-\sigma^i\sigma^j/3\).  The requested answer is
\[
 P_{\ell m}^{\mu\nu}(p)=
 \left\{
 Q_{3/2}^{\mu\nu}(p)
 [-ip^\rho\gamma_\rho+m]\beta
 \right\}_{\ell m}.
\]
Equivalently, suppressing spinor indices, ordinary matrix multiplication
is understood:
\[
 P^{\mu\nu}=Q_{3/2}^{\mu\nu}[-i\slashed p+m]\beta.
\]
It immediately satisfies
\(p_\mu P^{\mu\nu}=\gamma_\mu P^{\mu\nu}=0\).

The field is fermionic.  If
\[
 \widehat\Theta^{\mu\nu}
 =\eta^{\mu\nu}-\frac{\partial^\mu\partial^\nu}{m^2},
 \qquad
 \widehat\Gamma^\mu
 =\gamma^\mu-\frac{\partial^\mu\gamma^\rho\partial_\rho}{m^2},
 \qquad
 \widehat Q_{3/2}^{\mu\nu}
 =\widehat\Theta^{\mu\nu}
 -\frac13\widehat\Gamma^\mu\widehat\Gamma^\nu,
\]
then its causal anticommutator is
\[
 \{\psi^\mu_\ell(x),\psi^{\nu\dagger}_m(y)\}
 =
 \left\{\widehat Q_{3/2}^{\mu\nu}
 [-\gamma^\rho\partial_\rho+m]\beta\right\}_{\ell m}
 \Delta(x-y),
\]
and anticommutators without an adjoint are fixed by the Majorana condition.
They vanish at spacelike separation.

Up to the conventional intrinsic phases, the inversions are the tensor
product of the vector and Dirac rules:
\[
\begin{split}
 P\psi^\mu(x)P^{-1}
 &=-\eta^*\,\mathcal P^\mu{}_\nu\beta\,
   \psi^\nu(\mathcal P x),\\
 C\psi^\mu(x)C^{-1}
 &=-\xi^*\,\beta\mathcal C\,\psi^{\mu *}(x),\\
 T\psi^\mu(x)T^{-1}
 &=\zeta^*\,\mathcal P^\mu{}_\nu\gamma^5\mathcal C\,
   \psi^\nu(-\mathcal T x).
\end{split}
\]
A rephasing can move the overall sign in the last line into \(\zeta\).
For a self-conjugate field, \(\xi=\pm1\) and
\(\eta=\pm i\); the usual relations
\(\xi^c=\xi^*\), \(\zeta^c=\zeta^*\), and
\(\eta^c=-\eta^*\) hold.

\WeinbergSolution{3}
Use the spin-one polarization vectors and project the product \(1\otimes1\)
onto its spin-two part:
\[
 e^{\mu\nu}(\mathbf p,\sigma)=
 \sum_{r,s=-1}^{1}\braket{1r,1s}{2\sigma}\,
 e^\mu(\mathbf p,r)e^\nu(\mathbf p,s),
 \qquad
 u^{\mu\nu}(\mathbf p,\sigma)=
 \frac{e^{\mu\nu}(\mathbf p,\sigma)}{\sqrt{2p^0}}.
\]
The Clebsch--Gordan coefficient makes this tensor symmetric and traceless.
Since every vector polarization is transverse, the resulting free field
satisfies
\[
 h^{\mu\nu}=h^{\nu\mu},\qquad
 h^\mu{}_\mu=0,\qquad
 \partial_\mu h^{\mu\nu}=0,\qquad
 (\Box-m^2)h^{\mu\nu}=0.
\]
These conditions leave \(10-1-4=5\) independent components, the required
\(2j+1\) polarizations.

With
\(\Theta^{\mu\nu}=\eta^{\mu\nu}+p^\mu p^\nu/m^2\), completeness of the
spin-two polarization tensors gives
\[
 P^{\mu\nu,\kappa\lambda}(p)=
 \frac12\left(
 \Theta^{\mu\kappa}\Theta^{\nu\lambda}
 +\Theta^{\mu\lambda}\Theta^{\nu\kappa}\right)
 -\frac13\Theta^{\mu\nu}\Theta^{\kappa\lambda}.
\]
It is symmetric within and between the index pairs, transverse in every
index, traceless, and has trace five.  The field is bosonic, and
\[
 [h^{\mu\nu}(x),h^{\kappa\lambda\dagger}(y)]
 =
 \widehat P^{\mu\nu,\kappa\lambda}(\partial)\Delta(x-y),
\]
where
\[
 \widehat\Theta^{\mu\nu}
 =\eta^{\mu\nu}-\frac{\partial^\mu\partial^\nu}{m^2}
\]
is inserted in the preceding expression for \(P\).  This commutator vanishes
at spacelike separation.  For a self-charge-conjugate particle the field can
be chosen Hermitian.

The inversion laws, with intrinsic phases left explicit, are
\[
\begin{split}
 Ph^{\mu\nu}(x)P^{-1}
 &=\eta^*\mathcal P^\mu{}_\rho\mathcal P^\nu{}_\sigma
 h^{\rho\sigma}(\mathcal P x),\\
 Ch^{\mu\nu}(x)C^{-1}
 &=\xi^*h^{\mu\nu\dagger}(x),\\
 Th^{\mu\nu}(x)T^{-1}
 &=\zeta^*\mathcal P^\mu{}_\rho\mathcal P^\nu{}_\sigma
 h^{\rho\sigma}(-\mathcal P x).
\end{split}
\]
Here \(\eta^c=\eta^*\), \(\xi^c=\xi^*\), and
\(\zeta^c=\zeta^*\); all three phases may be chosen real for a neutral
self-conjugate spin-two particle.

\WeinbergSolution{4}
It is clearest to use two-component spinor indices.  A derivative
\(\partial_{\alpha\dot\alpha}\) transforms as
\((\tfrac12,\tfrac12)\).  Let
\(\phi_{\dot\beta_1\cdots\dot\beta_{2j}}\) be a totally symmetric
\((0,j)\) field.  The completely irreducible part of
\[
 \partial_{\alpha_1\dot\alpha_1}\cdots
 \partial_{\alpha_{2A}\dot\alpha_{2A}}
 \phi_{\dot\beta_1\cdots\dot\beta_{2j}}
\]
that is symmetric separately in all undotted and all dotted indices has
\(2A\) undotted indices and \(2A+2j\) dotted indices.  It therefore
transforms as \((A,A+j)\).  On each plane wave the derivatives merely
multiply the coefficient function by
\((ip_{\alpha\dot\alpha})^{2A}\), so they neither introduce a new particle
species nor change the helicity.  Uniqueness of the covariant field of a
fixed Lorentz type then shows that every \((A,A+j)\) field is proportional
to this derivative field.

Starting instead with a symmetric \((j,0)\) field and taking the separately
symmetric irreducible part of \(2B\) derivatives gives \(2B+2j\) undotted
and \(2B\) dotted indices, hence type \((B+j,B)\).  This proves both
statements.

\WeinbergSolution{5}
Write the two chiral fields as
\[
 \Psi(x)=
 \begin{pmatrix}\phi_L(x)\\ \phi_R(x)\end{pmatrix},
 \qquad
 \phi_L\sim(j,0),\quad \phi_R\sim(0,j).
\]
Equation \eqref{eq:5.9.41} says that \(\phi_L\) annihilates helicity
\(-j\) and creates the opposite-helicity antiparticle, whereas \(\phi_R\)
annihilates helicity \(+j\) and creates its opposite-helicity antiparticle.
Let
\[
 (\mathsf C_j)_{mn}=(-1)^{j-m}\delta_{m,-n}
\]
be the spin-\(j\) reversal matrix.  With harmless convention-dependent
signs absorbed into the intrinsic phases, the inversion laws can be written
\[
\begin{split}
 P\Psi(x)P^{-1}
 &=\eta^*
 \begin{pmatrix}0&\mathbf1\\ \mathbf1&0\end{pmatrix}
 \Psi(\mathcal P x),\\
 C\Psi(x)C^{-1}
 &=\xi^*
 \begin{pmatrix}0&\mathsf C_j\\
 (-1)^{2j}\mathsf C_j&0\end{pmatrix}
 \Psi^*(x),\\
 T\Psi(x)T^{-1}
 &=\zeta^*
 \begin{pmatrix}\mathsf C_j&0\\0&\mathsf C_j\end{pmatrix}
 \Psi(\mathbf x,-x^0).
\end{split}
\]
The operator \(T\) is antiunitary, so the numerical coefficients in the last
line are complex conjugated when the rule is used.

These formulas display the physics independently of phase convention:
\(P\) reverses momentum but not spin and therefore reverses helicity, so it
interchanges \((j,0)\) and \((0,j)\); \(C\) exchanges particles and
antiparticles; and \(T\) reverses both momentum and spin, preserving
helicity and each chiral transformation type.  The particle and antiparticle
phases obey
\[
 \eta^c=(-1)^{2j}\eta^*,\qquad
 \xi^c=\xi^*,\qquad
 \zeta^c=\zeta^*.
\]

\WeinbergSolution{6}
Decompose the generalized Dirac field as
\(\psi=(\psi_L,\psi_R)\), with
\(\psi_L\sim(j,0)\) and \(\psi_R\sim(0,j)\).
The four kinds of bilinear products transform as
\[
\begin{array}{c|c}
\text{bilinear}&\text{Lorentz content}\\ \hline
\psi_L^\dagger\psi_L&(j,j)\\
\psi_R^\dagger\psi_R&(j,j)\\
\psi_R^\dagger\psi_L&
(j,0)\otimes(j,0)=\displaystyle\bigoplus_{k=0}^{2j}(k,0)\\
\psi_L^\dagger\psi_R&
(0,j)\otimes(0,j)=\displaystyle\bigoplus_{k=0}^{2j}(0,k).
\end{array}
\]
Thus the complete list is
\[
 2(j,j)\ \oplus\
 \bigoplus_{k=0}^{2j}\bigl[(k,0)\oplus(0,k)\bigr].
\]
For each pair related by parity one may take a sum and a difference to form
ordinary and pseudo tensors.  Dimension counting checks completeness:
\[
 2(2j+1)^2+
 2\sum_{k=0}^{2j}(2k+1)
 =4(2j+1)^2,
\]
which is the dimension of the full matrix algebra acting on the
\(2(2j+1)\)-component field.

For \(j=\tfrac12\), the two \((0,0)\)'s are the scalar and pseudoscalar,
the two \((\tfrac12,\tfrac12)\)'s are the vector and axial vector, and
\((1,0)\oplus(0,1)\) is the antisymmetric tensor.  These are exactly the
sixteen standard Dirac bilinears
\(\bar\psi\psi\), \(\bar\psi\gamma^5\psi\),
\(\bar\psi\gamma^\mu\psi\),
\(\bar\psi\gamma^\mu\gamma^5\psi\), and
\(\bar\psi[\gamma^\mu,\gamma^\nu]\psi\).

\WeinbergSolution{7}
Choose the emitted momentum along the 3-axis.  Equation
\eqref{eq:5.7.14} then reduces to
\[
 u_{ab}(\mathbf p,\sigma)=
 \frac{C_{AB}(j\sigma;ab)}{\sqrt{2E}}\,
 e^{(-a+b)\theta},
 \qquad a+b=\sigma,
\]
where
\[
 e^\theta=\frac{E+|\mathbf p|}{m}
 =\frac{2E}{m}\left[1+O\!\left(\frac{m^2}{E^2}\right)\right].
\]
Because the Fourier components \(J^{ab}(p)\) are assumed to be comparable,
the largest exponent \(-a+b\) among nonzero Clebsch--Gordan coefficients
sets the generic high-energy behavior.  Subject to
\(-A\leq a\leq A\), \(-B\leq b\leq B\), and \(a+b=\sigma\), it is
\[
 r_\sigma=\max(-a+b)
 =\min(2A+\sigma,\,2B-\sigma).
\]
Consequently a definite-helicity emission matrix element behaves as
\[
 \mathcal A_\sigma(E)\sim
 E^{-1/2}\left(\frac{E}{m}\right)^{r_\sigma}
\]
up to an \(E\)-independent current component, a numerical
Clebsch--Gordan coefficient, and the overall spacetime
momentum-conservation factor.  Accidental cancellations or current
conservation can soften this leading power.

As checks, a scalar has \(r_0=0\); for a four-vector
\((A,B)=(\tfrac12,\tfrac12)\), the transverse helicities have
\(r_{\pm1}=0\) while the longitudinal helicity has \(r_0=1\), producing the
familiar \(E/m\) enhancement; and for a chiral \((j,0)\) field the helicity
\(-j\) is the unsuppressed massless component while the opposite helicity is
power suppressed.
\end{enumerate}
