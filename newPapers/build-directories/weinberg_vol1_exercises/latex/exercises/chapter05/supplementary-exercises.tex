\SupplementaryExercise{4}{The Lorentz Algebra in the Vector Representation}
  {(Adapted from B. Mistlberger and K. Zhou, Stanford PHYS-330, Problem Set 2, 2022, Problem 1.)}
  {zhou-stanford-phys330-ps2-2022-curated-parent-1}
\begin{enumerate}
\item
Starting from preservation of inner products under
\(x'^\mu=\Lambda^\mu{}_\nu x^\nu\), prove
\[
 \eta_{\mu\nu}
 =\eta_{\rho\sigma}\Lambda^\rho{}_\mu\Lambda^\sigma{}_\nu.
\]
\item
For \(\Lambda^\mu{}_\nu
=\delta^\mu{}_\nu+\varepsilon\omega^\mu{}_\nu\), prove
\(\omega_{\mu\nu}=-\omega_{\nu\mu}\).  Explain why proper,
orthochronous Lorentz transformations can be built from such
infinitesimal transformations.
\item
Write the \(4\times4\) matrix \(iJ_3\) for an infinitesimal rotation
\(\varepsilon=d\theta\) about the 3-axis, and identify the exponential
that gives a finite rotation through \(\theta\).
\item
Write the matrix \(iK_1\) for an infinitesimal boost
\(\varepsilon=dv\) along the 1-axis.  Express the finite boost of
velocity \(v\) as an exponential of \(K_1\).
\item
Define the remaining generators analogously.  The Lorentz algebra is
\[
 [J_i,J_j]=i\epsilon_{ijk}J_k,\quad
 [J_i,K_j]=i\epsilon_{ijk}K_k,\quad
 [K_i,K_j]=-i\epsilon_{ijk}J_k.
\]
Verify these relations explicitly for \((i,j)=(1,1)\) and
\((i,j)=(1,2)\); the remaining cases follow in the same way.  Interpret
the first two commutators as the vector transformation laws for angular
velocity and velocity, and the third as the possibility of obtaining a
rotation by composing non-collinear boosts.
\end{enumerate}

\SupplementaryExercise{5}{Quantization and Noether Charges of a Complex Scalar}
  {(Adapted from B. Mistlberger and K. Zhou, Stanford PHYS-330, Problem Set 2, 2022, Problem 2.)}
  {zhou-stanford-phys330-ps2-2022-curated-parent-2}
Consider
\[
 \mathcal L=-\partial_\mu\Phi^\dagger\partial^\mu\Phi
 -m^2\Phi^\dagger\Phi,
\qquad E_{\mathbf k}=\sqrt{\mathbf k^2+m^2}.
\]
At \(t=0\), use
\[
\begin{split}
\Phi(\mathbf x)&=\int\frac{d^3k}{(2\pi)^3\sqrt{2E_{\mathbf k}}}
\left[a(\mathbf k)e^{i\mathbf k\cdot\mathbf x}
+b^\dagger(\mathbf k)e^{-i\mathbf k\cdot\mathbf x}\right],\\
\Phi^\dagger(\mathbf x)&=\int\frac{d^3k}{(2\pi)^3\sqrt{2E_{\mathbf k}}}
\left[b(\mathbf k)e^{i\mathbf k\cdot\mathbf x}
+a^\dagger(\mathbf k)e^{-i\mathbf k\cdot\mathbf x}\right],\\
\Pi_\Phi(\mathbf x)&=-i\int\frac{d^3k}{(2\pi)^3}
\sqrt{\frac{E_{\mathbf k}}2}
\left[b(\mathbf k)e^{i\mathbf k\cdot\mathbf x}
-a^\dagger(\mathbf k)e^{-i\mathbf k\cdot\mathbf x}\right],\\
\Pi_{\Phi^\dagger}(\mathbf x)&=-i\int\frac{d^3k}{(2\pi)^3}
\sqrt{\frac{E_{\mathbf k}}2}
\left[a(\mathbf k)e^{i\mathbf k\cdot\mathbf x}
-b^\dagger(\mathbf k)e^{-i\mathbf k\cdot\mathbf x}\right].
\end{split}
\]
Quantize with
\[
 [\Phi(\mathbf x),\Pi_\Phi(\mathbf y)]
 =[\Phi^\dagger(\mathbf x),\Pi_{\Phi^\dagger}(\mathbf y)]
 =i\delta^3(\mathbf x-\mathbf y),
\]
all other canonical commutators being zero.  The vacuum obeys
\(a(\mathbf k)\ket0=b(\mathbf k)\ket0=0\).  As in the source
assignment, parallel calculations need not be repeated once the common
argument has been made explicit.
\begin{enumerate}
\item
Derive the commutators of \(a,a^\dagger,b,b^\dagger\), including the
vanishing mixed commutators.
\item
For
\(\Phi\to e^{i\alpha}\Phi\),
\(\Phi^\dagger\to e^{-i\alpha}\Phi^\dagger\), derive the Noether current
and conserved charge \(Q\).
\item
Normal order \(Q\), impose \(Q\ket0=0\), and find the charge of
\(a^\dagger(\mathbf k)\ket0\),
\(b^\dagger(\mathbf k)\ket0\), and
\(a^\dagger(\mathbf k_1)a^\dagger(\mathbf k_2)\ket0\).
\item
Prove
\[
 e^{i\alpha Q}\Phi(x)e^{-i\alpha Q}=e^{i\alpha}\Phi(x),
 \qquad
 e^{i\alpha Q}\Phi^\dagger(x)e^{-i\alpha Q}
 =e^{-i\alpha}\Phi^\dagger(x),
\]
first by deriving the infinitesimal commutators with \(Q\).
\item
Use Noether's theorem to derive \(T^{\mu\nu}\) and
\(P^\mu=\int d^3x\,T^{0\mu}\).
\item
Normal order \(H=P^0\) and \(\mathbf P\) in oscillator variables, set
\(P^\mu\ket0=0\), and find the four-momentum of the three states in
part (c).
\item
Show
\[
 e^{ia\cdot P}\Phi(x)e^{-ia\cdot P}=\Phi(x-a),
 \qquad
 [P^i,\Phi(x)]=i\partial^i\Phi(x),
\]
and verify the commutator from the oscillator expression for
\(\mathbf P\).
\end{enumerate}

\SupplementaryExercise{6}{Boost Charges on Scalar Creation Operators}
  {(Adapted from B. Mistlberger and K. Zhou, Stanford PHYS-330, Problem Set 2, 2022, Problem 3.)}
  {zhou-stanford-phys330-ps2-2022-curated-parent-3}
Under a Lorentz transformation, a scalar profile obeys
\[
 \phi'(x')=\phi(x),\qquad
 \phi'(x)=\phi(\Lambda^{-1}x).
\]
For
\(\Lambda^\mu{}_\nu
=\delta^\mu{}_\nu+\varepsilon\omega^\mu{}_\nu\), this gives, through
first order,
\[
 \phi'(x)=\phi(x)-\varepsilon\omega^\mu{}_\nu
 x^\nu\partial_\mu\phi(x).
\]
Work with a real scalar field.  Equivalently, one may obtain the same
real field by identifying the two oscillator families in a
complex-scalar mode expansion,
\(a(\mathbf k)=b(\mathbf k)\).
\begin{enumerate}
\item
For a real scalar, derive
\[
 M^{\mu\nu\rho}=x^\nu T^{\mu\rho}-x^\rho T^{\mu\nu}
\]
and the associated charges \(M^{\nu\rho}\).
\item
The six antisymmetric charges split into \(M^{ij}\) and \(M^{0i}\).
Explain the physical meaning of the three \(M^{0i}\).
\item
After normal ordering, prove
\[
 M^{ij}=i\int\frac{d^3k}{(2\pi)^3}\,
 a^\dagger(\mathbf k)
 \left(k^j\frac{\partial}{\partial k_i}
 -k^i\frac{\partial}{\partial k_j}\right)a(\mathbf k),
\]
and explain why this shows that scalar quanta have no intrinsic angular
momentum.
\end{enumerate}

\SupplementaryExercise{7}{Representation-Independent Clifford Identities}
  {(Adapted from D. Tong, \emph{Quantum Field Theory: Example Sheet 3}, 2006, Problem 3.)}
  {tong-qft-sheet3-2006-curated-parent-1}
Work in four dimensions with
\[
 \{\gamma^\mu,\gamma^\nu\}=2\eta^{\mu\nu},\qquad
 \gamma_5=i\gamma^0\gamma^1\gamma^2\gamma^3,\qquad
 \slashed p=p_\mu\gamma^\mu,\qquad
 S^{\mu\nu}=\frac14[\gamma^\mu,\gamma^\nu].
\]
Useful methods include cyclicity of the trace and inserting
\(\gamma_5^2=\mathbf1\) into a trace.
Without choosing a representation, prove:
\begin{enumerate}
\item \(\operatorname{tr}\gamma^\mu=0\);
\item \(\operatorname{tr}(\gamma^\mu\gamma^\nu)=4\eta^{\mu\nu}\);
\item \(\operatorname{tr}(\gamma^\mu\gamma^\nu\gamma^\rho)=0\);
\item \(\gamma_5^2=1\);
\item \(\operatorname{tr}\gamma_5=0\);
\item
\(\slashed p\slashed q
=2p\cdot q-\slashed q\slashed p
=p\cdot q+2S^{\mu\nu}p_\mu q_\nu\);
\item \(\operatorname{tr}(\slashed p\slashed q)=4p\cdot q\);
\item
\(\operatorname{tr}(\slashed p_1\cdots\slashed p_n)=0\) for odd \(n\);
\item
\[
\operatorname{tr}(\slashed p_1\slashed p_2\slashed p_3\slashed p_4)
=4\bigl[
(p_1\cdot p_2)(p_3\cdot p_4)
+(p_1\cdot p_4)(p_2\cdot p_3)
-(p_1\cdot p_3)(p_2\cdot p_4)\bigr];
\]
\item \(\operatorname{tr}(\gamma_5\slashed p\slashed q)=0\);
\item \(\gamma_\mu\slashed p\gamma^\mu=-2\slashed p\);
\item
\(\gamma_\mu\slashed p_1\slashed p_2\gamma^\mu
=4p_1\cdot p_2\);
\item
\(\gamma_\mu\slashed p_1\slashed p_2\slashed p_3\gamma^\mu
=-2\slashed p_3\slashed p_2\slashed p_1\);
\item
\[
\operatorname{tr}(\gamma_5
\slashed p_1\slashed p_2\slashed p_3\slashed p_4)
=-4i\epsilon^{\mu\nu\rho\sigma}
p_{1\mu}p_{2\nu}p_{3\rho}p_{4\sigma}.
\]
\end{enumerate}

\SupplementaryExercise{8}{The Clifford Representation of the Lorentz Algebra}
  {(Adapted from D. Tong, \emph{Quantum Field Theory: Example Sheet 3}, 2006, Problem 2.)}
  {tong-qft-sheet3-2006-curated-parent-2}
Starting only from the Clifford algebra, prove
\[
\begin{split}
[\gamma^\kappa\gamma^\lambda,\gamma^\mu\gamma^\nu]
={}&2\eta^{\lambda\mu}\gamma^\kappa\gamma^\nu
-2\eta^{\kappa\mu}\gamma^\lambda\gamma^\nu\\
&+2\eta^{\lambda\nu}\gamma^\mu\gamma^\kappa
-2\eta^{\kappa\nu}\gamma^\mu\gamma^\lambda.
\end{split}
\]
Then show
\[
 S^{\mu\nu}\equiv\frac14[\gamma^\mu,\gamma^\nu]
 =\frac12(\gamma^\mu\gamma^\nu-\eta^{\mu\nu}),
\]
and use the first identity to verify that \(S^{\mu\nu}\) furnish a
spinor representation of the Lorentz Lie algebra.

\SupplementaryExercise{9}{Dirac Spinor Norms and Orthogonality}
  {(Adapted from D. Tong, \emph{Quantum Field Theory: Example Sheet 3}, 2006, Problem 4.)}
  {tong-qft-sheet3-2006-curated-parent-3}
Use the local Weyl basis
\[
 \gamma^0=-i\begin{pmatrix}0&\mathbf1\\ \mathbf1&0\end{pmatrix},
 \qquad
 \gamma^i=-i\begin{pmatrix}0&\sigma^i\\-\sigma^i&0\end{pmatrix},
 \qquad
 \beta=i\gamma^0,
\]
and let \(\xi^s\), \(s=1,2\), be orthonormal two-component spinors.
For \(p^0=E_{\mathbf p}>0\), define
\[
 p\cdot\sigma=E_{\mathbf p}\mathbf1-\mathbf p\cdot\boldsymbol\sigma,
 \qquad
 p\cdot\bar\sigma=E_{\mathbf p}\mathbf1+\mathbf p\cdot\boldsymbol\sigma,
\]
\[
 u^s(\mathbf p)=
 \begin{pmatrix}\sqrt{p\cdot\sigma}\,\xi^s\\
 \sqrt{p\cdot\bar\sigma}\,\xi^s\end{pmatrix},
 \qquad
 v^s(\mathbf p)=
 \begin{pmatrix}\sqrt{p\cdot\sigma}\,\xi^s\\
 -\sqrt{p\cdot\bar\sigma}\,\xi^s\end{pmatrix}.
\]
Prove
\[
\begin{array}{rcl@{\qquad}rcl}
u^r(\mathbf p)^\dagger u^s(\mathbf p)&=&2E_{\mathbf p}\delta^{rs},
&\bar u^r(\mathbf p)u^s(\mathbf p)&=&2m\delta^{rs},\\
v^r(\mathbf p)^\dagger v^s(\mathbf p)&=&2E_{\mathbf p}\delta^{rs},
&\bar v^r(\mathbf p)v^s(\mathbf p)&=&-2m\delta^{rs}.
\end{array}
\]
as well as
\[
 \bar u^s(\mathbf p)v^r(\mathbf p)=0,\qquad
 u^r(\mathbf p)^\dagger v^s(-\mathbf p)=0.
\]

\SupplementaryExercise{11}{Equal-Time Dirac Anticommutators}
  {(Adapted from D. Tong, \emph{Quantum Field Theory: Example Sheet 3}, 2006, Problem 6.)}
  {tong-qft-sheet3-2006-curated-parent-5}
At fixed time, take
\[
\begin{split}
\psi(\mathbf x)
={}&\sum_{s=1}^2\int\frac{d^3p}{(2\pi)^3\sqrt{2E_{\mathbf p}}}
\left[
b^s_{\mathbf p}u^s(\mathbf p)e^{i\mathbf p\cdot\mathbf x}
+c_{\mathbf p}^{s\dagger}v^s(\mathbf p)e^{-i\mathbf p\cdot\mathbf x}
\right],\\
\psi^\dagger(\mathbf x)
={}&\sum_{s=1}^2\int\frac{d^3p}{(2\pi)^3\sqrt{2E_{\mathbf p}}}
\left[
b_{\mathbf p}^{s\dagger}u^s(\mathbf p)^\dagger
 e^{-i\mathbf p\cdot\mathbf x}
+c^s_{\mathbf p}v^s(\mathbf p)^\dagger
 e^{i\mathbf p\cdot\mathbf x}
\right].
\end{split}
\]
Assume
\[
\{b^r_{\mathbf p},b_{\mathbf q}^{s\dagger}\}
=\{c^r_{\mathbf p},c_{\mathbf q}^{s\dagger}\}
=(2\pi)^3\delta^{rs}\delta^3(\mathbf p-\mathbf q),
\]
with every other oscillator anticommutator zero.  Prove
\[
\{\psi_\alpha(\mathbf x),\psi_\beta(\mathbf y)\}
=\{\psi_\alpha^\dagger(\mathbf x),
\psi_\beta^\dagger(\mathbf y)\}=0,
\]
\[
 \{\psi_\ell(\mathbf x),\psi_m^\dagger(\mathbf y)\}
 =\delta_{\ell m}\delta^3(\mathbf x-\mathbf y),
\]
using the two completeness identities in S.5.10.
