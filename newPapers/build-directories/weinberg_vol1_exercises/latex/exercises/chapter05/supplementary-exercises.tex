\SupplementaryExercise{1}{The Klein--Gordon Current Is Not a Probability}
  {(Adapted from H. Liu, MIT 8.323, Assignment 2, 2023, Problem 1.)}
  {mit-823-s23-ps2-curated-parent-1}
\begin{enumerate}
\item Derive the probability continuity equation and current from the
nonrelativistic Schrödinger equation.
\item Find both energy branches of the relativistic scalar plane-wave
dispersion relation.
\item For a complex Klein--Gordon solution, derive its conserved current
and evaluate the density on each energy branch.
\item Explain precisely why this density cannot be interpreted as a
one-particle probability, and why it is instead a charge density.
\end{enumerate}

\SupplementaryExercise{2}{Scalar Mode Recovery, Four-Momentum, and One-Particle Matrix Elements}
  {(Inspired by H. Liu, MIT 8.323, Assignment 2, 2023, Problem 2(b)--(c), pp. 3--4 | H. Liu, MIT 8.323, Assignment 2, 2023, Problem 3(a)--(b), p. 4 | H. Liu, MIT 8.323, Assignment 2, 2023, Problem 3(c), p. 4 | H. Liu, MIT 8.323, Assignment 2, 2023, Problems 2--3, pp. 3--4.)}
  {mit-823-s23-ps2-curated-parent-2}
\begin{enumerate}
\item
At \(t=0\), invert the real scalar mode expansion to express
\(a(\mathbf p)\) and \(a^\dagger(\mathbf p)\) in terms of the spatial
Fourier transforms of \(\phi\) and its conjugate momentum.  Derive the
oscillator commutator from the canonical equal-time commutators.
\item
Starting from the symmetric stress tensor of a real scalar field, show after
normal ordering that
\[
 P^\mu=\int\frac{d^3p}{(2\pi)^3}\,
 p^\mu a^\dagger(\mathbf p)a(\mathbf p).
\]
Verify its action on a one-particle state.
\item
Use the mode expression for \(P^\mu\) to prove
\([P^\mu,\phi(x)]=-i\partial^\mu\phi(x)\), and exponentiate the
commutator to obtain the finite translation law.
\item
With relativistically normalized states
\(\ket p=\sqrt{2p^0}\,a^\dagger(\mathbf p)\ket\Omega\), compute
\(\bra\Omega\phi(x)\ket p\).  Check its Klein--Gordon equation and
Lorentz transformation law.
\end{enumerate}

\SupplementaryExercise{3}{Orbital Lorentz Charge and Spin Zero}
  {(Adapted from H. Liu, MIT 8.323, Assignment 2, 2023, Problem 4.)}
  {mit-823-s23-ps2-curated-parent-3}
\begin{enumerate}
\item Verify that the free real-scalar action is Lorentz invariant and
that \(\phi'(x')=\phi(x)\); derive the corresponding field equation.
\item Find the canonical momentum, Hamiltonian density, and conserved
four-momentum, and express the latter in oscillator variables.
\item Derive
\(M^{\rho\mu\nu}=x^\mu T^{\rho\nu}-x^\nu T^{\rho\mu}\), prove its

intrinsic spin.
\end{enumerate}

\SupplementaryExercise{4}{The Lorentz Algebra in the Vector Representation}
  {(Adapted from B. Mistlberger and K. Zhou, Stanford PHYS-330, Problem Set 2, 2022, Problem 1.)}
  {zhou-stanford-phys330-ps2-2022-curated-parent-1}
Construct the \(4\times4\) generators \(J_i\) and \(K_i\) acting on
four-vectors and verify
\[
 [J_i,J_j]=i\epsilon_{ijk}J_k,\quad
 [J_i,K_j]=i\epsilon_{ijk}K_k,\quad
 [K_i,K_j]=-i\epsilon_{ijk}J_k.
\]
Explain the minus sign in the last commutator.  Then exponentiate one
rotation and one boost, checking directly that both finite matrices preserve
the Minkowski metric.

\SupplementaryExercise{5}{Charged-Scalar Quantization, Charge, Energy, and Locality}
  {(Adapted from B. Mistlberger and K. Zhou, Stanford PHYS-330, Problem Set 2, 2022, Problem 2.)}
  {zhou-stanford-phys330-ps2-2022-curated-parent-2}
\begin{enumerate}
\item Derive the complex scalar equation of motion and canonical
momenta, treating \(\Phi\) and \(\Phi^\dagger\) as independent fields.
\item Quantize the field and show that two commuting oscillator families
\(a(\mathbf p)\), \(b(\mathbf p)\) are required.
\item Derive the Hamiltonian before and after normal ordering, and show

\item Derive the global \(U(1)\) current and charge.
\item Put the normal-ordered charge in oscillator form and determine the

\item Compute the four-momentum and verify that both species have the
same positive mass shell despite opposite charge.
\item Calculate \([\Phi(x),\Phi(y)]\) and
\([\Phi(x),\Phi^\dagger(y)]\), proving microcausality.
\end{enumerate}

\SupplementaryExercise{7}{Representation-Independent Gamma Traces}
  {(Adapted from D. Tong, \emph{Quantum Field Theory: Example Sheet 3}, 2006, Problem 3.)}
  {tong-qft-sheet3-2006-curated-parent-1}
Work in four dimensions using only the Clifford algebra and
\(\operatorname{tr}\mathbf1=4\).  Without choosing a representation,
derive the following fourteen consequences:
\begin{enumerate}
\item \(\gamma^\mu\gamma_\mu=4\);
\item \(\gamma^\mu\gamma^\nu\gamma_\mu=-2\gamma^\nu\);
\item \(\gamma^\mu\gamma^\nu\gamma^\rho\gamma_\mu=4\eta^{\nu\rho}\);
\item
\(\gamma^\mu\gamma^\nu\gamma^\rho\gamma^\sigma\gamma_\mu
=-2\gamma^\sigma\gamma^\rho\gamma^\nu\);
\item \(\operatorname{tr}\gamma^\mu=0\);
\item the trace of every odd product of ordinary gamma matrices vanishes;
\item \(\operatorname{tr}(\gamma^\mu\gamma^\nu)=4\eta^{\mu\nu}\);
\item the trace of four gamma matrices;
\item \((\gamma^5)^2=1\);
\item \(\{\gamma^5,\gamma^\mu\}=0\);
\item \(\operatorname{tr}\gamma^5=0\);
\item
\(\operatorname{tr}(\gamma^5\gamma^\mu\gamma^\nu)=0\);
\item
\(\operatorname{tr}(\gamma^5\gamma^\mu\gamma^\nu\gamma^\rho)=0\);
\item
\(\operatorname{tr}(\gamma^5\gamma^\mu\gamma^\nu\gamma^\rho
\gamma^\sigma)=4i\epsilon^{\mu\nu\rho\sigma}\), with the sign checked
against the stated metric and \(\gamma^5\) convention.
\end{enumerate}

\SupplementaryExercise{8}{Spinor Generators of Lorentz Transformations}
  {(Adapted from D. Tong, \emph{Quantum Field Theory: Example Sheet 3}, 2006, Problem 2.)}
  {tong-qft-sheet3-2006-curated-parent-2}
For an infinitesimal antisymmetric Lorentz parameter, define
\(\mathcal J^{\mu\nu}=-\tfrac i4[\gamma^\mu,\gamma^\nu]\).
Verify that these matrices obey the Lorentz Lie algebra and prove that
\(D(\Lambda)^{-1}\gamma^\rho D(\Lambda)
=\Lambda^\rho{}_\sigma\gamma^\sigma\) infinitesimally.  Keep every sign
fixed by the mostly-plus metric rather than by an explicit matrix basis.

\SupplementaryExercise{9}{Dirac Spinor Norms and Orthogonality}
  {(Adapted from D. Tong, \emph{Quantum Field Theory: Example Sheet 3}, 2006, Problem 4.)}
  {tong-qft-sheet3-2006-curated-parent-3}

\SupplementaryExercise{10}{Completeness of Free Dirac Spinors}
  {(Adapted from D. Tong, \emph{Quantum Field Theory: Example Sheet 3}, 2006, Problem 5.)}

\SupplementaryExercise{11}{Equal-Time Dirac Anticommutators}
  {(Adapted from D. Tong, \emph{Quantum Field Theory: Example Sheet 3}, 2006, Problem 6.)}
  {tong-qft-sheet3-2006-curated-parent-5}
Keep the particle and antiparticle contributions separate until after the
spin sums are inserted.  Use the free Dirac mode expansion to prove
\[
 \{\psi_\ell(\mathbf x,t),\psi_m^\dagger(\mathbf y,t)\}
 =\delta_{\ell m}\delta^3(\mathbf x-\mathbf y),
\]
with the other equal-time anticommutators zero.  Show explicitly how the
change of momentum variable makes the spatial derivative terms cancel.

\SupplementaryExercise{12}{What Fails with Bosonic Spinors}
  {(Adapted from D. Tong, \emph{Quantum Field Theory: Example Sheet 3}, 2006, Problem 8.)}
  {tong-qft-sheet3-2006-curated-parent-6}

\SupplementaryExercise{13}{Massive-Vector Constraints and Polarizations}
  {(Inspired by D. Tong, \emph{Lectures on Quantum Field Theory}, 2006, Section 6.1 | D. Tong, \emph{Lectures on Quantum Field Theory}, 2006, Sections 6.1--6.2.)}
  {tong-qft-free-fields-2006-curated-parent-1}
\begin{enumerate}
\item
Starting from the Proca equation
\(\partial_\mu F^{\mu\nu}-m^2A^\nu=0\) in mostly-plus convention, derive
the transversality condition and the Klein--Gordon equation.  Count the
propagating polarizations.
\item
Construct three polarization vectors for a massive vector with momentum
along the 3-axis and prove
\[
 \sum_{\sigma=-1}^{1}e^\mu(\mathbf p,\sigma)
 e^{\nu *}(\mathbf p,\sigma)
 =\eta^{\mu\nu}+\frac{p^\mu p^\nu}{m^2}.
\]
\item
Find the high-energy behavior of the longitudinal massive-vector
polarization.  Show that its coupling to \(J_\mu\) has a smooth massless
limit when \(p_\mu J^\mu=0\), and determine the leading residual behavior.
\end{enumerate}

\SupplementaryExercise{14}{Helicity, Gauge Shifts, and the Covariant Field Strength}
  {(Inspired by D. Tong, \emph{Lectures on Quantum Field Theory}, 2006, Section 6.2 | D. Tong, \emph{Lectures on Quantum Field Theory}, 2006, Sections 6.1--6.2.)}
  {tong-qft-free-fields-2006-curated-parent-2}
\begin{enumerate}
\item
For a massless spin-one polarization, show that the translation part of the
little group changes \(e^\mu\) by a multiple of \(p^\mu\).  Prove that a
Lorentz-covariant emission amplitude is insensitive to this change exactly
when the current is conserved.
\item
Define
\(f^{\mu\nu}(p,\sigma)=i(p^\mu e^\nu-p^\nu e^\mu)\)
for helicity \(\sigma=\pm1\).  Show that it is invariant under
\(e^\mu\to e^\mu+\alpha p^\mu\), obeys the source-free Maxwell identities,
and carries the two helicities in the
\((1,0)\oplus(0,1)\) representation.
\item
In Coulomb gauge, derive
\[
 [A_i(\mathbf x,t),E_j(\mathbf y,t)]
 =i\left(\delta_{ij}-\frac{\partial_i\partial_j}{\nabla^2}\right)
 \delta^3(\mathbf x-\mathbf y).
\]
Explain how this projector implements the two physical photon degrees of
freedom.
\end{enumerate}

\SupplementaryExercise{15}{Antiparticles and the Locality of Charged Fields}
  {(Inspired by D. Harlow, \emph{Lecture Notes for Physics 8.323}, 2024, Section 3.3, pp. 35--36 | D. Harlow, \emph{Lecture Notes for Physics 8.323}, 2024, Section 3.5, pp. 37--38 | D. Harlow, \emph{Lecture Notes for Physics 8.323}, 2024, Section 3.4, pp. 35--37.)}
  {harlow-qft1-fields-2024-curated-parent-1}
\begin{enumerate}
\item
At fixed time, solve for the positive-frequency part \(\phi^{(+)}\) of a
real scalar field in terms of \(\phi\) and \(\pi\).  Show that the operator
\(\sqrt{m^2-\nabla^2}\) makes this separation nonlocal in space even though
the full field is local.
\item
Try to build a charged scalar field from particle annihilation operators and
their adjoints alone.  Show why a local field of definite charge instead
requires an independent antiparticle oscillator with the opposite charge.
\item
Write the free scalar commutator as the Pauli--Jordan integral.  Prove its
vanishing at spacelike separation by evaluating it at equal time in a
suitable Lorentz frame, and explain why the separate positive-frequency
function does not vanish.
\end{enumerate}

\SupplementaryExercise{16}{CPT, Local Interactions, and Spin--Statistics}
  {(Inspired by D. Harlow, \emph{Lecture Notes for Physics 8.323}, 2024, Section 6.1, pp. 69--71 | D. Harlow, \emph{Lecture Notes for Physics 8.323}, 2024, Section 6.2, pp. 71--74.)}
  {harlow-qft1-fields-2024-curated-parent-2}
\begin{enumerate}
\item
Let \(\Theta\) be antiunitary and implement CPT on local fields.  Show how a
vacuum correlator transforms, including complex conjugation and reversal of
operator order.  Explain why locality is needed to compare the transformed
correlator with the original one.
\item
Using the CPT rules for scalar, vector, and Dirac bilinears, prove that a
local Hermitian Lorentz-scalar interaction density satisfies
\(\Theta\mathcal H_I(x)\Theta^{-1}=\mathcal H_I(-x)\).
Deduce invariance of \(H_I(0)\).
\item
For a covariant free field of spin \(j\), compare its two-point function at
spacelike-separated points before and after a rotation that exchanges the
points.  Show how positivity and locality require the exchange sign
\((-1)^{2j}\), and state the resulting statistics assignment.
\end{enumerate}
\end{enumerate}
