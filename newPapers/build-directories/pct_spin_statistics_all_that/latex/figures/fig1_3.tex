\begin{figure}[t]
  \centering
  \begin{tikzpicture}[x=1.5cm,y=1.5cm,>=Stealth,
    line cap=round,line join=round,
    shell/.style={line width=.9pt},
    cone/.style={line width=.9pt},
    hatch/.style={line width=.35pt},
    legend/.style={font=\small,anchor=west}]
    % Two-particle continuum: vertical hatch above the threshold at 2m.
    \begin{scope}
      \clip plot[domain=-2.99:2.99,samples=140] (\x,{sqrt(\x*\x+4)})
        -- (2.99,3.62) -- (-2.99,3.62) -- cycle;
      \foreach \x in {-2.96,-2.80,...,3.00}
        \draw[hatch] (\x,1.90) -- (\x,3.70);
    \end{scope}

    % Three-particle continuum: the same region crossed by horizontal rules
    % above the threshold at 3m.
    \begin{scope}
      \clip plot[domain=-1.99:1.99,samples=140] (\x,{sqrt(\x*\x+9)})
        -- (1.99,3.62) -- (-1.99,3.62) -- cycle;
      \foreach \y in {2.98,3.11,...,3.64}
        \draw[hatch] (-2.05,\y) -- (2.05,\y);
    \end{scope}

    % The light cone and the three mass hyperboloids.
    \draw[cone] (-3.25,3.25) -- (0,0) -- (3.25,3.25);
    \draw[cone] (-0.36,-0.36) -- (0,0) -- (0.36,-0.36);
    \draw[shell] plot[domain=-2.90:2.90,samples=160] (\x,{sqrt(\x*\x+1)});
    \draw[shell] plot[domain=-2.99:2.99,samples=160] (\x,{sqrt(\x*\x+4)});
    \draw[shell] plot[domain=-1.99:1.99,samples=160] (\x,{sqrt(\x*\x+9)});

    % Axes, the vacuum, and the one-particle labels.
    \draw[->,line width=.6pt] (0,0) -- (0,3.88) node[above right=-1pt] {$p^0$};
    \draw[->,line width=.6pt] (0,0) -- (1.38,0);
    \fill (0,0) circle (1.6pt);
    \node[font=\small] at (1.22,-0.17) {$\mathbf{p}$};
    \node[font=\small] at (0.64,-0.24) {$\ket{\Omega}$};

    \draw[line width=.45pt] (1.30,0.60) -- (0.52,1.12);
    \node[font=\small] at (1.78,0.42) {$p^2=-m^2$};
    \node[legend] at (2.50,0.62) {one-particle};
    \node[legend] at (2.50,0.32) {states $[m,0]$};

    % Key to the two hatch patterns.
    \foreach \x in {-3.80,-3.68,...,-3.32}
      \draw[line width=.5pt] (\x,1.50) -- (\x,1.85);
    \node[legend] at (-3.85,1.20) {two-particle states};
    \node[legend] at (-3.85,0.88) {$([m,0]\otimes[m,0])_{\mathrm{s}}$};
    \foreach \y in {0.34,0.46,0.58}
      \draw[line width=.5pt] (-3.80,\y) -- (-3.34,\y);
    \node[legend] at (-3.85,-0.02) {three-particle states};
    \node[legend] at (-3.85,-0.34)
      {$([m,0]\otimes[m,0]\otimes[m,0])_{\mathrm{s}}$};
  \end{tikzpicture}
  \caption{The spectrum of a theory of neutral scalar mesons of mass $m$
  without bound states.}
  \label{fig:1-3}
\end{figure}%
