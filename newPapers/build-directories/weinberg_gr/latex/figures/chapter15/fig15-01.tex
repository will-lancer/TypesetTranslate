% Source: Weinberg GR, physical PDF p. 506 (printed p. 482).
% Coverage: Figure 15.1, matter-dominated Friedmann solutions.
% Status: source-reviewed and compile-clean; redrawn in TikZ with the
%         binding scale factor a.
% Geometry: the three curves are the exact matter-dominated solutions, in the
%           normalization that makes them agree as t -> 0:
%             k = +1   a = 1 - cos(theta),   t = theta - sin(theta)
%             k = -1   a = cosh(psi) - 1,    t = sinh(psi) - psi
%             k =  0   a = (6t)^(2/3)/2 = 1.65096 t^(2/3)
%           The k = 0 branch is sampled in u = t^(1/3) so that the curve stays
%           smooth through the origin, where da/dt diverges.
%           Marker positions come from the q values quoted in the text:
%           q = (1 + cos theta)^(-1) on k = +1 and q = (1 + cosh psi)^(-1) on
%           k = -1, so
%             q = 1     theta = pi/2      q = 0.3   cosh psi = 7/3
%             q = 2     theta = 2pi/3     q = 0.25  cosh psi = 3
%             q = 7.5   cos theta = -13/15  q = 0.21  cosh psi = 79/21
%             q = inf   theta = pi
%           and q = 1/2 at the common origin.  Measured against the source
%           plot, every marker lands within about one percent of the box
%           width, which confirms this normalization is the one drawn there.
% Uncertainties: none.

\begingroup
\renewcommand{\thefigure}{15.1}
\begin{figure}[htbp]
  \centering
  \begin{tikzpicture}[x=1.45cm,y=1.81cm,>=Stealth]
    \draw[->] (0,0) -- (6.90,0) node[right,inner sep=3pt] {\(t\)};
    \draw[->] (0,0) -- (0,3.95) node[above,inner sep=3pt] {\(a\)};

    % k = -1: a = cosh(psi) - 1, t = sinh(psi) - psi.
    \draw[thick]
      plot[domain=0:2.06,samples=120,variable=\p]
        ({0.5*(exp(\p)-exp(-\p))-\p},{0.5*(exp(\p)+exp(-\p))-1});
    \node[anchor=south east,inner sep=1pt] at (1.95,3.02) {\(k=-1\)};

    % k = 0: a = 1.65096 t^(2/3), sampled in u = t^(1/3).
    \draw[thick]
      plot[domain=0:1.4345,samples=90,variable=\u]
        ({\u*\u*\u},{1.65096*\u*\u});
    \node[anchor=south east,inner sep=1pt] at (3.02,3.42) {\(k=0\)};

    % k = +1: the cycloid a = 1 - cos(theta), t = theta - sin(theta).
    \draw[thick]
      plot[domain=0:6.2832,samples=200,variable=\t]
        ({\t-sin(\t r)},{1-cos(\t r)});
    \node[anchor=west,inner sep=1pt] at (5.00,1.95) {\(k=+1\)};

    % Values of q along the k = -1 branch.
    \foreach \x/\y/\lab in {
      0.6096/1.3333/{0.3},
      1.0657/2.0000/{0.25},
      1.6244/2.7619/{0.21}
    }{
      \fill[white] (\x,\y) circle (2.2pt);
      \draw (\x,\y) circle (2.2pt);
      \node[anchor=east,inner sep=4pt] at (\x,\y) {\(\lab\)};
    }

    % Values of q along the k = +1 branch.
    \foreach \x/\y in {0.5708/1.0000, 1.2284/1.5000,
                       2.1180/1.8660, 3.1416/2.0000}{
      \fill[white] (\x,\y) circle (2.2pt);
      \draw (\x,\y) circle (2.2pt);
    }
    \node[anchor=north west,inner sep=3.5pt] at (0.5708,1.0000) {\(1\)};
    \node[anchor=north west,inner sep=3.5pt] at (1.2284,1.5000) {\(2\)};
    \node[anchor=north,inner sep=3.5pt] at (2.1180,1.6000) {\(7.5\)};
    \node[anchor=north,inner sep=6pt] at (3.1416,2.0000) {\(\infty\)};

    % The common origin, where q = 1/2 for every branch.
    \fill[white] (0,0) circle (2.2pt);
    \draw (0,0) circle (2.2pt);
    \node[anchor=south west,inner sep=2pt] at (0.12,0.05) {\(0.5\)};
  \end{tikzpicture}
  \caption{Solutions of Einstein's equations for a Robertson--Walker
  universe with curvature \(k=+1\), \(k=0\), and \(k=-1\). The numbers
  along the curves \(k=\pm1\) give the values of the deceleration parameter
  \(q\) at various epochs.}
  \label{fig:15.1}
\end{figure}
