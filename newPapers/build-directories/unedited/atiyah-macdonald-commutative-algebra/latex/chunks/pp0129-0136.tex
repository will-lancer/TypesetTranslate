\noindent\textit{Proposition 11.8.\amtarget{11.8} Let $A$, $\mathfrak m$, $\mathfrak q$ be as before. Let $M$ be a finitely-generated $A$-module, $x\in A$ a non-zero-divisor in $M$ and $M'=M/xM$. Then}
\[
 \deg\chi_{\mathfrak q}^{M'}\leq \deg\chi_{\mathfrak q}^{M}-1.
\]
\textit{Proof.} Let $N=xM$; then $N\cong M$ as $A$-modules, by virtue of the assumption on $x$. Let $N_n=N\cap\mathfrak q^nM$. Then we have exact sequences
\[
 0\longrightarrow N/N_n\longrightarrow M/\mathfrak q^nM\longrightarrow M'/\mathfrak q^nM'\longrightarrow 0.
\]
Hence, if $g(n)=l(N/N_n)$, we have
\[
 g(n)-\chi_{\mathfrak q}^{M}(n)+\chi_{\mathfrak q}^{M'}(n)=0
\]
for all large $n$. Now by Artin--Rees \amref{10.9}, $(N_n)$ is a stable $\mathfrak q$-filtration of $N$. Since $N\cong M$, \amref{11.4} iii then implies that $g(n)$ and $\chi_{\mathfrak q}^{M}(n)$ have the same leading term; hence the result. \proofbox

\medskip
\noindent\textit{Corollary 11.9.\amtarget{11.9} If $A$ is a Noetherian local ring, $x$ a non-zero-divisor in $A$, then}
\[
 d(A/(x))\leq d(A)-1.
\]
\textit{Proof.} Put $M=A$ in \amref{11.8}. \proofbox

We can now prove the crucial result:

\medskip
\noindent\textit{Proposition 11.10.\amtarget{11.10} $d(A)\geq\dim A$.}

\textit{Proof.} By induction on $d=d(A)$. If $d=0$ then $l(A/\mathfrak m^n)$ is constant for all large $n$, hence $\mathfrak m^n=\mathfrak m^{n+1}$ for some $n$, hence $\mathfrak m^n=0$ by Nakayama's lemma \amref{2.6}. Thus $A$ is an Artin ring and $\dim A=0$.

Suppose $d>0$ and let $\mathfrak p_0\subset\mathfrak p_1\subset\cdots\subset\mathfrak p_r$ be any chain of prime ideals in $A$. Let $x\in\mathfrak p_1$, $x\notin\mathfrak p_0$; let $A'=A/\mathfrak p_0$, and let $x'$ be the image of $x$ in $A'$. Then $x'\neq0$, and $A'$ is an integral domain, hence by \amref{11.9} we have
\[
 d(A'/(x'))\leq d(A')-1.
\]
Also, if $\mathfrak m'$ is the maximal ideal of $A'$, $A'/\mathfrak m'^n$ is a homomorphic image of $A/\mathfrak m^n$, hence $l(A/\mathfrak m^n)\geq l(A'/\mathfrak m'^n)$ and therefore $d(A)\geq d(A')$. Consequently
\[
 d(A'/(x'))\leq d(A)-1=d-1.
\]
Hence, by the inductive hypothesis, the length of any chain of prime ideals in $A'/(x')$ is $\leq d-1$. But the images of $\mathfrak p_1,\ldots,\mathfrak p_r$ in $A'/(x')$ form a chain of length $r-1$, hence $r-1\leq d-1$ and consequently $r\leq d$. Hence $\dim A\leq d$. \proofbox

\medskip
\noindent\textit{Corollary 11.11.\amtarget{11.11} If $A$ is a Noetherian local ring, $\dim A$ is finite.} \proofbox

If $A$ is any ring, $\mathfrak p$ a prime ideal in $A$, then the \textit{height} of $\mathfrak p$ is defined to be the supremum of chains of prime ideals $\mathfrak p_0\subset\mathfrak p_1\subset\cdots\subset\mathfrak p_r=\mathfrak p$ which end at $\mathfrak p$: by \amref{3.13}, height $\mathfrak p=\dim A_{\mathfrak p}$. Hence, from \amref{11.11}:

\medskip
\noindent\textit{Corollary 11.12.\amtarget{11.12} In a Noetherian ring every prime ideal has finite height, and therefore the set of prime ideals in a Noetherian ring satisfies the descending chain condition.} \proofbox

\textit{Remark.} Likewise we may define the \textit{depth} of $\mathfrak p$, by considering chains of prime ideals which start at $\mathfrak p$: clearly depth $\mathfrak p=\dim A/\mathfrak p$. But the depth of a prime ideal, even in a Noetherian ring, may be infinite (unless the ring is local). See Exercise 4.

\medskip
\noindent\textit{Proposition 11.13.\amtarget{11.13} Let $A$ be a Noetherian local ring of dimension $d$. Then there exists an $\mathfrak m$-primary ideal in $A$ generated by $d$ elements $x_1,\ldots,x_d$, and therefore $\dim A\geq\delta(A)$.}

\textit{Proof.} Construct $x_1,\ldots,x_d$ inductively in such a way that every prime ideal containing $(x_1,\ldots,x_i)$ has height $\geq i$, for each $i$. Suppose $i>0$ and $x_1,\ldots,x_{i-1}$ constructed. Let $\mathfrak p_j$ $(1\leq j\leq s)$ be the minimal prime ideals (if any) of $(x_1,\ldots,x_{i-1})$ which have height exactly $i-1$. Since $i-1<d=\dim A=\operatorname{height}\mathfrak m$, we have $\mathfrak m\neq\mathfrak p_j$ $(1\leq j\leq s)$, hence $\mathfrak m\neq\bigcup_{j=1}^s\mathfrak p_j$ by \amref{1.11}. Choose $x_i\in\mathfrak m$, $x_i\notin\bigcup\mathfrak p_j$, and let $\mathfrak q$ be any prime containing $(x_1,\ldots,x_i)$. Then $\mathfrak q$ contains some minimal prime ideal $\mathfrak p$ of $(x_1,\ldots,x_{i-1})$. If $\mathfrak p=\mathfrak p_j$ for some $j$, we have $x_i\in\mathfrak q$, $x_i\notin\mathfrak p$, hence $\mathfrak q\supset\mathfrak p$ and therefore height $\mathfrak q\geq i$; if $\mathfrak p\neq\mathfrak p_j$ $(1\leq j\leq s)$, then height $\mathfrak p\geq i$, hence height $\mathfrak q\geq i$. Thus every prime ideal containing $(x_1,\ldots,x_i)$ has height $\geq i$.

Consider then $(x_1,\ldots,x_d)$. If $\mathfrak p$ is a prime ideal of this ideal, $\mathfrak p$ has height $\geq d$, hence $\mathfrak p=\mathfrak m$ (for $\mathfrak p\subset\mathfrak m\Rightarrow\operatorname{height}\mathfrak p<\operatorname{height}\mathfrak m=d$). Hence the ideal $(x_1,\ldots,x_d)$ is $\mathfrak m$-primary. \proofbox

\medskip
\noindent\textit{Theorem 11.14.\amtarget{11.14} (Dimension theorem.) For any Noetherian local ring $A$ the following three integers are equal:}
\begin{enumerate}
\item[i)] \textit{the maximum length of chains of prime ideals in $A$;}
\item[ii)] \textit{the degree of the characteristic polynomial $\chi_{\mathfrak m}(n)=l(A/\mathfrak m^n)$;}
\item[iii)] \textit{the least number of generators of an $\mathfrak m$-primary ideal of $A$.}
\end{enumerate}
\textit{Proof.} \amref{11.7}, \amref{11.10}, \amref{11.13}. \proofbox

\textit{Example.} Let $A$ be the polynomial ring $k[x_1,\ldots,x_n]$ localized at the maximal ideal $\mathfrak m=(x_1,\ldots,x_n)$. Then $G_{\mathfrak m}(A)$ is a polynomial ring in $n$ indeterminates and so its Poincar\'e series is $(1-t)^{-n}$. Hence, using the equivalence of (i) and (ii) in \amref{11.14}, we deduce that $\dim A_{\mathfrak m}=n$.

\medskip
\noindent\textit{Corollary 11.15.\amtarget{11.15} $\dim A\leq\dim_k(\mathfrak m/\mathfrak m^2)$.}

\textit{Proof.} If $x_i\in\mathfrak m$ $(1\leq i\leq s)$ are such that their images in $\mathfrak m/\mathfrak m^2$ form a basis of this vector space, then the $x_i$ generate $\mathfrak m$ by \amref{2.8}; hence $\dim_k(\mathfrak m/\mathfrak m^2)=s\geq\dim A$ by \amref{11.13}. \proofbox

\medskip
\noindent\textit{Corollary 11.16.\amtarget{11.16} Let $A$ be a Noetherian ring, $x_1,\ldots,x_r\in A$. Then every minimal ideal $\mathfrak p$ belonging to $(x_1,\ldots,x_r)$ has height $\leq r$.}

\textit{Proof.} In $A_{\mathfrak p}$ the ideal $(x_1,\ldots,x_r)$ becomes $\mathfrak p^e$-primary, hence $r\geq\dim A_{\mathfrak p}=\operatorname{height}\mathfrak p$. \proofbox

\medskip
\noindent\textit{Corollary 11.17.\amtarget{11.17} (Krull's principal ideal theorem). Let $A$ be a Noetherian ring and let $x$ be an element of $A$ which is neither a zero-divisor nor a unit. Then every minimal prime ideal $\mathfrak p$ of $(x)$ has height 1.}

\textit{Proof.} By \amref{11.16}, height $\mathfrak p\leq1$. If height $\mathfrak p=0$, then $\mathfrak p$ is a prime ideal belonging to 0, hence every element of $\mathfrak p$ is a zero-divisor by \amref{4.7}: contradiction, since $x\in\mathfrak p$. \proofbox

\medskip
\noindent\textit{Corollary 11.18.\amtarget{11.18} Let $A$ be a Noetherian local ring, $x$ an element of $\mathfrak m$ which is not a zero-divisor. Then $\dim A/(x)=\dim A-1$.}

\textit{Proof.} Let $d=\dim A/(x)$. By \amref{11.9} and \amref{11.14} we have $d\leq\dim A-1$. On the other hand, let $x_i$ $(1\leq i\leq d)$ be elements of $\mathfrak m$ whose images in $A/(x)$ generate an $\mathfrak m/(x)$-primary ideal. Then the ideal $(x,x_1,\ldots,x_d)$ in $A$ is $\mathfrak m$-primary, hence $d+1\geq\dim A$. \proofbox

\medskip
\noindent\textit{Corollary 11.19.\amtarget{11.19} Let $\widehat A$ be the $\mathfrak m$-adic completion of $A$. Then $\dim\widehat A=\dim A$.}

\textit{Proof.} $A/\mathfrak m^n\cong\widehat A/\widehat{\mathfrak m}^{,n}$ from \amref{10.15}, hence $\chi_{\mathfrak m}(n)=\chi_{\widehat{\mathfrak m}}(n)$. \proofbox

If $x_1,\ldots,x_d$ generate an $\mathfrak m$-primary ideal, and $d=\dim A$, we call $x_1,\ldots,x_d$ a \textit{system of parameters}. They have a certain independence property described in the following proposition.

\medskip
\noindent\textit{Proposition 11.20.\amtarget{11.20} Let $x_1,\ldots,x_d$ be a system of parameters for $A$ and let $\mathfrak q=(x_1,\ldots,x_d)$ be the $\mathfrak m$-primary ideal generated by them. Let $f(t_1,\ldots,t_d)$ be a homogeneous polynomial of degree $s$ with coefficients in $A$, and assume that}
\[
 f(x_1,\ldots,x_d)\in\mathfrak q^{s+1}.
\]
\textit{Then all the coefficients of $f$ lie in $\mathfrak m$.}

\textit{Proof.} Consider the epimorphism of graded rings
\[
 \alpha:(A/\mathfrak q)[t_1,\ldots,t_d]\longrightarrow G_{\mathfrak q}(A)
\]
given by $t_i\mapsto\bar x_i$, where $t_i$ are indeterminates and $\bar x_i$ is $x_i\bmod\mathfrak q$. The hypothesis on $f$ implies that $\bar f(t_1,\ldots,t_d)$ (the reduction of $f\bmod\mathfrak q$) is in the kernel of $\alpha$. Assume if possible that some coefficient of $f$ is a unit, then $\bar f$ is not a zero-divisor (cf. Chapter 1, Exercise 3). Then we have
\begin{align*}
 d(G_{\mathfrak q}(A))&\leq d((A/\mathfrak q)[t_1,\ldots,t_d]/(\bar f)) &&\text{because $\bar f\in\operatorname{Ker}(\alpha)$}\\
 &=d((A/\mathfrak q)[t_1,\ldots,t_d])-1 &&\text{by \amref{11.3}}\\
 &=d-1 &&\text{by the example following \amref{11.3}.}
\end{align*}
But $d(G_{\mathfrak q}(A))=d$ by the main theorem \amref{11.14}. This gives the required contradiction. \proofbox

This proposition takes a simple form if $A$ contains a field $k$ mapping isomorphically onto the residue field $A/\mathfrak m$:

\medskip
\noindent\textit{Corollary 11.21.\amtarget{11.21} If $k\subset A$ is a field mapping isomorphically onto $A/\mathfrak m$ and if $x_1,\ldots,x_d$ is a system of parameters, then $x_1,\ldots,x_d$ are algebraically independent over $k$.}

\textit{Proof.} Assume $f(x_1,\ldots,x_d)=0$ where $f$ is a polynomial with coefficients in $k$. If $f\neq0$ we can write $f=f_s+$ higher terms, where $f_s$ is homogeneous of degree $s$ and $f_s\neq0$. Apply \amref{11.20} to $f_s$ and we deduce that $f_s$ has all its coefficients in $\mathfrak m$. Since $f_s$ has coefficients in $k$ this implies $f_s\equiv0$, a contradiction. Hence $x_1,\ldots,x_d$ are algebraically independent over $k$. \proofbox

\subsection{Regular local rings}

In algebraic geometry there is an important distinction between \textit{singular} and \textit{non-singular} points (see Exercise 1). The local rings of non-singular points have as their generalization (to the non-geometric case) what are called \textit{regular} local rings: these are rings satisfying any of the (equivalent) conditions i)--iii) of the next theorem.

\medskip
\noindent\textit{Theorem 11.22.\amtarget{11.22} Let $A$ be a Noetherian local ring of dimension $d$, $\mathfrak m$ its maximal ideal, $k=A/\mathfrak m$. Then the following are equivalent:}
\begin{enumerate}
\item[i)] $G_{\mathfrak m}(A)\cong k[t_1,\ldots,t_d]$ \textit{where the $t_i$ are independent indeterminates;}
\item[ii)] $\dim_k(\mathfrak m/\mathfrak m^2)=d$;
\item[iii)] $\mathfrak m$ \textit{can be generated by $d$ elements.}
\end{enumerate}
\textit{Proof.} i) $\Rightarrow$ ii) is clear. ii) $\Rightarrow$ iii) by \amref{2.8}: see the proof of \amref{11.15}. iii) $\Rightarrow$ i): let $\mathfrak m=(x_1,\ldots,x_d)$, then by \amref{11.20} the map $\alpha:k[\bar x_1,\ldots,\bar x_d]\longrightarrow G_{\mathfrak m}(A)$ is an isomorphism of graded rings. \proofbox

A regular local ring is necessarily an \textit{integral domain}: this is a consequence of the following more general result.

\medskip
\noindent\textit{Lemma 11.23.\amtarget{11.23} Let $A$ be a ring, $\mathfrak a$ an ideal of $A$ such that $\bigcap_n\mathfrak a^n=0$. Suppose that $G_{\mathfrak a}(A)$ is an integral domain. Then $A$ is an integral domain.}

\textit{Proof.} Let $x,y$ be non-zero elements of $A$. Then since $\bigcap\mathfrak a^n=0$ there exist integers $r,s\geq0$ such that $x\in\mathfrak a^r$, $x\notin\mathfrak a^{r+1}$, $y\in\mathfrak a^s$, $y\notin\mathfrak a^{s+1}$. Let $\bar x,\bar y$ denote the images of $x,y$ in $G_r(A)$, $G_s(A)$ respectively. Then $\bar x\neq0$, $\bar y\neq0$, hence $\overline{xy}=\bar x\cdot\bar y\neq0$, hence $xy\neq0$. \proofbox

Hence by \amref{9.2} the regular local rings of dimension 1 are precisely the discrete valuation rings.

It can also be shown that if $A$ is a local ring and $G_{\mathfrak m}(A)$ is an integrally closed integral domain, then $A$ is integrally closed. It follows that a regular local ring is integrally closed; but there are integrally closed local domains of dimension $>1$ which are not regular.

\medskip
\noindent\textit{Proposition 11.24.\amtarget{11.24} Let $A$ be a Noetherian local ring. Then $A$ is regular if and only if $\widehat A$ is regular.}

\textit{Proof.} By \amref{10.16}, \amref{10.26} and \amref{11.19} we know that $\widehat A$ is a Noetherian local ring of the same dimension as $A$ and with $\widehat{\mathfrak m}$ as maximal ideal. Now use \amref{10.22} which asserts that $G_{\mathfrak m}(A)\cong G_{\widehat{\mathfrak m}}(\widehat A)$ and the result follows. \proofbox

\textit{Remarks.} 1) It follows from what we have said above that $\widehat A$ is also an integral domain. Geometrically speaking this means that (locally)
\[
 \text{non-singularity}\Rightarrow\text{analytic irreducibility}
\]
or that, at a non-singular point, there is only one analytic ``branch''.

2) If $A$ contains a field $k$ mapping isomorphically onto $A/\mathfrak m$ (the geometric case) then \amref{11.22} implies that $\widehat A$ is a formal power series ring over $k$ in $d$ indeterminates. Thus the completions of local rings of non-singular points on $d$-dimensional varieties over $k$ are all isomorphic.

\textit{Example.} Let $A=k[x_1,\ldots,x_n]$ ($k$ any field, $x_i$ independent indeterminates); let $\mathfrak m=(x_1,\ldots,x_n)$. Then $A_{\mathfrak m}$ (the local ring of affine space $k^n$ at the origin) is a regular local ring: for $G_{\mathfrak m}(A)$ is a polynomial ring in $n$ variables.

\subsection{Transcendental dimension}

We shall conclude this brief treatment of dimension theory by showing how the dimension of local rings connects up with the dimension of varieties defined classically in terms of the function field.

Assume for simplicity that $k$ is an algebraically closed field and let $V$ be an irreducible affine variety over $k$. Thus the coordinate ring $A(V)$ is of the form
\[
 A(V)=k[x_1,\ldots,x_n]/\mathfrak p
\]
where $\mathfrak p$ is a prime ideal. The field of fractions of the integral domain $A(V)$ is called the field of rational functions on $V$ and is denoted by $k(V)$. It is a finitely-generated extension of $k$ and so has a finite transcendence degree over $k$---the maximum number of algebraically independent elements. This number is defined to be the \textit{dimension} of $V$. Now recall that, by the Nullstellensatz, the points of $V$ correspond bijectively with the maximal ideals of $A(V)$. If $P$ is a point with maximal ideal $\mathfrak m$ we shall call $\dim A(V)_{\mathfrak m}$ the \textit{local dimension} of $V$ at $P$. We propose to prove

\medskip
\noindent\textit{Theorem 11.25.\amtarget{11.25} For any irreducible variety $V$ over $k$ the local dimension of $V$ at any point is equal to $\dim V$.}

\textit{Remark.} We already know by \amref{11.21} that $\dim V\geq\dim A_{\mathfrak m}$ for all $\mathfrak m$. The problem is to prove the opposite inequality, and for this purpose the main lemma is:

\medskip
\noindent\textit{Lemma 11.26.\amtarget{11.26} Let $B\subseteq A$ be integral domains with $B$ integrally closed and $A$ integral over $B$. Let $\mathfrak m$ be a maximal ideal of $A$, and let $\mathfrak n=\mathfrak m\cap B$. Then $\mathfrak n$ is maximal and $\dim A_{\mathfrak m}=\dim B_{\mathfrak n}$.}

\textit{Proof.} This is an easy consequence of the results of Chapter 5. First $\mathfrak n$ is maximal by \amref{5.8}. Next if
\begin{equation}
 \mathfrak m\supset\mathfrak q_1\supset\mathfrak q_2\supset\cdots\supset\mathfrak q_d
 \tag{1}
\end{equation}
is a strict chain of primes in $A$, its intersection with $B$ is by \amref{5.9} a strict chain of primes
\begin{equation}
 \mathfrak n\supset\mathfrak p_1\supset\mathfrak p_2\supset\cdots\supset\mathfrak p_d.
 \tag{2}
\end{equation}
This proves $\dim B_{\mathfrak n}\geq\dim A_{\mathfrak m}$. Conversely given the strict chain (2) we can, by \amref{5.16}, lift this to a chain (1) (necessarily strict): thus $\dim A_{\mathfrak m}\geq\dim B_{\mathfrak n}$. \proofbox

We can now proceed to:

\textit{Proof of \amref{11.25}.} By the Normalization Lemma (Chapter 5, Exercise 16), we can find a polynomial ring $B=k[x_1,\ldots,x_d]$ contained in $A(V)$ such that $d=\dim V$ and $A(V)$ is integral over $B$. Since $B$ is integrally closed (remark following \amref{5.12}) we can apply \amref{11.26} and this reduces our task to proving \amref{11.25} for the ring $B$, i.e. for affine space. But any point of affine space can be taken as the origin of coordinates and, as we have already seen, $k[x_1,\ldots,x_d]$ localized at the maximal ideal $(x_1,\ldots,x_d)$ is a local ring of dimension $d$. \proofbox

\medskip
\noindent\textit{Corollary 11.27.\amtarget{11.27} For every maximal ideal $\mathfrak m$ of $A(V)$ we have}
\[
 \dim A(V)=\dim A(V)_{\mathfrak m}.
\]
\textit{Proof.} By definition we have $\dim A(V)=\sup_{\mathfrak m}\dim A(V)_{\mathfrak m}$. But by \amref{11.25} all $A(V)_{\mathfrak m}$ have the same dimension. \proofbox
