\chapterbackmatter{Problems}

\begin{enumerate}
\item What sort of term would need to be added to the action for Goldstone
boson fields in four-dimensional Euclidean spacetime to make it possible to
have topologically non-trivial field configurations at which the action is
stationary?

\item Consider a theory of scalar fields in six space dimensions in which
the chiral symmetry $SU(2)\times SU(2)$ of the Lagrangian is spontaneously
broken to the $SU(2)$ of isospin. Suppose that enough higher derivative
terms are added to the Lagrangian so that skyrmions are stabilized. What
sort of conservation law do these skyrmions obey? (Hint: Note that, as
shown in \hyperref[sec:2.7]{Section 2.7}, $SU(2)$ is topologically the same
as $S_3$.)

\item Show that all $\pi_n(\mathcal M)$ for $n>1$ and arbitrary manifolds
$\mathcal M$ are Abelian.

\item What is the coupling-constant dependence of the contribution of
instantons of unit winding number in an $SU(4)$ gauge theory?

\item Derive a formula for the axion mass in the case in which $m_u$,
$m_d$, and $m_s$ are all small.

\item Consider the theory of a real scalar field $\phi$ with Lagrangian
density
\[
 \mathcal L=-\frac12\partial_\mu\phi\,\partial^\mu\phi
 +\frac12m^2\phi^2-\frac14\lambda\phi^4-g\phi,
\]
where $m^2$ and $\lambda$ are positive, and $g$ is very small. Calculate
the exponential suppression factor $\exp(-B)$ in the rate of decay of the
metastable vacuum state in terms of $m^2$, $\lambda$, and $g$.
\end{enumerate}
