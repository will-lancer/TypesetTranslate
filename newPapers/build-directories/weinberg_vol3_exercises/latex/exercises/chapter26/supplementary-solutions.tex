\begin{enumerate}
  \SupplementarySolution{1}{Off-Shell Superfields and Closure}
  \begin{enumerate}
    \item \emph{Counting a General Real Superfield.}
    Reality makes \(C,M,N,D,V_\mu\) real: there are four scalar components
      plus four vector components, hence eight bosonic components.  Each of
      \(\omega,\lambda\) is Majorana and has four real components, hence eight
      fermionic components.  No equation has been used.  In particular,
      \(V_\mu\) is not yet transverse and \(M,N,D,\lambda\) are not eliminated;
      the equality is therefore an off-shell statement.
    \item \emph{Closure on the Wess--Zumino Scalar.}
    From \(\delta A=\bar\alpha\psi\),
      \[
      \begin{aligned}
        [\delta_\alpha,\delta_\beta]A
         ={}&\bar\beta\!
           \left[\partial_\mu(A+\ii\gamma_5B)\gamma^\mu\alpha
                 +(F-\ii\gamma_5G)\alpha\right]-(\alpha\leftrightarrow\beta).
      \end{aligned}
      \]
      Majorana interchange identities make the \(F,G\) terms and the
      \(\partial B\) terms cancel, while the vector bilinear is antisymmetric.
      Thus
      \[
        [\delta_\alpha,\delta_\beta]A
          =2(\bar\beta\gamma^\mu\alpha)\partial_\mu A.
      \]
      Direct substitution gives the same translation on
      \(B,\psi,F,G\).  The auxiliaries are precisely what permits this closure
      without imposing a field equation on \(\psi\).
  \end{enumerate}

  \SupplementarySolution{2}{Covariant Superderivatives, Chiral Coordinates, and Integration}
  \begin{enumerate}
    \item \emph{Why the Covariant Superderivative Works.}
    Write \(\SuperQ=A+B\) and \(\SuperD=A-B\), where \(A\) differentiates
      with respect to \(\theta\) and \(B\) is linear in \(\theta\partial_\mu\).
      Then \(\{A,A\}=\{B,B\}=0\), while the two cross terms cancel in
      \(\{\SuperD,\SuperQ\}\) and add in
      \(\{\SuperD,\bar{\SuperD}\}\), giving
      \[
        \{\SuperD_\alpha,\SuperQ_\beta\}=0,\qquad
        \{\SuperD_\alpha,\bar{\SuperD}_\beta\}
          =-2\gamma^\mu_{\alpha\beta}\partial_\mu .
      \]
      Hence \((\bar\alpha\SuperQ)\SuperD S=\SuperD(\bar\alpha\SuperQ)S\).
      A bare \(\theta\)-derivative lacks the compensating
      \(-\gamma^\mu\theta\partial_\mu\) term and does not commute with
      supersymmetry.
    \item \emph{A Superspace Integration by Parts.}
    The \(\theta\)-derivative part of \(\SuperD_\alpha S\) has Grassmann
      degree at most three and cannot contain a nonderivative \(D\)-component.
      Its remaining part is
      \(-(\gamma^\mu\theta)_\alpha\partial_\mu S\), whose \(D\)-component is a
      spacetime derivative.  Therefore
      \(\int \dd^4x[\SuperD_\alpha S]_D=0\).  Applying this to \(S_1S_2\) and
      using the graded Leibniz rule yields
      \[
        \int[S_1\SuperD_\alpha S_2]_D
          =-\int[(\SuperD_\alpha S_1)S_2]_D
          =-\int[S_2\SuperD_\alpha S_1]_D .
      \]
    \item \emph{Solving the Chiral Constraint.}
    Substitution in Eq.~\eqref{eq:26.2.26} gives
      \(\SuperD_Rx_+^\mu=0\) and \(\SuperD_R\theta_L=0\).
      A scalar solution is therefore an arbitrary polynomial in the two
      components of \(\theta_L\):
      \[
        \Phi=\phi(x_+)-\sqrt2\,
          (\theta_L^{\mathsf T}\epsilon\psi_L(x_+))
          +\mathcal F(x_+)(\theta_L^{\mathsf T}\epsilon\theta_L).
      \]
      This is Eq.~\eqref{eq:26.3.21}.  The complex \(\phi,\mathcal F\) give
      four real bosonic components, and the Majorana field represented by
      \(\psi_L\) gives four real fermionic components.
  \end{enumerate}

  \SupplementarySolution{3}{Chiral Superspace and Non-Renormalization}
  \begin{enumerate}
    \item \emph{Covariant-Derivative Algebra.}
    The terms with two Grassmann derivatives cancel by antisymmetry, while the
      cross terms between a Grassmann derivative and a coordinate-dependent
      translation survive.  Thus
      \[
        \{D_\alpha,D_\beta\}=0,\qquad
        \{\bar D_{\dot\alpha},\bar D_{\dot\beta}\}=0,\qquad
        \{D_\alpha,\bar D_{\dot\alpha}\}
          =-2\ii\sigma^\mu_{\alpha\dot\alpha}\partial_\mu .
      \]
      Therefore \(P_\mu=-\ii\partial_\mu\), and the mixed relation is
      \(2\sigma^\mu_{\alpha\dot\alpha}P_\mu\).  The signs agree with the
      four-component operators \(\SuperD_L,\SuperD_R\) after the chiral
      projections are made.
    \item \emph{Chiral and Antichiral Superfields.}
    A chiral scalar obeys
      \(\bar D_{\dot\alpha}\Phi=0\), while an antichiral scalar obeys
      \(D_\alpha\Phi^\dagger=0\).  Since
      \(\bar D_{\dot\alpha}y^\mu=0\), the general chiral solution is
      \[
        \Phi(y,\theta)=\phi(y)+\sqrt2\,\theta\psi(y)+\theta^2F(y).
      \]
      Taylor expansion back at \(x^\mu\) gives
      \[
      \begin{aligned}
        \Phi={}&\phi+\sqrt2\,\theta\psi+\theta^2F
          +\ii\theta\sigma^\mu\bar\theta\,\partial_\mu\phi\\
          &-\frac{\ii}{\sqrt2}\theta^2
             \partial_\mu\psi\,\sigma^\mu\bar\theta
          +\frac14\theta^2\bar\theta^2\Box\phi .
      \end{aligned}
      \]
      Complex conjugation, with
      \(y^{\dagger\mu}=x^\mu-\ii\theta\sigma^\mu\bar\theta\), gives the
      antichiral expansion.
    \item \emph{Half-Superspace Actions.}
    The supersymmetry transformation of the top component is
      \[
        \delta F
          =\ii\sqrt2\,\bar\epsilon\bar\sigma^\mu\partial_\mu\psi,
      \]
      a spacetime divergence.  A holomorphic function of chiral superfields
      is itself chiral, so its \(\theta^2\) coefficient has the same property.
      Hence
      \(\int \dd^4x\,\dd^2\theta\,W(\Phi)+\mathrm{H.c.}\) is supersymmetric for
      fields with the usual boundary behavior.
    \item \emph{Holomorphy and Physical Couplings.}
    Regard \(m\) and \(\lambda\) as background chiral spurions.  Holomorphy,
      dimensional analysis, and the two independent phase symmetries of
      \(\Phi,m,\lambda\) first restrict the Wilsonian answer to
      \[
        W_{\rm eff}=m\Phi^2
          f\!\left(\frac{\lambda\Phi}{m}\right)
      \]
      for a holomorphic function \(f\).  Expanding \(f(z)=\sum_nc_nz^n\),
      every term with \(n\ge2\) contains a negative power of \(m\) and is
      singular in the perturbative limit \(m\to0\).  The \(n=0\) and \(n=1\)
      coefficients cannot acquire loop corrections either: after the
      spurion charges and dimensions have been imposed, no holomorphic
      coupling remains on which such a correction could depend while also
      vanishing in the free limit \(\lambda\to0\).  Matching to the classical
      action therefore gives
      \(c_0=\tfrac12,\ c_1=\tfrac13\), and all higher \(c_n=0\).
      Thus the Wilsonian superpotential is not perturbatively renormalized.

      The Kähler term is not protected:
      \(Z(\mu)\int \dd^4\theta\,\Phi^\dagger\Phi\) is generated.  With
      \(\Phi_c=Z^{1/2}\Phi\), the physical parameters are
      \[
        m_{\rm phys}=\frac{m}{Z},\qquad
        \lambda_{\rm phys}=\frac{\lambda}{Z^{3/2}}.
      \]
      Their running is therefore entirely wavefunction renormalization and
      does not contradict superpotential non-renormalization.
  \end{enumerate}

  \SupplementarySolution{4}{Kähler Equivalence and Renormalizable Chiral Actions}
  \begin{enumerate}
    \item \emph{Kahler Transformations.}
    Both \(h(\Phi)\) and \(h(\Phi)^*\) are chiral superfields.  Chiral
      superfields have \(D\)-components that are spacetime derivatives, so
      \[
        \int \dd^4x\,[h(\Phi)+h(\Phi)^*]_D=0 .
      \]
      Also
      \(\partial_n\partial_{\bar m}h=
        \partial_n\partial_{\bar m}h^*=0\).  Hence
      \(K_{n\bar m}\), and therefore the component kinetic terms, are
      unchanged.
    \item \emph{Renormalizability by Dimension Counting.}
    Since an \(\mathcal F\)-component raises dimension by one and a
      \(D\)-component by two, a dimension-four Lagrangian requires
      \(d(f)\le3\) and \(d(K)\le2\).  With \(d(\Phi)=1\), the derivative-free
      possibilities are a cubic holomorphic polynomial for \(f\) and a
      quadratic function for \(K\).  Purely holomorphic or antiholomorphic
      terms in \(K\) have vanishing integrated \(D\)-terms.  Positivity and
      reality then leave
      \[
        K=g_{n\bar m}\Phi_n\Phi_m^*,\qquad
        g_{n\bar m}=g_{m\bar n}^*,
      \]
      while derivative terms are either too large in dimension or removable
      by the integrations described in Section 26.3.
  \end{enumerate}

  \SupplementarySolution{5}{Auxiliary Elimination, Masses, and \(F\)-Term Vacua}
  \begin{enumerate}
    \item \emph{Eliminating the Chiral Auxiliary Field.}
    The auxiliary part is
      \[
        \mathcal L_{\rm aux}
          =\sum_n\left(\mathcal F_n^*\mathcal F_n
           +\mathcal F_nf_n+\mathcal F_n^*f_n^*\right)
          =\sum_n|\mathcal F_n+f_n^*|^2-\sum_n|f_n|^2 .
      \]
      Therefore \(\mathcal F_n=-f_n^*\), and the on-shell Lagrangian contains
      \(-V\), with
      \[
        V(\phi,\phi^*)=\sum_n|f_n(\phi)|^2.
      \]
    \item \emph{When Does an \(F\)-Term Vacuum Exist?.}
    The potential is a sum of nonnegative terms.  If all \(f_n\) vanish at a
      point, then \(V=0\), the auxiliary expectation values vanish, and the
      supersymmetry variations of the fermions have no constant term.
      Conversely a supersymmetric vacuum is annihilated by the supercharges and
      has zero energy.  Positivity of
      \(V=\sum|f_n|^2\) then forces every \(f_n\) separately to vanish.  Thus a
      common zero is necessary and sufficient at tree level.
    \item \emph{Symmetry of the Fermion Mass Matrix.}
    Around \(\phi=v+\varphi\),
      \[
        f=f(v)+f_n(v)\varphi_n+\frac12f_{nm}(v)\varphi_n\varphi_m+\cdots .
      \]
      Eq.~\eqref{eq:26.4.4} supplies
      \(-\tfrac12f_{nm}\bar\psi_n\psi_{mL}\).  Ordinary mixed partial
      derivatives commute, so \(f_{nm}=f_{mn}\), as also required by the
      symmetry of the Majorana chiral bilinear.  Complex conjugation gives
      \(-\tfrac12f_{nm}^*\bar\psi_n\psi_{mR}\).
  \end{enumerate}

  \SupplementarySolution{6}{Moduli and the Goldstino Null Vector}
  \begin{enumerate}
    \item \emph{A Moduli-Space Tangent Vector.}
    Differentiating the vacuum equations gives
      \[
        0=\frac{\dd f_n}{\dd s}
          =f_{nm}(\phi(s))\frac{\dd\phi_m}{\dd s}.
      \]
      Hence the tangent \(v_m=\dd\phi_m/\dd s\) is a zero eigenvector of the
      fermion mass matrix.  Supersymmetry pairs this massless fermion with the
      two real scalar fluctuations tangent to the complex vacuum family, giving
      one massless chiral multiplet.
    \item \emph{The Goldstino Null Eigenvector.}
    Stationarity with respect to \(\phi_m\) gives
      \[
        0=\frac{\partial V}{\partial\phi_m}
          =f_{nm}f_n^*.
      \]
      Thus \(f_n^*\) is a right null vector of the symmetric fermion mass
      matrix.  If \(F^2=\sum_n|f_n|^2>0\), the normalized field
      \[
        \psi_G=\frac{f_n^*\psi_{nL}}{F}
      \]
      is massless.  Its supersymmetry variation has the nonzero constant
      \(\delta\psi_G\propto F\alpha_L\), identifying it as the goldstino.
  \end{enumerate}

  \SupplementarySolution{7}{Supertrace and the O'Raifeartaigh Spectrum}
  \begin{enumerate}
    \item \emph{The Chiral-Multiplet Supertrace.}
    Write \(m_{ij}=W_{ij}\).  For canonical kinetic terms the fermion
      squared-mass trace is
      \(\operatorname{tr}(m^\dagger m)\).  In the basis
      \((\delta\phi^i,\delta\phi^{*\bar\imath})\), the real-scalar
      squared-mass matrix has the block form
      \[
        \mathcal M_0^2=
        \begin{pmatrix}
          m^\dagger m&W_{ijk}^*W_k\\
          W_{ijk}W_k^*&mm^\dagger
        \end{pmatrix}.
      \]
      Its trace is \(2\operatorname{tr}(m^\dagger m)\); the
      supersymmetry-breaking off-diagonal blocks do not contribute.
      A Weyl fermion carries two spin states in the supertrace convention, so
      \[
        \operatorname{Str}M^2
          =\operatorname{tr}\mathcal M_0^2
           -2\operatorname{tr}(m^\dagger m)=0.
      \]
      Applied directly to visible MSSM multiplets, this rule prevents all
      scalar partners from being uniformly raised above their fermion
      partners at tree level without compensating light or tachyonic states.
      Realistic spectra therefore require a separate breaking/mediation
      sector, gauge or noncanonical effects, and radiative corrections.
    \item \emph{O'Raifeartaigh Vacuum and Spectrum.}
    The auxiliary equations follow from
      \[
        W_X=\lambda(Z^2-\mu^2),\qquad
        W_Y=MZ,\qquad
        W_Z=2\lambda XZ+MY .
      \]
      Since \(W_X=W_Y=0\) would require simultaneously \(Z^2=\mu^2\) and
      \(Z=0\), supersymmetry is broken.  At \(x=y=z=0\),
      \(F_X=\lambda\mu^2\), \(V=\lambda^2\mu^4\), and \(\psi_X\) is the
      goldstino.  The complex \(x\) direction is a tree-level pseudomodulus.

      The quadratic scalar potential is
      \[
        V^{(2)}
          =M^2(|y|^2+|z|^2)
             -\lambda^2\mu^2(z^2+z^{*2}).
      \]
      For \(z=(z_R+\ii z_I)/\sqrt2\), the six real scalar masses are
      \[
      \begin{array}{c|cccccc}
        \text{field}&x_R&x_I&y_R&y_I&z_R&z_I\\ \hline
        m^2&0&0&M^2&M^2&
          M^2-2\lambda^2\mu^2&M^2+2\lambda^2\mu^2 .
      \end{array}
      \]
      Thus the origin is locally stable when
      \(M^2>2\lambda^2\mu^2\).

      At the same point
      \[
        W_{ij}=
        \begin{pmatrix}
          0&0&0\\
          0&0&M\\
          0&M&0
        \end{pmatrix}_{X,Y,Z}.
      \]
      Hence \(\psi_X\) is massless and \(\psi_Y,\psi_Z\) form a Dirac
      fermion of mass \(M\).  The bosonic squared-mass sum is
      \(4M^2\); the two massive Weyl fermions contribute
      \(2M^2+2M^2=4M^2\).  The explicit spectrum therefore verifies
      \(\operatorname{Str}M^2=0\).
  \end{enumerate}

  \SupplementarySolution{8}{Moment Maps and Current Multiplets}
  \begin{enumerate}
    \item \emph{The Current as a Moment Map.}
    Infinitesimal invariance of \(K\) gives
      \[
        0=\ii\epsilon\left(
          K_n\mathcal T_{nm}\Phi_m
          -K_{\bar n}\mathcal T_{mn}\Phi_m^*\right).
      \]
      Therefore the two expressions for \(\mathcal J\) are equal and are
      complex conjugates, so \(\mathcal J\) is real.  Adding a real constant
      preserves this property and changes none of its superderivatives or its
      vector current, which is the usual moment-map ambiguity.
    \item \emph{The Canonical \(U(1)\) Current Multiplet.}
    The product formulas give
      \[
      \begin{aligned}
        C^{\mathcal J}&=|\phi|^2,\\
        \omega^{\mathcal J}&=-\ii\sqrt2\gamma_5
           (\phi^*\psi_L+\phi\psi_R),\\
        V_\mu^{\mathcal J}&=-\ii(\phi^*\partial_\mu\phi
           -\phi\partial_\mu\phi^*)
           +\ii\bar\psi_R\gamma_\mu\psi_L.
      \end{aligned}
      \]
      Varying the canonical scalar and fermion kinetic terms under
      \(\delta\phi=\ii\epsilon\phi,\ \delta\psi_L=\ii\epsilon\psi_L\) yields this
      same vector, up to the overall sign fixed by whether the current is
      defined from \(\delta\mathcal L=-(\partial_\mu\epsilon)J^\mu\) or with
      the opposite convention.
  \end{enumerate}

  \SupplementarySolution{9}{On-Shell Wess--Zumino Supersymmetry}
  Put
  \[
    P=A+\ii\gamma_5B,\qquad Q=A-\ii\gamma_5B,\qquad
    \mathscr D=\gamma^\mu\partial_\mu+m.
  \]
  The supplied charge-conjugation identity, together with the
  anticommutation of Majorana components, gives the interchange rules
  \[
    \bar s_1s_2=\bar s_2s_1,\qquad
    \bar s_1\gamma_5s_2=\bar s_2\gamma_5s_1,\qquad
    \bar s_1\gamma^\mu s_2=-\bar s_2\gamma^\mu s_1.
  \]
  These identities are the only spinor-specific input needed below.

  Since \(\alpha\) is constant, variation of the two scalar kinetic terms
  and the two scalar mass terms, followed by one integration by parts, gives
  \[
    \delta_S\mathcal L_{\rm bos}
      \doteq \bar\alpha(\Box-m^2)Q\psi,
  \]
  where \(\doteq\) denotes equality modulo a spacetime divergence.

  For the Majorana quadratic form, its two variations are equal after a
  Majorana interchange and an integration by parts.  The factor \(1/2\)
  therefore disappears:
  \[
    \delta_S\mathcal L_{\rm f}
      \doteq-\bar\psi\,\mathscr D(\delta_S\psi).
  \]
  Now
  \[
    \delta_S\psi=(\partial_\mu P)\gamma^\mu\alpha-mQ\alpha.
  \]
  Anticommutation of \(\gamma_5\) with every \(\gamma^\mu\) implies
  \[
    \gamma^\nu\partial_\nu Q=(\partial_\nu P)\gamma^\nu,
    \qquad
    \gamma^\nu(\partial_\nu\partial_\mu P)\gamma^\mu
      =\Box Q.
  \]
  The two terms linear in \(m\partial A\) or \(m\partial B\) consequently
  cancel, and hence
  \[
    \mathscr D(\delta_S\psi)=(\Box-m^2)Q\alpha.
  \]
  Because \(Q\) is a linear combination of \(\mathbf 1\) and \(\gamma_5\),
  the first two Majorana interchange rules yield
  \[
    \delta_S\mathcal L_{\rm f}
      \doteq-\bar\alpha(\Box-m^2)Q\psi.
  \]
  This cancels \(\delta_S\mathcal L_{\rm bos}\).  Thus
  \(\delta_S\mathcal L\) is a total derivative and the action is invariant.
  No field equation was used: ``on shell'' here means that the auxiliary
  fields have already been eliminated, not that the propagating fields were
  put on their equations of motion.

  \SupplementarySolution{10}{Nonlinear Kähler Sigma Models}
  \begin{enumerate}
    \item \emph{The Superfield Euler--Lagrange Equation.}
    With \(\delta\Phi_n=\SuperD_R^2\delta S_n\), superspace integration by
      parts moves both derivatives onto \(K_n\):
      \[
        \delta I_K\propto
          \int \dd^4x\,\dd^4\theta\,
          \delta S_n\SuperD_R^2K_n .
      \]
      The chiral projection identity
      \(\int[(\SuperD_R^2h)]_{\mathcal F}
        =2\int[h]_D\), together with the normalizations
      \eqref{eq:26.6.5}--\eqref{eq:26.6.6}, turns the superpotential variation
      into \(4f_n\) in the same equation.  Arbitrary \(\delta S_n\) therefore
      gives
      \[
        \SuperD_R^2K_n=-4f_n,
      \]
      as in Eq.~\eqref{eq:26.6.15}.
    \item \emph{The Kahler Metric in Components.}
    The mixed quadratic term in the \(\theta\)-expansion of \(K\) yields
      \[
        \mathcal L_{\rm scalar}
          =-K_{n\bar m}(\phi,\phi^*)
            \partial_\mu\phi_n\partial^\mu\phi_m^* .
      \]
      Around any constant background, diagonalize the Hermitian matrix
      \(K_{n\bar m}\).  A negative eigenvalue gives a scalar mode with a
      wrong-sign kinetic term and hence negative norm or unbounded energy.
      Absence of such ghosts requires, and locally is guaranteed by, positive
      definiteness of the Kahler metric.
    \item \emph{Auxiliaries with Noncanonical \(K\).}
    Varying Eq.~\eqref{eq:26.8.6} with respect to
      \(\mathcal F_{\bar n}^*\) gives
      \[
        K_{m\bar n}\mathcal F_m
          -\frac12K_{mp\bar n}(\bar\psi_m\psi_{pL})
          +f_{\bar n}^*=0.
      \]
      Its conjugate determines \(\mathcal F_m^*\).  With fermions set to zero,
      \(\mathcal F_m=-(K^{-1})^{m\bar n}f_{\bar n}^*\).  Completing the
      square then yields
      \[
        V=f_n(K^{-1})^{n\bar m}f_{\bar m}^*.
      \]
    \item \emph{Kahler Curvature and Four Fermions.}
    Solving the auxiliary equation produces a term quadratic in the Kahler
      connection \(K_{np\bar q}\).  Combining it with the explicit
      \(K_{np\bar m\bar q}\) term gives
      \[
        R_{n\bar m p\bar q}
          =K_{n\bar m p\bar q}
           -K_{np\bar r}(K^{-1})^{s\bar r}K_{s\bar m\bar q}.
      \]
      Hence the four-fermion interaction is proportional to
      \(R_{n\bar m p\bar q}
      (\bar\psi_n\psi_{pL})(\bar\psi_q\psi_{mR})\).
      Reordering the Grassmann fields can move an overall sign or a factor
      between this formula and the definition of \(R\), but the covariant
      curvature tensor is invariant.
  \end{enumerate}

  \SupplementarySolution{11}{Supercurrent Conservation and Improvement}
  \begin{enumerate}
    \item \emph{Conservation of the Supersymmetry Current.}
    Differentiating Eq.~\eqref{eq:26.7.10} produces terms containing
      \(\Box\phi_n\), \(\sl{\partial}\psi_{nL,R}\), and
      \(\partial_\mu f_n=f_{nm}\partial_\mu\phi_m\).
      Use Eq.~\eqref{eq:26.7.17} for the fermions and the scalar equation
      \[
        \Box\phi_n=f_{nm}^*f_m
          +\frac12f_{nmp}^*(\bar\psi_m\psi_{pR})
      \]
      with its conjugate.  The \(f_{nm}\partial\phi\) terms cancel the
      differentiated superpotential terms, while the remaining Yukawa terms
      cancel pairwise by \(f_{nm}=f_{mn}\).  Thus
      \(\partial_\mu S^\mu=0\).
    \item \emph{An Identically Conserved Improvement.}
    The derivatives commute and are symmetric in \(\mu,\nu\), whereas
      \([\gamma^\mu,\gamma^\nu]\) is antisymmetric:
      \[
        \partial_\mu\partial_\nu
          ([\gamma^\mu,\gamma^\nu]\chi)=0.
      \]
      The time component of the improvement is a spatial divergence because
      the \(\nu=0\) term vanishes.  Its integral is therefore a boundary term,
      so it changes neither the charge nor the commutators of that charge with
      fields.
  \end{enumerate}

  \SupplementarySolution{12}{Gamma Trace and the Improved Current Multiplet}
  \begin{enumerate}
    \item \emph{Gamma Trace and Scale Invariance.}
    For homogeneous cubic \(f\), Euler's theorem gives
      \(\phi_mf_m=3f\).  Differentiating with respect to \(\phi_n\) yields
      \[
        \phi_m f_{nm}=2f_n,
      \]
      so every parenthesis in Eq.~\eqref{eq:26.7.19} vanishes.  A quadratic
      term instead gives \(\phi_mf_{nm}=f_n\), and a linear term gives zero;
      both leave a nonzero difference from \(2f_n\).  These dimensionful mass
      and tadpole couplings are precisely the classical obstructions to a
      gamma-traceless improved current.
    \item \emph{Inverting the Supercurrent Relation.}
    Gamma tracing gives
      \[
        \gamma_\mu S^\mu
          =-2\gamma_\mu\omega^\mu
           +2\gamma_\mu\gamma^\mu\gamma^\nu\omega_\nu
          =6\gamma^\nu\omega_\nu.
      \]
      Thus
      \[
        \omega_\mu=-\frac12S_\mu
          +\frac16\gamma_\mu\gamma^\nu S_\nu.
      \]
      Substitution back into the original relation verifies the inverse.
  \end{enumerate}

  \SupplementarySolution{13}{Parity, Time Reversal, and Chirality}
  \begin{enumerate}
    \item \emph{Parity Exchanges Chiralities.}
    The parity substitution gives
      \(x_+\mapsto\Lambda_Px_-\) and
      \(P_L(-\ii\beta\theta)=-\ii\beta P_R\theta\).  Hence a function only of
      \(x_+,\theta_L\) becomes one only of \(x_-,\theta_R\), so left chirality
      becomes right chirality.  Up to the intrinsic phase,
      \[
        \begin{aligned}
          \phi(x)&\mapsto\phi(\Lambda_Px),\\
          \psi_L(x)&\mapsto-\ii\beta\psi_L(\Lambda_Px)
            &&\text{in the right slot},\\
          \mathcal F(x)&\mapsto-\mathcal F(\Lambda_Px),
        \end{aligned}
      \]
      consistently with Eqs.~\eqref{eq:26.3.35}--\eqref{eq:26.3.36}.
    \item \emph{Time Reversal Preserves Chirality.}
    The matrix \(B_T\) intertwines complex-conjugate gamma matrices with the
      time-reflected ones.  When \(T\) is moved through
      \(\SuperD_R\Phi=0\), its antiunitarity also sends \(i\to-\ii\).  This
      extra sign compensates the interchange that occurred for unitary parity,
      giving
      \[
        \SuperD_R\Phi^T=0.
      \]
      Equivalently, time reversal flips both momentum and spin, so helicity is
      unchanged; parity flips momentum but not angular momentum, so helicity
      and chirality are exchanged.
  \end{enumerate}

  \SupplementarySolution{14}{K\"ahler Data, Superspace Components, and Chiral Vacua}
  \begin{enumerate}
    \item[(a)] The K\"ahler potential is a real function
    \(K(\Phi^i,\Phi^{\dagger\bar\jmath})\) integrated over full superspace.
    Its mixed second derivatives
    \[
      g_{i\bar\jmath}=\partial_i\partial_{\bar\jmath}K
    \]
    form the K\"ahler metric that controls the scalar and fermion kinetic
    terms.  The replacement
    \(K\mapsto K+h(\Phi)+h(\Phi)^*\) changes the action only by a superspace
    derivative.  The superpotential \(W(\Phi)\) is holomorphic and is
    integrated over chiral superspace; its first derivatives determine the
    auxiliary equations and scalar potential, and its second derivatives
    determine the fermion masses and Yukawa couplings.

    \item[(b)] In multiplying \(\Phi^\dagger\Phi\), the purely bosonic
    \(\theta^2\bar\theta^2\) coefficient receives contributions from
    \(\mathcal F^*\mathcal F\), the two displayed \(\Box\phi\) terms, and
    the product of the two
    \(\theta\sigma^\mu\bar\theta\,\partial_\mu\phi\) terms.  With the
    sheet's Berezin normalization and after an ordinary integration by parts,
    \[
      \boxed{
      I_K\big|_{\rm bos}
        =\int \dd^4x\left[
          -\partial_\mu\phi^*\partial^\mu\phi
          +\mathcal F^*\mathcal F
        \right].}
    \]
    The omitted difference is a spacetime boundary term.

    \item[(c)] Taylor expansion in the chiral increment gives
    \[
      \begin{aligned}
        W(\Phi)={}&W(\phi)
          +W'(\phi)\bigl(\sqrt2\,\theta\psi+\theta^2\mathcal F\bigr)\\
          &+\frac12W''(\phi)(\sqrt2\,\theta\psi)^2+\cdots .
      \end{aligned}
    \]
    Its bosonic \(\theta^2\) coefficient is therefore
    \(\mathcal F W'(\phi)\), so the action exactly as printed has
    \[
      \boxed{
      I_W\big|_{\rm bos}
        =\int \dd^4x\,\mathcal F W'(\phi).}
    \]
    This expression is generally complex.  A real Lagrangian contains
    \(I_W+I_W^\dagger\), whose bosonic contribution is
    \[
      \int \dd^4x\left[
        \mathcal F W'(\phi)
        +\mathcal F^*W'(\phi)^*
      \right].
    \]
    For several canonical chiral fields, eliminating the auxiliaries gives
    \[
      \mathcal F_i=-W_i^*,\qquad
      V=\sum_i|W_i|^2,\qquad W_i\equiv\frac{\partial W}{\partial\phi_i}.
    \]

    \item[(d)]
    \begin{enumerate}
      \item[(i)] For \(W=X^2Z\),
      \[
        W_X=2XZ,\qquad W_Z=X^2,\qquad
        V=4|x|^2|z|^2+|x|^4.
      \]
      Thus the complete vacuum set is
      \[
        x=0,\qquad z\ \hbox{arbitrary},\qquad V_{\rm vac}=0.
      \]
      It is one complex moduli line of supersymmetric vacua.

      \item[(ii)] For \(W=XYZ\),
      \[
        \begin{aligned}
          W_X&=YZ,& W_Y&=XZ,& W_Z&=XY,\\
          V&=|yz|^2+|xz|^2+|xy|^2.
        \end{aligned}
      \]
      Vanishing energy requires all three pairwise products to vanish.
      Hence the vacuum space is the union of the three complex coordinate
      axes,
      \[
        (x,0,0),\qquad(0,y,0),\qquad(0,0,z),
      \]
      meeting at the origin.  Every point has \(V_{\rm vac}=0\) and preserves
      supersymmetry.

      \item[(iii)] Here
      \[
        W_X=2\beta XY+\gamma Z,\qquad
        W_Y=\alpha+\beta X^2,\qquad
        W_Z=\gamma X,
      \]
      and therefore
      \[
        V=|2\beta xy+\gamma z|^2
          +|\alpha+\beta x^2|^2+|\gamma x|^2.
      \]
      Since \(\gamma\ne0\), the equation \(W_Z=0\) would require \(x=0\),
      but then \(W_Y=\alpha\ne0\).  No supersymmetric vacuum exists.

      For fixed \(x,y\), minimize the first square by taking
      \[
        z=-\frac{2\beta}{\gamma}xy.
      \]
      If \(\rho=|x|\), the phase of \(x\) can make
      \(\operatorname{Re}(\alpha^*\beta x^2)=-|\alpha\beta|\rho^2\).
      The remaining minimum at fixed \(\rho\) is
      \[
        V_{\min}(\rho)=|\alpha|^2
          +\bigl(|\gamma|^2-2|\alpha\beta|\bigr)\rho^2
          +|\beta|^2\rho^4.
      \]
      The exact condition \(|\gamma|^2>2|\alpha\beta|\), which is implied by
      the stated hierarchy, selects \(\rho=0\).  Consequently
      \[
        \boxed{
        x=z=0,\qquad y\ \hbox{arbitrary},\qquad
        V_{\rm vac}=|\alpha|^2.}
      \]
      The nonzero \(\mathcal F_Y\) breaks supersymmetry, while \(y\) is a classical
      pseudomodulus.
    \end{enumerate}
  \end{enumerate}

\end{enumerate}
