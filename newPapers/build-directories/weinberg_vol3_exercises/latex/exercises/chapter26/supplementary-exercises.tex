\begin{enumerate}
  \SupplementaryExercise{1}{Off-Shell Superfields and Closure}
    {(Inspired by Philip C. Argyres, \emph{An Introduction to Global Supersymmetry}, Sections 1.5.2--1.5.5; independently formulated.)}
    {ch26-s01-off-shell-superfields-and-closure}
  \begin{enumerate}
    \item \emph{Counting a General Real Superfield.}
    Starting from the finite Grassmann expansion
      \eqref{eq:26.2.10}, count the independent real bosonic and fermionic
      components of a real scalar superfield.  Explain why the equality holds
      off shell, before any field equation or gauge invariance is imposed.
    \item \emph{Closure on the Wess--Zumino Scalar.}
    Use Eq.~\eqref{eq:26.1.21} to evaluate
      \([\delta_\alpha,\delta_\beta]A\).  Show explicitly that the auxiliary
      fields cancel and that the result is a translation.  State the
      corresponding result for \(B,\psi,F,G\).
  \end{enumerate}

  \SupplementaryExercise{2}{Covariant Superderivatives, Chiral Coordinates, and Integration}
    {(Adapted from David Tong, \emph{Supersymmetry Example Sheet 1}, Question 7.)}
    {ch26-s02-covariant-superderivatives-chiral-coordinates-and-integration}
  \begin{enumerate}
    \item \emph{Why the Covariant Superderivative Works.}
    Starting from Eqs.~\eqref{eq:26.2.2} and \eqref{eq:26.2.26}, verify
      \(\{\SuperD_\alpha,\SuperQ_\beta\}=0\) and
      \(\{\SuperD_\alpha,\bar{\SuperD}_\beta\}
      =-2\gamma^\mu_{\alpha\beta}\partial_\mu\).  Explain why
      \(\SuperD_\alpha S\) is a superfield although
      \(\partial S/\partial\theta_\alpha\) alone is not.
    \item \emph{A Superspace Integration by Parts.}
    Prove Eq.~\eqref{eq:26.3.32} by Grassmann degree counting and an ordinary
      spacetime integration by parts.  Deduce Eq.~\eqref{eq:26.3.33} for two
      bosonic superfields.
    \item \emph{Solving the Chiral Constraint.}
    Show directly that
      \(x_+^\mu=x^\mu+\bar\theta_R\gamma^\mu\theta_L\) and \(\theta_L\)
      are annihilated by \(\SuperD_R\).  Derive the expansion
      \eqref{eq:26.3.21} of a left-chiral scalar superfield and count its
      independent off-shell components.
  \end{enumerate}

  \SupplementaryExercise{3}{Chiral Superspace and Non-Renormalization}
    {(Adapted from University of Cambridge, 2022 Part III Paper 307, \emph{Supersymmetry}, Question 1.)}
    {ch26-s03-chiral-multiplets-products-and-invariant-terms}
  \begin{enumerate}
    \item \emph{Covariant-Derivative Algebra.}
    In two-component shorthand define
      \[
        D_\alpha=\partial_\alpha
          +i\sigma^\mu_{\alpha\dot\alpha}\bar\theta^{\dot\alpha}\partial_\mu,
        \qquad
        \bar D_{\dot\alpha}=-\bar\partial_{\dot\alpha}
          -i\theta^\alpha\sigma^\mu_{\alpha\dot\alpha}\partial_\mu .
      \]
      Compute all their anticommutators and identify the translation operator
      \(P_\mu\) for which
      \(\{D_\alpha,\bar D_{\dot\alpha}\}
        =2\sigma^\mu_{\alpha\dot\alpha}P_\mu\).
    \item \emph{Chiral and Antichiral Superfields.}
    Define both constraints.  Introduce
      \(y^\mu=x^\mu+i\theta\sigma^\mu\bar\theta\), solve the chiral
      constraint, and derive the complete expansion of a scalar chiral
      superfield through order \(\theta^2\bar\theta^2\).
    \item \emph{Half-Superspace Actions.}
    Explain why the \(\theta^2\) component of a holomorphic
      superpotential is invariant up to a spacetime divergence, and hence why
      \(\int d^2\theta\,W(\Phi)+\mathrm{H.c.}\) is supersymmetric.
    \item \emph{Holomorphy and Physical Couplings.}
    For
      \(W_{\rm tree}=\tfrac12m\Phi^2+\tfrac13\lambda\Phi^3\), use
      holomorphy and spurion charges to explain why perturbation theory cannot
      generate a new superpotential.  Then explain why the canonically
      normalized mass and cubic coupling nevertheless run.
  \end{enumerate}

  \SupplementaryExercise{4}{Kähler Equivalence and Renormalizable Chiral Actions}
    {(Inspired by Stephen P. Martin, \emph{A Supersymmetry Primer}, version 7, Sections 3.1--3.2 and 4.6--4.7; independently formulated.)}
    {ch26-s04-k-hler-equivalence-and-renormalizable-chiral}
  \begin{enumerate}
    \item \emph{Kahler Transformations.}
    Prove that the action is unchanged by
      \[
        K(\Phi,\Phi^*)\longrightarrow K(\Phi,\Phi^*)+
          h(\Phi)+h(\Phi)^*
      \]
      for any holomorphic \(h\).  Show that the Kahler metric
      \(K_{n\bar m}\) is unchanged.
    \item \emph{Renormalizability by Dimension Counting.}
    With \(d(\Phi)=1\), \(d(\SuperD)=1/2\), derive the bounds
      \(d(f)\le3\) and \(d(K)\le2\) for a renormalizable theory.  After removing
      chiral \(D\)-terms and derivative redundancies, show that \(f\) is at
      most cubic and \(K\) is a Hermitian quadratic form.
  \end{enumerate}

  \SupplementaryExercise{5}{Auxiliary Elimination, Masses, and \(F\)-Term Vacua}
    {(Adapted from David Tong, \emph{Supersymmetry Example Sheet 1}, Question 9.)}
    {ch26-s05-auxiliary-elimination-masses-and-term-vacua}
  \begin{enumerate}
    \item \emph{Eliminating the Chiral Auxiliary Field.}
    For canonical \(K=\sum_n\Phi_n^*\Phi_n\), isolate all terms containing
      \(\mathcal F_n\) in the component Lagrangian.  Complete the square,
      eliminate \(\mathcal F_n\), and obtain the scalar potential.
    \item \emph{When Does an \(F\)-Term Vacuum Exist?.}
    For canonical kinetic terms prove that a classical supersymmetric vacuum
      exists precisely when the equations \(f_n(\phi)=0\) have a common
      solution.  Explain how positivity of \(V=\sum_n|f_n|^2\) enters both
      directions.
    \item \emph{Symmetry of the Fermion Mass Matrix.}
    Expand a holomorphic superpotential to quadratic order around a constant
      background.  Show that the chiral-fermion mass matrix is \(f_{nm}\) and
      is symmetric.  Relate its complex conjugate to the right-chiral mass
      term in four-component notation.
  \end{enumerate}

  \SupplementaryExercise{6}{Moduli and the Goldstino Null Vector}
    {(Inspired by David Tong, \emph{Supersymmetric Field Theory}, Sections 3.3--3.4; independently formulated.)}
    {ch26-s06-moduli-and-the-goldstino-null-vector}
  \begin{enumerate}
    \item \emph{A Moduli-Space Tangent Vector.}
    Suppose \(f_n(\phi(s))=0\) along a smooth one-complex-parameter family of
      supersymmetric vacua.  Differentiate with respect to \(s\) and show that
      the tangent vector is a zero mode of \(f_{nm}\).  Identify the associated
      massless chiral multiplet.
    \item \emph{The Goldstino Null Eigenvector.}
    At a stationary point of \(V=\sum_n|f_n|^2\), prove
      \(f_{mn}f_n^*=0\).  Conclude that when some \(f_n\ne0\), the normalized
      combination \(f_n^*\psi_{nL}/(\sum|f|^2)^{1/2}\) is massless.
  \end{enumerate}

  \SupplementaryExercise{7}{Supertrace and the O'Raifeartaigh Spectrum}
    {(Adapted from University of Cambridge, 2016 Part III Paper 307, \emph{Supersymmetry and Extra Dimensions}, Question 4.)}
    {ch26-s07-the-o-raifeartaigh-model}
  \begin{enumerate}
    \item \emph{The Chiral-Multiplet Supertrace.}
    For a renormalizable \(N=1\) theory of chiral superfields with canonical
      Kähler metric and arbitrary \(F\)-term breaking, derive the scalar and
      fermion mass matrices and prove
      \[
        \operatorname{Str}M^2
          =\sum_j(-1)^{2j}(2j+1)m_j^2=0.
      \]
      Here the supertrace uses the conventional boson-minus-fermion sign;
      the source sheet prints its overall negative, which is equivalent for
      a vanishing result.
      State the tree-level implication for attempts to make every MSSM
      superpartner heavier than its standard-model partner.
    \item \emph{O'Raifeartaigh Vacuum and Spectrum.}
    Take three chirals \(X,Y,Z\) with
      \[
        W=\lambda X(Z^2-\mu^2)+MYZ,\qquad M\gg\mu,
      \]
      and choose phases so that \(\lambda,M,\mu>0\).  Show that supersymmetry
      is broken at \(x=y=z=0\).  Find every real-scalar and Weyl-fermion
      mass there, identify the goldstino and pseudomodulus, give the local
      stability condition, and verify the supertrace explicitly.
  \end{enumerate}

  \SupplementaryExercise{8}{Moment Maps and Current Multiplets}
    {(Inspired by Adel Bilal, \emph{Introduction to Supersymmetry}, Sections 4.1--4.3 and 7.1; independently formulated.)}
    {ch26-s08-moment-maps-and-current-multiplets}
  \begin{enumerate}
    \item \emph{The Current as a Moment Map.}
    Let \(\delta\Phi_n=i\epsilon\mathcal T_{nm}\Phi_m\) be an isometry of
      \(K\).  Show that
      \(\mathcal J=K_n\mathcal T_{nm}\Phi_m\) is real, using only the
      infinitesimal invariance of \(K\).  Explain why it is defined only up to a
      field-independent real constant.
    \item \emph{The Canonical \(U(1)\) Current Multiplet.}
    For one chiral field with \(K=\Phi^*\Phi\) and
      \(\Phi\mapsto e^{i\epsilon}\Phi\), expand
      \(\mathcal J=\Phi^*\Phi\) through its vector component.  Verify directly
      that this vector is the scalar-plus-fermion Noether current.
  \end{enumerate}

  \SupplementaryExercise{9}{The Linear Superfield Constraint}
    {(Inspired by Adel Bilal, \emph{Introduction to Supersymmetry}, Sections 4.1--4.3; independently formulated.)}
    {ch26-s09-the-linear-superfield-constraint}
Apply \(\SuperD_R^2\) and \(\SuperD_L^2\) to a general real superfield.
  Derive
  \(M=N=0,\ \partial_\mu V^\mu=0,\ 
  \lambda=-\sl{\partial}\omega,\ D=-\Box C\).
  Count the remaining independent off-shell components.

  \SupplementaryExercise{10}{Nonlinear Kähler Sigma Models}
    {(Inspired by David Tong, \emph{Supersymmetric Field Theory}, Section 3.3; independently formulated.)}
    {ch26-s10-nonlinear-k-hler-sigma-models}
  \begin{enumerate}
    \item \emph{The Superfield Euler--Lagrange Equation.}
    Vary the action \eqref{eq:26.6.9} while preserving chirality by writing
      \(\delta\Phi_n=\SuperD_R^2\delta S_n\).  Integrate by parts in superspace
      and derive Eq.~\eqref{eq:26.6.15}, including its factor of \(-4\).
    \item \emph{The Kahler Metric in Components.}
    From the \(D\)-term of a general derivative-free \(K(\Phi,\Phi^*)\),
      show that the scalar kinetic term is
      \(-K_{n\bar m}\partial_\mu\phi_n\partial^\mu\phi_m^*\).
      Prove that positivity requires \(K_{n\bar m}\) to be positive definite.
    \item \emph{Auxiliaries with Noncanonical \(K\).}
    Retain the two-fermion terms and derive the algebraic field equation for
      \(\mathcal F_n\) from Eq.~\eqref{eq:26.8.6}.  In the purely bosonic
      limit show that
      \(V=f_n(K^{-1})^{n\bar m}f_{\bar m}^*\).
    \item \emph{Kahler Curvature and Four Fermions.}
    Eliminate the auxiliary fields in a nonlinear sigma model and combine
      the resulting four-fermion term with the last term of
      Eq.~\eqref{eq:26.8.6}.  Show that its coefficient is the Kahler curvature
      tensor \(R_{n\bar m p\bar q}\), up to the spinor-ordering convention.
  \end{enumerate}

  \SupplementaryExercise{11}{Supercurrent Conservation and Improvement}
    {(Adapted from A.~C. Davis, \emph{Supergravity 1 Example Sheet}, Question 5.)}
    {ch26-s11-supercurrent-conservation-and-improvement}
  \begin{enumerate}
    \item \emph{Conservation of the Supersymmetry Current.}
    Take the divergence of Eq.~\eqref{eq:26.7.10}.  Use the scalar and
      fermion field equations to prove \(\partial_\mu S^\mu=0\), keeping
      \(f_{nm}\) explicit until the final cancellation.
    \item \emph{An Identically Conserved Improvement.}
    Prove without field equations that
      \[
        \partial_\mu\partial_\nu\!
          \left([\gamma^\mu,\gamma^\nu]\chi\right)=0
      \]
      for any spinor \(\chi\).  Explain why the improvement
      \eqref{eq:26.7.18} changes neither the conserved charge nor its
      supersymmetry transformations.
  \end{enumerate}

  \SupplementaryExercise{12}{Gamma Trace and the Improved Current Multiplet}
    {(Inspired by Philip C. Argyres, \emph{An Introduction to Global Supersymmetry}, Sections 1.6.1 and 3.4; independently formulated.)}
    {ch26-s12-gamma-trace-and-the-improved-current}
  \begin{enumerate}
    \item \emph{Gamma Trace and Scale Invariance.}
    Use Euler's theorem for a homogeneous cubic superpotential to show that
      the right-hand side of Eq.~\eqref{eq:26.7.19} vanishes.  Check the
      converse for a polynomial superpotential with canonical dimensions:
      identify which mass and linear terms obstruct gamma-tracelessness.
    \item \emph{Inverting the Supercurrent Relation.}
    Starting from
      \(S^\mu=-2\omega^\mu+2\gamma^\mu\gamma^\nu\omega_\nu\),
      solve algebraically for \(\omega_\mu\) in terms of \(S_\mu\).  Use only
      \(\gamma_\mu\gamma^\mu=4\).
  \end{enumerate}

  \SupplementaryExercise{13}{Parity, Time Reversal, and Chirality}
    {(Inspired by Philip C. Argyres, \emph{An Introduction to Global Supersymmetry}, Sections 1.3.2 and 1.5.3; independently formulated.)}
    {ch26-s13-parity-time-reversal-and-chirality}
  \begin{enumerate}
    \item \emph{Parity Exchanges Chiralities.}
    Apply \(x\mapsto\Lambda_Px,\ \theta\mapsto-i\beta\theta\) to
      \(x_\pm\) and to the chiral projectors.  Prove that parity maps a
      left-chiral scalar superfield into a right-chiral one and write the
      induced map on \((\phi,\psi_L,\mathcal F)\).
    \item \emph{Time Reversal Preserves Chirality.}
    Let \(B_T=-\gamma_5\mathcal C\) be the Majorana time-reversal matrix.
      Taking account of antiunitarity, show that the time reverse of
      \(\SuperD_R\Phi=0\) again obeys a right-superderivative constraint rather
      than a left-superderivative constraint.  Explain the contrast with
      parity in terms of massless helicity.
  \end{enumerate}

  \SupplementaryExercise{14}{Dimension-Five Supersymmetric Operators}
    {(Inspired by Stephen P. Martin, \emph{A Supersymmetry Primer}, version 7, Section 4.10; independently formulated.)}
    {ch26-s14-dimension-five-supersymmetric-operators}
For one neutral left-chiral scalar superfield, classify all Lorentz-scalar
  supersymmetric operators of component dimension five.  Work modulo
  complex conjugation, ordinary and superspace integration by parts, and
  chiral \(D\)-terms.  Include the higher-derivative class.

\end{enumerate}
