\chapterbackmatter{Supplementary Exercises}

\begin{enumerate}
  \SupplementaryExercise{1}{The Supertrace Test}
    {(Adapted from Haag, {\L}opusza\'nski, and Sohnius, ``All Possible
    Generators of Supersymmetries of the S-Matrix,'' Sections 2--3)}
    {hls-1975-supergenerators}
  Let \(V=V_{\bar0}\oplus V_{\bar1}\), and write endomorphisms in even and
  odd block form.  Define
  \(\operatorname{str}\begin{psmallmatrix}A&B\\C&D\end{psmallmatrix}
  =\operatorname{tr}A-\operatorname{tr}D\).
  Prove for homogeneous \(X,Y\) that
  \(\operatorname{str}[X,Y\}=0\), and conclude that the supertraceless
  matrices close under the graded bracket.

  \SupplementaryExercise{2}{The Signed Jacobi Identity}
    {(Adapted from Haag, {\L}opusza\'nski, and Sohnius, ``All Possible
    Generators of Supersymmetries of the S-Matrix,'' Section 2)}
    {hls-1975-supergenerators}
  For homogeneous elements of an associative graded algebra, set
  \([X,Y\}=XY-(-1)^{|X||Y|}YX\).  Expand all products and verify
  \[
    (-1)^{|X||Z|}[X,[Y,Z\}\}
    +(-1)^{|Y||X|}[Y,[Z,X\}\}
    +(-1)^{|Z||Y|}[Z,[X,Y\}\}=0.
  \]
  Explain why the case of three odd elements involves three
  commutators with anticommutators.

  \SupplementaryExercise{3}{Why a SUSY Variation Is Even}
    {(Adapted from Haag, {\L}opusza\'nski, and Sohnius, ``All Possible
    Generators of Supersymmetries of the S-Matrix,'' Sections 2--3)}
    {hls-1975-supergenerators}
  Let \(Q_\alpha\) and a constant parameter \(\epsilon^\alpha\) both be
  odd.  Show that \(G=\epsilon^\alpha Q_\alpha+
  \epsilon^\dagger_{\dot\alpha}\bar Q^{\dot\alpha}\) is even and that
  \(\delta_\epsilon X=i[G,X]\) preserves the statistics of a homogeneous
  operator \(X\).  Then explain why treating \(\epsilon\) as commuting
  gives the wrong bracket in the commutator of two transformations.

  \SupplementaryExercise{4}{Why Supercharges Have Spin One-Half}
    {(Adapted from Haag, {\L}opusza\'nski, and Sohnius, ``All Possible
    Generators of Supersymmetries of the S-Matrix,'' Sections 3--4)}
    {hls-1975-supergenerators}
  Under the hypotheses used in the Coleman--Mandula and
  Haag--{\L}opusza\'nski--Sohnius arguments, suppose an odd conserved charge
  transforms in Lorentz spin \(A\).  Use the highest-spin component of
  \(A\otimes A\) in its anticommutator with its adjoint to explain why
  \(A>1/2\) is excluded in an interacting theory.  Why are \(A=0\) and
  \(A=1/2\) qualitatively different?

  \SupplementaryExercise{5}{Central Charges in Three Dimensions}
    {(Adapted from Haag, {\L}opusza\'nski, and Sohnius, ``All Possible
    Generators of Supersymmetries of the S-Matrix,'' Section 4)}
    {hls-1975-supergenerators}
  In \(2+1\) dimensions let \(Q^I_\alpha\) be Majorana supercharges and take
  \(C_{\alpha\beta}\) antisymmetric while
  \((\gamma^\mu C)_{\alpha\beta}\) is symmetric.  Starting from the most
  general Lorentz-covariant anticommutator containing only \(P_\mu\) and
  scalar charges, show that its coefficient of \(P_\mu\) is symmetric in
  \(I,J\), while the scalar central-charge matrix \(Z^{IJ}\) is
  antisymmetric.  How many independent real central charges are possible
  for \(N\) supercharges?

  \SupplementaryExercise{6}{Positive Energy from the Algebra}
    {(Adapted from Philip C. Argyres, \emph{An Introduction to Global
    Supersymmetry}, Sections 1.1 and 1.4)}
    {argyres-global-susy-2001}
  In four dimensions use
  \(\{Q_\alpha,Q^\dagger_{\dot\beta}\}
  =2\sigma^\mu_{\alpha\dot\beta}P_\mu\).
  Sum suitable diagonal anticommutators to express \(H=P^0\) as a sum of
  positive operators.  Prove that every normalized state has nonnegative
  energy.

  \SupplementaryExercise{7}{The Supersymmetric Vacuum Criterion}
    {(Adapted from Philip C. Argyres, \emph{An Introduction to Global
    Supersymmetry}, Section 1.4)}
    {argyres-global-susy-2001}
  Prove that a translationally invariant vacuum has zero energy if and only
  if it is annihilated by every component of \(Q\) and \(Q^\dagger\).
  State precisely which implication uses positivity of the Hilbert-space
  norm.

  \SupplementaryExercise{8}{Boson--Fermion Pairing}
    {(Adapted from Philip C. Argyres, \emph{An Introduction to Global
    Supersymmetry}, Section 1.4)}
    {argyres-global-susy-2001}
  Let \((-1)^F\) anticommute with every supercharge and commute with \(H\).
  Show that in any finite-dimensional eigenspace with \(E>0\), the numbers
  of bosonic and fermionic states are equal.  Give a proof using the
  invertible odd operator \(Q_1+Q_1^\dagger\) after choosing a rest frame
  and normalization.

  \SupplementaryExercise{9}{Mass Degeneracy in a Supermultiplet}
    {(Adapted from Philip C. Argyres, \emph{An Introduction to Global
    Supersymmetry}, Section 1.4.2)}
    {argyres-global-susy-2001}
  Given \([P_\mu,Q_\alpha]=0\), prove that \(P^2\) commutes with all
  supercharges.  Show that every state in an irreducible massive
  supermultiplet has the same mass, while its spin need not be the same.
  If an internal charge \(T\) commutes with all supercharges, what can be
  said about its eigenvalue within the multiplet?

  \SupplementaryExercise{10}{The \(U(1)_R\) Automorphism}
    {(Adapted from Philip C. Argyres, \emph{An Introduction to Global
    Supersymmetry}, Sections 1.4 and 1.6.2)}
    {argyres-global-susy-2001}
  For simple supersymmetry with no central charge, verify that
  \(Q_\alpha\mapsto e^{-i\theta}Q_\alpha\) and
  \(Q^\dagger_{\dot\alpha}\mapsto
  e^{i\theta}Q^\dagger_{\dot\alpha}\) leaves the algebra invariant.
  Introduce its generator \(R\), find \([R,Q_\alpha]\), and explain why this
  is an automorphism of supersymmetry rather than an ordinary internal
  symmetry commuting with \(Q\).

  \SupplementaryExercise{11}{Antisymmetry of \(Z_{rs}\)}
    {(Adapted from Adel Bilal, \emph{Introduction to Supersymmetry},
    Sections 3.1--3.2)}
    {bilal-introduction-susy-2001}
  In the extended algebra write
  \(\{Q_{\alpha r},Q_{\beta s}\}
  =\epsilon_{\alpha\beta}Z_{rs}\).
  Exchange the two complete labels \((\alpha,r)\) and \((\beta,s)\) and
  prove \(Z_{rs}=-Z_{sr}\).  Deduce that simple supersymmetry admits no
  scalar central charge of this form.

  \SupplementaryExercise{12}{Canonical Blocks of the Central Charge}
    {(Adapted from Adel Bilal, \emph{Introduction to Supersymmetry},
    Sections 3.1 and 8)}
    {bilal-introduction-susy-2001}
  A complex antisymmetric \(N\times N\) matrix can be brought by a unitary
  congruence to \(2\times2\) blocks
  \(z_i\begin{psmallmatrix}0&1\\-1&0\end{psmallmatrix}\), plus a zero block
  when \(N\) is odd.  Use this form to find the eigenvalues of
  \(Z^\dagger Z\), their multiplicities, and
  \(\operatorname{Tr}\sqrt{Z^\dagger Z}\).

  \SupplementaryExercise{13}{The \(N=2\) BPS Bound}
    {(Adapted from Adel Bilal, \emph{Introduction to Supersymmetry},
    Sections 8 and 9.2.3)}
    {bilal-introduction-susy-2001}
  In the rest frame of an \(N=2\) multiplet, choose phases so that the single
  skew eigenvalue \(z\) is real and nonnegative.  Form linear combinations
  of \(Q_{\alpha1}\) and \(\epsilon_{\alpha\beta}
  Q^\dagger_{\dot\beta2}\) whose anticommutators are proportional to
  \(2M\pm |z|\).  Derive the mass bound, noting explicitly how it translates
  to the convention \(M\geq |Z_{12}|/2\) used in this chapter.

  \SupplementaryExercise{14}{Null Oscillators and Shortening}
    {(Adapted from Adel Bilal, \emph{Introduction to Supersymmetry},
    Sections 3.4 and 9.2.3)}
    {bilal-introduction-susy-2001}
  Continue the preceding construction.  Show that at saturation one
  oscillator pair has zero norm and therefore vanishes in a unitary
  irreducible representation.  Compare the dimensions of a long \(N=2\)
  massive multiplet and a half-BPS short multiplet built on a spin-zero
  Clifford vacuum.

  \SupplementaryExercise{15}{A Long Massive \(N=1\) Multiplet}
    {(Adapted from Adel Bilal, \emph{Introduction to Supersymmetry},
    Section 3.4)}
    {bilal-introduction-susy-2001}
  Start from a spin-\(j\) Clifford vacuum annihilated by the two rest-frame
  annihilation operators.  Apply the two creation operators and decompose
  the one-creation level under rotations.  For \(j>0\), determine every spin
  and its multiplicity, and verify equality of bosonic and fermionic
  degrees of freedom for both integral and half-integral \(j\).

  \SupplementaryExercise{16}{The Collapsed Massive Multiplet}
    {(Adapted from Adel Bilal, \emph{Introduction to Supersymmetry},
    Section 3.4)}
    {bilal-introduction-susy-2001}
  Repeat the massive \(N=1\) construction for a spin-zero Clifford vacuum.
  Explain why the formal \(j-1/2\) representation is absent, and show that
  the spectrum consists of two spin-zero particles and one spin-\(1/2\)
  particle.  Count physical spin states.

  \SupplementaryExercise{17}{Parity Along the Massive Ladder}
    {(Adapted from Adel Bilal, \emph{Introduction to Supersymmetry},
    Sections 3.2 and 3.4)}
    {bilal-introduction-susy-2001}
  Assume parity is conserved and choose the phase convention
  \(P^{-1}Q_aP=i\epsilon_{ab}Q_b^\dagger\).  If the one-creation states of
  spin \(j\pm1/2\) have parity \(\eta\), prove that the two spin-\(j\)
  combinations have parities \(+i\eta\) and \(-i\eta\).  Specialize the
  result to the collapsed \(j=0\) multiplet.

  \SupplementaryExercise{18}{The Massless-Frame Oscillator}
    {(Adapted from David Tong, \emph{Supersymmetric Field Theory},
    Section 2.3.2)}
    {tong-supersymmetric-field-theory-2022}
  Take \(p^\mu=(E,0,0,E)\) in a convention where
  \(\{Q_\alpha,Q^\dagger_{\dot\beta}\}
  =2\sigma^\mu_{\alpha\dot\beta}p_\mu\).  Diagonalize the resulting
  \(2\times2\) matrix.  Show that one component of \(Q\) has zero norm and
  hence acts trivially, while the other becomes a single fermionic
  oscillator.  State how its adjoint changes helicity.

  \SupplementaryExercise{19}{The Exterior Algebra of Helicity}
    {(Adapted from David Tong, \emph{Supersymmetric Field Theory},
    Section 2.4.1)}
    {tong-supersymmetric-field-theory-2022}
  For \(N\) massless supercharges, identify the state space generated from a
  highest-helicity state with \(\bigwedge^\bullet\mathbb C^N\).
  Derive the multiplicity \(\binom Nk\) and helicity
  \(\lambda_{\max}-k/2\) at level \(k\).  Prove that the total number of
  states is \(2^N\).

  \SupplementaryExercise{20}{The CPT Self-Conjugacy Test}
    {(Adapted from David Tong, \emph{Supersymmetric Field Theory},
    Section 2.4.1)}
    {tong-supersymmetric-field-theory-2022}
  A massless multiplet runs from \(\lambda_{\max}\) to
  \(\lambda_{\max}-N/2\).  Derive the condition for this helicity interval
  to be invariant under CPT.  If it is not invariant, write the top
  helicity of the required CPT-conjugate multiplet.

  \SupplementaryExercise{21}{The Massless Chiral Multiplet}
    {(Adapted from Stephen P. Martin, \emph{A Supersymmetry Primer},
    version 7, Sections 1 and 3.1)}
    {martin-susy-primer-v7}
  Construct the CPT-complete massless \(N=1\) multiplet containing a
  left-handed Weyl fermion.  List the helicities before and after CPT
  completion, interpret the result as one complex scalar plus one Weyl
  fermion, and count on-shell bosonic and fermionic degrees of freedom.

  \SupplementaryExercise{22}{The Massless Vector Multiplet}
    {(Adapted from Stephen P. Martin, \emph{A Supersymmetry Primer},
    version 7, Sections 1 and 3.3)}
    {martin-susy-primer-v7}
  Starting from helicity \(+1\), construct the \(N=1\) massless multiplet
  and its CPT conjugate.  Show that the completed spectrum is a massless
  vector plus a Majorana gaugino, and verify the equality of physical
  bosonic and fermionic polarizations.

  \SupplementaryExercise{23}{The \(N=2\) Hypermultiplet}
    {(Adapted from David Tong, \emph{Supersymmetric Field Theory},
    Section 2.4.1)}
    {tong-supersymmetric-field-theory-2022}
  Begin with a massless \(N=2\) Clifford vacuum of helicity \(+1/2\).
  Construct all levels, add the CPT conjugate when required by the
  \(SU(2)_R\) representation, and show that the physical content can be
  organized as one Dirac fermion and two complex scalars.  Give the
  \(SU(2)_R\) transformation of the scalars.

  \SupplementaryExercise{24}{The \(N=2\) Vector Multiplet}
    {(Adapted from David Tong, \emph{Supersymmetric Field Theory},
    Section 2.4.1)}
    {tong-supersymmetric-field-theory-2022}
  Begin with helicity \(+1\) and construct the CPT-complete massless \(N=2\)
  vector multiplet.  Tabulate the multiplicities at helicities
  \(1,1/2,0,-1/2,-1\), identify the \(SU(2)_R\) doublets, and count the
  physical degrees of freedom.

  \SupplementaryExercise{25}{Pascal's Triangle and \(N=4\)}
    {(Adapted from David Tong, \emph{Supersymmetric Field Theory},
    Section 2.4.1)}
    {tong-supersymmetric-field-theory-2022}
  Starting from helicity \(+1\), construct the massless \(N=4\) vector
  multiplet.  Show directly that it is CPT self-conjugate, list the
  multiplicities \(1,4,6,4,1\), and identify the spins and the corresponding
  \(SU(4)_R\) representations.

  \SupplementaryExercise{26}{The \(N=8\) Gravity Multiplet}
    {(Adapted from David Tong, \emph{Supersymmetric Field Theory},
    Section 2.4.1)}
    {tong-supersymmetric-field-theory-2022}
  Starting from helicity \(+2\), use the exterior-algebra construction to
  list the complete \(N=8\) massless supergravity multiplet.  Identify the
  spin-2, spin-\(3/2\), spin-1, spin-\(1/2\), and spin-0 multiplicities and
  verify CPT self-conjugacy.

  \SupplementaryExercise{27}{Helicity Cutoffs on Extended SUSY}
    {(Adapted from David Tong, \emph{Supersymmetric Field Theory},
    Section 2.4.1)}
    {tong-supersymmetric-field-theory-2022}
  Use the helicity width \(N/2\) of a massless multiplet to prove that a
  CPT-self-conjugate theory with no particles of spin above \(1\) has
  \(N\leq4\), while one with no particles of spin above \(2\) has
  \(N\leq8\).  State the assumptions under which these are bounds on global
  supersymmetry and supergravity, respectively.

  \SupplementaryExercise{28}{Why Perturbative \(N=3\) Becomes \(N=4\)}
    {(Adapted from David Tong, \emph{Supersymmetric Field Theory},
    Section 2.4.1)}
    {tong-supersymmetric-field-theory-2022}
  Construct an \(N=3\) massless multiplet beginning at helicity \(+1\) and
  add its CPT conjugate.  Sum the multiplicities helicity by helicity and
  compare them with an \(N=4\) vector multiplet.  Explain both the
  enhancement suggested by this count and why the single-particle argument
  is explicitly perturbative.

  \SupplementaryExercise{29}{Helicity Supertraces}
    {(Adapted from David Tong, \emph{Supersymmetric Field Theory},
    Sections 2.3--2.4)}
    {tong-supersymmetric-field-theory-2022}
  For a massless \(N\)-extended multiplet define
  \(B_m=\sum_{\text{states}}(-1)^F\lambda^m\).
  Using the generating function
  \(\sum(-1)^F e^{t\lambda}
  =e^{t\lambda_{\max}}(1-e^{-t/2})^N\), prove that
  \(B_m=0\) for \(m<N\).  Compute \(B_N\).

  \SupplementaryExercise{30}{Parity and the Central-Charge Phase}
    {(Adapted from David Tong, \emph{Supersymmetric Field Theory},
    Sections 2.2.1 and 2.4.2)}
    {tong-supersymmetric-field-theory-2022}
  With
  \(P^{-1}Q_{\alpha r}P=i\epsilon_{\alpha\beta}
  Q^\dagger_{\dot\beta r}\) and
  \(\{Q_{\alpha r},Q_{\beta s}\}
  =\epsilon_{\alpha\beta}Z_{rs}\), determine the parity transform of
  \(Z_{rs}\).  Explain when a single BPS central-charge sector can carry
  parity eigenstates and when parity must exchange two sectors.  Relate this
  to the freedom to rephase the supercharges.
\end{enumerate}
