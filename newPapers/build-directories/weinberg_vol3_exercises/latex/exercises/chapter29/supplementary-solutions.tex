\begin{enumerate}
  \SupplementarySolution{1}{Supersymmetric Quantum Mechanics and Normalizable Ground States}
  \begin{enumerate}
    \item Complex conjugation reverses the order of the Grassmann factors.
    In particular,
    \[
      \bigl(i\bar\psi\dot\psi\bigr)^*
       =i\bar\psi\dot\psi
        -i\frac{d}{dt}(\bar\psi\psi),
    \]
    while the remaining terms in (29.1.1) are real.  Direct substitution
    of (29.1.2), using \(\psi^2=\bar\psi^2=0\), cancels all bulk terms;
    the terms left over are total time derivatives proportional to
    \(d(h'\psi)/dt\) and \(d(h'\bar\psi)/dt\).

    The canonical bosonic momentum is \(p=\dot x\).  A convenient phase
    convention for the two Noether charges is
    \[
      \boxed{
      Q=-\psi\bigl(p-ih'(x)\bigr),
      \qquad
      \bar Q=-\bar\psi\bigl(p+ih'(x)\bigr),}
      \tag{29.1.S1}
    \]
    where \(Q\) multiplies \(\bar\epsilon\), \(\bar Q\) multiplies
    \(\epsilon\), and \(\bar Q=Q^\dagger\).  A simultaneous overall phase
    change of the charges and supersymmetry parameters is physically
    immaterial.

    \item The second-class fermionic constraint from the first-order
    kinetic term gives, after quantization,
    \[
      [x,p]=i,
      \qquad \{\psi,\bar\psi\}=1,
      \qquad \{\psi,\psi\}=\{\bar\psi,\bar\psi\}=0,
      \tag{29.1.S2}
    \]
    with all bosonic operators commuting with the fermionic ones.  Take
    \(p=-i\,d/dx\), identify \(\bar\psi\) with exterior multiplication by
    \(dx\), and identify \(\psi\) with contraction by
    \(\partial_x\).  The Hilbert space is then
    \[
      \mathcal H=L^2\!\left(\mathbb R,\Lambda^*T^*\mathbb R\right),
      \qquad
      \Psi(x)=f_0(x)+f_1(x)\,dx.
      \tag{29.1.S3}
    \]
    In this representation,
    \[
      Q=i\,\iota_{\partial_x}\bigl(\partial_x+h'\bigr),
      \qquad
      \bar Q=i\,dx\wedge\bigl(\partial_x-h'\bigr).
      \tag{29.1.S4}
    \]
    Thus the supercharges are the adjoint pair of differentials obtained
    by conjugating the de Rham differential by \(e^{-h}\).  Their
    anticommutator gives
    \[
      H=\frac12\{Q,\bar Q\}
       =\frac12\left[-\partial_x^2+(h')^2
          +h''\bigl(1-2\bar\psi\psi\bigr)\right]\geq0.
      \tag{29.1.S5}
    \]

    \item A zero-energy state must be annihilated by both supercharges.
    Equation (29.1.S4) gives the two formal possibilities
    \[
      f_0(x)=C_0e^{h(x)},
      \qquad
      f_1(x)=C_1e^{-h(x)}.
      \tag{29.1.S6}
    \]
    For \(h=-x^4\), only the zero-form is square integrable:
    \[
      \boxed{\Psi_0(x)=C e^{-x^4},\qquad
      |C|^{-2}=\int_{-\infty}^{\infty}e^{-2x^4}\,dx<\infty.}
      \tag{29.1.S7}
    \]
    The one-form candidate grows as \(e^{x^4}\), so the ground state is
    unique.  A generic cubic has opposite signs at the two ends of the real
    line.  Consequently \(e^h\) diverges at one end and \(e^{-h}\) at the
    other; neither formal state in (29.1.S6) is normalizable, and there is
    no zero-energy state.
  \end{enumerate}

  \SupplementarySolution{2}{The Goldstino Direction}
At a stationary vacuum the tree-level fermion mass matrix annihilates
  the vector
  \[
    v_n=\sqrt2\,\mathcal F_{n0},\qquad
    v_A=iD_{A0}.
  \]
  The same statement holds nonperturbatively for the one-particle pole
  residues of the corresponding fermion fields.  Define
  \[
    \mathcal N^2
      =2\sum_n\lvert\mathcal F_{n0}\rvert^2
       +\sum_A D_{A0}^2 .
  \]
  Up to an overall Majorana phase, the normalized left-handed goldstino
  is therefore
  \[
    G_L=\frac{
      \sqrt2\sum_n\mathcal F_{n0}\psi_{nL}
      +i\sum_A D_{A0}\lambda_{AL}}
      {\mathcal N}.
  \]
  Section 29.2 gives
  \[
    \rho_{\rm VAC}
      =\sum_n\lvert\mathcal F_{n0}\rvert^2
       +\frac12\sum_A D_{A0}^2
      =\frac{\mathcal N^2}{2}.
  \]
  Thus the norm of the goldstino direction is fixed directly by the
  positive vacuum energy.

  \SupplementarySolution{3}{Two-Dimensional Gauge Superspace and the Axial Anomaly}
  \begin{enumerate}
    \item Let \(\Lambda\) be a chiral gauge parameter,
    \(\bar D_\pm\Lambda=0\).  In one standard Abelian convention,
    \[
      \Phi^i\longmapsto e^{iq_i\Lambda}\Phi^i,
      \qquad
      \bar\Phi^{\bar i}\longmapsto
        \bar\Phi^{\bar i}e^{-iq_i\bar\Lambda},
      \qquad
      V\longmapsto V+i(\bar\Lambda-\Lambda).
      \tag{29.3.S1}
    \]
    These transformations leave
    \(\bar\Phi^{\bar i}e^{q_iV}\Phi^i\) invariant.  Wess--Zumino gauge
    uses the nonphysical components of \(\Lambda\) to remove the lowest
    scalar and spinor components of \(V\); the surviving components are
    the gauge field, the complex vector-multiplet scalar, the gauginos,
    and the real auxiliary field.  Ordinary gauge transformations remain
    as the residual gauge freedom.

    Define
    \[
      \boxed{\Sigma=\frac1{\sqrt2}\bar D_+D_-V.}
      \tag{29.3.S2}
    \]
    The normalization is conventional.  Its variation contains
    \(\bar D_+D_-\Lambda\), which vanishes because
    \(\bar D_+\Lambda=0\) and \(\{\bar D_+,D_-\}=0\), and
    \(\bar D_+D_-\bar\Lambda=0\), which vanishes because
    \(D_-\bar\Lambda=0\).  Hence \(\Sigma\) is gauge invariant.  Moreover,
    \[
      \bar D_+\Sigma=0,
      \qquad D_-\Sigma=0,
      \tag{29.3.S3}
    \]
    by nilpotence and the same anticommutator.  These are precisely the
    twisted-chiral constraints.

    \item In the Cambridge-course axial convention,
    \(R_A(\theta^\pm)=\mp1\) and
    \(R_A(\bar\theta^\pm)=\pm1\).  The twisted-chiral measure
    \(d\theta^+d\bar\theta^-\) therefore has charge \(+2\).  Treating the
    constant FI parameter \(t\) as neutral, invariance of (29.3.1) requires
    \[
      \boxed{R_A(\Sigma)=-2.}
      \tag{29.3.S4}
    \]
    Reversing the sign convention for the axial generator reverses both
    charges and gives the equivalent assignment
    \(R_A(d\theta^+d\bar\theta^-)=-2\), \(R_A(\Sigma)=+2\).

    \item The two matter fermions in a charged \(N=(2,2)\) chiral
    multiplet have opposite world-sheet chirality, so their vector charges
    cancel in the vector-current anomaly but add in the axial-current
    anomaly.  For background flux
    \[
      k=\frac1{2\pi}\int_{T^2}F\in\mathbb Z,
    \]
    the charged Dirac operator has index \(q_i k\).  A Fujikawa axial
    rotation by \(\alpha\) therefore changes the complete fermion measure
    by a phase proportional to
    \[
      \exp\!\left(2i\alpha k\sum_iq_i\right),
      \tag{29.3.S5}
    \]
    equivalently
    \[
      \partial_\mu j_A^\mu
        =\frac1\pi\left(\sum_iq_i\right)F_{01}
      \tag{29.3.S6}
    \]
    in these normalizations.  Since the torus admits arbitrary integral
    flux, a continuous axial symmetry is present for every background
    precisely when
    \[
      \boxed{\sum_iq_i=0.}
    \]
    Otherwise the functional measure is anomalous (and the axial rotation
    shifts the theta angle), even though the classical superspace action is
    invariant.
  \end{enumerate}

  \SupplementarySolution{4}{The Supersymmetric One-Loop Gauge Coefficient}
In a convention with
  \[
    \mu\frac{dg}{d\mu}=-\frac{b_0}{16\pi^2}g^3,
  \]
  gauge bosons contribute \(11C_1/3\), a Majorana gaugino contributes
  \(-2C_1/3\), a left-handed Weyl fermion contributes
  \(-2T(R)/3\), and a complex scalar contributes \(-T(R)/3\).
  The vector multiplet therefore contributes \(3C_1\), while each
  chiral multiplet contributes \(-T(R)\).  Summing all chiral
  representations gives
  \[
    b_0=3C_1-C_2.
  \]
  For \(SU(N_c)\), \(C_1=N_c\).  Each fundamental or antifundamental has
  \(T=1/2\), so \(N_f\) flavor pairs give \(C_2=N_f\), and
  \[
    b_0=3N_c-N_f.
  \]
  This is positive precisely in the asymptotically free range.

  \SupplementarySolution{5}{The Instanton Weight as a Chiral Parameter}
A Euclidean configuration of winding number \(\nu>0\) has
  \[
    I_E=\frac{8\pi^2\nu}{g^2}-i\nu\theta.
  \]
  Its semiclassical weight is consequently
  \[
    e^{-I_E}
      =\exp\left(-\frac{8\pi^2\nu}{g^2}+i\nu\theta\right)
      =\exp(2\pi i\nu\tau),
    \qquad
    \tau=\frac{4\pi i}{g^2}+\frac{\theta}{2\pi}.
  \]
  A Wilsonian \(\mathcal F\)-term is holomorphic in the chiral
  parameter \(\tau\), so positive-winding instantons supply positive
  powers of \(e^{2\pi i\tau}\).  Negative powers would grow like
  \(e^{+8\pi^2/g^2}\) as \(g\to0\), contradicting the weak-coupling
  expansion.  Anti-instantons instead depend on \(\tau^*\) and
  contribute to the conjugate \(\mathcal F\)-term, not to the same
  holomorphic function.

  \SupplementarySolution{6}{\(SU(2)\) SQCD: Runaways, Quantum Moduli, and Anomaly Matching}
  \begin{enumerate}
    \item A holomorphic instanton factor transforms with the charges of
    the fermion zero-mode measure.  The baryon-number contributions of
    \(\Phi\) and \(\widetilde\Phi\) cancel.  For \(U(1)_A\), each of the
    \(2N_f\) matter zero modes has charge one.  For \(U(1)_R\), the four
    adjoint gaugino zero modes have charge \(+1\), whereas each matter
    fermion has charge
    \(R[\Phi]-1=-2/N_f\).  Hence
    \[
      \begin{array}{c|ccc}
        &U(1)_B&U(1)_A&U(1)_R\\ \hline
        \Lambda^{6-N_f}&0&2N_f&0 .
      \end{array}
      \tag{29.6.S1}
    \]
    Equivalently, an axial rotation by \(\alpha\) sends
    \(\Lambda^{6-N_f}\to e^{2iN_f\alpha}\Lambda^{6-N_f}\), while the
    other two symmetries leave it invariant.  The vanishing total
    \(R\)-charge, \(4+2N_f(-2/N_f)=0\), is the anomaly-cancellation
    condition.

    \item For \(N_f=1\), the only nonzero gauge invariant is
    \[
      M=\widetilde\Phi_a\Phi^a.
    \]
    A would-be baryon made from two identical commuting \(\Phi\)'s
    vanishes after contraction with \(\epsilon_{ab}\), and similarly for
    \(\widetilde\Phi\).  There is therefore no algebraic constraint on
    \(M\).  Its charges are
    \((B,A,R)=(0,2,-2)\), while \(\Lambda^5\) has charges
    \((0,2,0)\).  Holomorphy, dimension, and all symmetries fix
    \[
      \boxed{W_{N_f=1}=\frac{\Lambda^5}{M}.}
      \tag{29.6.S2}
    \]
    Since \(\partial_MW=-\Lambda^5/M^2\) never vanishes at finite
    \(M\), there is no supersymmetric vacuum at finite distance.  The
    potential approaches zero only as \(|M|\to\infty\): the theory has a
    supersymmetric runaway rather than a normalizable finite-field ground
    state.

    \item For \(N_f=2\), write
    \[
      M_i{}^j=\widetilde\Phi^j_a\Phi_i^a,
      \qquad
      B=\epsilon_{ab}\Phi_1^a\Phi_2^b,
      \qquad
      \widetilde B=\epsilon^{ab}\widetilde\Phi^1_a
        \widetilde\Phi^2_b.
      \tag{29.6.S3}
    \]
    The rank-two determinant identity gives the classical constraint
    \[
      \det M-B\widetilde B=0.
      \tag{29.6.S4}
    \]
    Every quadratic invariant has axial charge two, baryon charge zero or
    \(\pm2\), and \(R\)-charge zero.  No holomorphic function regular on
    the moduli space has both dimension three and \(R\)-charge two, so no
    dynamical superpotential is generated.  However, both sides of
    (29.6.S4) have the same charges and dimension as \(\Lambda^4\).
    The symmetry-allowed quantum moduli space is therefore
    \[
      \boxed{\det M-B\widetilde B=\Lambda^4.}
      \tag{29.6.S5}
    \]
    The nonzero right-hand side smooths the classical conical singularity
    and removes the origin.

    \item For \(N_f=3\), the infrared chiral coordinates are
    \[
      M_i{}^j=\widetilde\Phi^j\Phi_i,
      \qquad
      B_{ij}=\epsilon_{ab}\Phi_i^a\Phi_j^b,
      \qquad
      \widetilde B^{ij}=\epsilon^{ab}\widetilde\Phi^i_a
        \widetilde\Phi^j_b,
      \tag{29.6.S6}
    \]
    with \(B_{ij}\) and \(\widetilde B^{ij}\) antisymmetric.  At the
    origin they provide \(9+3+3=15\) unconstrained chiral multiplets.

    A microscopic matter fermion has \(R=-2/3\); there are
    \(2\times2\times3=12\) of them.  The three gauginos have charge one.
    Thus
    \[
      \mathcal A_R^{\rm UV}=3+12\left(-\frac23\right)=-5,
      \qquad
      \mathcal A_{R^3}^{\rm UV}
       =3+12\left(-\frac23\right)^3=-\frac59.
      \tag{29.6.S7}
    \]
    Each composite superfield has \(R=2/3\), so each of its fifteen Weyl
    fermions has charge \(-1/3\).  Consequently
    \[
      \boxed{
      \mathcal A_R^{\rm IR}=15\left(-\frac13\right)=-5,
      \qquad
      \mathcal A_{R^3}^{\rm IR}
       =15\left(-\frac13\right)^3=-\frac59,}
      \tag{29.6.S8}
    \]
    exactly matching the ultraviolet anomalies.
  \end{enumerate}

  \SupplementarySolution{7}{Special Geometry and Abelian Dualization}
  \begin{enumerate}
    \item \emph{Metric from the \(N=2\) Holomorphic Function.}
    Differentiating the Kahler potential gives
      \[
        K_{aa^*}
          =\frac{1}{4\pi}\operatorname{Im}h'(a)
          =\operatorname{Re}T(a).
      \]
      This is the common coefficient of the scalar, two fermion, and Abelian
      gauge kinetic terms required by \(N=2\) supersymmetry.  Local unitarity
      therefore requires
      \[
        \operatorname{Im}h'(a)>0.
      \]
      Perturbatively,
      \(h'(a)\sim(2i/\pi)\ln(a/\Lambda)\), up to an additive constant.
      Its imaginary part decreases without bound when \(\lvert a\rvert\) is
      taken sufficiently small.  Because a negative kinetic coefficient is
      impossible, the one-loop logarithm cannot describe the origin; new
      singularities and a dual weakly coupled description must intervene.
    \item \emph{First-Order Abelian Dualization.}
    Add a real multiplier \(\widetilde V\) so that its field strength
      \(\widetilde W_L\) couples linearly to the unconstrained \(W_L\).
      The \(W_L\)-dependent part of the action is quadratic, and stationarity
      gives
      \[
        W_L=-\frac{\widetilde W_L}{h'(\Phi)}.
      \]
      Substitution reverses the quadratic coefficient.  Defining
      \[
        \Phi_D=h(\Phi),\qquad h_D(\Phi_D)=-\Phi
      \]
      preserves the original \(N=2\) form because
      \[
        h_D'(\Phi_D)=-\frac{1}{h'(\Phi)}.
      \]
      Thus \(a_D=h(a)\).  Since
      \(T=h'/(4\pi i)\), the dual coefficient is
      \[
        T_D=\frac{h_D'}{4\pi i}
          =-\frac{1}{4\pi i\,h'}
          =\frac{1}{16\pi^2T}.
      \]
      Integrating out \(\widetilde V\) instead simply restores the original
      Bianchi identity, proving equivalence of the two formulations.
  \end{enumerate}

  \SupplementarySolution{8}{Integral Duality and the Dyon Lattice}
  \begin{enumerate}
    \item \emph{Generators of the Integral Duality Group.}
    Electric--magnetic interchange and an integral theta shift act as
      \[
        S=\begin{pmatrix}0&1\\-1&0\end{pmatrix},
        \qquad
        T_b=\begin{pmatrix}1&0\\b&1\end{pmatrix},
        \quad b\in\mathbb Z.
      \]
      Both matrices have determinant one, so every product has integral
      entries and unit determinant; they generate \(SL(2,\mathbb Z)\).
      For
      \[
        M=\begin{pmatrix}n&m\\k&l\end{pmatrix},
        \qquad
        \binom{a'}{a_D'}=M\binom{a}{a_D},
      \]
      set \(\tau=h'(a)=da_D/da\).  Differentiation gives
      \[
        \tau'=\frac{da_D'}{da'}
          =\frac{k+l\tau}{n+m\tau}.
      \]
      The identity
      \(\operatorname{Im}\tau'
       =\operatorname{Im}\tau/\lvert n+m\tau\rvert^2\)
      shows that the positive kinetic region is preserved.
    \item \emph{Theta Shift and the Dyon Lattice.}
    The shift \(a_D\to a_D+ba\) is represented by \(T_b\) from the
      preceding part.  To keep
      \(na+ma_D\) invariant, the integer row of charges must transform with
      the inverse matrix:
      \[
        (n,m)\longrightarrow(n,m)T_b^{-1}=(n-bm,m).
      \]
      Therefore the central charge and
      \(M_{\rm BPS}=\sqrt2\lvert na+ma_D\rvert\) are unchanged; only the
      labels assigned in the shifted electric frame differ.  Since the real
      part of \(a_D/a\) determines the electric charge induced on a magnetic
      state, increasing the theta angle by \(2\pi b\) changes a unit
      monopole's electric charge by \(be\).  Relabeling \(n\) by
      \(n-bm\) leaves the full dyon lattice invariant, which is the Witten
      effect and theta periodicity together.
  \end{enumerate}

  \SupplementarySolution{9}{Seiberg--Witten Monodromies and BPS Covariance}
  \begin{enumerate}
    \item \emph{Weak-Coupling Monodromy at Infinity.}
    When \(u\) goes once counterclockwise around a sufficiently large
      circle,
      \[
        \sqrt{2u}\longrightarrow-\sqrt{2u},\qquad
        \ln(2u/\Lambda^2)\longrightarrow
          \ln(2u/\Lambda^2)+2\pi i.
      \]
      Hence \(a\to-a\).  Substituting both changes in the asymptotic formula
      for \(a_D\) gives
      \[
        a_D\longrightarrow 2a-a_D.
      \]
      In the ordering used in Chapter 29,
      \[
        \binom a{a_D}\longrightarrow
        M_\infty\binom a{a_D},
        \qquad
        M_\infty=
          \begin{pmatrix}-1&0\\2&-1\end{pmatrix}.
      \]
      Its determinant is one, as required for an allowed integral duality
      transformation.
    \item \emph{Monodromy of a Massless Monopole.}
    Encircling \(u_0\) once sends
      \[
        \ln(a_D/\Lambda_0)\longrightarrow
          \ln(a_D/\Lambda_0)+2\pi i,
      \]
      while the analytic coordinate \(a_D\) returns to itself.  The
      logarithmic term in \(a\) consequently changes by
      \[
        \frac{i}{\pi}a_D(2\pi i)=-2a_D.
      \]
      Therefore
      \[
        a\longrightarrow a-2a_D,\qquad a_D\longrightarrow a_D,
      \]
      or
      \[
        M_{\rm mono}=
          \begin{pmatrix}1&-2\\0&1\end{pmatrix}.
      \]
      The invariant vanishing combination is \(a_D\), confirming that the
      light state has electric--magnetic charge vector \((0,1)\).
    \item \emph{Factorization of Seiberg--Witten Monodromies.}
    The ordered product is
      \[
        M_+M_-=
        \begin{pmatrix}1&-2\\0&1\end{pmatrix}
        \begin{pmatrix}3&-2\\2&-1\end{pmatrix}
        =
        \begin{pmatrix}-1&0\\2&-1\end{pmatrix}
        =M_\infty.
      \]
      Thus a large contour can be deformed into contours around the two
      finite singularities with the stated ordering.  At \(u=+1\), the
      vanishing period is \(a_D\), so the massless state has charge
      \((n,m)=(0,1)\).  At \(u=-1\), it is \(a_D-a\), corresponding to
      \((n,m)=(-1,1)\).  Their charges have unit symplectic pairing, so no
      single electric frame makes both particles elementary everywhere.
    \item \emph{Duality Covariance of the BPS Spectrum.}
    Write the period column as \(v=(a,a_D)^{\mathsf T}\) and the charge row
      as \(q=(n,m)\).  Since the central charge is proportional to \(qv\),
      the transformation \(v\to Mv\) must be accompanied by
      \[
        q\longrightarrow q'=qM^{-1}.
      \]
      An \(SL(2,\mathbb Z)\) matrix has an integral inverse, so integer
      electric and magnetic charges remain integral.  Then
      \[
        q'v'=qM^{-1}Mv=qv,
      \]
      proving exact invariance of \(Z\) and
      \(M_{\rm BPS}=\lvert Z\rvert/2\).  The transformation changes the
      electric--magnetic description of a state, not its physical mass.
  \end{enumerate}

  \SupplementarySolution{10}{\(Sp(N_c)\) Conformal Window and Seiberg Duality}
  \begin{enumerate}
    \item[(i)]
    The fermion in a chiral multiplet of superfield charge \(R[\Phi]\)
    has charge \(R[\Phi]-1\), whereas the gaugino has charge \(+1\).
    Using the indices supplied in (29.10.5), cancellation of the mixed
    \(Sp(N_c)^2U(1)_R\) anomaly gives
    \[
      2(N_c+1)+2N_f\bigl(R[\Phi]-1\bigr)=0.
    \]
    Therefore
    \[
      \boxed{R[\Phi]=1-\frac{N_c+1}{N_f}
        =\frac{N_f-N_c-1}{N_f}.}
      \tag{29.10.S1}
    \]

    \item[(ii)]
    The one-loop coefficient (29.10.4) is
    \[
      b_0^{\rm el}
       =\frac32\,2(N_c+1)-\frac12(2N_f)
       =3(N_c+1)-N_f.
      \tag{29.10.S2}
    \]
    The theory is asymptotically free for \(b_0^{\rm el}>0\), and it
    ceases to be so at
    \[
      \boxed{N_f=3(N_c+1).}
    \]

    \item[(iii)]
    At a four-dimensional \(N=1\) superconformal fixed point, a chiral
    primary obeys \(\Delta=\tfrac32R\).  Since
    \(R[M]=2R[\Phi]\),
    \[
      \Delta(M)=3\left(1-\frac{N_c+1}{N_f}\right).
      \tag{29.10.S3}
    \]
    At \(N_f=3(N_c+1)\), this becomes
    \(\boxed{\Delta(M)=2}\), as requested.

    \item[(iv)]
    A gauge-invariant scalar saturates the free-field unitarity bound at
    \(\Delta=1\).  Equation (29.10.S3) reaches this value when
    \[
      N_f=\frac32(N_c+1).
    \]
    The expected interacting conformal window is therefore
    \[
      \boxed{\frac32(N_c+1)<N_f<3(N_c+1).}
      \tag{29.10.S4}
    \]

    \item[(v)]
    Put
    \(\widetilde N_c=N_f-N_c-2\).  The magnetic mixed anomaly vanishes
    when
    \[
      2(\widetilde N_c+1)
       +2N_f\bigl(R[q]-1\bigr)=0.
    \]
    Hence
    \[
      \boxed{R[q]
       =1-\frac{\widetilde N_c+1}{N_f}
       =\frac{N_c+1}{N_f}.}
      \tag{29.10.S5}
    \]

    \item[(vi)]
    The superpotential \(qMq\) must have \(R\)-charge two, so
    \[
      \boxed{R[M]=2-2R[q]
        =2\left(1-\frac{N_c+1}{N_f}\right).}
    \]
    Therefore
    \[
      \boxed{\Delta(M)=\frac32R[M]
        =3\left(1-\frac{N_c+1}{N_f}\right),}
    \]
    which is exactly the charge and dimension of the electric composite
    \(\Phi^i\Phi^j\).

    \item[(vii)]
    The magnetic coefficient is
    \[
      \begin{split}
      b_0^{\rm mag}
       &=3(\widetilde N_c+1)-N_f\\
       &=2N_f-3(N_c+1).
      \end{split}
      \tag{29.10.S6}
    \]
    Thus the magnetic theory ceases to be asymptotically free at
    \(N_f=\tfrac32(N_c+1)\).  For a nontrivial magnetic gauge group, its
    infrared-free range is
    \[
      \boxed{N_c+3\leq N_f\leq\frac32(N_c+1),}
    \]
    where the upper equality is the marginally-free boundary.  This is
    precisely the lower edge inferred from the electric meson's unitarity
    bound.  Inside the strict conformal window both descriptions flow to
    the same interacting infrared fixed point; in the range displayed
    above, the magnetic variables furnish the weakly coupled description.

    \item[(viii)]
    The electric quark fermions are flavor fundamentals, with one copy
    for each of the \(2N_c\) gauge components:
    \[
      \mathcal A_{SU(2N_f)^3}^{\rm el}=2N_c.
    \]
    In the magnetic theory, the \(2\widetilde N_c\) gauge components of
    \(q\) transform as flavor antifundamentals and contribute
    \(-2\widetilde N_c\).  The antisymmetric meson fermion has anomaly
    coefficient \(2N_f-4\).  Consequently
    \[
      \boxed{\mathcal A_{SU(2N_f)^3}^{\rm mag}
        =-2(N_f-N_c-2)+(2N_f-4)
        =2N_c
        =\mathcal A_{SU(2N_f)^3}^{\rm el}.}
      \tag{29.10.S7}
    \]
  \end{enumerate}

  \SupplementarySolution{11}{Chiral Gauge Theory and 't Hooft Anomaly Matching}
  \begin{enumerate}
    \item[(a)]
    A gauge anomaly violates a redundancy needed to remove negative-norm
    polarizations.  It spoils the Ward or BRST identities and makes the
    gauge theory inconsistent, so it must cancel.  An ABJ anomaly instead
    says that a classical global current is not conserved after
    quantization; the quantum theory can remain consistent, but the
    symmetry and its selection rules are lost.  A 't~Hooft anomaly is an
    obstruction to gauging a global symmetry, detected by coupling it to
    background fields.  It is invariant under renormalization-group flow,
    so it must be matched in the infrared by massless fields, spontaneous
    symmetry breaking, or suitable topological degrees of freedom.

    \item[(i)]
    With \(A(\wedge^2\boldsymbol N)=N-4\) and
    \(A(\overline{\boldsymbol N})=-1\), the gauge anomaly is
    \[
      \mathcal A_{SU(N)^3}=(N-4)-p.
    \]
    Consistency therefore requires
    \[
      \boxed{p=N-4.}
      \tag{29.11.S1}
    \]

    \item[(ii)]
    Classically, the \(p\) identical antifundamentals have a \(U(p)\)
    flavor symmetry, and \(\lambda\) has an independent phase rotation.
    Equivalently, this is
    \(SU(p)\times U(1)_\psi\times U(1)_\lambda\).
    One Abelian combination has a mixed \(SU(N)^2U(1)\) anomaly.  The
    surviving combination obeys
    \[
      (N-2)q_\lambda+p q_\psi=0.
    \]
    A convenient integral normalization is
    \[
      \boxed{q_\lambda=p=N-4,\qquad q_\psi=-(N-2).}
      \tag{29.11.S2}
    \]
    The exact continuous quantum global symmetry is therefore
    \(SU(p)\times U(1)\), up to discrete identifications.

    \item[(iii)]
    There are \(N\) gauge-color copies of the flavor fundamental
    \(\psi^i\).  Using the normalization in (29.11.2), the ultraviolet
    anomalies are
    \[
    \begin{aligned}
      \mathcal A_{SU(p)^3}^{\rm UV}
        &=N,\\
      \mathcal A_{SU(p)^2U(1)}^{\rm UV}
        &=Nq_\psi=-N(N-2),\\
      \mathcal A_{U(1)^3}^{\rm UV}
        &=\frac{N(N-1)}2\,(N-4)^3
          +N(N-4)\bigl(-(N-2)\bigr)^3,\\
      \mathcal A_{\mathrm{grav}^2U(1)}^{\rm UV}
        &=\frac{N(N-1)}2\,(N-4)
          +N(N-4)\bigl(-(N-2)\bigr).
    \end{aligned}
    \tag{29.11.S3}
    \]
    The first terms in the last two lines are the
    \(N(N-1)/2\) components of \(\lambda\); the second terms are the
    \(Np=N(N-4)\) components of the \(\psi^i\).

    \item[(iv)]
    The composite \(\chi^{ij}\) is symmetric in its flavor indices and
    has charge
    \[
      q_\chi=2q_\psi+q_\lambda=-N.
      \tag{29.11.S4}
    \]
    For \(\mathrm{Sym}^2\boldsymbol p\),
    \[
      \dim=\frac{p(p+1)}2,\qquad I=p+2,\qquad A=p+4.
    \]
    Its infrared anomalies are consequently
    \[
    \begin{aligned}
      \mathcal A_{SU(p)^3}^{\rm IR}
        &=p+4=N,\\
      \mathcal A_{SU(p)^2U(1)}^{\rm IR}
        &=(p+2)(-N)=-N(N-2),\\
      \mathcal A_{U(1)^3}^{\rm IR}
        &=\frac{p(p+1)}2(-N)^3,\\
      \mathcal A_{\mathrm{grav}^2U(1)}^{\rm IR}
        &=\frac{p(p+1)}2(-N).
    \end{aligned}
    \tag{29.11.S5}
    \]
    Substituting \(p=N-4\) into (29.11.S3) reduces its last two lines to
    the corresponding expressions in (29.11.S5).  All four 't~Hooft
    anomalies therefore match, so the proposed confining spectrum passes
    this necessary infrared consistency test.
  \end{enumerate}

  \SupplementarySolution{12}{Pseudomoduli in Dynamical Supersymmetry Breaking}
  \begin{enumerate}
    \item \emph{Broken Vacuum or Runaway?.}
    For chiral fields with smooth positive Kahler metric \(g_{i\bar j}\),
      eliminating auxiliary fields gives
      \[
        V=g^{i\bar j}f_i f_{\bar j}^*
          +\frac12\sum_A D_A^2\geq0.
      \]
      At a finite stationary point, failure of all equations
      \(f_i=D_A=0\) means \(V>0\).  The supersymmetry algebra then implies
      that the vacuum is not annihilated by the supercharges, so
      supersymmetry is spontaneously broken and a goldstino exists.
      This conclusion presupposes a normalizable ground state.  If instead
      there is a path with fields tending to infinity and \(V\to0\), the
      infimum is zero but is not attained at finite distance.  The system
      has no vacuum state at all; it is a runaway, not a stable
      supersymmetry-breaking vacuum.  This distinction is also why an index
      may change when large-field asymptotics change.
    \item \emph{O'Raifeartaigh Pseudomodulus and Goldstino.}
    The three \(\mathcal F\)-flatness conditions are
      \[
        f_X=f_0+\frac h2\phi_1^2,\qquad
        f_{\phi_1}=hX\phi_1+m\phi_2,\qquad
        f_{\phi_2}=m\phi_1.
      \]
      The last equation forces \(\phi_1=0\); then the first cannot vanish
      when \(f_0\ne0\), while the second sets \(\phi_2=0\).  Therefore
      \[
        \phi_1=\phi_2=0,\qquad X\ {\rm arbitrary},\qquad
        V_{\rm tree}=\lvert f_0\rvert^2.
      \]
      The condition \(m^2>\lvert hf_0\rvert\) keeps both real scalar
      eigenvalues \(m^2\pm\lvert hf_0\rvert\) positive at the origin.
      The complex scalar \(X\) is a tree-level pseudomodulus.  Only
      \(\mathcal F_X=-f_0^*\) has a vacuum expectation value, so the
      normalized goldstino is simply the Majorana fermion \(\psi_X\), up to
      an overall phase.
    \item \emph{Radiative Lifting of a Pseudomodulus.}
    The one-loop potential for a constant \(X\) is
      \[
        V_1(X)=\frac{1}{64\pi^2}
          \operatorname{STr}\left[
           \mathcal M^4(X)
           \left(\ln\frac{\mathcal M^2(X)}{\mu^2}-\frac32\right)
          \right].
      \]
      Supersymmetry breaking splits the two bosonic eigenvalues associated
      with \(\phi_1,\phi_2\) from their fermionic partners by quantities of
      order \(hf_0\), while the \(X\)-dependent supersymmetric masses vary
      through \(hX\).  Expanding the supertrace near \(X=0\) gives
      \[
        m_X^2
          \sim\frac{\lvert h\rvert^4}{16\pi^2}
           \frac{\lvert f_0\rvert^2}{\lvert m\rvert^2}
           \,\mathcal G\!\left(
            \frac{\lvert hf_0\rvert}{\lvert m\rvert^2}
           \right),
      \]
      where the basic stable model has a positive dimensionless function
      \(\mathcal G\).  Thus the scalar pseudomodulus is lifted
      radiatively.  The fermion \(\psi_X\) remains exactly massless because
      the broken supercurrent Ward identity protects the goldstino, even
      though no supersymmetry pairs it with the now-massive scalar.
  \end{enumerate}

  \SupplementarySolution{13}{\(R\)-Symmetry and the \(3\)-\(2\) Rank Obstruction}
  \begin{enumerate}
    \item \emph{A Generic \(R\)-Symmetry Counting Test.}
    At an \(R\)-symmetric point, all fields with nonzero \(R\) charge have
      zero expectation value.  A generic superpotential can then be written
      to leading order as
      \[
        f=\sum_{i=1}^{N_X}X_i\,g_i(Y_1,\ldots,Y_{N_Y})
          +\text{terms containing other charged fields},
      \]
      where \(R(X_i)=2\) and \(R(Y_a)=0\).  All auxiliary equations except
      those of the \(X_i\) vanish automatically at the symmetric locus, and
      the remaining conditions are
      \[
        g_i(Y)=0,\qquad i=1,\ldots,N_X.
      \]
      For generic functions, more independent equations than variables
      have no common solution.  Hence \(N_X>N_Y\) generically implies
      supersymmetry breaking at an \(R\)-symmetric vacuum.  When
      \(N_X\leq N_Y\), solutions generically exist; special coefficients,
      non-generic superpotentials, runaways, or vacua that spontaneously
      break \(R\) require a separate analysis.
    \item \emph{The \(3\)-\(2\) Model Rank Obstruction.}
    Denote the two \(SU(2)\) components of the \(SU(3)\) triplet \(Q\) by
      two flavors.  Strong \(SU(3)\) dynamics generates
      \[
        f_{\rm ADS}
          =\frac{\Lambda_3^7}
           {\det(\bar Q Q)}.
      \]
      The full superpotential is
      \(f=\lambda Q\bar D L+f_{\rm ADS}\).  The equation obtained by varying
      \(L\) is
      \[
        Q\bar D=0.
      \]
      Thus one column of the \(2\times2\) meson matrix
      \((\bar Q Q)\) vanishes, so its determinant is zero.  At precisely
      that locus the ADS term and its derivatives are singular; the
      remaining \(\mathcal F\)-term equations cannot vanish.  Moving toward
      large fields can reduce some \(\mathcal F\)-terms, but the
      \(SU(3)\) and \(SU(2)\) \(D\)-flatness conditions do not permit all
      dangerous directions simultaneously.  Their potential stabilizes the
      fields at finite values, leaving positive vacuum energy and dynamical
      supersymmetry breaking.
  \end{enumerate}

  \SupplementarySolution{14}{Quantum Rank Conditions and the Goldstino Scale}
  \begin{enumerate}
    \item \emph{Quantum-Constraint Rank Condition.}
    The confined coordinates are the antisymmetric mesons
      \(M_{ij}=Q_iQ_j\), constrained by
      \[
        \operatorname{Pf}M=\Lambda^{2N+2}.
      \]
      The singlet coupling gives
      \[
        \mathcal F_{S^{ij}}^*
          =-\lambda M_{ij}.
      \]
      Unbroken supersymmetry would therefore require every component of
      \(M\) to vanish.  That is impossible because its Pfaffian is fixed to
      a nonzero value by the exact quantum constraint.  Hence at least one
      singlet auxiliary field has a nonzero expectation value and the
      vacuum energy is positive.  A non-anomalous \(R\) symmetry assigns
      \(R(S)=2\) and \(R(M)=0\); it forbids additional generic terms that
      could remove the rank obstruction.  The goldstino is the normalized
      linear combination of singlet fermions weighted by their nonzero
      \(\mathcal F_S\) expectation values.
    \item \emph{Current Normalization of the Goldstino Scale.}
    Lorentz invariance and current conservation allow the leading
      vacuum-to-one-goldstino matrix element to be written
      \[
        \langle{\rm VAC}\lvert S^\mu(0)
          \rvert\mathbf p,\lambda\rangle
          =(2\pi)^{-3/2}
           \left[
            \frac{1+\gamma_5}{2}\gamma^\mu F_G
            +\frac{1-\gamma_5}{2}\gamma^\mu F_G^*
           \right]u(\mathbf p,\lambda)
      \]
      up to terms proportional to \(p^\mu\).  Inserting the two helicity
      states into the vacuum expectation value of the supercharge
      anticommutator and using
      \(\sum_\lambda u\bar u=-i\sl{p}/(2p^0)\) gives
      \[
        \rho_{\rm VAC}=\frac{\lvert F_G\rvert^2}{2}.
      \]
      The goldstino residues in matter fermions and gauginos are proportional
      to \(\sqrt2\langle\mathcal F_n\rangle/F_G\) and
      \(-\langle D_A\rangle/F_G\), respectively.  Normalizing the one-particle
      state therefore yields
      \[
        \lvert F_G\rvert^2
          =2\sum_n\lvert\langle\mathcal F_n\rangle\rvert^2
           +\sum_A\lvert\langle D_A\rangle\rvert^2,
      \]
      reproducing the auxiliary-field expression for the positive vacuum
      energy.
  \end{enumerate}

  \SupplementarySolution{15}{A Chiral \(SU(N)\)--\(\operatorname{Spin}(8)\) Seiberg Dual Pair}
  Seiberg-dual theories have different ultraviolet gauge fields and
  elementary matter, but their renormalization-group flows approach the
  same infrared quantum field theory.  Gauge-invariant operators and
  deformations must admit a consistent map, exact global symmetries and
  their 't~Hooft anomalies must agree, and correlation functions of the
  mapped infrared observables coincide.
  \begin{enumerate}
    \item[(i)] The cubic gauge-anomaly coefficient of the symmetric tensor
    is \(A(\mathrm{Sym}^2\boldsymbol N)=N+4\), while every gauge
    antifundamental contributes \(-1\).  Hence
    \[
      \mathcal A_{SU(N)^3}=N+4-p,
    \]
    and consistency fixes
    \[
      \boxed{p=N+4.}
      \tag{29.15.S1}
    \]

    \item[(ii)] The fermion in a chiral multiplet has charge one less than
    the superfield.  Thus
    \[
      R[\psi_S]=\frac{2-N}{2+N}-1=-\frac{2N}{N+2},
      \qquad R[\psi_{\widetilde\Phi}]=0.
    \]
    The mixed gauge anomaly is
    \[
      \mathcal A_{SU(N)^2U(1)_R}
       =2N+(N+2)\left(-\frac{2N}{N+2}\right)+p(1)(0)=0,
      \tag{29.15.S2}
    \]
    where the first term comes from the gaugino.  The proposed
    \(R\)-symmetry therefore survives quantization.

    \item[(iii)] Using \(p=N+4\), the electric coefficient is
    \[
      \begin{split}
      b_0^{\rm el}
       &=\frac32(2N)-\frac12\bigl[(N+2)+p\bigr]\\
       &=2N-3.
      \end{split}
      \tag{29.15.S3}
    \]
    It is positive for every nontrivial \(SU(N)\), namely \(N\geq2\), so
    the electric theory is asymptotically free.

    \item[(iv)] Only the magnetic gaugino, \(q\), and \(t\) can contribute
    to the mixed \(\operatorname{Spin}(8)^2U(1)_R\) anomaly.  Their
    fermion charges are \(1\), \(R[q]-1=0\), and
    \(R[t]-1=-6\), respectively.  Consequently
    \[
      \mathcal A_{\operatorname{Spin}(8)^2U(1)_R}
        =6+p(1)(0)+1(-6)=0.
      \tag{29.15.S4}
    \]
    The magnetic \(R\)-symmetry is nonanomalous as well.

    \item[(v)] The Spin(8) invariant in
    \(\boldsymbol8_v\otimes\boldsymbol8_v\) is symmetric, so it contracts
    with the flavor-symmetric singlet \(M^{ij}\).  Likewise two spinors
    \(t\) admit a singlet bilinear.  Both terms have \(R\)-charge two, and
    the symmetry-allowed superpotential is
    \[
      \boxed{W=y\,M^{ij}q_iq_j+\kappa\,Utt,}
      \tag{29.15.S5}
    \]
    with gauge contractions implicit.  The constants \(y\) and \(\kappa\)
    may be changed by field normalizations.

    \item[(vi)] The magnetic theory contains \(p\) vector multiplets
    \(q_i\) in \(\boldsymbol8_v\) and one \(t\) in
    \(\boldsymbol8_s\).  Therefore
    \[
      b_0^{\rm mag}
       =\frac32(6)-\frac12(p+1)
       =\frac{13-N}{2}.
      \tag{29.15.S6}
    \]
    In the phase-diagram convention of (29.15.4), the infrared-free range
    begins at
    \[
      \boxed{N\geq13.}
    \]
    For \(N>13\), the one-loop coefficient is strictly negative; at
    \(N=13\), it vanishes and the magnetic theory lies at the
    marginally-free boundary.  In this range, duality says
    that the strongly coupled infrared limit of the asymptotically free
    electric \(SU(N)\) theory is equivalently described by weakly coupled
    Spin(8) gauge fields, the magnetic quarks, and the singlets in
    (29.15.2), with their mapped interactions (29.15.S5).

    \item[(vii)] The electric \(\widetilde\Phi^i\) is a flavor
    fundamental, with one copy for each of the \(N\) gauge-color
    components.  Thus
    \[
      \mathcal A_{SU(p)^3}^{\rm el}=N.
    \]
    In the magnetic theory, the eight Spin(8) components of \(q_i\) are
    flavor antifundamentals and contribute \(-8\).  The fermion in
    \(M^{ij}\) is in \(\mathrm{Sym}^2\boldsymbol p\) and contributes
    \(p+4\); \(t\) and \(U\) are flavor singlets.  Hence
    \[
      \boxed{\mathcal A_{SU(p)^3}^{\rm mag}
        =-8+(p+4)=p-4=N
        =\mathcal A_{SU(p)^3}^{\rm el}.}
      \tag{29.15.S7}
    \]
  \end{enumerate}

\end{enumerate}
