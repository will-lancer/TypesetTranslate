\chapterbackmatter{Weinberg Exercises}

\begin{enumerate}
  \WeinbergExercise{1} What is the Witten index of the model in Problem 4 of Chapter 26
  for $a\ne 0$? For $a=0$? Can supersymmetry be broken by higher-order
  effects in this model for $a=0$? Explain.

  \WeinbergExercise{2} Consider the renormalizable supersymmetric theory with
  $SO(N_c)$ gauge symmetry and $N_f$ left-chiral scalar superfields
  $\Phi_n$ in the $N$-vector representation. What can you say about the
  structure of the non-perturbative Wilsonian Lagrangian density when
  the bare superpotential vanishes? What if there is a bare
  superpotential $\sum_n\Phi_n\Phi_n$?

  \WeinbergExercise{3} In the theory with Lagrangian density \eqref{eq:29.5.22}, what
  is the relation between the component fields of the spinor field
  strength superfield $W_\alpha$ and its dual
  $\widetilde W_\alpha$?
\end{enumerate}
