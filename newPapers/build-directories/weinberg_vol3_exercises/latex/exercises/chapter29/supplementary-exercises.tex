  \SupplementaryExercise{1}{Supersymmetric Quantum Mechanics and Normalizable Ground States}
    {Adapted from Cambridge Part III, \emph{Supersymmetry}, 2021
    Paper 307, Question 1.}
    {cambridge-part-iii-susy-2021-p307-q1-v3ch29-s01}
  Consider the action
  \[
    S[x,\psi]=\int dt\left[
      \frac12\dot x^2+i\bar\psi\dot\psi
      -\frac12\bigl(h'(x)\bigr)^2
      +\bar\psi\psi\,h''(x)\right],
    \tag{29.1.1}
  \]
  where \(x(t)\) is real, \(\psi(t)\) is a complex Grassmann variable
  with conjugate \(\bar\psi(t)\), and \(h\) is a smooth real function.
  \begin{enumerate}
    \item Show, up to boundary terms, that the action is invariant under
    complex conjugation and under
    \[
      \delta x=\epsilon\bar\psi-\bar\epsilon\psi,
      \qquad
      \delta\psi=\epsilon(i\dot x-h'),
      \qquad
      \delta\bar\psi=-\bar\epsilon(i\dot x+h').
      \tag{29.1.2}
    \]
    Find the Noether charges \(Q\) and \(\bar Q\) associated with
    \(\bar\epsilon\) and \(\epsilon\), respectively.

    \item Quantize the system.  State the canonical (anti)commutation
    relations, explain why its Hilbert space can be represented by
    square-integrable complex differential forms on \(\mathbb R\), and
    identify the actions of \(Q\) and \(\bar Q\) in that representation.

    \item Find the unique normalizable zero-energy state when
    \(h(x)=-x^4\).  Its overall normalization need not be evaluated, but
    must be shown to be finite.  Finally, show that a generic cubic
    polynomial \(h(x)\) admits no normalizable zero-energy state.
  \end{enumerate}

  \SupplementaryExercise{3}{Two-Dimensional Gauge Superspace and the Axial Anomaly}
    {Adapted from Cambridge Part III, \emph{Supersymmetry}, 2021
    Paper 307, Question 2.}
    {cambridge-part-iii-susy-2021-p307-q2-v3ch29-s03}
  Consider a two-dimensional \(N=(2,2)\) supersymmetric \(U(1)\) gauge
  theory.  Let
  \(\Phi^i(x^\pm,\theta^\pm,\bar\theta^\pm)\) be a chiral superfield of
  charge \(q_i\), and let
  \(V(x^\pm,\theta^\pm,\bar\theta^\pm)\) be the vector superfield
  containing the photon \(A_\mu(x^\pm)\).
  \begin{enumerate}
    \item Write the gauge transformations of \(\Phi^i\),
    \(\bar\Phi^{\bar i}\), and \(V\), and explain Wess--Zumino gauge.
    Define the field-strength superfield \(\Sigma\), then prove that it is
    both gauge invariant and twisted chiral.

    \item Determine the axial charge of \(\Sigma\) required for the
    Fayet--Iliopoulos term
    \[
      \frac t2\int d^2x\,d\theta^+d\bar\theta^-\,\Sigma
      \tag{29.3.1}
    \]
    to be invariant under axial transformations.  You may use
    \(\{\bar D_+,D_-\}=0\) and
    \(\bar D_+^2=D_-^2=0\).

    \item Put the theory on a world-sheet torus and take \(V\) to be a
    nondynamical background field, with action
    \[
      S[\Phi^i,V]
       =\int d^2x\,d^4\theta\sum_i
          \bar\Phi^{\bar i}e^{q_iV}\Phi^i
       -\left(
          \frac t2\int d^2x\,d\theta^+d\bar\theta^-\,\Sigma
          +\mathrm{c.c.}\right).
      \tag{29.3.2}
    \]
    Explain why the axial \(U(1)\) symmetry survives quantization only if
    \[
      \sum_iq_i=0.
      \tag{29.3.3}
    \]
  \end{enumerate}

  \SupplementaryExercise{6}{\(SU(2)\) SQCD: Runaways, Quantum Moduli, and Anomaly Matching}
    {Adapted from Cambridge Part III, \emph{Supersymmetry and Duality},
    2023 Paper 307, Question 3.}
    {cambridge-part-iii-susy-2023-p307-q3-v3ch29-s06}
  Consider \(SU(2)\) supersymmetric gauge theory with \(N_f\) chiral
  multiplets \(\Phi_i\) in the fundamental and \(N_f\) chiral
  multiplets \(\widetilde\Phi_i\) in the antifundamental,
  \(i=1,\ldots,N_f\).  Its classical Abelian symmetries are
  \[
  \begin{array}{c|ccc}
       &U(1)_B&U(1)_A&U(1)_R\\ \hline
    \Phi&1&1&1-2/N_f\\
    \widetilde\Phi&-1&1&1-2/N_f .
  \end{array}
  \tag{29.6.1}
  \]
  Here \(U(1)_R\) is an \(R\)-symmetry, the one-loop coefficient is
  \(b_0=6-N_f\), and \(\Lambda\) denotes the holomorphic strong-coupling
  scale.
  \begin{enumerate}
    \item Determine how \(\Lambda^{b_0}\) transforms under each of the
    three \(U(1)\) symmetries in (29.6.1).  You may use
    \(I(\mathrm{adj})=4\) for \(SU(2)\).

    \item For \(N_f=1\), list the gauge-invariant chiral fields and their
    constraints.  Determine the dynamically generated superpotential
    allowed by all the symmetries, and describe the resulting
    supersymmetric ground-state structure.

    \item For \(N_f=2\), list the gauge-invariant chiral fields and derive
    their classical constraint.  Decide whether a superpotential can be
    generated consistently with all symmetries, and determine the allowed
    quantum deformation of the classical constraint.

    \item For \(N_f=3\), list the gauge-invariant chiral fields.  Show that
    unconstrained mesons and baryons at the origin of moduli space match
    both the \(U(1)_R\) and \(U(1)_R^3\) 't~Hooft anomalies of the
    microscopic theory.
  \end{enumerate}

  \SupplementaryExercise{7}{Special Geometry and Abelian Dualization}
    {(Inspired by Adel Bilal, \emph{Introduction to Supersymmetry}, Sections 8.2 and 9.1; independently formulated.)}
    {ch29-s07-special-geometry-and-abelian-dualization}
  \begin{enumerate}
    \item \emph{Metric from the \(N=2\) Holomorphic Function.}
    Given \(T(a)=h'(a)/(4\pi i)\) and
      \(K=\operatorname{Im}[a^*h(a)]/(4\pi)\), derive the scalar kinetic
      coefficient.  State the local positivity condition and explain why a
      logarithmic perturbative \(h(a)\) cannot remain valid near the origin.
    \item \emph{First-Order Abelian Dualization.}
    Treat \(W_L\) as unconstrained and impose its supersymmetric Bianchi
      identity with a real multiplier superfield.  Integrate out \(W_L\) and
      show that the dual scalar coordinate and coupling satisfy
      \(a_D=h(a)\) and \(T_D=1/(16\pi^2T)\).
  \end{enumerate}

  \SupplementaryExercise{8}{Integral Duality and the Dyon Lattice}
    {(Inspired by Adel Bilal, \emph{Introduction to Supersymmetry}, Sections 9.1--9.2; independently formulated.)}
    {ch29-s08-integral-duality-and-the-dyon-lattice}
  \begin{enumerate}
    \item \emph{Generators of the Integral Duality Group.}
    Represent electric--magnetic interchange and an integral theta shift
      as matrices acting on \((a,a_D)^{\mathsf T}\).  Show that their

      induced fractional-linear transformation of \(h'(a)\).
    \item \emph{Theta Shift and the Dyon Lattice.}
    Use \(a_D\to a_D+ba\), with integral \(b\), to find the corresponding

  \end{enumerate}

  \SupplementaryExercise{9}{Seiberg--Witten Monodromies and BPS Covariance}
    {(Inspired by Adel Bilal, \emph{Introduction to Supersymmetry}, Sections 9.2--9.4; independently formulated.)}
    {ch29-s09-seiberg-witten-monodromies-and-bps-covariance}
  \begin{enumerate}
    \item \emph{Weak-Coupling Monodromy at Infinity.}
    Starting from the one-loop asymptotic forms
      \(a\simeq\sqrt{2u}\) and
      \(a_D\simeq i\sqrt{2u}[\ln(2u/\Lambda^2)-2]/\pi\), carry \(u\)
      counterclockwise around infinity and derive its monodromy matrix.
    \item \emph{Monodromy of a Massless Monopole.}
    Near a point where a unit monopole becomes massless, take
      \(a_D\sim c(u-u_0)\) and
      \(a\sim a_0+(i/\pi)a_D\ln(a_D/\Lambda_0)\).
      Determine the monodromy acting on \((a,a_D)^{\mathsf T}\).
    \item \emph{Factorization of Seiberg--Witten Monodromies.}
    Multiply the proposed monodromies at \(u=+1\) and \(u=-1\),
      \[
        M_+=\begin{pmatrix}1&-2\\0&1\end{pmatrix},
        \qquad
        M_-=\begin{pmatrix}3&-2\\2&-1\end{pmatrix}.
      \]
      Compare their ordered product with the weak-coupling monodromy and
      identify the massless charge vector at each singularity.
    \item \emph{Duality Covariance of the BPS Spectrum.}
    For \(Z=2\sqrt2\,i(na+ma_D)\), determine how the integer charge row
      \((n,m)\) must transform when
      \((a,a_D)^{\mathsf T}\to M(a,a_D)^{\mathsf T}\) with
      \(M\in SL(2,\mathbb Z)\).  Prove that the BPS mass is invariant.
  \end{enumerate}

  \SupplementaryExercise{10}{\(Sp(N_c)\) Conformal Window and Seiberg Duality}
    {Adapted from Cambridge Part III, \emph{Supersymmetry}, 2022
    Paper 307, Question 4.}
    {cambridge-part-iii-susy-2022-p307-q4-v3ch29-s10}
  Consider \(Sp(N_c)\) supersymmetric gauge theory minimally coupled to
  \(2N_f\) chiral multiplets \(\Phi^i\) in the fundamental
  representation, \(i=1,\ldots,2N_f\).  Its classical symmetries are
  \[
  \begin{array}{c|cccc}
       &Sp(N_c)&SU(2N_f)&U(1)_A&U(1)_{R_0}\\ \hline
    \Phi&\boldsymbol{2N_c}&\boldsymbol{2N_f}&1&0 .
  \end{array}
  \tag{29.10.1}
  \]
  \begin{enumerate}
    \item[(i)]
    Show that there is a nonanomalous \(R\)-symmetry under which the gluinos
    have charge \(+1\) and
    \[
      R[\Phi]=\frac{N_f-N_c-1}{N_f}.
      \tag{29.10.2}
    \]

    \item[(ii)]
    For what value of \(N_f\) does the theory cease to be asymptotically
    free?

    \item[(iii)]
    Use the relation between \(R\)-charge and scaling dimension to show that
    the meson \(M^{ij}=\Phi^i\Phi^j\) has dimension \(2\) at this point.

    \item[(iv)]
    For what value of \(N_f\) does \(M^{ij}\) have the dimension of a free
    scalar?  Hence determine the expected range of the conformal window.

    It is conjectured that the theory has a dual description with gauge
    group \(Sp(N_f-N_c-2)\), \(2N_f\) dual chiral multiplets \(q_i\), and
    gauge-singlet chiral multiplets \(M^{ij}=-M^{ji}\).  They interact
    through \(W\sim q_iM^{ij}q_j\), and their classical symmetries are
    \[
    \begin{array}{c|cccc}
       &Sp(N_f-N_c-2)&SU(2N_f)&U(1)_A&U(1)_{R_0}\\ \hline
      q&\boldsymbol{2(N_f-N_c-2)}
        &\overline{\boldsymbol{2N_f}}&1&0\\
      M&\boldsymbol1&\wedge^2\boldsymbol{2N_f}&-2&2 .
    \end{array}
    \tag{29.10.3}
    \]

    \item[(v)]
    Determine the nonanomalous \(R\)-charge of \(q\).

    \item[(vi)]
    Determine the \(R\)-charge and scaling dimension of \(M\) at the fixed
    point, and show that they coincide with those of the electric meson.

    \item[(vii)]
    For what values of \(N_f\) is the magnetic theory no longer
    asymptotically free?  Explain the relevance of this result for the
    duality.

    \item[(viii)]
    Compute the \(SU(2N_f)^3\) 't~Hooft anomaly in both the electric and
    magnetic theories.
  \end{enumerate}

  You may use the one-loop coefficient
  \[
    b_0=\frac32 I(\mathrm{adj})
      -\frac12\sum_{\text{chiral multiplets}} I(R),
    \tag{29.10.4}
  \]
  together with
  \[
  \begin{gathered}
    \dim(\boldsymbol{2N})=2N,\qquad I(\boldsymbol{2N})=1,\\
    \dim(\mathrm{adj}_{Sp(N)})=N(2N+1),\qquad
    I(\mathrm{adj}_{Sp(N)})=2(N+1),\\
    \dim(\wedge^2\boldsymbol N)=\frac12N(N-1),\qquad
    I(\wedge^2\boldsymbol N)=N-2,\qquad
    A(\wedge^2\boldsymbol N)=N-4,
  \end{gathered}
  \tag{29.10.5}
  \]
  where the last line refers to \(SU(N)\).

  \SupplementaryExercise{11}{Chiral Gauge Theory and 't Hooft Anomaly Matching}
    {Adapted from Cambridge Part III, \emph{Supersymmetry}, 2022
    Paper 307, Question 3.}
    {cambridge-part-iii-susy-2022-p307-q3-v3ch29-s11}
  \begin{enumerate}
    \item[(a)]
    Briefly describe the implications when a symmetry has, respectively, a
    gauge anomaly, a chiral or Adler--Bell--Jackiw anomaly, and a 't~Hooft
    anomaly.

    Now consider a nonsupersymmetric \(SU(N)\) gauge theory coupled to one
    left-handed Weyl fermion \(\lambda\) in the two-index antisymmetric
    representation and \(p\) left-handed Weyl fermions \(\psi^i\),
    \(i=1,\ldots,p\), each in the gauge antifundamental.

    \item[(i)]
    For what value of \(p\) is the quantum gauge theory consistent?

    \item[(ii)]
    Write down the classical global symmetries.  Show that the quantum
    theory has an \(SU(p)\times U(1)\) global symmetry.

    \item[(iii)]
    Compute its \(SU(p)^3\), \(SU(p)^2U(1)\), \(U(1)^3\), and mixed
    \(U(1)\)-gravitational 't~Hooft anomalies.

    It is conjectured that this theory confines without spontaneously
    breaking a global symmetry.  The proposed massless fields are the
    gauge-singlet fermions
    \[
      \chi^{ij}=\psi^{(i}(\lambda\psi^{j)}),
      \qquad i,j=1,\ldots,p,
      \tag{29.11.1}
    \]
    which transform in the symmetric representation of the \(SU(p)\)
    global symmetry.

    \item[(iv)]
    Show that the 't~Hooft anomalies of \(\chi\) match those of the
    microscopic gauge theory.
  \end{enumerate}

  You may use
  \[
  \begin{array}{c|ccc}
    R&\boldsymbol N&\mathrm{Sym}^2\boldsymbol N&
      \wedge^2\boldsymbol N\\ \hline
    \dim(R)&N&\frac12N(N+1)&\frac12N(N-1)\\
    I(R)&1&N+2&N-2\\
    A(R)&1&N+4&N-4 .
  \end{array}
  \tag{29.11.2}
  \]

  \SupplementaryExercise{12}{Pseudomoduli in Dynamical Supersymmetry Breaking}
    {(Inspired by Poppitz and Trivedi, ``Dynamical Supersymmetry Breaking'', Sections 1--3; independently formulated.)}
    {ch29-s12-pseudomoduli-in-dynamical-supersymmetry-breaking}
  \begin{enumerate}
    \item \emph{Broken Vacuum or Runaway?.}
    For a smooth positive Kahler metric, prove that a finite stationary
      point with no solution of \(f_i=0\) has positive energy and broken

    \item \emph{O'Raifeartaigh Pseudomodulus and Goldstino.}
    Consider
      \(f=f_0X+\tfrac12hX\phi_1^2+m\phi_1\phi_2\), with

      energy, the tree-level pseudomodulus, and the normalized goldstino
      direction.
    \item \emph{Radiative Lifting of a Pseudomodulus.}
    For the O'Raifeartaigh model in the preceding part, write the
      Coleman--Weinberg supertrace formula.  Use the \(X\)-dependent mass
      matrices to determine the parametric size of the induced
      pseudomodulus squared mass and explain why the goldstino remains
      massless.
  \end{enumerate}

  \SupplementaryExercise{14}{Quantum Rank Conditions and the Goldstino Scale}
    {(Inspired by Poppitz and Trivedi, ``Dynamical Supersymmetry Breaking'', Section 4.1; independently formulated.)}
    {ch29-s14-quantum-rank-conditions-and-the-goldstino}
  \begin{enumerate}
    \item \emph{Quantum-Constraint Rank Condition.}
    An \(Sp(N)\) theory with \(2N+2\) fundamentals has antisymmetric mesons
      satisfying \(\operatorname{Pf}M=\Lambda^{2N+2}\).  Couple singlets by

    \item \emph{Current Normalization of the Goldstino Scale.}

      supercurrent by a constant \(F_G\).  Insert one-particle states into

      \(\rho_{\rm VAC}=\lvert F_G\rvert^2/2\) and the relation to chiral and
      gauge auxiliary expectation values.
  \end{enumerate}

  \SupplementaryExercise{15}{A Chiral \(SU(N)\)--\(\operatorname{Spin}(8)\) Seiberg Dual Pair}
    {Adapted from Cambridge Part III, \emph{Supersymmetry and Duality},
    2023 Paper 307, Question 4.}
    {cambridge-part-iii-susy-2023-p307-q4-v3ch29-s15}
  Briefly explain what it means for two quantum field theories to be
  Seiberg dual.  Then consider \(SU(N)\) supersymmetric gauge theory with
  one chiral multiplet \(S\) in \(\mathrm{Sym}^2\boldsymbol N\) and
  \(p\) chiral multiplets \(\widetilde\Phi^i\) in the gauge
  antifundamental, transforming as a fundamental of a global \(SU(p)\).
  \begin{enumerate}
    \item[(i)] For what value of \(p\) is the quantum gauge theory
    consistent?

    \item[(ii)] Show that the theory has a nonanomalous \(R\)-symmetry with
    \[
      R[S]=\frac{2-N}{2+N},
      \qquad R[\widetilde\Phi]=1.
      \tag{29.15.1}
    \]

    \item[(iii)] Show that the electric theory is asymptotically free for
    every nontrivial \(SU(N)\).

    It is conjectured that the theory has a dual description with gauge
    group \(\operatorname{Spin}(8)\) and chiral multiplets
    \[
    \begin{array}{c|ccc}
       &\operatorname{Spin}(8)&SU(p)&U(1)_R\\ \hline
      q&\boldsymbol8_v&\overline{\boldsymbol p}&1\\
      t&\boldsymbol8_s&\boldsymbol1&-5\\
      M&\boldsymbol1&\mathrm{Sym}^2\boldsymbol p&0\\
      U&\boldsymbol1&\boldsymbol1&12 .
    \end{array}
    \tag{29.15.2}
    \]

    \item[(iv)] Show that this \(R\)-symmetry is nonanomalous in the
    magnetic quantum gauge theory.

    \item[(v)] Write the superpotential allowed by the symmetries.

    \item[(vi)] For what values of \(N\) is the magnetic theory infrared
    free?  Explain what Seiberg duality then implies for the low-energy
    physics of the electric \(SU(N)\) theory.

    \item[(vii)] Compute and match the \(SU(p)^3\) 't~Hooft anomaly in the
    electric and magnetic descriptions.
  \end{enumerate}

  You may use
  \[
  \begin{array}{c|ccc}
    SU(N)\text{ irrep}&\boldsymbol N&\mathrm{adj}&
       \mathrm{Sym}^2\boldsymbol N\\ \hline
    \dim(R)&N&N^2-1&\frac12N(N+1)\\
    I(R)&1&2N&N+2\\
    A(R)&1&0&N+4
  \end{array}
  \qquad
  \begin{array}{c|ccc}
    \operatorname{Spin}(8)\text{ irrep}&\boldsymbol8_v&\boldsymbol8_s&
       \mathrm{adj}\\ \hline
    \dim(R)&8&8&28\\
    I(R)&1&1&6\\
    A(R)&0&0&0,
  \end{array}
  \tag{29.15.3}
  \]
  together with \(I(\overline R)=I(R)\),
  \(A(\overline R)=-A(R)\), and
  \[
    b_0=\frac32I(\mathrm{adj})
      -\frac12\sum_{\text{chiral multiplets}}I(R).
    \tag{29.15.4}
  \]

\end{enumerate}
