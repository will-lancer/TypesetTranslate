\chapterbackmatter{Supplementary Exercises}

\begin{enumerate}
  \SupplementaryExercise{1}{Witten Index and Vacua at Infinity}
    {(Adapted from David Tong, \emph{Supersymmetric Field Theory and Example Sheets 2--4}, Example Sheet 2, Question 1; Chapter 3, Section 3.4.)}
    {ch29-s01-witten-index-and-vacua-at-infinity}
  \begin{enumerate}
    \item \emph{Cancellation Behind the Witten Index.}
    Let a Hermitian supercharge obey \(Q^2=H\) and anticommute with
      \((-1)^F\).  Prove that every positive-energy eigenspace has equal
      bosonic and fermionic dimensions, and determine which states can
      contribute to \(\operatorname{Tr}(-1)^F e^{-\beta H}\).
    \item \emph{A Vacuum Escaping to Infinity.}
    For one chiral superfield with
      \(f_\epsilon(X)=fX+\epsilon X^2/2\), find all supersymmetric vacua for
      \(\epsilon\ne0\), follow them as \(\epsilon\to0\), and explain precisely
      why index invariance does not forbid the limiting theory from breaking
      supersymmetry.
  \end{enumerate}

  \SupplementaryExercise{2}{The Goldstino Direction}
    {(Adapted from David Tong, \emph{Supersymmetric Field Theory and Example Sheets 2--4}, Example Sheet 2, Question 3.)}
    {ch29-s02-the-goldstino-direction}
In a general theory with chiral auxiliary expectation values
  \(\mathcal F_{n0}\) and gauge auxiliary expectation values \(D_{A0}\),
  construct the normalized massless fermion direction and express its
  normalization in terms of the vacuum energy density.

  \SupplementaryExercise{3}{Holomorphic Spurions}
    {(Adapted from David Tong, \emph{Supersymmetric Field Theory and Example Sheets 2--4}, Example Sheet 3, Question 1.)}
    {ch29-s03-holomorphic-spurions}
Promote the coefficients in
  \(f(\Phi)=\mu_2\Phi^2+\mu_3\Phi^3\) to background chiral superfields.
  Assign spurionic phase and \(R\) charges, and use holomorphy plus the
  weak-coupling limit to rule out perturbative corrections to the
  Wilsonian superpotential.

  \SupplementaryExercise{4}{The Supersymmetric One-Loop Gauge Coefficient}
    {(Adapted from David Tong, \emph{Supersymmetric Field Theory and Example Sheets 2--4}, Example Sheet 3, Question 4.)}
    {ch29-s04-the-supersymmetric-one-loop-gauge-coefficient}
Sum the gauge-boson, gaugino, Weyl-fermion, and complex-scalar
  contributions to derive \(b_0=3C_1-C_2\) for an \(N=1\)
  supersymmetric gauge theory.  Apply the result to \(SU(N_c)\) SQCD
  with \(N_f\) flavor pairs.

  \SupplementaryExercise{5}{The Instanton Weight as a Chiral Parameter}
    {(Adapted from David Tong, \emph{Supersymmetric Field Theory and Example Sheets 2--4}, Example Sheet 3, Question 5.)}
    {ch29-s05-the-instanton-weight-as-a-chiral}
Starting from the Euclidean instanton action and topological charge,
  show that a winding-\(\nu\) contribution carries
  \(\exp(2\pi i\nu\tau)\).  Explain why holomorphic Wilsonian
  \(\mathcal F\)-terms contain positive rather than negative powers of
  this quantity.

  \SupplementaryExercise{6}{Mass-Deformed ADS Vacua}
    {(Adapted from David Tong, \emph{Supersymmetric Field Theory and Example Sheets 2--4}, Example Sheet 4, Question 1.)}
    {ch29-s06-mass-deformed-ads-vacua}
For \(SU(N_c)\) SQCD with \(N_f<N_c\), extremize
  \[
    f_{\rm eff}=(N_c-N_f)
    \left(\frac{\Lambda^{3N_c-N_f}}{\det M}\right)^{1/(N_c-N_f)}
    +\operatorname{Tr}(mM).
  \]
  Find \(M\), count the supersymmetric vacua, and derive the scale
  matching relation after all flavors are integrated out.

  \SupplementaryExercise{7}{Special Geometry and Abelian Dualization}
    {(Inspired by Adel Bilal, \emph{Introduction to Supersymmetry}, Sections 8.2 and 9.1; independently formulated.)}
    {ch29-s07-special-geometry-and-abelian-dualization}
  \begin{enumerate}
    \item \emph{Metric from the \(N=2\) Holomorphic Function.}
    Given \(T(a)=h'(a)/(4\pi i)\) and
      \(K=\operatorname{Im}[a^*h(a)]/(4\pi)\), derive the scalar kinetic
      coefficient.  State the local positivity condition and explain why a
      logarithmic perturbative \(h(a)\) cannot remain valid near the origin.
    \item \emph{First-Order Abelian Dualization.}
    Treat \(W_L\) as unconstrained and impose its supersymmetric Bianchi
      identity with a real multiplier superfield.  Integrate out \(W_L\) and
      show that the dual scalar coordinate and coupling satisfy
      \(a_D=h(a)\) and \(T_D=1/(16\pi^2T)\).
  \end{enumerate}

  \SupplementaryExercise{8}{Integral Duality and the Dyon Lattice}
    {(Inspired by Adel Bilal, \emph{Introduction to Supersymmetry}, Sections 9.1--9.2; independently formulated.)}
    {ch29-s08-integral-duality-and-the-dyon-lattice}
  \begin{enumerate}
    \item \emph{Generators of the Integral Duality Group.}
    Represent electric--magnetic interchange and an integral theta shift
      as matrices acting on \((a,a_D)^{\mathsf T}\).  Show that their
      products have integral entries and unit determinant, and determine the
      induced fractional-linear transformation of \(h'(a)\).
    \item \emph{Theta Shift and the Dyon Lattice.}
    Use \(a_D\to a_D+ba\), with integral \(b\), to find the corresponding
      relabeling of electric and magnetic quantum numbers.  Verify that the
      BPS central charge and mass are unchanged and recover the Witten shift
      of a monopole's electric charge.
  \end{enumerate}

  \SupplementaryExercise{9}{Seiberg--Witten Monodromies and BPS Covariance}
    {(Inspired by Adel Bilal, \emph{Introduction to Supersymmetry}, Sections 9.2--9.4; independently formulated.)}
    {ch29-s09-seiberg-witten-monodromies-and-bps-covariance}
  \begin{enumerate}
    \item \emph{Weak-Coupling Monodromy at Infinity.}
    Starting from the one-loop asymptotic forms
      \(a\simeq\sqrt{2u}\) and
      \(a_D\simeq i\sqrt{2u}[\ln(2u/\Lambda^2)-2]/\pi\), carry \(u\)
      counterclockwise around infinity and derive its monodromy matrix.
    \item \emph{Monodromy of a Massless Monopole.}
    Near a point where a unit monopole becomes massless, take
      \(a_D\sim c(u-u_0)\) and
      \(a\sim a_0+(i/\pi)a_D\ln(a_D/\Lambda_0)\).
      Determine the monodromy acting on \((a,a_D)^{\mathsf T}\).
    \item \emph{Factorization of Seiberg--Witten Monodromies.}
    Multiply the proposed monodromies at \(u=+1\) and \(u=-1\),
      \[
        M_+=\begin{pmatrix}1&-2\\0&1\end{pmatrix},
        \qquad
        M_-=\begin{pmatrix}3&-2\\2&-1\end{pmatrix}.
      \]
      Compare their ordered product with the weak-coupling monodromy and
      identify the massless charge vector at each singularity.
    \item \emph{Duality Covariance of the BPS Spectrum.}
    For \(Z=2\sqrt2\,i(na+ma_D)\), determine how the integer charge row
      \((n,m)\) must transform when
      \((a,a_D)^{\mathsf T}\to M(a,a_D)^{\mathsf T}\) with
      \(M\in SL(2,\mathbb Z)\).  Prove that the BPS mass is invariant.
  \end{enumerate}

  \SupplementaryExercise{10}{\(Sp(N_c)\) Conformal Window and Seiberg Duality}
    {(Adapted from University of Cambridge, 2022 Part III Paper 307, \emph{Supersymmetry}, Question 4.)}
    {ch29-s10-sp-n-c-conformal-window-and}
Consider four-dimensional \(N=1\) gauge theory with gauge group
  \(Sp(N_c)\) and \(2N_f\) fundamental chiral multiplets
  \(\Phi^i\).
  \begin{enumerate}
    \item Determine the anomaly-free \(R\)-charge of \(\Phi\).
    From the one-loop coefficient, find where the electric theory
    ceases to be asymptotically free.
    \item The meson is \(M^{ij}=\Phi^i\Phi^j\).  Use the
    superconformal relation between \(R\)-charge and dimension to find
    the upper and lower edges of the conformal window.
    \item Analyze the proposed magnetic theory with gauge group
    \(Sp(N_f-N_c-2)\), \(2N_f\) dual quarks \(q_i\), singlets
    \(M^{ij}\), and \(W=M^{ij}q_iq_j\).  Determine \(R[q]\),
    \(R[M]\), and the magnetic asymptotic-freedom condition.
    \item Check the \(SU(2N_f)^3\) anomaly on both sides.  Use
    \(A(\boldsymbol{2N_f})=1\) and
    \(A(\text{antisymmetric})=2N_f-4\).
  \end{enumerate}

  \SupplementaryExercise{11}{Chiral Gauge Theory and 't Hooft Anomaly Matching}
    {(Adapted from University of Cambridge, 2022 Part III Paper 307, \emph{Supersymmetry}, Question 3.)}
    {ch29-s11-chiral-gauge-theory-and-t-hooft}
An \(SU(N)\) gauge theory contains one left-handed Weyl fermion
  \(\lambda\) in the two-index antisymmetric representation and
  \(p\) left-handed fermions \(\psi_i\) in the gauge
  antifundamental.
  \begin{enumerate}
    \item Cancel the gauge anomaly and determine \(p\).  Find the
    nonanomalous \(U(1)\) flavor symmetry by assigning integer charges
    to \(\lambda\) and \(\psi_i\).
    \item Compute the \(SU(p)^3\), \(SU(p)^2U(1)\), \(U(1)^3\), and
    mixed gravitational-\(U(1)\) anomalies of the ultraviolet theory.
    \item Suppose the theory confines without breaking
    \(SU(p)\times U(1)\), and the only massless composite is
    \(\chi^{ij}=\psi^{(i}\lambda\psi^{j)}\).  Determine its
    representation and charge, and verify all four anomaly-matching
    conditions.
  \end{enumerate}

  \SupplementaryExercise{12}{Pseudomoduli in Dynamical Supersymmetry Breaking}
    {(Inspired by Poppitz and Trivedi, ``Dynamical Supersymmetry Breaking'', Sections 1--3; independently formulated.)}
    {ch29-s12-pseudomoduli-in-dynamical-supersymmetry-breaking}
  \begin{enumerate}
    \item \emph{Broken Vacuum or Runaway?.}
    For a smooth positive Kahler metric, prove that a finite stationary
      point with no solution of \(f_i=0\) has positive energy and broken
      supersymmetry.  Explain why a direction with \(V\to0\) at infinite
      field values instead describes a runaway without a ground state.
    \item \emph{O'Raifeartaigh Pseudomodulus and Goldstino.}
    Consider
      \(f=f_0X+\tfrac12hX\phi_1^2+m\phi_1\phi_2\), with
      \(m^2>\lvert hf_0\rvert\).  Find the classical vacuum manifold, its
      energy, the tree-level pseudomodulus, and the normalized goldstino
      direction.
    \item \emph{Radiative Lifting of a Pseudomodulus.}
    For the O'Raifeartaigh model in the preceding part, write the
      Coleman--Weinberg supertrace formula.  Use the \(X\)-dependent mass
      matrices to determine the parametric size of the induced
      pseudomodulus squared mass and explain why the goldstino remains
      massless.
  \end{enumerate}

  \SupplementaryExercise{13}{\(R\)-Symmetry and the \(3\)-\(2\) Rank Obstruction}
    {(Inspired by Poppitz and Trivedi, ``Dynamical Supersymmetry Breaking'', Sections 3--4.1; independently formulated.)}
    {ch29-s13-symmetry-and-the-rank-obstruction}
  \begin{enumerate}
    \item \emph{A Generic \(R\)-Symmetry Counting Test.}
    In a generic \(R\)-symmetric Wess--Zumino model, let \(N_X\) fields
      have \(R=2\) and \(N_Y\) fields have \(R=0\).  Analyze the
      \(\mathcal F\)-term equations at an \(R\)-symmetric point and state the
      condition under which supersymmetry is generically broken.
    \item \emph{The \(3\)-\(2\) Model Rank Obstruction.}
    In the \(SU(3)\times SU(2)\) model with
      \(f_{\rm tree}=\lambda Q\bar D L\), combine the \(SU(3)\) ADS term with
      the \(L\)-field equation.  Show that no finite configuration is both
      \(\mathcal F\)-flat and nonsingular, and explain the role of
      \(D\)-terms in stabilizing the runaway.
  \end{enumerate}

  \SupplementaryExercise{14}{Quantum Rank Conditions and the Goldstino Scale}
    {(Inspired by Poppitz and Trivedi, ``Dynamical Supersymmetry Breaking'', Section 4.1; independently formulated.)}
    {ch29-s14-quantum-rank-conditions-and-the-goldstino}
  \begin{enumerate}
    \item \emph{Quantum-Constraint Rank Condition.}
    An \(Sp(N)\) theory with \(2N+2\) fundamentals has antisymmetric mesons
      satisfying \(\operatorname{Pf}M=\Lambda^{2N+2}\).  Couple singlets by
      \(f=\lambda S^{ij}M_{ij}\).  Prove that the singlet
      \(\mathcal F\)-terms are incompatible with the quantum constraint and
      identify the symmetry reason for the resulting goldstino.
    \item \emph{Current Normalization of the Goldstino Scale.}
    Parameterize the vacuum-to-goldstino matrix element of the conserved
      supercurrent by a constant \(F_G\).  Insert one-particle states into
      the supersymmetry algebra and derive both
      \(\rho_{\rm VAC}=\lvert F_G\rvert^2/2\) and the relation to chiral and
      gauge auxiliary expectation values.
  \end{enumerate}

\end{enumerate}
