  \SupplementaryExercise{1}{Discrete Symmetries and Dangerous MSSM Operators}
    {(Inspired by Chung et al., ``The Soft Supersymmetry-Breaking Lagrangian: Theory and Applications'', Sections 2.3 and 10.2; independently formulated.)}
    {ch28-s01-discrete-symmetries-and-dangerous-mssm-operators}
  \begin{enumerate}
    \item \emph{Matter-Parity Test of the Renormalizable Superpotential.}
    Assign matter parity \(-1\) to every quark and lepton chiral superfield
      and \(+1\) to both Higgs superfields.  Classify all gauge-invariant
      renormalizable superpotential monomials and identify which violate
      baryon or lepton number.
    \item \emph{Proton Decay from Simultaneous RPV Couplings.}
    Combine one \(LQ\bar D\) coupling with one
      \(\bar U\bar D\bar D\) coupling and integrate out the appropriate
      right-handed down squark.  Write a representative four-fermion operator
      and derive its coefficient including flavor labels.
    \item \emph{Baryon Triality Check for Dimension-Four Operators.}
    Construct a multiplicative \(\mathbb Z_3\) charge assignment that
      permits the ordinary Yukawa and \(\mu\) terms and the lepton-number
      violating renormalizable monomials, but forbids
      \(\bar U\bar D\bar D\).  Check every monomial explicitly.
  \end{enumerate}

  \SupplementaryExercise{2}{Supersymmetric Gauge Unification}
    {Adapted from Cambridge Part III, \emph{Supersymmetry}, 2024
    Paper 307, Question 3.}
    {cambridge-part-iii-susy-2024-p307-q3-v3ch28-s02}
  \begin{enumerate}
    \item[(a)]
    Describe supersymmetric gauge unification.  Include graphs depicting the
    evolution of the three gauge couplings for both the minimal
    supersymmetric Standard Model and the nonsupersymmetric Standard Model.
    Also draw a Feynman diagram for a contribution to the evolution of
    \(g_2\) from a superpartner and one from its Standard-Model counterpart.
    No explicit loop calculation is required in this part.

    \item[(b)]
    The one-loop evolution of each MSSM gauge coupling \(g_a(\mu)\) with the
    renormalization scale \(\mu\) is
    \[
      \mu\frac{\dd g_a(\mu)}{\dd\mu}
        =\beta_a g_a^3(\mu)
      \qquad\text{(no sum on \(a\))},
      \tag{28.2.1}
    \]
    where the \(\beta_a\) are real numerical constants.  Solve this equation,
    finding \(g_a(\mu)\) in terms of input data \(g_a(\mu_0)\).

    \item[(c)]
    The accurately measured values \(g_1(M_Z)\) and \(g_2(M_Z)\) can be used,
    assuming MSSM gauge unification, to predict \(g_3(M_Z)\).  Determine the
    unification scale \(M_{\rm GUT}\) in terms of
    \(M_Z\), \(g_2(M_Z)\), and \(g_1(M_Z)\).

    \item[(d)]
    Derive the gauge-unification prediction
    \[
      g_3^{-2}(M_Z)=g_2^{-2}(M_Z)
        +A\bigl[g_2^{-2}(M_Z)-g_1^{-2}(M_Z)\bigr],
      \tag{28.2.2}
    \]
    and evaluate \(A\) in terms of the three distinct \(\beta_a\).  Do you
    expect (28.2.2) to be approximately satisfied by measurements?  Give your
    reasoning.
  \end{enumerate}

  \SupplementaryExercise{3}{Right-Handed Neutrinos and the Dimension-Five Operator}
    {(Adapted from John McGreevy, \emph{Physics 215C Quantum Field Theory}, Spring 2025, Assignment 6, Problem 2.)}
    {mcgreevy-215c-s25-hw6-p2-v3ch28-s03}
Add to the standard-model matter content one right-handed neutrino
  \(N_R\), neutral under every gauge factor.  Define
  \(N_R^c=C\bar N_R^{\mathsf T}\), with
  \(C=\ii\gamma^2\gamma^0\) in the Dirac representation, and write
  \[
    X\equiv H_i^{\mathsf T}L_j\epsilon^{ij},
    \qquad \epsilon^{12}=+1.
  \]
  Up to the overall mostly-plus kinetic convention, the complete set of
  operators of dimension at most four involving \(N_R\) is
  \[
    \mathcal L_N
      =-\bar N_R\gamma^\mu\partial_\mu N_R
       -\left[
         \frac{M}{2}
          \bigl(\bar N_R^cN_R+\bar N_RN_R^c\bigr)
         +g_\nu\bar N_RX+\mathrm{H.c.}
       \right].
  \]
  Take \(|M|\) much larger than the electroweak scale.
  \begin{enumerate}
    \item Introduce the Majorana field \(N=N_R+N_R^c\), which satisfies
      \(N=N^c\), and integrate it out at tree level.  Expand the result in
      powers of \(1/M\) and display the leading operator made only from
      standard-model fields, including its coefficient, mass dimension, and
      Hermitian conjugate.
    \item Give the Higgs a vacuum expectation value
      \(\langle H\rangle=(0,v/\sqrt2)^{\mathsf T}\).  Determine the induced
      light-neutrino mass and explain why it is Majorana and violates lepton
      number by two units.
    \item Using \(m_\nu<0.5\,\mathrm{eV}\), obtain the corresponding bound on
      \(|M|/|g_\nu|^2\) and evaluate its order of magnitude for
      \(v=246\,\mathrm{GeV}\).
  \end{enumerate}

  \SupplementaryExercise{4}{Soft Flavor Tensors and Mass-Insertion Bounds}
    {(Inspired by Chung et al., ``The Soft Supersymmetry-Breaking Lagrangian: Theory and Applications'', Sections 5.1--5.2; independently formulated.)}
    {ch28-s04-soft-flavor-tensors-and-mass-insertion}
  \begin{enumerate}
    \item \emph{Soft-Term Flavor Counting in a Super-CKM Basis.}
    Diagonalize the quark Yukawa matrices and enumerate the independent
      flavor structures that remain in the left- and right-squark squared-mass
      matrices and trilinear couplings.  Explain why gauge symmetry alone does
      not align them with the quark masses.
    \item \emph{A Neutral-Kaon Mass-Insertion Bound.}
    Define \((\delta^d_{12})_{LL}=(\Delta M_{\widetilde \dd}^2)_{12}/
      \widetilde M^2\).  Use a gluino box and dimensional analysis to show
      that its \(\Delta S=2\) coefficient scales quadratically in this
      insertion, and compare it with the standard-model charm contribution.
    \item \emph{A Slepton-Mixing Bound from Radiative Muon Decay.}
    Insert one off-diagonal slepton squared mass into a neutralino--slepton
      loop for \(\mu\to e\gamma\).  Determine the dependence of the dipole
      coefficient and branching ratio on the insertion, electroweak coupling,
      and common superpartner mass.
  \end{enumerate}

  \SupplementaryExercise{5}{Physical Soft Phases and Dipole Decoupling}
    {(Inspired by Chung et al., ``The Soft Supersymmetry-Breaking Lagrangian: Theory and Applications'', Sections 5.1--5.2; independently formulated.)}
    {ch28-s05-physical-soft-phases-and-dipole-decoupling}
  \begin{enumerate}
    \item \emph{Rephasing-Invariant Soft CP Phases.}
    In a one-generation MSSM with complex \(\mu\), \(B\), a common gaugino
      mass \(M_{1/2}\), and a common trilinear parameter \(A\), use field
      rephasings to construct two independent CP-odd invariant phase
      combinations.
    \item \emph{Parametric Decoupling of a Supersymmetric EDM.}
    Estimate a light-fermion electric dipole moment generated by one gaugino,
      two sfermion propagators, and one left--right mass insertion.  Show
      explicitly how the answer decouples when every soft mass is scaled by a
      common factor.
  \end{enumerate}

  \SupplementaryExercise{6}{A General Two-Higgs-Doublet Sector}
    {(Adapted from David Tong, \emph{The Standard Model Example Sheet 4}, Question 6.)}
    {ch28-s06-two-higgs-superfields-and-electroweak-breaking}
  \begin{enumerate}
    \item \emph{Charges.}
    Besides the standard Higgs \(H\sim(\mathbf2,\mathbf{1})_{+1/2}\), introduce
      \(\phi\sim(\mathbf2,\mathbf{1})_{-1/2}\).  Determine the electric charge
      of each component of \(\phi\).
    \item \emph{General Renormalizable Potential.}
    Write every gauge-invariant Hermitian operator of dimension at most four
      built from \(H\) and \(\phi\).  Count the independent real parameters,
      allowing all coefficients permitted by gauge symmetry to be complex.
    \item \emph{Vacuum and Gauge Spectrum.}
    Suppose
      \[
        \langle H\rangle=\frac1{\sqrt2}\binom0v,\qquad
        \langle\phi\rangle=\frac1{\sqrt2}\binom{u+\ii w}0,
        \qquad u,v,w\in\mathbb R.
      \]
      Decide whether electromagnetism is broken.  Compute the masses of
      \(W^\pm\), \(Z\), and the photon, and identify the electric charges of
      the remaining physical scalar fields.
    \item \emph{Yukawa Couplings.}
    List all gauge-invariant quark and charged-lepton Yukawa couplings to the
      two doublets.  Explain why gauge symmetry alone permits both doublets to
      couple to each fermion species.
  \end{enumerate}

  \SupplementaryExercise{14}{Naturalness, \(R\)-Parity, and Proton Decay}
    {Adapted from Cambridge Part III, \emph{Supersymmetry}, 2025
    Paper 307, Question 2.}
    {cambridge-part-iii-susy-2025-p307-q2-v3ch28-s14}
  \begin{enumerate}
    \item[(a)]
    Describe why theoretical naturalness arguments lead us to expect
    sparticle masses to be of order a few TeV or less.  Include the relevant
    Feynman diagrams and order-of-magnitude estimates.

    \item[(b)]
    Define \(R\)-parity in terms of spin \(s\), baryon number \(B\), and
    lepton number \(L\).  Write the complete renormalizable
    \(R\)-parity-violating superpotential of the MSSM, leaving gauge indices
    suppressed if desired.  State the antisymmetries of its couplings.

    \item[(c)]
    Draw a Feynman diagram showing how one baryon-number-violating term and
    one lepton-number-violating term in this superpotential can induce proton
    decay into a pion and an invisible particle.  Is the invisible particle
    a neutrino or an antineutrino?  From which family can it come, and what is
    the electric charge of the pion?

    \item[(d)]
    By dimensional analysis, estimate the order of magnitude of the decay
    rate through this channel in terms of the two relevant
    \(R\)-parity-violating couplings, taking the exchanged superpartner mass
    to be \(10\,\mathrm{TeV}\).

    \item[(e)]
    The examination supplies the experimental benchmark
    \[
      \tau_{p\to\mu^+\pi^0}>10^{34}\ \mathrm{years}.
      \tag{28.14.1}
    \]
    Use this lifetime scale and dimensional analysis to find an
    order-of-magnitude upper bound on the product of the two couplings in
    part~(d).  The printed benchmark labels a charged-lepton channel whereas
    part~(c) asks for an invisible channel; use (28.14.1) as the requested
    common order-of-magnitude proton-lifetime bound.

    \item[(f)]
    Show that \(R\)-parity forbids every term in the superpotential written
    in part~(b), and now impose \(R\)-parity on the theory.

    \item[(g)]
    The MSSM Lagrangian contains a direct coupling among a nongauge particle,
    its antiparticle, an \(SU(2)_L\) \(W^-\) boson, and a gluon.  Identify the
    fields and the part of the Lagrangian density from which this interaction
    comes.  Draw its Feynman diagram and write its Feynman rule.

    \emph{Hint.}  In natural units, one second is of order
    \(10^{24}\,\mathrm{GeV}^{-1}\), and the proton mass is of order
    \(1\,\mathrm{GeV}\).
  \end{enumerate}

  \SupplementaryExercise{15}{Unitary Gauge for a General Higgs Mechanism}
    {(Adapted from Daniel Harlow, \emph{Quantum Field Theory III}, Spring 2026, Section 10.7,
      Homework Problem 1.)}
    {harlow-qft3-2026-s10p7-hw1-v3ch28-s15}
  Let a compact gauge group \(G\) act on a complex scalar \(\phi\) through
  a unitary representation \(D\), with Hermitian infinitesimal generators
  \(\tau_a\).  Consider
  \[
    \mathcal L=-\frac14\sum_a\frac1{g_a^2}F^a_{\mu\nu}F_a^{\mu\nu}
      -(D_\mu\phi)^\dagger D^\mu\phi-V(\phi),
    \qquad D_\mu\phi=\partial_\mu\phi-\ii\tau_aA^a_\mu\phi,
  \]
  and write \(\langle\phi\rangle=v/\sqrt2\).  The unbroken subgroup is
  \[
    H=\{h\in G\mid D(h)v=v\}.
  \]
  On each gauge orbit choose \(g_\phi(x)\) to maximize
  \[
    f_\phi(g)=\operatorname{Re}\!\left[\phi^\dagger(x)D(g)v\right],
  \]
  and define
  \[
    \widetilde\phi=D(g_\phi^{-1})\phi,
    \qquad
    \widetilde A_\mu=g_\phi^{-1}A_\mu g_\phi
       +\ii g_\phi^{-1}\partial_\mu g_\phi .
  \]
  \begin{enumerate}
    \item Show that the maximizing condition gives
      \(\operatorname{Im}(\widetilde\phi^\dagger\tau_av)=0\).
      Show also that the transformed and original fields have the same
      vacuum expectation value.
    \item Prove that right multiplication of \(g_\phi\) by an element of
      \(H\) gives another maximum.  Decomposing
      \(\mathfrak g=\mathfrak h\oplus\mathfrak h^\perp\), determine how the
      corresponding components of \(\widetilde A_\mu\) transform under the
      residual local \(H\) symmetry.
    \item Expand
      \[
        f_{\widetilde\phi}(e^{\ii\theta^aT_a})
          =\operatorname{Re}\!\left(
             \widetilde\phi^\dagger e^{\ii\theta^a\tau_a}v\right)
      \]
      through second order.  Show that, for \(\widetilde\phi\) sufficiently
      close to \(v\), the maximum is locally unique on \(G/H\): equality
      away from \(\theta^a=0\) can occur only when
      \(\theta^a\tau_av=0\), which is an unbroken direction.
    \item Parametrize unitary gauge by
      \(\widetilde\phi=(v+h)/\sqrt2\), and write the full Lagrangian in
      terms of \(h\) and \(\widetilde A_\mu\), using the condition from
      part (a).  For the canonical fields
      \(\widehat A^a_\mu=\widetilde A^a_\mu/g_a\), derive
      \[
        (\mu^2)_{ab}=\frac{g_ag_b}{2}
          v^\dagger\{\tau_a,\tau_b\}v .
      \]
      Prove that this matrix is real, symmetric, and positive semidefinite,
      and show that \(c^a\) is a null vector precisely when
      \(\sum_a g_ac^aT_a\in\mathfrak h\).
    \item Specialize to \(G=U(1)\) and a scalar of integer charge \(p\).
      Find \(g_\phi\), \(\widetilde\phi\), the residual group, and the
      gauge-boson mass.
    \item Finally take \(G=SU(2)\times U(1)_Y\), with the Higgs in
      \((\boldsymbol2,-\tfrac12)\) and
      \[
        \langle\phi\rangle=\frac1{\sqrt2}\binom v0,
        \qquad v>0.
      \]
      Find its unitary-gauge form and verify, in the basis consisting of the
      three \(SU(2)\) generators followed by hypercharge, that
      \[
        \mu^2=\frac{v^2}{4}
        \begin{pmatrix}
          g_2^2&0&0&0\\
          0&g_2^2&0&0\\
          0&0&g_2^2&-g_1g_2\\
          0&0&-g_1g_2&g_1^2
        \end{pmatrix}.
      \]
      Identify the massive and massless vector eigenstates.
  \end{enumerate}

\end{enumerate}
