\chapterbackmatter{Solutions to Weinberg Exercises}
\ExerciseEditorialNote

\begin{enumerate}
  \WeinbergSolution{1}
  Let \(\lambda'\) denote a representative coefficient of an
  \(LQ\bar D\) or \(LL\bar E\) interaction in
  Eq.~\eqref{eq:28.1.4}, and let \(\lambda''\) denote a representative
  coefficient of the \(\bar U\bar D\bar D\) interaction in
  Eq.~\eqref{eq:28.1.5}.  Exchanging the scalar component common to a
  lepton-number violating vertex and a baryon-number violating vertex
  produces a four-fermion operator with coefficient
  \[
    g_6\sim\frac{\lambda'\lambda''}{m_{\widetilde f}^{\,2}} .
    \tag{1}
  \]
  Color, flavor, and hadronic matrix elements change the answer by
  factors of order unity to a few orders of magnitude, but not by the
  many powers of ten at issue here.

  A dimension-six proton-decay interaction gives
  \[
    \Gamma_p\sim c_p m_p^5|g_6|^2,\qquad
    c_p\sim10^{-3}\text{--}1.
    \tag{2}
  \]
  Using a lifetime bound of order \(10^{33}\)--\(10^{34}\) years gives
  \[
    m_{\widetilde f}
      \gtrsim
      (c_pm_p^5\tau_p)^{1/4}
      |\lambda'\lambda''|^{1/2}
      \sim
      10^{15}\text{--}10^{16}\ {\rm GeV}\,
      |\lambda'\lambda''|^{1/2}.
    \tag{3}
  \]
  Thus order-one coefficients would force the relevant squarks or
  sleptons up near a grand-unification scale, wholly defeating
  weak-scale supersymmetry.  Conversely, for a \(1\ {\rm TeV}\)
  sfermion, the product \(|\lambda'\lambda''|\) must be extraordinarily
  small, roughly \(10^{-24}\)--\(10^{-26}\), with the precise limit
  depending on the flavors and decay channel.  This is the practical
  reason for imposing \(R\) parity or another symmetry that forbids at
  least one of the two classes of interactions.

  \WeinbergSolution{2}
  Put \(s^2\equiv\sin^2\theta\),
  \[
    L=\ln\frac{M}{m},\qquad \ell=\ln\frac{m}{m_Z},
  \]
  and take \(n_g=3\).  Above \(m\), complete supermultiplets contribute,
  and Eqs.~\eqref{eq:28.2.8}--\eqref{eq:28.2.10} give
  \[
  \begin{aligned}
    \mu\frac{\dd g'}{\dd\mu}
      &=\frac{g'^3}{4\pi^2}
        \left(\frac52+\frac{n_s}{8}\right),\\
    \mu\frac{\dd g}{\dd\mu}
      &=\frac{g^3}{4\pi^2}\frac{n_s}{8},\\
    \mu\frac{\dd g_s}{\dd\mu}
      &=-\frac{3g_s^3}{16\pi^2}.
  \end{aligned}
    \tag{4}
  \]
  Below \(m\), the gauginos, higgsinos, squarks, and sleptons are absent.
  Retaining \(n_s\) light complex scalar doublets gives
  \[
  \begin{aligned}
    \mu\frac{\dd g'}{\dd\mu}
      &=\frac{g'^3}{4\pi^2}
        \left(\frac53+\frac{n_s}{24}\right),\\
    \mu\frac{\dd g}{\dd\mu}
      &=\frac{g^3}{4\pi^2}
        \left(-\frac56+\frac{n_s}{24}\right),\\
    \mu\frac{\dd g_s}{\dd\mu}
      &=-\frac{7g_s^3}{16\pi^2}.
  \end{aligned}
    \tag{5}
  \]
  If only one scalar doublet remains below \(m\), \(n_s\) in Eq.~(5)
  is simply replaced by one; the following formulas use the spectrum
  stipulated above.

  Integrating piecewise and imposing
  \(g_s^2(M)=g^2(M)=5g'^2(M)/3\), the two independent differences are
  \[
  \begin{aligned}
    \frac{s^2}{e^2}-\frac1{g_s^2}
      &=\frac1{2\pi^2}
        \left[
          \frac{6+n_s}{8}L
          +\frac{22+n_s}{24}\ell
        \right],\\
    \frac{8s^2-3}{5e^2}
      &=\frac1{2\pi^2}
        \left[
          \left(-\frac32+\frac{n_s}{20}\right)L
          +\left(-\frac{11}{6}+\frac{n_s}{60}\right)\ell
        \right],
  \end{aligned}
    \tag{6}
  \]
  where \(e=e(m_Z)\) and \(g_s=g_s(m_Z)\).  Solving gives
  \[
  \begin{aligned}
    s^2={}&
    \frac{18+3n_s+(60-2n_s)e^2/g_s^2}{108+6n_s}
    -\frac{2n_se^2}{3\pi^2(18+n_s)}
      \ln\frac{m}{m_Z},\\
    \ln\frac{M}{m_Z}={}&
    \frac{8\pi^2}{e^2}
      \frac{1-8e^2/(3g_s^2)}{18+n_s}
    +\frac{2(n_s-6)}{3(18+n_s)}
      \ln\frac{m}{m_Z}.
  \end{aligned}
    \tag{7}
  \]
  For the MSSM value \(n_s=2\), taking
  \(e^2/4\pi\simeq1/128\) and \(g_s^2/4\pi\simeq0.118\) gives
  \[
    s^2\simeq0.2309-6.6\times10^{-4}\ln(m/m_Z),
    \qquad
    M\simeq2.2\times10^{16}{\rm GeV}\,
      (m/m_Z)^{-2/15}.
    \tag{8}
  \]
  The unification-scale constraint is weak because of the small exponent.
  The accurately measured weak angle is decisive: allowing a one-loop
  mismatch of only \(10^{-3}\)--\(2\times10^{-3}\) restricts \(m\) to
  roughly the few-hundred-GeV to few-TeV range.  Given matching and
  two-loop uncertainties, the robust rough conclusion is
  \(m\lesssim O(1\ {\rm TeV})\), rather than an arbitrarily high
  supersymmetry-breaking threshold.

  \WeinbergSolution{3}
  Let the canonically normalized CP-even neutral fields be
  \[
    \phi_1=\sqrt2\,\operatorname{Re}\varphi_1,\qquad
    \phi_2=\sqrt2\,\operatorname{Re}\varphi_2,
  \]
  and define the light and heavy mass eigenstates by
  \[
    h=-\sin\alpha\,\phi_1+\cos\alpha\,\phi_2,\qquad
    H=\cos\alpha\,\phi_1+\sin\alpha\,\phi_2,
    \qquad -\frac{\pi}{2}<\alpha<0.
    \tag{9}
  \]
  Diagonalizing Eq.~\eqref{eq:28.5.27} gives
  \[
  \begin{aligned}
    \Delta&=\sqrt{(m_A^2+m_Z^2)^2
      -4m_A^2m_Z^2\cos^22\beta},\\
    \sin2\alpha&=-\frac{(m_A^2+m_Z^2)\sin2\beta}{\Delta},\\
    \cos2\alpha&=-\frac{(m_A^2-m_Z^2)\cos2\beta}{\Delta},
  \end{aligned}
    \tag{10}
  \]
  equivalently
  \[
    \tan2\alpha=\tan2\beta\,
      \frac{m_A^2+m_Z^2}{m_A^2-m_Z^2}
  \]
  with the range in Eq.~(9) selecting the light eigenstate.

  The up-type masses arise from \(\mathcal H_2^0\), whereas the
  down-type quark and charged-lepton masses arise from
  \(\mathcal H_1^0\).  Since
  \(1/v_{\rm SM}=(\sqrt2G_F)^{1/2}\), the interaction may be written
  \[
    \mathcal L_{hff}
      =-h\left[
        \sum_{u}g_{huu}\bar uu
        +\sum_{d}g_{hdd}\bar \dd d
        +\sum_{\ell}g_{h\ell\ell}\bar\ell\ell
      \right],
    \tag{11}
  \]
  with
  \[
  \begin{aligned}
    g_{huu}
      &=(\sqrt2G_F)^{1/2}m_u\,\frac{\cos\alpha}{\sin\beta},\\
    g_{hdd}
      &=-(\sqrt2G_F)^{1/2}m_d\,\frac{\sin\alpha}{\cos\beta},\\
    g_{h\ell\ell}
      &=-(\sqrt2G_F)^{1/2}m_\ell\,\frac{\sin\alpha}{\cos\beta}.
  \end{aligned}
    \tag{12}
  \]
  These formulas also provide a sign check: for
  \(m_A\gg m_Z\), \(\alpha\to\beta-\pi/2\), so every ratio in
  Eq.~(12) approaches one and the light scalar has the standard-model
  Yukawa couplings.

  \WeinbergSolution{4}
  For each messenger pair let \(T_3(R_n)\) be the Dynkin index of the
  representation of \(\Phi_n\), normalized by
  \(T_3(\boldsymbol 3)=1/2\).  The pair
  \(\Phi_n+\bar\Phi_n\) therefore has total index
  \(T_{3n}^{\rm pair}=2T_3(R_n)\).  Its holomorphic mass is the chiral
  background
  \[
    \mathcal M_n(S_n)=\lambda_nS_n.
  \]
  Below all messenger thresholds, the one-loop Wilsonian gauge kinetic
  function consequently contains
  \[
    N_3(S_n,\mu)
      =N_3^{(0)}(\mu)
       -\frac1{16\pi^2}
        \sum_nT_{3n}^{\rm pair}\ln(\lambda_nS_n).
    \tag{13}
  \]
  Holomorphy is the important step: the real threshold
  \(\ln|\lambda_n\mathcal S_n|\) uniquely continues to the chiral
  function \(\ln(\lambda_nS_n)\).

  Write
  \(S_n=\mathcal S_n+\theta^2\mathcal F_n\).  In the regime
  \(|\mathcal F_n|\ll|\lambda_n||\mathcal S_n|^2\),
  \[
    [\ln(\lambda_nS_n)]_{\mathcal F}
      =\frac{\mathcal F_n}{\mathcal S_n}.
    \tag{14}
  \]
  Canonically normalizing the gluino as in
  Eq.~\eqref{eq:28.6.18} then gives the complex one-loop mass parameter
  \[
    \mathcal M_{\widetilde g}
      =\frac{g_s^2(\mu)}{16\pi^2}
        \sum_n2T_3(R_n)\frac{\mathcal F_n}{\mathcal S_n}
        \left[
          1+O\!\left(
            \frac{|\mathcal F_n|^2}
                 {|\lambda_n|^2|\mathcal S_n|^4}
          \right)
        \right],
    \qquad
    m_{\widetilde g}=|\mathcal M_{\widetilde g}|.
    \tag{15}
  \]
  If \(T_{3n}\) is defined to include both members of the pair, Eq.~(15)
  is instead written with \(T_{3n}\) and no explicit factor two.
  Notice that \(\lambda_n\) cancels: at leading order it only fixes the
  messenger threshold, whereas the supersymmetry-breaking splitting is
  governed by \(\mathcal F_n/\mathcal S_n\).

  \WeinbergSolution{5}
  Let \(g_5\) be a coefficient of one of the dimension-five operators
  in Eq.~\eqref{eq:28.7.3}.  A single gluino-dressing graph would
  nominally produce
  \[
    g_6^{(\widetilde g)}
      \sim\frac{g_s^2g_5}{8\pi^2}
        \frac{m_{\widetilde g}}{\widetilde M^2},
    \tag{16}
  \]
  where \(\widetilde M\) is the larger of the gluino and squark masses.
  For flavor-independent squark masses, however, the sum of graphs
  vanishes by the color/flavor antisymmetry and Fermi-statistics argument
  of Section 28.7.  Expanding the squark propagator about a common squared
  mass leaves one mass insertion:
  \[
    g_6^{(\widetilde g)}
      \sim\frac{g_s^2g_5}{8\pi^2}
        \frac{m_{\widetilde g}}{\widetilde M^2}\,
        \delta_{qq'},\qquad
    \delta_{qq'}\equiv
      \sum_iV_{qi}V_{q'i}^*
      \frac{\Delta M_i^2}{\widetilde M^2}.
    \tag{17}
  \]
  Which \(q,q'\) occurs depends on the flavor tensor of the original
  dimension-five operator.

  The same mass insertion enters the gluino box responsible for
  \(K^0\)-\(\bar K^0\) conversion twice.  Equation
  \eqref{eq:28.4.6} therefore supplies the useful bound
  \[
    |\delta_{\dd s}|
      <1.5\times10^{-3}
        \left(\frac{\widetilde M}{100\ {\rm GeV}}\right).
    \tag{18}
  \]
  Combining Eqs.~(17) and (18),
  \[
    |g_6^{(\widetilde g)}|
      \lesssim
      \frac{g_s^2|g_5|}{8\pi^2}
      \frac{m_{\widetilde g}}{\widetilde M^2}
      1.5\times10^{-3}
      \left(\frac{\widetilde M}{100\ {\rm GeV}}\right).
    \tag{19}
  \]
  For \(m_{\widetilde g}\simeq\widetilde M\) and
  \(g_s^2/4\pi\simeq0.118\), this is approximately
  \[
    |g_6^{(\widetilde g)}|
      \lesssim3\times10^{-7}\ {\rm GeV}^{-1}|g_5|.
    \tag{20}
  \]
  The cancellation has therefore converted the apparently strong
  gluino dressing into a flavor-breaking effect already tightly
  constrained by neutral-kaon mixing.  The estimate is deliberately
  conservative: additional antisymmetry, small mixing angles, or
  near-degeneracy can suppress it further.
\end{enumerate}
