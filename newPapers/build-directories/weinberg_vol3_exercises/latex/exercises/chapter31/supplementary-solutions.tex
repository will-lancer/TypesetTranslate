\begin{enumerate}
  \SupplementarySolution{1}{The Linearized Supergravity Current}
  Denote the Fierz--Pauli part of the Lagrangian by \(\mathcal L_{\rm FP}\)
  and define
  \[
    \Pi^{\kappa\mid\mu\nu}
      \equiv\frac{\partial\mathcal L_{\rm FP}}
                   {\partial(\partial_\kappa h_{\mu\nu})}
      =-\partial^\kappa h^{\mu\nu}
       +2\eta^{\kappa(\mu}\partial_\rho h^{\nu)\rho}
       -\eta^{\kappa(\mu}\partial^{\nu)}h
       -\eta^{\mu\nu}\partial_\rho h^{\rho\kappa}
       +\eta^{\mu\nu}\partial^\kappa h .
  \]
  When \(\epsilon\) is local, the part of the graviton variation containing
  a derivative of the parameter is therefore
  \[
    \delta\mathcal L_{\rm FP}\big|_{\partial\epsilon}
      =-i(\partial_\kappa\bar\epsilon)
        \Pi^{\kappa\mid\mu\nu}\gamma_\mu\psi_\nu .
    \tag{1}
  \]
  Terms without \(\partial\epsilon\) cancel because the transformations
  with constant parameter are a symmetry.

  Next vary both gravitino factors in \(\mathcal L_{\rm RS}\).  In the
  variation of \(\partial_\rho\psi_\sigma\), separate
  \[
    \partial_\rho\delta\psi_\sigma
      =-i\sigma^{\lambda\tau}
       \left[(\partial_\rho\partial_\lambda h_{\tau\sigma})\epsilon
       +(\partial_\lambda h_{\tau\sigma})\partial_\rho\epsilon\right].
  \]
  Integrate the second-derivative terms once by parts, use the Majorana
  interchange identities, and reduce gamma products with
  \(\{\gamma^\mu,\gamma^\nu\}=2\eta^{\mu\nu}\).  The terms proportional to
  \(\partial\bar\epsilon\) are exactly the negative of (1), while the
  uncancelled part is
  \[
    \delta\mathcal L_{\rm RS}\big|_{\partial\epsilon}
      =i(\partial_\kappa\bar\epsilon)
        \Pi^{\kappa\mid\mu\nu}\gamma_\mu\psi_\nu
       +\frac{i}{2}\epsilon^{\kappa\nu\rho\sigma}
        \bar\psi_\rho\gamma_5\gamma_\nu\sigma^{\lambda\tau}
        \partial_\lambda h_{\tau\sigma}\,\partial_\kappa\epsilon
       +\partial_\kappa K^\kappa .
    \tag{2}
  \]
  Here \(K^\kappa\) is the surface term generated by the integration by
  parts.  Adding (1) and (2) gives
  \[
    \delta S_{\rm lin}
      =\int d^4x\,
        \frac{i}{2}\epsilon^{\mu\nu\rho\sigma}
        \bar\psi_\rho\gamma_5\gamma_\nu\sigma^{\lambda\tau}
        \partial_\lambda h_{\tau\sigma}\,\partial_\mu\epsilon,
  \]
  which identifies the stated \(j^\mu\).

  This is Noether's local-parameter construction: the coefficient of
  \(\partial_\mu\epsilon\) is the current associated with the symmetry for
  constant \(\epsilon\).  Integrating by parts gives
  \[
    \delta S_{\rm lin}
      =-\int d^4x\,(\partial_\mu j^\mu)\epsilon
  \]
  up to a boundary term.  On the free Fierz--Pauli and Rarita--Schwinger
  equations, the original variation vanishes for arbitrary local test
  parameter modulo equations-of-motion terms, and hence
  \(\partial_\mu j^\mu=0\).  Directly, the derivative acting twice on
  \(h_{\tau\sigma}\) drops out against the antisymmetry in
  \(\mu,\sigma\), while the remaining term vanishes after inserting the
  Rarita--Schwinger equation and its gamma trace.

  \SupplementarySolution{2}{Rarita--Schwinger Spin and Off-Shell Balance}
  \begin{enumerate}
    \item \emph{Why the Field Has Spin \(3/2\).}
    Under spatial rotations a vector-spinor contains
      \[
        \mathbf{1}\otimes\mathbf{\frac12}
          =\mathbf{\frac32}\oplus\mathbf{\frac12}.
      \]
      The gamma trace \(\gamma^\mu\psi_\mu\) is precisely a spinor projection.
      The free Rarita--Schwinger equation implies, after a convenient gauge
      choice,
      \[
        \gamma^\mu\psi_\mu=0,\qquad
        \partial^\mu\psi_\mu=0,\qquad
        \slashed\partial\psi_\mu=0 .
      \]
      The gauge invariance
      \(\psi_\mu\to\psi_\mu+\partial_\mu\varepsilon\) removes the residual
      longitudinal spinor.  On shell only helicities
      \(+3/2\) and \(-3/2\) remain, showing that the field describes a
      massless spin-\(3/2\) particle.
    \item \emph{Off-Shell Component Counts.}
    Before equations of motion, a Majorana vector-spinor has
      \(4\times4=16\) real components.  Four local Majorana gauge functions
      remove four, leaving
      \[
        n_F^{\rm off}=16-4=12.
      \]
      A symmetric tensor \(h_{\mu\nu}\) has ten real components.
      Four linearized diffeomorphism functions remove four, leaving
      \[
        n_B^{\rm off}=10-4=6.
      \]
      The mismatch is six bosonic components.
    \item \emph{Auxiliary Fields.}
    Old-minimal supergravity adds a real scalar \(s\), a real pseudoscalar
      \(p\), and a real vector \(b_\mu\).  They contribute
      \[
        1_s+1_p+4_{b_\mu}=6
      \]
      off-shell bosonic components and have algebraic field equations.
      Consequently
      \(6+6=12=n_F^{\rm off}\), so supersymmetry can close without imposing
      dynamical equations.
  \end{enumerate}

  \SupplementarySolution{3}{Closure of Linearized Supergravity}
Substitution of the gravitino transformation into the variation of
  \(h_{\mu\nu}\) gives
  \[
  \begin{aligned}
    [\delta_1,\delta_2]h_{\mu\nu}
      =-\frac12\big[
       &\bar\epsilon_2\gamma_\mu\sigma^{\rho\sigma}\epsilon_1\,
          \partial_\rho h_{\sigma\nu}
        +\bar\epsilon_2\gamma_\nu\sigma^{\rho\sigma}\epsilon_1\,
          \partial_\rho h_{\sigma\mu}\\
       &-(1\leftrightarrow2)\big].
  \end{aligned}
  \]
  With
  \(\gamma_\mu{}^{\rho\sigma}
    \equiv\gamma_{[\mu}\gamma^\rho\gamma^{\sigma]}\), the required
  decomposition is
  \[
    \gamma_\mu\sigma^{\rho\sigma}
      =i\left(
        \delta_\mu^\rho\gamma^\sigma
        -\delta_\mu^\sigma\gamma^\rho
        +\gamma_\mu{}^{\rho\sigma}\right).
  \]
  For anticommuting Majorana parameters, the vector bilinear changes sign
  under \(1\leftrightarrow2\), while the three-gamma bilinear does not.
  Hence the three-gamma contribution cancels between the two parameter
  orderings and the vector contribution doubles.  Symmetry of
  \(h_{\mu\nu}\) then leaves
  \[
    [\delta_1,\delta_2]h_{\mu\nu}
      =-iv^\rho
        \left(\partial_\mu h_{\rho\nu}
              +\partial_\nu h_{\rho\mu}
              -2\partial_\rho h_{\mu\nu}\right),
    \qquad
    v^\rho=\bar\epsilon_2\gamma^\rho\epsilon_1 .
  \]

  The last term is the translation
  \(2iv^\rho\partial_\rho h_{\mu\nu}\).  Defining
  \(\xi_\nu=-iv^\rho h_{\rho\nu}\), the first two terms are
  \(\partial_\mu\xi_\nu+\partial_\nu\xi_\mu\), a linearized
  diffeomorphism.  Thus the supersymmetry commutator closes on a translation
  modulo the gauge symmetry of the graviton.

  \SupplementarySolution{4}{No-Scale Supergravity with \(S,T,C\)}
  \begin{enumerate}
    \item \emph{A Three-Field Model.}
    Set
      \[
        s=S+S^*,\qquad
        t=T+T^*,\qquad
        U=t-|C|^2,\qquad
        A=ae^{-\alpha S}.
      \]
      The nonzero metric components are
      \[
      K_{S\bar S}=\frac1{s^2},\qquad
      (K_{i\bar j})_{i,j=T,C}
        =\frac3{U^2}
          \begin{pmatrix}1&-C\\-C^*&t\end{pmatrix},
      \]
      with inverse
      \[
      K^{S\bar S}=s^2,\qquad
      (K^{i\bar j})_{i,j=T,C}
        =\frac U3
          \begin{pmatrix}t&C^*\\C&1\end{pmatrix}.
      \]
      Here the displayed inverse is indexed by row \(i\) and column
      \(\bar j\); direct contraction gives
      \(K_{i\bar j}K^{k\bar j}=\delta_i{}^k\).
      The Kähler derivatives are
      \[
        D_SW=-\alpha A-\frac Ws,\qquad
        D_TW=-\frac{3W}{U},\qquad
        D_CW=3C^2+\frac{3C^*W}{U}.
      \]
      Therefore
      \[
      \begin{aligned}
        V=e^K\Bigg\{&
          s^2|D_SW|^2-3|W|^2\\
          &+\frac U3\left[
            t|D_TW|^2+|D_CW|^2\right.\\
          &\hspace{4.5em}\left.
            +2\operatorname{Re}\!\left(
              C^*D_TW\,\overline{D_CW}\right)\right]\Bigg\},
      \end{aligned}
      \]
      where \(e^K=(sU^3)^{-1}\).  In the convention
      \(F^i=-e^{K/2}K^{i\bar j}\overline{D_jW}\), the auxiliaries are
      \[
      \begin{aligned}
        F^S&=-e^{K/2}s^2\overline{D_SW},\\
        F^T&=-e^{K/2}\frac U3
          \left(t\overline{D_TW}+C^*\overline{D_CW}\right),\\
        F^C&=-e^{K/2}\frac U3
          \left(C\overline{D_TW}+\overline{D_CW}\right).
      \end{aligned}
      \]
    \item \emph{Broken Supersymmetry and Flat Directions.}
    At \(C=0\), \(U=t\), \(D_CW=0\), and the \(T\)-term cancels
      \(-3|W|^2\).  The potential becomes
      \[
        V=\frac{|W+\alpha sA|^2}{s\,t^3}\ge0.
      \]
      Hence a zero-energy stationary locus is
      \[
        C=0,\qquad W+\alpha sA=0,
        \quad\text{or}\quad
        b=-(1+\alpha s)ae^{-\alpha S}.
      \]
      On this locus \(F^S=F^C=0\), whereas
      \[
        F^T=e^{K/2}t\,\overline W
      \]
      is nonzero for generic \(a,b\).  Supersymmetry is therefore broken
      with zero classical vacuum energy.  The complex modulus \(T\) is
      undetermined subject only to \(t>0\); both its real scale and axionic
      direction are tree-level flat.
    \item \emph{General No-Scale Identity.}
    Homogeneity gives
      \(\tau_i\Gamma_i=\Gamma\) and, after differentiation,
      \(\tau_i\Gamma_{ij}=0\).  Since
      \[
        K_{ij}=3\left(
          \frac{\Gamma_i\Gamma_j}{\Gamma^2}
          -\frac{\Gamma_{ij}}{\Gamma}\right),
      \]
      contraction yields
      \[
        \tau_iK_{ij}=\frac{3\Gamma_j}{\Gamma}.
      \]
      Multiplying by \(K^{-1}_{jk}\) and then by
      \(\Gamma_k/\Gamma\) gives
      \[
        1=\frac{\tau_k\Gamma_k}{\Gamma}
          =3\frac{\Gamma_jK^{-1}_{jk}\Gamma_k}{\Gamma^2},
      \]
      proving the second identity.  Finally
      \(K_i=-3\Gamma_i/\Gamma\), so
      \[
        K_iK^{-1}_{ij}K_j=3.
      \]
      If \(W\) is independent of these moduli, \(D_iW=K_iW\);
      their positive contribution to
      \(e^K(K^{i\bar j}D_iW\overline{D_jW}-3|W|^2)\) is exactly
      \(3e^K|W|^2\) and cancels the universal negative term.
  \end{enumerate}

  \SupplementarySolution{5}{Anomaly-Mediated Gaugino and Trilinear Terms}
  \begin{enumerate}
    \item \emph{Anomaly-Mediated Gaugino Mass.}
    Introduce a chiral compensator
      \(\mathcal C=1+\theta^2m_g\).  Super-Weyl invariance makes the running
      coupling depend on \(\mu/\mathcal C\).  Expanding the holomorphic
      gauge coefficient,
      \[
        \frac1{g^2(\mu/\mathcal C)}
          =\frac1{g^2(\mu)}
           -m_g\theta^2\frac{d}{d\ln\mu}\frac1{g^2}+\cdots .
      \]
      The \(\theta^2W^\alpha W_\alpha\) component is a gaugino mass.
      Canonical normalization gives
      \[
        M_\lambda=\frac{\beta_g}{g}m_g
      \]
      up to the phase chosen for \(m_g\).  The compensator is part of the
      supergravity multiplet, so its auxiliary component is present even
      when no gauge-singlet hidden field can appear in the gauge kinetic
      function.  This is why the quantum contribution evades the
      tree-level singlet requirement.
    \item \emph{One-Loop Gauge Example.}
    Substitution of
      \(\beta_g=bg^3/(16\pi^2)\) gives
      \[
        M_\lambda=\frac{bg^2}{16\pi^2}m_g.
      \]
      It is suppressed by one loop relative to \(m_g\).  A direct
      tree-level contribution from a field-dependent gauge kinetic
      function has the form
      \[
        M_A^{\rm tree}
          =\frac{F^i\partial_i f_A}{2\operatorname{Re}f_A}.
      \]
      If \(\partial_i f_A=O(\kappa)\) and \(F^i=O(m_g/\kappa)\), it is
      \(O(m_g)\), parametrically larger than anomaly mediation.  If a
      symmetry forbids the derivative or no singlet exists, the tree term
      vanishes and the loop result is leading.
    \item \emph{Anomaly-Mediated Trilinear Term.}
    Canonical fields are obtained from wavefunction factors \(Z_i\).
      The physical Yukawa coupling is
      \(y^{\rm phys}_{ijk}=y_{ijk}(Z_iZ_jZ_k)^{-1/2}\).  Replacing
      \(\mu\) by \(\mu/\mathcal C\) and expanding each \(Z\) in
      \(\theta^2m_g\) produces a soft scalar vertex.  With
      \(\gamma_i=d\ln Z_i/d\ln\mu\),
      \[
        a_{ijk}
          =-\frac12(\gamma_i+\gamma_j+\gamma_k)m_g\,y_{ijk}
      \]
      in this definition of \(\gamma\).  If instead
      \(\gamma_i=\tfrac12d\ln Z_i/d\ln\mu\), the factor \(1/2\) is absent:
      \[
        a_{ijk}=-(\gamma_i+\gamma_j+\gamma_k)m_g\,y_{ijk}
                =-m_g\beta_{y_{ijk}}.
      \]
      The invariant statement is that the \(A\)-term is minus \(m_g\) times
      the logarithmic beta function of the corresponding Yukawa coupling.
  \end{enumerate}

  \SupplementarySolution{6}{Anomaly-Mediated Scalar Masses and Threshold Insensitivity}
  \begin{enumerate}
    \item \emph{Scalar Mass from Running.}
    The real wavefunction factor must depend on the invariant combination
      \(\mu/(\mathcal C\mathcal C^*)^{1/2}\).  Expanding twice,
      \[
        \ln Z_i
          =\ln Z_i(\mu)
           -\frac12(\theta^2m_g+\bar\theta^2m_g^*)\gamma_i
           +\frac14\theta^2\bar\theta^2|m_g|^2
             \frac{d\gamma_i}{d\ln\mu}.
      \]
      The last component becomes a scalar squared mass after canonical
      normalization:
      \[
        m_i^2=-\frac14|m_g|^2\frac{d\gamma_i}{d\ln\mu}.
      \]
      Some authors define \(\gamma_i=-d\ln Z_i/d\ln\mu\), or absorb a
      factor \(1/2\) in it.  Both the displayed sign and factor then change.
      The convention-independent statement is that \(m_i^2\) is fixed by
      the RG derivative of the anomalous dimension and is two-loop order.
    \item \emph{Ultraviolet Insensitivity.}
    Above a supersymmetric threshold,
      \(M_\lambda^+=(\beta_g^+/g)m_g\).  The mass \(M\) occurs as the
      compensator-dressed chiral parameter \(M\mathcal C\).  Integrating out
      the massive multiplet therefore gives a gaugino threshold
      \[
        \Delta M_\lambda
          =-\frac{\Delta\beta_g}{g}m_g,\qquad
        \Delta\beta_g\equiv\beta_g^+-\beta_g^-.
      \]
      Matching yields
      \[
        M_\lambda^-
          =M_\lambda^++\Delta M_\lambda
          =\frac{\beta_g^-}{g}m_g.
      \]
      Thus a supersymmetric heavy threshold leaves no independent memory of
      ultraviolet soft parameters.  Nonsupersymmetric threshold splittings
      would spoil the cancellation and are outside the premise.
    \item \emph{The Slepton-Sign Problem.}
    With negligible lepton Yukawa couplings, the lepton anomalous
      dimensions are dominated by electroweak gauge terms.  In the standard
      convention
      \(\gamma_{\tilde\ell}\sim-[C_1g_1^2+C_2g_2^2]/(8\pi^2)\).
      Both electroweak couplings have positive one-loop beta coefficients in
      the MSSM, so \(d\gamma_{\tilde\ell}/d\ln\mu<0\).  The standard
      anomaly-mediated formula
      \(m_i^2=\tfrac12|m_g|^2\dot\gamma_i\) then gives
      \[
        m_{\tilde L}^2<0,\qquad m_{\tilde e}^2<0.
      \]
      Equivalent anomalous-dimension conventions reproduce the same signs.
      A negative squared mass makes the origin a maximum of the scalar
      potential; charged sleptons acquire expectation values and break
      electromagnetism.  It is therefore a physical tachyonic instability,
      not a harmless sign convention.
  \end{enumerate}

  \SupplementarySolution{7}{Kähler Invariance and Minkowski Tuning}
  \begin{enumerate}
    \item \emph{Kähler-Transformation Invariance.}
    The first combination transforms as
      \[
        e^{\kappa^2(d+a+a^*)}
          |fe^{-\kappa^2a}|^2
          =e^{\kappa^2d}|f|^2.
      \]
      The metric is unchanged because
      \(\partial_n\partial_{\bar m}(a+a^*)=0\).  For the covariant
      derivative,
      \[
        L'_n
          =\partial_n(fe^{-\kappa^2a})
           +\kappa^2fe^{-\kappa^2a}(d_n+a_n)
          =e^{-\kappa^2a}L_n.
      \]
      Therefore
      \[
        e^{\kappa^2d'}g^{n\bar m}L'_nL_{\bar m}^{\prime *}
          =e^{\kappa^2d}g^{n\bar m}L_nL_{\bar m}^*.
      \]
      Both pieces of the \(F\)-term potential are invariant separately.
    \item \emph{Minkowski \(F\)-Term Sum Rule.}
    With the definition in the question,
      \[
        g_{n\bar m}F^nF^{*\bar m}
          =e^{\kappa^2d}g^{n\bar m}L_nL_{\bar m}^*.
      \]
      Also
      \(m_g^2=\kappa^4e^{\kappa^2d}|f|^2\).  Thus the potential is
      \[
        V_F=g_{n\bar m}F^nF^{*\bar m}
            -\frac{3m_g^2}{\kappa^2}.
      \]
      Vanishing \(D\)-terms and zero vacuum energy imply
      \[
        \boxed{\quad
        g_{n\bar m}F^nF^{*\bar m}
          =\frac{3m_g^2}{\kappa^2}\quad .}
      \]
      This is the multi-field form of the cancellation used in
      Supplementary Exercise 6.
  \end{enumerate}

  \SupplementarySolution{8}{Coupling the Wess--Zumino Supercurrent to the Gravitino}
  Put
  \[
    C_+=A+i\gamma_5B,
    \qquad C_-=A-i\gamma_5B,
    \qquad
    J_\sigma=A\overleftrightarrow{\partial_\sigma}B.
  \]
  A convenient symmetric representative of the free stress tensor is
  \[
  \begin{split}
    T_{\mu\nu}={}&\partial_\mu A\partial_\nu A
      +\partial_\mu B\partial_\nu B
      -\frac12\eta_{\mu\nu}
        \bigl[(\partial A)^2+(\partial B)^2\bigr]\\
      &+\frac{i}{4}\bar\psi
        \bigl(\gamma_\mu\overleftrightarrow{\partial_\nu}
             +\gamma_\nu\overleftrightarrow{\partial_\mu}\bigr)\psi .
  \end{split}
  \tag{1}
  \]
  Terms proportional to the free fermion equation may be moved between
  (1) and an improvement without changing the following Noether argument.

  First vary the gravitino in the interaction.  Since
  \(\delta\bar\psi_\mu=(2/\kappa)\partial_\mu\bar\epsilon\),
  \[
    -\frac{\kappa}{2}(\delta\bar\psi_\mu)j^\mu
      =-(\partial_\mu\bar\epsilon)j^\mu.
  \tag{2}
  \]
  This is already the negative of the local-parameter variation of the
  free Wess--Zumino Lagrangian.

  It remains to vary the matter fields in \(j^\mu\).  Starting from
  \[
    j^\mu=(\gamma^\alpha\partial_\alpha C_-)\gamma^\mu\psi,
  \]
  insert \(\delta A\), \(\delta B\), and
  \(\delta\psi=-i\slashed\partial C_+\epsilon\).  Reorder Majorana
  bilinears and use
  \[
    \gamma^\alpha\gamma^\mu\gamma^\beta
      =\eta^{\alpha\mu}\gamma^\beta
       -\eta^{\alpha\beta}\gamma^\mu
       +\eta^{\mu\beta}\gamma^\alpha
       +i\epsilon^{\alpha\mu\beta\nu}\gamma_5\gamma_\nu,
  \tag{3}
  \]
  with the corresponding sign if the opposite convention for
  \(\epsilon^{0123}\) is chosen.  The terms symmetric in the two scalar
  derivatives and the fermion bilinears assemble into (1); the
  antisymmetric part of (3) gives the scalar rotation current.  For the
  improved current representative used in the question, the result is
  \[
    \delta j^\mu
      =-2i\gamma^\nu\epsilon\,T^\mu{}_{\nu}
       +i\epsilon^{\mu\nu\rho\sigma}\gamma_\nu\epsilon\,
          \partial_\rho J_\sigma,
  \tag{4}
  \]
  up to the free matter equations and a divergence of an antisymmetric
  improvement tensor.  Equation (4) is the component supercurrent
  multiplet identity needed here.

  Combining (2) and (4) gives
  \[
  \begin{split}
    \delta\mathcal L_1={}&-(\partial_\mu\bar\epsilon)j^\mu
      +i\kappa\bar\psi^\mu\gamma^\nu\epsilon T_{\mu\nu}\\
      &-\frac{i\kappa}{2}\epsilon^{\mu\nu\rho\sigma}
       \bar\psi_\mu\gamma_\nu\epsilon\,\partial_\rho J_\sigma.
  \end{split}
  \tag{5}
  \]
  Integrating the last line by parts,
  \[
  \begin{split}
    -\bar\psi_\mu\gamma_\nu\epsilon\,\partial_\rho J_\sigma
      ={}&(\partial_\rho\bar\psi_\mu)\gamma_\nu\epsilon J_\sigma
       +\bar\psi_\mu\gamma_\nu(\partial_\rho\epsilon)J_\sigma\\
       &-\partial_\rho
          (\bar\psi_\mu\gamma_\nu\epsilon J_\sigma),
  \end{split}
  \]
  turns (5) into precisely
  \[
  \begin{split}
    \delta\mathcal L_1={}&-(\partial_\mu\bar\epsilon)j^\mu
      +i\kappa\bar\psi^\mu\gamma^\nu\epsilon T_{\mu\nu}\\
      &+\frac{i\kappa}{2}\epsilon^{\mu\nu\rho\sigma}
        \bar\psi_\mu\gamma_\nu\partial_\rho\epsilon J_\sigma
       +\frac{i\kappa}{2}\epsilon^{\mu\nu\rho\sigma}
        (\partial_\rho\bar\psi_\mu)\gamma_\nu\epsilon J_\sigma,
  \end{split}
  \]
  modulo the displayed total derivative.

  The first term cancels \(\delta\mathcal L\).  The stress-tensor term
  demands the usual coupling of the graviton to \(T_{\mu\nu}\), together
  with the supersymmetry variation of the graviton.  The two
  \(J_\sigma\) terms demand further order-\(\kappa\) contact terms and
  supercovariant completions.  Iterating these cancellations is the
  Noether construction of the locally supersymmetric theory.

  \SupplementarySolution{9}{Scalar Soft Terms, Trilinears, and Flavor}
  \begin{enumerate}
    \item \emph{Visible Scalar Soft Mass.}
    Near \(Q=0\), write
      \(d=d_{\rm hid}(z,z^*)+Z_Q(z,z^*)|Q|^2+\cdots\).
      The canonically normalized field is \(Q_c=Z_Q^{1/2}Q\).  Expanding the
      supergravity potential to quadratic order and using the stationary
      hidden-sector equations gives
      \[
        m_Q^2=m_g^2+\frac{V_0\kappa^2}{3}
          -F^mF^{*\bar n}
           \partial_m\partial_{\bar n}\ln Z_Q.
      \]
      For a tuned Minkowski vacuum, \(V_0=0\), hence
      \[
        \boxed{\quad m_Q^2=m_g^2
          -F^mF^{*\bar n}
           \partial_m\partial_{\bar n}\ln Z_Q\quad .}
      \]
      The second term is the sectional Kähler curvature sampled along the
      supersymmetry-breaking direction.
    \item \emph{Gravity-Mediated \(A\)-Term.}
    The physical Yukawa coupling is
      \[
        Y^{\rm phys}_{ijk}
          =e^{\kappa^2d/2}
            \frac{Y_{ijk}}{(Z_iZ_jZ_k)^{1/2}}.
      \]
      Varying it along the hidden auxiliary direction gives the scalar
      trilinear.  With the convention
      \(V_{\rm soft}\supset A_{ijk}Y^{\rm phys}_{ijk}q_iq_jq_k+\mathrm{c.c.}\),
      \[
        A_{ijk}
          =F^m\left[
            \kappa^2d_m+\partial_m\ln Y_{ijk}
            -\partial_m\ln(Z_iZ_jZ_k)
          \right],
      \]
      where a factor \(1/2\) moves between \(A\) and the definition of the
      canonically normalized Yukawa in alternative conventions.  Equivalently,
      the hidden dependence is summarized by
      \[
        A_{ijk}\propto
        F^m\partial_m\ln\!
          \left(\frac{e^{\kappa^2d}Y_{ijk}}{Z_iZ_jZ_k}\right).
      \]
    \item \emph{Flavor Universality Criterion.}
    A sufficient condition is
      \(Z_{i\bar j}=Z(z,z^*)\delta_{i\bar j}\) for fields with identical
      gauge quantum numbers, together with hidden-independent ratios among
      the corresponding Yukawa tensors.  Then
      \[
        \partial_m\partial_{\bar n}\ln Z_i
      \]
      is generation independent, so the scalar mass matrix is proportional
      to the identity.  Likewise
      \(\partial_m\ln(Z_iZ_jZ_k)\) is the same for every entry of a given
      Yukawa tensor, making the trilinear matrix proportional to that
      tensor.

      Flavor violation arises from nonuniversal first derivatives
      \(\partial_mZ_{i\bar j}\), which misalign \(A\)-terms, and especially
      from nonuniversal mixed second derivatives
      \(\partial_m\partial_{\bar n}Z_{i\bar j}\), which generate off-diagonal
      scalar squared masses.  Gauge symmetries alone do not generally forbid
      these Kähler couplings.
  \end{enumerate}

  \SupplementarySolution{10}{Planck-Suppressed Mediation and the Goldstino Direction}
  \begin{enumerate}
    \item \emph{Planck-Suppressed Mediation Scale.}
    A representative operator is
      \[
        \int d^2\theta\,\frac{X}{M_{\rm Pl}}W^\alpha W_\alpha
        \quad\hbox{or}\quad
        \int d^4\theta\,\frac{X^*X}{M_{\rm Pl}^2}Q^*Q,
      \]
      with \(M_{\rm Pl}=\kappa^{-1}\).  Replacing
      \(X|_{\theta^2}\) by \(F\) gives
      \[
        m_{\rm soft}=O\!\left(\frac{F}{M_{\rm Pl}}\right)
                    =O(\kappa F).
      \]
      Tuning the vacuum energy gives
      \(F_{\rm std}=\sqrt3\,m_g/\kappa\), so
      \(m_{\rm soft}=O(m_g)\), with coefficients controlled by unknown
      Planck-scale operators.  This is the characteristic scale of gravity
      mediation.
    \item \emph{Goldstino Direction.}
    Let the positive breaking energy be
      \[
        F_{\rm tot}^2=\sum_i|F_i|^2+\frac12\sum_A D_A^2.
      \]
      The normalized goldstino is, up to phases,
      \[
        \widetilde G
          =\frac1{F_{\rm tot}}\left(
            \sum_iF_i^*\psi_i
            -\frac{i}{\sqrt2}\sum_AD_A\gamma_5\lambda_A
          \right).
      \]
      The fermion variations contain
      \(\delta\psi_i\supset F_i\alpha\) and
      \(\delta\lambda_A\supset iD_A\gamma_5\alpha\).  Taking the same linear
      combination gives
      \[
        \delta\widetilde G=F_{\rm tot}\alpha+\text{derivative terms}.
      \]
      A nonzero constant shift is the defining signature of the Goldstone
      fermion of broken global supersymmetry.  In supergravity it is removed
      in unitary gauge and supplies the longitudinal gravitino.
  \end{enumerate}

  \SupplementarySolution{11}{Longitudinal-Gravitino Equivalence and Decay}
  \begin{enumerate}
    \item \emph{Longitudinal-Gravitino Equivalence.}
    A high-energy helicity-\(1/2\) polarization of a massive gravitino has
      the leading form
      \[
        u_\mu(q,\tfrac12)
          =\sqrt{\frac23}\frac{q_\mu}{m_g}u(q,\tfrac12)
           +O(m_g/E),
      \]
      and similarly for helicity \(-1/2\).  Contracting this with the
      supercurrent coupling replaces \(q_\mu S^\mu/m_g\) by the divergence
      of the supercurrent.  The broken-supersymmetry Ward identity relates
      that divergence to the goldstino coupling.  Hence
      \[
        \mathcal M(\psi_{\mu,\rm long})
          =\mathcal M(\widetilde G)+O(m_g/E).
      \]
      The theorem concerns only the longitudinal helicities; the transverse
      \(\pm3/2\) amplitudes remain gravitationally coupled.
    \item \emph{Gravitino Decay Scaling.}
    Goldstino interactions are derivative couplings suppressed by \(F\).
      For a parent of mass \(m\), the amplitude therefore scales as
      \(m^2/F\).  Two-body phase space supplies one inverse power of \(m\),
      so
      \[
        \Gamma(\widetilde X\to X+\widetilde G)
          =C\,\frac{m^5}{16\pi F^2},
      \]
      where \(C\) contains mixing angles and
      \((1-m_X^2/m^2)^4\).  With Weinberg's flat-vacuum relation
      \(F^2=6m_g^2/\kappa^2\),
      \[
        \Gamma
          =C\,\frac{\kappa^2m^5}{96\pi m_g^2}.
      \]
      Thus a lighter gravitino couples more strongly through its
      longitudinal component, while a Planck-scale \(F\) produces a very
      long-lived parent.
  \end{enumerate}

  \SupplementarySolution{12}{The Polonyi Model}
  \begin{enumerate}
    \item Since \(K_{z\bar z}=K^{z\bar z}=1\),
      \[
        D_zW=m^2\bigl[1+z^*(z+\beta)\bigr].
      \tag{1}
      \]
      For \(m\ne0\), supersymmetry at a vacuum would require
      \[
        1+\langle z\rangle^*(\langle z\rangle+\beta)=0.
      \tag{2}
      \]
      Writing \(z=x+iy\) and using real \(\beta>0\), the imaginary part of
      (2) gives \(y=0\), while the real part is
      \(x^2+\beta x+1=0\).  Thus a real supersymmetric solution can exist
      only for \(\beta\geq2\).  An unbroken Minkowski vacuum would also
      require \(W=0\), but \(z=-\beta\) makes
      \(D_zW=m^2\), so no such vacuum exists for \(m\ne0\).  The tuned
      stable solution found below has \(\beta=2-\sqrt3<2\) and therefore
      breaks supersymmetry.  The exceptional case \(m=0\) is the trivial
      theory.

    \item Substitution of (1) into the supplied supergravity formula gives
      the complete potential
      \[
        \boxed{V(z)=|m|^4e^{|z|^2}
          \left(\left|1+z^*(z+\beta)\right|^2
                -3|z+\beta|^2\right)}.
      \tag{3}
      \]

    \item At a stationary vacuum the phase is fixed at
      \(\operatorname{Im}z=0\), so put \(z=x\),
      \(u=x+\beta\), and \(f=1+xu\).  Apart from the positive factor
      \(|m|^4e^{x^2}\), the potential is
      \[
        Q(x)=f^2-3u^2.
      \]
      Zero vacuum energy says \(f=s\sqrt3\,u\), with \(s=\pm1\).
      Because \(Q=0\), stationarity of the full exponential potential is
      simply \(Q'=0\):
      \[
        0=2f(2x+\beta)-6u
          =2u\bigl[s\sqrt3(x+u)-3\bigr].
      \]
      The branch \(u=0\) is incompatible with \(f=1\), so
      \(x+u=s\sqrt3\).  Combining this with
      \(1+xu=s\sqrt3u\) gives \(u^2=1\).  The locally stable solution is
      \[
        u=1,qquad s=+1,qquad
        x=\sqrt3-1,qquad
        \beta=2-\sqrt3.
      \tag{4}
      \]
      Indeed, at (4) the two real Hessian eigenvalues are proportional to
      \(4\sqrt3\) and \(8-4\sqrt3\), both positive.  The alternative
      positive-\(\beta\) stationary branch
      \(\beta=2+\sqrt3\), \(x=-1-\sqrt3\) has a negative real-direction
      eigenvalue and is not the vacuum.  Comparing (4) with
      \(\beta=-\sqrt A+B\) gives
      \[
        \boxed{A=3,\qquad B=2.}
      \]

    \item The same solution has
      \[
        \langle z\rangle=\sqrt3-1.
      \]
      Hence
      \[
        \boxed{C=3,\qquad D=-1.}
      \]
      Finally,
      \(D_zW=m^2\sqrt3\ne0\) at this point, explicitly confirming the
      spontaneous breaking asserted in part (a).
  \end{enumerate}

\end{enumerate}
