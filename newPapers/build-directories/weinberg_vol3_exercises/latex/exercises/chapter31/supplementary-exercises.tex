  \SupplementaryExercise{1}{The Linearized Supergravity Current}
    {(Adapted from A.~C. Davis, \emph{Supergravity 1 Example Sheet},
      Question 2.)}
    {cambridge-supergravity1-q2-v3ch31-s01}
  Let \(h_{\mu\nu}=h_{\nu\mu}\), \(h=h^\mu{}_{\mu}\), and let
  \(\psi_\mu\) be a Majorana vector--spinor.  Work with the quadratic
  action
  \[
  \begin{split}
    S_{\rm lin}=\int d^4x\,\bigg[&
      -\frac12\partial_\lambda h_{\mu\nu}\partial^\lambda h^{\mu\nu}
      +\partial_\mu h^{\mu\nu}\partial^\rho h_{\rho\nu}
      -\partial_\mu h^{\mu\nu}\partial_\nu h
      +\frac12\partial_\lambda h\partial^\lambda h\\
      &-\frac{i}{2}\epsilon^{\mu\nu\rho\sigma}
        \bar\psi_\mu\gamma_5\gamma_\nu\partial_\rho\psi_\sigma
      \bigg]
  \end{split}
  \]
  and the linearized transformations
  \[
    \sigma^{\rho\sigma}=\frac{i}{2}[\gamma^\rho,\gamma^\sigma],
    \qquad
    \delta h_{\mu\nu}=-\frac{i}{2}\bar\epsilon
      (\gamma_\mu\psi_\nu+\gamma_\nu\psi_\mu),
    \qquad
    \delta\psi_\mu=-i\sigma^{\rho\sigma}
      \partial_\rho h_{\sigma\mu}\epsilon .
  \]
  These are a symmetry for constant Majorana \(\epsilon\).  Now promote
  the parameter to \(\epsilon(x)\), retain every term in which a derivative
  acts on it, and show, up to a surface term, that
  \[
    \delta S_{\rm lin}=\int d^4x\,j^\mu\partial_\mu\epsilon,
    \qquad
    j^\mu=\frac{i}{2}\epsilon^{\mu\nu\rho\sigma}
      \bar\psi_\rho\gamma_5\gamma_\nu\sigma^{\lambda\tau}
      \partial_\lambda h_{\tau\sigma}.
  \]
  Explain why this coefficient is the Noether supercurrent and verify its
  conservation using the free equations of motion.

  \SupplementaryExercise{2}{Rarita--Schwinger Spin and Off-Shell Balance}
    {(Adapted from A.~C. Davis, \emph{Supergravity 1 Example Sheet}, Question 3.)}
    {ch31-s02-gravitino-degrees-of-freedom-and-the}
  \begin{enumerate}
    \item \emph{Why the Field Has Spin \(3/2\).}
    Starting from a Majorana vector-spinor \(\psi_\mu\), decompose the
      vector--spinor product into spin-\(3/2\) and spin-\(1/2\) pieces.
      Explain how the Rarita--Schwinger equation and gauge invariance remove
      the unwanted spin-\(1/2\) component.
    \item \emph{Off-Shell Component Counts.}
    In four dimensions count the independent off-shell components of a
      massless Majorana gravitino after local fermionic gauge redundancy, and
      compare with a massless graviton after linearized diffeomorphisms.
    \item \emph{Auxiliary Fields.}
    Deduce the number of bosonic auxiliary components required for off-shell
      supersymmetry and show how the old-minimal set
      \((s,p,b_\mu)\) supplies them.
  \end{enumerate}

  \SupplementaryExercise{3}{Closure of Linearized Supergravity}
    {(Adapted from A.~C. Davis, \emph{Supergravity 1 Example Sheet}, Question 1.)}
    {ch31-s03-torsion-one-and-a-half-order}
Let the fields of linearized four-dimensional supergravity transform
  under constant Majorana parameters as
  \[
    \sigma^{\rho\sigma}
      \equiv\frac{i}{2}[\gamma^\rho,\gamma^\sigma]
      =-2\mathcal J^{\rho\sigma},
    \qquad
    \delta_\epsilon h_{\mu\nu}
      =-\frac{i}{2}\bar\epsilon
        (\gamma_\mu\psi_\nu+\gamma_\nu\psi_\mu),
    \qquad
    \delta_\epsilon\psi_\mu
      =-i\sigma^{\rho\sigma}\partial_\rho h_{\sigma\mu}\epsilon .
  \]
  Compute the commutator on \(h_{\mu\nu}\) and prove
  \[
  \begin{aligned}
    [\delta_{\epsilon_1},\delta_{\epsilon_2}]h_{\mu\nu}
      =-i\bar\epsilon_2\gamma^\rho\epsilon_1
       \big(&\partial_\mu h_{\rho\nu}
             +\partial_\nu h_{\rho\mu}\\
             &-2\partial_\rho h_{\mu\nu}\big).
  \end{aligned}
  \]
  Separate the result into a translation and a linearized
  diffeomorphism.

  \SupplementaryExercise{4}{No-Scale Supergravity with \(S,T,C\)}
    {(Adapted from University of Cambridge, \emph{Supersymmetry and Extra Dimensions Example Sheet 4}, Problem 10 in Section 4 (Exercise 4.10).)}
    {ch31-s04-off-shell-balance-ads-and-no}
  \begin{enumerate}
    \item \emph{A Three-Field Model.}
    In Planck units take
      \[
        K=-\ln(S+S^*)-3\ln(T+T^*-CC^*),\qquad
        W=C^3+ae^{-\alpha S}+b,
      \]
      with complex \(a,b\) and \(\alpha>0\).  Compute the Kähler metric,
      its inverse, the three chiral auxiliary fields, and the complete
      \(F\)-term scalar potential.
    \item \emph{Broken Supersymmetry and Flat Directions.}
    Treat \(C\) as a matter field with vanishing expectation value.  Find a
      zero-energy stationary locus, determine which auxiliary field breaks
      supersymmetry there, and identify every tree-level flat direction.
    \item \emph{General No-Scale Identity.}
    Let \(K=-3\ln\Gamma(\tau_i)\), where \(\Gamma\) is homogeneous of degree
      one in real moduli \(\tau_i\).  Starting from
      \(\tau_i\Gamma_i=\Gamma\) and
      \(\tau_i\Gamma_{ij}=0\), prove
      \[
        \tau_iK_{ij}=\frac{3\Gamma_j}{\Gamma},
        \qquad
        \frac{\Gamma_iK^{-1}_{ij}\Gamma_j}{\Gamma^2}=\frac13.
      \]
      Deduce that moduli absent from \(W\) make a vanishing contribution to
      the \(N=1\) supergravity scalar potential.
  \end{enumerate}

  \SupplementaryExercise{5}{Anomaly-Mediated Gaugino and Trilinear Terms}
    {(Inspired by Giudice, Luty, Murayama, and Rattazzi, ``Gaugino Mass without Singlets'', Sections 2--4; independently formulated.)}
    {ch31-s05-anomaly-mediated-gaugino-and-trilinear-terms}
  \begin{enumerate}
    \item \emph{Anomaly-Mediated Gaugino Mass.}
    Treat the renormalization scale as a chiral compensator spurion and
      derive
      \(M_\lambda=(\beta_g/g)m_g\), up to the common phase convention.
      Why does this contribution survive without a hidden-sector singlet?
    \item \emph{One-Loop Gauge Example.}
    For \(\beta_g=bg^3/(16\pi^2)\), evaluate the anomaly-mediated
      gaugino mass.  Compare its loop order and parametric size with a
      tree-level mass \(F^i\partial_i f_{AB}\).
    \item \emph{Anomaly-Mediated Trilinear Term.}
    Let \(W=y_{ijk}\Phi_i\Phi_j\Phi_k/6\).  Use wavefunction
      renormalization to show that the anomaly-mediated trilinear
      coefficient is proportional to
      \(-(\gamma_i+\gamma_j+\gamma_k)m_g y_{ijk}\).
  \end{enumerate}

  \SupplementaryExercise{6}{Anomaly-Mediated Scalar Masses and Threshold Insensitivity}
    {(Inspired by Giudice, Luty, Murayama, and Rattazzi, ``Gaugino Mass without Singlets'', Sections 3--5; independently formulated.)}
    {ch31-s06-anomaly-mediated-scalar-masses-and-threshold}
  \begin{enumerate}
    \item \emph{Scalar Mass from Running.}
    Show that the compensator dependence of a wavefunction factor
      \(Z_i(\mu)\) produces
      \(m_i^2=-|m_g|^2\,d\gamma_i/d\ln\mu\,/4\) in the convention
      \(\gamma_i=d\ln Z_i/d\ln\mu\).  Explain the convention dependence of
      the overall sign and factor.
    \item \emph{Ultraviolet Insensitivity.}
    A supersymmetric multiplet of mass \(M\) is integrated out.  Show that
      the threshold change of \(\beta_g\) is accompanied by a matching
      correction to \(M_\lambda\), leaving the low-energy expression
      dependent only on the low-energy beta function.
    \item \emph{The Slepton-Sign Problem.}
    Neglect lepton Yukawa couplings and use the signs of the electroweak
      beta functions to explain why minimal anomaly mediation gives
      negative slepton squared masses.  Why is this a genuine instability
      rather than merely a negative Lagrangian mass parameter?
  \end{enumerate}

  \SupplementaryExercise{7}{Kähler Invariance and Minkowski Tuning}
    {(Inspired by Brignole, Ib\'a\~nez, and Mu\~noz, ``Soft Supersymmetry-Breaking Terms from Supergravity and Superstring Models'', Section 2; independently formulated.)}
    {ch31-s07-k-hler-invariance-and-minkowski-tuning}
  \begin{enumerate}
    \item \emph{Kähler-Transformation Invariance.}
    Under
      \(d\to d+a+a^*\) and \(f\to fe^{-\kappa^2a}\), prove that
      \(e^{\kappa^2d}|f|^2\) and
      \(e^{\kappa^2d}g^{n\bar m}L_nL_{\bar m}^*\) are invariant.
    \item \emph{Minkowski \(F\)-Term Sum Rule.}
    Define
      \(F^n=-e^{\kappa^2d/2}g^{n\bar m}L_{\bar m}^*\).
      Derive the condition on \(g_{n\bar m}F^nF^{*\bar m}\) and \(m_g\)
      in a stationary Minkowski vacuum with vanishing \(D\)-terms.
  \end{enumerate}

  \SupplementaryExercise{9}{Scalar Soft Terms, Trilinears, and Flavor}
    {(Inspired by Brignole, Ib\'a\~nez, and Mu\~noz, ``Soft Supersymmetry-Breaking Terms from Supergravity and Superstring Models'', Sections 2--3; independently formulated.)}
    {ch31-s09-scalar-soft-terms-trilinears-and-flavor}
  \begin{enumerate}
    \item \emph{Visible Scalar Soft Mass.}
    A visible field \(Q\) has Kähler metric \(Z_Q(z,z^*)\) and vanishing
      vacuum expectation value, while hidden fields \(z^m\) break
      supersymmetry.  Derive
      \(m_Q^2=m_g^2-F^mF^{*\bar n}
      \partial_m\partial_{\bar n}\ln Z_Q\) for zero vacuum energy.
    \item \emph{Gravity-Mediated \(A\)-Term.}
    For a visible Yukawa coupling \(Y_{ijk}(z)\) and diagonal visible
      metrics \(Z_i\), derive the hidden-\(F\) contribution proportional to
      \[
        F^m\partial_m
        \ln\!\left(\frac{e^{\kappa^2d}Y_{ijk}}
          {Z_iZ_jZ_k}\right).
      \]
    \item \emph{Flavor Universality Criterion.}
    Use the result of the preceding two parts to give sufficient
      conditions for generation-universal scalar masses and aligned
      trilinear terms.  Explain which derivatives of the Kähler metric
      threaten flavor conservation.
  \end{enumerate}

  \SupplementaryExercise{10}{Planck-Suppressed Mediation and the Goldstino Direction}
    {(Inspired by Stephen P. Martin, \emph{A Supersymmetry Primer}, version 7, Sections 7 and 12; independently formulated.)}
    {ch31-s10-planck-suppressed-mediation-and-the-goldstino}
  \begin{enumerate}
    \item \emph{Planck-Suppressed Mediation Scale.}
    A hidden sector has \(\sqrt F\ll\kappa^{-1}\).  By dimensional
      analysis determine the scale of visible soft masses generated by
      operators suppressed by one power of the reduced Planck mass.  Relate
      it to the gravitino mass when the cosmological constant is tuned.
    \item \emph{Goldstino Direction.}
    For chiral fermions \(\psi_i\) and gauginos \(\lambda_A\) with vacuum
      auxiliary values \(F_i,D_A\), find the normalized goldstino linear
      combination.  Show that it shifts by a constant under the broken
      supercharge.
  \end{enumerate}

  \SupplementaryExercise{11}{Longitudinal-Gravitino Equivalence and Decay}
    {(Inspired by Stephen P. Martin, \emph{A Supersymmetry Primer}, version 7, Section 12; independently formulated.)}
    {ch31-s11-longitudinal-gravitino-equivalence-and-decay}
  \begin{enumerate}
    \item \emph{Longitudinal-Gravitino Equivalence.}
    For energy \(E\gg m_g\), write the leading longitudinal
      spin-\(3/2\) polarization in terms of \(q_\mu/m_g\) and a spin-\(1/2\)
      wavefunction.  Explain why high-energy amplitudes with longitudinal
      gravitinos equal goldstino amplitudes up to \(O(m_g/E)\).
    \item \emph{Gravitino Decay Scaling.}
    Use the goldstino coupling to estimate the two-body decay width of a
      superpartner of mass \(m\) to its partner plus a light gravitino.
      Determine the dependence on \(m\), \(F\), and \(m_g\).
  \end{enumerate}

  \SupplementaryExercise{12}{Polonyi Stability and Hidden Scales}
    {(Inspired by Stephen P. Martin, \emph{A Supersymmetry Primer}, version 7, Sections 12--13; independently formulated.)}
    {ch31-s12-polonyi-stability-and-hidden-scales}
  \begin{enumerate}
    \item \emph{Polonyi Stability Check.}
    For the model of Weinberg Exercise 6, compute the two real scalar
      squared masses at the Minkowski stationary point.  Verify that both
      are positive and express them in units of the gravitino mass.
    \item \emph{Hidden-Sector Scale Comparison.}
    Compare a hidden sector with \(F=O(\Lambda^2)\) to one whose
      nonperturbative superpotential is \(O(\Lambda^3)\).  Derive the
      characteristic gravity-mediated scales
      \(\kappa\Lambda^2\) and \(\kappa^2\Lambda^3\), and solve for
      \(\Lambda\) when the visible scale is \(1\) TeV.
  \end{enumerate}
