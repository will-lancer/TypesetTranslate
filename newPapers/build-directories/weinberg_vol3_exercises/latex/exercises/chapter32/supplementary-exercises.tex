  \SupplementaryExercise{1}{Clifford Modules, Weyl Projection, and Reality}
    {(Inspired by Antoine Van Proeyen, \emph{Tools for Supersymmetry}, Section 3; independently formulated.)}
    {ch32-s01-clifford-modules-weyl-projection-and-reality}
  \begin{enumerate}
    \item \emph{Fermionic Clifford Module.}
    Let \(a_i,a_i^\dagger\), \(i=1,\ldots,n\), obey
      \(\{a_i,a_j^\dagger\}=\delta_{ij}\) with all other anticommutators
      zero.  Construct their irreducible module and prove that it contains
      \(2^n\) states, equally divided between even and odd fermion number
      for \(n>0\).
    \item \emph{Weyl Projectors in Even Dimension.}
    In \(d=2m\), define a chirality matrix proportional to the ordered
      product of all gamma matrices.  Prove that its square can be
      normalized to one, that it anticommutes with every \(\gamma^\mu\),
      and that the projectors \(P_\pm\) each select half of a Dirac spinor.
    \item \emph{Reality Conditions Modulo Eight.}
    Starting from \(B\gamma^\mu B^{-1}=(\gamma^\mu)^*\), show that
      applying complex conjugation twice gives \(B^*B=\epsilon\).
      Explain why \(\epsilon=+1\) permits a Majorana condition, whereas
      \(\epsilon=-1\) requires a symplectic Majorana doublet.  Apply the
      latter result to a six-dimensional Weyl supercharge.
  \end{enumerate}

  \SupplementaryExercise{2}{Spinor Bilinears and Automorphisms of the Superalgebra}
    {(Inspired by Antoine Van Proeyen, \emph{Tools for Supersymmetry}, Sections 3--4; independently formulated.)}
    {ch32-s02-spinor-bilinears-and-automorphisms-of-the}
  \begin{enumerate}
    \item \emph{Symmetric Gamma-Matrix Bilinears.}
    Suppose \(C^T=\eta_C C\) and
      \(C\gamma^\mu C^{-1}=-(\gamma^\mu)^T\).  Derive the transpose sign of
      \(\gamma^{[\mu_1\cdots\mu_p]}C\).  Explain how this sign decides
      whether a \(p\)-form charge is symmetric or antisymmetric in
      supersymmetry labels.
    \item \emph{Six-Dimensional Automorphism Group.}
    A six-dimensional Weyl spinor is pseudoreal.  Show that a consistent
      reality condition on \(2p\) spinors uses a nondegenerate
      antisymmetric form \(\Omega\), and prove that its compact
      automorphism group is \(USp(2p)\).  Deduce the automorphism group for
      \((p,q)\) supersymmetry.
    \item \emph{Why Supercharges Are Spinors.}
    Revisit the finite-dimensional angular-momentum argument behind the
      supersymmetry extension of the Poincare algebra.  Show that an odd
      generator of Lorentz spin greater than \(1/2\) produces, under
      repeated action on one-particle states, an unacceptable unbounded
      sequence of helicities.  State the assumption about a nontrivial
      scattering matrix used in the conclusion.
  \end{enumerate}

  \SupplementaryExercise{3}{Graviton and Gravitino Counts in Even Dimensions}
    {(Adapted from A.~C. Davis, \emph{Supergravity 1 Example Sheet}, Question 4.)}
    {ch32-s03-massless-little-groups-and-multiplet-counting}
  \begin{enumerate}
    \item \emph{Massless Graviton.}
    In even spacetime dimension \(D\), use the little group \(SO(D-2)\) to
      count the physical polarizations of a massless graviton.
    \item \emph{Massless Gravitino.}
    Let \(s_{D-2}\) denote the dimension, with the appropriate reality and
      chirality condition, of the little-group spinor carried by a minimal
      massless supercharge.  Count the polarizations of a gamma-traceless
      vector-spinor and hence of a massless gravitino.
    \item \emph{The Ten-Dimensional Multiplet.}
    Evaluate the counts in \(D=10\).  Explain why a graviton and gravitino
      alone cannot form a complete supergravity multiplet and identify a
      minimal set of additional bosonic and fermionic field types that
      restores equality.
  \end{enumerate}

  \SupplementaryExercise{4}{A Spin-Three Supersymmetry Bound}
    {(Adapted from University of Cambridge, \emph{Supersymmetry and Extra Dimensions Example Sheet 4}, Problem 8 in Section 4 (Exercise 4.8).)}
    {ch32-s04-the-thirty-two-supercharge-bound}
Imagine that elementary massless particles of every spin through
  \(j=3\) were allowed, but no particle of spin greater than three.
  Determine the largest spacetime dimension compatible with ordinary
  supersymmetry.  Relate the real size of a minimal higher-dimensional
  spinor to the helicity span of the four-dimensional multiplet obtained by
  dimensional reduction.

  \SupplementaryExercise{5}{Type-II and Eleven-Dimensional Reductions}
    {(Adapted from University of Cambridge, \emph{Supersymmetry and Extra Dimensions Example Sheet 4}, Problem 9 in Section 4 (Exercise 4.9).)}
    {ch32-s05-eleven-dimensional-and-type-ii-supergravity}
  \begin{enumerate}
    \item \emph{Type IIA and IIB from Ten to Nine Dimensions.}
    Reduce the complete bosonic and fermionic massless field content of each
      ten-dimensional type-II supergravity on a circle.  Count the
      nine-dimensional polarizations field by field, show that both reductions
      give the same \(128_B+128_F\) maximal multiplet, and decide whether the
      nine-dimensional spectrum is chiral.
    \item \emph{Direct Reduction from Eleven Dimensions.}
    Compactify eleven-dimensional supergravity on a two-torus directly to
      nine dimensions.  Decompose the metric, three-form, and gravitino and
      compare the result with the common type-II spectrum from part (a).
    \item \emph{Type IIB from Ten to Four Dimensions.}
    Reduce the type-IIB massless multiplet on a six-torus.  Without discarding
      any zero modes, organize the result into the four-dimensional maximal
      supergravity multiplet and verify its bosonic and fermionic
      polarization counts.
  \end{enumerate}

  \SupplementaryExercise{6}{Kaluza--Klein BPS Towers and the Eleventh Dimension}
    {(Inspired by Edward Witten, ``String Theory Dynamics in Various Dimensions'', Sections 1--2; independently formulated.)}
    {ch32-s06-kaluza-klein-bps-towers-and-the}
  \begin{enumerate}
    \item \emph{Kaluza--Klein Momentum as Central Charge.}
    Reduce the eleven-dimensional super-Poincare algebra on a circle.
      Show that \(P_{10}\) is a Lorentz scalar in ten dimensions and occurs
      as a scalar central charge in the type-IIA algebra.
    \item \emph{The Kaluza--Klein BPS Tower.}
    For a circle of radius \(R\), use momentum quantization and the
      ten-dimensional BPS inequality to find the masses of states with
      integer Kaluza--Klein number \(n\).  Explain why each nonzero level
      forms a \(256\)-state short multiplet.
    \item \emph{The Emerging Eleventh Dimension.}
    Assume the type-IIA D0-brane mass scales as
      \(1/(g_s\ell_s)\).  Identifying it with the first Kaluza--Klein mass,
      derive \(R_{11}=g_s\ell_s\).  Explain what happens to the
      eleven-dimensional radius at strong coupling.
  \end{enumerate}

  \SupplementaryExercise{7}{Type-II Chirality and Dimensional Reduction}
    {(Inspired by Edward Witten, ``String Theory Dynamics in Various Dimensions'', Sections 1--3; independently formulated.)}
    {ch32-s07-type-ii-chirality-and-dimensional-reduction}
  \begin{enumerate}
    \item \emph{IIA and IIB Chirality.}
    Decompose the eleven-dimensional Majorana supercharge into
      ten-dimensional Weyl components and show that they have opposite
      chirality.  Contrast this with type IIB and explain why IIB cannot be
      obtained by an ordinary circle reduction of eleven-dimensional
      supergravity.
    \item \emph{Reducing the Eleven-Dimensional Three-Form.}
    Decompose an eleven-dimensional three-form potential under circle
      reduction.  Identify the ten-dimensional three-form and two-form,
      and similarly decompose the two-form and five-form charges in the
      eleven-dimensional supersymmetry algebra.
    \item \emph{Nine-Dimensional Unification.}
    Reduce both type-IIA and type-IIB supersymmetry to nine dimensions.
      Show that ten-dimensional chirality ceases to distinguish the
      supercharges and that the two algebras become equivalent after a
      change of basis.  Relate this fact to T-duality.
  \end{enumerate}

  \SupplementaryExercise{8}{Tensorial Charges and the \(528\)-Component M-Algebra}
    {(Inspired by Paul K. Townsend, ``\(p\)-Brane Democracy'', Sections 1--3; independently formulated.)}
    {ch32-s08-tensorial-charges-and-the-component-m}
  \begin{enumerate}
    \item \emph{Topological \(p\)-Brane Charge.}
    For a static \(p\)-brane described asymptotically by coordinates
      \(X^{\mu_1},\ldots,X^{\mu_p}\), consider the spatial integral of the
      pullback \(dX^{\mu_1}\wedge\cdots\wedge dX^{\mu_p}\).  Show that its
      variation is a boundary term and explain when it defines a
      topological tensor charge.
    \item \emph{Completeness of the \(p\)-Form Basis.}
    Prove that antisymmetrized gamma matrices form a basis of the full
      spinor-matrix algebra, with Hodge duality removing redundant ranks.
      Explain why the symmetric subset exhausts all possible bosonic terms
      in an anticommutator of two supercharges.
    \item \emph{The \(528\) Components of the M-Algebra.}
    A single eleven-dimensional Majorana supercharge has \(32\) real
      components.  Count the entries of its symmetric anticommutator and
      show that they decompose into an eleven-vector, a two-form, and a
      five-form.
  \end{enumerate}

  \SupplementaryExercise{9}{Dual Charges, Brane Couplings, and M2/M5 Projectors}
    {(Inspired by Paul K. Townsend, ``\(p\)-Brane Democracy'', Sections 2--4; independently formulated.)}
    {ch32-s09-dual-charges-brane-couplings-and-m2}
  \begin{enumerate}
    \item \emph{Hodge-Dual Charge Redundancy.}
    Derive the gamma-matrix relation between a rank-\(p\) matrix and its
      rank-\((d-p)\) Hodge dual.  Use it to explain why listing both a
      \(p\)-form and its dual generally double counts central extensions.
    \item \emph{Which Potential Couples to a Brane?.}
    Starting from reparametrization invariance of a \(p\)-dimensional
      spatial brane, construct its minimal Wess--Zumino coupling.  Show
      that an electrically charged \(p\)-brane couples to a
      \((p+1)\)-form gauge potential and identify the rank of its
      supersymmetry charge.
    \item \emph{M2 and M5 BPS Projectors.}
    For a planar M2 along \(x^1,x^2\), show that saturation of its
      energy-charge bound imposes
      \(\Gamma^{012}\epsilon=\pm\epsilon\).  Establish the analogous
      condition for an M5 along \(x^1,\ldots,x^5\), and prove that each
      projector preserves sixteen supercharges.
  \end{enumerate}

  \SupplementaryExercise{10}{Dp-Brane Spectra and Type-II Parity}
    {(Inspired by Joseph Polchinski, \emph{TASI Lectures on D-Branes}, Sections 1--3; independently formulated.)}
    {ch32-s10-dp-brane-spectra-and-type-ii}
  \begin{enumerate}
    \item \emph{Dp-Brane Worldvolume Degrees of Freedom.}
    For \(1\leq p\leq9\), the massless open-string modes on one Dp-brane
      consist of a worldvolume gauge field and transverse scalars.  Count
      their bosonic on-shell degrees of freedom and show that the answer is
      eight, independent of \(p\).  State why the \(p=0\) worldline case
      requires a separate constrained-Hamiltonian interpretation.
    \item \emph{Even and Odd D-Branes.}
    Use the kappa-symmetry or supersymmetry projector for a Dp-brane to
      determine how it relates the two ten-dimensional supercharges.
      Explain why type IIA admits even \(p\) and type IIB admits odd \(p\).
  \end{enumerate}

  \SupplementaryExercise{11}{Open-String Vector Multiplets and No Force}
    {(Inspired by Joseph Polchinski, \emph{TASI Lectures on D-Branes}, Sections 1--3; independently formulated.)}
    {ch32-s11-open-string-vector-multiplets-and-no}
  \begin{enumerate}
    \item \emph{BPS Tension, Charge, and No Force.}
    Use a central-charge inequality to explain why a supersymmetric
      D-brane has tension equal to the magnitude of its Ramond--Ramond
      charge in suitable units.  Show qualitatively why the gravitational
      and form-field forces between two parallel identical branes cancel.
    \item \emph{The Open-String Vector Multiplet.}
    Count the massless Ramond-sector degrees of freedom on a single
      Dp-brane after the GSO projection.  Show that they match the eight
      bosonic degrees of freedom and identify the resulting maximally
      supersymmetric worldvolume vector multiplet.
  \end{enumerate}

  \SupplementaryExercise{12}{T-Duality and Threshold Bounds}
    {(Inspired by Joseph Polchinski, \emph{TASI Lectures on D-Branes}, Sections 3 and 5; independently formulated.)}
    {ch32-s12-t-duality-and-threshold-bounds}
  \begin{enumerate}
    \item \emph{T-Duality Changes Brane Dimension.}
    Apply T-duality to an open string coordinate parallel and then
      transverse to a Dp-brane.  Track Neumann and Dirichlet boundary
      conditions and prove that the brane becomes respectively a
      D\((p-1)\)- or D\((p+1)\)-brane.
    \item \emph{Brane Threshold Bound.}
    Consider a state carrying several mutually compatible brane charges.
      Diagonalize the corresponding central-charge matrix to obtain its
      mass bound.  Explain the difference between a threshold marginal
      bound state and a nonthreshold BPS bound state in terms of the
      charge eigenvalues and binding energy.
  \end{enumerate}

\end{enumerate}
