\chapterbackmatter{Supplementary Exercises}

\begin{enumerate}
  \SupplementaryExercise{1}{Fermionic Clifford Module}
    {(Adapted from Van Proeyen, \emph{Tools for Supersymmetry},
    Sections 3--4)}
    {van-proeyen-tools-susy-ch32}
  Let \(a_i,a_i^\dagger\), \(i=1,\ldots,n\), obey
  \(\{a_i,a_j^\dagger\}=\delta_{ij}\) with all other anticommutators
  zero.  Construct their irreducible module and prove that it contains
  \(2^n\) states, equally divided between even and odd fermion number
  for \(n>0\).

  \SupplementaryExercise{2}{Weyl Projectors in Even Dimension}
    {(Adapted from Van Proeyen, \emph{Tools for Supersymmetry},
    Section 3)}
    {van-proeyen-tools-susy-ch32}
  In \(d=2m\), define a chirality matrix proportional to the ordered
  product of all gamma matrices.  Prove that its square can be
  normalized to one, that it anticommutes with every \(\gamma^\mu\),
  and that the projectors \(P_\pm\) each select half of a Dirac spinor.

  \SupplementaryExercise{3}{Reality Conditions Modulo Eight}
    {(Adapted from Van Proeyen, \emph{Tools for Supersymmetry},
    Section 3)}
    {van-proeyen-tools-susy-ch32}
  Starting from \(B\gamma^\mu B^{-1}=(\gamma^\mu)^*\), show that
  applying complex conjugation twice gives \(B^*B=\epsilon\).
  Explain why \(\epsilon=+1\) permits a Majorana condition, whereas
  \(\epsilon=-1\) requires a symplectic Majorana doublet.  Apply the
  latter result to a six-dimensional Weyl supercharge.

  \SupplementaryExercise{4}{Symmetric Gamma-Matrix Bilinears}
    {(Adapted from Van Proeyen, \emph{Tools for Supersymmetry},
    Sections 3--4)}
    {van-proeyen-tools-susy-ch32}
  Suppose \(C^T=\eta_C C\) and
  \(C\gamma^\mu C^{-1}=-(\gamma^\mu)^T\).  Derive the transpose sign of
  \(\gamma^{[\mu_1\cdots\mu_p]}C\).  Explain how this sign decides
  whether a \(p\)-form charge is symmetric or antisymmetric in
  supersymmetry labels.

  \SupplementaryExercise{5}{Six-Dimensional Automorphism Group}
    {(Adapted from Van Proeyen, \emph{Tools for Supersymmetry},
    Sections 3--4)}
    {van-proeyen-tools-susy-ch32}
  A six-dimensional Weyl spinor is pseudoreal.  Show that a consistent
  reality condition on \(2p\) spinors uses a nondegenerate
  antisymmetric form \(\Omega\), and prove that its compact
  automorphism group is \(USp(2p)\).  Deduce the automorphism group for
  \((p,q)\) supersymmetry.

  \SupplementaryExercise{6}{Why Supercharges Are Spinors}
    {(Adapted from Van Proeyen, \emph{Tools for Supersymmetry},
    Section 4)}
    {van-proeyen-tools-susy-ch32}
  Revisit the finite-dimensional angular-momentum argument behind the
  supersymmetry extension of the Poincare algebra.  Show that an odd
  generator of Lorentz spin greater than \(1/2\) produces, under
  repeated action on one-particle states, an unacceptable unbounded
  sequence of helicities.  State the assumption about a nontrivial
  scattering matrix used in the conclusion.

  \SupplementaryExercise{7}{Null-Momentum Stabilizer}
    {(Adapted from de Wit, \emph{Supergravity}, Sections 2--4)}
    {de-wit-supergravity-2002-ch32}
  Fix a standard null momentum in \(d\) dimensions.  Determine its
  Lorentz stabilizer and show that finite-spin unitary particle
  representations make the translational part act trivially, leaving
  \(SO(d-2)\) as the physical little group.

  \SupplementaryExercise{8}{Binomial Multiplet Counting}
    {(Adapted from de Wit, \emph{Supergravity}, Sections 2--4)}
    {de-wit-supergravity-2002-ch32}
  A massless algebra with \({\cal N}\) real supercharge components has
  \({\cal N}/4\) fermionic creators.  Show that the level with \(r\)
  creators has multiplicity \(\binom{{\cal N}/4}{r}\), and verify
  equality of total bosonic and fermionic state counts.

  \SupplementaryExercise{9}{The Thirty-Two-Supercharge Bound}
    {(Adapted from de Wit, \emph{Supergravity}, Sections 2--4)}
    {de-wit-supergravity-2002-ch32}
  For a massless multiplet extending from helicity \(+2\) to no lower
  than \(-2\), derive the bound \({\cal N}\leq32\) on real supercharge
  components.  Explain why a theory with an ordinary graviton cannot
  evade the bound merely by choosing a different Clifford vacuum.

  \SupplementaryExercise{10}{Eleven-Dimensional State Count}
    {(Adapted from de Wit, \emph{Supergravity}, Sections 4 and 6)}
    {de-wit-supergravity-2002-ch32}
  Under the eleven-dimensional massless little group \(SO(9)\), count
  the components of a symmetric traceless tensor, a three-form, and a
  gamma-traceless vector-spinor.  Verify the supergravity balance
  \(44+84=128\).

  \SupplementaryExercise{11}{Branching from \(SO(9)\) to \(SO(8)\)}
    {(Adapted from de Wit, \emph{Supergravity}, Sections 4 and 6)}
    {de-wit-supergravity-2002-ch32}
  Compactify one spatial direction.  Decompose the \(SO(9)\)
  representations \(\mathbf{44}\), \(\mathbf{84}\), and
  \(\mathbf{128}\) into \(SO(8)\) representations, and identify the
  resulting type-IIA graviton, forms, gravitini, dilatino, and scalar.

  \SupplementaryExercise{12}{Type-IIB Polarization Balance}
    {(Adapted from de Wit, \emph{Supergravity}, Sections 4 and 6)}
    {de-wit-supergravity-2002-ch32}
  Count the physical polarizations of the ten-dimensional type-IIB
  graviton, two gravitini, two two-forms, two spin-\(1/2\) fields, two
  scalars, and a self-dual four-form.  Verify \(128_B=128_F\).

  \SupplementaryExercise{13}{Kaluza--Klein Momentum as Central Charge}
    {(Adapted from Witten, \emph{String Theory Dynamics in Various
    Dimensions}, Sections 1--2)}
    {witten-string-dynamics-1995-ch32}
  Reduce the eleven-dimensional super-Poincare algebra on a circle.
  Show that \(P_{10}\) is a Lorentz scalar in ten dimensions and occurs
  as a scalar central charge in the type-IIA algebra.

  \SupplementaryExercise{14}{The Kaluza--Klein BPS Tower}
    {(Adapted from Witten, \emph{String Theory Dynamics in Various
    Dimensions}, Sections 1--2)}
    {witten-string-dynamics-1995-ch32}
  For a circle of radius \(R\), use momentum quantization and the
  ten-dimensional BPS inequality to find the masses of states with
  integer Kaluza--Klein number \(n\).  Explain why each nonzero level
  forms a \(256\)-state short multiplet.

  \SupplementaryExercise{15}{IIA and IIB Chirality}
    {(Adapted from Witten, \emph{String Theory Dynamics in Various
    Dimensions}, Sections 1--3)}
    {witten-string-dynamics-1995-ch32}
  Decompose the eleven-dimensional Majorana supercharge into
  ten-dimensional Weyl components and show that they have opposite
  chirality.  Contrast this with type IIB and explain why IIB cannot be
  obtained by an ordinary circle reduction of eleven-dimensional
  supergravity.

  \SupplementaryExercise{16}{The Emerging Eleventh Dimension}
    {(Adapted from Witten, \emph{String Theory Dynamics in Various
    Dimensions}, Sections 1--2)}
    {witten-string-dynamics-1995-ch32}
  Assume the type-IIA D0-brane mass scales as
  \(1/(g_s\ell_s)\).  Identifying it with the first Kaluza--Klein mass,
  derive \(R_{11}=g_s\ell_s\).  Explain what happens to the
  eleven-dimensional radius at strong coupling.

  \SupplementaryExercise{17}{Reducing the Eleven-Dimensional Three-Form}
    {(Adapted from Witten, \emph{String Theory Dynamics in Various
    Dimensions}, Sections 1--3)}
    {witten-string-dynamics-1995-ch32}
  Decompose an eleven-dimensional three-form potential under circle
  reduction.  Identify the ten-dimensional three-form and two-form,
  and similarly decompose the two-form and five-form charges in the
  eleven-dimensional supersymmetry algebra.

  \SupplementaryExercise{18}{Nine-Dimensional Unification}
    {(Adapted from Witten, \emph{String Theory Dynamics in Various
    Dimensions}, Sections 2--3)}
    {witten-string-dynamics-1995-ch32}
  Reduce both type-IIA and type-IIB supersymmetry to nine dimensions.
  Show that ten-dimensional chirality ceases to distinguish the
  supercharges and that the two algebras become equivalent after a
  change of basis.  Relate this fact to T-duality.

  \SupplementaryExercise{19}{Topological \(p\)-Brane Charge}
    {(Adapted from Townsend, \emph{\(p\)-Brane Democracy},
    Sections 1--2)}
    {townsend-p-brane-democracy-ch32}
  For a static \(p\)-brane described asymptotically by coordinates
  \(X^{\mu_1},\ldots,X^{\mu_p}\), consider the spatial integral of the
  pullback \(dX^{\mu_1}\wedge\cdots\wedge dX^{\mu_p}\).  Show that its
  variation is a boundary term and explain when it defines a
  topological tensor charge.

  \SupplementaryExercise{20}{Completeness of the \(p\)-Form Basis}
    {(Adapted from Townsend, \emph{\(p\)-Brane Democracy},
    Sections 1--3)}
    {townsend-p-brane-democracy-ch32}
  Prove that antisymmetrized gamma matrices form a basis of the full
  spinor-matrix algebra, with Hodge duality removing redundant ranks.
  Explain why the symmetric subset exhausts all possible bosonic terms
  in an anticommutator of two supercharges.

  \SupplementaryExercise{21}{The \(528\) Components of the M-Algebra}
    {(Adapted from Townsend, \emph{\(p\)-Brane Democracy},
    Sections 1--3)}
    {townsend-p-brane-democracy-ch32}
  A single eleven-dimensional Majorana supercharge has \(32\) real
  components.  Count the entries of its symmetric anticommutator and
  show that they decompose into an eleven-vector, a two-form, and a
  five-form.

  \SupplementaryExercise{22}{Hodge-Dual Charge Redundancy}
    {(Adapted from Townsend, \emph{\(p\)-Brane Democracy},
    Sections 2--3)}
    {townsend-p-brane-democracy-ch32}
  Derive the gamma-matrix relation between a rank-\(p\) matrix and its
  rank-\((d-p)\) Hodge dual.  Use it to explain why listing both a
  \(p\)-form and its dual generally double counts central extensions.

  \SupplementaryExercise{23}{Which Potential Couples to a Brane?}
    {(Adapted from Townsend, \emph{\(p\)-Brane Democracy},
    Sections 2--4)}
    {townsend-p-brane-democracy-ch32}
  Starting from reparametrization invariance of a \(p\)-dimensional
  spatial brane, construct its minimal Wess--Zumino coupling.  Show
  that an electrically charged \(p\)-brane couples to a
  \((p+1)\)-form gauge potential and identify the rank of its
  supersymmetry charge.

  \SupplementaryExercise{24}{M2 and M5 BPS Projectors}
    {(Adapted from Townsend, \emph{\(p\)-Brane Democracy},
    Sections 2--4)}
    {townsend-p-brane-democracy-ch32}
  For a planar M2 along \(x^1,x^2\), show that saturation of its
  energy-charge bound imposes
  \(\Gamma^{012}\epsilon=\pm\epsilon\).  Establish the analogous
  condition for an M5 along \(x^1,\ldots,x^5\), and prove that each
  projector preserves sixteen supercharges.

  \SupplementaryExercise{25}{Dp-Brane Worldvolume Degrees of Freedom}
    {(Adapted from Polchinski, \emph{TASI Lectures on D-Branes},
    Sections 1--3)}
    {polchinski-tasi-d-branes-ch32}
  For \(1\leq p\leq9\), the massless open-string modes on one Dp-brane
  consist of a worldvolume gauge field and transverse scalars.  Count
  their bosonic on-shell degrees of freedom and show that the answer is
  eight, independent of \(p\).  State why the \(p=0\) worldline case
  requires a separate constrained-Hamiltonian interpretation.

  \SupplementaryExercise{26}{BPS Tension, Charge, and No Force}
    {(Adapted from Polchinski, \emph{TASI Lectures on D-Branes},
    Sections 2--3)}
    {polchinski-tasi-d-branes-ch32}
  Use a central-charge inequality to explain why a supersymmetric
  D-brane has tension equal to the magnitude of its Ramond--Ramond
  charge in suitable units.  Show qualitatively why the gravitational
  and form-field forces between two parallel identical branes cancel.

  \SupplementaryExercise{27}{Even and Odd D-Branes}
    {(Adapted from Polchinski, \emph{TASI Lectures on D-Branes},
    Sections 2--3)}
    {polchinski-tasi-d-branes-ch32}
  Use the kappa-symmetry or supersymmetry projector for a Dp-brane to
  determine how it relates the two ten-dimensional supercharges.
  Explain why type IIA admits even \(p\) and type IIB admits odd \(p\).

  \SupplementaryExercise{28}{The Open-String Vector Multiplet}
    {(Adapted from Polchinski, \emph{TASI Lectures on D-Branes},
    Sections 1--3)}
    {polchinski-tasi-d-branes-ch32}
  Count the massless Ramond-sector degrees of freedom on a single
  Dp-brane after the GSO projection.  Show that they match the eight
  bosonic degrees of freedom and identify the resulting maximally
  supersymmetric worldvolume vector multiplet.

  \SupplementaryExercise{29}{T-Duality Changes Brane Dimension}
    {(Adapted from Polchinski, \emph{TASI Lectures on D-Branes},
    Sections 2--3)}
    {polchinski-tasi-d-branes-ch32}
  Apply T-duality to an open string coordinate parallel and then
  transverse to a Dp-brane.  Track Neumann and Dirichlet boundary
  conditions and prove that the brane becomes respectively a
  D\((p-1)\)- or D\((p+1)\)-brane.

  \SupplementaryExercise{30}{Brane Threshold Bound}
    {(Adapted from Polchinski, \emph{TASI Lectures on D-Branes},
    Sections 2--3 and 5)}
    {polchinski-tasi-d-branes-ch32}
  Consider a state carrying several mutually compatible brane charges.
  Diagonalize the corresponding central-charge matrix to obtain its
  mass bound.  Explain the difference between a threshold marginal
  bound state and a nonthreshold BPS bound state in terms of the
  charge eigenvalues and binding energy.
\end{enumerate}
