  \SupplementaryExercise{1}{Axion--Kalb--Ramond Duality from a Parent Action}
    {(Adapted from University of Cambridge, \emph{Supersymmetry and Extra
      Dimensions Example Sheet 4}, Problem 2 in Section 4 (Exercise 4.2).)}
    {cambridge-susy-extra-sheet4-ex4p2-v3ch32-s01}
  In four dimensions let \(H_{\mu\nu\rho}\) be an independent real
  three-form and let \(a\) be a real Lagrange multiplier.  Consider the
  source-normalized parent action
  \[
    S=\int d^4x\left[
      \frac1{g^2}H_{\mu\nu\rho}H^{\mu\nu\rho}
      +a\,\epsilon^{\mu\nu\rho\sigma}\partial_\mu H_{\nu\rho\sigma}
      \right].
  \]
  Overall Lorentzian signs and factorials may equivalently be fixed by using
  \(S=-\frac1{2g^2}\int H\wedge *H+\int a\,dH\); use one normalization
  consistently.
  \begin{enumerate}
    \item Eliminate \(a\).  Solve its equation locally, introduce the
      massless Kalb--Ramond potential \(B_{\mu\nu}\), and obtain the
      two-form action including its gauge invariance.
    \item Instead eliminate \(H\) and derive the propagating axion action.
      Determine what happens to the coupling \(g\) under the duality.
    \item Generalize the parent-action construction to a massless
      \(p\)-form potential in arbitrary dimension \(D\).  State the rank
      of its dual potential and the map between the two field strengths.
  \end{enumerate}

  \SupplementaryExercise{2}{Massive Antisymmetric-Tensor Duality}
    {(Adapted from University of Cambridge, \emph{Supersymmetry and Extra
      Dimensions Example Sheet 4}, Problem 3 in Section 4 (Exercise 4.3).)}
    {cambridge-susy-extra-sheet4-ex4p3-v3ch32-s02}
  Let \(A_q\) be a real massive antisymmetric tensor of rank \(q\) in
  \(D\) dimensions, with \(F_{q+1}=dA_q\).
  \begin{enumerate}
    \item Write its local Lagrangian containing at most two derivatives,
      including the kinetic and mass terms and all factorials in component
      notation.
    \item Introduce a \((D-q-1)\)-form \(B\) and construct a first-order
      parent Lagrangian that yields the original theory when \(B\) is
      eliminated and its massive dual when \(A\) is eliminated.  Display
      both elimination steps and the field-strength duality relation.
    \item Count the propagating degrees of freedom on both sides and prove
      that the counts agree.
    \item Interpret the parent functional integral as a functional Fourier
      transform.  Analyze the limit \(m\to0\): explain what remains valid,
      what becomes singular in the massive derivation, and why the
      massless dual potential has a different rank.
  \end{enumerate}

  \SupplementaryExercise{3}{Graviton and Gravitino Counts in Even Dimensions}
    {(Adapted from A.~C. Davis, \emph{Supergravity 1 Example Sheet}, Question 4.)}
    {ch32-s03-massless-little-groups-and-multiplet-counting}
  \begin{enumerate}
    \item \emph{Massless Graviton.}
    In even spacetime dimension \(D\), use the little group \(SO(D-2)\) to
      count the physical polarizations of a massless graviton.
    \item \emph{Massless Gravitino.}
    Let \(s_{D-2}\) denote the real physical dimension, after the appropriate
      reality and chirality condition, of the little-group spinor carried by
      a minimal massless supercharge.  Count the polarizations of a
      gamma-traceless vector-spinor and hence of a massless gravitino.
    \item \emph{The Ten-Dimensional Multiplet.}
    Evaluate the counts in \(D=10\).  Explain why a graviton and gravitino
      alone cannot form a complete supergravity multiplet and identify a
      minimal set of additional bosonic and fermionic field types that
      restores equality.
  \end{enumerate}

  \SupplementaryExercise{4}{A Spin-Three Supersymmetry Bound}
    {(Adapted from University of Cambridge, \emph{Supersymmetry and Extra Dimensions Example Sheet 4}, Problem 8 in Section 4 (Exercise 4.8).)}
    {ch32-s04-the-thirty-two-supercharge-bound}
Imagine that elementary massless particles of every spin through
  \(j=3\) were allowed, but no particle of spin greater than three.
  Determine the largest spacetime dimension compatible with ordinary
  supersymmetry.  Relate the real size of a minimal higher-dimensional
  spinor to the helicity span of the four-dimensional multiplet obtained by
  dimensional reduction.

  \SupplementaryExercise{5}{Type-II and Eleven-Dimensional Reductions}
    {(Adapted from University of Cambridge, \emph{Supersymmetry and Extra Dimensions Example Sheet 4}, Problem 9 in Section 4 (Exercise 4.9).)}
    {ch32-s05-eleven-dimensional-and-type-ii-supergravity}
  \begin{enumerate}
    \item \emph{Type IIA and IIB from Ten to Nine Dimensions.}
    Reduce the complete bosonic and fermionic massless field content of each
      ten-dimensional type-II supergravity on a circle.  Count the
      nine-dimensional polarizations field by field, show that both reductions
      give the same \(128_B+128_F\) maximal multiplet, and decide whether the
      nine-dimensional spectrum is chiral.
    \item \emph{Direct Reduction from Eleven Dimensions.}
    Compactify eleven-dimensional supergravity on a two-torus directly to
      nine dimensions.  Decompose the metric, three-form, and gravitino and
      compare the result with the common type-II spectrum from part (a).
    \item \emph{Type IIB from Ten to Four Dimensions.}
    Reduce the type-IIB massless multiplet on a six-torus.  Without discarding
      any zero modes, organize the result into the four-dimensional maximal
      supergravity multiplet and verify its bosonic and fermionic
      polarization counts.
  \end{enumerate}

  \SupplementaryExercise{6}{Kaluza--Klein BPS Towers and the Eleventh Dimension}
    {(Inspired by Edward Witten, ``String Theory Dynamics in Various Dimensions'', Sections 1--2; independently formulated.)}
    {ch32-s06-kaluza-klein-bps-towers-and-the}
  \begin{enumerate}
    \item \emph{Kaluza--Klein Momentum as Central Charge.}
    Reduce the eleven-dimensional super-Poincare algebra on a circle.
      Show that \(P_{10}\) is a Lorentz scalar in ten dimensions and occurs
      as a scalar central charge in the type-IIA algebra.
    \item \emph{The Kaluza--Klein BPS Tower.}
    For a circle of radius \(R\), use momentum quantization and the
      ten-dimensional BPS inequality to find the masses of states with
      integer Kaluza--Klein number \(n\).  Explain why each nonzero level
      forms a \(256\)-state short multiplet.
    \item \emph{The Emerging Eleventh Dimension.}
    Assume the type-IIA D0-brane mass scales as
      \(1/(g_s\ell_s)\).  Identifying it with the first Kaluza--Klein mass,
      derive \(R_{11}=g_s\ell_s\).  Explain what happens to the
      eleven-dimensional radius at strong coupling.
  \end{enumerate}

  \SupplementaryExercise{7}{Type-II Chirality and Dimensional Reduction}
    {(Inspired by Edward Witten, ``String Theory Dynamics in Various Dimensions'', Sections 1--3; independently formulated.)}
    {ch32-s07-type-ii-chirality-and-dimensional-reduction}
  \begin{enumerate}
    \item \emph{IIA and IIB Chirality.}
    Decompose the eleven-dimensional Majorana supercharge into
      ten-dimensional Weyl components and show that they have opposite
      chirality.  Contrast this with type IIB and explain why IIB cannot be
      obtained by an ordinary circle reduction of eleven-dimensional
      supergravity.
    \item \emph{Reducing the Eleven-Dimensional Three-Form.}
    Decompose an eleven-dimensional three-form potential under circle
      reduction.  Identify the ten-dimensional three-form and two-form,
      and similarly decompose the two-form and five-form charges in the
      eleven-dimensional supersymmetry algebra.
    \item \emph{Nine-Dimensional Unification.}
    Reduce both type-IIA and type-IIB supersymmetry to nine dimensions.
      Show that ten-dimensional chirality ceases to distinguish the
      supercharges and that the two algebras become equivalent after a
      change of basis.  Relate this fact to T-duality.
  \end{enumerate}

  \SupplementaryExercise{8}{Tensorial Charges and the \(528\)-Component M-Algebra}
    {(Inspired by Paul K. Townsend, ``\(p\)-Brane Democracy'', Sections 1--3; independently formulated.)}
    {ch32-s08-tensorial-charges-and-the-component-m}
  \begin{enumerate}
    \item \emph{Topological \(p\)-Brane Charge.}
    For a static \(p\)-brane described asymptotically by coordinates
      \(X^{\mu_1},\ldots,X^{\mu_p}\), consider the spatial integral of the
      pullback \(dX^{\mu_1}\wedge\cdots\wedge dX^{\mu_p}\).  Show that its
      variation is a boundary term and explain when it defines a
      topological tensor charge.
    \item \emph{Completeness of the \(p\)-Form Basis.}
    Prove that antisymmetrized gamma matrices form a basis of the full
      spinor-matrix algebra, with Hodge duality removing redundant ranks.
      Explain why the symmetric subset exhausts all possible bosonic terms
      in an anticommutator of two supercharges.
    \item \emph{The \(528\) Components of the M-Algebra.}
    A single eleven-dimensional Majorana supercharge has \(32\) real
      components.  Count the entries of its symmetric anticommutator and
      show that they decompose into an eleven-vector, a two-form, and a
      five-form.
  \end{enumerate}

  \SupplementaryExercise{9}{Self-Dual Tensor Fields and Their Branes}
    {(Adapted from University of Cambridge, \emph{Supersymmetry and Extra
      Dimensions Example Sheet 4}, Problem 4 in Section 4 (Exercise 4.4).)}
    {cambridge-susy-extra-sheet4-ex4p4-v3ch32-s09}
  Let \(F_n\) be a real middle-degree field strength in even spacetime
  dimension \(D=2n\).
  \begin{enumerate}
    \item In Lorentzian signature compute \(*^2\) on \(n\)-forms and show
      that a real condition \(F_n=\pm *F_n\) is possible only for
      \(D=4k+2\).
    \item Count the on-shell degrees of freedom of such a self-dual field.
      Make clear which unconstrained massless potential supplies the count
      before self-duality is imposed.
    \item Determine the spatial dimension \(p\) of an electrically charged
      brane coupled to the corresponding potential, and explain why the
      electric and magnetic branes have the same dimension.
    \item Repeat the Hodge-star argument in Euclidean signature.  State in
      which dimensions real self-duality is possible and what replaces it
      in the remaining even dimensions.
  \end{enumerate}

  \SupplementaryExercise{10}{Kaluza--Klein Reduction of Five-Dimensional Gravity}
    {(Adapted from University of Cambridge, \emph{Supersymmetry and Extra
      Dimensions Example Sheet 4}, Problem 5 in Section 4 (Exercise 4.5).)}
    {cambridge-susy-extra-sheet4-ex4p5-v3ch32-s10}
  Let \(y\) be a periodic fifth coordinate and impose the cylinder condition
  that all fields are independent of \(y\).  Use the Kaluza--Klein metric
  \[
    ds_5^2=g_{\mu\nu}dx^\mu dx^\nu
      +e^{2\sigma}(dy+A_\mu dx^\mu)^2,
    \qquad G_{55}=e^{2\sigma},
  \]
  together with
  \[
    {}^{(5)}R={}^{(4)}R-2e^{-\sigma}\nabla^2e^\sigma
      -\frac14e^{2\sigma}F_{\mu\nu}F^{\mu\nu},
    \qquad F_{\mu\nu}=\partial_\mu A_\nu-\partial_\nu A_\mu,
  \]
  to reduce the pure five-dimensional Einstein action to four dimensions.
  Display both the Jordan-frame action and its Einstein-frame form,
  including the radion and gauge-field terms.  Finally, show that the
  four-dimensional gauge transformation of \(A_\mu\) is the local form of
  the \(U(1)\) isometry of the compact circle.

  \SupplementaryExercise{11}{Spheres, Higher-Dimensional Potentials, and the Planck Scale}
    {(Adapted from University of Cambridge, 2015 Part III Paper 48,
      \emph{Supersymmetry and Extra Dimensions}, Question 3.)}
    {cambridge-susy-extra-exam-2015-q3-v3ch32-s11}
  \begin{enumerate}
    \item Prove that the area of an \((N-1)\)-sphere of radius \(r\) is
      \[
        V_{N-1}=\frac{2\pi^{N/2}}{\Gamma(N/2)}r^{N-1}.
      \]
      As a hint, evaluate
      \[
        I_N=\int d^Nx\,e^{-\rho^2},
        \qquad \rho^2=\sum_{i=1}^N x_i^2,
      \]
      once in Cartesian coordinates and once in spherical coordinates.
    \item Use this result to derive the radial electric and gravitational
      potentials in \(D\) spacetime dimensions.
    \item If the \(n\) extra dimensions form an \(n\)-sphere of radius
      \(R\), express the four-dimensional reduced Planck scale
      \(M_{\rm Pl}\) in terms of the \((n+4)\)-dimensional scale \(M_*\).
      For \(M_*=1\,\mathrm{TeV}\), estimate \(R\) for
      \(n=1,2,6\).
    \item Write the image sum for the potential of a point source when the
      fifth dimension is a circle.  Show that at distances much larger than
      the circle it reduces to a four-dimensional \(1/r\) potential.
  \end{enumerate}

  \SupplementaryExercise{12}{Warped Compactification and the Hierarchy}
    {(Adapted from University of Cambridge, \emph{Supersymmetry and Extra
      Dimensions Example Sheet 4}, Problem 7 in Section 4 (Exercise 4.7).)}
    {cambridge-susy-extra-sheet4-ex4p7-v3ch32-s12}
  Consider five-dimensional gravity with cosmological constant \(\Lambda<0\),
  compactified on an interval \(0\leq\theta\leq\pi\).  The endpoints are
  three-branes with tensions \(\pm\Lambda/k\).  Let
  \[
    ds^2=e^{-2A(\theta)}\eta_{\mu\nu}dx^\mu dx^\nu-r^2d\theta^2,
  \]
  where \(r\) is constant.  You may use the reduced Einstein equations
  \[
    \frac{6A'^2}{r^2}=-\frac{\Lambda}{2M_*^3},
    \qquad
    \frac{3A''}{r^2}=\frac{\Lambda}{2M_*^3kr}
      \bigl[\delta(\theta-\pi)-\delta(\theta)\bigr].
  \]
  \begin{enumerate}
    \item Solve for \(A(\theta)\), determine \(k\) in terms of \(\Lambda\)
      and \(M_*\), and verify both bulk and brane terms in the equations.
    \item Use the warp factor to show that
      \[
        M_{\rm Pl}^2=M_*^3r\int_{-\pi}^{\pi}d\theta\,e^{-2A(\theta)}
          =\frac{M_*^3}{k}\left(1-e^{-2k\pi r}\right).
      \]
    \item Put a Higgs field on the brane at \(\theta=\pi\).  Bring its
      Lagrangian to canonical form and show that a fundamental mass \(m_0\)
      is observed as \(m=e^{-k\pi r}m_0\).
    \item Estimate the size of \(r\) needed to generate the electroweak
      scale from the Planck scale.  Explain in what sense the construction
      addresses the hierarchy problem and compare \(M_{\rm Pl}\) with the
      five-dimensional scale \(M_*\).
  \end{enumerate}

\end{enumerate}
