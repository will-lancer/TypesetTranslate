  \SupplementaryExercise{1}{Vector Superfields and Supersymmetric Maxwell Theory}
    {(Adapted from University of Cambridge, 2022 Part III Paper 307, \emph{Supersymmetry}, Question 2.)}
    {cambridge-part-iii-supersymmetry-2022-p307-q2-v3ch27-p01}
For this problem only, the displayed two-component formulas retain the
  source's metric
  \(\eta_{\mathrm s}=\operatorname{diag}(+1,-1,-1,-1)=-\eta\), with
  \(\epsilon_{\mathrm s}^{0123}=+1\).  This convention is marked by the
  subscript \({\mathrm s}\) so that it is not confused with the mostly-plus
  convention used elsewhere in this volume.  Take
  \(s^\mu=(\mathbf 1,\boldsymbol\sigma)\) and
  \(\bar s^\mu=(\mathbf 1,-\boldsymbol\sigma)\).

  A real superfield \(V\) has component expansion
  \[
  \begin{aligned}
    V(x,\theta,\bar\theta)={}&C+\theta\chi+\bar\theta\bar\chi
      +\ii\theta^2M-\ii\bar\theta^2M^\dagger
      +\theta s^\mu\bar\theta\,A_\mu\\
    &+\theta^2\bar\theta
       \left(\bar\lambda+\frac{\ii}{2}\bar s^\mu\partial_\mu\chi\right)
      +\bar\theta^2\theta
       \left(\lambda+\frac{\ii}{2}s^\mu\partial_\mu\bar\chi\right)\\
    &+\frac12\theta^2\bar\theta^2
       \left(D-\frac12\Box_{\mathrm s}C\right).
  \end{aligned}
  \]
  A chiral multiplet has the expansion
  \[
    \Omega=\omega+\sqrt2\,\theta\psi+\theta^2F
      +\ii\theta s^\mu\bar\theta\,\partial_\mu\omega
      -\frac{\ii}{\sqrt2}\theta^2
         \partial_\mu\psi\,s^\mu\bar\theta
      -\frac14\theta^2\bar\theta^2\Box_{\mathrm s}\omega.
  \]
  Explain how the shift
  \[
    V\longmapsto V+\ii(\Omega-\Omega^\dagger)
  \]
  can be used to implement gauge transformations of \(A_\mu\) and remove
  some of the fields in \(V\).

  A superfield is defined by
  \[
    W_a=-\frac14\bar{\mathcal D}^{\,2}\mathcal D_aV.
  \]
  Explain briefly how a supersymmetric action can be constructed using this
  superfield.  You need not compute the component form of this action.

  Supersymmetric Maxwell theory has the action
  \[
    S=\int \dd^4x\left[
      -\frac1{4e^2}F_{\mu\nu}F^{\mu\nu}
      -\frac{\ii}{e^2}\lambda s^\mu\partial_\mu\bar\lambda
    \right],
    \qquad
    F_{\mu\nu}=\partial_\mu A_\nu-\partial_\nu A_\mu,
  \]
  where spacetime contractions in this displayed source formula use
  \(\eta_{\mathrm s}\).  Show that the action is invariant under
  \[
    \delta A_\mu=\epsilon s_\mu\bar\lambda
                   +\lambda s_\mu\bar\epsilon,
    \qquad
    \delta\lambda=F_{\mu\nu}\Sigma^{\mu\nu}\epsilon,
  \]
  together with the conjugate transformation, where \(\epsilon\) is a
  Grassmann-valued Weyl spinor and
  \[
    \Sigma^{\mu\nu}=\frac{\ii}{4}
       (s^\mu\bar s^\nu-s^\nu\bar s^\mu).
  \]
  You may use the supplied sigma-matrix identity
  \[
    s^\nu\bar s^\mu s^\rho
      =\eta_{\mathrm s}^{\mu\nu}s^\rho
       +\eta_{\mathrm s}^{\mu\rho}s^\nu
       -\eta_{\mathrm s}^{\nu\rho}s^\mu
       +\ii\epsilon_{\mathrm s}^{\nu\mu\rho\kappa}s_\kappa.
  \]

  \SupplementaryExercise{2}{The Superspace Bianchi Identity and Abelian Gauge Action}
    {(Adapted from David Tong, \emph{Supersymmetry Example Sheet 2}, Question 3.)}
    {ch27-s02-the-superspace-bianchi-identity-and-abelian}
  \begin{enumerate}
    \item \emph{The Superspace Bianchi Identity.}
    Derive Eq.~\eqref{eq:27.2.20} from the superderivative definition of
      \(W_L,W_R\).  Show at component level that its tensor equation is
      \(\partial_{[\rho}f_{\mu\nu]}=0\), and state the local solution.
    \item \emph{The Abelian Gauge Action and the Theta Term.}
    Extract the \(\mathcal F\)-component of
      \(W_L^{\mathsf T}\epsilon W_L\).  Show that its real part gives the Maxwell,
      gaugino, and \(D^2\) terms, whereas its imaginary part is a spacetime
      derivative proportional to
      \(\epsilon^{\mu\nu\rho\sigma}f_{\mu\nu}f_{\rho\sigma}\).
  \end{enumerate}

  \SupplementaryExercise{3}{Non-Abelian Matter and Gauge Covariance}
    {(Inspired by Adel Bilal, \emph{Introduction to Supersymmetry}, Sections 5.1--5.3; independently formulated.)}
    {ch27-s03-non-abelian-matter-and-gauge-covariance}
  \begin{enumerate}
    \item \emph{Gauge-Invariant Matter Kinetic Term.}
    Use the finite transformations of \(\Phi\), \(\Phi^\dagger\), and
      \(\Gamma\) to prove that
      \([\Phi^\dagger\Gamma\Phi]_D\) is invariant under the extended gauge
      group.  Explain why \([\Phi^\dagger\Phi]_D\) is not invariant when the
      chiral gauge parameter depends on superspace.
    \item \emph{The Covariant Non-Abelian Field Strength.}
    Starting from \(\Gamma=e^{-2t_AV^A}\), prove that the non-Abelian chiral
      field strength transforms by conjugation.  Deduce that
      \(\operatorname{Tr}(W_L^{\mathsf T}\epsilon W_L)\) is invariant and that its
      component tensor is the Yang--Mills field strength
      \eqref{eq:27.3.5}.
    \item \emph{Closure of the Non-Abelian Bianchi Identity.}
    Use the Jacobi identity for gauge-covariant derivatives to prove
      \(D_{[\rho}f_{\mu\nu]}=0\).  Contract with
      \(\epsilon^{\mu\nu\rho\sigma}\), and explain how this identity appears
      in the constrained non-Abelian spinor field-strength superfield.
    \item \emph{Gauge Invariance of the Superpotential.}
    Derive Eq.~\eqref{eq:27.4.12} by varying a holomorphic superpotential.
      Differentiate the identity once and evaluate it at a vacuum to obtain
      the relation between the chiral-fermion mass matrix and the gauge-orbit
      tangent vectors.
  \end{enumerate}

  \SupplementaryExercise{4}{Matrix \(D\)-Flatness in \(U(N_c)\) Gauge Theory}
    {(Adapted from David Tong, \emph{Supersymmetry Example Sheet 2}, Question 5.)}
    {ch27-s04-auxiliary-fields-fayet-iliopoulos-terms-and}
Consider \(U(N_c)\) gauge theory with \(N_f\) chiral multiplets
  \(Q_i\) in the fundamental and \(\widetilde Q_i\) in the
  antifundamental.  With Hermitian generators \(T^A\), the vanishing
  \(D\)-term conditions are
  \[
    D^A=\sum_{i=1}^{N_f}
      \left(q_i^\dagger T^Aq_i
       -\widetilde q_iT^A\widetilde q_i^\dagger\right)=0
       \quad\text{for every }A.
  \]
  Construct an explicit Hermitian basis for the \(N_c\times N_c\) matrices
  and prove that these \(N_c^2\) equations are equivalent to the single
  matrix equation
  \[
    \sum_i q_i^{*a}q_{ib}
      -\sum_i\widetilde q_i^{\,a}
              \widetilde q_b^{*i}=0
      \qquad(a,b=1,\ldots,N_c).
  \]

  \end{enumerate}

  \SupplementaryExercise{6}{Supersymmetry of Pure Yang--Mills Theory}
    {(Adapted from David Tong, \emph{Supersymmetry Example Sheet 2}, Question 4.)}
    {ch27-s06-gauge-invariant-supersymmetry-and-compensated-closure}
For a compact gauge group and vanishing theta angle, set
  \[
    s^\mu=(\mathbf{1},\boldsymbol{\sigma}),\qquad
    \bar s^\mu=(\mathbf{1},-\boldsymbol{\sigma}),\qquad
    \Sigma^{\mu\nu}
      =\frac{\ii}{4}(s^\mu\bar s^\nu-s^\nu\bar s^\mu).
  \]
  Thus
  \(s^\mu\bar s^\nu+s^\nu\bar s^\mu
    =-2\eta^{\mu\nu}\mathbf{1}\) in the mostly-plus convention of this
  volume.  Take the on-shell pure-super-Yang--Mills action
  \[
    S=\frac1{g^2}\int \dd^4x\,\operatorname{Tr}
      \left(-\frac12F_{\mu\nu}F^{\mu\nu}
            -2\ii\lambda s^\mu D_\mu\bar\lambda\right),
  \]
  where \(D_\mu X=\partial_\mu X-\ii[A_\mu,X]\) on an adjoint field and
  \(F_{\mu\nu}=\partial_\mu A_\nu-\partial_\nu A_\mu-\ii[A_\mu,A_\nu]\).
  Verify its invariance under
  \[
    \delta A_\mu=\epsilon s_\mu\bar\lambda
                   +\lambda s_\mu\bar\epsilon,\qquad
    \delta\lambda=\Sigma^{\mu\nu}\epsilon F_{\mu\nu},
  \]
  together with the conjugate gaugino transformation.  Give the complete
  cancellation, including the terms from varying the gauge connection in
  \(D_\mu\bar\lambda\), and state explicitly where the non-Abelian Bianchi
  identity is used.  As a hint translated to the metric convention of this
  volume, you may use
  \[
    s^\nu\bar s^\mu s^\rho
      =-\eta^{\mu\nu}s^\rho-\eta^{\mu\rho}s^\nu
       +\eta^{\nu\rho}s^\mu
       -\ii\epsilon^{\nu\mu\rho\kappa}s_\kappa .
  \]

  \SupplementaryExercise{7}{Chiral Superfields and a Gauged \(U(1)\) Model}
    {(Adapted from University of Cambridge, 2024 Part III Paper 307, \emph{Supersymmetry}, Question 2.)}
    {cambridge-part-iii-supersymmetry-2024-p307-q2-v3ch27-p07}
For this problem only, let
  \(\eta_{\mathrm s}=\operatorname{diag}(+1,-1,-1,-1)=-\eta\) denote the
  source's mostly-minus metric.  Its two-component conventions are
  \[
    \varepsilon_{\mathrm s}^{12}=-\varepsilon^{\mathrm s}_{12}
      =\varepsilon_{\mathrm s}^{\dot1\dot2}
      =-\varepsilon^{\mathrm s}_{\dot1\dot2}=1,
    \qquad
    \{s^\mu,\bar s^\nu\}=2\eta_{\mathrm s}^{\mu\nu}\mathbf1,
  \]
  with
  \((\bar s^\mu)^{\dot a a}
    =\varepsilon_{\mathrm s}^{ab}
      \varepsilon_{\mathrm s}^{\dot a\dot b}
      (s^\mu)_{b\dot b}\), and
  \[
    \bar{\mathcal D}_{\dot a}
      =-\bar\partial_{\dot a}
       -\ii\theta^b(s^\mu)_{b\dot a}\partial_\mu.
  \]
  \begin{enumerate}
    \item Define what it means for a superfield
      \(\Phi(x,\theta,\bar\theta)\) to be chiral.  Give an example of a
      chiral superfield in the MSSM and list the physical component fields
      contained within it.

      Show that, when written in terms of
      \(y^\mu=x^\mu+\ii\theta s^\mu\bar\theta\), a chiral superfield has a
      particularly simple component expansion.  Hence deduce that
      \[
      \begin{aligned}
        \Phi(x,\theta,\bar\theta)={}&\phi(x)+\sqrt2\,\theta\psi(x)
          +(\theta\theta)F(x)
          +A\,\theta s^\mu\bar\theta\,\partial_\mu\phi(x)\\
        &+B(\theta\theta)\partial_\mu\psi(x)s^\mu\bar\theta
          +C(\theta\theta)(\bar\theta\bar\theta)
             \partial_\mu\partial^\mu\phi(x),
      \end{aligned}
      \]
      where \(A,B,C\) are constants to be determined and contractions in
      this display use \(\eta_{\mathrm s}\).

    \item Consider a globally \(N=1\) supersymmetric theory with only three
      chiral superfields \(X_{2+},X_0,X_{2-}\), where the suffix labels the
      charge under a gauged \(U(1)\) symmetry.  Do you expect the theory to
      possess a gauge anomaly?  Explain your answer using Feynman diagrams
      and a mathematical expression.

      Write down the most general superpotential of the model.  Now set to
      zero every parameter multiplying a quadratic or linear superpotential
      term.  State in one sentence why quantum corrections are expected to
      respect these conditions.  Write down the scalar potential, including
      any Fayet--Iliopoulos terms, and determine whether and how the model
      spontaneously breaks supersymmetry, the \(U(1)\) symmetry, or both.
  \end{enumerate}

  \SupplementaryExercise{8}{Wilsonian Gauge Couplings and FI Protection}
    {(Inspired by David Tong, \emph{Supersymmetric Field Theory}, Sections 3.3--3.4 and 4.3; independently formulated.)}
    {ch27-s08-wilsonian-gauge-couplings-and-fi-protection}
  \begin{enumerate}
    \item \emph{A Holomorphic Non-Renormalization Argument.}
    Promote the mass and cubic couplings of a renormalizable superpotential
      to background chiral superfields.  Use holomorphy, mass dimension, and
      the restored phase symmetries to show that perturbation theory cannot
      generate a new Wilsonian superpotential term.
    \item \emph{One-Loop Exactness of the Holomorphic Gauge Coupling.}
    Treat the complex gauge coupling multiplying
      \([W_L^{\mathsf T}\epsilon W_L]_{\mathcal F}\) as a chiral background.
      Explain why its Wilsonian perturbative renormalization is restricted to
      one loop, distinguishing this holomorphic statement from the running of
      a canonically normalized physical coupling.
    \item \emph{Why the FI Term Is Not Perturbatively Generated.}
    Show that a one-loop \(D\)-tadpole is proportional to
      \(\operatorname{Tr}t\) over chiral multiplets.  Relate this trace to the
      mixed gauge--gravitational anomaly and explain why an anomaly-free theory
      with no bare FI term has no perturbatively generated FI term.
  \end{enumerate}

  \SupplementaryExercise{9}{Soft Breaking and Power Counting}
    {(Inspired by Stephen P. Martin, \emph{A Supersymmetry Primer}, version 7, Section 5; independently formulated.)}
    {ch27-s09-soft-breaking-and-power-counting}
  \begin{enumerate}
    \item \emph{Classifying Soft Breaking Terms.}
    For a renormalizable gauge theory with chiral scalars, list the
      gauge-invariant scalar masses, holomorphic bilinear and trilinear scalar
      couplings, gaugino masses, and allowed linear terms.  Introduce a
      background spurion superfield for each class and identify the component

      property.
  \end{enumerate}

  \SupplementaryExercise{10}{Superspace and Finite \(SU(3)\) Gauge Transformations}
    {(Adapted from University of Cambridge, 2025 Part III Paper 307, \emph{Supersymmetry}, Question 1.)}
    {cambridge-part-iii-supersymmetry-2025-p307-q1-v3ch27-p10}
This question concerns a renormalizable, globally \(N=1\) supersymmetric
  \(SU(3)\) gauge theory with one chiral superfield \(\Phi\) in the
  fundamental representation and a vector superfield \(V=V_AT_A\), where
  \(T_A\) are fundamental-representation generators.  Strictly speaking,
  the theory has a gauge anomaly, but ignore it here.  The supersymmetry
  generators \(Q_a,\overline Q_{\dot a}\) are represented on functions of
  superspace by \(\mathcal Q_a,\overline{\mathcal Q}_{\dot a}\); denote the
  supersymmetry-covariant derivatives by
  \(\mathcal D_a,\overline{\mathcal D}_{\dot a}\).
  \begin{enumerate}
    \item[(i)] Under a supersymmetry transformation a general superfield
      transforms as
      \[
        S(x,\theta,\bar\theta)\longmapsto
          \exp\!\left[\ii(\epsilon\mathcal Q
                       +\bar\epsilon\overline{\mathcal Q})\right]
          S(x,\theta,\bar\theta),
      \]
      where \(\epsilon^a,\bar\epsilon^{\dot a}\) are Grassmann-valued
      transformation parameters.  Find the explicit differential-operator
      forms of \(\mathcal Q_a,\overline{\mathcal Q}_{\dot a}\).  You may
      assume any properties of the super-Poincar\'e algebra from the text,
      provided they are stated clearly.
    \item[(ii)] What is the defining property of a vector superfield \(V_A\)?
    \item[(iii)] What is the defining property of a chiral superfield
      \(\Phi\)?
    \item[(iv)] A generalized \(SU(3)\) gauge transformation is
      \[
        \Phi\longmapsto e^{-2\ii\Lambda}\Phi,
        \qquad \Lambda=\Lambda_AT_A.
      \]
      Give the mass dimension of the superfields \(\Lambda_A\) and state
      clearly all restrictions on their nature.
  \end{enumerate}
  The supersymmetric \(SU(3)\) field-strength chiral superfield is
  \[
    W_a=-\frac18\overline{\mathcal D}^{\,2}
      \left(e^{-2V}\mathcal D_a e^{2V}\right).
  \]
  \begin{enumerate}
    \item[(v)] Write the theory's Lagrangian density as superspace integrals
      involving functions of \(V_A,\Phi\), and \(W_a\).
    \item[(vi)] Assuming generalized \(SU(3)\) gauge invariance of the
      standard K\"ahler potential, relate \(V\) and \(V'\), where
      \(V\mapsto V'=V'_AT_A\) under a finite generalized gauge
      transformation.
    \item[(vii)] Using the relation from part (vi), show that the entire
      Lagrangian density is invariant under a generalized gauge
      transformation.
  \end{enumerate}

  \SupplementaryExercise{11}{SQED Vacua and the Moduli-Space Metric}
    {(Adapted from University of Cambridge, 2023 Part III Paper 307, \emph{Supersymmetry and Duality}, Question 2.)}
    {cambridge-part-iii-supersymmetry-and-duality-2023-p307-q2-v3ch27-p11}
Supersymmetric QED consists of a \(U(1)\) vector multiplet \(V\) coupled
  to \(N\) chiral multiplets \(\Phi^i\) of charge \(+1\) and \(N\) chiral
  multiplets \(\widetilde\Phi_i\) of charge \(-1\), with
  \(i=1,\ldots,N\).  A Fayet--Iliopoulos term \(\zeta\) is added to the
  action.  A neutral chiral multiplet \(X\) is coupled through the
  superpotential
  \[
    W=\sum_{i=1}^N\widetilde\Phi_i(X-m_i)\Phi^i.
  \]
  Write down the potential energy \(V\).

  For each of the following cases, describe the \(V=0\) ground states.  If
  there is a moduli space of vacua, compute its dimension; if there are
  isolated vacua, compute how many:
  \begin{enumerate}
    \item[(i)] \(\zeta\ne0\) and \(m_i=0\);
    \item[(ii)] \(\zeta\ne0\) and the \(m_i\) are all distinct;
    \item[(iii)] \(\zeta=0\) and the \(m_i\) are all distinct;
    \item[(iv)] \(m_i=\zeta=0\).
  \end{enumerate}
  Finally, consider \(N=1\) and \(\zeta=m_1=0\).  Compute the K\"ahler
  potential and associated metric on the moduli space of vacua parametrized
  by \(M=\widetilde\Phi\Phi\).  What are the singularities of this metric,
  and what is their physical meaning?

  \SupplementaryExercise{12}{The \(SU(3)\) Coulomb Branch}
    {(Adapted from John McGreevy, \emph{MIT 8.821 Problem Set 2}, Problem 6.)}
    {ch27-s12-the-su-3-coulomb-branch}
  \begin{enumerate}
    \item \emph{The Coulomb Branch of Pure \(N=2\) Theory.}
    Rewrite the pure \(N=2\) scalar potential as a norm of
      \([\phi^\dagger,\phi]\).  Prove that \(V=0\) precisely when
      \(\operatorname{Re}\phi\) and \(\operatorname{Im}\phi\) commute, and
      show that a generic vacuum can be gauge rotated into a Cartan subalgebra.
    \item \emph{Root-Space Masses on the \(SU(3)\) Coulomb Branch.}
    Let
      \(\phi_\infty=\operatorname{diag}(a_1,a_2,a_3)\), with
      \(\sum_i a_i=0\).  Using the roots of \(SU(3)\), find the masses of the
      six off-diagonal vector multiplets.  Identify all loci where the
      unbroken gauge group is larger than \(U(1)^2\).
  \end{enumerate}
