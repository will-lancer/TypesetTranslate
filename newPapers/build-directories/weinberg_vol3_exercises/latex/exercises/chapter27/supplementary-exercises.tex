\chapterbackmatter{Supplementary Exercises}

\begin{enumerate}
  \SupplementaryExercise{1}{Residual Gauge Transformations in Wess--Zumino Gauge}
    {(Adapted from Philip C. Argyres, \emph{An Introduction to Global
    Supersymmetry}, Section 1.7.1)}
    {argyres-global-susy-2001-ch27}
  In the Abelian theory apply Eq.~\eqref{eq:27.1.12} to a vector
  superfield already in Wess--Zumino gauge.  Determine the restrictions on
  \(W,w,\mathcal W\) that preserve the gauge and show that the remaining
  freedom is \(V_\mu\mapsto V_\mu+\partial_\mu\Lambda\).

  \SupplementaryExercise{2}{Gauge-Invariant Matter Kinetic Term}
    {(Adapted from Adel Bilal, \emph{Introduction to Supersymmetry},
    Sections 5.1--5.2)}
    {bilal-introduction-susy-2001-ch27}
  Use the finite transformations of \(\Phi\), \(\Phi^\dagger\), and
  \(\Gamma\) to prove that
  \([\Phi^\dagger\Gamma\Phi]_D\) is invariant under the extended gauge
  group.  Explain why \([\Phi^\dagger\Phi]_D\) is not invariant when the
  chiral gauge parameter depends on superspace.

  \SupplementaryExercise{3}{Chirality and Gauge Invariance of \(W_\alpha\)}
    {(Adapted from David Tong, \emph{Supersymmetric Field Theory},
    Section 4.1.1)}
    {tong-supersymmetric-field-theory-2022-ch27}
  In the Abelian theory start with Eq.~\eqref{eq:27.2.7}.  Prove directly
  that \(W_L\) is left chiral and unchanged by
  \(V\mapsto V+\tfrac i2(\Omega-\Omega^*)\).  Identify the superderivative
  identities used in both steps.

  \SupplementaryExercise{4}{Recovering the Abelian Components}
    {(Adapted from Philip C. Argyres, \emph{An Introduction to Global
    Supersymmetry}, Section 1.7.1)}
    {argyres-global-susy-2001-ch27}
  Insert the Wess--Zumino expansion \eqref{eq:27.1.19} into the definition
  of \(W_L\).  Derive the three terms in Eq.~\eqref{eq:27.2.11}, including
  the coefficient of \(f_{\mu\nu}\), the sign of \(D\), and the derivative
  of the opposite-chirality gaugino.

  \SupplementaryExercise{5}{The Superspace Bianchi Identity}
    {(Adapted from Stephen P. Martin, \emph{A Supersymmetry Primer},
    version 7, Sections 4.7 and 4.8)}
    {martin-susy-primer-v7-ch27}
  Derive Eq.~\eqref{eq:27.2.20} from the superderivative definition of
  \(W_L,W_R\).  Show at component level that its tensor equation is
  \(\partial_{[\rho}f_{\mu\nu]}=0\), and state the local solution.

  \SupplementaryExercise{6}{The Abelian Gauge Action and the Theta Term}
    {(Adapted from Adel Bilal, \emph{Introduction to Supersymmetry},
    Section 5.2)}
    {bilal-introduction-susy-2001-ch27}
  Extract the \(\mathcal F\)-component of
  \(W_L^{\rm T}\epsilon W_L\).  Show that its real part gives the Maxwell,
  gaugino, and \(D^2\) terms, whereas its imaginary part is a spacetime
  derivative proportional to
  \(\epsilon^{\mu\nu\rho\sigma}f_{\mu\nu}f_{\rho\sigma}\).

  \SupplementaryExercise{7}{When Is a Fayet--Iliopoulos Term Allowed?}
    {(Adapted from David Tong, \emph{Supersymmetric Field Theory},
    Sections 4.1.1--4.1.2)}
    {tong-supersymmetric-field-theory-2022-ch27}
  Show that \(-\xi[V]_D\) is invariant in an Abelian theory.  Then use the
  non-Abelian transformation law to show why a linear term
  \(-\xi_A[V^A]_D\) is allowed only for generators in Abelian factors (or,
  equivalently, for an invariant vector of the adjoint representation).

  \SupplementaryExercise{8}{The Covariant Non-Abelian Field Strength}
    {(Adapted from Philip C. Argyres, \emph{An Introduction to Global
    Supersymmetry}, Sections 1.8.1--1.8.2)}
    {argyres-global-susy-2001-ch27}
  Starting from \(\Gamma=e^{-2t_AV^A}\), prove that the non-Abelian chiral
  field strength transforms by conjugation.  Deduce that
  \(\operatorname{Tr}(W_L^{\rm T}\epsilon W_L)\) is invariant and that its
  component tensor is the Yang--Mills field strength
  \eqref{eq:27.3.5}.

  \SupplementaryExercise{9}{Closure of the Non-Abelian Bianchi Identity}
    {(Adapted from Adel Bilal, \emph{Introduction to Supersymmetry},
    Section 5.3)}
    {bilal-introduction-susy-2001-ch27}
  Use the Jacobi identity for gauge-covariant derivatives to prove
  \(D_{[\rho}f_{\mu\nu]}=0\).  Contract with
  \(\epsilon^{\mu\nu\rho\sigma}\), and explain how this identity appears
  in the constrained non-Abelian spinor field-strength superfield.

  \SupplementaryExercise{10}{Eliminating All Auxiliary Fields}
    {(Adapted from Stephen P. Martin, \emph{A Supersymmetry Primer},
    version 7, Sections 3.4 and 4.9)}
    {martin-susy-primer-v7-ch27}
  Isolate the \(\mathcal F_n\) and \(D_A\) dependence in
  Eq.~\eqref{eq:27.4.1}, complete the squares, and derive
  Eqs.~\eqref{eq:27.4.6}--\eqref{eq:27.4.9}.  State the necessary and
  sufficient conditions for a tree-level supersymmetric vacuum.

  \SupplementaryExercise{11}{Gauge Invariance of the Superpotential}
    {(Adapted from Adel Bilal, \emph{Introduction to Supersymmetry},
    Sections 5.2--5.3)}
    {bilal-introduction-susy-2001-ch27}
  Derive Eq.~\eqref{eq:27.4.12} by varying a holomorphic superpotential.
  Differentiate the identity once and evaluate it at a vacuum to obtain
  the relation between the chiral-fermion mass matrix and the gauge-orbit
  tangent vectors.

  \SupplementaryExercise{12}{Complexified Gauge Orbits and \(D\)-Flatness}
    {(Adapted from David Tong, \emph{Supersymmetric Field Theory},
    Section 4.3)}
    {tong-supersymmetric-field-theory-2022-ch27}
  Suppose all Fayet--Iliopoulos constants vanish and an \(F\)-flat point
  exists.  Explain why its complexified gauge orbit remains \(F\)-flat.
  For one \(U(1)\) with charges \(q_i\), minimize
  \(\sum_i|e^{q_i s}\phi_i|^2\) over real \(s\) and relate its stationary
  condition to \(D=0\).

  \SupplementaryExercise{13}{Unbroken Generators and Vector Masses}
    {(Adapted from Philip C. Argyres, \emph{An Introduction to Global
    Supersymmetry}, Sections 1.7.4 and 1.8.2)}
    {argyres-global-susy-2001-ch27}
  Expand the scalar kinetic term around \(\phi_0\) and derive
  \((\mu^2)_{AB}=\phi_0^\dagger\{t_A,t_B\}\phi_0\).
  Prove that a real vector \(c_A\) is a zero mode precisely when
  \(c_At_A\phi_0=0\).

  \SupplementaryExercise{14}{The Fermions of a Massive Vector Multiplet}
    {(Adapted from David Tong, \emph{Supersymmetric Field Theory},
    Sections 4.1.2 and 4.3)}
    {tong-supersymmetric-field-theory-2022-ch27}
  At an \(F\)- and \(D\)-flat Higgs vacuum, use the matter--gaugino mixing
  in Eq.~\eqref{eq:27.4.8} to show that every broken generator pairs one
  gaugino combination with the chiral fermion tangent to the complexified
  gauge orbit.  Show that their mass equals the associated vector mass.

  \SupplementaryExercise{15}{The Two Phases of Fayet--Iliopoulos SQED}
    {(Adapted from Stephen P. Martin, \emph{A Supersymmetry Primer},
    version 7, Sections 7.2 and 7.5)}
    {martin-susy-primer-v7-ch27}
  For two chiral fields of charges \(\pm q\), superpotential
  \(f=m\Phi_+\Phi_-\), and FI parameter \(\xi\), minimize
  \[
    V=|m|^2(|\phi_+|^2+|\phi_-|^2)
      +\frac12\{\xi+q(|\phi_+|^2-|\phi_-|^2)\}^2.
  \]
  Locate the phase boundary, determine whether the gauge symmetry is
  broken in each phase, and calculate the vacuum energy.

  \SupplementaryExercise{16}{The General Goldstino Direction}
    {(Adapted from Adel Bilal, \emph{Introduction to Supersymmetry},
    Sections 6.2--6.3)}
    {bilal-introduction-susy-2001-ch27}
  At a stationary point of the general potential, prove directly that
  \((i\sqrt2\mathcal F_{n0},D_{A0})\) is a null vector of the full fermion
  mass-squared matrix.  Normalize the goldstino and state how much of it
  lies in chiral fermions and how much in gauginos.

  \SupplementaryExercise{17}{The Tree-Level Mass Supertrace}
    {(Adapted from Stephen P. Martin, \emph{A Supersymmetry Primer},
    version 7, Sections 7.3--7.4)}
    {martin-susy-primer-v7-ch27}
  Take the traces of Eqs.~\eqref{eq:27.5.4},
  \eqref{eq:27.5.6}, and \eqref{eq:27.5.7}.  Derive the supertrace rule
  \eqref{eq:27.5.10} and explain why anomaly cancellation reduces its
  right-hand side to zero in a consistent renormalizable theory.

  \SupplementaryExercise{18}{A Holomorphic Non-Renormalization Argument}
    {(Adapted from David Tong, \emph{Supersymmetric Field Theory},
    Sections 3.3.3 and 3.4)}
    {tong-supersymmetric-field-theory-2022-ch27}
  Promote the mass and cubic couplings of a renormalizable superpotential
  to background chiral superfields.  Use holomorphy, mass dimension, and
  the restored phase symmetries to show that perturbation theory cannot
  generate a new Wilsonian superpotential term.

  \SupplementaryExercise{19}{One-Loop Exactness of the Holomorphic Gauge Coupling}
    {(Adapted from Adel Bilal, \emph{Introduction to Supersymmetry},
    Sections 7.2 and 8.1)}
    {bilal-introduction-susy-2001-ch27}
  Treat the complex gauge coupling multiplying
  \([W_L^{\rm T}\epsilon W_L]_{\mathcal F}\) as a chiral background.
  Explain why its Wilsonian perturbative renormalization is restricted to
  one loop, distinguishing this holomorphic statement from the running of
  a canonically normalized physical coupling.

  \SupplementaryExercise{20}{Why the FI Term Is Not Perturbatively Generated}
    {(Adapted from Philip C. Argyres, \emph{An Introduction to Global
    Supersymmetry}, Sections 1.7.2--1.7.3)}
    {argyres-global-susy-2001-ch27}
  Show that a one-loop \(D\)-tadpole is proportional to
  \(\operatorname{Tr}t\) over chiral multiplets.  Relate this trace to the
  mixed gauge--gravitational anomaly and explain why an anomaly-free theory
  with no bare FI term has no perturbatively generated FI term.

  \SupplementaryExercise{21}{Classifying Soft Breaking Terms}
    {(Adapted from Stephen P. Martin, \emph{A Supersymmetry Primer},
    version 7, Section 5)}
    {martin-susy-primer-v7-ch27}
  For a renormalizable gauge theory with chiral scalars, list the
  gauge-invariant scalar masses, holomorphic bilinear and trilinear scalar
  couplings, gaugino masses, and allowed linear terms.  Introduce a
  background spurion superfield for each class and identify the component
  that breaks supersymmetry.

  \SupplementaryExercise{22}{Power Counting for Softness}
    {(Adapted from Adel Bilal, \emph{Introduction to Supersymmetry},
    Sections 6.3.2 and 7.2)}
    {bilal-introduction-susy-2001-ch27}
  Use the superficial degree of divergence to show that insertions of the
  standard dimension-two and dimension-three soft operators cannot create
  new quadratic divergences in scalar masses.  Explain why generic
  dimension-four supersymmetry-breaking interactions do not share this
  property.

  \SupplementaryExercise{23}{Gauge-Invariant Supersymmetry in the Abelian Theory}
    {(Adapted from Philip C. Argyres, \emph{An Introduction to Global
    Supersymmetry}, Sections 1.7.1--1.7.2)}
    {argyres-global-susy-2001-ch27}
  Apply an ordinary supersymmetry transformation to
  \(V\) in Wess--Zumino gauge and identify the components that leave the
  gauge slice.  Construct the compensating extended gauge transformation
  and derive gauge-covariant transformations of \(V_\mu,\lambda,D\).

  \SupplementaryExercise{24}{Closure up to a Gauge Transformation}
    {(Adapted from David Tong, \emph{Supersymmetric Field Theory},
    Sections 4.1.1 and 4.2.1)}
    {tong-supersymmetric-field-theory-2022-ch27}
  Compute the commutator of two gauge-covariant supersymmetry
  transformations on \(V_\mu\) and on a charged scalar.  Show that it is a
  translation plus a field-dependent gauge transformation, and give the
  gauge parameter.

  \SupplementaryExercise{25}{The Second Supersymmetry of Pure \(N=2\) Yang--Mills}
    {(Adapted from Adel Bilal, \emph{Introduction to Supersymmetry},
    Sections 8.1--8.2)}
    {bilal-introduction-susy-2001-ch27}
  Specialize Eq.~\eqref{eq:27.9.3} to an adjoint chiral multiplet.  Verify
  that the exchange \(\psi_A\leftrightarrow\lambda_A\), accompanied by the
  auxiliary-field transformation in Section 27.9, leaves the action
  invariant.  Explain why this produces a second independent supersymmetry.

  \SupplementaryExercise{26}{The Coulomb Branch of Pure \(N=2\) Theory}
    {(Adapted from David Tong, \emph{Supersymmetric Field Theory},
    Section 4.4.1)}
    {tong-supersymmetric-field-theory-2022-ch27}
  Rewrite the pure \(N=2\) scalar potential as a norm of
  \([\phi^\dagger,\phi]\).  Prove that \(V=0\) precisely when
  \(\operatorname{Re}\phi\) and \(\operatorname{Im}\phi\) commute, and
  show that a generic vacuum can be gauge rotated into a Cartan subalgebra.

  \SupplementaryExercise{27}{Root-Space Masses on the \(SU(3)\) Coulomb Branch}
    {(Adapted from Adel Bilal, \emph{Introduction to Supersymmetry},
    Sections 8.2 and 9.1)}
    {bilal-introduction-susy-2001-ch27}
  Let
  \(\phi_\infty=\operatorname{diag}(a_1,a_2,a_3)\), with
  \(\sum_i a_i=0\).  Using the roots of \(SU(3)\), find the masses of the
  six off-diagonal vector multiplets.  Identify all loci where the
  unbroken gauge group is larger than \(U(1)^2\).

  \SupplementaryExercise{28}{Central Charge as a Surface Flux}
    {(Adapted from Philip C. Argyres, \emph{An Introduction to Global
    Supersymmetry}, Sections 1.8.2 and 2.4)}
    {argyres-global-susy-2001-ch27}
  Starting from Eqs.~\eqref{eq:27.9.17}--\eqref{eq:27.9.20}, derive the
  central charge for a rank-\(r\) Coulomb vacuum in terms of the \(r\)
  electric and magnetic surface fluxes.  Prove that the answer is
  independent of the chosen orthonormal Cartan basis.

  \SupplementaryExercise{29}{Electric, Magnetic, and Dyonic BPS Masses}
    {(Adapted from Adel Bilal, \emph{Introduction to Supersymmetry},
    Sections 8.2 and 9.2)}
    {bilal-introduction-susy-2001-ch27}
  Combine the central charge of Eq.~\eqref{eq:27.9.22} with the \(N=2\)
  positivity bound.  Obtain the masses of a purely electric state, a
  monopole of winding \(\nu\), and a dyon, and state precisely when the
  inequalities are saturated.

  \SupplementaryExercise{30}{Allowed \(N=2\) Hypermultiplet Couplings}
    {(Adapted from Stephen P. Martin, \emph{A Supersymmetry Primer},
    version 7, Sections 4.10 and 5.5)}
    {martin-susy-primer-v7-ch27}
  Represent a hypermultiplet by two \(N=1\) chiral superfields in conjugate
  gauge representations.  Use gauge invariance, renormalizability, and the
  \(SU(2)\) symmetry rotating the two supersymmetries to recover the
  superpotential structure of Eq.~\eqref{eq:27.9.29}.  Explain why the
  other quadratic and cubic terms are excluded.
\end{enumerate}
