\begin{enumerate}
  \SupplementarySolution{1}{Vector Superfields in Wess--Zumino Gauge}
  \begin{enumerate}
    \item \emph{Residual Gauge Transformations in Wess--Zumino Gauge.}
    In the Abelian limit Eq.~\eqref{eq:27.1.17} is exact.  Starting with
      \(C=\omega=M=N=0\), preservation of the gauge requires
      \[
        \operatorname{Im}W=0,\qquad w=0,\qquad\mathcal W=0.
      \]
      Thus the only independent lowest component is a real function
      \(\Lambda(x)=\operatorname{Re}W(x)\).  The derivative terms forced by
      chirality do not restore any of the eliminated components; they give
      \[
        V_\mu\longmapsto V_\mu+\partial_\mu\Lambda,\qquad
        \lambda\longmapsto\lambda,\qquad D\longmapsto D.
      \]
      Wess--Zumino gauge therefore fixes the extended part of the gauge group,
      but not the ordinary gauge symmetry.
    \item \emph{Chirality and Gauge Invariance of \(W_\alpha\).}
    The Abelian definition is a chiral projection of a superderivative of
      \(V\).  Acting with \(\SuperD_R\) gives zero because the projected product
      contains three identical right-chiral Grassmann derivatives, of which
      only two are independent.  Under
      \(\delta V=\tfrac i2(\Omega-\Omega^*)\), the term containing \(\Omega\)
      vanishes because \(\Omega\) is left chiral, and the term containing
      \(\Omega^*\) vanishes after the opposite chiral projection.  Thus
      \[
        \SuperD_RW_L=0,\qquad \delta W_L=0.
      \]
      The two ingredients are chirality,
      \[
        \SuperD_R\Omega=0,\qquad \SuperD_L\Omega^*=0,
      \]
      and the nilpotence and anticommutation algebra of the covariant
      superderivatives.
    \item \emph{Recovering the Abelian Components.}
    On the left-chiral slice only the left-chiral Grassmann coordinate
      remains.  Applying the field-strength projection to
      Eq.~\eqref{eq:27.1.19} gives
      \[
        W_L=\lambda_L(x_+)
          +\frac12\gamma^\mu\gamma^\nu\theta_Lf_{\mu\nu}(x_+)
          -i\theta_LD(x_+)
          +(\theta_L^{\rm T}\epsilon\theta_L)
             \sl{\partial}\lambda_R(x_+).
      \]
      The symmetric part of \(\gamma^\mu\gamma^\nu\) drops out against the
      antisymmetry of \(f_{\mu\nu}\).  The derivative term arises from the
      shifted coordinate \(x_+\), not from a new independent component.
      This reproduces Eq.~\eqref{eq:27.2.11}; complex conjugation gives the
      right-chiral expression.
  \end{enumerate}

  \SupplementarySolution{2}{The Superspace Bianchi Identity and Abelian Gauge Action}
  \begin{enumerate}
    \item \emph{The Superspace Bianchi Identity.}
    Substitute the two chiral projections of the same real \(V\) into the
      two sides of Eq.~\eqref{eq:27.2.20}.  Moving superderivatives past one
      another with their anticommutator reduces both sides to the same
      spacetime derivative of \(V\).  At tensor order the remaining statement
      is
      \[
        \partial_\rho f_{\mu\nu}
         +\partial_\mu f_{\nu\rho}
         +\partial_\nu f_{\rho\mu}=0,
        \qquad
        \epsilon^{\mu\nu\rho\sigma}\partial_\rho f_{\mu\nu}=0.
      \]
      Thus \(f\) is a closed two-form.  On a contractible patch the Poincare
      lemma gives
      \(f_{\mu\nu}=\partial_\mu V_\nu-\partial_\nu V_\mu\).
    \item \emph{The Abelian Gauge Action and the Theta Term.}
    Multiplying the expansion of \(W_L\) and extracting its top component
      yields, up to a spacetime derivative,
      \[
        -\frac12\operatorname{Re}
          [W_L^{\rm T}\epsilon W_L]_{\mathcal F}
          =-\frac14f_{\mu\nu}f^{\mu\nu}
           -\frac12\bar\lambda\sl{\partial}\lambda+\frac12D^2.
      \]
      The imaginary part contains no new local field equation:
      \[
        \operatorname{Im}
          [W_L^{\rm T}\epsilon W_L]_{\mathcal F}
          \ \propto\
          \epsilon^{\mu\nu\rho\sigma}f_{\mu\nu}f_{\rho\sigma}
          =4\partial_\mu\!
            \left(\epsilon^{\mu\nu\rho\sigma}
              V_\nu\partial_\rho V_\sigma\right).
      \]
      The gaugino derivative also differs from its manifestly real form only
      by an integration by parts.  These are the component terms in the
      gauge action at the end of Section 27.2.
  \end{enumerate}

  \SupplementarySolution{3}{Non-Abelian Matter and Gauge Covariance}
  \begin{enumerate}
    \item \emph{Gauge-Invariant Matter Kinetic Term.}
    Write \(U=e^{i\Omega}\).  The defining finite laws imply that the
      covariant row made from \(\Phi^\dagger\) and the connection transforms
      contragrediently:
      \[
        \Phi'=U\Phi,\qquad
        (\Phi^\dagger\Gamma)'=(\Phi^\dagger\Gamma)U^{-1}.
      \]
      Adjacent inverse factors therefore cancel and
      \((\Phi^\dagger\Gamma\Phi)'=\Phi^\dagger\Gamma\Phi\).
      Its \(D\)-component is therefore invariant.  Without \(\Gamma\),
      \(\Phi^{\dagger\prime}\Phi'\) contains unequal chiral and antichiral
      transformation matrices, which do not cancel because a local chiral
      parameter is not real.  For an ordinary transformation
      \(\Omega=\Omega^*=\Lambda(x)\), the familiar cancellation is recovered.
    \item \emph{The Covariant Non-Abelian Field Strength.}
    The definition uses \(\Gamma\) and its inverse so that all derivatives of
      the chiral gauge parameter cancel.  What remains is homogeneous
      conjugation,
      \[
        W_L\longmapsto e^{-i\Omega}W_Le^{i\Omega}.
      \]
      Cyclicity of the trace then gives
      \[
        \operatorname{Tr}(W_L^{\rm T}\epsilon W_L)
          \longmapsto
        \operatorname{Tr}(W_L^{\rm T}\epsilon W_L).
      \]
      Expanding to the term linear in \(\theta_L\), the commutator part supplies
      \[
        f_{A\mu\nu}=\partial_\mu V_{A\nu}-\partial_\nu V_{A\mu}
          +C_{ABC}V_{B\mu}V_{C\nu}.
      \]
      Thus both the superfield and its component tensor transform covariantly.
    \item \emph{Closure of the Non-Abelian Bianchi Identity.}
    Acting on any representation,
      \([D_\mu,D_\nu]=-it_Af^A_{\mu\nu}\).  The Jacobi identity gives
      \[
        [D_\rho,[D_\mu,D_\nu]]+\text{cyclic}=0
        \quad\Longrightarrow\quad D_{[\rho}f_{\mu\nu]}=0.
      \]
      Therefore
      \[
        \epsilon^{\mu\nu\rho\sigma}(D_\rho f_{\mu\nu})_A=0.
      \]
      This is the tensor component of the covariant superspace Bianchi
      identity.  Unlike the Abelian equation it does not imply
      \(f=dV\) without the commutator term; it instead fixes the curvature of
      the non-Abelian connection.
    \item \emph{Gauge Invariance of the Superpotential.}
    An infinitesimal transformation gives
      \[
        0=\delta f=i\Lambda_A f_n(t_A\phi)_n,
        \qquad f_n(t_A\phi)_n=0.
      \]
      Differentiating with respect to \(\phi_m\) yields
      \[
        f_{mn}(t_A\phi)_n+f_n(t_A)_{nm}=0.
      \]
      At an \(F\)-flat vacuum the second term vanishes, so
      \[
        \mathcal M_{mn}(t_A\phi_0)_n=0.
      \]
      The chiral fermion tangent to every broken complexified gauge direction
      is therefore a zero mode of the chiral mass matrix before it mixes with
      the corresponding gaugino.
  \end{enumerate}

  \SupplementarySolution{4}{Matrix \(D\)-Flatness in \(U(N_c)\) Gauge Theory}
Define the Hermitian color matrix
  \[
    H^a{}_b
      =\sum_iq_i^{*a}q_{ib}
        -\sum_i\widetilde q_i^{\,a}\widetilde q_b^{*i}.
  \]
  The auxiliary condition is simply
  \(D^A=\operatorname{tr}(T^AH)=0\).

  Let \(E_{ab}\) denote the matrix with a single unit in row \(a\), column
  \(b\).  A real basis of all Hermitian matrices is
  \[
    E_{aa},\qquad
    \frac{E_{ab}+E_{ba}}{\sqrt2},\qquad
    \frac{-i(E_{ab}-E_{ba})}{\sqrt2}
    \quad (a<b).
  \]
  There are
  \(N_c+2\binom{N_c}{2}=N_c^2\) such matrices, including the trace
  generator of \(U(N_c)\).  Taking \(T^A\) successively from this basis,
  \(D^A=0\) sets every diagonal entry of \(H\), every real part of an
  off-diagonal entry, and every imaginary part of an off-diagonal entry to
  zero.  Hence \(H=0\).

  Conversely, \(H=0\) plainly implies
  \(\operatorname{tr}(T^AH)=0\) for every generator.  Therefore the
  component conditions are equivalent to
  \[
    \sum_iq_i^{*a}q_{ib}
      =\sum_i\widetilde q_i^{\,a}\widetilde q_b^{*i}
      \quad\text{for all }a,b,
  \]
  as required.

  \SupplementarySolution{5}{Supersymmetric Higgsing and Massive Vector Multiplets}
  \begin{enumerate}
    \item \emph{Unbroken Generators and Vector Masses.}
    Put \(\phi=\phi_0+\varphi\) in
      \(-|D_\mu\phi|^2\).  Its term quadratic in constant gauge fields is
      \[
        -\frac12V_A^\mu
           \left(\phi_0^\dagger\{t_A,t_B\}\phi_0\right)V_{B\mu},
      \]
      so the bracket is \((\mu^2)_{AB}\).  For real \(c_A\),
      \[
        c_A(\mu^2)_{AB}c_B
           =2\left\|\left(c_At_A\right)\phi_0\right\|^2.
      \]
      Positivity shows that this vanishes precisely when
      \(c_At_A\phi_0=0\), which is exactly the condition that the corresponding
      generator leave the vacuum invariant.
    \item \emph{The Fermions of a Massive Vector Multiplet.}
    For a broken generator \(A\), the orbit tangent is
      \(u_A=t_A\phi_0\).  Gauge invariance and \(F\)-flatness give
      \(\mathcal M u_A=0\), while the Yukawa term mixes
      \(u_A^\dagger\psi_L/\|u_A\|\) with \(\lambda_A\).  Its coefficient is
      \(\sqrt2\|u_A\|\), or the corresponding singular value when broken
      directions mix.  The same singular values are obtained from
      \[
        (\mu^2)_{AB}=2\operatorname{Re}(u_A^\dagger u_B).
      \]
      Thus the orbit fermion and gaugino form a Dirac fermion whose mass equals
      the vector mass.  The eaten real scalar supplies the longitudinal vector
      polarization, and the orthogonal real scalar completes the massive
      \(N=1\) vector multiplet.
  \end{enumerate}

  \SupplementarySolution{6}{Supersymmetry of Pure Yang--Mills Theory}
Because the action is real it is enough to keep the parameter
  \(\epsilon\) and set \(\bar\epsilon=0\); the conjugate calculation then
  supplies the other half.  In this sector
  \[
    \delta_\epsilon A_\mu=\epsilon\sigma_\mu\bar\lambda,
    \qquad
    \delta_\epsilon\lambda=\sigma^{\rho\sigma}\epsilon F_{\rho\sigma},
    \qquad
    \delta_\epsilon\bar\lambda=0.
  \]
  Gauge covariance gives
  \[
    \delta F_{\mu\nu}
      =D_\mu(\epsilon\sigma_\nu\bar\lambda)
       -D_\nu(\epsilon\sigma_\mu\bar\lambda).
  \]
  After one integration by parts, the gauge-field variation is proportional
  to
  \((D_\mu F^{\mu\nu})\epsilon\sigma_\nu\bar\lambda\), plus a divergence.

  Varying \(\lambda\) in the fermion kinetic term and using
  \[
    \sigma^\nu\bar\sigma^\mu\sigma^\rho
      =\eta^{\mu\nu}\sigma^\rho+\eta^{\mu\rho}\sigma^\nu
       -\eta^{\nu\rho}\sigma^\mu
       +i\epsilon^{\nu\mu\rho\kappa}\sigma_\kappa
  \]
  produces the negative of that equation-of-motion term.  The remaining
  derivative terms combine, up to the same total divergence, into
  \[
    \epsilon\sigma_\nu\bar\lambda\,
      D_\mu\,{}^\star F^{\mu\nu}.
  \]
  This vanishes by the non-Abelian Bianchi identity
  \(D_{[\mu}F_{\nu\rho]}=0\).

  There is one further contribution because \(D_\mu\bar\lambda\) contains
  the gauge connection.  It is proportional to
  \[
    \operatorname{Tr}\!
      \left(\lambda\sigma^\mu
        [\,\epsilon\sigma_\mu\bar\lambda,\bar\lambda\,]\right).
  \]
  The two-component Fierz identity converts the product of the two
  \(\bar\lambda\)'s into its antisymmetric spinor contraction; cyclicity of
  the color trace then makes the two commutator terms equal and opposite.
  Thus the cubic contribution vanishes.  The entire
  \(\epsilon\)-variation is a boundary term, and complex conjugation proves
  invariance under the full transformation.

  \SupplementarySolution{7}{Fayet--Iliopoulos Breaking, Goldstino, and Supertrace}
  \begin{enumerate}
    \item \emph{The Two Phases of Fayet--Iliopoulos SQED.}
    Write \(x=|\phi_+|^2\), \(y=|\phi_-|^2\), and
      \(D=\xi+q(x-y)\).  At an interior stationary point,
      \(m^2+qD=0\) if \(x\ne0\) and
      \(m^2-qD=0\) if \(y\ne0\); both cannot hold for \(m\ne0\).
      Hence
      \[
      \begin{array}{c|c|c}
        \text{range}&\text{vacuum}&V_{\min}\\ \hline
        |\xi|\le |m|^2/|q|&x=y=0&\frac12\xi^2\\
        \xi>|m|^2/q&x=0,\,
          y=\dfrac{\xi-|m|^2/q}{q}
          &\dfrac{|m|^2\xi}{q}-\dfrac{|m|^4}{2q^2}\\
        \xi<-|m|^2/q&y=0,\,
          x=\dfrac{-\xi-|m|^2/q}{q}
          &-\dfrac{|m|^2\xi}{q}-\dfrac{|m|^4}{2q^2}.
      \end{array}
      \]
      Here \(q>0\) was chosen without loss of generality.  The gauge group is
      unbroken in the first phase and Higgsed in the other two.  For
      \(\xi\ne0\), every listed vacuum has positive energy, so supersymmetry is
      broken in both phases.
    \item \emph{The General Goldstino Direction.}
    Multiplying the mass-squared matrix in Eq.~\eqref{eq:27.5.6} by
      \((i\sqrt2\mathcal F_0,D_0)\), the chiral block cancels by the
      stationarity condition \eqref{eq:27.5.13}, and the gauge block cancels
      by Eq.~\eqref{eq:27.5.14}.  Therefore
      \[
        M^\dagger M
           \binom{i\sqrt2\mathcal F_0}{D_0}=0.
      \]
      With
      \[
        \mathcal N^2=2\sum_n|\mathcal F_{n0}|^2+\sum_AD_{A0}^2,
      \]
      the original fields contain the normalized goldstino as
      \[
        \psi_{nL}=\frac{i\sqrt2\mathcal F_{n0}}{\mathcal N}G_L+\cdots,
        \qquad
        \lambda_{AL}=\frac{D_{A0}}{\mathcal N}G_L+\cdots.
      \]
      The chiral and gaugino fractions are respectively
      \(2\sum|\mathcal F_0|^2/\mathcal N^2\) and
      \(\sum D_0^2/\mathcal N^2\).
    \item \emph{The Tree-Level Mass Supertrace.}
    The off-diagonal blocks do not contribute to a trace.  The three mass
      matrices give
      \[
      \begin{aligned}
        \operatorname{Tr}M_0^2
          &=2\operatorname{Tr}(\mathcal M^*\mathcal M)
            +\operatorname{Tr}\mu^2
            +2D_{A0}\operatorname{Tr}t_A,\\
        \operatorname{Tr}(M^\dagger M)
          &=\operatorname{Tr}(\mathcal M^*\mathcal M)
            +2\operatorname{Tr}\mu^2.
      \end{aligned}
      \]
      Weighting a real scalar, Majorana fermion, and vector by \(1,-2,3\)
      yields
      \[
        \operatorname{STr}M^2=-2D_{A0}\operatorname{Tr}t_A.
      \]
      Semisimple generators are traceless.  For a \(U(1)\),
      \(\operatorname{Tr}t\) is the mixed gauge--gravitational anomaly
      coefficient, so consistency sets it to zero and hence
      \(\operatorname{STr}M^2=0\).
  \end{enumerate}

  \SupplementarySolution{8}{Wilsonian Gauge Couplings and FI Protection}
  \begin{enumerate}
    \item \emph{A Holomorphic Non-Renormalization Argument.}
    Regard every \(m_{ij}\) and \(y_{ijk}\) as the lowest component of a
      background chiral multiplet.  Assign spurion charges so that the
      classical superpotential is invariant under independent phase rotations
      of the matter fields.  A Wilsonian superpotential must be holomorphic in
      \(\Phi,m,y\), have dimension three and \(R\)-charge two, and be regular
      when the couplings vanish.  These conditions allow only the monomials
      already present classically.  Perturbative inverse powers or logarithms
      of chiral couplings would be singular at the free point, and
      antiholomorphic couplings are forbidden.  Thus perturbation theory
      renormalizes wave functions but generates no new Wilsonian
      superpotential.
    \item \emph{One-Loop Exactness of the Holomorphic Gauge Coupling.}
    Combine the inverse squared coupling and theta angle into the lowest
      component of a chiral parameter \(\tau\).  Holomorphy and the periodicity
      of the theta angle restrict a perturbative Wilsonian correction to a
      constant plus a single logarithm of the scale.  Its coefficient is fixed
      by the one-loop chiral anomaly.  Higher-loop powers would violate the
      spurionic shift symmetry of \(\tau\).  Therefore the holomorphic
      Wilsonian coupling is one-loop exact.  Canonical normalization requires
      anomalous, coupling-dependent rescalings of matter and vector fields, so
      the beta function of the physical canonical coupling can and generally
      does contain higher-loop terms.
    \item \emph{Why the FI Term Is Not Perturbatively Generated.}
    The auxiliary \(D\) couples to a chiral scalar with coefficient \(t\).
      Closing the scalar line gives
      \[
        \Gamma_D^{(1)}\propto
          D\sum_i(t)_i{}^i\int^\Lambda\frac{d^4p}{p^2}
          =D\,\operatorname{Tr}t\int^\Lambda\frac{d^4p}{p^2}.
      \]
      Thus the possible quadratically divergent FI counterterm is
      proportional to \(\operatorname{Tr}t\).  For a \(U(1)\) the same sum of
      charges is the mixed gauge--gravitational anomaly coefficient.  Anomaly
      cancellation sets it to zero.  Supersymmetric Ward identities and the
      non-renormalization argument of Section 27.6 then prevent higher-loop
      generation when the bare FI constant vanishes.
  \end{enumerate}

  \SupplementarySolution{9}{Soft Breaking and Power Counting}
  \begin{enumerate}
    \item \emph{Classifying Soft Breaking Terms.}
    Up to Hermitian conjugation, the standard gauge-invariant soft
      Lagrangian is
      \[
        \begin{aligned}
          -\mathcal L_{\rm soft}
            &=(m^2)^i{}_j\phi_i^*\phi^j\\
            &\quad+\left[
              \frac12M_A(\lambda_{AL}^{\mathsf T}\epsilon\lambda_{AL})
              +\frac16a_{ijk}\phi_i\phi_j\phi_k \right.\\
            &\hspace{5.5em}\left.
              +\frac12b_{ij}\phi_i\phi_j+t_i\phi_i
              +\text{c.c.}\right].
        \end{aligned}
      \]
      Gauge invariance restricts every tensor exactly as for the corresponding
      supersymmetric operator; \(t_i\) requires a singlet.  Scalar masses come
      from a real spurion with a \(\theta^2\bar\theta^2\) component.  Gaugino
      masses and holomorphic \(a,b,t\) terms come from chiral spurions with
      nonzero \(\mathcal F\)-components.  Freezing these backgrounds breaks
      supersymmetry explicitly while keeping the bookkeeping manifestly
      supersymmetric.
    \item \emph{Power Counting for Softness.}
    A scalar mass insertion has dimension two and a gaugino mass or
      holomorphic trilinear insertion has dimension three.  Relative to a
      dimension-four vertex, each insertion lowers the superficial degree of
      divergence by at least two or one, respectively.  Supersymmetry cancels
      the quadratic divergence before insertion; after insertion the remaining
      diagrams are at worst logarithmically divergent in the corresponding
      soft parameters.  A generic dimension-four breaking coupling supplies
      no such reduction, so repeated insertions can leave a quadratic scalar
      divergence.  This power-counting distinction is the meaning of
      ``soft''.
  \end{enumerate}

  \SupplementarySolution{10}{The Second Supersymmetry of Pure \(N=2\) Yang--Mills}
In the adjoint representation
  \((t_A)_{BC}=-iC_{ABC}\).  Substitution in
  Eq.~\eqref{eq:27.4.1} gives Eq.~\eqref{eq:27.9.3}: the two Majorana
  fields \(\psi_A,\lambda_A\) have identical covariant kinetic terms, and
  the Yukawa coupling is rotated into itself by their \(SU(2)\) doublet
  rotation.  The three real auxiliaries
  \((\operatorname{Re}\mathcal F_A,\operatorname{Im}\mathcal F_A,D_A)\)
  rotate so that their quadratic terms and their couplings to \(\phi\)
  are invariant.  Conjugating the known \(N=1\) supersymmetry by the
  exchange \(\psi_A\leftrightarrow\lambda_A\) gives a second transformation
  with an independent Majorana parameter.  The action therefore has
  \(N=2\) supersymmetry.

  \SupplementarySolution{11}{Allowed \(N=2\) Hypermultiplet Couplings}
Let \(Q_n,\widetilde Q_m\) be the two \(N=1\) chiral fields and
  \(\Phi_A\) the adjoint chiral field in the \(N=2\) vector multiplet.
  Gauge invariance pairs \(Q\) with \(\widetilde Q\); the \(SU(2)\)
  automorphism rotating the two supersymmetries fixes the adjoint Yukawa
  coupling to the gauge representation matrices.  The allowed
  superpotential is therefore
  \[
    f=\frac12(s_A)_{nm}Q_n\widetilde Q_m\Phi_A
       +\frac12\mu_{nm}Q_n\widetilde Q_m.
  \]
  Terms with two adjoint fields would give \(\psi_A\) masses or
  interactions without matching ones for \(\lambda_A\).  Cubics made only
  from hypermultiplet fields and bilinears \(QQ\) or
  \(\widetilde Q\widetilde Q\) similarly violate the required \(SU(2)\)
  pairing.  Renormalizability excludes higher powers, leaving exactly
  Eq.~\eqref{eq:27.9.29}.

  \SupplementarySolution{12}{The \(SU(3)\) Coulomb Branch}
  \begin{enumerate}
    \item \emph{The Coulomb Branch of Pure \(N=2\) Theory.}
    Eliminating the auxiliaries gives, in matrix notation,
      \[
        V=\frac12\operatorname{Tr}
           \bigl([\phi^\dagger,\phi]^\dagger[\phi^\dagger,\phi]\bigr)
          \ge0
      \]
      up to the normalization of the generators.  Since
      \[
        [\phi^\dagger,\phi]
           =2i[\operatorname{Re}\phi,\operatorname{Im}\phi],
      \]
      the potential vanishes exactly when the real and imaginary parts
      commute.  They are commuting Hermitian adjoint matrices and can therefore
      be simultaneously conjugated into a Cartan subalgebra.  Dividing by the
      residual Weyl group gives the classical Coulomb-branch moduli space.
    \item \emph{Root-Space Masses on the \(SU(3)\) Coulomb Branch.}
    The off-diagonal generator \(E_{ij}\) obeys
      \[
        [\phi_\infty,E_{ij}]=(a_i-a_j)E_{ij}.
      \]
      Hence, with the normalization of Section 27.9, the vector multiplet
      associated with the roots \(\pm(e_i-e_j)\) has
      \[
        M_{ij}=\sqrt2\,g\,|a_i-a_j|,\qquad i<j.
      \]
      For three distinct eigenvalues all six are massive and the unbroken
      group is \(U(1)^2\).  On any of the three walls \(a_i=a_j\), one root
      pair becomes massless and the group is \(SU(2)\times U(1)\).  Tracelessness
      makes simultaneous equality of all three eigenvalues possible only at
      the origin, where the full \(SU(3)\) is restored.
  \end{enumerate}

  \SupplementarySolution{13}{Surface Central Charge and the Dyonic BPS Spectrum}
  \begin{enumerate}
    \item \emph{Central Charge as a Surface Flux.}
    In a rank-\(r\) vacuum choose orthonormal Cartan generators and write
      \(\phi_\infty=v_IH_I\).  Only the Cartan field strengths survive at
      spatial infinity.  Substitution in the surface integral
      \eqref{eq:27.9.20} yields
      \[
        Z_{12}=2\sqrt2\sum_{I=1}^r
          v_I(iq_I-\mathcal M_I),\qquad
        q_I=\int dS_iE_I^i,\quad
        \mathcal M_I=\int dS_iB_I^i .
      \]
      Under an orthogonal change of Cartan basis, each of the three vectors
      \(v,q,\mathcal M\) is multiplied by the same orthogonal matrix.  Their
      scalar products, and hence \(Z_{12}\), are unchanged.
    \item \emph{Electric, Magnetic, and Dyonic BPS Masses.}
    The \(N=2\) algebra gives \(M\ge|Z_{12}|/2\).  With
      Eq.~\eqref{eq:27.9.22},
      \[
        M\ge\sqrt2\,|v|\,|iq-\mathcal M|.
      \]
      Thus a purely electric state of charge \(q\) obeys
      \(M\ge\sqrt2|vq|\).  A monopole with
      \(\mathcal M=4\pi\nu/e\) obeys
      \[
        M\ge\frac{4\pi\sqrt2\,|\nu v|}{|e|},
      \]
      and a dyon obeys
      \[
        M\ge\sqrt2\,|v|\sqrt{q^2+\mathcal M^2}.
      \]
      Equality holds precisely for a shortened half-BPS multiplet.  The
      elementary charged vector multiplets and the Bogomol'nyi monopoles
      described in Section 27.9 saturate their respective bounds.
  \end{enumerate}

\end{enumerate}
