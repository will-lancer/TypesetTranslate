\begin{enumerate}
  \SupplementarySolution{1}{Vector Superfields and Supersymmetric Maxwell Theory}
The lowest components of the shift
  \(\delta V=i(\Omega-\Omega^\dagger)\) give
  \[
    \delta C=i(\omega-\omega^*),\qquad
    \delta\chi=i\sqrt2\,\psi,\qquad
    \delta M=F,\qquad
    \delta A_\mu=-\partial_\mu(\omega+\omega^*).
  \]
  Thus \(\operatorname{Im}\omega,\psi,F\) may be chosen to set
  \(C=\chi=M=0\).  This is Wess--Zumino gauge:
  \[
    V_{\mathrm{WZ}}
      =\theta s^\mu\bar\theta\,A_\mu
       +\theta^2\bar\theta\,\bar\lambda
       +\bar\theta^2\theta\,\lambda
       +\frac12\theta^2\bar\theta^2D.
  \]
  The unused real part of \(\omega\) remains.  With
  \(\Lambda=-2\operatorname{Re}\omega\), it acts as
  \[
    A_\mu\longmapsto A_\mu+\partial_\mu\Lambda,
    \qquad \lambda\longmapsto\lambda,\qquad D\longmapsto D.
  \]
  Hence the extended superfield gauge freedom removes the extra fields,
  while its residual freedom is the ordinary Abelian gauge transformation.

  The projection
  \[
    W_a=-\frac14\bar{\mathcal D}^{\,2}\mathcal D_aV
  \]
  is chiral because three identical dotted Grassmann derivatives vanish:
  \(\bar{\mathcal D}_{\dot a}W_b=0\).  It is gauge invariant as well.
  Indeed, \(\bar{\mathcal D}^{\,2}\mathcal D_a\Omega=0\) for chiral
  \(\Omega\), while
  \(\mathcal D_a\Omega^\dagger=0\) for antichiral
  \(\Omega^\dagger\).  Therefore the \(F\)-term
  \[
    S_{\mathrm{gauge}}
      =\frac1{4e^2}\int d^4x
       \left[\int d^2\theta\,W^aW_a
             +\int d^2\bar\theta\,
                \overline W_{\dot a}\overline W^{\dot a}\right]
  \]
  is both supersymmetric and gauge invariant.  Its component expansion
  gives the displayed Maxwell--gaugino action, together with the auxiliary
  term \(D^2/(2e^2)\); eliminating \(D\) gives the on-shell action in the
  question.

  It remains to check that action directly.  Antisymmetry of \(F_{\mu\nu}\)
  and one integration by parts give
  \[
  \begin{aligned}
    \delta\mathcal L_{\mathrm{Maxwell}}
      &=-\frac1{2e^2}F^{\mu\nu}\delta F_{\mu\nu}\\
      &=\frac1{e^2}(\partial_\mu F^{\mu\nu})
         \bigl(\epsilon s_\nu\bar\lambda
               +\lambda s_\nu\bar\epsilon\bigr)
        +\partial_\mu K_1^\mu .
  \end{aligned}
  \]
  Varying both gauginos and using the supplied triple-sigma identity yields
  \[
  \begin{aligned}
    \delta\mathcal L_{\mathrm{gaugino}}
      ={}&-\frac1{e^2}(\partial_\mu F^{\mu\nu})
         \bigl(\epsilon s_\nu\bar\lambda
               +\lambda s_\nu\bar\epsilon\bigr)\\
       &+\frac{i}{2e^2}\epsilon_{\mathrm s}^{\mu\nu\rho\sigma}
          (\partial_\mu F_{\nu\rho})
          \bigl(\epsilon s_\sigma\bar\lambda
                -\lambda s_\sigma\bar\epsilon\bigr)
        +\partial_\mu K_2^\mu .
  \end{aligned}
  \]
  The first line cancels the Maxwell variation.  The second vanishes by
  \(\partial_{[\mu}F_{\nu\rho]}=0\), and the remaining terms are a boundary
  term.  Hence \(\delta S=0\).

  \SupplementarySolution{2}{The Superspace Bianchi Identity and Abelian Gauge Action}
  \begin{enumerate}
    \item \emph{The Superspace Bianchi Identity.}
    Substitute the two chiral projections of the same real \(V\) into the
      two sides of Eq.~\eqref{eq:27.2.20}.  Moving superderivatives past one
      another with their anticommutator reduces both sides to the same
      spacetime derivative of \(V\).  At tensor order the remaining statement
      is
      \[
        \partial_\rho f_{\mu\nu}
         +\partial_\mu f_{\nu\rho}
         +\partial_\nu f_{\rho\mu}=0,
        \qquad
        \epsilon^{\mu\nu\rho\sigma}\partial_\rho f_{\mu\nu}=0.
      \]
      Thus \(f\) is a closed two-form.  On a contractible patch the Poincare
      lemma gives
      \(f_{\mu\nu}=\partial_\mu V_\nu-\partial_\nu V_\mu\).
    \item \emph{The Abelian Gauge Action and the Theta Term.}
    Multiplying the expansion of \(W_L\) and extracting its top component
      yields, up to a spacetime derivative,
      \[
        -\frac12\operatorname{Re}
          [W_L^{\mathsf T}\epsilon W_L]_{\mathcal F}
          =-\frac14f_{\mu\nu}f^{\mu\nu}
           -\frac12\bar\lambda\sl{\partial}\lambda+\frac12D^2.
      \]
      The imaginary part contains no new local field equation:
      \[
        \operatorname{Im}
          [W_L^{\mathsf T}\epsilon W_L]_{\mathcal F}
          \ \propto\
          \epsilon^{\mu\nu\rho\sigma}f_{\mu\nu}f_{\rho\sigma}
          =4\partial_\mu\!
            \left(\epsilon^{\mu\nu\rho\sigma}
              V_\nu\partial_\rho V_\sigma\right).
      \]
      The gaugino derivative also differs from its manifestly real form only
      by an integration by parts.  These are the component terms in the
      gauge action at the end of Section 27.2.
  \end{enumerate}

  \SupplementarySolution{3}{Non-Abelian Matter and Gauge Covariance}
  \begin{enumerate}
    \item \emph{Gauge-Invariant Matter Kinetic Term.}
    Write \(U=e^{i\Omega}\).  The defining finite laws imply that the
      covariant row made from \(\Phi^\dagger\) and the connection transforms
      contragrediently:
      \[
        \Phi'=U\Phi,\qquad
        (\Phi^\dagger\Gamma)'=(\Phi^\dagger\Gamma)U^{-1}.
      \]
      Adjacent inverse factors therefore cancel and
      \((\Phi^\dagger\Gamma\Phi)'=\Phi^\dagger\Gamma\Phi\).
      Its \(D\)-component is therefore invariant.  Without \(\Gamma\),
      \(\Phi^{\dagger\prime}\Phi'\) contains unequal chiral and antichiral
      transformation matrices, which do not cancel because a local chiral
      parameter is not real.  For an ordinary transformation
      \(\Omega=\Omega^*=\Lambda(x)\), the familiar cancellation is recovered.
    \item \emph{The Covariant Non-Abelian Field Strength.}
    The definition uses \(\Gamma\) and its inverse so that all derivatives of
      the chiral gauge parameter cancel.  What remains is homogeneous
      conjugation,
      \[
        W_L\longmapsto e^{-i\Omega}W_Le^{i\Omega}.
      \]
      Cyclicity of the trace then gives
      \[
        \operatorname{Tr}(W_L^{\mathsf T}\epsilon W_L)
          \longmapsto
        \operatorname{Tr}(W_L^{\mathsf T}\epsilon W_L).
      \]
      Expanding to the term linear in \(\theta_L\), the commutator part supplies
      \[
        f_{A\mu\nu}=\partial_\mu V_{A\nu}-\partial_\nu V_{A\mu}
          +C_{ABC}V_{B\mu}V_{C\nu}.
      \]
      Thus both the superfield and its component tensor transform covariantly.
    \item \emph{Closure of the Non-Abelian Bianchi Identity.}
    Acting on any representation,
      \([D_\mu,D_\nu]=-it_Af^A_{\mu\nu}\).  The Jacobi identity gives
      \[
        [D_\rho,[D_\mu,D_\nu]]+\text{cyclic}=0
        \quad\Longrightarrow\quad D_{[\rho}f_{\mu\nu]}=0.
      \]
      Therefore
      \[
        \epsilon^{\mu\nu\rho\sigma}(D_\rho f_{\mu\nu})_A=0.
      \]
      This is the tensor component of the covariant superspace Bianchi
      identity.  Unlike the Abelian equation it does not imply
      \(f=dV\) without the commutator term; it instead fixes the curvature of
      the non-Abelian connection.
    \item \emph{Gauge Invariance of the Superpotential.}
    An infinitesimal transformation gives
      \[
        0=\delta f=i\Lambda_A f_n(t_A\phi)_n,
        \qquad f_n(t_A\phi)_n=0.
      \]
      Differentiating with respect to \(\phi_m\) yields
      \[
        f_{mn}(t_A\phi)_n+f_n(t_A)_{nm}=0.
      \]
      At an \(F\)-flat vacuum the second term vanishes, so
      \[
        \mathcal M_{mn}(t_A\phi_0)_n=0.
      \]
      The chiral fermion tangent to every broken complexified gauge direction
      is therefore a zero mode of the chiral mass matrix before it mixes with
      the corresponding gaugino.
  \end{enumerate}

  \SupplementarySolution{4}{Matrix \(D\)-Flatness in \(U(N_c)\) Gauge Theory}
Define the Hermitian color matrix
  \[
    H^a{}_b
      =\sum_iq_i^{*a}q_{ib}
        -\sum_i\widetilde q_i^{\,a}\widetilde q_b^{*i}.
  \]
  The auxiliary condition is simply
  \(D^A=\operatorname{tr}(T^AH)=0\).

  Let \(E_{ab}\) denote the matrix with a single unit in row \(a\), column
  \(b\).  A real basis of all Hermitian matrices is
  \[
    E_{aa},\qquad
    \frac{E_{ab}+E_{ba}}{\sqrt2},\qquad
    \frac{-i(E_{ab}-E_{ba})}{\sqrt2}
    \quad (a<b).
  \]
  There are
  \(N_c+2\binom{N_c}{2}=N_c^2\) such matrices, including the trace
  generator of \(U(N_c)\).  Taking \(T^A\) successively from this basis,
  \(D^A=0\) sets every diagonal entry of \(H\), every real part of an
  off-diagonal entry, and every imaginary part of an off-diagonal entry to
  zero.  Hence \(H=0\).

  Conversely, \(H=0\) plainly implies
  \(\operatorname{tr}(T^AH)=0\) for every generator.  Therefore the
  component conditions are equivalent to
  \[
    \sum_iq_i^{*a}q_{ib}
      =\sum_i\widetilde q_i^{\,a}\widetilde q_b^{*i}
      \quad\text{for all }a,b,
  \]
  as required.

  \SupplementarySolution{5}{Supersymmetric Higgsing and Massive Vector Multiplets}
  \begin{enumerate}
    \item \emph{Unbroken Generators and Vector Masses.}
    Put \(\phi=\phi_0+\varphi\) in
      \(-|D_\mu\phi|^2\).  Its term quadratic in constant gauge fields is
      \[
        -\frac12V_A^\mu
           \left(\phi_0^\dagger\{t_A,t_B\}\phi_0\right)V_{B\mu},
      \]
      so the bracket is \((\mu^2)_{AB}\).  For real \(c_A\),
      \[
        c_A(\mu^2)_{AB}c_B
           =2\left\|\left(c_At_A\right)\phi_0\right\|^2.
      \]
      Positivity shows that this vanishes precisely when
      \(c_At_A\phi_0=0\), which is exactly the condition that the corresponding
      generator leave the vacuum invariant.
    \item \emph{The Fermions of a Massive Vector Multiplet.}
    For a broken generator \(A\), the orbit tangent is
      \(u_A=t_A\phi_0\).  Gauge invariance and \(F\)-flatness give
      \(\mathcal M u_A=0\), while the Yukawa term mixes
      \(u_A^\dagger\psi_L/\|u_A\|\) with \(\lambda_A\).  Its coefficient is
      \(\sqrt2\|u_A\|\), or the corresponding singular value when broken
      directions mix.  The same singular values are obtained from
      \[
        (\mu^2)_{AB}=2\operatorname{Re}(u_A^\dagger u_B).
      \]
      Thus the orbit fermion and gaugino form a Dirac fermion whose mass equals
      the vector mass.  The eaten real scalar supplies the longitudinal vector
      polarization, and the orthogonal real scalar completes the massive
      \(N=1\) vector multiplet.
  \end{enumerate}

  \SupplementarySolution{6}{Supersymmetry of Pure Yang--Mills Theory}
Because the action is real it is enough to keep the parameter
  \(\epsilon\) and set \(\bar\epsilon=0\); the conjugate calculation then
  supplies the other half.  In this sector
  \[
    \delta_\epsilon A_\mu=\epsilon s_\mu\bar\lambda,
    \qquad
    \delta_\epsilon\lambda=\Sigma^{\rho\sigma}\epsilon F_{\rho\sigma},
    \qquad
    \delta_\epsilon\bar\lambda=0.
  \]
  Gauge covariance gives
  \[
    \delta F_{\mu\nu}
      =D_\mu(\epsilon s_\nu\bar\lambda)
       -D_\nu(\epsilon s_\mu\bar\lambda).
  \]
  After one integration by parts, the gauge-field variation is
  \[
    \delta\mathcal L_{\rm gauge}
      =2\operatorname{Tr}\!\left[
        (D_\mu F^{\mu\nu})\epsilon s_\nu\bar\lambda\right]
        +\partial_\mu(\cdots).
  \]

  The supplied triple-product identity gives
  \[
    \Sigma^{\rho\sigma}F_{\rho\sigma}s^\mu
      =iF^\mu{}_\nu s^\nu+{}^\star F^{\mu\nu}s_\nu .
  \]
  Therefore variation of \(\lambda\), followed by a covariant integration
  by parts, yields
  \[
  \begin{aligned}
    \delta_\lambda\mathcal L_{\rm ferm}
      ={}&-2\operatorname{Tr}\!\left[
        (D_\mu F^{\mu\nu})\epsilon s_\nu\bar\lambda\right]\\
       &+2i\operatorname{Tr}\!\left[
        (D_\mu{}^\star F^{\mu\nu})\epsilon s_\nu\bar\lambda\right]
        +\partial_\mu(\cdots).
  \end{aligned}
  \]
  The first line cancels \(\delta\mathcal L_{\rm gauge}\).  The second
  vanishes by the non-Abelian Bianchi identity
  \(D_{[\mu}F_{\nu\rho]}=0\).

  There is one further contribution because \(D_\mu\bar\lambda\) contains
  the gauge connection.  It is proportional to
  \[
    -2\operatorname{Tr}\!
      \left(\lambda s^\mu
        [\,\epsilon s_\mu\bar\lambda,\bar\lambda\,]\right).
  \]
  The two-component Fierz identity converts the product of the two
  \(\bar\lambda\)'s into its antisymmetric spinor contraction; cyclicity of
  the color trace then makes the two commutator terms equal and opposite.
  Thus the cubic contribution vanishes.  The entire
  \(\epsilon\)-variation is a boundary term, and complex conjugation proves
  invariance under the full transformation.

  \SupplementarySolution{7}{Chiral Superfields and a Gauged \(U(1)\) Model}
  \begin{enumerate}
    \item Chirality means
      \[
        \bar{\mathcal D}_{\dot a}\Phi=0.
      \]
      For example, the MSSM superfield \(H_u\) contains the physical complex
      Higgs scalar \(h_u\) and a left-handed Weyl higgsino
      \(\widetilde h_u\); its \(F\)-component is auxiliary rather than
      physical.

      The chiral constraint is solved by a function of \(y^\mu\) and
      \(\theta\) alone:
      \[
        \Phi(y,\theta)=\phi(y)+\sqrt2\,\theta\psi(y)+\theta^2F(y),
        \qquad y^\mu=x^\mu+i\theta s^\mu\bar\theta.
      \]
      Taylor expansion about \(x\), followed by the two-spinor identities
      for \(\theta^a\theta^b\), gives
      \[
      \begin{aligned}
        \Phi(x,\theta,\bar\theta)={}&
          \phi+\sqrt2\,\theta\psi+\theta^2F
          +i\theta s^\mu\bar\theta\,\partial_\mu\phi\\
        &-\frac{i}{\sqrt2}\theta^2
           \partial_\mu\psi\,s^\mu\bar\theta
          -\frac14\theta^2\bar\theta^2
           \Box_{\mathrm s}\phi .
      \end{aligned}
      \]
      Hence
      \[
        A=i,\qquad B=-\frac{i}{\sqrt2},\qquad C=-\frac14.
      \]

    \item Only the chiral fermions contribute to the Abelian gauge anomaly.
      The relevant one-loop graphs are the gauge-current triangle and its
      mixed gauge--gravitational counterpart:
      \begin{center}
      \begin{tikzpicture}[x=0.72cm,y=0.68cm,baseline=-0.4ex,font=\scriptsize]
        \begin{scope}
          \coordinate (a) at (0,0);
          \coordinate (b) at (1.5,1.45);
          \coordinate (c) at (3,0);
          \draw[thick,->] (a)--(b);
          \draw[thick,->] (b)--(c);
          \draw[thick,->] (c)--(a);
          \draw[thick] (a)--(-0.75,-0.45) node[left] {\(A_\mu\)};
          \draw[thick] (b)--(1.5,2.25) node[above] {\(A_\nu\)};
          \draw[thick] (c)--(3.75,-0.45) node[right] {\(A_\rho\)};
          \node at (1.5,0.55) {Weyl fermion};
          \node at (1.5,-0.72) {\(U(1)^3\)};
        \end{scope}
        \begin{scope}[xshift=5.7cm]
          \coordinate (a) at (0,0);
          \coordinate (b) at (1.5,1.45);
          \coordinate (c) at (3,0);
          \draw[thick,->] (a)--(b);
          \draw[thick,->] (b)--(c);
          \draw[thick,->] (c)--(a);
          \draw[thick] (a)--(-0.75,-0.45) node[left] {\(A_\mu\)};
          \draw[thick,double] (b)--(1.5,2.25) node[above] {\(h_{\nu\lambda}\)};
          \draw[thick,double] (c)--(3.75,-0.45) node[right] {\(h_{\rho\sigma}\)};
          \node at (1.5,0.55) {Weyl fermion};
          \node at (1.5,-0.72) {\(U(1)\text{--gravity}^2\)};
        \end{scope}
      \end{tikzpicture}
      \end{center}
      Their coefficients are
      \[
        \mathcal A_{U(1)^3}\propto\sum_iq_i^3
           =(+2)^3+0^3+(-2)^3=0,
        \qquad
        \mathcal A_{U(1)\text{-grav}^2}\propto\sum_iq_i=0.
      \]
      The Abelian gaugino is neutral.  Thus the two oppositely charged
      chiral multiplets cancel both anomalies, and the theory is
      gauge-anomaly free.

      Up to an irrelevant constant, the most general renormalizable
      gauge-invariant superpotential is
      \[
        W=aX_0+\frac12m_0X_0^2+mX_{2+}X_{2-}
          +\frac{\kappa}{3}X_0^3
          +\lambda X_0X_{2+}X_{2-}.
      \]
      Setting \(a=m_0=m=0\) leaves
      \[
        W=\frac{\kappa}{3}X_0^3+\lambda X_0X_{2+}X_{2-}.
      \]
      The perturbative non-renormalization theorem for the Wilsonian
      superpotential prevents the absent linear and quadratic holomorphic
      terms from being generated.

      Let \(x_0,x_+,x_-\) be the scalar components.  With the convention
      \(D=2|x_+|^2-2|x_-|^2+\xi\), the scalar potential is
      \[
      \begin{aligned}
        V={}&|\lambda x_0x_-|^2+|\lambda x_0x_+|^2
          +|\kappa x_0^2+\lambda x_+x_-|^2\\
          &+\frac{g^2}{2}
            \left(2|x_+|^2-2|x_-|^2+\xi\right)^2.
      \end{aligned}
      \]
      For generic nonzero \(\lambda,\kappa\), \(F_+=F_-=F_0=0\) first
      implies \(x_0=0\) and \(x_+x_-=0\).  The remaining \(D\)-flatness
      equation then has the solutions
      \[
      \begin{array}{c|c}
        \xi>0 & x_+=x_0=0,\quad |x_-|^2=\xi/2,\\[2pt]
        \xi<0 & x_-=x_0=0,\quad |x_+|^2=-\xi/2,\\[2pt]
        \xi=0 & x_+=x_-=x_0=0.
      \end{array}
      \]
      A gauge transformation removes the phase of the nonzero field.
      Consequently supersymmetry is unbroken in every case:
      \(V_{\min}=0\).  The \(U(1)\) gauge symmetry is Higgsed for
      \(\xi\ne0\) and unbroken at \(\xi=0\).  Nongeneric choices such as
      \(\lambda=0\) or \(\kappa=0\) introduce additional flat directions,
      but do not describe the most general cubic model.
  \end{enumerate}

  \SupplementarySolution{8}{Wilsonian Gauge Couplings and FI Protection}
  \begin{enumerate}
    \item \emph{A Holomorphic Non-Renormalization Argument.}
    Regard every \(m_{ij}\) and \(y_{ijk}\) as the lowest component of a
      background chiral multiplet.  Assign spurion charges so that the
      classical superpotential is invariant under independent phase rotations
      of the matter fields.  A Wilsonian superpotential must be holomorphic in
      \(\Phi,m,y\), have dimension three and \(R\)-charge two, and be regular
      when the couplings vanish.  These conditions allow only the monomials
      already present classically.  Perturbative inverse powers or logarithms
      of chiral couplings would be singular at the free point, and
      antiholomorphic couplings are forbidden.  Thus perturbation theory
      renormalizes wave functions but generates no new Wilsonian
      superpotential.
    \item \emph{One-Loop Exactness of the Holomorphic Gauge Coupling.}
    Combine the inverse squared coupling and theta angle into the lowest
      component of a chiral parameter \(\tau\).  Holomorphy and the periodicity
      of the theta angle restrict a perturbative Wilsonian correction to a
      constant plus a single logarithm of the scale.  Its coefficient is fixed
      by the one-loop chiral anomaly.  Higher-loop powers would violate the
      spurionic shift symmetry of \(\tau\).  Therefore the holomorphic
      Wilsonian coupling is one-loop exact.  Canonical normalization requires
      anomalous, coupling-dependent rescalings of matter and vector fields, so
      the beta function of the physical canonical coupling can and generally
      does contain higher-loop terms.
    \item \emph{Why the FI Term Is Not Perturbatively Generated.}
    The auxiliary \(D\) couples to a chiral scalar with coefficient \(t\).
      Closing the scalar line gives
      \[
        \Gamma_D^{(1)}\propto
          D\sum_i(t)_i{}^i\int^\Lambda\frac{d^4p}{p^2}
          =D\,\operatorname{Tr}t\int^\Lambda\frac{d^4p}{p^2}.
      \]
      Thus the possible quadratically divergent FI counterterm is
      proportional to \(\operatorname{Tr}t\).  For a \(U(1)\) the same sum of
      charges is the mixed gauge--gravitational anomaly coefficient.  Anomaly
      cancellation sets it to zero.  Supersymmetric Ward identities and the
      non-renormalization argument of Section 27.6 then prevent higher-loop
      generation when the bare FI constant vanishes.
  \end{enumerate}

  \SupplementarySolution{9}{Soft Breaking and Power Counting}
  \begin{enumerate}
    \item \emph{Classifying Soft Breaking Terms.}
    Up to Hermitian conjugation, the standard gauge-invariant soft
      Lagrangian is
      \[
        \begin{aligned}
          -\mathcal L_{\rm soft}
            &=(m^2)^i{}_j\phi_i^*\phi^j\\
            &\quad+\left[
              \frac12M_A(\lambda_{AL}^{\mathsf T}\epsilon\lambda_{AL})
              +\frac16a_{ijk}\phi_i\phi_j\phi_k \right.\\
            &\hspace{5.5em}\left.
              +\frac12b_{ij}\phi_i\phi_j+t_i\phi_i
              +\mathrm{c.c.}\right].
        \end{aligned}
      \]
      Gauge invariance restricts every tensor exactly as for the corresponding
      supersymmetric operator; \(t_i\) requires a singlet.  Scalar masses come
      from a real spurion with a \(\theta^2\bar\theta^2\) component.  Gaugino
      masses and holomorphic \(a,b,t\) terms come from chiral spurions with
      nonzero \(\mathcal F\)-components.  Freezing these backgrounds breaks
      supersymmetry explicitly while keeping the bookkeeping manifestly
      supersymmetric.
    \item \emph{Power Counting for Softness.}
    A scalar mass insertion has dimension two and a gaugino mass or
      holomorphic trilinear insertion has dimension three.  Relative to a
      dimension-four vertex, each insertion lowers the superficial degree of
      divergence by at least two or one, respectively.  Supersymmetry cancels
      the quadratic divergence before insertion; after insertion the remaining
      diagrams are at worst logarithmically divergent in the corresponding
      soft parameters.  A generic dimension-four breaking coupling supplies
      no such reduction, so repeated insertions can leave a quadratic scalar
      divergence.  This power-counting distinction is the meaning of
      ``soft''.
  \end{enumerate}

  \SupplementarySolution{10}{Superspace and Finite \(SU(3)\) Gauge Transformations}
  \begin{enumerate}
    \item[(i)] Use left Grassmann derivatives and put
      \(\mathcal P_\mu=-i\partial_\mu\).  A representation consistent with
      the transformation in the question is
      \[
      \begin{aligned}
        \mathcal Q_a
          &=-i\partial_a
            -(s^\mu)_{a\dot b}\bar\theta^{\dot b}\partial_\mu,\\
        \overline{\mathcal Q}_{\dot a}
          &=+i\bar\partial_{\dot a}
            +\theta^b(s^\mu)_{b\dot a}\partial_\mu .
      \end{aligned}
      \]
      These operators generate
      \[
        \delta\theta=\epsilon,\qquad
        \delta\bar\theta=\bar\epsilon,\qquad
        \delta x^\mu
          =i\theta s^\mu\bar\epsilon
             -i\epsilon s^\mu\bar\theta,
      \]
      and obey
      \[
        \{\mathcal Q_a,\overline{\mathcal Q}_{\dot b}\}
          =2(s^\mu)_{a\dot b}\mathcal P_\mu,\qquad
        \{\mathcal Q_a,\mathcal Q_b\}
          =\{\overline{\mathcal Q}_{\dot a},
              \overline{\mathcal Q}_{\dot b}\}=0.
      \]

    \item[(ii)] A vector superfield is real:
      \[
        V_A^\dagger=V_A,\qquad V^\dagger=V
      \]
      for Hermitian \(T_A\).

    \item[(iii)] A chiral superfield satisfies
      \[
        \overline{\mathcal D}_{\dot a}\Phi=0.
      \]
      Its Hermitian conjugate is antichiral.

    \item[(iv)] Since \(\Phi\) has mass dimension one, the exponent in
      \(e^{-2i\Lambda}\Phi\) shows that
      \[
        [\Lambda_A]=0.
      \]
      Each \(\Lambda_A\) is a complex chiral scalar superfield,
      \(\overline{\mathcal D}_{\dot a}\Lambda_A=0\), and
      \(\Lambda=\Lambda_AT_A\) is valued in the complexified
      \(\mathfrak{su}(3)\) algebra.  Its conjugate
      \(\Lambda^\dagger\) is antichiral.  Restricting its lowest component
      to be real recovers an ordinary \(SU(3)\) gauge transformation, but
      the generalized superspace parameter itself need not be real.

    \item[(v)] With conventional normalization, the renormalizable
      Lagrangian is
      \[
      \begin{aligned}
        \mathcal L={}&
          \int d^4\theta\,\Phi^\dagger e^{2V}\Phi\\
        &+\left[
          \frac1{4g^2}\int d^2\theta\,
             \operatorname{Tr}(W^aW_a)
          +\int d^2\theta\,W_{\mathrm{tree}}(\Phi)
          +\mathrm{H.c.}\right].
      \end{aligned}
      \]
      There is no nonconstant gauge-invariant renormalizable
      \(W_{\mathrm{tree}}\) for a single fundamental of \(SU(3)\).
      In particular,
      \(\epsilon_{ijk}\Phi^i\Phi^j\Phi^k=0\), because identical chiral
      superfields commute.

    \item[(vi)] Put \(U=e^{-2i\Lambda}\).  Invariance of the K\"ahler
      expression,
      \[
        \Phi^{\prime\dagger}e^{2V'}\Phi'
          =\Phi^\dagger e^{2V}\Phi,
      \]
      requires
      \[
        \boxed{\;
        e^{2V'}=e^{-2i\Lambda^\dagger}
                  e^{2V}e^{2i\Lambda}\; }.
      \]
      This finite matrix relation, rather than an additive shift of \(V\),
      is essential for a non-Abelian group.

    \item[(vii)] The preceding relation immediately proves invariance of
      the matter term:
      \[
      \begin{aligned}
        \Phi^{\prime\dagger}e^{2V'}\Phi'
          &=\Phi^\dagger e^{2i\Lambda^\dagger}
             \left(e^{-2i\Lambda^\dagger}
                   e^{2V}e^{2i\Lambda}\right)
             e^{-2i\Lambda}\Phi\\
          &=\Phi^\dagger e^{2V}\Phi.
      \end{aligned}
      \]
      Because \(\Lambda\) is chiral and \(\Lambda^\dagger\) antichiral,
      substituting the finite relation into the field-strength definition
      makes all inhomogeneous derivative terms disappear under
      \(\overline{\mathcal D}^{\,2}\).  The result is the homogeneous law
      \[
        W'_a=e^{-2i\Lambda}W_a e^{2i\Lambda}.
      \]
      Cyclicity of the trace then gives
      \[
        \operatorname{Tr}(W^{\prime a}W'_a)
          =\operatorname{Tr}(W^aW_a).
      \]
      The chiral and full superspace measures are supersymmetry invariant,
      and the only possible superpotential is a gauge-invariant constant.
      Therefore every term in \(\mathcal L\) is invariant.
  \end{enumerate}

  \SupplementarySolution{11}{SQED Vacua and the Moduli-Space Metric}
Let \(\phi^i,\widetilde\phi_i,x\) denote the scalar components of
  \(\Phi^i,\widetilde\Phi_i,X\).  For canonical K\"ahler normalization and
  \(D=\sum_i|\phi^i|^2-\sum_i|\widetilde\phi_i|^2-\zeta\), eliminating the
  auxiliary fields gives
  \[
  \begin{aligned}
    V={}&\sum_{i=1}^N|x-m_i|^2
       \left(|\phi^i|^2+|\widetilde\phi_i|^2\right)
       +\left|\sum_{i=1}^N\widetilde\phi_i\phi^i\right|^2\\
      &+\frac{e^2}{2}
       \left(\sum_i|\phi^i|^2
             -\sum_i|\widetilde\phi_i|^2-\zeta\right)^2.
  \end{aligned}
  \]
  A different positive normalization for the \(X\) kinetic term merely
  multiplies the second term by a positive constant and leaves all vacuum
  equations unchanged.  Thus \(V=0\) is equivalent to
  \[
  \begin{gathered}
    (x-m_i)\phi^i=(x-m_i)\widetilde\phi_i=0
       \quad\text{for every }i,\\
    \sum_i\widetilde\phi_i\phi^i=0,\qquad
    \sum_i|\phi^i|^2-\sum_i|\widetilde\phi_i|^2=\zeta,
  \end{gathered}
  \]
  modulo the \(U(1)\) action
  \(\phi^i\mapsto e^{i\alpha}\phi^i\),
  \(\widetilde\phi_i\mapsto e^{-i\alpha}\widetilde\phi_i\).

  \begin{enumerate}
    \item[(i)] If all \(m_i=0\) and \(\zeta\ne0\), the \(D\)-equation forces
      some charged scalar to be nonzero, so \(x=0\).  The remaining vacuum
      space is
      \[
        \left\{(\phi,\widetilde\phi)\in
           \mathbb C^N\oplus\mathbb C^N:
           \widetilde\phi\cdot\phi=0,\;
           |\phi|^2-|\widetilde\phi|^2=\zeta\right\}\big/U(1).
      \]
      One complex equation, one real equation, and the one-dimensional
      gauge quotient leave
      \[
        \dim_{\mathbb C}\mathcal M=2N-2,\qquad
        \dim_{\mathbb R}\mathcal M=4N-4.
      \]
      For \(\zeta>0\) this is the smooth cotangent bundle
      \(T^*\mathbb{CP}^{N-1}\); for \(\zeta<0\) the roles of
      \(\phi\) and \(\widetilde\phi\) are exchanged.

    \item[(ii)] Suppose the masses are distinct.  A nonzero field of flavor
      \(j\) fixes \(x=m_j\), and no other flavor can then be nonzero.  For
      \(\zeta>0\), the equations give
      \[
        x=m_j,\qquad |\phi^j|^2=\zeta,\qquad
        \widetilde\phi_j=0,
      \]
      with all other fields zero.  For \(\zeta<0\), instead
      \[
        x=m_j,\qquad |\widetilde\phi_j|^2=-\zeta,\qquad
        \phi^j=0.
      \]
      The gauge quotient removes the remaining phase.  There are therefore
      exactly \(N\) isolated vacua.

    \item[(iii)] With \(\zeta=0\) and distinct masses, a putative nonzero
      flavor must satisfy both
      \(|\phi^j|=|\widetilde\phi_j|\) and
      \(\widetilde\phi_j\phi^j=0\), so both fields vanish.  Consequently
      \[
        \phi=\widetilde\phi=0,\qquad x\in\mathbb C.
      \]
      This is a Coulomb branch of complex dimension one.  At
      \(x=m_i\), the \(i\)th charged pair becomes massless, although no
      additional classical Higgs vacuum opens when the masses are distinct.

    \item[(iv)] When \(m_i=\zeta=0\), the vacuum space is the union of two
      branches.  The Coulomb branch is
      \[
        \phi=\widetilde\phi=0,\qquad x\in\mathbb C,
        \qquad \dim_{\mathbb C}\mathcal M_C=1.
      \]
      The Higgs branch has \(x=0\) and is the zero-level quotient
      \[
        \mathcal M_H=
        \{\widetilde\phi\cdot\phi=0,\;
          |\phi|^2=|\widetilde\phi|^2\}/U(1),
        \qquad
        \dim_{\mathbb C}\mathcal M_H=2N-2.
      \]
      The two branches meet at the origin.  For \(N=1\), the Higgs branch
      collapses to that single point.
  \end{enumerate}

  There is a literal inconsistency in the last paragraph of the printed
  parent problem.  In the model as stated, \(X\) is dynamical, and its
  auxiliary equation is
  \[
    F_X^*=-\widetilde\phi\phi=-M.
  \]
  Thus a supersymmetric vacuum has \(M=0\), and for one flavor there is no
  nontrivial vacuum moduli space parametrized by \(M\).  The actual moduli
  space of the printed model is the Coulomb branch parametrized by \(x\).

  The requested metric is instead the standard result for one-flavor
  \(N=1\) SQED with the neutral \(X\) coupling omitted or decoupled.  For
  that intended quotient, \(D=0\) gives
  \(|\phi|=|\widetilde\phi|\), and
  \(M=\widetilde\phi\phi\) gives
  \[
    K=\Phi^\dagger\Phi+
       \widetilde\Phi^\dagger\widetilde\Phi
      =2|\phi|^2
      =2\sqrt{M^\dagger M}.
  \]
  Therefore
  \[
    g_{M\bar M}=\partial_M\partial_{\bar M}K
      =\frac1{2|M|},
    \qquad
    ds^2=\frac{|dM|^2}{2|M|}.
  \]
  The metric is singular at \(M=0\).  Away from the origin the photon and
  gaugino are massive by the supersymmetric Higgs mechanism and may be
  integrated out.  At the origin the gauge symmetry is restored and this
  vector multiplet becomes massless, so the low-energy description in
  terms of \(M\) alone necessarily breaks down.  Equivalently,
  \(w=\sqrt{2M}\) makes the metric locally flat but identifies
  \(w\sim-w\), displaying the conical \(\mathbb C/\mathbb Z_2\)
  singularity.

  \SupplementarySolution{12}{The \(SU(3)\) Coulomb Branch}
  \begin{enumerate}
    \item \emph{The Coulomb Branch of Pure \(N=2\) Theory.}
    Eliminating the auxiliaries gives, in matrix notation,
      \[
        V=\frac12\operatorname{Tr}
           \bigl([\phi^\dagger,\phi]^\dagger[\phi^\dagger,\phi]\bigr)
          \ge0
      \]
      up to the normalization of the generators.  Since
      \[
        [\phi^\dagger,\phi]
           =2i[\operatorname{Re}\phi,\operatorname{Im}\phi],
      \]
      the potential vanishes exactly when the real and imaginary parts
      commute.  They are commuting Hermitian adjoint matrices and can therefore
      be simultaneously conjugated into a Cartan subalgebra.  Dividing by the
      residual Weyl group gives the classical Coulomb-branch moduli space.
    \item \emph{Root-Space Masses on the \(SU(3)\) Coulomb Branch.}
    The off-diagonal generator \(E_{ij}\) obeys
      \[
        [\phi_\infty,E_{ij}]=(a_i-a_j)E_{ij}.
      \]
      Hence, with the normalization of Section 27.9, the vector multiplet
      associated with the roots \(\pm(e_i-e_j)\) has
      \[
        M_{ij}=\sqrt2\,g\,|a_i-a_j|,\qquad i<j.
      \]
      For three distinct eigenvalues all six are massive and the unbroken
      group is \(U(1)^2\).  On any of the three walls \(a_i=a_j\), one root
      pair becomes massless and the group is \(SU(2)\times U(1)\).  Tracelessness
      makes simultaneous equality of all three eigenvalues possible only at
      the origin, where the full \(SU(3)\) is restored.
  \end{enumerate}

  \SupplementarySolution{13}{Surface Central Charge and the Dyonic BPS Spectrum}
  \begin{enumerate}
    \item \emph{Central Charge as a Surface Flux.}
    In a rank-\(r\) vacuum choose orthonormal Cartan generators and write
      \(\phi_\infty=v_IH_I\).  Only the Cartan field strengths survive at
      spatial infinity.  Substitution in the surface integral
      \eqref{eq:27.9.20} yields
      \[
        Z_{12}=2\sqrt2\sum_{I=1}^r
          v_I(iq_I-\mathcal M_I),\qquad
        q_I=\int dS_iE_I^i,\quad
        \mathcal M_I=\int dS_iB_I^i .
      \]
      Under an orthogonal change of Cartan basis, each of the three vectors
      \(v,q,\mathcal M\) is multiplied by the same orthogonal matrix.  Their
      scalar products, and hence \(Z_{12}\), are unchanged.
    \item \emph{Electric, Magnetic, and Dyonic BPS Masses.}
    The \(N=2\) algebra gives \(M\ge|Z_{12}|/2\).  With
      Eq.~\eqref{eq:27.9.22},
      \[
        M\ge\sqrt2\,|v|\,|iq-\mathcal M|.
      \]
      Thus a purely electric state of charge \(q\) obeys
      \(M\ge\sqrt2|vq|\).  A monopole with
      \(\mathcal M=4\pi\nu/e\) obeys
      \[
        M\ge\frac{4\pi\sqrt2\,|\nu v|}{|e|},
      \]
      and a dyon obeys
      \[
        M\ge\sqrt2\,|v|\sqrt{q^2+\mathcal M^2}.
      \]
      Equality holds precisely for a shortened half-BPS multiplet.  The
      elementary charged vector multiplets and the Bogomol'nyi monopoles
      described in Section 27.9 saturate their respective bounds.
  \end{enumerate}

\end{enumerate}
