  \SupplementaryExercise{1}{Chiral and Fermi Multiplets in \texorpdfstring{\((0,2)\)}{(0,2)} Superspace}
    {(Adapted from University of Cambridge, 2021 Part III Paper 307, \emph{Supersymmetry}, Question 3.)}
    {cambridge-part-iii-susy-2021-p307-q3-v3ch30-s01}
  In two bosonic dimensions, \((0,2)\) superspace is defined as the surface
  \(\theta^- = \bar\theta^- = 0\) inside \((2,2)\) superspace.  A \((0,2)\)
  chiral field is a superfield \(\Phi\) that obeys \(\bar D_+\Phi=0\), where
  \[
    \bar D_+=-\frac{\partial}{\partial\bar\theta^+}
      +i\theta^+\frac{\partial}{\partial x^+}
  \]
  is the chiral derivative.  What is the component expansion of a \((0,2)\)
  chiral superfield?  Construct a supersymmetric action \(S_1[\Phi]\) whose
  component expansion includes the standard kinetic term for a complex
  scalar field.

  Now let \(\Lambda_-\) be a \((0,2)\) superfield whose leading component
  field is a fermionic spinor \(\lambda_-\), and let \(\Lambda_-\) obey
  \(\bar D_+\Lambda_-=f(\Phi^i)\), where \(f\) is a holomorphic function of
  chiral superfields \(\Phi^i\).  Find the component expansion of \(\Lambda_-\)
  and the component-field expression for the action
  \[
    S_2=-\frac12\int \bar\Lambda_-\Lambda_-\,
      d\theta^+\,d\bar\theta^+\,d^2x.
  \]
  Identify which components of \(\Lambda_-\) are auxiliary.

  In a theory of several such fermionic superfields \(\Lambda_{-a}\), each
  obeying \(\bar D_+\Lambda_{-a}=f_a(\Phi^i)\), it is desired to add the terms
  \[
    S_3=-\left(\int \Lambda_{-a}J^a(\Phi^i)\,
      d\theta^+\,d^2y+\text{complex conjugate}\right)
  \]
  to the action, where \(J^a(\Phi^i)\) are further holomorphic functions of
  \((0,2)\) chiral superfields and
  \(y^+=x^++i\theta^+\bar\theta^+\), while \(y^-=x^-\).  Under what
  condition(s) are these terms also supersymmetric?  Find the potential
  experienced by the scalars \(\phi^i\) in the theory \(S_1+S_2+S_3\) after
  eliminating the auxiliary field(s).

  \SupplementaryExercise{2}{Nilpotent and Orthogonally Constrained Chiral Superfields}
    {(Adapted from University of Cambridge, 2017 Part III Paper 307, \emph{Supersymmetry}, Question 2.)}
    {cambridge-part-iii-susy-2017-p307-q2-v3ch30-s02}
  Using the definition of the supercovariant derivatives
  \[
    D_\alpha\equiv\partial_\alpha
      +i(\sigma^\mu)_{\alpha\dot\beta}\bar\theta^{\dot\beta}\partial_\mu,
    \qquad
    \bar D_{\dot\alpha}\equiv\bar\partial_{\dot\alpha}
      +i\theta^\beta(\sigma^\mu)_{\beta\dot\alpha}\partial_\mu,
  \]
  define a chiral superfield.  Write a general expression for a chiral
  superfield \(\Phi(y,\theta)\), with
  \(y^\mu=x^\mu+i\theta\sigma^\mu\bar\theta\), in terms of its components.
  Use this to write the full expression of \(\Phi(x,\theta,\bar\theta)\).
  Consider now a chiral superfield \(X\) with the extra nilpotency constraint
  \(X^2=0\).  Solve for each of the components of \(X\).  Write the most
  general superpotential and K\"ahler potential for the field \(X\).  Compute
  the corresponding contribution to the scalar potential.  What can be said
  about supersymmetry breaking in this case?  If an extra chiral superfield
  \(Y\) is introduced satisfying \(XY=0\), solve for the components of \(Y\).
  Write the most general superpotential and K\"ahler potential as a function
  of \(X\) and \(Y\).

  \SupplementaryExercise{3}{D-Algebra in a One-Loop Supergraph}
    {(Inspired by Gates, Grisaru, Ro\v{c}ek, and Siegel, \emph{Superspace}, Section 6.3, pp. 352--365; independently formulated.)}
    {ch30-s03-d-algebra-in-a-one-loop}
  \begin{enumerate}
    \item \emph{Derivative Transfer Rule.}
    Prove the superspace integration-by-parts rule that transfers
      \(\mathcal D_{L,R}\) from one end of a propagator to the other.
      Include the Grassmann parity sign and apply the result to a product
      of two fermionic superfields.
    \item \emph{One-Loop Self-Energy Algebra.}
    In the massless cubic chiral theory, perform only the \(D\)-algebra
      of the one-loop two-point supergraph.  Show that its local
      divergence is proportional to a full-superspace integral
      \([\Phi^*\Phi]_D\), not an \(\mathcal F\)-term.
    \item \emph{Vanishing Vacuum Loop.}
    Consider a connected vacuum supergraph made only of massless chiral
      propagators and superpotential vertices.  Use unsaturated Grassmann
      delta functions, rather than component cancellations, to show that
      its contribution to the vacuum energy vanishes.
  \end{enumerate}

  \SupplementaryExercise{4}{The Wess--Zumino Model and Non-Renormalization}
    {(Adapted from University of Cambridge, 2018 Part III Paper 307, \emph{Supersymmetry}, Question 3.)}
    {cambridge-part-iii-susy-2018-p307-q3-v3ch30-s04}
  Consider the Wess--Zumino model for a chiral superfield
  \(\Phi(x,\theta)=\phi(x)+\sqrt2\,\theta\psi(x)+\theta\theta F(x)+\cdots\),
  with K\"ahler potential \(K=\Phi^\dagger\Phi\) and superpotential
  \[
    W=\frac{m}{2}\Phi^2+\frac{g}{3}\Phi^3.
  \]
  Write down the Lagrangian in superspace.

  Compute the \(F\)-dependent part of the Lagrangian.  [Hint: Expand the
  superpotential \(W\) in a Taylor series around \(\Phi=\phi\).]  By solving
  for \(F\), determine the scalar potential.  Verify the following:
  \begin{itemize}
    \item The scalar potential is positive semidefinite.
    \item The mass of the fermion \(\psi\) equals the mass of the boson
      \(\phi\).
    \item The strength of the scalar quartic coupling \(|\phi|^4\) is
      determined by the strength of the Yukawa coupling \((\phi\psi\psi)\).
  \end{itemize}
  Outline the arguments establishing that the superpotential for this model
  does not receive perturbative quantum corrections.

  \SupplementaryExercise{5}{Tensor Beta Functions for Superpotential Operators}
    {(Inspired by Antusch and Ratz, ``Supergraph Techniques and Two-Loop Beta-Functions for Renormalizable and Non-Renormalizable Operators'', Sections 2.1 and 3.1; independently formulated.)}
    {ch30-s05-tensor-beta-functions-for-superpotential-operators}
  \begin{enumerate}
    \item \emph{Yukawa Beta Tensor.}
    Let \(W=Y_{ijk}\Phi_i\Phi_j\Phi_k/6\), with \(Y_{ijk}\) symmetric.
      Assuming the superpotential has no independent counterterm, derive
      its beta function in terms of the anomalous-dimension matrix
      \(\gamma_i{}^j\).
    \item \emph{Mass-Matrix Running.}
    For \(W=M_{ij}\Phi_i\Phi_j/2\), derive the renormalization-group
      equation for the symmetric mass matrix \(M\).  Check that the answer
      transforms covariantly under a constant unitary change of chiral
      field basis.
    \item \emph{Higher-Operator Running.}
    A superpotential contains the operator
      \[
        \frac{C_{i_1\cdots i_n}}{n!\,M^{n-3}}\,
          \Phi_{i_1}\cdots\Phi_{i_n}.
      \]
      Derive the beta function of \(C\) from wavefunction renormalization
      and explain why the result remains valid for \(n>3\).
  \end{enumerate}

  \SupplementaryExercise{6}{Anomalous Dimensions and Group Factors}
    {(Inspired by Antusch and Ratz, ``Supergraph Techniques and Two-Loop Beta-Functions for Renormalizable and Non-Renormalizable Operators'', Sections 2--3; independently formulated.)}
    {ch30-s06-anomalous-dimensions-and-group-factors}
  \begin{enumerate}
    \item \emph{One-Loop Anomalous Dimension.}
    For chiral fields with cubic couplings \(Y_{ijk}\) and gauge
      coupling \(g\), show that the one-loop anomalous dimension must have
      the tensor form
      \(aY_{imn}Y^{*jmn}-bg^2C_2(R_i)\delta_i{}^j\).
      Determine the signs from physical wavefunction screening.
    \item \emph{Casimir Conversion.}
    Prove
      \(C_2(R)\dim R=T(R)\dim G\) by tracing the defining Casimir equation.
      Evaluate \(C_2\) and \(T\) for the fundamental representation of
      \(SU(N)\) with \(T(\mathbf N)=1/2\).
    \item \emph{Abelian Charge Factor.}
    Replace a non-Abelian gauge factor by \(U(1)\).  Demonstrate from the
      two gauge vertices on a chiral self-energy line that
      \(C_2(R)\) is replaced by \(q^2\), and explain why no term linear in
      \(q\) can survive.
  \end{enumerate}

  \SupplementaryExercise{7}{Field Redefinitions and Effective-Operator Running}
    {(Inspired by Antusch and Ratz, ``Supergraph Techniques and Two-Loop Beta-Functions for Renormalizable and Non-Renormalizable Operators'', Sections 2--3; independently formulated.)}
    {ch30-s07-field-redefinitions-and-effective-operator-running}
  \begin{enumerate}
    \item \emph{Field-Redefinition Contribution.}
    Let \(\Phi_B=Z^{1/2}\Phi\) with matrix-valued \(Z\).  Differentiate
      this relation at fixed bare field and derive the contribution of
      each external chiral leg to the running of an arbitrary
      superpotential tensor.
    \item \emph{EFT Triangularity.}
    Explain by dimensional analysis why an operator suppressed by
      \(M^{-k}\) does not affect renormalizable wavefunction counterterms
      at leading order in \(1/M\), while renormalizable interactions can
      renormalize that higher-dimensional operator.
    \item \emph{Holomorphic versus Canonical Coupling.}
    A holomorphic cubic coefficient is scale independent before fields
      are canonically normalized.  Show explicitly how a nonzero physical
      beta function nevertheless arises when
      \(\Phi_{\mathrm c}=Z^{1/2}\Phi\).
  \end{enumerate}

  \SupplementaryExercise{8}{Background Hessian and the One-Loop Kähler Term}
    {(Inspired by Kuzenko and Tyler, ``The One-Loop Effective Potential of the Wess--Zumino Model Revisited'', Sections 2.1--3; independently formulated.)}
    {ch30-s08-background-hessian-and-the-one-loop}
  \begin{enumerate}
    \item \emph{Background Hessian.}
    Expand a general Wess--Zumino action about a chiral background
      \(\Phi\).  Construct the quadratic chiral--antichiral Hessian in
      terms of \(\Psi=W''(\Phi)\), and show why cubic quantum vertices
      cannot contribute at one loop.
    \item \emph{Logarithm of the Hessian.}
    Evaluate the Gaussian functional integral for the background
      Hessian and obtain
      \(\Gamma^{(1)}=(i/2)\operatorname{Tr}\ln(H/H_0)\).
      Expand the logarithm and identify the symmetry factor of an
      \(n\)-vertex ring supergraph.
    \item \emph{Kahler Dimensional Check.}
    In four dimensions \([\Phi]=1\) and \([\Psi=W''(\Phi)]=1\).
      Use dimensions and phase invariance to determine the possible
      one-loop local Kahler correction for constant \(\Psi\), up to its
      overall coefficient and a local subtraction.
  \end{enumerate}

  \SupplementaryExercise{9}{Momentum Integration and Regulator Ordering}
    {(Inspired by Kuzenko and Tyler, ``The One-Loop Effective Potential of the Wess--Zumino Model Revisited'', Sections 3--4.1 and Appendix A; independently formulated.)}
    {ch30-s09-momentum-integration-and-regulator-ordering}
  \begin{enumerate}
    \item \emph{Logarithmic Momentum Integral.}
    Differentiate
      \(J(M^2)=\int d^dp\,(2\pi)^{-d}\ln(p^2+M^2)\) with respect to
      \(M^2\), evaluate the pole in dimensional regularization, and
      reconstruct the \(M^2\ln(M^2/\mu^2)\) dependence.
    \item \emph{Regulator Ordering Warning.}
    Consider a conditionally convergent proper-time and momentum
      integral arising after \(D\)-algebra.  Explain why exchanging the
      order of integration can change a finite coefficient, and formulate
      a regulator-first prescription that removes the ambiguity.
  \end{enumerate}

  \SupplementaryExercise{10}{Higher-Derivative Terms, Scaling, and Decoupling}
    {(Inspired by Kuzenko and Tyler, ``The One-Loop Effective Potential of the Wess--Zumino Model Revisited'', Sections 4.1--4.3; independently formulated.)}
    {ch30-s10-higher-derivative-terms-scaling-and-decoupling}
  \begin{enumerate}
    \item \emph{Four-Derivative Invariant.}
    For a constant chiral background, build the lowest real scalar
      full-superspace integrand that vanishes when all spinor derivatives
      of \(\Phi\) vanish.  Show that it begins with two left and two right
      superderivatives.
    \item \emph{Auxiliary-Field Order.}
    Set spacetime derivatives and fermions to zero in the invariant of
      part (a).  Determine its dependence on the scalar and
      auxiliary components and explain why it belongs to an auxiliary
      field potential rather than the ordinary Kahler potential.
    \item \emph{Massless Scaling Form.}
    In the massless cubic model, use dimensions and constant phase
      transformations of \(\Phi\) to find the denominator required beneath
      the four-superderivative numerator of part (a).  Verify that
      the resulting integrand is dimension two.
    \item \emph{Heavy-Field Decoupling.}
    For \(W''(\Phi)=m+\lambda\Phi\) with
      \(|\lambda\Phi|\ll |m|\), expand the one-loop Kahler correction
      through quadratic order in \(\Phi\).  Identify the removable linear
      term and the threshold correction to the kinetic term.
  \end{enumerate}

\end{enumerate}
