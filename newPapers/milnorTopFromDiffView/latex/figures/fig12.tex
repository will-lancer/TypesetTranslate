% Figure 12: Examples of plane vector fields (indices -1, 0, +1, +2)
\begin{tikzpicture}[line cap=round,line join=round,>=Stealth,scale=0.9,
    midarrow/.style={postaction={decorate,
      decoration={markings,mark=at position #1 with {\arrow{Stealth}}}}}]
  % saddle: iota = -1
  \begin{scope}[shift={(-2.2,1.55)},rotate=45,scale=0.85]
    \draw[thin,midarrow=0.75] (0,0) -- (1.15,0);
    \draw[thin,midarrow=0.75] (0,0) -- (-1.15,0);
    \draw[thin,midarrow=0.6] (0,1.15) -- (0,0);
    \draw[thin,midarrow=0.6] (0,-1.15) -- (0,0);
    \foreach \c in {0.12,0.34} {
      \draw[thin] plot[domain={\c/1.05}:1.05,samples=25] (\x,{\c/\x});
      \draw[thin] plot[domain={\c/1.05}:1.05,samples=25] (-\x,{-\c/\x});
      \draw[thin] plot[domain={\c/1.05}:1.05,samples=25] (\x,{-\c/\x});
      \draw[thin] plot[domain={\c/1.05}:1.05,samples=25] (-\x,{\c/\x});
    }
  \end{scope}
  \node at (-2.2,0.05) {$\iota = -1$};
  % no zero: iota = 0
  \begin{scope}[shift={(2.2,1.55)}]
    \foreach \y in {-0.8,-0.4,0,0.4,0.8} {
      \draw[thin,midarrow=0.58] (-1.15,\y)
        .. controls (-0.4,{\y+0.28}) and (0.4,{\y-0.28}) .. (1.15,\y);
    }
  \end{scope}
  \node at (2.2,0.05) {$\iota = 0$};
  % center: iota = +1
  \begin{scope}[shift={(-2.2,-1.75)}]
    \foreach \r in {0.3,0.6,0.9} {
      \draw[thin,midarrow=0.12,midarrow=0.62] (0,0) circle (\r);
    }
    \fill (0,0) circle (1.2pt);
  \end{scope}
  \node at (-2.2,-3.05) {$\iota = +1$};
  % dipole: iota = +2
  \begin{scope}[shift={(2.2,-1.75)}]
    \foreach \r in {0.24,0.48,0.78} {
      \draw[thin,midarrow=0.3,midarrow=0.8] (0,\r) circle (\r);
      \draw[thin,midarrow=0.3,midarrow=0.8] (0,-\r) circle (\r);
    }
    \fill (0,0) circle (1.2pt);
  \end{scope}
  \node at (2.2,-3.65) {$\iota = +2$};
\end{tikzpicture}
