% Figure 16: Framed submanifolds and a framed cobordism
\begin{tikzpicture}[line cap=round,line join=round,>=Stealth,scale=0.86]
  % (a) closed framed submanifold N inside the manifold M
  \begin{scope}[shift={(-2.7,1.5)}]
    \draw[thick] (0,0) ellipse (1.9 and 1.15);
    \node at (-1.55,1.1) {$M$};
    \draw[very thick] (-0.15,-0.05) ellipse (0.75 and 0.45);
    \node at (-1.35,-0.35) {$N$};
    \foreach \a in {35,90,145,215,270,325} {
      \draw[->] ({-0.15+0.75*cos(\a)},{-0.05+0.45*sin(\a)})
        -- ({-0.15+1.15*cos(\a)},{-0.05+0.78*sin(\a)});
    }
  \end{scope}
  % (b) a framed arc N
  \begin{scope}[shift={(2.4,1.35)}]
    \draw[very thick] (-0.8,-0.75) .. controls (-0.55,0.15) and (0.15,0.75) .. (1.15,0.85);
    \draw[->] (-0.76,-0.6) -- (-1.5,-0.45);
    \draw[->] (-0.62,-0.25) -- (-1.32,0.0);
    \draw[->] (-0.38,0.15) -- (-1.0,0.55);
    \draw[->] (0.05,0.5) -- (-0.45,1.05);
    \draw[->] (0.6,0.75) -- (0.3,1.4);
    \draw[->] (1.1,0.84) -- (1.0,1.55);
    \node at (1.4,0.55) {$N$};
  \end{scope}
  % (c) a framed cobordism X in M x [0,1]
  \begin{scope}[shift={(0,-1.6)}]
    \draw[thick] (-3.1,0.75) -- (3.1,0.75);
    \draw[thick] (-3.1,-0.75) -- (3.1,-0.75);
    \draw[thick] (-3.1,0) ellipse (0.24 and 0.75);
    \draw[thick] (3.1,0.75) arc (90:-90:0.24 and 0.75);
    \draw[thick,dashed] (3.1,0.75) arc (90:270:0.24 and 0.75);
    % the cobordism surface X with a hump
    \draw[very thick] (-2.88,0.1) .. controls (-1.9,0.3) and (-1.2,-0.2) .. (-0.4,-0.1)
      .. controls (0.3,-0.02) and (0.5,0.35) .. (1.0,0.3)
      .. controls (1.7,0.22) and (2.3,-0.15) .. (2.87,-0.05);
    \fill (-2.88,0.1) circle (1.4pt);
    \fill (2.87,-0.05) circle (1.4pt);
    % framing arrows along X
    \draw[->] (-2.4,0.22) -- (-2.4,0.75);
    \draw[->] (-1.6,0.12) -- (-1.6,0.65);
    \draw[->] (-0.8,-0.14) -- (-0.8,0.4);
    \draw[->] (0.1,-0.02) -- (0.1,0.52);
    \draw[->] (0.9,0.31) -- (0.9,0.85);
    \draw[->] (1.8,0.13) -- (1.8,0.67);
    \draw[->] (2.5,-0.1) -- (2.5,0.44);
    \node at (-0.35,-0.42) {$X$};
    \node[below] at (0,-0.85) {$M \times [0,1]$};
  \end{scope}
\end{tikzpicture}
